import ast, hashlib
NAMED=(ast.ClassDef,ast.FunctionDef,ast.AsyncFunctionDef)
def source_segment(raw_lines,node):
    if node.lineno==node.end_lineno:
        return raw_lines[node.lineno-1][node.col_offset:node.end_col_offset]
    return b''.join([raw_lines[node.lineno-1][node.col_offset:],*raw_lines[node.lineno:node.end_lineno-1],raw_lines[node.end_lineno-1][:node.end_col_offset]])
def named_rows(tree,source,module,module_sha):
    lines=source.splitlines(keepends=True)
    result=[]
    def visit(node,scope):
        inner=scope
        if isinstance(node,NAMED):
            inner=scope+(node.name,)
            row={'module':module,'qualname':'.'.join(inner),'type':type(node).__name__,'line':node.lineno,'end':node.end_lineno,'body_sha256':hashlib.sha256(source_segment(lines,node)).hexdigest(),'full_module_sha256':module_sha}
            if isinstance(node,ast.ClassDef):row['bases']=[ast.unparse(x) for x in node.bases]
            result.append(row)
        for child in ast.iter_child_nodes(node):visit(child,inner)
    visit(tree,())
    return result
def row_key(row):
    return (row['module'],row['qualname'],row['type'],row['line'],row['end'])
def pinned_bytes(row):
    from pathlib import Path
    import stat
    p=Path(row['path']);s=p.lstat()
    assert stat.S_ISREG(s.st_mode) and stat.S_IMODE(s.st_mode)==0o400 and s.st_nlink==1
    b=p.read_bytes()
    assert len(b)==row['bytes'] and hashlib.sha256(b).hexdigest()==row['sha256']
    return b
