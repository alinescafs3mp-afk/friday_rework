static void history_loss(Adoption *a,int reason)
{
 a->history_failed=1; /* local sticky ORIGINAL adoption state, never cleared */
 if(a->a.session && a->a.epoch)frw_client_history_loss(a->a.session,a->a.epoch,reason);
}
static int current_handles(Adoption *a)
{

 rc=frw_client_validate_current(a->a.session,a->a.epoch);
 if(rc)history_loss(a,rc);return rc;
}
static void history_free(HistoryBatch *b)
{
 if(b->owner && !b->attempted)history_loss(b->owner,-256);
 if(b->receipt)frw_client_put_delivery(b->receipt); /* never implicit ACK */
 Py_XDECREF(b->bytes);Py_XDECREF(b->owner);Py_TYPE(b)->tp_free((PyObject *)b);
}
