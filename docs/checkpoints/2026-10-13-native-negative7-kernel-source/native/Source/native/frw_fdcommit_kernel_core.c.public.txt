/* Kept zero: no complete, kernel-source-qualified native insertion map exists
 * in supplied evidence. Only a separately reviewed NEW Source replacement can
 * change this. Caller flags, JSON, reload, counters or BTF cannot qualify it. */
static const u64 frwk_source_coverage = 0;

static int sticky(struct frwk_epoch *e, u64 loss)
{
 e->fence_seq=0;
 e->loss |= loss;
 e->final_only = true;
 return -EUCLEAN;
}
static u64 generation(struct frwk_epoch *e)
{
 if (e->generation >= U64_MAX-1 || e->records_n == FRWK_RECORDS) {
  sticky(e, 16); return 0;
 }
 e->records_n++;
 return ++e->generation;
}
static struct frwk_task *task_ref(struct frwk_epoch *e, struct task_struct *p)
{
 u32 i;
 for (i=0;i<e->tasks_n;i++) if (e->tasks[i].ref==p) return &e->tasks[i];
 return NULL;
}
static struct frwk_table *table_ref(struct frwk_epoch *e, struct files_struct *p)
{
 u32 i;
 for (i=0;i<e->tables_n;i++) if (e->tables[i].ref==p) return &e->tables[i];
 return NULL;
}
static struct frwk_ofd *ofd_ref(struct frwk_epoch *e, struct file *p)
{
 u32 i;

}
static struct frwk_slot *slot_ref(struct frwk_epoch *e, u64 table, u32 fd)
{

  return &e->slots[i];
 return NULL;
}
int frwk_publish_locked(struct frwk_epoch *e, struct frwk_event *v)
{
 if (e->loss) return -EUCLEAN;
 if (!e->prearmed || e->coverage!=FRWK_REQUIRED || frwk_source_coverage!=FRWK_REQUIRED)
  return sticky(e, 4);
 if (e->seq>=U64_MAX-1 || e->events_n==FRWK_EVENTS) return sticky(e, 2|16);
 e->fence_seq=0;
 v->magic=0x46525731; v->version=1; v->epoch_hi=e->hi; v->epoch_lo=e->lo;
 v->mono_ns=ktime_get_ns();
 v->seq=++e->seq; v->coverage=e->coverage; v->loss=e->loss; v->reserved=0;
 e->events[e->producer]=*v;
 e->producer=(e->producer+1)%FRWK_EVENTS; e->events_n++;
 return 0;
}

  return sticky(e,64);

  struct frwk_slot *old=&e->slots[i],*next;u32 j;struct frwk_ofd *ofd=NULL;

  v.ofd=old->ofd;v.src_table=s->id;v.src_fd=old->fd;v.src_slot=old->id;
  v.src_ofd=old->ofd;v.flags=16;v.aux=old->flags;
  if(frwk_publish_locked(e,&v))return -EUCLEAN;
  next=&e->slots[e->slots_n++];*next=*old;next->table=d->id;next->id=gen;ofd->aliases++;

int frwk_slot_commit_locked(struct frwk_epoch *e, struct task_struct *actor,

}

int frwk_take_retired_refs_locked(struct frwk_epoch *e,struct frwk_reference_batch *b)
{
 u32 i;
 /* Caller provides the already charged bounded batch. Detach each owned
  * reference ONCE under lock; dispose outside lock. No unbounded free list. */
 if(!b || b->tasks_n || b->tables_n || b->files_n)return sticky(e,64);
 for(i=0;i<e->tasks_n;i++)if(!e->tasks[i].live && e->tasks[i].ref){b->tasks[b->tasks_n++]=e->tasks[i].ref;e->tasks[i].ref=NULL;}
 for(i=0;i<e->tables_n;i++)if(!e->tables[i].live && e->tables[i].ref){b->tables[b->tables_n++]=e->tables[i].ref;e->tables[i].ref=NULL;}
 for(i=0;i<e->ofds_n;i++)if(!e->ofds[i].live && e->ofds[i].ref){b->files[b->files_n++]=e->ofds[i].ref;e->ofds[i].ref=NULL;}
 for(i=0;i<e->rings_n;i++)if(!e->rings[i].live && e->rings[i].ref){b->files[b->files_n++]=e->rings[i].ref;e->rings[i].ref=NULL;}
 for(i=0;i<e->retained_n;i++)if(!e->retained[i].live && e->retained[i].held){b->files[b->files_n++]=e->retained[i].held;e->retained[i].held=NULL;}
 return 0;
}
void frwk_put_reference_batch(struct frwk_reference_batch *b)
{
 u32 i;
 /* Internal ownership disposal only; fput can defer actual kernel teardown.
  * This does NOT claim native quiescence, ResourceGo, or deadline acceptance. */
 for(i=0;i<b->tasks_n;i++){put_task_struct(b->tasks[i]);b->tasks[i]=NULL;}
 for(i=0;i<b->tables_n;i++){put_files_struct(b->tables[i]);b->tables[i]=NULL;}
 for(i=0;i<b->files_n;i++){fput(b->files[i]);b->files[i]=NULL;}
 b->tasks_n=b->tables_n=b->files_n=0;
}
