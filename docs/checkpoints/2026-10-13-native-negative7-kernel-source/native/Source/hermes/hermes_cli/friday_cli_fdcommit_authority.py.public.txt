def require_fixed_native_bridge(native):
    names = ("validate_adoption", "session_of", "epoch_of", "qualification_of", "role_of",
             "check_registration", "validate_binding", "validate_current", "binding_fields",
             "reserve_birth", "bind_child", "retain_child", "validate_retained_cleanup", "retain_descriptor",
             "validate_descriptor", "release_descriptor", "read_owner_history",
             "adopt_owner", "adopt_child", "descriptor_anchor", "reopen_descriptor",
             "descriptor_reader_fd", "release_descriptor_reader", "release_child_handle",
             "history_bytes", "ack_owner_history")
    if (type(native) is not types.ModuleType or native.__name__ != "_frw_fdcommit_native"
            or sys.modules.get(native.__name__) is not native
            or type(getattr(native, "__loader__", None)) is not ExtensionFileLoader):
        raise BindingUnbound("fixed_native_extension_loader_unbound")
    for name in names:
        method = getattr(native, name, None)
        if (type(method) is not types.BuiltinFunctionType or method.__name__ != name
                or method.__module__ != native.__name__ or method.__self__ is not native):
            raise BindingUnbound("arbitrary_callback_native_backend_refused")
    # These shape checks cannot qualify an arbitrary extension. Its internal
    # sealed independent qualifier must verify the entire actual native mapping.
    # No independently anchored loaded-module/layout/resource issuer exists
    # in this proposal. Do not invoke ANY candidate native method on shapes.
    # A future genuine implementation requires new whole-dependency review.
    raise BindingUnbound("genuine_independent_loaded_bridge_bootstrap_missing")

class CommitHistoryAuthority(NativeAcceptanceAuthority):
    __slots__ = ("_native", "_adoption", "_session", "_epoch", "_qualification", "_failure")

    def __init__(self):
        raise BindingUnbound("native_opaque_adoption_required")

    @classmethod
    def from_native_adoption(cls, adoption, native):
        require_fixed_native_bridge(native)
        if (cls is not CommitHistoryAuthority or type(native) is not types.ModuleType
                or native.__name__ != "_frw_fdcommit_native"
                or sys.modules.get(native.__name__) is not native
                or type(adoption) is not native.Adoption):
            raise BindingUnbound("foreign_native_adoption")
        # Native fixed implementation verifies its sealed independent qualifier,
        # native session, exact current task/exec/role and charged original bounds.
        native.validate_adoption(adoption)
        self = object.__new__(cls)
        self._native, self._adoption = native, adoption
        self._session = native.session_of(adoption)
        self._epoch = native.epoch_of(adoption)
        self._qualification = native.qualification_of(adoption)
        self._failure = None
        if (type(self._session) is not native.Session or type(self._epoch) is not native.Epoch
                or type(self._qualification) is not native.Qualification):
            raise BindingUnbound("native_handle_or_epoch_unbound")
        return self

    def _registered(self, binding, qualification):
        if (qualification is not self._qualification or binding.authority is not self
                or binding.acceptance_epoch is not self._epoch
                or type(self._session) is not self._native.Session
                or self._native is not sys.modules.get("_frw_fdcommit_native")):
            raise BindingUnbound("native_registered_epoch_or_handle_drift")
        require_fixed_native_bridge(self._native)
        self._native.validate_binding(self._adoption, binding)
        self._native.check_registration(
            self._adoption, qualification, type(self),
            CommitHistoryAuthority.validate_current.__code__,
            CommitHistoryAuthority.validate_retained_cleanup.__code__, type(binding))

    def validate_current(self, binding, phase):
        if self._failure is not None:
            raise BindingUnbound("native_prior_current_refusal:" + self._failure)
        try:
            self._registered(binding, self._qualification)
            if type(phase) is not str or not phase or len(phase.encode()) > 512:
                raise BindingUnbound("native_phase_bound")
            self._native.validate_current(self._session, self._epoch)
        except BaseException as error:
            self._failure = type(error).__name__
            raise

    def validate_retained_cleanup(self, binding, phase, *, native_boundary, child, deadline):
        # Admission/history refusal remains sticky. Only a separately retained
        # original custody capability is eligible; this method cannot open a
        # child/handle, extend a deadline, wait or clear current refusal.
        self._registered(binding, self._qualification)
        if (type(phase) is not str or not phase or len(phase.encode()) > 512
                or native_boundary.binding is not binding or child.binding is not binding
                or not any(item is child for item in native_boundary.children)
                or type(deadline) not in (int, float) or not math.isfinite(deadline)
                or not 0 < deadline <= binding.session_deadline):
            raise BindingUnbound("native_retained_cleanup_foreign_or_deadline")
        receipt = getattr(child, "native_retained_custody", None)
        if type(receipt) is not self._native.RetainedChild:
            raise BindingUnbound("native_child_retained_custody_unbound")
        # Child PID/pidfd numbers are descriptive; genuine receipt is native
        # same-epoch true-birth identity and exact held pidfd/directory refs.
        self._native.validate_retained_cleanup(self._session, self._epoch, receipt,
                                               0 if child.pid is None else child.pid,
                                               -1 if child.pidfd is None else child.pidfd, deadline)

    def _current_handles(self):
        if self._failure is not None:
            raise BindingUnbound("native_prior_current_refusal:" + self._failure)
        require_fixed_native_bridge(self._native)
        try:
            self._native.validate_current(self._session, self._epoch)
        except BaseException as error:
            self._failure = type(error).__name__
            raise

    def reserve_birth(self, native_boundary, child):
        if (native_boundary.binding.authority is not self or child.binding is not native_boundary.binding
                or not any(item is child for item in native_boundary.children)
                or child.pid is not None or child.native_retained_custody is not None):
            raise BindingUnbound("native_birth_reservation_exact_preclone_custody")
        self._current_handles()
        # Reserves exact original cgroup/next clone scope BEFORE native call.
        # This native retained capability survives PREPID or call-no-return;
        # it cannot prove PID absence or authorize a retry/replacement.
        return self._native.reserve_birth(self._session, self._epoch)

    def retain_child(self, child):
        self._current_handles()
        if child.binding.authority is not self or child.pid is None or child.pidfd is None:
            raise BindingUnbound("native_child_identity_not_known")
        receipt = child.native_retained_custody
        if type(receipt) is not self._native.RetainedChild:
            raise BindingUnbound("native_preclone_reservation_missing")
        self._native.bind_child(self._session, self._epoch, receipt, child.pid, child.pidfd,
                                child.gatefd, child.errorfd)
        return receipt

    def close_child_handle(self, child, kind):
        require_fixed_native_bridge(self._native)
        receipt = child.native_retained_custody
        if (child.binding.authority is not self or type(receipt) is not self._native.RetainedChild
                or kind not in (0, 1, 2)):
            raise BindingUnbound("foreign_child_handle_close")
        # Same retained capability closes the original recorded generations;
        # mutable child FD observations never identify a descriptor to close.
        self._native.release_child_handle(self._session, self._epoch, receipt, kind)

    def retain_descriptor(self, fd):
        self._current_handles()
        if type(fd) is not int or fd < 0:
            raise BindingUnbound("descriptor_number_shape")
        return self._native.retain_descriptor(self._session, self._epoch, fd)

    def descriptor_anchor(self, receipt):
        self.validate_descriptor(receipt)
        return self._native.descriptor_anchor(self._session, self._epoch, receipt)

    def validate_descriptor(self, receipt):
        self._current_handles()
        if type(receipt) is not self._native.RetainedDescriptor:
            raise BindingUnbound("native_descriptor_retained_capability_required")
        self._native.validate_descriptor(self._session, self._epoch, receipt)

    def reopen_descriptor(self, receipt):
        self.validate_descriptor(receipt)
        return self._native.reopen_descriptor(self._session, self._epoch, receipt)

    def descriptor_reader_fd(self, receipt):
        self._current_handles()
        if type(receipt) is not self._native.RetainedReader:
            raise BindingUnbound("native_descriptor_reader_capability_required")
        return self._native.descriptor_reader_fd(self._session, self._epoch, receipt)

    def release_descriptor_reader(self, receipt):
        require_fixed_native_bridge(self._native)
        if type(receipt) is not self._native.RetainedReader:
            raise BindingUnbound("foreign_descriptor_reader_release")
        self._native.release_descriptor_reader(self._session, self._epoch, receipt)

    def release_descriptor(self, receipt):
        require_fixed_native_bridge(self._native)
        if type(receipt) is not self._native.RetainedDescriptor:
            raise BindingUnbound("native_foreign_descriptor_release")
        self._native.release_descriptor(self._session, self._epoch, receipt)

    def read_owner_history(self):
        self._current_handles()
        return self._native.read_owner_history(self._session, self._epoch)

    def history_bytes(self, batch):
        self._current_handles()
        if type(batch) is not self._native.HistoryBatch:
            raise BindingUnbound("foreign_history_delivery")
        return self._native.history_bytes(self._session, self._epoch, batch)

    def accept_owner_history(self, batch):
        self._current_handles()
        if type(batch) is not self._native.HistoryBatch:
            raise BindingUnbound("foreign_history_ack")
        # The missing native client must have authenticated actual acceptance.
        # Calling this Python method or possessing bytes is never that receipt.
        self._native.ack_owner_history(self._session, self._epoch, batch)
