def _historical_identity(info):
    return [info.st_dev,info.st_ino,info.st_mode,info.st_uid,info.st_gid,
            info.st_nlink,info.st_size,info.st_mtime_ns,info.st_ctime_ns]

def _compile_tree_checker(onboarding_raw,sealed,expected_files,expected_directories,roots):
    import os
    import stat
    node=function(onboarding_raw,'_inactive_worker_tree')
    original=copy.deepcopy(node)
    anchor='stat.S_IMODE(before.st_mode) != 384'
    changed=0
    for parent in ast.walk(node):
        for field,value in ast.iter_fields(parent):
            if isinstance(value,list):
                for i,child in enumerate(value):
                    if isinstance(child,ast.Compare) and ast.unparse(child)==anchor:
                        value[i]=ast.parse('not _S356_carrier_mode(name,before)',mode='eval').body
                        changed+=1
    need(changed==1,'current6_exact_tree_mode_anchor_required')
    inverse=copy.deepcopy(node)
    for parent in ast.walk(inverse):
        for field,value in ast.iter_fields(parent):
            if isinstance(value,list):
                for i,child in enumerate(value):
                    if isinstance(child,ast.UnaryOp) and ast.unparse(child)=='not _S356_carrier_mode(name, before)':
                        value[i]=ast.parse(anchor,mode='eval').body
    need(ast.dump(inverse,include_attributes=False)==ast.dump(original,include_attributes=False),
         'current6_exact_tree_inverse_failed')
    def mode(name,info):
        if name not in sealed:return stat.S_IMODE(info.st_mode)==0o600
        row=sealed[name]
        return (stat.S_IMODE(info.st_mode)==0o400 and _historical_identity(info)==row['identity']
                and info.st_size==row['bytes'])
    ns={'os':os,'hashlib':hashlib,'_S356_carrier_mode':mode}
    exec(compile(ast.fix_missing_locations(ast.Module(body=[node],type_ignores=[])),
         '<current6-exact-local-sealed-tree>','exec'),ns)
    def check(folder):
        need(folder in roots,'current6_exact_tree_location_required')
        need(folder.resolve()==folder,'current6_tree_parent_link_forbidden')
        parent=folder.parent.lstat()
        need(stat.S_ISDIR(parent.st_mode) and parent.st_uid==os.getuid()
             and stat.S_IMODE(parent.st_mode)==0o700,'current6_private_tree_parent_required')
        tree=ns['_inactive_worker_tree'](folder)
        need(set(tree['files'])==set(expected_files)
             and set(tree['directories'])==set(expected_directories),
             'current6_exact_complete_tree_required')
        for name,row in tree['files'].items():
            need(row['sha256']==expected_files[name]['sha256'], 'current6_exact_tree_content_required')
        return tree
    check.inverse_exact=True
    check.canonical_source=ast.unparse(ast.fix_missing_locations(node))+'\n'
    return check
