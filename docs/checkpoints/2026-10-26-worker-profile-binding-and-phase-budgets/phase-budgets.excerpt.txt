    def start_worker(self, Budget, seconds, boot):
        """Allocate the first worker clock once, after read-only preparation."""
        need(self.entered and self.worker_budget is None, 'first_worker_clock_once_required')
        need(type(seconds) is int and 30 <= seconds <= 600,
             'current_runtime_bounded_worker_seconds_required')
        parent = self.budget
        need(parent.seconds == 1800 and parent.started == self.accepted
             and parent.deadline == parent.original_deadline == self.accepted + 1800,
             'original_preparation_parent_required')
        parent.check(reserve=seconds + 180)
        started = time.monotonic()
        worker = Budget(seconds, started=started)
        self.worker_budget = worker  # A failed sink cannot be retried in this invocation.
        need(worker.deadline <= parent.deadline - 180,
             'worker_clock_and_parent_cleanup_reserve_required')
        require_policy_window(self.policy_successor, worker)
        claim = {'boot_id':boot, 'started_mono':worker.started,
                 'deadline_mono':worker.deadline, 'budget_seconds':seconds}
        self.preserve('worker_clock', {
            'schema':'frw.Root.R460.first-worker-clock.v1',
            'parent':{'started_monotonic':parent.started, 'seconds':1800,
                      'reserve':180, 'deadline_monotonic':parent.original_deadline},
            'worker':claim, 'allocated_after_initial_sources':True,
            'worker_effect_started':False, 'retry':False})
        worker.check(); parent.check(reserve=180)
        return worker, claim

    def commit_parent(self):
        need(self.worker_budget is not None and self.budget.started == self.accepted
             and self.budget.seconds == 1800
             and self.budget.deadline == self.budget.original_deadline == self.accepted + 1800,
             'completed_worker_original_commit_parent_required')
        self.budget.check(reserve=180)
        return self.budget
