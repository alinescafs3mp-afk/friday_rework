"""Original frontend descriptor custody. Source only; kernel qualification pending.

The anchor's duplication is the admission linearization point. F_DUPFD_QUERY
compares open file descriptions, including same-inode close/reopen. Writes
use a nonblocking description opened from the retained anchor, never a later
numeric frontend FD. An unobserved swap-and-restore cannot redirect a write;
NEW proposal additionally requires a retained native commit-history capability;
without independently qualified complete companion coverage it refuses. Snapshot
comparisons alone still cannot prove historical transitions.
"""
import fcntl
import io
import os
import stat
import sys

QUERY = 1027  # exact selected /usr/include/linux/fcntl.h, BASE1024 + 3
LIMIT = 384 * 1024

def identity(s):
    return s.st_dev, s.st_ino, s.st_mode, s.st_rdev

class DescriptorUnconfirmed(RuntimeError):
    pass

class _ReaderCustody:
    """Fixed slots reserved before acquisition; uncertainty remains owned."""
    __slots__ = ("receipt", "fd", "attempted", "acquisition_started", "uncertain", "confirmed", "tracked")

    def __init__(self):
        self.receipt = self.fd = None
        self.attempted = self.acquisition_started = self.uncertain = self.confirmed = self.tracked = False


class OriginalDescriptor:
    def __init__(self, stream):
        self.stream = stream
        self._birth_stream = stream
        self.failure = None
        self.anchor = None
        self.native_history = None
        self.native_authority = None
        self.readers = {}
        self.reader_custody = []
        self.reader_total = 0  # lifetime32; never replenished by release
        self.close_attempted = False
        self.close_errors = []
        self.closed = False
        self.memory = type(stream) is io.StringIO
        if self.memory:
            if stream.closed or stream.tell() > LIMIT or stream.tell() != len(stream.getvalue()):
                raise ValueError("original_memory_not_append")
            self.initial_position = stream.tell()
            return
        if sys.platform != "linux" or type(stream) is not io.TextIOWrapper or stream.closed:
            raise ValueError("original_stream_not_supported")
        if stream.encoding.lower().replace("_", "-") not in {"utf-8", "ascii"}:
            raise ValueError("original_encoding_not_supported")
        buf = stream.buffer
        self.raw = buf.raw if type(buf) is io.BufferedWriter else buf
        if type(self.raw) is not io.FileIO or self.raw.closed:
            raise ValueError("original_raw_not_exact")
        self.number = self.raw.fileno()
        before = os.fstat(self.number)
        self.stat = identity(before)
        self.flags = fcntl.fcntl(self.number, fcntl.F_GETFL)
        if not ((stat.S_ISCHR(before.st_mode) and os.isatty(self.number)) or stat.S_ISFIFO(before.st_mode)):
            raise ValueError("original_descriptor_not_tty_or_pipe")
        if self.flags & os.O_ACCMODE == os.O_RDONLY:
            raise ValueError("original_descriptor_readonly")
        # Exact native held custody MUST acquire/duplicate atomically. Python
        # never closes a later numeric FD as a replacement for that capability.
        anchor = None
        self._birth_raw = self.raw
        self._birth_number = self.number
        self._birth_anchor = anchor
        self._birth_stat = self.stat
        self._birth_flags = self.flags
        try:
            # Unsupported/denied kernel comparison refuses this capability;
            # inode equality alone is never an open-description attestation.
            from hermes_cli.friday_cli_binding import current_binding
            binding = current_binding()
            self.native_authority = binding.authority
            self.native_history = self.native_authority.retain_descriptor(self.number)
            anchor = self.native_authority.descriptor_anchor(self.native_history)
            if type(anchor) is not int or anchor < 0:
                raise DescriptorUnconfirmed("native_anchor_observation_shape")
            self.anchor = self._birth_anchor = anchor
            self.validate()
        except BaseException as primary:
            try:
                self.close()
            except BaseException as cleanup:
                group = BaseExceptionGroup("original_descriptor_admission_and_close", [primary, cleanup])
                group.descriptor_custody = self
                raise group
            raise

    def validate(self):
        if self.failure is not None:
            raise DescriptorUnconfirmed("original_descriptor_prior_observed_failure")
        try:
            self._validate_birth()
        except BaseException as exc:
            # A restored dynamic field or numeric FD never clears an observed
            # replacement. Invisible swap-and-restore history is not claimed.
            self.failure = type(exc).__name__
            raise

    def _validate_birth(self):
        if self.stream is not self._birth_stream:
            raise DescriptorUnconfirmed("original_stream_object_replaced")
        if self.closed:
            raise DescriptorUnconfirmed("original_descriptor_closed")
        if self.memory:
            if self.stream.closed or self.stream.tell() > 2 * LIMIT:
                raise DescriptorUnconfirmed("original_memory_closed_or_over_limit")
            return
        if self.native_history is None or self.native_authority is None:
            raise DescriptorUnconfirmed("native_true_birth_descriptor_history_unbound")
        self.native_authority.validate_descriptor(self.native_history)
        if (self.raw is not self._birth_raw or self.number != self._birth_number
                or self.anchor != self._birth_anchor or self.stat != self._birth_stat
                or self.flags != self._birth_flags
                or self.raw.closed or self.raw.fileno() != self._birth_number
                or identity(os.fstat(self.number)) != self.stat
                or identity(os.fstat(self.anchor)) != self.stat
                or fcntl.fcntl(self.number, fcntl.F_GETFL) != self.flags
                or fcntl.fcntl(self.anchor, fcntl.F_GETFL) != self.flags
                or fcntl.fcntl(self.number, QUERY, self.anchor) != 1):
            raise DescriptorUnconfirmed("original_open_description_replaced")

    def reopen(self):
        self.validate()
        row = None
        receipt = None
        try:
            if self.reader_total >= 32:
                self.failure = "original_private_descriptor_capacity"
                raise DescriptorUnconfirmed(self.failure)
            row = _ReaderCustody()
            self.reader_custody.append(row)  # fallible allocation BEFORE acquisition
            row.tracked = True
            self.reader_total += 1
            row.acquisition_started = True
            receipt = self.native_authority.reopen_descriptor(self.native_history)
            row.receipt = receipt  # existing slot, no dictionary allocation
            fd = self.native_authority.descriptor_reader_fd(receipt)
            if type(fd) is not int or fd < 0 or fd in self.readers:
                raise DescriptorUnconfirmed("native_reader_observation_shape")
            row.fd = fd
            self.readers[fd] = row
            self.validate()
            if (identity(os.fstat(fd)) != self.stat
                    or not fcntl.fcntl(fd, fcntl.F_GETFL) & os.O_NONBLOCK
                    or not fcntl.fcntl(fd, fcntl.F_GETFD) & fcntl.FD_CLOEXEC):
                raise DescriptorUnconfirmed("anchored_reopen_changed")
            return fd
        except BaseException as primary:
            if self.failure is None:
                self.failure = "reader_admission_or_publication_failed"
            if row is not None:
                row.uncertain = True
                if receipt is not None:
                    row.receipt = receipt
                    try:
                        self._release_reader(row)
                    except BaseException as cleanup:
                        raise BaseExceptionGroup("reader_admission_and_retained_close", [primary, cleanup])
                elif not row.acquisition_started:
                    if row.tracked:
                        self.reader_custody.remove(row)  # no native attempt/held custody
                # Unknown native effect: keep the row; missing provider must own
                # the original ledger. Dropping Python refs never confirms close.
            raise

    def _release_reader(self, row):
        try:
            found = False
            for value in self.reader_custody:
                if value is row:
                    found = True
            if not found or row.attempted or row.receipt is None:
                raise DescriptorUnconfirmed("foreign_duplicate_or_unknown_descriptor_release")
            row.attempted = True  # BEFORE native effect; never retry unknown close
            try:
                self.native_authority.release_descriptor_reader(row.receipt)
            except BaseException:
                row.uncertain = True
                self.failure = "private_descriptor_close_unknown"
                raise  # preserve cause; no fallible append needed to retain custody
            row.confirmed = True  # only authenticated provider success may confirm
            row.uncertain = False
            if row.fd is not None and self.readers.get(row.fd) is row:
                del self.readers[row.fd]
            self.reader_custody.remove(row)
        except BaseException:
            if self.failure is None:
                self.failure = "private_descriptor_close_unknown"
            row.uncertain = True
            raise

    def release(self, fd):
        if type(fd) is not int or fd not in self.readers:
            raise DescriptorUnconfirmed("foreign_descriptor_release")
        self._release_reader(self.readers[fd])

    def close(self):
        if self.closed:
            return
        if self.close_attempted:
            raise DescriptorUnconfirmed("original_descriptor_close_not_repeated")
        if self.reader_custody or self.readers or self.close_errors:
            raise DescriptorUnconfirmed("original_descriptor_readers_not_closed")
        self.close_attempted = True
        if self.memory:
            self.closed = True
            return
        if self.native_history is None or self.native_authority is None:
            self.failure = "native_original_close_custody_missing"
            raise DescriptorUnconfirmed(self.failure)
        try:
            self.native_authority.release_descriptor(self.native_history)
        except BaseException as exc:
            self.failure = "native_anchor_close_unknown"
            self.close_errors.append({"fd": self._birth_anchor, "error": type(exc).__name__})
            raise DescriptorUnconfirmed(self.failure) from exc
        self.native_history = None
        self.anchor = None
        self.closed = True
