static bool frwk_bounds_locked(const struct frwk_epoch *e)
{
 return e && e->tasks_n<=FRWK_TASKS && e->tables_n<=FRWK_TABLES &&
  e->ofds_n<=FRWK_OFDS && e->slots_n<=FRWK_SLOTS &&
  e->retained_n<=FRWK_RETAINS && e->rings_n<=FRWK_RINGS &&
  e->records_n<=FRWK_RECORDS && e->events_n<=FRWK_EVENTS &&
  e->producer<FRWK_EVENTS && e->consumer<FRWK_EVENTS;
}
static bool frwk_hold_layout_locked(const struct frwk_epoch *e);
static int frwk_admission_locked(const struct frwk_epoch *e)
{
 if (!e || e->close_state!=FRWK_EPOCH_OPEN || e->close_hi || e->close_lo ||
     e->close_final_ns || e->final_refs.state!=FRWK_BATCH_EMPTY) return -ESHUTDOWN;
 if (!frwk_bounds_locked(e) || !frwk_hold_layout_locked(e)) return -EUCLEAN;
 return 0;
}
/* Counts and pointer ownership, not pointer identity, define the inventory.
 * A retired NULL tombstone owns nothing; a live NULL entry or an owned pointer
 * hidden beyond a lifetime count is corruption. No deduplication is allowed. */
static bool frwk_hold_layout_locked(const struct frwk_epoch *e)
{
 u32 i;
 if (!frwk_bounds_locked(e)) return false;
 for (i=0;i<FRWK_TASKS;i++)
  if ((i<e->tasks_n && (!e->tasks[i].id || (e->tasks[i].live && !e->tasks[i].ref))) ||
      (i>=e->tasks_n && (e->tasks[i].ref || e->tasks[i].live || e->tasks[i].id))) return false;
 for (i=0;i<FRWK_TABLES;i++)
  if ((i<e->tables_n && (!e->tables[i].id || (e->tables[i].live && !e->tables[i].ref))) ||
      (i>=e->tables_n && (e->tables[i].ref || e->tables[i].live || e->tables[i].id))) return false;
 for (i=0;i<FRWK_OFDS;i++)
  if ((i<e->ofds_n && (!e->ofds[i].id || (e->ofds[i].live && !e->ofds[i].ref))) ||
      (i>=e->ofds_n && (e->ofds[i].ref || e->ofds[i].live || e->ofds[i].id))) return false;
 for (i=0;i<FRWK_RINGS;i++)
  if ((i<e->rings_n && (!e->rings[i].id || (e->rings[i].live && !e->rings[i].ref))) ||
      (i>=e->rings_n && (e->rings[i].ref || e->rings[i].live || e->rings[i].id))) return false;
 for (i=0;i<FRWK_RETAINS;i++)
  if ((i<e->retained_n && (!e->retained[i].id || (e->retained[i].live && !e->retained[i].held))) ||
      (i>=e->retained_n && (e->retained[i].held || e->retained[i].live || e->retained[i].id))) return false;
 return true;
}

static bool frwk_original_custody_locked(const struct frwk_epoch *e)
{
 /* owner_ref must be separately acquired from any equal tasks[].ref.
  * scope_ref must own cgroup_get custody. That bootstrap is still absent. */
 return e && e->hi && e->lo && e->owner_ref && e->scope_ref &&
  e->useful_ns && e->useful_ns<=e->final_ns;
}
static bool frwk_batch_bound_locked(const struct frwk_epoch *e,
 const struct frwk_reference_batch *b)
{
 return b && b->owner_epoch==e && b->hi==e->hi && b->lo==e->lo &&
  b->final_ns && b->final_ns==e->final_ns &&
  ((b->kind==FRWK_BATCH_RETIRED && e->retired_batch==b &&
    (e->close_state==FRWK_EPOCH_OPEN || e->close_state==FRWK_EPOCH_CLOSING)) ||
   (b->kind==FRWK_BATCH_FINAL && b==&e->final_refs &&
    e->close_state==FRWK_EPOCH_TRANSFERRED && e->close_hi==e->hi &&
    e->close_lo==e->lo && e->close_final_ns==e->final_ns));
}
static bool frwk_batch_layout(const struct frwk_reference_batch *b)
{
 u32 i;
 if (b->tasks_n>FRWK_TASKS || b->tables_n>FRWK_TABLES ||
     b->files_n>FRWK_OFDS+FRWK_RINGS+FRWK_RETAINS ||
     b->tasks_i>b->tasks_n || b->tables_i>b->tables_n || b->files_i>b->files_n)
  return false;
 for (i=0;i<FRWK_TASKS;i++)
  if ((i>=b->tasks_i && i<b->tasks_n) ? !b->tasks[i] : !!b->tasks[i]) return false;
 for (i=0;i<FRWK_TABLES;i++)
  if ((i>=b->tables_i && i<b->tables_n) ? !b->tables[i] : !!b->tables[i]) return false;
 for (i=0;i<FRWK_OFDS+FRWK_RINGS+FRWK_RETAINS;i++)
  if ((i>=b->files_i && i<b->files_n) ? !b->files[i] : !!b->files[i]) return false;
 if (b->kind==FRWK_BATCH_RETIRED)
  return !b->owner && !b->scope && !b->owner_issued && !b->scope_issued;
 if (b->kind!=FRWK_BATCH_FINAL) return false;
 return (b->owner_issued ? !b->owner : !!b->owner) &&
        (b->scope_issued ? !b->scope : !!b->scope);
}
static int frwk_partial_locked(struct frwk_epoch *e,
 struct frwk_reference_batch *b, int error)
{
 b->state=FRWK_BATCH_PARTIAL;
 if (b==&e->final_refs) e->close_state=FRWK_EPOCH_PARTIAL;
 /* Keep pending slots, original deadline and uncertain in-flight marker.
  * No counter reset, authority renewal, slot reuse or retry is provided. */
 return error;
}

static int sticky(struct frwk_epoch *e, u64 loss)
{
 e->fence_seq=0;
 e->loss |= loss;
 e->final_only = true;
 return -EUCLEAN;
}
static u64 generation(struct frwk_epoch *e)
{
 if (e->generation >= U64_MAX-1 || e->records_n == FRWK_RECORDS) {
  sticky(e, 16); return 0;
 }
 e->records_n++;
 return ++e->generation;
}

int frwk_publish_locked(struct frwk_epoch *e, struct frwk_event *v)
{
 int admission=frwk_admission_locked(e);
 if (admission) return admission;
 {
 if (e->loss) return -EUCLEAN;
 if (!e->prearmed || e->coverage!=FRWK_REQUIRED || frwk_source_coverage!=FRWK_REQUIRED)
  return sticky(e, 4);
 if (e->seq>=U64_MAX-1 || e->events_n==FRWK_EVENTS) return sticky(e, 2|16);
 e->fence_seq=0;
 v->magic=0x46525731; v->version=1; v->epoch_hi=e->hi; v->epoch_lo=e->lo;
 v->mono_ns=ktime_get_ns();

 e->events[e->producer]=*v;
 e->producer=(e->producer+1)%FRWK_EVENTS; e->events_n++;

 }
}

 if (!e->close_final_ns || e->close_hi!=e->hi || e->close_lo!=e->lo ||
     e->close_final_ns!=e->final_ns || !frwk_original_custody_locked(e) ||
     !frwk_hold_layout_locked(e)) return -EUCLEAN;
 b=&e->final_refs;
 if (!frwk_batch_empty(b) || e->retired_batch) return -EBUSY;
 if (ktime_get_ns()>=e->close_final_ns) {
  e->close_state=FRWK_EPOCH_PARTIAL;return -ETIME;
 }
 /* This actual absent native precondition is fail-closed, not a fabricated
  * detach fence. The bounded transfer below is a Source facility, UNATTACHED.
  * A lost epoch is not required to regain coverage/current/scope permission.
  * Real hook exclusion and original epoch memory lifetime are still missing. */
 if (frwk_source_disposal_lifetime!=FRWK_REQUIRED) return -EOPNOTSUPP;
 for (i=0;i<e->tasks_n;i++) if (e->tasks[i].ref) nt++;
 for (i=0;i<e->tables_n;i++) if (e->tables[i].ref) nb++;
 for (i=0;i<e->ofds_n;i++) if (e->ofds[i].ref) nf++;
 for (i=0;i<e->rings_n;i++) if (e->rings[i].ref) nf++;
 for (i=0;i<e->retained_n;i++) if (e->retained[i].held) nf++;
 if (nt>FRWK_TASKS || nb>FRWK_TABLES || nf>FRWK_OFDS+FRWK_RINGS+FRWK_RETAINS) return -EOVERFLOW;
 b->owner_epoch=e;b->hi=e->close_hi;b->lo=e->close_lo;b->final_ns=e->close_final_ns;
 b->kind=FRWK_BATCH_FINAL;
 for (i=0;i<e->tasks_n;i++) {
  if (e->tasks[i].ref) {b->tasks[b->tasks_n++]=e->tasks[i].ref;e->tasks[i].ref=NULL;}
  e->tasks[i].live=false;
 }
 for (i=0;i<e->tables_n;i++) {
  if (e->tables[i].ref) {b->tables[b->tables_n++]=e->tables[i].ref;e->tables[i].ref=NULL;}
  e->tables[i].live=false;
 }
 for (i=0;i<e->ofds_n;i++) {
  if (e->ofds[i].ref) {b->files[b->files_n++]=e->ofds[i].ref;e->ofds[i].ref=NULL;}
  e->ofds[i].live=false;
 }
 for (i=0;i<e->rings_n;i++) {
  if (e->rings[i].ref) {b->files[b->files_n++]=e->rings[i].ref;e->rings[i].ref=NULL;}
  e->rings[i].live=false;
 }
 for (i=0;i<e->retained_n;i++) {
  if (e->retained[i].held) {b->files[b->files_n++]=e->retained[i].held;e->retained[i].held=NULL;}
  e->retained[i].live=false;
 }
 /* These are additional, independently acquired original holds. In
  * particular owner and an equal tasks[] pointer each require their own put. */
 b->owner=e->owner_ref;e->owner_ref=NULL;
 b->scope=e->scope_ref;e->scope_ref=NULL;
 b->state=FRWK_BATCH_READY;e->close_state=FRWK_EPOCH_TRANSFERRED;
 /* Lifetime counts/retained tombstones, loss, deadlines, sequence, delivery
  * and incomplete copy evidence are preserved. No quiescence is asserted. */
 return 0;
}
static int frwk_put_one(struct frwk_epoch *e,struct frwk_reference_batch *b,
 enum frwk_put_kind kind)
{
 struct task_struct *task=NULL;struct files_struct *table=NULL;
 struct file *file=NULL;struct cgroup *scope=NULL;unsigned long flags;int rc=0;
 raw_spin_lock_irqsave(&e->lock,flags);
 if (!frwk_batch_bound_locked(e,b) || b->state!=FRWK_BATCH_DRAINING ||
     b->put_inflight || !frwk_batch_layout(b)) {
  rc=frwk_partial_locked(e,b,-EUCLEAN);goto unlock;
 }
 if (ktime_get_ns()>=b->final_ns) {
  rc=frwk_partial_locked(e,b,-ETIME);goto unlock;
 }
 switch (kind) {
 case FRWK_PUT_TASK:
  if (b->tasks_i>=b->tasks_n || !b->tasks[b->tasks_i]) {rc=-EUCLEAN;break;}
  b->put_index=b->tasks_i;task=b->tasks[b->tasks_i];
  b->tasks[b->tasks_i++]=NULL;break;
 case FRWK_PUT_TABLE:
  if (b->tables_i>=b->tables_n || !b->tables[b->tables_i]) {rc=-EUCLEAN;break;}
  b->put_index=b->tables_i;table=b->tables[b->tables_i];
  b->tables[b->tables_i++]=NULL;break;
 case FRWK_PUT_FILE:
  if (b->files_i>=b->files_n || !b->files[b->files_i]) {rc=-EUCLEAN;break;}
  b->put_index=b->files_i;file=b->files[b->files_i];
  b->files[b->files_i++]=NULL;break;
 case FRWK_PUT_OWNER:
  if (b->kind!=FRWK_BATCH_FINAL || b->owner_issued || !b->owner) {rc=-EUCLEAN;break;}
  b->put_index=0;task=b->owner;b->owner=NULL;b->owner_issued=true;break;
 case FRWK_PUT_SCOPE:
  if (b->kind!=FRWK_BATCH_FINAL || b->scope_issued || !b->scope) {rc=-EUCLEAN;break;}
  b->put_index=0;scope=b->scope;b->scope=NULL;b->scope_issued=true;break;
 default:rc=-EINVAL;
 }
 if (rc) {rc=frwk_partial_locked(e,b,rc);goto unlock;}
 b->put_kind=kind;b->put_inflight=true;
 raw_spin_unlock_irqrestore(&e->lock,flags);
 /* Each slot's owning pointer was cleared ONCE before its concrete put.
  * No allocator, callback, sleep or put occurs under the epoch raw spinlock.
  * These void primitives provide no native final-completion/error receipt. */
 if (task) put_task_struct(task);
 else if (table) put_files_struct(table);
 else if (file) fput(file);
 else cgroup_put(scope);
 raw_spin_lock_irqsave(&e->lock,flags);
 if (!frwk_batch_bound_locked(e,b) || b->state!=FRWK_BATCH_DRAINING ||
     !b->put_inflight || b->put_kind!=kind) {
  /* A put was invoked, but its associated receipt is uncertain. Never replay. */
  rc=frwk_partial_locked(e,b,-EUCLEAN);goto unlock;
 }
 b->put_inflight=false;
 if (ktime_get_ns()>=b->final_ns) rc=frwk_partial_locked(e,b,-ETIME);
unlock:
 raw_spin_unlock_irqrestore(&e->lock,flags);return rc;
}
int frwk_put_reference_batch(struct frwk_epoch *e,struct frwk_reference_batch *b)
{
 unsigned long flags;u32 i,nt,nb,nf;enum frwk_batch_kind kind;int rc;
 if (!e || !b) return -EINVAL;
 /* The still-missing native custodian must hold original epoch/batch memory
  * alive and supply real sleepable, exclusive lifetime. Context predicates
  * reject atomic callers; they do not manufacture that missing qualification. */
 if (in_atomic() || irqs_disabled()) return -EWOULDBLOCK;
 raw_spin_lock_irqsave(&e->lock,flags);
 if (!frwk_bounds_locked(e) || !frwk_batch_bound_locked(e,b)) {rc=-ESTALE;goto unlock;}
 if (b->state!=FRWK_BATCH_READY) {rc=-EALREADY;goto unlock;}
 if (!frwk_batch_layout(b) || b->put_inflight || b->tasks_i || b->tables_i ||
     b->files_i || b->owner_issued || b->scope_issued) {rc=-EUCLEAN;goto unlock;}
 if (ktime_get_ns()>=b->final_ns) {rc=frwk_partial_locked(e,b,-ETIME);goto unlock;}
 if (frwk_source_disposal_lifetime!=FRWK_REQUIRED) {rc=-EOPNOTSUPP;goto unlock;}
 nt=b->tasks_n;nb=b->tables_n;nf=b->files_n;kind=b->kind;
 b->state=FRWK_BATCH_DRAINING;
 raw_spin_unlock_irqrestore(&e->lock,flags);
 /* At most 64+64+(1024+64+32)+1+1 = 1250 issued puts.
  * This is one drain attempt. Partial/deadline/uncertain states are terminal. */
 for (i=0;i<nt;i++) {rc=frwk_put_one(e,b,FRWK_PUT_TASK);if (rc) return rc;}
 for (i=0;i<nb;i++) {rc=frwk_put_one(e,b,FRWK_PUT_TABLE);if (rc) return rc;}
 for (i=0;i<nf;i++) {rc=frwk_put_one(e,b,FRWK_PUT_FILE);if (rc) return rc;}
 if (kind==FRWK_BATCH_FINAL) {
  rc=frwk_put_one(e,b,FRWK_PUT_OWNER);if (rc) return rc;
  rc=frwk_put_one(e,b,FRWK_PUT_SCOPE);if (rc) return rc;
 }
 raw_spin_lock_irqsave(&e->lock,flags);
 if (!frwk_batch_bound_locked(e,b) || b->state!=FRWK_BATCH_DRAINING ||
     b->put_inflight || !frwk_batch_layout(b) || b->tasks_i!=b->tasks_n ||
     b->tables_i!=b->tables_n || b->files_i!=b->files_n ||
     (kind==FRWK_BATCH_FINAL && (!b->owner_issued || !b->scope_issued))) {
  rc=frwk_partial_locked(e,b,-EUCLEAN);goto unlock;
 }
 if (ktime_get_ns()>=b->final_ns) {rc=frwk_partial_locked(e,b,-ETIME);goto unlock;}
 b->state=FRWK_BATCH_PUTS_ISSUED;
 if (kind==FRWK_BATCH_RETIRED) e->retired_batch=NULL;
 else e->close_state=FRWK_EPOCH_PUTS_ISSUED;
 /* fput may defer final release; these calls also do not prove cgroup resource
  * release, observer/hook quiescence, epoch memory free or ResourceGo. */
 rc=0;
unlock:
 raw_spin_unlock_irqrestore(&e->lock,flags);return rc;
}
