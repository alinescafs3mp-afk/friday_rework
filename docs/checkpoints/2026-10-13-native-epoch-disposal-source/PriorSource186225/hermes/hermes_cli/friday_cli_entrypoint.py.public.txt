"""Existing product CLI entrypoint ownership extension; NEW UNQUALIFIED Source.

One fixed outside owner stays outside the whole original CLI cgroup. Its only
child is the SAME existing product entrypoint, original caller/UI/proxy/loop.
There is no developer TUI, broker, new model, retry or replacement frontend.
"""
from dataclasses import dataclass
import fcntl
import json
import os
from pathlib import Path
import select
import socket
import sys
import time

from hermes_cli.friday_cli_binding import current_binding, BindingUnbound
from hermes_cli.friday_cli_birth import OriginalBirthBoundary, birth, group
from hermes_cli.friday_cli_ipc import OriginalIPCClient, encode, decode, request, recv_credentialed

_CHILD = None


@dataclass(eq=False)
class ChildContext:
    binding: object
    ipc: OriginalIPCClient


def current_child():
    return _CHILD


def own_existing_entrypoint():
    """Called by EXISTING hermes_cli.main.main, before chat/model/native effects."""
    if _CHILD is not None:
        return None
    # No serialized options/environment can turn this refusal into Go.
    binding=current_binding()
    owner=FixedOriginalCLIOwner(binding)
    return owner.run(tuple(sys.argv))


class FixedOriginalCLIOwner:
    def __init__(self,binding):
        from hermes_cli.friday_native_supervision import NativeSupervisor
        self.binding=binding.require("outside_owner_constructor_before_effects")
        self.birth=birth(os.getpid())[0]
        self.native=OriginalBirthBoundary(binding)
        self.supervisor=NativeSupervisor()
        self.child=None
        self.sock=None
        self.inputfd=None
        self.sequence=0
        self.leases={}
        self.active=None
        self.primary=None
        self.secondary=[]
        self.closed=False
        self.launch_intent="NOT_ENTERED"
        self.whole_stop_started=False

    def _current(self, *, cleanup_child=None, deadline=None):
        if cleanup_child is None:
            self.binding.require("outside_owner_current_native_boundary")
        else:
            self.binding.require_retained_cleanup(
                "outside_owner_retained_cleanup", native_boundary=self.native,
                child=cleanup_child, deadline=deadline)
        if birth(os.getpid())[0]!=self.birth or group(os.getpid())==self.binding.exclusive_cgroup or group(os.getpid()).startswith(self.binding.exclusive_cgroup+"/"):
            raise BindingUnbound("whole_original_owner_inside_target_or_replaced")

    def _send(self,response,end):
        packet=encode(response)
        while True:
            left=end-time.monotonic()
            if left<=0:raise TimeoutError("outside_owner_response_cutoff")
            try:
                if self.sock.send(packet)!=len(packet):raise RuntimeError("outside_owner_short_response")
                return
            except BlockingIOError:
                select.select([],[self.sock],[],left)

    def _body(self,operation,body):
        fields={
          "CLAIM":{"call_id","task_id","entry","deadline","native_tid","authority_generation"},
          "CHECK":{"phase","deadline"},"NARROW":{"deadline"},"ACTIVITY":{"ordinal"},
          "PROXY":{"ordinal","deadline"},"REPAINT":{"ordinal","deadline"},"PROMPT":{"ordinal","deadline"},
          "TASK_RESERVE":{"ordinal","deadline"},"TASK_ATTACH":{"ordinal"},
          "RECEIPTS":{"candidates"},"FINISH":{"returned_at","deadline","candidates"},
          "STOP":{"deadline","primary"}}
        if set(body)!=fields[operation]:
            raise BindingUnbound("outside_owner_fixed_operation_schema")
        if "deadline" in body and (type(body["deadline"]) not in (float,int) or not 0<body["deadline"]):
            raise BindingUnbound("outside_owner_invalid_absolute_cutoff")
        if "ordinal" in body and (type(body["ordinal"]) is not int or not 1<=body["ordinal"]<=4096):
            raise BindingUnbound("outside_owner_invalid_participant_ordinal")

    def _candidates(self, value):
        import re
        if type(value) is not list or len(value) > 256:
            raise BindingUnbound("outside_owner_candidate_quota")
        for item in value:
            if (type(item) is not dict or set(item) != {"selection", "commit", "generation"}
                    or type(item["generation"]) is not int or item["generation"] < 1
                    or type(item["selection"]) is not str or not re.fullmatch(r"[0-9a-f]{64}", item["selection"])
                    or type(item["commit"]) is not str or not re.fullmatch(r"[0-9a-f]{64}", item["commit"])):
                raise BindingUnbound("outside_owner_candidate_exact_schema")

    def _dispatch(self,value):
        import secrets
        self._current()
        request(value)
        if value["seq"]!=self.sequence+1:
            raise BindingUnbound("outside_owner_replay_or_missing_sequence")
        self.sequence=value["seq"]
        op=value["operation"];body=value["body"];self._body(op,body)
        if op=="CLAIM":
            if self.active is not None or value["lease"] is not None:
                raise BindingUnbound("outside_owner_overlapping_original_claim")
            entry=body["entry"];end=body["deadline"];now=time.monotonic()
            if (type(entry) not in (float,int) or not 0<entry<=now<end<=entry+30
                    or end>self.binding.session_deadline
                    or type(body["native_tid"]) is not int
                    or not Path(f"/proc/{self.child.pid}/task/{body['native_tid']}").exists()
                    or type(body["authority_generation"]) is not int or body["authority_generation"]<1
                    or any(type(body[k]) is not str or not body[k] or len(body[k].encode())>512 for k in ("call_id","task_id"))):
                raise BindingUnbound("outside_owner_original_claim_identity_or_clock")
            native_id=secrets.token_hex(16)
            row={"id":native_id,"entry":entry,"end":end,"useful":end-3,
                 "call_id":body["call_id"],"task_id":body["task_id"],"state":"LIVE",
                 "participants":[],"task_attachments":[],"candidates":[],"primary":None,"stop_started":False}
            self.leases[native_id]=row;self.active=row
        else:
            row=self.leases.get(value["lease"])
            if row is None or row is not self.active or time.monotonic()>=row["end"]:
                raise BindingUnbound("outside_owner_foreign_or_late_original_lease")
            native_id=row["id"];end=row["end"]
            if "deadline" in body and body["deadline"]>end:
                raise BindingUnbound("outside_owner_original_deadline_widened")
            if op=="NARROW":
                if body["deadline"]<=time.monotonic():
                    raise BindingUnbound("outside_owner_narrowed_expired")
                row["end"]=body["deadline"];row["useful"]=min(row["useful"],row["end"]-3)
            elif op == "TASK_ATTACH":
                ordinal = body["ordinal"]
                if (ordinal in row["task_attachments"]
                        or not any(kind == "TASK_RESERVE" and value["ordinal"] == ordinal
                                   for kind,value in row["participants"])):
                    raise BindingUnbound("outside_owner_task_attach_without_exact_reservation")
                row["task_attachments"].append(ordinal)
            elif op in {"ACTIVITY","PROXY","REPAINT","PROMPT","TASK_RESERVE"}:
                if len(row["participants"])>=4096:
                    raise BindingUnbound("outside_owner_participant_quota")
                row["participants"].append((op,dict(body)))
            elif op=="RECEIPTS":
                if row["state"]!="LIVE" or type(body["candidates"]) is not list or len(body["candidates"])>256:
                    raise BindingUnbound("outside_owner_receipt_phase_or_quota")
                self._candidates(body["candidates"])
                row["candidates"]=body["candidates"];row["state"]="CANDIDATES_DURABLE_NOT_ACCEPTED"
            elif op=="FINISH":
                self._candidates(body["candidates"])
                if (row["state"]!="CANDIDATES_DURABLE_NOT_ACCEPTED"
                        or body["candidates"]!=row["candidates"]
                        or type(body["returned_at"]) not in (int,float)
                        or not row["entry"]<=body["returned_at"]<time.monotonic()<row["end"]):
                    raise BindingUnbound("outside_owner_actual_final_return_or_commit_mismatch")
                # A trusted Source participant reports its ACTUAL direct return,
                # real store returns and settled custody. This is a supervision
                # lease close, not consumer delivery acceptance or self-reap.
                row["state"]="FINAL_ACK_PENDING_NOT_DELIVERED"
                row["finished_at"]=time.monotonic()
                self.active=None
            elif op=="STOP":
                row["primary"]=body["primary"];row["state"]="UNKNOWN"
                self._stop(row)
                raise BindingUnbound("outside_owner_original_stopped_no_client_return")
        self._send({"v":1,"seq":value["seq"],"lease":native_id,
                    "state":"OBSERVED","body":{"state":row["state"]}},min(end,self.binding.session_deadline))

    def _stop(self,row, *, child=None):
        if self.whole_stop_started:
            raise BindingUnbound("outside_owner_duplicate_whole_stop")
        self.whole_stop_started=True
        if row is not None:
            if row["stop_started"]:raise BindingUnbound("outside_owner_duplicate_stop")
            row["stop_started"]=True
            end=min(row["end"],self.binding.session_deadline)
        else:
            end=self.binding.session_deadline
        retained = self.child if child is None else child
        if retained is None and self.native.children:
            retained=self.native.children[0]
        try:
            if retained is None:
                raise BindingUnbound("STOP_UNCONFIRMED:original_PREPID_custody_missing")
            self._current(cleanup_child=retained,deadline=end)
            # Includes a launch failure before assignment to self.child.
            self.supervisor.stop_original_cli(
                self.binding.original_association,binding=self.binding,
                native_birth=self.native,child=retained,deadline=end)
        except BaseException as error:
            self.secondary.append(error)
            if row is not None:row["state"]="STOP_UNCONFIRMED"
            raise

    def run(self,argv):
        self._current()
        peer=None
        try:
            if any(type(a) is not str or "\x00" in a or len(a.encode())>4096 for a in argv) or len(argv)>256:
                raise ValueError("original_product_argv_quota")
            payload=encode({"schema":"friday.original.cli-input.v1","argv":list(argv),"owner_pid":os.getpid()})
            self.inputfd=os.memfd_create("friday-original-cli-input",os.MFD_CLOEXEC|os.MFD_ALLOW_SEALING)
            if os.write(self.inputfd,payload)!=len(payload):raise OSError("short_original_input_write")
            os.lseek(self.inputfd,0,os.SEEK_SET)
            fcntl.fcntl(self.inputfd,fcntl.F_ADD_SEALS,fcntl.F_SEAL_WRITE|fcntl.F_SEAL_GROW|fcntl.F_SEAL_SHRINK|fcntl.F_SEAL_SEAL)
            self.sock,peer=socket.socketpair(socket.AF_UNIX,socket.SOCK_SEQPACKET)
            self.sock.setsockopt(socket.SOL_SOCKET,socket.SO_PASSCRED,1);self.sock.setblocking(False)
            self.launch_intent="PREPID_UNKNOWN"
            self.supervisor.verify_original_launch(self.binding.original_association,binding=self.binding,
                                                    deadline=self.binding.session_deadline)
            self.child=self.native.launch(self.inputfd,peer.fileno())
            self.launch_intent="KNOWN_ORIGINAL_PID_EXEC_PENDING"
            peer.close();peer=None
            fd=self.inputfd;self.inputfd=None;os.close(fd)
            while time.monotonic()<self.binding.session_deadline:
                self.native.observe_exec(self.child)
                if self.native.reap(self.child):
                    self.native.close(self.child)
                    if self.active is not None or self.child.errors:
                        raise BindingUnbound("original_exit_with_open_call_or_close_failure")
                    after = self.supervisor.stop(self.binding.original_association,
                                                 deadline=self.binding.session_deadline)
                    if not after.quiescent or not self.supervisor.commands.all_closed():
                        raise BindingUnbound("normal_original_unit_or_control_children_not_closed")
                    self.closed=True
                    return os.waitstatus_to_exitcode(self.child.raw_wait)
                now=time.monotonic()
                if self.active is not None and now>=self.active["useful"]:
                    self.active["primary"]="original_useful27_exhausted"
                    self._stop(self.active)
                    raise BindingUnbound("whole_original_call_UNKNOWN")
                end=self.binding.session_deadline if self.active is None else self.active["useful"]
                readable=select.select([self.sock],[],[],min(.01,max(0,end-now)))[0]
                if readable:
                    self._dispatch(recv_credentialed(self.sock,expected_pid=self.child.pid,expected_uid=os.getuid()))
            raise TimeoutError("original_session_cutoff")
        except BaseException as primary:
            self.primary=primary
            # Whole-unit custody remains necessary even when the direct child
            # already reaped or launch failed before self.child assignment.
            if self.native.children and not self.whole_stop_started:
                try:self._stop(self.active,child=self.child or self.native.children[0])
                except BaseException as secondary:self.secondary.append(secondary)
            # Whole stop is never repeated. Preserve every retained birth and
            # close/reap remaining direct children under the SAME original end.
            for child in self.native.children:
                if not child.closed and not child.stop_attempted:
                    try:self.native.stop(child,deadline=min(self.active["end"],self.binding.session_deadline) if self.active is not None else self.binding.session_deadline)
                    except BaseException as secondary:self.secondary.append(secondary)
            raise
        finally:
            for handle in (peer,self.sock):
                if handle is not None:
                    try:handle.close()
                    except BaseException as error:self.secondary.append(error);self.closed=False
            if self.inputfd is not None:
                fd=self.inputfd;self.inputfd=None
                try:os.close(fd)
                except BaseException as error:self.secondary.append(error);self.closed=False
            if self.secondary:
                self.closed=False
                failures = ([self.primary] if self.primary is not None else []) + self.secondary
                raise BaseExceptionGroup("outside_owner_primary_and_final_close_UNKNOWN", failures)


def _native_child():
    global _CHILD
    # NEW task-bound companion lookup after actual exec; input FD3/IPC FD4
    # are not adoption credentials. No extra inherited FD or birth-C change.
    from hermes_cli.friday_cli_fdcommit_authority import adopt_original_child
    adopt_original_child()
    binding=current_binding()
    seals=fcntl.fcntl(3,fcntl.F_GET_SEALS)
    need=fcntl.F_SEAL_WRITE|fcntl.F_SEAL_GROW|fcntl.F_SEAL_SHRINK|fcntl.F_SEAL_SEAL
    if seals&need!=need:raise BindingUnbound("original_input_not_kernel_sealed")
    body=os.read(3,65537)
    value=decode(body)
    if (type(value) is not dict or set(value)!={"schema","argv","owner_pid"}
            or value["schema"]!="friday.original.cli-input.v1"
            or type(value["argv"]) is not list or type(value["owner_pid"]) is not int):
        raise BindingUnbound("original_native_input_schema")
    os.close(3)
    sock=socket.socket(fileno=4)
    ipc=OriginalIPCClient(sock,binding,value["owner_pid"])
    _CHILD=ChildContext(binding,ipc)
    sys.argv=value["argv"]
    from hermes_cli.main import main
    main()


if __name__=="__main__":
    if sys.argv[1:]!=["--native-child"]:
        raise BindingUnbound("fixed_original_entrypoint_only")
    # -m executes this wrapper as __main__. Adoption and all existing
    # consumers use the one canonical module; never alias/replace its object.
    from hermes_cli.friday_cli_entrypoint import _native_child as adopt_original_child
    adopt_original_child()
