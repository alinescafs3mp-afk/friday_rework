"""Exact current native placement admission; never accepts serialized Go.

This package supplies topology/code, not a native acceptance authority. The
missing FD mutation history authority is explicit and cannot be minted here.
The future independent native binder must implement the pinned Authority ABI
from true kernel birth, including all mandatory resource/delegation checks.
"""
from dataclasses import dataclass, fields
import hashlib
import os
import json
import math
from pathlib import Path
import sys
import time
import types
from types import MappingProxyType

# Single immutable pair is the ONLY registry/current publication edge.
_CURRENT = (None, MappingProxyType({}))
_ADOPTION_ATTEMPTED = False
_ADOPTION_FAILURE = None


class BindingUnbound(RuntimeError):
    pass


class NativeAcceptanceAuthority:
    """Residual native qualification ABI, deliberately no Python substitute."""
    def validate_current(self, binding, phase):
        raise BindingUnbound("native_Source_Resource_Go_and_true_birth_FD_history_UNBOUND")

    def validate_retained_cleanup(self, binding, phase, *, native_boundary, child, deadline):
        # Separate native authority for exact previously admitted custody. It
        # must verify the current unit invocation, retained directory/pidfd
        # generations, all owned descendants and exclusion of foreign actors.
        # This ABI grants no new launch, lease, writer, admission or clock.
        raise BindingUnbound("native_retained_cleanup_custody_UNBOUND")


@dataclass(frozen=True, eq=False)
class CurrentQualifiedBinding:
    authority: object
    authority_type: type
    authority_code: object
    native_library: str
    native_library_sha256: str
    interpreter: str
    interpreter_sha256: str
    sources: tuple
    exclusive_cgroup_fd: int
    exclusive_cgroup: str
    original_association: dict
    session_deadline: float
    acceptance_epoch: object
    environment: tuple
    exclusive_cgroup_identity: tuple
    original_association_body: bytes
    loaded_functions: tuple
    authority_cleanup_code: object

    def _require_authority(self):
        current, registry = _CURRENT
        if (current is not self or self.authority_type not in registry
                or type(self.authority) is not self.authority_type
                or self.authority_type is NativeAcceptanceAuthority
                or not issubclass(self.authority_type, NativeAcceptanceAuthority)
                or self.authority_type.validate_current.__code__ is not self.authority_code
                or self.acceptance_epoch is None):
            raise BindingUnbound("native_authority_implementation_UNBOUND")
        qualification = registry[self.authority_type]
        self.authority._registered(self, qualification)
        return self

    def _require_metadata_shape(self):
        import re
        def path_hash(path, digest):
            return (type(path) is str and path.startswith("/") and "\x00" not in path
                    and len(path.encode()) <= 4096 and type(digest) is str
                    and re.fullmatch(r"[0-9a-f]{64}", digest) is not None)
        if (not path_hash(self.interpreter, self.interpreter_sha256)
                or not path_hash(self.native_library, self.native_library_sha256)
                or type(self.sources) is not tuple or not 0 < len(self.sources) <= 8192
                or any(type(row) is not tuple or len(row) != 2 or not path_hash(*row) for row in self.sources)
                or len({row[0] for row in self.sources}) != len(self.sources)
                or type(self.exclusive_cgroup_fd) is not int or self.exclusive_cgroup_fd < 0
                or type(self.exclusive_cgroup) is not str or not self.exclusive_cgroup.startswith("/")
                or "\x00" in self.exclusive_cgroup or len(self.exclusive_cgroup.encode()) > 4096
                or type(self.exclusive_cgroup_identity) is not tuple or len(self.exclusive_cgroup_identity) != 2
                or any(type(v) is not int or v < 0 for v in self.exclusive_cgroup_identity)
                or type(self.original_association) is not dict
                or type(self.original_association_body) is not bytes or len(self.original_association_body) > 65536
                or type(self.environment) is not tuple or len(self.environment) > 256
                or any(type(v) is not str or "\x00" in v or len(v.encode()) > 4096 for v in self.environment)
                or type(self.loaded_functions) is not tuple or not 0 < len(self.loaded_functions) <= 8192
                or any(type(row) is not tuple or len(row) != 2 or type(row[0]) is not types.FunctionType
                       or type(row[1]) is not types.CodeType or row[0].__code__ is not row[1]
                       for row in self.loaded_functions)
                or type(self.authority_code) is not types.CodeType
                or type(self.authority_cleanup_code) is not types.CodeType
                or type(self.session_deadline) not in (int, float)
                or not math.isfinite(self.session_deadline) or self.session_deadline <= 0):
            raise BindingUnbound("native_binding_full_metadata_shape")
        # Native validate_binding must additionally bind ALL values, recursive
        # association schema, native map, resource charge and original deadline.
        return self

    def _require_retained_source_and_namespace(self):
        self._require_metadata_shape()
        if type(self.session_deadline) not in (float, int) or not math.isfinite(self.session_deadline):
            raise BindingUnbound("session_original_clock_UNBOUND")
        if os.path.realpath(sys.executable) != self.interpreter:
            raise BindingUnbound("original_interpreter_changed")
        for path, expected in ((self.interpreter, self.interpreter_sha256),
                               (self.native_library, self.native_library_sha256), *self.sources):
            if hashlib.sha256(Path(path).read_bytes()).hexdigest() != expected:
                raise BindingUnbound("loaded_native_Source_bytes_changed")
        cgroup = os.fstat(self.exclusive_cgroup_fd)
        if (not self.exclusive_cgroup.startswith("/")
                or (cgroup.st_dev,cgroup.st_ino) != self.exclusive_cgroup_identity
                or json.dumps(self.original_association,sort_keys=True,separators=(",",":"),allow_nan=False).encode()
                   != self.original_association_body):
            raise BindingUnbound("exclusive_original_cgroup_UNBOUND")
        if not self.loaded_functions:
            raise BindingUnbound("current_loaded_code_closure_UNBOUND")
        for function, code in self.loaded_functions:
            if type(function) is not types.FunctionType or function.__code__ is not code:
                raise BindingUnbound("actual_loaded_original_function_changed")
        return self

    def require(self, phase):
        try:
            self._require_retained_source_and_namespace()
            if time.monotonic() >= self.session_deadline:
                raise BindingUnbound("original_session_admission_expired")
            self._require_authority()
            self.authority.validate_current(self, phase)
        except BaseException as error:
            from hermes_cli.friday_cli_fdcommit_authority import CommitHistoryAuthority
            if type(self.authority) is CommitHistoryAuthority and self.authority._failure is None:
                self.authority._failure = type(error).__name__
            raise
        return self

    def require_retained_cleanup(self, phase, *, native_boundary, child, deadline):
        self._require_authority()
        from hermes_cli.friday_cli_birth import OriginalBirthBoundary, OriginalNativeChild
        if (type(native_boundary) is not OriginalBirthBoundary
                or native_boundary.binding is not self
                or type(child) is not OriginalNativeChild or child.binding is not self
                or not any(item is child for item in native_boundary.children)
                or type(deadline) not in (int, float) or not math.isfinite(deadline)
                or not 0 < deadline <= self.session_deadline
                or self.authority_type.validate_retained_cleanup.__code__ is not self.authority_cleanup_code):
            raise BindingUnbound("foreign_or_widened_retained_cleanup")
        self.authority.validate_retained_cleanup(
            self, phase, native_boundary=native_boundary, child=child, deadline=deadline)
        self._require_retained_source_and_namespace()
        # Expiry forbids new useful work. It does not erase retained custody.
        # No later deadline, new authority or post-cutoff wait is supplied.
        return self


def current_binding():
    binding = _CURRENT[0]
    if type(binding) is not CurrentQualifiedBinding:
        raise BindingUnbound("current_original_CLI_placement_UNBOUND")
    return binding.require("current_constructor_before_effects")


def bind_cli_provider(cli):
    if os.name != "posix":
        return None
    # Ordinary construction must not import Linux/native dependencies. A real
    # native child has already adopted in this exact canonical module.
    module = sys.modules.get("hermes_cli.friday_cli_entrypoint")
    if module is None:
        return None
    child = module.current_child()
    if child is None:
        # Ordinary non-Friday CLI keeps its API; Friday result refuses later,
        # before parser/spinner/admission. No developer TUI is touched.
        return None
    child.binding.require("existing_CLI_constructor_before_model_or_frontend_effects")
    from hermes_cli.friday_cli_guardian import GuardianProvider
    provider = GuardianProvider(child, cli)
    cli._friday_original_caller_guard = provider
    return provider


def _begin_native_adoption(native, expected_role):
    global _ADOPTION_ATTEMPTED, _ADOPTION_FAILURE
    from hermes_cli.friday_cli_fdcommit_authority import require_fixed_native_bridge
    if _ADOPTION_ATTEMPTED or _CURRENT[0] is not None or _CURRENT[1]:
        raise BindingUnbound("original_native_adoption_attempt_not_repeated")
    _ADOPTION_ATTEMPTED = True  # one attempt; even failed preflight cannot retry
    try:
        if type(expected_role) is not str or expected_role not in ("owner", "child"):
            raise BindingUnbound("native_adoption_fixed_role")
        require_fixed_native_bridge(native)  # before EVERY native capability call
    except BaseException as error:
        _ADOPTION_FAILURE = type(error).__name__
        raise


def adopt_native_current(expected_role):
    global _ADOPTION_FAILURE
    # No import of an arbitrary extension executes before provenance preflight.
    native = sys.modules.get("_frw_fdcommit_native")
    _begin_native_adoption(native, expected_role)
    try:
        adoption = native.adopt_owner() if expected_role == "owner" else native.adopt_child()
        return _install_native_adoption_prevalidated(adoption, native, expected_role=expected_role)
    except BaseException as error:
        _ADOPTION_FAILURE = type(error).__name__
        raise


def install_native_adoption(adoption, native, *, expected_role):
    global _ADOPTION_FAILURE
    _begin_native_adoption(native, expected_role)
    try:
        return _install_native_adoption_prevalidated(adoption, native, expected_role=expected_role)
    except BaseException as error:
        _ADOPTION_FAILURE = type(error).__name__
        raise


def _install_native_adoption_prevalidated(adoption, native, *, expected_role):
    global _CURRENT
    from hermes_cli.friday_cli_fdcommit_authority import CommitHistoryAuthority, require_fixed_native_bridge
    require_fixed_native_bridge(native)
    if (not _ADOPTION_ATTEMPTED or _ADOPTION_FAILURE is not None or _CURRENT[0] is not None or _CURRENT[1]
            or expected_role not in ("owner", "child") or type(adoption) is not native.Adoption):
        raise BindingUnbound("native_adoption_original_attempt_or_role")
    native.validate_adoption(adoption)
    if native.role_of(adoption) != expected_role:
        raise BindingUnbound("native_adoption_role_mismatch")
    authority = CommitHistoryAuthority.from_native_adoption(adoption, native)
    native_fields = native.binding_fields(adoption)
    names = {f.name for f in fields(CurrentQualifiedBinding)}
    protected = {"authority", "authority_type", "authority_code", "authority_cleanup_code", "acceptance_epoch"}
    if type(native_fields) is not dict or set(native_fields) != names - protected:
        raise BindingUnbound("native_adoption_metadata_exact_schema")
    binding = CurrentQualifiedBinding(
        authority=authority, authority_type=CommitHistoryAuthority,
        authority_code=CommitHistoryAuthority.validate_current.__code__,
        authority_cleanup_code=CommitHistoryAuthority.validate_retained_cleanup.__code__,
        acceptance_epoch=authority._epoch, **native_fields)
    binding._require_metadata_shape()
    native.check_registration(adoption, authority._qualification, CommitHistoryAuthority,
        binding.authority_code, binding.authority_cleanup_code, CurrentQualifiedBinding)
    authority._registered(binding, authority._qualification)
    binding._require_retained_source_and_namespace()
    authority.validate_current(binding, "canonical_native_adoption_before_install")
    if time.monotonic() >= binding.session_deadline:
        raise BindingUnbound("native_adoption_original_deadline_expired")
    # Prepare ALL fallible objects/checks before the one atomic pointer store.
    publication = (binding, MappingProxyType({CommitHistoryAuthority: authority._qualification}))
    if _CURRENT[0] is not None or _CURRENT[1] or _ADOPTION_FAILURE is not None:
        raise BindingUnbound("native_adoption_concurrent_or_refused")
    _CURRENT = publication
    return binding  # no fallible post-publication validation/callback/call
