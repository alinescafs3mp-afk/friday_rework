/* Fixed-purpose whole-original CLI birth boundary, NEW UNQUALIFIED Source.
 * No binary built or loaded by Source author. Linux clone3(CLONE_INTO_CGROUP)
 * creates the original child in its exclusive admitted cgroup from kernel birth.
 * This DOES NOT observe every descriptor-table transition or qualify deadlines.
 */
#define _GNU_SOURCE
#include <errno.h>
#include <fcntl.h>
#include <limits.h>
#include <linux/sched.h>
#include <signal.h>
#include <stdint.h>
#include <stdlib.h>
#include <sys/syscall.h>
#include <sys/types.h>
#include <unistd.h>

struct friday_birth {
    int pid;
    int pidfd;
    int gate_write;
    int error_read;
    int parent_close_errno;
};
/* Effect boundary: caller must retain PREPID_UNKNOWN intent before entering.
 * exe is the exact pinned original interpreter; envp is the original admitted
 * environment. argv is fixed here; no command strings, broker or model.
 * Fixed FDs 3/4 carry sealed original kwargs and authenticated native IPC.
 */
int friday_cli_spawn(int cgroup_fd, int input_fd, int ipc_fd, const char *exe,
                     char *const envp[], struct friday_birth *out) {
    int gate[2] = {-1,-1}, error[2] = {-1,-1}, input_copy=-1, ipc_copy=-1;
    int pidfd=-1;
    if (!out || !exe || exe[0] != '/' || cgroup_fd < 0 || input_fd < 0 || ipc_fd < 0) {
        errno=EINVAL; return -1;
    }
    out->pid=-1; out->pidfd=-1; out->gate_write=-1; out->error_read=-1; out->parent_close_errno=0;
    /* Copies above reserved fd slots prevent dup2 alias clobber. */
    input_copy=fcntl(input_fd, F_DUPFD_CLOEXEC, 10);
    if (input_copy < 0) goto fail;
    ipc_copy=fcntl(ipc_fd, F_DUPFD_CLOEXEC, 10);
    if (ipc_copy < 0) goto fail;
    if (pipe2(gate,O_CLOEXEC) || pipe2(error,O_CLOEXEC|O_NONBLOCK)) goto fail;
    struct clone_args args = {0};
    args.flags=CLONE_PIDFD|CLONE_INTO_CGROUP;
    args.pidfd=(uint64_t)(uintptr_t)&pidfd;
    args.cgroup=(uint64_t)cgroup_fd;
    args.exit_signal=SIGCHLD;
    long pid=syscall(SYS_clone3,&args,sizeof(args));
    if (pid < 0) goto fail;
    if (!pid) {
        close(gate[1]); close(error[0]);
        char permit=0;
        ssize_t n;
        do {n=read(gate[0],&permit,1);} while(n<0 && errno==EINTR);
        if (n != 1 || permit != 'G') _exit(125);
        close(gate[0]);
        int failure=0;
        if (dup2(input_copy,3)<0 || dup2(ipc_copy,4)<0) failure=errno;
        /* The exec-error pipe stays CLOEXEC. Every inherited other fd is
         * closed before exec; fd mutation history remains separately UNBOUND. */
        if (!failure) {
            long maxfd=sysconf(_SC_OPEN_MAX);
            if (maxfd < 0 || maxfd > 1048576) maxfd=1048576;
            for (int fd=5;fd<maxfd;fd++) if(fd!=error[1]) close(fd);
            char *const argv[]={(char *)exe,"-B","-m","hermes_cli.friday_cli_entrypoint",
                                "--native-child",NULL};
            execve(exe,argv,envp);
            failure=errno;
        }
        (void)write(error[1],&failure,sizeof(failure));
        _exit(126);
    }
    out->pid=(int)pid; out->pidfd=pidfd;
    out->gate_write=gate[1]; out->error_read=error[0];
    int owned[4]={gate[0],error[1],input_copy,ipc_copy};
    for(int k=0;k<4;k++) {
        if(close(owned[k])<0 && !out->parent_close_errno) out->parent_close_errno=errno;
    }
    return 0;
fail:
    {
        int saved=errno;
        if(input_copy>=0) close(input_copy);
        if(ipc_copy>=0) close(ipc_copy);
        for(int k=0;k<2;k++){if(gate[k]>=0)close(gate[k]);if(error[k]>=0)close(error[k]);}
        errno=saved; return -1;
    }
}
