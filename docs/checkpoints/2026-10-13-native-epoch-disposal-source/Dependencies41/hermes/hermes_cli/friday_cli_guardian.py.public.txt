"""Concrete original guardian provider and all eleven retained-lease methods.

Cooperative objects remain in their original process/loop. The fixed outside
owner owns the whole original OS boundary and actual raw wait. A RPC reply is
never a durable commit, actual caller return, FD history or delivered credit.
"""
import asyncio
import copy
from dataclasses import dataclass
import hashlib
import json
import os
import sys
import threading
import time
import types

from hermes_cli.friday_cli_binding import BindingUnbound


@dataclass(eq=False)
class ForegroundReservation:
    lease: object
    context: object
    coroutine: object
    loop: object
    deadline: float
    task: object = None


@dataclass(frozen=True, eq=False)
class AcceptedDelivery:
    lease: object
    selection: object
    receipt_body: bytes
    commit_return: object
    final_return: object
    accepted_at: float


class GuardianProvider:
    def __init__(self, child, cli):
        from hermes_cli.friday_cli_entrypoint import current_child
        from hermes_cli.friday_cli_binding import CurrentQualifiedBinding
        if child is not current_child() or type(child.binding) is not CurrentQualifiedBinding:
            raise BindingUnbound("native_child_constructor_custody_UNBOUND")
        self.child=child
        self.cli=cli
        self.binding=child.binding.require("original_provider_constructor")
        self.leases={}
        self.poison=None

    def claim_original_caller(self, *, agent, call_id, task_id, thread, native_tid,
                              entry, deadline, frontend, session):
        self.binding.require("claim_original_before_parser_spinner_admission")
        if (self.poison is not None or self.cli.agent is not agent
                or frontend.cli is not self.cli or frontend.cap is not session
                or thread is not threading.current_thread() or native_tid!=threading.get_native_id()
                or call_id in self.leases or not entry <= time.monotonic() < deadline <= entry+30
                or deadline>self.binding.session_deadline):
            raise BindingUnbound("original_guardian_foreign_or_duplicate_claim")
        frontend.check_lifetime();frontend.anchor.validate()
        response=self.child.ipc.call("CLAIM",{"call_id":call_id,"task_id":task_id,
            "entry":entry,"deadline":deadline,"native_tid":native_tid,
            "authority_generation":session.session_authority["generation"]},deadline=deadline)
        lease=OriginalGuardianLease(self,agent,call_id,task_id,thread,native_tid,frontend,session,
                                    entry,deadline,response["lease"])
        self.leases[call_id]=lease
        return lease


class OriginalGuardianLease:
    def __init__(self, provider, agent, call_id, task_id, thread, native_tid, owner, session, entry, end, native_id):
        self.provider=provider
        self.agent=agent;self.call_id=call_id;self.task_id=task_id
        self.thread=thread;self.native_tid=native_tid
        self.owner=owner;self.session=session
        self.entry=entry;self.end=end;self.native_id=native_id
        self.phase="CLAIMED"
        self.span=None
        self.activities=[];self.callbacks=[];self.tasks=[];self.candidates=[]
        self.primary=None;self.secondary=[];self.stop_started=False

    def _same(self, span, deadline, *, cleanup=False):
        self.provider.binding.require("retained_original_guardian")
        if (self.provider.leases.get(self.call_id) is not self
                or span.guard is not self or span.agent is not self.agent
                or span.call_id!=self.call_id or span.task_id!=self.task_id
                or span.thread is not self.thread or span.thread_id!=self.native_tid
                or span.owner is not self.owner or span.entry!=self.entry
                or not self.entry<deadline<=span.end<=self.end
                or time.monotonic()>=deadline or self.phase in {"CLOSED","STOP_REQUESTED","UNKNOWN"}):
            raise BindingUnbound("original_guardian_lifetime_or_cutoff_changed")
        if self.span is not None and self.span is not span:
            raise BindingUnbound("foreign_original_span")
        self.span=span
        self.owner.check_lifetime();self.owner.anchor.validate()
        if self.owner.record_poison is not None and not cleanup:
            raise BindingUnbound("original_frontend_partial_prefix_poisoned")

    def _call(self, op, body, deadline):
        return self.provider.child.ipc.call(op,body,lease=self.native_id,deadline=deadline)["body"]

    def check_original_caller(self, span, *, phase, deadline):
        self._same(span,deadline,cleanup=deadline>span.useful)
        self._call("CHECK",{"phase":phase,"deadline":deadline},deadline)

    def narrow_original_caller(self, span, *, deadline):
        self._same(span,deadline)
        self._call("NARROW",{"deadline":deadline},deadline)
        self.end=deadline

    def own_activity_thread(self, span, *, thread, stop):
        self._same(span,span.useful)
        if (type(thread) is not threading.Thread or type(stop) is not threading.Event
                or thread.ident is not None or not any(r["thread"] is thread and r["stop"] is stop for r in span.activities)
                or any(t is thread for t,s in self.activities)):
            raise BindingUnbound("foreign_or_started_original_activity")
        self.activities.append((thread,stop))
        self._call("ACTIVITY",{"ordinal":len(self.activities)},span.useful)

    def _own_callback(self, span, kind, record, app, loop, deadline):
        self._same(span,deadline,cleanup=deadline>span.useful)
        prompt=self.owner.prompt
        if (prompt is None or prompt.app is not app or prompt.loop is not loop
                or app.loop is not loop or app._loop_thread is not prompt.loop_thread
                or not any(c.record is record and c.app is app and c.loop is loop for c in span.custody.callbacks)
                or any(r is record for k,r in self.callbacks)):
            raise BindingUnbound("foreign_original_callback_or_loop")
        self.callbacks.append((kind,record))
        self._call(kind,{"ordinal":len(self.callbacks),"deadline":deadline},deadline)

    def own_native_proxy_tail(self, span, *, record, proxy, app, native_loop, deadline):
        if self.owner.frontend.proxy is not proxy or self.owner.prompt.proxy is not proxy:
            raise BindingUnbound("foreign_original_proxy")
        self._own_callback(span,"PROXY",record,app,native_loop,deadline)

    def own_native_repaint_barrier(self, span, *, binding, record, app, loop, deadline):
        if self.owner.prompt is not binding:
            raise BindingUnbound("foreign_original_repaint_binding")
        self._own_callback(span,"REPAINT",record,app,loop,deadline)

    def own_native_prompt_phase(self, span, *, operation, phase, app, native_loop, context, deadline):
        prompt=self.owner.prompt
        if (prompt is None or operation.binding is not prompt or prompt.app_context is not context
                or not any(c.record.get("operation") is operation and c.record.get("phase")==phase
                           for c in span.custody.callbacks if type(c.record) is dict)):
            raise BindingUnbound("foreign_original_prompt_operation")
        record=next(c.record for c in span.custody.callbacks
                    if type(c.record) is dict and c.record.get("operation") is operation and c.record.get("phase")==phase)
        self._own_callback(span,"PROMPT",record,app,native_loop,deadline)

    def reserve_original_foreground_task(self, span, *, context, coroutine, loop, deadline):
        self._same(span,deadline)
        manager=context._manager
        if (manager._cli_ref is not self.provider.cli or context.plugin_id!="friday_rework"
                or not asyncio.iscoroutine(coroutine) or loop is not asyncio.get_running_loop()
                or not loop.is_running() or loop.is_closed()
                or any(r.coroutine is coroutine for r in self.tasks)):
            raise BindingUnbound("foreign_original_foreground_reservation")
        item=ForegroundReservation(self,context,coroutine,loop,deadline)
        self.tasks.append(item)
        self._call("TASK_RESERVE",{"ordinal":len(self.tasks),"deadline":deadline},deadline)
        return item

    def attach_original_foreground_task(self, span, *, custody, task):
        self._same(span,custody.deadline)
        if (type(custody) is not ForegroundReservation or custody.lease is not self or custody not in self.tasks
                or custody.task is not None or not isinstance(task,asyncio.Task)
                or task.get_loop() is not custody.loop or task.get_coro() is not custody.coroutine
                or not any(t.task is task and t.context is custody.context for t in span.custody.tasks)):
            raise BindingUnbound("foreign_original_foreground_attachment")
        custody.task=task
        self._call("TASK_ATTACH",{"ordinal":self.tasks.index(custody)+1},custody.deadline)

    def commit_original_receipts(self, span, *, proof, operations, deadline, native_tail, frontend):
        self._same(span,deadline,cleanup=True)
        if (proof.custody.span is not span or proof.custody.witness is not proof
                or proof.custody.state!="WITNESS_CONSUMED" or frontend is not self.owner.frontend
                or tuple(span.phases)!=native_tail or self.phase!="CLAIMED"):
            raise BindingUnbound("original_actual_direct_return_not_consumed")
        span.custody.wait_natural_settlement(deadline)
        self.phase="COMMITTING"
        for function,args in operations:
            module=sys.modules.get(getattr(function,"__module__",""))
            if (type(function) is not types.FunctionType or module is None
                    or getattr(module,"finish_delivery",None) is not function
                    or function.__name__!="finish_delivery" or type(args) is not tuple or len(args)!=7):
                raise BindingUnbound("foreign_original_receipt_operation")
            store,task_id,owner,reference,number,receipt,selection=args
            if (selection is not span.receipt_selections.get((id(store),task_id,reference,number))
                    or selection.span is not span or selection.store is not store):
                raise BindingUnbound("foreign_original_exact_receipt_selection")
            function(*args)  # actual original store/compound lock/finally return
            selection.observe_finish_boundary_return()
            self.candidates.append(selection)
        self._call("RECEIPTS",{"candidates":[
            {"selection":s.sha256,"commit":s.commit_return.commit.content_sha256,
             "generation":s.commit_return.commit.generation} for s in self.candidates]},deadline)
        self.phase="CANDIDATES_DURABLE"

    def finish_original_caller(self, span, *, proof, outcome, phases, deadline):
        from hermes_cli.friday_cli_return import require_final_boundary_return
        final=require_final_boundary_return(span,proof)
        self._same(span,deadline,cleanup=True)
        if outcome!="COMPLETED" or self.phase!="CANDIDATES_DURABLE" or tuple(span.phases)!=phases:
            raise BindingUnbound("original_final_lifecycle_not_ready")
        span.custody.wait_natural_settlement(deadline)
        if self.owner.anchor.readers or self.owner.anchor.close_errors or self.owner.frontend.pending or self.owner.frontend.errors:
            raise BindingUnbound("original_temporary_resources_not_closed")
        if any(r.task is None or not r.task.done() or r.task.cancelled() or r.task.exception() is not None for r in self.tasks):
            raise BindingUnbound("original_foreground_not_actually_settled")
        body={"returned_at":final.returned_at,"deadline":deadline,"candidates":[
            {"selection":s.sha256,"commit":s.commit_return.commit.content_sha256,
             "generation":s.commit_return.commit.generation} for s in self.candidates]}
        # Server retains FINAL_ACK_PENDING, never consumer DELIVERED. Actual
        # response return + local resource/lifecycle acceptance is a later edge.
        self._call("FINISH",body,deadline)
        self._same(span,deadline,cleanup=True)
        accepted_at=time.monotonic()
        for selection in self.candidates:
            selection.validate_accept(self,final,accepted_at)
        for selection in self.candidates:
            selection.accept(self,final,accepted_at)
        self.phase="CLOSED"
        return final

    def stop_original_caller(self, span, *, deadline, activity, frontend):
        if self.stop_started:
            raise BindingUnbound("duplicate_original_native_stop")
        self.stop_started=True
        self.primary=self.primary or span.primary_error or (span.errors[0] if span.errors else "original_unknown")
        self.phase="STOP_REQUESTED"
        try:
            # The client requests; only the outside owner can kill/reap this
            # original unit. Client cannot claim its own post-stop return.
            self._call("STOP",{"deadline":deadline,"primary":str(self.primary)},deadline)
        except BaseException as error:
            self.secondary.append(error)
        self.provider.poison=self.primary
        self.phase="UNKNOWN"
        raise BindingUnbound("STOP_UNCONFIRMED:client_cannot_self_reap")
