"""Finite real Linux workload fixtures, never a Codex/daemon substitute.
No admission is issued here. Root must first bind/review these exact bytes and
launch only through the existing owner/native_spawn barrier and stock guards.
"""

MEMORY = 2147483648
TASKS = 128
TMP = 134217728
INODES = 512
OUTPUT = 33554432

# Explicit native stacks, one malloc arena, and conservative virtual headroom
# precede EVERY thread stress birth. Actual mappings/controller refusal still
# require independent Root observations; these bounds are never run credit.
THREAD_STACK_BYTES=262144
THREAD_GUARD_RESERVE=65536
THREAD_RUNTIME_RESERVE=1048576
THREAD_BASE_MAX=67108864
THREAD_SHARED_RESERVE=16777216
THREAD_MAX_NEW=126

def bounded_thread_stacks():
    assert len(list(Path('/proc/self/task').iterdir()))==1, 'configure before any stress thread'
    assert resource.getrlimit(resource.RLIMIT_AS) in ((268435456,268435456),(1073741824,1073741824))
    page=os.sysconf('SC_PAGE_SIZE')
    assert type(page)is int and 0<page<=THREAD_GUARD_RESERVE and THREAD_STACK_BYTES%page==0
    libc=ctypes.CDLL(None,use_errno=True)
    # GNU libc M_ARENA_MAX=-8 is an explicitly supported fixture precondition.
    # Unsupported allocator is concrete fixture refusal, never pids credit.
    version=libc.gnu_get_libc_version;version.restype=ctypes.c_char_p
    assert version(), 'exact supported GNU libc allocator required'
    mallopt=libc.mallopt;mallopt.argtypes=[ctypes.c_int,ctypes.c_int];mallopt.restype=ctypes.c_int
    assert mallopt(-8,1)==1, 'single malloc arena before thread birth'
    threading.stack_size(THREAD_STACK_BYTES)
    assert threading.stack_size()==THREAD_STACK_BYTES, 'explicit bounded pthread stack accepted'
    fd=os.open('/proc/self/statm',os.O_RDONLY|os.O_NOFOLLOW)
    try:raw=os.read(fd,257);assert len(raw)<=256
    finally:os.close(fd)
    base=int(raw.split()[0])*page
    reserved=THREAD_BASE_MAX+THREAD_MAX_NEW*(THREAD_STACK_BYTES+THREAD_GUARD_RESERVE+THREAD_RUNTIME_RESERVE)+THREAD_SHARED_RESERVE
    assert 0<base<=THREAD_BASE_MAX and reserved<=268435456, 'all128 tasks fit conservative AS256 reservation'
    return {'stack_bytes':THREAD_STACK_BYTES,'allocator_max_arenas':1,'baseline_virtual_bytes':base,
            'guard_reserve_each':THREAD_GUARD_RESERVE,'runtime_reserve_each':THREAD_RUNTIME_RESERVE,
            'maximum_new_threads':THREAD_MAX_NEW,'conservative_virtual_upper':reserved,
            'actual_maps_and_kernel_refusal_require_Root':True}



def bounded_environment(group, temp):
    group, temp = Path(group), Path(temp)
    assert os.getpid() > 1
    # Caller PID1 and this child MUST share the independently admitted namespace.
    assert os.stat('/proc/self/ns/pid').st_ino == os.stat('/proc/1/ns/pid').st_ino
    assert group.is_absolute() and temp.is_absolute() and '..' not in group.parts + temp.parts
    actual = Path('/proc/self/cgroup').read_text().strip()
    assert actual == '0::/' + str(group.relative_to('/sys/fs/cgroup'))
    for name, expected in [('memory.max', str(MEMORY)), ('memory.swap.max', '0'),
                           ('pids.max', str(TASKS)), ('cpu.max', '200000 100000')]:
        assert (group/name).read_text().strip() == expected
    assert resource.getrlimit(resource.RLIMIT_AS) == (1073741824,1073741824)
    assert resource.getrlimit(resource.RLIMIT_CPU) == (60,60)
    assert resource.getrlimit(resource.RLIMIT_FSIZE) == (67108864,67108864)
    assert resource.getrlimit(resource.RLIMIT_CORE) == (0,0)
    assert 0 < len(os.sched_getaffinity(0)) <= 2
    rows=[s for s in Path('/proc/self/mountinfo').read_text().splitlines() if s.split()[4]==str(temp)]
    assert len(rows)==1 and ' - tmpfs ' in rows[0]
    v=os.statvfs(temp)
    assert v.f_blocks*v.f_frsize <= TMP and v.f_files <= INODES
    return group,temp


def emit(value):
    raw=(json.dumps(value,sort_keys=True,separators=(',',':'))+'\n').encode()
    assert len(raw)<=8192
    pos=0
    while pos<len(raw):
        n=os.write(1,raw[pos:]);assert n>0;pos+=n


def counters(group, name):
    return {k:int(v) for k,v in (line.split() for line in (group/name).read_text().splitlines())}


def birth():
    v=Path('/proc/self/stat').read_text().rsplit(')',1)[1].split()
    return {'pid':os.getpid(),'birth':v[19],'ppid':int(v[1]),'sid':int(v[3]),
            'namespace':os.stat('/proc/self/ns/pid').st_ino,
            'group':Path('/proc/self/cgroup').read_text().strip()}


def positive_eof(group,temp):
    # Distinct exact streams including non-UTF8 bytes; no early-success marker.
    raw=bytes(range(256))*1024
    for fd in [1,2]:
        for pos in range(0,len(raw),8192):
            chunk=raw[pos:pos+8192];off=0
            while off<len(chunk):n=os.write(fd,chunk[off:]);assert n>0;off+=n
    return 0


def output_exhaust(group,temp):
    # Cross the EXISTING whole32MiB budget, not a smaller fixture threshold.
    for i in range(OUTPUT//8192+1):
        fd=1 if i%2==0 else 2;raw=b'C40_OUTPUT_'+b'x'*(8192-11)
        assert len(raw)==8192
        off=0
        while off<len(raw):n=os.write(fd,raw[off:]);assert n>0;off+=n
    raise AssertionError('existing whole-output collector did not refuse')


def strict_buffer_exhaust(group,temp):
    # Real oversized stdio, NOT an initialized Codex app-server. This only
    # exercises its collector when separately bound to the strict fixture case.
    raw=b'x'*8192
    for _ in range(8388608//len(raw)+1):
        off=0
        while off<len(raw):n=os.write(1,raw[off:]);assert n>0;off+=n
    return 0


def pids_threads(group,temp):
    stack_plan=bounded_thread_stacks()
    baseline=int((group/'pids.current').read_text());assert 2<=baseline<=128
    assert TASKS-baseline<=THREAD_MAX_NEW
    before=counters(group,'pids.events');release=threading.Event();threads=[];error=None
    try:
        for _ in range(TASKS+1):
            t=threading.Thread(target=release.wait,args=(12,),daemon=False)
            try:t.start()
            except (OSError,RuntimeError) as exc:
                error={'type':type(exc).__name__,'errno':getattr(exc,'errno',None),'detail':str(exc)[:120]};break
            threads.append(t)
        after=counters(group,'pids.events')
        emit({'fixture':'pids_threads','before':before,'after':after,'threads_started':len(threads),
              'pids_current':int((group/'pids.current').read_text()),'error':error,'identity':birth(),
              'stack_plan':stack_plan,'baseline_tasks':baseline})
        assert len(threads)==TASKS-baseline and int((group/'pids.current').read_text())==TASKS
        assert after.get('max',0)>before.get('max',0), 'no actual kernel pids.max refusal'
        assert error is not None, '129 threads unexpectedly admitted under whole128tasks'
    finally:
        release.set()
        for t in threads:t.join(timeout=1)
        assert all(not t.is_alive() for t in threads)
    return 0


def memory_oom(group,temp):
    # Four real 600MiB resident allocations stay below each AS1GiB but exceed
    # the exact inherited2GiB total. Kernel memory.events/oom_kill and original
    # outside wait must be collected by Root even if this producer is OOM-killed.
    children=[];emit({'fixture':'memory_oom','before':counters(group,'memory.events'),'identity':birth()})
    for _ in range(3):
        pid=os.fork()
        if pid==0:
            children=[]
            break
        children.append(pid)
    raw=bytearray(600*1024*1024)
    for p in range(0,len(raw),4096):raw[p]=1
    select.select([],[],[],8)
    del raw
    if not children:os._exit(0)
    waits=[]
    for pid in children:
        got,status=os.waitpid(pid,0);assert got==pid;waits.append([pid,status])
    emit({'fixture':'memory_oom','after':counters(group,'memory.events'),'waits':waits})
    return 0


def cpu_quota(group,temp):
    before=counters(group,'cpu.stat');children=[]
    for _ in range(3):
        pid=os.fork()
        if pid==0:
            children=[]
            break
        children.append(pid)
    end=time.monotonic()+10;count=0
    while time.monotonic()<end:count=(count*1664525+1013904223)&0xffffffff
    if not children:os._exit(0)
    for pid in children:got,_=os.waitpid(pid,0);assert got==pid
    after=counters(group,'cpu.stat')
    emit({'fixture':'cpu_quota','before':before,'after':after,'identity':birth()})
    assert after.get('nr_throttled',0)>before.get('nr_throttled',0), 'actual CPU throttling not observed'
    return 0


def detached_adopted(group,temp):
    first=os.fork()
    if first==0:
        os.setsid();second=os.fork()
        if second:os._exit(0)
        emit({'fixture':'detached_adopted','identity':birth()})
        # Intentionally outlives direct producer, yet finite and same kernel
        # boundary. Existing PID-init MUST observe/register/wait this descendant.
        select.select([],[],[],3);os._exit(0)
    got,status=os.waitpid(first,0);assert got==first
    emit({'fixture':'detached_adopted','direct_wait':[got,status]})
    return 0


def tmp_bytes(group,temp):
    root=temp/'c40-tmp-bytes';root.mkdir(mode=0o700);failure=None;total=0
    for i in range(3):
        fd=os.open(root/str(i),os.O_WRONLY|os.O_CREAT|os.O_EXCL,0o400)
        try:
            for _ in range(45):
                raw=b't'*1048576;pos=0
                while pos<len(raw):n=os.write(fd,raw[pos:]);assert n>0;pos+=n;total+=n
        except OSError as exc:failure=exc.errno;break
        finally:os.close(fd)
    emit({'fixture':'tmp_bytes','written':total,'errno':failure,'statvfs':os.statvfs(temp).f_bavail})
    assert failure==errno.ENOSPC and total<=TMP,'actual tmpfs whole-cap cause absent'
    return 0


def tmp_inodes(group,temp):
    root=temp/'c40-tmp-inodes';root.mkdir(mode=0o700);failure=None;made=0
    for i in range(INODES+1):
        try:fd=os.open(root/str(i),os.O_WRONLY|os.O_CREAT|os.O_EXCL,0o400)
        except OSError as exc:failure=exc.errno;break
        os.close(fd);made+=1
    emit({'fixture':'tmp_inodes','files':made,'errno':failure,'remaining':os.statvfs(temp).f_favail})
    assert failure==errno.ENOSPC and made<=INODES,'actual inode exhaustion absent'
    return 0

MODES={'positive_eof':positive_eof,'output_exhaust':output_exhaust,
