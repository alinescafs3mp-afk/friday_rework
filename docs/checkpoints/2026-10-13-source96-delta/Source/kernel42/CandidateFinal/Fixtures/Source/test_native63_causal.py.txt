"""Finite original63 causal test assertions, AST-only Source delivery.
Root supplies a SAME-READ actual outside collector artifact through an ALREADY
retained FD and independently pinned input. These tests issue no grant, launch,
receipt, kill, config, queue, model or capacity-release operation. No CLI or
new acceptance service. Missing actual evidence fails, never fills zeros.
"""
import hashlib
import json
import os
from pathlib import Path
import stat


def _actual(fd,pin,binding,case_id):
    assert binding['case_id']==case_id
    for key in ['Root_native_admission','independent_source_acceptance','final_source_pins',
                'fixture_source_pins','effective_kernel_admission','original_native_clock',
                'existing_Root_outside_observations']:
        assert binding[key] is not None,'unbound '+key
    before=os.fstat(fd)
    assert stat.S_ISREG(before.st_mode) and before.st_nlink==1 and before.st_size==pin['bytes']<=2097152
    raw=os.pread(fd,before.st_size+1,0);after=os.fstat(fd)
    key=lambda q:(q.st_dev,q.st_ino,q.st_mode,q.st_size,q.st_mtime_ns,q.st_ctime_ns)
    assert len(raw)==before.st_size and key(before)==key(after)==key(Path(pin['path']).lstat())
    assert hashlib.sha256(raw).hexdigest()==pin['sha256']
    actual=json.loads(raw)
    assert actual['case_id']==case_id and actual['origin']=='ROOT_ACTUAL_EXISTING_STOCK_COLLECTOR'
    assert actual['simulated'] is False and actual['Source_JSON_is_terminal'] is False
    assert actual['final_source_pins']==binding['final_source_pins']
    assert actual['fixture_source_pins']==binding['fixture_source_pins']
    assert actual['original_native_clock']==binding['original_native_clock']
    assert actual['clock_reset'] is False and actual['replayed'] is False
    assert actual['Root_actual_terminal_ref']==binding['existing_Root_outside_observations']
    assert type(actual['cause'])is dict and actual['cause']['operation_observed'] is True
    assert actual['cause']['producer_observed'] is True and actual['cause']['value_observed'] is True
    assert actual['cause']['Source_fault_is_kernel_fault'] is False
    assert actual['inside_capacity_release'] is False
    return actual


def _before_effect(a,phase):
    v=a['before_effect']
    assert v['actual_original_clocks_read'] is True and v['unknown_budget_as_zero'] is False
    assert v['whole_role_limits_complete'] is True or a['cause']['key']=='resource_missing'
    assert v['kernel_reservation_before_first_opaque_birth'] is True or a['actual_whole_births']==0
    assert a['bootstrap_births_observed_separately_from_genuine_commands'] is True
    assert v['exclusive_existing_heavy_slot_actual_fd'] is True
    assert v['same_update_then_senderinstall_order'] is True
    assert v['no_protected_writer_or_shared_scope_change'] is True
    assert v['Source21_credit_as_native'] is False
    if phase=='before_imports':assert a['updater_imports']==0
    if phase in ('before_imports','before_birth'):
        assert a['genuine_command_GO']==0 and a['genuine_command_births']==0
        assert a['unknown_bootstrap_births_as_zero'] is False
    if phase=='before_GO':assert a['opaque_GO']==0 and a['retained_actual_child_pidfd'] is True
    if phase in ('before_imports','before_birth','before_GO'):assert a['genuine_receipt_writes']==0

def _kernel_subcases(a,modes):
    assert set(modes)<=set(a['actual_linux_subcases'])
    for mode in modes:
        s=a['actual_linux_subcases'][mode]
        assert s['actual_payload_pin_verified'] is True and s['existing_owner_spawn_barrier'] is True
        assert s['real_kernel_effect'] is True and s['simulated'] is False
        assert s['effective_limits']=={'memory.max':'2147483648','memory.swap.max':'0','pids.max':'128','cpu.max':'200000 100000'}
        assert s['no_threshold_reduction'] is True
        assert s['actual_direct_wait_and_fresh_absence'] is True
        assert s['actual_adopted_wait_and_fresh_absence'] is True
        if mode=='memory_oom':assert s['after']['oom_kill']>s['before']['oom_kill']
        elif mode=='pids_threads':
            assert s['after']['max']>s['before']['max']
            assert s['actual_threads_in_total_tasks'] is True and s['spawn_failure_observed'] is True
        elif mode=='cpu_quota':assert s['after']['nr_throttled']>s['before']['nr_throttled']
        elif mode in ('tmp_bytes','tmp_inodes'):assert s['actual_errno']==28
        elif mode=='output_exhaust':
            assert s['actual_raw_output_bytes']>33554432 and s['existing_whole32MiB_refusal'] is True
            assert s['truncated_output_as_success'] is False
        elif mode=='positive_eof':
            assert s['actual_raw_output_bytes']==524288 and s['actual_both_streams_EOF'] is True
            assert s['actual_raw_stream_hashes_match'] is True
        elif mode=='strict_buffer_exhaust':
            assert s['actual_raw_buffer_bytes']>8388608 and s['actual_existing_strict_refusal'] is True
            assert s['fixture_as_genuine_strict_credit'] is False
        elif mode=='schema_bytes':assert s['actual_input_bytes']>8388608 and s['actual_admission_refusal'] is True
        elif mode=='clone_bytes':assert s['actual_input_bytes']>1048576 and s['actual_admission_refusal'] is True
        elif mode=='detached_adopted':
            assert s['actual_detach_and_reparent_observed'] is True
            assert s['all_observed_descendants_same_kernel_scope_and_namespace'] is True
            assert s['stop_CLI_zero_as_closure'] is False
