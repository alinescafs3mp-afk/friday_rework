"""Corrective C45 finite consistency controls Source; NOT_RUN/NOT_ADMITTED.
No positive full32 plan is manufactured. Synthetic bindings below exist only
to reach negative branches and are explicitly never authority.
"""
import copy

def _refuses(fn,code,*args):
    try:fn(*args)
    except ValueError as error:assert error.code==code
    else:raise AssertionError('expected concrete refusal')

def test_c45_missing_actual_root_provider(adapter):
    _refuses(adapter.verify_existing_root_facility,'MISSING_EXISTING_ROOT_FACILITY',{}, {}, {},0,0,'boot')

def test_c45_wrong_original_source_budget(adapter,valid_clock):
    for seconds in (1800,3600,True,120.0):
        clock=copy.deepcopy(valid_clock);clock['original_seconds']=seconds
        _refuses(adapter.original_root_clock,'ORIGINAL_CLOCK',clock,clock['boot_id'],clock['started_at_monotonic']+20,clock['started_at_unix']+20)

def test_c45_dualclock_drift(adapter,valid_clock):
    c=valid_clock
    _refuses(adapter.original_root_clock,'ORIGINAL_CLOCK',c,c['boot_id'],c['started_at_monotonic']+20,c['started_at_unix']+23)

def test_c45_renewed_final(adapter,valid_clock):
    c=copy.deepcopy(valid_clock);c['final_deadline_unix']+=20
    _refuses(adapter.original_root_clock,'ORIGINAL_CLOCK',c,c['boot_id'],c['started_at_monotonic']+20,c['started_at_unix']+20)

def test_c45_nonfinite_bool_clock(adapter,valid_clock):
    for value in (True,float('inf'),float('nan')):
        c=copy.deepcopy(valid_clock);c['started_at_unix']=value
        _refuses(adapter.original_root_clock,'ORIGINAL_CLOCK',c,c['boot_id'],20,20)

def test_c45_complete_original_clock_positive(adapter,valid_clock):
    c=valid_clock;r=adapter.original_root_clock(c,c['boot_id'],c['started_at_monotonic']+20,c['started_at_unix']+20)
    assert r['actor_stop_m']==c['started_at_monotonic']+100 and r['stock_stop_m']==c['started_at_monotonic']+110
    assert r['final_u']==c['final_deadline_unix']

def test_c45_wait_after_firstbirth(adapter,valid_spec,valid_expected,valid_clock):
    e=copy.deepcopy(valid_expected);p=e['actual_wait_observation'];p['wait_entered_m']=p['first_birth_m']
    _refuses(adapter.verify_existing_root_facility,'WAIT_ARM_ORDER',valid_spec,e,valid_clock,p['first_birth_m']+1,p['first_birth_u']+1,valid_clock['boot_id'])

def test_c45_candidate_json_is_not_wait_proof(adapter,valid_spec,valid_expected,valid_clock):
    e=copy.deepcopy(valid_expected);p=e['actual_wait_observation'];p['Root_independently_verified_before_firstbirth']=False
    _refuses(adapter.verify_existing_root_facility,'WAIT_PROVIDER_NOT_VERIFIED',valid_spec,e,valid_clock,p['first_birth_m']+1,p['first_birth_u']+1,valid_clock['boot_id'])

def memory_clock():
    return {'boot_id':'SYNTHETIC_MEMORY_ONLY','original_seconds':120,'cleanup_seconds':30,
            'started_at_monotonic':1000,'started_at_unix':2000,
            'useful_deadline_unix':2090,'final_deadline_unix':2120}

def memory_wait_inputs():
    token='SYNTHETIC_MEMORY_ONLY_NOT_AUTHORITY';clock_pin='SYNTHETIC_MEMORY_CLOCK_NOT_AUTHORITY'
    spec={'domain':'kernel120_qualification','clock':clock_pin,'native':{'action_id':'SYNTHETIC_MEMORY_ONLY'}}
    expected={k:token for k in ('outside_wait_provider_source','before_firstbirth_absolute_wait_ref',
        'direct_and_adopted_wait_provider_source','final_helper_wait_fresh_release_source',
        'independent_current_source_acceptance','actual_specific_action_admission','original_final_helper_guard_binding_ref')}
    expected['original_clock_pin']=clock_pin
    expected['actual_wait_observation']={'action_id':'SYNTHETIC_MEMORY_ONLY','original_clock_pin':clock_pin,
        'actual_provider_source':token,'Root_independently_verified_before_firstbirth':True,
        'wait_entered_m':1019,'wait_entered_u':2019,'first_birth_m':1020,'first_birth_u':2020,
        'actor_stop_m':1100,'actor_stop_u':2100,'stock_stop_m':1110,'stock_stop_u':2110,
        'final_m':1120,'final_u':2120,'retained_owned_stock_actor_pidfds_before_firstbirth':True,
        'already_admitted_dedicated_context_subreaper':True,'protected_or_TUI_context':False}
    return spec,expected,memory_clock()

def memory_conflict_plan(batch):
    """Current API pure-memory fixture, never admission or physical evidence."""
    pin={'path':'/SYNTHETIC','bytes':0,'sha256':'0'*64};clock=memory_clock()
    planned={case_id:{'case_id':case_id,'kind':kind,'cause':cause,
             'original_clock_ref':pin,'actual_entry':None,
             'planned_age_window':[19,21] if kind=='source_uncertainty_after_actual_operation' else [0,120]}
             for case_id,kind,name,cause in batch.CASES}
    catalog=batch.Original32Catalog(clock,pin,pin,pin,planned,pin)
    p=batch.make_original32_plan(catalog)
    absolute={'start_m':1000,'start_u':2000,'useful_m':1090,'useful_u':2090,'final_m':1120,'final_u':2120}
    return p,pin,absolute

def memory_negative_binding(batch,plan,pin,absolute):
    return {'original_catalog_object':plan['catalog_object'],
            'independent_current_source_acceptance':'SYNTHETIC_NOT_AUTHORITY',
            'actual_specific_action_admission':'SYNTHETIC_NOT_AUTHORITY'}

def test_c45_original_cut_window_not_common_batch_assumption(batch,p,pin,absolute,expected):
    catalog=p['catalog_object'];saved=dict(catalog.actual)
    try:
        for case_id,kind,name,cause in batch.CASES:
            age=20 if kind=='source_uncertainty_after_actual_operation' else 1
            catalog.actual[case_id]={'original_clock_ref':pin,'Root_expected_original_clock_ref':pin,
              'clock_start_m':1000,'clock_start_u':2000,'entry_m':1000+age,'entry_u':2000+age,
              'original_final_m':1120,'original_final_u':2120,'clock_reset':False,
              'accepted_original_clock_elapsed_at_entry':age}
        cut=next(r[0] for r in batch.CASES if r[1]=='source_uncertainty_after_actual_operation')
        catalog.actual[cut].update(entry_m=1022,entry_u=2022,accepted_original_clock_elapsed_at_entry=22)
        _refuses(batch.original_case_clock_bindings,'BATCH_OWNER_CUT_WINDOW',expected,pin)
    finally:catalog.actual=saved

    finally:catalog.phase_intervals=saved

def test_c45_trace_driver_full32_not_partial(batch,owner,tests,expected,trace_refs,clock_pin):
