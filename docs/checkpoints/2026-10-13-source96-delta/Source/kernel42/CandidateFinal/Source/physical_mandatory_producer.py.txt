_SEAL=object()
class ActualMandatoryScope:
    __slots__=('seal','scope','producer_pin','source_pins','clock_identity','raw_ref','output','credit')
    def __init__(self,seal,scope,producer_pin,registration,raw_ref,output,credit):
        physical.need(seal is _SEAL,'CURRENT_MANDATORY_SCOPE_CONSTRUCTOR_FORBIDDEN')
        self.seal=seal;self.scope=scope;self.producer_pin=producer_pin
        self.source_pins=registration['complete_current_source_pins']
        self.clock_identity=registration['original_clock_identity']
        self.raw_ref=raw_ref;self.output=output;self.credit=credit

def _admitted(scope,registration,live_external_grant):
    from physical_admission_qualification import qualify_external_live_grant
    qualification=qualify_external_live_grant(live_external_grant,registration,('validate_exact_mandatory_scope',))
    live_external_grant.validate_exact_mandatory_scope(scope,registration)
    clock=registration['original_clock_identity']
    physical.need(clock['original_seconds']==120 and clock['cleanup_seconds']==30,
                  'SAME_ORIGINAL_MANDATORY_120_INCLUDING30')
    for pin in registration['complete_current_source_pins']:physical.same_read_source(pin)
    physical.hard_caps()
    physical.need(time.monotonic()<clock['started_at_monotonic']+90 and
        time.time()<clock['started_at_unix']+90,'CURRENT_MANDATORY_ORIGINAL_USEFUL_NOT_RENEWED')
    return qualification

def run_current_h991_fd8(registration,live_external_grant,modules):
    _admitted('H991_FD8_MEMORY_ONLY',registration,live_external_grant)
    pins=registration['mandatory_producer_pins'];inputs=registration['current_memory_inputs']
    h991=_qualified_test(modules['memory_harness'],pins['memory_harness'],'qualify')
    fd8=_qualified_test(modules['root_fd_memory_adapter'],pins['root_fd_memory_adapter'],'run_fd_memory_controls')
    hblobs={name:physical.same_read_source(pin) for name,pin in inputs['h991_source_pins'].items()}
    fblobs={name:physical.same_read_source(pin) for name,pin in inputs['fd_source_pins'].items()}
    # Consistency records are new current bindings supplied after external live
    # admission. Historical ROOT-GO/PASS records are never copied or read here.
    h=h991(hblobs,inputs['h991_binding'],inputs['h991_go'],
        physical.same_read_source(pins['memory_harness']),physical.same_read_source(inputs['testdata']),
        physical.same_read_source(inputs['resources']),ast)
    f=fd8(fblobs,inputs['fd_context'],ast,types,json,hashlib,stat)
    physical.need(h['credit']=='BOUNDED_SYNTHETIC_EXACT_BODY_ONLY_NO_LIVE_KERNEL_NATIVE_ACCEPTANCE',
                  'H991_ORIGINAL_MEMORY_SCOPE_NOT_LIVE')
    physical.need(h['full_sequence']['operations']==991 and
        h['full_sequence']['births']==128 and h['eighteen_segments']['segments']==18,
                  'FULL_H991_991_OPERATIONS128_BIRTHS18_SEGMENTS')
    physical.need(f['observations']['controls']==8 and f['observations']['close_ambiguity_variants']==2 and
        f['observations']['actual_kernel_calls']==0,'ALL_FD8_TWO_AMBIGUITY_CONTROLS_MEMORY_ONLY')
    _admitted('H991_FD8_MEMORY_ONLY',registration,live_external_grant)
    output={'H991':h,'FD8':f,'source_pins':registration['complete_current_source_pins'],
        'clock_identity':registration['original_clock_identity'],'runtime_acceptance':False}
    ref=_exclusive_output(registration['raw_output_parent_capability'],registration['h991_fd8_component'],output)
    return (ActualMandatoryScope(_SEAL,'H991',pins['memory_harness'],registration,ref,h,'MEMORY_ONLY'),
            ActualMandatoryScope(_SEAL,'FD8',pins['root_fd_memory_adapter'],registration,ref,f,'MEMORY_ONLY'))

def validate_mandatory_scope(observation,registration):
    physical.need(type(observation)is ActualMandatoryScope and observation.seal is _SEAL,
                  'ACTUAL_CURRENT_MANDATORY_PRODUCER_RETURN_REQUIRED')
    physical.need(observation.source_pins==registration['complete_current_source_pins'] and
        observation.clock_identity==registration['original_clock_identity'] and observation.raw_ref is not None,
                  'NO_HISTORICAL_SOURCE_CLOCK_OR_PASS_TRANSFER')
    return observation
