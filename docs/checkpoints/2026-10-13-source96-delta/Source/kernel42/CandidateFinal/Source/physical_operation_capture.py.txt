import ctypes

import physical_observation_primitives as physical

PEEKUSER=3
POKEUSER=6
SETREGS=13
DEBUG_BASE=848

def mallopt_symbol(raw):
    physical.need(raw[:6]==b"\x7fELF\x02\x01" and len(raw)>=64,"PINNED_LIBC_ELF64")
    header=struct.unpack_from("<16sHHIQQQIHHHHHH",raw)
    physical.need(header[1:3]==(3,62) and header[11]==64 and 0<header[12]<=8192,"LIBC_ELF64_X86_SECTION_TABLE")
    sections=[struct.unpack_from("<IIQQQQIIQQ",raw,header[6]+64*i) for i in range(header[12])]
    values=[]
    for section in sections:
        if section[1]!=11:continue
        physical.need(section[9]==24 and section[5]%24==0 and section[6]<len(sections),"LIBC_DYNSYM_LAYOUT")
        strings=sections[section[6]];table=raw[strings[4]:strings[4]+strings[5]]
        for offset in range(section[4],section[4]+section[5],24):
            name,info,other,index,value,size=struct.unpack_from("<IBBHQQ",raw,offset)
            end=table.find(b"\0",name)
            if table[name:end]==b"mallopt":
                physical.need(info&15==2 and index and value and size,"MALLOPT_REAL_DEFINED_FUNCTION")
                values.append((value,size))
    physical.need(len(values)==1,"ONE_REAL_LIBC_MALLOPT_SYMBOL")
    physical.need(header[9]==56 and 0<header[10]<=256,"LIBC_PROGRAM_HEADERS")
    loads=[struct.unpack_from("<IIQQQQQQ",raw,header[5]+56*i) for i in range(header[10])]
    value,size=values[0];physical.need(0<size<=1048576,'BOUNDED_REAL_MALLOPT_FUNCTION_BODY')
    executable=[p for p in loads if p[0]==1 and p[1]&1 and p[3]<=value and value+size<=p[3]+p[5]]
    physical.need(len(executable)==1,'ONE_PINNED_MALLOPT_EXECUTABLE_FILE_RANGE')
    segment=executable[0];offset=segment[2]+value-segment[3]
    body=raw[offset:offset+size];physical.need(len(body)==size,'COMPLETE_PINNED_MALLOPT_BODY')
    return value,tuple(p for p in loads if p[0]==1),body

class ActualMalloptCapture:
    def __init__(self,observer):
        self.observer=observer;self.active={};self.records=[]
    def _debug(self,pid,index,value=None):
        address=DEBUG_BASE+8*index
        if value is None:return self.observer.ptrace(PEEKUSER,pid,address)&((1<<64)-1)
        return self.observer.ptrace(POKEUSER,pid,address,value)
    def _mapped(self,pid,pin):
        raw=physical.same_read_source(pin);q=Path(pin['path']).lstat()
        text=(Path('/proc')/str(pid)/'maps').read_text();maps=[]
        for line in text.splitlines():
            row=line.split(None,5)
            if len(row)!=6:continue
            lo,hi=(int(x,16) for x in row[0].split('-'));major,minor=(int(x,16) for x in row[3].split(':'))
            if (os.makedev(major,minor),int(row[4]))==(q.st_dev,q.st_ino):
                physical.need(row[5]==pin['path'],'EXACT_MAPPED_ELF_PATH')
                maps.append((lo,hi,row[1],int(row[2],16)))
        physical.need(maps,'ACTUAL_PINNED_ELF_MAPPING_UNBOUND')
        return raw,maps,text,(q.st_dev,q.st_ino)
    def arm(self,pid,operation):
        o=self.observer
        physical.need(pid not in self.active,'NO_SECOND_ALLOCATOR_INTERVAL')
        import physical_admission_qualification as admission
        admission.qualify_external_live_grant(o.external_grant,o.registration,('validate_exact_function_capture',))
        o.external_grant.validate_exact_function_capture('libc.mallopt(-8,1)',pid,o.registration)
        pin=o.registration['allocator_libc_pin'];raw,maps,text,identity=self._mapped(pid,pin)
        symbol,loads,body=mallopt_symbol(raw);page=os.sysconf('SC_PAGE_SIZE');biases=set()
        for lo,hi,perms,offset in maps:
            for load in loads:
                if offset==load[2]//page*page:biases.add(lo-load[3]//page*page)
        physical.need(len(biases)==1,'ONE_ACTUAL_LIBC_LOAD_BIAS')
        address=biases.pop()+symbol
        physical.need(any(lo<=address<hi and 'x' in perms for lo,hi,perms,offset in maps),'MALLOPT_ACTUAL_EXECUTABLE_MAPPING')
        physical.need(o._bytes(pid,address,len(body))==body,'ACTUAL_MAPPED_MALLOPT_COMPLETE_BODY_BYTES')
        saved={i:self._debug(pid,i) for i in (0,1,2,3,6,7)}
        physical.need(saved[7]==0,'NO_PREEXISTING_DEBUG_REGISTER_OWNER')
        state={'pid':pid,'birth':o.subjects[pid][0],'operation':operation,'entry_address':address,
               'saved':saved,'libc_pin':pin,'mapped_identity':identity,'maps_raw':text,'body_sha256':hashlib.sha256(body).hexdigest(),'body_bytes':body,'entry':None,'return':None}
        self.active[pid]=state # retain before first fallible register effect.
        self._debug(pid,0,address);self._debug(pid,6,0);self._debug(pid,7,1)
    def trap(self,pid,status):
        if pid not in self.active:return False
        state=self.active[pid];dr6=self._debug(pid,6)
        if not dr6&3:return False
        physical.need(dr6&3 in (1,2),'UNEXPECTED_DEBUG_BREAKPOINT_COMBINATION')
        regs=self.observer._regs(pid)
        physical.need(self.observer._bytes(pid,state['entry_address'],len(state['body_bytes']))==state['body_bytes'],
                      'MALLOPT_MAPPED_BODY_CHANGED_DURING_ACTUAL_CALL')
        if dr6&1:
            physical.need(state['entry'] is None and regs.rip==state['entry_address'],'REAL_MALLOPT_ENTRY_NO_REPLAY')
            physical.need(ctypes.c_int(regs.rdi).value==-8 and ctypes.c_int(regs.rsi).value==1,'ACTUAL_MALLOPT_ARGUMENTS')
            returned_to=int.from_bytes(self.observer._bytes(pid,regs.rsp,8,8),'little')
            qualified=[]
            for pin in self.observer.registration['allocator_return_elf_pins']:
                raw,maps,text,identity=self._mapped(pid,pin)
                if any(lo<=returned_to<hi and 'x' in perms for lo,hi,perms,offset in maps):qualified.append((pin,identity,text))
            physical.need(len(qualified)==1,'ACTUAL_LIBFFI_RETURN_MAPPING_UNBOUND')
            state['entry']={'kernel_sequence':len(self.observer.kernel),'raw_regs':tuple(getattr(regs,n) for n in self.observer.register_names),
               'at_monotonic':time.monotonic(),'at_unix':time.time(),'wait_status':status,'return_address':returned_to,'return_mapping':qualified[0]}
            self._debug(pid,0,0);self._debug(pid,1,returned_to);self._debug(pid,7,4)
        else:
            physical.need(state['entry'] is not None and state['return'] is None and
                regs.rip==state['entry']['return_address'],'ACTUAL_MALLOPT_RETURN_PAIR')
            state['return']={'kernel_sequence':len(self.observer.kernel),'raw_regs':tuple(getattr(regs,n) for n in self.observer.register_names),
                             'at_monotonic':time.monotonic(),'at_unix':time.time(),'wait_status':status,'result':ctypes.c_int(regs.rax).value}
            self._debug(pid,7,0)
        self._debug(pid,6,0);regs.eflags|=1<<16
        self.observer.ptrace(SETREGS,pid,0,ctypes.byref(regs))
        return True
    def restore(self,pid,require_return=False):
        if pid not in self.active:return
        state=self.active[pid]
        physical.need(not require_return or state['entry'] is not None and state['return'] is not None,
                      'LIBC_ENTRY_OR_RETURN_NOT_ACTUALLY_CAPTURED')
        self._debug(pid,7,0)
        for index,value in state['saved'].items():
            if index not in (4,5,7):self._debug(pid,index,value)
        self._debug(pid,7,state['saved'][7]);self.records.append(state);del self.active[pid]
