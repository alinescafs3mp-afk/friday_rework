def _qualified_test(module,pin,function_name):
    raw=physical.same_read_source(pin)
    physical.need(type(module)is types.ModuleType and module.__file__==pin['path'],
                  'CURRENT_EXACT_ASSERTION_MODULE_OBJECT')
    tree=ast.parse(raw,filename=pin['path'])
    code=compile(tree,pin['path'],'exec',dont_inherit=True,optimize=0)
    canonical={c.co_name:c for c in code.co_consts if type(c)is types.CodeType}
    function=module.__dict__.get(function_name)
    physical.need(type(function)is types.FunctionType and function.__globals__ is module.__dict__
        and function_name in canonical and
        marshal.dumps(function.__code__)==marshal.dumps(canonical[function_name]),
        'CURRENT_EXACT_ASSERTION_BODY_OBJECT')
    return function


def _exclusive_output(parent,name,document):
    row=physical.fd_read(physical.process_read(os.getpid()),parent['fd'])
    physical.need(row.identity[:2]==tuple(parent['identity']) and
        row.policy()=={'uid':os.getuid(),'gid':os.getgid(),'mode':0o700} and
        row.link==parent['path'],'CURRENT_RESULT_REAL_PRIVATE_PARENT')
    physical.need(name and '/' not in name and name not in ('.','..'),'CURRENT_RESULT_COMPONENT')
    raw=json.dumps(document,sort_keys=True,separators=(',',':'),allow_nan=False).encode()
    physical.need(len(raw)<=67108864,'CURRENT_RESULT_WRITE_RESERVATION')
    fd=os.open(name,os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o400,dir_fd=parent['fd'])
    try:
        physical.need(os.fstat(fd).st_nlink==1,'CURRENT_RESULT_EXCLUSIVE_NLINK')
        view=memoryview(raw)
        while view:
            wrote=os.write(fd,view);physical.need(wrote>0,'CURRENT_RESULT_WRITE_PROGRESS');view=view[wrote:]
        os.fsync(fd)
    finally:os.close(fd)
    os.fsync(parent['fd'])
    return {'path':str(Path(parent['path'])/name),'bytes':len(raw),'sha256':hashlib.sha256(raw).hexdigest()}


def run_current_full_assertions(actual_runs,qualified_test_modules,registration,mandatory_scopes=()):
    """Run all exact114 consumer functions after genuine capture, preserving reds.
    Current H991/FD8/C45 are connected through physical_mandatory_producer.
    Missing C59_28 producer/dependency remains an explicit uncompleted gate.
    This cannot be used as whole-final runtime admission or physical-fit credit.
    """
    expected={r[0] for r in CASES}|set(CONTROLS)|{r[0] for r in C83}
    physical.need(set(actual_runs)==expected and len(expected)==114,'CURRENT_EXACT114_RUN_SCOPE')
    for observed in actual_runs.values():capture.validate_current_capture(observed)
    from physical_mandatory_producer import validate_mandatory_scope
    for scoped in mandatory_scopes:validate_mandatory_scope(scoped,registration)
    coverage={'schema':'C100.actual_causal_capture_coverage.v1',
        'ids':sorted(expected),'raw_runs':{key:r.raw_ref for key,r in actual_runs.items()},
        'complete_current_source_pins':registration['complete_current_source_pins'],
        'original_clock_identity':registration['original_clock_identity'],
        'all114_actual_captures_present':True,'all_mandatory_acceptance':False,
        'additional_mandatory_scopes':{r.scope:r.raw_ref for r in mandatory_scopes},
        'additional_current_missing_scopes':sorted({'H991','FD8','C45_MEMORY18','C45_ACTUAL32_CONSUMER','C59_28'}-{r.scope for r in mandatory_scopes}),
        'native69_plus_reserved2_and_full67_547_cli150_installed':'UNBOUND_NOT_RUN'}
    ref=_exclusive_output(registration['raw_output_parent_capability'],
        registration['coverage_component'],coverage)
    whole=reducers.CurrentMandatoryObservation(actual_runs,ref,
        registration['complete_current_source_pins'],registration['original_clock_identity'],mandatory_scopes)
    for observed in actual_runs.values():
        for member in capture.validate_current_capture(observed):
            member.registration['current_whole_mandatory_observation']=whole
    jobs=[]
    for case_id,kind,name,cause in CASES:
        jobs.append((case_id,'original32',name,lambda r=actual_runs[case_id],c=case_id:legacy.Original32Observation(r,c)))
    for control_id,definition in CONTROLS.items():
        suite=control_id[:3]
        jobs.append((control_id,suite,definition['function'],
            lambda r=actual_runs[control_id]:legacy.project_genuine_current_observation(r)))
    for control_id,kind,refusal,mutation in C83:
        jobs.append((control_id,'C83',reducers.C83_ASSERTIONS[control_id],
            lambda r=actual_runs[control_id]:reducers.project_c83(r)))
    outcomes=[]
    for control_id,suite,function_name,project in jobs:
        function=_qualified_test(qualified_test_modules[suite],registration['test_source_pins'][suite],function_name)
        try:function(project())
        except physical.ObservationMissing as error:
            state='UNKNOWN';reason=str(error)
        except BaseException as error:
            state='FAIL';reason=type(error).__name__+':'+str(error)
        else:state='PASS';reason=None
        outcomes.append({'id':control_id,'function':function_name,'state':state,'reason':reason})
    document={'schema':'C100.actual114.assertion_outcomes.v1','coverage':ref,'outcomes':outcomes,
        'Source_author_is_not_independent_acceptance':True,'runtime_acceptance':False,
        'capacity_release':False,'physical2sfit':'UNPROVEN','additional_mandatory_gates':'NOT_RUN',
        'all_original32_10_18_4_definitions_retained':True}
    result=_exclusive_output(registration['raw_output_parent_capability'],registration['outcomes_component'],document)
    return {'raw_result':result,'outcomes':outcomes,'runtime_acceptance':False,'capacity_release':False}

def run_current_connected_diagnostic(actual_runs,qualified_test_modules,registration,live_external_grant,modules):
    from physical_mandatory_producer import run_current_h991_fd8,run_current_c45,run_current_c45_actual32_consumer,refuse_uncompleted_current_c59_28
    scopes=[];missing=[]
    for name,produce in (('H991_FD8',run_current_h991_fd8),('C45_MEMORY18',run_current_c45),('C45_ACTUAL32_CONSUMER',run_current_c45_actual32_consumer)):
        try:
            observed=produce(registration,live_external_grant,modules)
            scopes.extend(observed if type(observed)is tuple else (observed,))
        except physical.ObservationMissing as error:missing.append({'scope':name,'state':'UNBOUND_NOT_RUN','reason':str(error)})
        except BaseException as error:missing.append({'scope':name,'state':'FAIL','reason':type(error).__name__+':'+str(error)})
    try:refuse_uncompleted_current_c59_28(registration,live_external_grant)
    except physical.ObservationMissing as error:missing.append({'scope':'C59_28','state':'UNBOUND_NOT_RUN','reason':str(error)})
    result=run_current_full_assertions(actual_runs,qualified_test_modules,registration,tuple(scopes))
    result['mandatory_scope_failures']=missing;result['whole_mandatory_acceptance']=False
    return result
