"""Qualify an already-live external gate; never issue a live authorization.

The two existing gate methods are compared to compiled code from the exact
independently accepted external Source. Compilation creates code objects only;
no external Source is executed. The actual gate still performs admission.
This file is inert SOURCE; no qualification or runtime occurred in C100.
"""
import ast
import inspect
import marshal
import os
import sys
import types
import hashlib
import physical_observation_primitives as physical

_METHODS=('validate_exact_observer_invocation','validate_exact_control')

def _nested_code(parent,name):
    found=[c for c in parent.co_consts if isinstance(c,types.CodeType) and c.co_name==name]
    physical.need(len(found)==1,'EXTERNAL_GATE_CODE_AMBIGUOUS:'+name)
    return found[0]

def qualify_external_live_grant(grant,registration,additional_methods=()):
    physical.need(type(additional_methods)is tuple and len(additional_methods)==len(set(additional_methods)) and
        set(additional_methods)<={'validate_exact_function_capture','validate_exact_mandatory_scope'},
        'EXACT_ADDITIONAL_REAL_GATE_METHODS')
    physical.need(grant is not None and not isinstance(grant,
        (dict,str,bytes,bool,int,float,list,tuple,type)),
        'LIVE_EXTERNAL_ROOT_ADMISSION_CAPABILITY_UNBOUND')
    pin=registration['external_admission_gate_source_pin']
    raw=physical.same_read_source(pin)
    klass=type(grant)
    module=sys.modules.get(klass.__module__)
    physical.need(type(module)is types.ModuleType and module.__dict__.get('__file__')==pin['path'],
                  'EXTERNAL_GATE_ACTUAL_MODULE_OBJECT')
    # Nested/local, fabricated module classes and subclasses are inadmissible.
    physical.need(klass.__qualname__==klass.__name__ and
        module.__dict__.get(klass.__name__) is klass,
        'EXTERNAL_GATE_EXACT_TOP_LEVEL_CLASS_OBJECT')
    tree=ast.parse(raw,filename=pin['path'])
    classes=[n for n in tree.body if isinstance(n,ast.ClassDef) and n.name==klass.__name__]
    physical.need(len(classes)==1,'EXTERNAL_GATE_EXACT_CLASS_SOURCE_BODY')
    body=classes[0]
    # Dynamic decorators/metaclasses can replace method objects at load time.
    # They need a separate explicit reviewed contract, never co_filename credit.
    physical.need(not body.decorator_list and not body.keywords,
                  'EXTERNAL_GATE_DYNAMIC_CLASS_NOT_QUALIFIED')
    compiled=compile(tree,pin['path'],'exec',dont_inherit=True,optimize=0)
    canonical_class=_nested_code(compiled,klass.__name__)
    identities=[]
    for name in _METHODS+additional_methods:
        nodes=[n for n in body.body if isinstance(n,ast.FunctionDef) and n.name==name]
        physical.need(len(nodes)==1 and not nodes[0].decorator_list,
                      'EXTERNAL_GATE_EXACT_METHOD_BODY:'+name)
        actual=inspect.getattr_static(klass,name)
        physical.need(type(actual)is types.FunctionType and
            inspect.getattr_static(grant,name) is actual and
            actual.__globals__ is module.__dict__ and
            actual.__closure__ is None,
            'EXTERNAL_GATE_ACTUAL_METHOD_OBJECT:'+name)
        canonical=_nested_code(canonical_class,name)
        physical.need(marshal.dumps(actual.__code__)==marshal.dumps(canonical),
                      'EXTERNAL_GATE_EXACT_COMPILED_BODY:'+name)
        physical.need(not nodes[0].args.defaults and not any(nodes[0].args.kw_defaults),
                      'EXTERNAL_GATE_MUTABLE_DEFAULTS_NOT_QUALIFIED')
        method=getattr(grant,name)
        physical.need(type(method)is types.MethodType and method.__self__ is grant
            and method.__func__ is actual,'EXTERNAL_GATE_REAL_BOUND_METHOD:'+name)
        identities.append((name,hashlib.sha256(marshal.dumps(canonical)).hexdigest()))
    # This is the ORIGINAL real authority call, with a real bound object.
    # Return values and mere artifact presence are never authorization substitutes.
    grant.validate_exact_observer_invocation(registration)
    return {'module':module,'class':klass,'methods':tuple(identities),'pin':dict(pin),
            'grant':grant,'observer_pid':os.getpid()}
