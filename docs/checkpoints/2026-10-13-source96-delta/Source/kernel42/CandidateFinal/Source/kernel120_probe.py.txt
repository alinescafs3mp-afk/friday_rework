def bounded_thread_stacks():
    assert len(list(Path('/proc/self/task').iterdir()))==1, 'configure before any stress thread'
    assert resource.getrlimit(resource.RLIMIT_AS) in ((268435456,268435456),(1073741824,1073741824))
    page=os.sysconf('SC_PAGE_SIZE')
    assert type(page)is int and 0<page<=THREAD_GUARD_RESERVE and THREAD_STACK_BYTES%page==0
    libc=ctypes.CDLL(None,use_errno=True)
    # GNU libc M_ARENA_MAX=-8 is an explicitly supported fixture precondition.
    # Unsupported allocator is concrete fixture refusal, never pids credit.
    version=libc.gnu_get_libc_version;version.restype=ctypes.c_char_p
    assert version(), 'exact supported GNU libc allocator required'
    mallopt=libc.mallopt;mallopt.argtypes=[ctypes.c_int,ctypes.c_int];mallopt.restype=ctypes.c_int
    require(_physical is not None,'INDEPENDENT_LIBC_ENTRY_RETURN_OBSERVER_UNBOUND')
    _physical.mark('operation_begin',3)
    returned=mallopt(-8,1)
    _physical.mark('operation_end',3)
    assert returned==1, 'single malloc arena before thread birth'
    # returned is never published as independent evidence; hardware entry/return
    # capture reads RDI/RSI/RAX from the executing pinned libc frame.
    threading.stack_size(THREAD_STACK_BYTES)

    fd=os.open('/proc/self/statm',os.O_RDONLY|os.O_NOFOLLOW)

    assert 0<base<=THREAD_BASE_MAX and reserved<=268435456, 'all128 tasks fit conservative AS256 reservation'
    return {'stack_bytes':THREAD_STACK_BYTES,'allocator_max_arenas':1,'baseline_virtual_bytes':base,
            'guard_reserve_each':THREAD_GUARD_RESERVE,'runtime_reserve_each':THREAD_RUNTIME_RESERVE,
            'maximum_new_threads':THREAD_MAX_NEW,'conservative_virtual_upper':reserved,
            'actual_maps_and_kernel_refusal_require_Root':True}

def read_small(path,limit=65536):
    fd=os.open(path,os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK)
    try:
        b=os.read(fd,limit+1);require(len(b)<=limit,'finite read');return b
    finally:os.close(fd)


def pin(ref):
    require(type(ref) is dict and set(ref)=={'path','bytes','sha256'},'exact probe input pin')
    p=Path(ref['path']);require(p.is_absolute() and p.resolve(strict=True)==p,'probe exact pin path')
    fd=os.open(p,os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK)
    try:
        a=os.fstat(fd);require(stat.S_ISREG(a.st_mode) and a.st_nlink==1 and a.st_size==ref['bytes']
            and 0<ref['bytes']<=2097152,'probe finite regular pin')
        b=bytearray()
        while len(b)<=ref['bytes']:
            x=os.read(fd,min(65536,ref['bytes']+1-len(b)))
            if not x:break
            b.extend(x)
        z=os.fstat(fd)
        require((a.st_dev,a.st_ino,a.st_size,a.st_mtime_ns,a.st_ctime_ns)
                ==(z.st_dev,z.st_ino,z.st_size,z.st_mtime_ns,z.st_ctime_ns)
                and len(b)==ref['bytes'] and hashlib.sha256(b).hexdigest()==ref['sha256'],
                'probe same-read captured bytes')
        return bytes(b)
    finally:os.close(fd)


def birth(pid):

    return {'pid':pid,'birth':s[19],'state':s[0],'ppid':int(s[1]),'pgid':int(s[2]),'sid':int(s[3])}


def absent(row):

def snapshot(spec,label,pid=None,tid=None):

            'actual inherited reservation from birth')

    require(controllers==CONTROLLERS and [q.st_dev,q.st_ino]==spec['native']['first_birth_group_identity'],
            'same actual inherited controller inode/values')
    tasks=int(read_small(group/'pids.current',4096));require(0<tasks<=128,'actual Linux tasks including threads')
    result={'label':label,'identity':row,'tid':tid,'controllers':controllers,'group_identity':[q.st_dev,q.st_ino],
        'kernel_pids_current':tasks,'pidns_inode':os.stat('/proc/'+str(pid)+'/ns/pid').st_ino,
        'userns_inode':os.stat('/proc/'+str(pid)+'/ns/user').st_ino,
        'mountns_inode':os.stat('/proc/'+str(pid)+'/ns/mnt').st_ino,
        'at_unix':time.time(),'at_monotonic':time.monotonic()}
    # Z has no live syscall/wchan/remote-FD evidence. Existence != actual wait.
    if row['state']=='Z':result['zombie_no_live_probe_or_reap_credit']=True

def deny_open(path):
    try:fd=os.open(path,os.O_WRONLY|os.O_NOFOLLOW|os.O_NONBLOCK)
    except OSError as e:
        require(e.errno in (errno.EACCES,errno.EROFS),'intended actual kernel write denial')
        return {'path':str(path),'actual_errno':e.errno,'write_count':0}
    else:
        os.close(fd);raise RuntimeError('write-open unexpectedly allowed; NO write/retry')


def real_thread_and_child(spec,context):
    rows=[];errors=[];gate=threading.Event()
    def target():
        try:
            rows.append(snapshot(spec,context+'_thread',tid=threading.get_native_id()))
            rows.append({'denial':deny_open(Path(spec['native']['group'])/'memory.max')})
        except BaseException as e:errors.append(type(e).__name__)
        finally:gate.set()
    thread=threading.Thread(target=target);thread.start()
    require(gate.wait(5),'bounded actual thread sample');thread.join(timeout=5)
    require(not thread.is_alive() and not errors and len(rows)==2,'actual thread joined')
    r,w=os.pipe();pid=os.fork()
    if pid==0:
        try:
            os.close(r);data={'snapshot':snapshot(spec,context+'_child'),
                'denial':deny_open(Path(spec['native']['group'])/'memory.max')}
            raw=json.dumps(data).encode();require(len(raw)<=8192,'bounded real child output')
            require(os.write(w,raw)==len(raw),'actual child pipe write');os.close(w);os._exit(0)
        except BaseException:os._exit(125)
    retained={'pid':pid,'pidfd':None,'pipe_r':r,'pipe_w':w,'waited':False}
    # The SAME finite probe retains all handles before fallible evidence work.
    retained['pidfd']=os.pidfd_open(pid);fdid=os.fstat(retained['pidfd']);os.close(w);retained['pipe_w']=None
    require(select.select([r,retained['pidfd']],[],[],5)[0],'actual child ready/terminal')
    raw=os.read(r,8193);require(0<len(raw)<=8192,'actual bounded child bytes');child=json.loads(raw)
    waited,status=os.waitpid(pid,0);require(waited==pid and status==0,'actual direct child wait')
    require(absent(child['snapshot']['identity']),'fresh exact child birth absent AFTER wait')
    q=os.fstat(retained['pidfd']);require((q.st_dev,q.st_ino)==(fdid.st_dev,fdid.st_ino),'same retained child pidfd')
    os.close(r);os.close(retained['pidfd']);retained['waited']=True
    return {'thread':rows,'child':child,'direct_wait_status':status,'fresh_exact_child_absent':True,
            'thread_joined':True,'real_fork_and_thread':True}

def quota_chunk(fd,chunk,written,call_budget=4096):
    # Positive partial writes are ordinary POSIX progress. Only a REAL ENOSPC
    # is quota cause evidence. Zero/invalid writes remain unknown fixture error.
    require(type(chunk)is bytes and 0<len(chunk)<=1048576,'bounded quota request')
    require(type(call_budget)is int and 0<call_budget<=4096,'remaining whole quota syscall reservation BEFORE write')
    pos=0;short=0;calls=0
    while pos<len(chunk):
        require(written+len(chunk)-pos<=MAX_WRITE,'finite aggregate lifetime reservation')
        calls+=1;require(calls<=call_budget,'finite remaining whole partial-write syscall reservation')
        try:n=os.write(fd,chunk[pos:])
        except OSError as e:
            require(e.errno==errno.ENOSPC,'actual bounded tmpfs byte exhaustion')
            return written,short,calls,e.errno
        require(type(n)is int and 0<n<=len(chunk)-pos,'positive finite partial-write progress')
        if n<len(chunk)-pos:short+=1
        pos+=n;written+=n
    return written,short,calls,None


def output_occupancy(spec):
    root=Path(spec['record']);files=[];written=0;short=0;calls=0;bytes_errno=None;inode_errno=None;chunk=b'Q'*1048576
    seed_bytes=0;v=os.statvfs(root)
    require(0<v.f_frsize<=65536 and 1048576%v.f_frsize==0,'supported finite tmpfs block layout')
    try:
        # Ensure this mandatory positive explicitly exercises a normal short
        # write even when initial free bytes were exactly MiB-aligned.
        if (v.f_bavail*v.f_frsize)%len(chunk)==0:
            p=root/'bytes-partial-seed';fd=os.open(p,os.O_WRONLY|os.O_CREAT|os.O_EXCL,0o400);files.append(p)
            try:
                written,s,c,e=quota_chunk(fd,b'S'*v.f_frsize,written)
                require(e is None,'actual bounded seed before exhaustion');short+=s;calls+=c;seed_bytes=v.f_frsize
            finally:os.close(fd)
        headroom=os.statvfs(root).f_bavail*v.f_frsize
        require(0<headroom<=67108864 and headroom%len(chunk)!=0,'actual unaligned free-byte fixture')
        for i in range(19):
            p=root/('bytes-'+str(i));fd=os.open(p,os.O_WRONLY|os.O_CREAT|os.O_EXCL,0o400);files.append(p)
            try:
                for _ in range(4):
                    written,s,c,bytes_errno=quota_chunk(fd,chunk,written,4096-calls);short+=s;calls+=c
                    require(calls<=4096,'whole bounded quota syscall inventory')
                    if bytes_errno is not None:break
            finally:os.close(fd)
            if bytes_errno is not None:break
        v=os.statvfs(root);byte_sample={'blocks':v.f_blocks,'available':v.f_bavail,'size':v.f_frsize}
        require(bytes_errno==errno.ENOSPC and v.f_bavail==0 and short>0,'actual short progress then tmpfs ENOSPC')
    finally:
        for p in files:p.unlink()
    files=[]
    try:
        for i in range(513):
            p=root/('inode-'+str(i))
            try:fd=os.open(p,os.O_WRONLY|os.O_CREAT|os.O_EXCL,0o400)
            except OSError as e:
                require(e.errno==errno.ENOSPC,'actual bounded tmpfs inode exhaustion');inode_errno=e.errno;break
            files.append(p);os.close(fd)
        v=os.statvfs(root);inode_sample={'files':v.f_files,'free':v.f_ffree}
        require(inode_errno==errno.ENOSPC and v.f_ffree==0,'intended actual tmpfs inode boundary')
    finally:
        for p in files:p.unlink()
    return {'actual_byte_errno':bytes_errno,'actual_inode_errno':inode_errno,'bytes_written':written,
            'positive_short_writes':short,'quota_write_syscalls':calls,'partial_seed_bytes':seed_bytes,
            'actual_unaligned_headroom_bytes':headroom,'lifetime_probe_write_upper':MAX_WRITE,
            'actual_byte_sample':byte_sample,'actual_inode_sample':inode_sample,
            'occupancy_not_cumulative_IO_credit':True}

def create_private_probe_record(spec):
    """Use an actual same-inode directory FD inside the existing private output."""
    root=Path(spec['record']);parent=root.parent
    require(root.is_absolute() and '..' not in root.parts and str(root)==os.path.normpath(str(root)),
            'normalized exact private probe record')
    for ancestor in reversed(parent.parents):
        require(stat.S_ISDIR(ancestor.lstat().st_mode),'private probe parent no symlink ancestor')
    q=parent.lstat()
    require(stat.S_ISDIR(q.st_mode) and stat.S_IMODE(q.st_mode)==0o700
            and q.st_uid==os.getuid() and q.st_gid==os.getgid()
            and [q.st_dev,q.st_ino]==spec['qualification_output_identity'],
            'same original admitted private probe output parent UID/identity')
    fd=os.open(parent,os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW)
    retained={'fd':fd,'close_attempted':False,'closed':False} # FIRST after acquisition.
    try:
        a=os.fstat(fd)
        require([a.st_dev,a.st_ino]==[q.st_dev,q.st_ino]
                and stat.S_ISDIR(a.st_mode) and stat.S_IMODE(a.st_mode)==0o700
                and a.st_uid==os.getuid() and a.st_gid==os.getgid(),
                'actual private probe parent FD before mkdir')
        require(not os.path.lexists(root),'one private qualification no replay or dangling symlink')
        os.mkdir(root.name,0o700,dir_fd=fd)
        after=os.stat(root.name,dir_fd=fd,follow_symlinks=False)
        require(stat.S_ISDIR(after.st_mode) and stat.S_IMODE(after.st_mode)==0o700
                and after.st_uid==os.getuid() and after.st_gid==os.getgid(),
                'own probe record explicit 0700 before output writes')
    finally:
        retained['close_attempted']=True
        os.close(fd) # One attempt only; ambiguity is an actual failure, no retry/PASS.
        retained['closed']=True
    return root
