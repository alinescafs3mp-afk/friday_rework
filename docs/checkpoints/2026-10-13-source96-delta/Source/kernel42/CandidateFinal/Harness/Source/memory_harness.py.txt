def clone(value):
    return json.loads(json.dumps(value, sort_keys=True, allow_nan=False))


def pin(raw, path):
    assert type(raw) is bytes
    return {'path': path, 'bytes': len(raw), 'sha256': hashlib.sha256(raw).hexdigest()}


def exact(raw, ref, maximum):
    assert type(ref) is dict and set(ref) == {'path', 'bytes', 'sha256'}
    assert type(ref['path']) is str and ref['path']
    assert type(raw) is bytes and type(ref['bytes']) is int
    assert 0 < len(raw) == ref['bytes'] <= maximum
    assert len(ref['sha256']) == 64 and hashlib.sha256(raw).hexdigest() == ref['sha256']

def extract(source_blobs, binding, ast_module, store):
    """The only Source exec is an exact, explicitly named AST subset AFTER GO.
    ast_module is Root's trusted stdlib AST parser, supplied by the external
    bootstrap, never imported from Source or discovered through a finder.
    Reject executable constants, decorators, hidden imports and forbidden globals.
    Source top-level imports and all host/caller/native classes are discarded.
    """
    ns = {'__builtins__': safe_builtins(), 'json': json, 'hashlib': hashlib,
          'zlib': zlib, 'types': types, 'sys': types.SimpleNamespace(modules={}),
          'kernel_snapshot': fake_kernel, 'same_read': store.same_read,
          '_real': synthetic_real_guard}
    fingerprints = {}
    forbidden = {'__import__', 'open', 'eval', 'exec', 'compile', 'os', 'Path',
                 'resource', 'time', 'threading', 'subprocess', 'importlib',
                 'globals', 'locals', '__loader__', '__spec__', '__subclasses__'}
    for filename, original_selection in SOURCE_SELECTION.items():
        selection = original_selection
        if filename in ('native63_observe.py', 'native41_evidence.py'):
            extra_constants = DEPENDENCY_EXTENSION['constants']
            extra_functions = DEPENDENCY_EXTENSION['functions']
            if filename == 'native63_observe.py':
                extra_constants = tuple(name for name in extra_constants if name != 'FAILURE_OUTPUT_OVERSHOOT_MAX')
                extra_functions = tuple(name for name in extra_functions if name != 'failure_owned_state')
            selection = {'constants': original_selection['constants'] + extra_constants,
                         'functions': original_selection['functions'] + extra_functions}
            if filename == 'native63_observe.py':
                selection['functions'] += DEPENDENCY_EXTENSION['observer_functions']
        raw = source_blobs[filename]
        exact(raw, binding['source_pins'][filename], 2097152)
        tree = ast_module.parse(raw, filename=filename)
        chosen = []; found = set(); wanted = set(selection['constants']) | set(selection['functions'])
        for node in tree.body:
            name = None
            if isinstance(node, ast_module.Assign) and len(node.targets) == 1 and isinstance(node.targets[0], ast_module.Name):
                name = node.targets[0].id
                if name in selection['constants']:
                    ast_module.literal_eval(node.value)  # Constants ONLY; no calls/effects.
                else:
                    continue
            elif isinstance(node, ast_module.FunctionDef):
                name = node.name
                if name not in selection['functions']:
                    continue
                assert not node.decorator_list and node.returns is None
                for arg in node.args.posonlyargs + node.args.args + node.args.kwonlyargs:
                    assert arg.annotation is None
                for default in node.args.defaults + [d for d in node.args.kw_defaults if d is not None]:
                    ast_module.literal_eval(default)
                for child in ast_module.walk(node):
                    assert not isinstance(child, (ast_module.Import, ast_module.ImportFrom))
                    if isinstance(child, ast_module.Name):
                        assert child.id not in forbidden
                    if isinstance(child, ast_module.Attribute):
                        assert child.attr not in ('__globals__', '__code__', '__subclasses__', '__mro__')
            else:
                continue
            assert name not in found
            found.add(name); chosen.append(node)
            text = ast_module.get_source_segment(raw.decode(), node).encode()
            fingerprints[filename + ':' + name] = hashlib.sha256(text).hexdigest()
        assert found == wanted
        selected = ast_module.Module(body=chosen, type_ignores=[])
        code = compile(ast_module.fix_missing_locations(selected), filename + ':exact-allowed-AST', 'exec')
        exec(code, ns, ns)  # Only selected literal assignments/function definitions.
    assert (ns['TRACE_COMPRESSED_MAX'], ns['TRACE_RAW_MAX'], ns['SEGMENT_COMPRESSED_MAX'],
            ns['SEGMENT_WIRE_MAX'], ns['FAILURE_WIRE_MAX']) == (430080, 16777216, 24576, 65536, 65536)
    assert (ns['KNOWN_BIRTH_MAX'], ns['DIRECT_BIRTH_MAX'], ns['CAPTURE_EXTENSION_MAX'],
            ns['NONBIRTH_HANDLE_MAX'], ns['LIVE_HANDLE_MAX']) == (128, 8, 51, 126, 254)
    assert len(ns['PREBIRTH_NEGATIVES']) == 13
    assert len(ns['DECLARED_APPLICATION_NAMES']) == len(set(ns['DECLARED_APPLICATION_NAMES'])) == 6
    assert set(ns['DECLARED_APPLICATION_NAMES']) == set(ns['DECLARED_APPLICATION_PINS'])
    return ns, fingerprints


class MemoryPath:
    __slots__ = ('text',)
    def __init__(self, text):
        assert type(text) is str and text.startswith('memory://')
        self.text = text
    def __truediv__(self, leaf):
        assert type(leaf) is str and '/' not in leaf and '..' not in leaf
        return MemoryPath(self.text + '/' + leaf)
    def __str__(self):
        return self.text


class MemoryStore:
    def __init__(self):
        self.rows = {}; self.serial = 0; self.total = 0
    def put(self, path, raw):
        assert type(raw) is bytes and 0 < len(raw) <= LIMITS['leaf']
        assert path not in self.rows
        assert self.total + len(raw) <= LIMITS['aggregate'] and len(self.rows) < 20
        self.serial += 1
        self.rows[path] = {'raw': raw, 'handle': 'MEMORY-HANDLE-%d' % self.serial}
        self.total += len(raw)
        return pin(raw, path)
    def replace(self, path, raw):
        if path in self.rows:
            self.total -= len(self.rows[path]['raw']); del self.rows[path]
        return self.put(path, raw)
    def handle(self, path):
        return self.rows[path]['handle']
    def same_read(self, handle, ref, maximum=65536):
        # No FD API exists. Handle and path identify the SAME in-memory bytes.
        assert type(handle) is str and handle.startswith('MEMORY-HANDLE-')
        row = self.rows[ref['path']]
        assert row['handle'] == handle
        exact(row['raw'], ref, min(maximum, LIMITS['leaf']))
        return row['raw']

def snapshot(owner):
    # Independent oracle normalizes the genuine producer's missing fields.
    fields=('pid','birth','state','ppid','waited','absent','returncode')
    return clone({'known': [{k: row.get(k) for k in fields} for row in owner.known.values()],
                  'direct_pids': list(owner.direct), 'unknown_spawn': owner.unknown_spawn,
                  'ambiguous': owner.ambiguous, 'output_bytes': owner.output_bytes,
                  'live_labels': [k for k, v in owner.fds.items() if v is not None],
                  'pending_receipt': owner.pending_receipt is not None})


class MemoryOwner:
    def __init__(self, ns, store, births=()):
        self.ns = ns; self.store = store; self.births = clone(list(births))
        self.known = {}; self.direct = []; self.unknown_spawn = False; self.ambiguous = {}
        self.output_bytes = 0; self.fds = {name: 'MEMORY_RETAINED' for name in
            ('lock_update', 'lock_install', 'native_host_pidfd', 'native_capture_capsule')}
        self.pending_receipt = None; self.temp_root = MemoryPath('memory://evidence')
        self.group = MemoryPath('memory://group'); self.calls = []; self.expected = []
        self.failures = {}; self.put_failure = {}; self.put_attempts = {}; self.put_success = []
        self.run_count = 0; self.extension_names = ()
        for method in ns['METHODS']:
            def invoke(owner, *args, _name=method, **kw):
                return owner.dispatch(_name, args, kw)
            setattr(self, method, types.MethodType(invoke, self))
    def native_check(self):
        assert len(self.known) <= 128 and len(self.direct) <= 8 and len(self.calls) <= 1024
    def dispatch(self, name, args, kw):
        assert name in self.ns['METHODS']
        self.calls.append(name); assert len(self.calls) <= LIMITS['operations']
        if name in self.failures:
            error, state = self.failures[name]
            if state == 'latched': self._c40_observation_failed = True
            if state == 'exhausted': self._c40_raw_bytes = LIMITS['raw']
            raise error
        if name == 'native_spawn':
            row = clone(self.births[args[0]])
            key = json.dumps([row['pid'], row['birth']], separators=(',', ':'))
            assert key not in self.known
            assert 'state' not in row, 'genuine identity(pid) has no state'
            self.fds['direct_pidfd_' + str(row['pid'])] = 'MEMORY_RETAINED_BIRTH'
            for stream in ('stdout', 'stderr'):
                self.fds['spawn_' + str(row['pid']) + '_' + stream] = 'MEMORY_RETAINED_STREAM'
            self.known[key] = row; self.direct.append(row['pid']); assert len(self.direct) <= 8
            return row['pid']
        if name == 'native_wait':
            row = self.known[args[0]]; self.direct.remove(row['pid'])
            row.update(waited=True, absent=True, returncode=0) # genuine native_wait; state remains unknown
            return row['pid'], 0
        if name == 'native_children_closed':
            if args:
                assert self.known[args[0]]['waited'] is True and self.known[args[0]]['absent'] is True
            return True
        if name == 'native_run':
            # No fabricated stat sample: direct genuine rows are never refreshed.
            self.run_count += 1
            return types.SimpleNamespace(returncode=0)
        if name == 'native_charge':
            self.output_bytes += len(args[0]); return len(args[0])
        if name in ('native_close_selector', 'native_close_stream', 'native_close_fd'):
            label = args[0] if args else 'memory_selector'
            if label in self.fds: self.fds[label] = None
            return None
        if name == 'native_module':
            for extension in self.extension_names:
                self.fds['capture_extension_' + extension] = 'MEMORY_CAPTURED_EXTENSION'
        if name == 'native_capture_restore':
            for label in self.fds:
                if label.startswith('capture_'): self.fds[label] = None
        if name == 'native_stage_receipt': self.pending_receipt = 'SYNTHETIC_NO_RECEIPT'
        if name == 'native_finish':
            assert not self.direct and all(row['waited'] and row['absent'] for row in self.known.values())
            for label in self.fds:
                if not label.startswith('lock_'): self.fds[label] = None
            self.pending_receipt = None
        return None
    def native_put(self, path, value):
        name = str(path); self.put_attempts[name] = self.put_attempts.get(name, 0) + 1
        assert self.put_attempts[name] == 1  # Fail closed against any retry.
        self.fds[name] = 'memory-tracked-before-fsync-close'
        raw = self.ns['wire'](value)
        assert 0 < len(raw) <= 65536
        if name in self.put_failure:
            stage, error = self.put_failure[name]
            assert stage in ('fsync', 'close', 'reserved')
            raise error  # SAME tracked memory custody remains; no OS call.
        self.store.put(name, raw); self.put_success.append(name); self.fds[name] = None

def eighteen_segments(ns):
    store = MemoryStore(); owner = MemoryOwner(ns, store)
    ns['attach'](owner, 'MEMORY_18_SEGMENTS')
    reached = False
    for ordinal in range(LIMITS['pressure_iterations']):
        padding = ''.join(hashlib.sha256(('memory-entropy:%d:%d' % (ordinal, n)).encode()).hexdigest() for n in range(64))
        owner._c40_add({'operation': 'memory_segment_pressure', 'phase': 'actual', 'padding': padding})
        # Size the genuine Source encoder's bounded terminal output without
        # flushing/mutating it or implementing a second observation encoder.
        projected = len(owner._c40_compressed) + len(owner._c40_encoder.copy().flush())
        if projected > 17 * 24576:
            assert projected <= 430080
            reached = True; break
    assert reached, 'bounded data sizing failed; never widen caps or retry'
    index = ns['persist'](owner)
    decoded = consume(ns, owner, index)
    assert len(index['segments']) == 18 and decoded['index'] == index
    leaves = [json.loads(store.rows[r['path']]['raw']) for r in index['segments']]
    assert all(len(bytes.fromhex(leaf['compressed_hex'])) == 24576 for leaf in leaves[:-1])
    assert 0 < len(bytes.fromhex(leaves[-1]['compressed_hex'])) < 24576
    ns['observation_failure'](owner, RuntimeError('synthetic reserved capacity control'))
    ns['reserved_failure'](owner, {'class': 'MemoryControlError', 'detail': 'SYNTHETIC_NO_NATIVE_CREDIT'})
    path = str(owner.temp_root / 'C41-OBSERVATION-FAILURE.json')
    ns['same_read'] = store.same_read
    failure = ns['failure_evidence'](store.handle(path), pin(store.rows[path]['raw'], path))
    assert failure['capacity_release'] is False and failure['normal_evidence_complete'] is False
    assert len(store.rows) == 20 and store.total <= 1048576
    assert all(len(row['raw']) <= 65536 for row in store.rows.values())
    return {'segments': 18, 'partial_final_bytes': len(bytes.fromhex(leaves[-1]['compressed_hex'])),
            'compressed_bytes': index['compressed_bytes'], 'actual_serialized_total_with_index_and_reserved': store.total,
            'leaf_max_actual': max(len(row['raw']) for row in store.rows.values()), 'leaves': 20}

def frame(ns, raw, template, compressed=None):
    """Construct exact-pinned hostile INPUT frames, not a validation mirror.
    All framing validation remains solely the exact Source caller_evidence body.
    """
    store = MemoryStore(); index = clone(template)
    compressed = zlib.compress(raw, 6) if compressed is None else compressed
    refs = []
    for start in range(0, len(compressed), 24576):
        path = 'memory://negative/segment-%03d' % len(refs)
        leaf = {'schema': 'friday.causal.lossless-segment.v1', 'case': index['case'],
                'index': len(refs), 'offset': start, 'compressed_hex': compressed[start:start + 24576].hex()}
        leafraw = ns['wire'](leaf); refs.append(store.put(path, leafraw))
    index.update(segments=refs, compressed_bytes=len(compressed), compressed_sha256=hashlib.sha256(compressed).hexdigest(),
                 raw_bytes=len(raw), raw_sha256=hashlib.sha256(raw).hexdigest())
    path = 'memory://negative/index'; ref = store.put(path, ns['wire'](index))
    return store, index, path, ref


def check_frame(ns, item):
    store, index, path, ref = item
    ns['same_read'] = store.same_read
    return ns['caller_evidence'](store.handle(path), ref, [store.handle(r['path']) for r in index['segments']])


def rewrite_index(ns, item):
    store, index, path, ref = item
    return store, index, path, store.replace(path, ns['wire'](index))

        raw = ns['wire'](failure).replace(b'"NUMERIC_SENTINEL"', literal); ref = store.replace(path, raw)
        ns['same_read'] = store.same_read
        results['reserved:' + label] = expect_refusal(lambda store=store, path=path, ref=ref: ns['failure_evidence'](store.handle(path), ref))
    assert ns['strict_json'](b'{"kernel":{"nested":[1e100,0.25]}}')['kernel']['nested'] == [1e100, 0.25]
    return results


def reserved_producer_failure_controls(ns):
    """Finite producer-shaped failures: no real Popen/pidfd/read/effects.
    Static immutable causality establishes ordering. These memory cases verify
    the actual reserved publisher/consumer retains the resulting truthful shape.
    Post-retention fstat ambiguity does NOT imply reserved publication succeeded.
    """
    result={}
    for shape in ('PIDFD_OPEN_BEFORE_TRACK', 'STRICT_READ_CHARGE_OVERSHOOT'):
        store=MemoryStore();owner=MemoryOwner(ns,store)
        ns['attach'](owner,'MEMORY_RESERVED_'+shape)
        if shape=='PIDFD_OPEN_BEFORE_TRACK':
            owner.direct.append(91);owner.unknown_spawn=True
        else:owner.output_bytes=33554432+131072
        primary=OSError(5,'MEMORY_ONLY_'+shape)
        ns['note_observation_failure'](owner,primary)
        ns['reserved_failure'](owner,ns['exception_summary'](primary))
        path=str(owner.temp_root/'C41-OBSERVATION-FAILURE.json')
        assert path in store.rows
        ns['same_read']=store.same_read
        failure=ns['failure_evidence'](store.handle(path),pin(store.rows[path]['raw'],path))
        value=failure['owned']
        assert failure['capacity_release'] is False and failure['normal_evidence_complete'] is False
        expect_refusal(lambda value=value:ns['owned_state'](value))
        if shape=='PIDFD_OPEN_BEFORE_TRACK':
            assert value['known']==[] and value['direct_pids']==[91] and value['unknown_spawn'] is True
            bad=clone(value);bad['unknown_spawn']=False
        else:
            assert value['output_bytes']==33554432+131072
            bad=clone(value);bad['output_bytes']+=1
        expect_refusal(lambda bad=bad:ns['failure_owned_state'](bad))
        assert len(store.rows)==1 and store.total<=65536
        result[shape]={'reserved_shape_preserved':True,'normal_credit':False,
                       'capacity_release':False,'actual_kernel_calls':0}
    return result


def qualify(source_blobs, binding, external_root_go, harness_bytes, testdata_bytes, resource_bytes, ast_module):
    """Root-only entry AFTER independently reviewed source+harness final bytes.
    Calls are finite; refusal/MemoryError/timeout are failures, never empty/zero
    observations or permission. Root owns external process stop/wait/closure.
    """
    data = admission(source_blobs, binding, external_root_go, harness_bytes, testdata_bytes, resource_bytes)
    store = MemoryStore()
    ns, fingerprints = extract(source_blobs, binding, ast_module, store)
    result = {'schema': 'friday.memory-only.observed-result.v1',
              'credit': 'BOUNDED_SYNTHETIC_EXACT_BODY_ONLY_NO_LIVE_KERNEL_NATIVE_ACCEPTANCE',
              'selected_body_hashes': fingerprints}
    result['full_sequence'] = full_sequence(ns, data)
    result['eighteen_segments'] = eighteen_segments(ns)
    result['decoder_negatives'] = decoder_negatives(ns, data['births'])
    result['genuine_producer_inventory'] = producer_inventory_controls(ns, data)
    result['strict_JSON_all_paths'] = strict_json_controls(ns, data['births'])
    result['reserved_producer_failures'] = reserved_producer_failure_controls(ns)
    result['original_exception_identity'] = original_exception_controls(ns)
    result['persistence_failures'] = persistence_failure_controls(ns, data['births'])
    result['assertion_structure'] = assertion_structure_controls(ns, data)
    result['original_live63_plus4_kernel32_genuine_all_gates_untouched_not_run'] = True
    result['Root_external_terminal_direct_wait_fresh_all_known_closure'] = 'REQUIRED_SEPARATELY_NOT_OBSERVED_INSIDE'
    return result
