def strict_number(value, name, low, high):
    return number(value, name, low, high)

def resolve_attribute(row):
    # Entire row has already passed pure exact JSON validation.
    module = sys.modules.get(text(row["module"], "module", 256))
    if module is None:
        raise Unbound("qualified_module_not_loaded")
    value = module
    for part in text(row["attribute"], "attribute", 1024).split("."):
        if not part.isidentifier() or part.startswith("__"):
            raise Unbound("qualified_attribute_shape")
        value = getattr(value, part)
    return value

def nested_code(function, qualname):
    function = function.__func__ if type(function) is types.MethodType else function
    if type(function) is not types.FunctionType:
        raise Unbound("site_not_original_python_function")
    pending, found = [function.__code__], []
    while pending:
        code = pending.pop()
        if code.co_qualname == qualname:
            found.append(code)
        pending.extend(x for x in code.co_consts if type(x) is types.CodeType)
    if len(found) != 1:
        raise Unbound("exact_nested_site_ambiguous")
    return found[0]

        self.guardian_sha256 = hashlib.sha256(canonical(guardian)).hexdigest()
        self.effects = frozenset(grant["effects"])

            raise Unbound("unavoidable_constructor_effects_not_granted")
        resources = grant["resources"]
        if type(resources) is not dict or set(resources) != {"AS_bytes", "CPU_seconds", "FS_bytes", "CORE_bytes", "affinity", "owned_fds", "owned_threads"}:

        self.functions = MappingProxyType(dict(self.functions))
        self.constructor_kwargs = MappingProxyType(dict(self.constructor_kwargs))
        self.used_cases = set()
        self._sealed = True

    def check(self, *, useful=True):
        if type(self) is not NativeRuntime:
            raise Unbound("runtime_not_exact_Source_type")
        current_scope = {k:v for k,v in self.grant.items() if k not in ("root_authorization","independent_safety")}
        if (hashlib.sha256(canonical(current_scope)).hexdigest() != self.scope_sha256
                or hashlib.sha256(canonical(self.protocol)).hexdigest() != self.protocol_sha256
                or frozenset(self.grant["effects"]) != self.effects):
            raise Unbound("qualified_whole_scope_or_protocol_mutated")
        if hashlib.sha256(canonical(self.guardian)).hexdigest() != self.guardian_sha256:
            raise Unbound("guardian_receipt_mutated")
        supervisor = self.guardian["supervisor_birth"]
        actual_guardian = self._qualification.proc_birth(supervisor["pid"])
        if actual_guardian["state"] not in {"R","S","D","I"} or any(actual_guardian[k] != supervisor[k] for k in supervisor):
            raise Unbound("native_guardian_birth_lost_before_effect")
        now_birth = self._qualification.proc_birth(os.getpid())
        if any(now_birth[k] != self.birth[k] for k in ("pid", "starttime", "boot_id")):
            raise Unbound("native_host_birth_changed")
        limit = self.original_deadline(final=not useful)
        if time.monotonic() >= limit:
            raise Unbound("original_native_parent_deadline")
        self.home_custody.verify()
        return limit

    def original_deadline(self, *, final=False):
        clock = self._original_clock
        if (type(clock) is not tuple or len(clock) != 3 or any(type(v) not in (int,float) for v in clock)
                or not clock[0] < clock[1] < clock[2]):
            raise Unbound("original_cleanup_clock_unknown_no_new_deadline")
        return number(clock[2] if final else clock[1],"retained_original_deadline",0,1e12)

    def cleanup_reserve(self):
        return number(self._cleanup_reserve,"retained_cleanup_reserve",.25,.25)

    def readiness(self, fd, *, writable, timeout):
        integer(fd,"original_readiness_FD",0,10**9)
        number(timeout,"bounded_readiness_wait",0,.05)
        if (type(writable) is not bool or sys.modules.get("select") is not self._select_module
                or self._select_module.select is not self._select_primitive
                or self._select_module.__spec__.origin != self._select_origin):
            raise Unbound("descriptor_free_select_original_primitive_lost")
        # Exact loaded C select primitive creates no Source FD. Native extension
        # bytes and interpreter belong to the already exact whole layout pin.
        readable,writes,_ = self._select_primitive([] if writable else [fd],[fd] if writable else [],[],timeout)
        return bool(writes if writable else readable)

    def source_FD_state(self):
        return self._fd_budget.snapshot()

    def original_function(self, name):
        return self._qualification.real_function(self.functions[name],self.grant["function_rows"][name])

    def require_effect(self, name, *, useful=True):
        text(name, "effect", 128)
        self.check(useful=useful)
        if name not in self.effects:
            raise Unbound("effect_not_explicitly_granted:" + name)

    def require_effects(self, names, *, useful=True):
        if type(names) not in (frozenset, tuple, list, set) or any(type(x) is not str for x in names):
            raise Unbound("effect_sequence_type")
        for name in sorted(names):
            self.require_effect(name, useful=useful)

    def qualified_function(self, name):
        text(name,"qualified_function_name",256)
        self.check(useful=False)
        if name not in self.functions:
            raise Unbound("native_function_not_in_sealed_whole_map")
        return self._qualification.real_function(self.functions[name],self.grant["function_rows"][name])

    def qualified_class(self, name):
        text(name,"qualified_class_name",256)
        self.check()
        if name not in self.classes:
            raise Unbound("native_class_not_in_sealed_whole_map")
        value = self.classes[name]
        self._qualification.real_function(value.__init__,self.grant["class_rows"][name]["constructor"])
        return value

    def constructor_parameters(self):
        self.check()
        if canonical(dict(self.constructor_kwargs)) != canonical(self.grant["constructor_kwargs"]):
            raise Unbound("actual_constructor_parameters_differ_from_frozen_scope")
        return pure(self.grant["constructor_kwargs"],"constructor_parameters")

    def begin(self, case_id):
        text(case_id, "case_id", 128)
        self.check()
        if case_id not in self.grant["allowed_cases"] or case_id in self.used_cases:
            raise Unbound("case_not_granted_or_replayed")
        self.require_effects(CONSTRUCTOR_EFFECTS)
        self.used_cases.add(case_id)
