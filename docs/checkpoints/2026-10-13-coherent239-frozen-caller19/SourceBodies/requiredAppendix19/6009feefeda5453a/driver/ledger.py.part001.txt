"""Witness actual supervised task creation from the original native return frame."""
from __future__ import annotations
import asyncio
import types
from .qualification import Unbound, code_digest
from .native import object_key

_ISSUER = object()


class LedgerWitness:
    def __init__(self, issuer, *, frame, returned, qualification, binding):
        if issuer is not _ISSUER:
            raise Unbound("no_actual_frame_issuer")
        code = frame.f_code
        if (code_digest(code) != qualification["code_sha256"]
                or code.co_filename != qualification["filename"]
                or code.co_qualname != qualification["qualname"]):
            raise Unbound("spawn_frame_not_exact_original")
        loc = frame.f_locals
        if (loc.get("self") is not binding.ctx
                or type(returned) is not asyncio.Task
                or loc.get("task") is not returned
                or loc.get("handle") is None
                or loc["handle"].active is not True
                or returned.get_loop() is not loc.get("loop")):
            raise Unbound("actual_task_registration_not_complete")
        self.task = returned
        self.handle = loc["handle"]
        self.context = binding.ctx
        self.manager = binding.manager
        self.manifest = binding.manifest
        self.state = binding.state
        self.loop = loc["loop"]
        self.qualification = dict(qualification)
        self.binding = binding
        self.consumed = False

    def require(self, *, task, context):
        if (task is not self.task or context is not self.context
                or self.context._manager is not self.manager
                or self.context.manifest is not self.manifest
                or self.context.state is not self.state
                or self.handle.active is not True
                or task.get_loop() is not self.loop
                or asyncio.get_running_loop() is not self.loop
                or asyncio.current_task() is not task
                or task.done()):
            raise Unbound("original_task_registration_lost")
        return self

    def observation(self):
        return {"task": object_key(self.task), "handle": object_key(self.handle),
                "loop": object_key(self.loop), "manager": object_key(self.manager),
                "manifest": object_key(self.manifest),
                "state": object_key(self.state),
                "active": self.handle.active, "task_done": self.task.done(),
                "source_qualification": self.qualification}


def witness_actual_return(*, event, frame, returned, qualification, binding):
    if event != "return":
        raise Unbound("spawn_task_witness_before_return")
    # This function is a trace callback consumer, not an operation producer.
    # Root must install its independently accepted exact site before original
    # spawn_task and before creation of the persistent native _run_async loop.
    return LedgerWitness(_ISSUER, frame=frame, returned=returned,
                         qualification=qualification, binding=binding)
