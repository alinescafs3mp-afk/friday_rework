"""Install observations before actual native construction, never fabricate it.

Root still must implement the genuine frontend/input/caller test producer.
This bootstrap ONLY captures actual qualified frames and original references.
It does not construct an alternative CLI, model, thread, task, writer or lease.
"""
from __future__ import annotations
import sys
import threading
import time
from .qualification import Unbound, code_digest
from .caller import witness_actual_entry
from .ledger import witness_actual_return


class BootstrapObservation:
    def __init__(self, *, qualified_sites, classes, original_final_deadline):
        if original_final_deadline <= time.monotonic():
            raise Unbound("bootstrap_original_clock_expired")
        self.sites = qualified_sites
        self.classes = classes
        self.final = original_final_deadline
        self.original_cli = None
        self.original_caller = None
        self.binding = None
        self.ledger = []
        self.events = []
        self.unknown = []
        self.installed = False
        self.code_cache = {}
        self.live_frames = []  # not a serialized native authority

    def code_matches(self, frame, row):
        digest = self.code_cache.get(frame.f_code)
        if digest is None:
            digest = code_digest(frame.f_code)
            self.code_cache[frame.f_code] = digest
        return (digest == row["code_sha256"]
                and frame.f_code.co_filename == row["filename"]
                and frame.f_code.co_qualname == row["qualname"])

    def set_actual_binding(self, binding):
        if self.binding is not None:
            raise Unbound("bootstrap_binding_replaced")
        if binding.cli is not self.original_cli:
            raise Unbound("not_observed_original_constructor_cli")
        binding.check()
        self.binding = binding
        if self.original_caller is not None:
            self.original_caller.require_original(binding)
            binding.caller_witness = self.original_caller

    def trace(self, frame, event, result):
        observed_at = time.monotonic()
        if observed_at >= self.final:
            self.unknown.append("bootstrap_final_boundary")
            raise Unbound("bootstrap_final_boundary")
        if len(self.events) >= 2048:
            self.unknown.append("bootstrap_event_limit")
            raise Unbound("bootstrap_event_limit")
        for name, row in self.sites.items():
            if event != row["event"] or not self.code_matches(frame, row):
                continue
            self.live_frames.append(frame)
            self.events.append({"site": name, "at": observed_at,
                                "frame": id(frame), "actor": threading.get_native_id()})
            if name == "original_constructor_return":
                cli = frame.f_locals.get("self")
                if type(cli) is not self.classes["cli"] or self.original_cli is not None:
                    raise Unbound("original_constructor_identity_ambiguous")
                self.original_cli = cli
            elif name == "original_caller_entry":
                if self.original_caller is not None:
                    raise Unbound("duplicate_original_caller")
                agent = frame.f_locals.get(row["agent_argument"])
                if type(agent) is not self.classes["agent"]:
                    raise Unbound("original_agent_not_real_class")
                self.original_caller = witness_actual_entry(
                    event=event, frame=frame, qualification=row, agent=agent,
                    approved_original_seconds=30, absolute_parent_deadline=self.final)
                if self.binding is not None:
                    self.original_caller.require_original(self.binding)
                    self.binding.caller_witness = self.original_caller
            elif name == "ledger_task_created":
                if self.binding is None:
                    self.unknown.append("actual_ledger_before_binding")
                    raise Unbound("actual_ledger_before_binding")
                witness = witness_actual_return(
                    event=event, frame=frame, returned=result,
                    qualification=row, binding=self.binding)
                self.ledger.append(witness)
                self.binding.ledger_witness = witness
        return self.trace

    def install_before_native_construction(self):
        if self.installed or sys.gettrace() is not None or threading.gettrace() is not None:
            raise Unbound("preexisting_trace_guard_or_duplicate_install")
        # Root must arrange this before intact native construction. Existing
        # threads cannot be silently counted as newly observed/qualified.
        sys.settrace(self.trace)
        threading.settrace(self.trace)
        self.installed = True

    def detach(self):
        if not self.installed:
            return
        if sys.gettrace() != self.trace or threading.gettrace() != self.trace:
            raise Unbound("bootstrap_trace_owner_lost")
        sys.settrace(None)
        threading.settrace(None)
        self.installed = False
        # Native per-thread hooks and exact live frame references remain UNKNOWN
        # until Root proves normal/adverse closure; no claimed automatic reap.

    def source_result(self):
        return {"events": self.events, "unknown": self.unknown,
                "original_constructor_observed": self.original_cli is not None,
                "original_caller_observed": self.original_caller is not None,
                "real_task_witnesses": len(self.ledger),
                "producer": "NOT_DONE",
                "runtime_qualification": "UNBOUND_NOT_ACCEPTED"}
