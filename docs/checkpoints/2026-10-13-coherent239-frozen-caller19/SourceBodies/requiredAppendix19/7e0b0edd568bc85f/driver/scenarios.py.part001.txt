turn, a quiescent worker result, a PluginContext ledger task or Root acceptance.
Whole ResultTool/native-caller orchestration is a separate NOT_DONE dependency.
Every current control is NOT_RUN; component observations cannot award suite PASS.
"""

class UnknownStream(io.TextIOBase):
    def write(self, data):
        raise AssertionError("unsupported_custom_writer_was_invoked")
    def flush(self):
        raise AssertionError("unsupported_custom_flush_was_invoked")


@contextlib.contextmanager
def changed_attribute(obj, name, value):
    existed = hasattr(obj, name)
    previous = getattr(obj, name, None)
    setattr(obj, name, value)
    try:
        yield
    finally:
        if getattr(obj, name, None) is not value:
            raise Unbound("fault_attribute_custody_changed")
        if existed:
            setattr(obj, name, previous)
        else:
            delattr(obj, name)


def ledger_task(binding):
    task = asyncio.current_task()
    if task is None:
        raise Unbound("no_existing_native_task")
    witness = getattr(binding, "ledger_witness", None)
    if witness is None:
        raise Unbound("actual_spawn_task_frame_witness_UNBOUND")
    # The witness must have been retained from the exact original spawn_task
    # return frame after actual task creation, not reconstructed from JSON.
    witness.require(task=task, context=binding.ctx)
    return task


def payload_parameters(case_id):
    if case_id in {"P02", "L093"}:
        return bytes(range(256)) * 1024, "я" * 127, "😀" * 2048
    return (("actual ordinary CLI output " + case_id + "\n").encode(),
            "ordinary-cli-" + case_id + ".txt", "genuine physical observation")


def preflight_variants(case_id):
    if case_id in {"N25", "L092"}:
        return [{"timeout": v} for v in (True, False, float("nan"),
                float("inf"), -1, 0, 30.000001)] + [
                {"deadline": v} for v in (True, float("nan"), float("inf"), -1, 0)]
    if case_id in {"N26", "L049"}:
        return [{"data": bytearray(b"not exact bytes")},
                {"data": b"x" * (256*1024+1)},
                {"file_name": ""}, {"file_name": "../foreign"},
                {"file_name": "😀" * 64},
                {"file_name": "x" * 256},
                {"caption": "x" * 2049}]
    raise Unbound("not_a_preflight_variant_case")


def foreign_mutation(case_id, binding, foreign):
    """Use another REAL qualified fixture when a distinct owner is required."""
    if foreign is None:
        raise Unbound("genuine_foreign_fixture_not_bound")
    foreign.check()
    mapping = {
        "N02": (binding.prompt.proxy, "app_session", foreign.session),
        "N03": (binding.prompt.proxy, "_output", foreign.app.output),
        "N04": (binding.cli, "_app", foreign.app),
        "N05": (binding.app, "future", foreign.app.future),
        "N30": (binding.prompt, "renderer", foreign.app.renderer),
        "L008": (binding.agent, "session_id", foreign.agent.session_id),
        "L009": (binding.cli, "agent", foreign.agent),
        "L010": (binding.agent, "_session_db", foreign.database),
        "L014": (binding.cli, "console", foreign.console),
        "L016": (binding.cli, "_friday_cli_scope", foreign.cap),
        "L045": (binding.ctx, "manifest", foreign.manifest),
        "L062": (binding.ctx, "state", foreign.state),
    }
    if case_id not in mapping:
        raise Unbound("foreign_mutation_not_implemented")
    return changed_attribute(*mapping[case_id])


def simple_mutation(case_id, binding):
    if case_id in {"N01", "L015", "L091"}:
        return changed_attribute(binding.console, "_file", UnknownStream())
    if case_id == "N06":
        return changed_attribute(binding.app.output, "enable_cpr", True)
    if case_id == "N07":
        return changed_attribute(binding.app, "full_screen", True)
    if case_id == "L011":
        return changed_attribute(binding.agent, "_session_db", None)
    if case_id == "L042":
        return changed_attribute(binding.ctx, "_load_abandoned", True)
    if case_id == "L043":
        return changed_attribute(binding.manager, "home_path",
                                 binding.cap.home / "unadmitted")
    if case_id == "L044":
        return changed_attribute(binding.manager, "_cli_ref", None)
    raise Unbound("simple_mutation_not_implemented")


def require_unchanged_history(before, after):
    if before != after:
        raise AssertionError("read_only_or_refused_operation_changed_history")


def require_no_document(before, after, *, phase):
    if before["wire"] != after["wire"]:
        # Native prompt redraw is not a document; parse the actual wire.
        from .physical import document_frames
        first, first_partial = document_frames(before["wire"])
        last, last_partial = document_frames(after["wire"])
        if len(last) != len(first) or len(last_partial) != len(first_partial):
            raise AssertionError("unexpected_document_effect:" + phase)


async def invoke_document(binding, *, data, file_name, caption,
                          timeout=30, deadline=None):
    from hermes_cli.friday_cli_work import deliver_cli_document, route_for
    turn = binding.check()
    task = ledger_task(binding)
    original = binding.next_operation()
    original.retain_original_task()
    if deadline is None:
        end = original.deadline
    elif type(deadline) not in (int, float) or not math.isfinite(deadline) or deadline <= 0:
        # Preserve invalid input EXACTLY for the real producer preflight.
        # Narrowing must not turn bool/NaN/infinity into a valid timeout.
        end = deadline
    else:
        end = min(original.deadline, deadline)
    result = await deliver_cli_document(
        binding.ctx, route=route_for(turn), data=data,
        file_name=file_name, caption=caption, timeout=timeout, deadline=end)
    if asyncio.current_task() is not task:
        raise Unbound("document_caller_task_substituted")
    # This return is ONLY the component return, not the full native caller.
    return result, original


async def component_case(case_id, binding, capture, *, foreign=None):
    """Run one actual component path; no synthetic syscall counts or PASS."""
    from hermes_cli.friday_cli_work import route_for
    initial_turn = binding.check()
    data, name, caption = payload_parameters(case_id)
    before = capture.snapshot()
    started = time.monotonic()
    mode = None
    result = None
    custody = None
    expected = None

    if case_id in {"P01", "P02", "P03", "P04", "P05", "L067", "L068",
                   "L069", "L093"}:
        if case_id in {"P01", "P02", "P03", "L068"} and (
                binding.prompt is None or not os.isatty(binding.owner.stream.fileno())):
            raise Unbound("real_original_prompt_and_tty_required")
        if case_id == "P04" and binding.prompt is not None:
            raise Unbound("direct_tty_case_must_use_original_direct_frontend")
        if case_id == "P05" and type(binding.owner.stream) is not io.StringIO:
            raise Unbound("exact_original_memory_frontend_required")
        result, custody = await invoke_document(
            binding, data=data, file_name=name, caption=caption)
        expected = "DELIVERED"
        mode = "actual_component_document"
    elif case_id in {"N25", "N26", "L092", "L049"}:
        evidence = []
        for variant in preflight_variants(case_id):
            before_variant = capture.snapshot()
            params = {"data": data, "file_name": name, "caption": caption}
            params.update(variant)
            answer, current = await invoke_document(binding, **params)
            capture.read_available()
            require_no_document(before_variant, capture.snapshot(), phase=case_id)
            if answer.get("state") != "FAILED":
                raise AssertionError("invalid_input_not_refused")
            evidence.append({"variant": repr(variant), "answer": answer,
                             "elapsed": time.monotonic()-current.started})
        return {"case": case_id, "component": evidence,
                "whole_native_caller": "NOT_DONE", "qualification": "UNBOUND",
                "status": "OBSERVATION_ONLY_NEVER_SUITE_PASS"}
    elif case_id in {"N01", "N06", "N07", "L011", "L015", "L042", "L043",
                     "L044", "L091", "N02", "N03", "N04", "N05", "N30",
                     "L008", "L009", "L010", "L014", "L016", "L045", "L062"}:
        if case_id in {"N01", "N06", "N07", "L011", "L015", "L042",
                       "L043", "L044", "L091"}:
            mutation = simple_mutation(case_id, binding)
        else:
            mutation = foreign_mutation(case_id, binding, foreign)
        # Verify actual native binding BEFORE the deliberate adverse mutation.
        # Do not bypass an adverse check by re-verifying it as a new capability.
        from hermes_cli.friday_cli_work import deliver_cli_document
        task = ledger_task(binding)
        end = binding.caller_witness.require_original(binding).deadline
        with mutation:
            result = await deliver_cli_document(
                binding.ctx, route=route_for(initial_turn), data=data,
                file_name=name, caption=caption, timeout=30, deadline=end)
        if asyncio.current_task() is not task:
            raise Unbound("refusal_caller_task_substituted")
        expected = "FAILED"
        mode = "actual_adverse_owner_preflight"
    else:
        raise Unbound("NOT_DONE_case_producer:" + case_id)

    capture.read_available()
    after = capture.snapshot()
    if result.get("state") != expected:
        raise AssertionError("unexpected_component_state:" + repr(result))
    if expected == "DELIVERED":
        content = require_exact_document(
            after, data=data, route=route_for(initial_turn),
            file_name=name, caption=caption)
        if result.get("human_read") != "UNCONFIRMED":
            raise AssertionError("unsupported_human_read_claim")
    else:
        require_no_document(before, after, phase=case_id)
        content = {"document_effect": "ABSENT"}
    return {"case": case_id, "mode": mode, "component_result": result,
            "content": content, "elapsed": time.monotonic()-started,
            "whole_native_caller": "NOT_DONE",
            "qualification": "UNBOUND",
            "status": "OBSERVATION_ONLY_NEVER_SUITE_PASS"}
