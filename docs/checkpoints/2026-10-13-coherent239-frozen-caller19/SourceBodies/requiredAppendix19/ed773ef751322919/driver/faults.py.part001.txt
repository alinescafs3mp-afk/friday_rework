"""Finite adverse-control mechanisms; proposals UNBOUND_NOT_GRANTED.

No mechanism runs at import. Each requires a Root-qualified exact site and the
already owned original task/descriptor. No syscall result/PASS is fabricated.
Real zero/bool-return backend injection is NOT_IMPLEMENTED here; those required
legacy controls remain NOT_DONE, rather than renamed to a different test.
"""
from __future__ import annotations
import asyncio
import fcntl
import os
import threading
import time
from .qualification import Unbound
from .native import descriptor, object_key


class CancelOriginalTask:
    def __init__(self, binding, *, phase, second_phase=None):
        if phase not in {"before_acquire", "after_begin", "cleanup"}:
            raise Unbound("unsupported_cancel_site")
        witness = binding.ledger_witness
        if witness is None:
            raise Unbound("actual_original_task_UNBOUND")
        self.binding = binding
        self.task = witness.task
        self.loop = witness.loop
        self.phase = phase
        self.second_phase = second_phase
        self.scheduled = []
        self.handles = []
        self.closed = False

    def at_actual_phase(self, *, site, operation):
        # Called ONLY by the separately accepted native frame observer.
        if self.closed or site not in {self.phase, self.second_phase}:
            return
        if operation.binding is not self.binding.prompt:
            raise Unbound("cancel_foreign_operation")
        if any(row["site"] == site for row in self.scheduled):
            raise Unbound("cancel_site_replayed")
        caller = self.binding.caller_witness
        caller.require_original(self.binding)
        if caller.stop_intent not in (None, "cancel"):
            raise Unbound("different_original_stop_intent")
        caller.stop_intent = "cancel"
        self.scheduled.append({"site": site, "at": time.monotonic(),
                               "task": object_key(self.task),
                               "loop": object_key(self.loop),
                               "operation": object_key(operation)})
        # Fixed native task.cancel callback only, never an arbitrary blocking
        # callback and never a replacement writer or new coroutine.
        handle = self.loop.call_soon_threadsafe(self.task.cancel,
                                               "qualified_private_adverse_control")
        self.handles.append(handle)

    def close(self):
        self.closed = True
        # Cancel only unexecuted callbacks that this mechanism itself scheduled.
        # A canceled handle does not prove native task/finally/FD cleanup.
        for handle in self.handles:
            if not handle.cancelled():
                handle.cancel()
        return {"scheduled": self.scheduled,
                "task_done": self.task.done(),
                "task_cancelled": self.task.cancelled(),
                "exact_reap": "ROOT_REQUIRED"}


class RevokeOriginalLifetime:
    def __init__(self, binding):
        self.binding = binding
        self.applied = None

    def at_actual_body(self, *, operation):
        if self.applied is not None:
            raise Unbound("lifetime_revoke_repeated")
        if operation.binding is not self.binding.prompt:
            raise Unbound("foreign_prompt_revoke")
        self.binding.caller_witness.require_original(self.binding)
        self.binding.owner.close()
        self.applied = {"at": time.monotonic(),
                        "owner": object_key(self.binding.owner),
                        "operation": object_key(operation)}
        # Closed original capability cannot be reset/recreated for this fixture.
        return self.applied


class ExactPrivateDescriptorClose:
    def __init__(self, binding, *, original_descriptor_birth):
        self.binding = binding
        self.birth = original_descriptor_birth
        self.applied = None
        if self.birth.get("provenance") != "ROOT_QUALIFIED_NEW_PRIVATE_FIXTURE_FD":
            raise Unbound("private_descriptor_allocation_UNBOUND")

    def at_actual_lease(self, *, lease):
        if self.applied is not None:
            raise Unbound("descriptor_fault_replayed")
        if lease.owner is not self.binding.owner or lease.fd is None:
            raise Unbound("not_exact_owned_product_lease")
        observed = descriptor(lease.original_fd)
        if (observed["identity"] != tuple(self.birth["identity"])
                or observed["fd"] != self.birth["fd"]):
            raise Unbound("original_private_fd_changed_before_fault")
        self.binding.caller_witness.require_original(self.binding)
        # Close the descriptor created by the NEW private fixture. No owner
        # terminal, unknown FD, process signal or generic cleanup is allowed.
        os.close(lease.original_fd)
        self.applied = {"at": time.monotonic(),
                        "original": observed,
                        "retained_product_fd": descriptor(lease.fd)}
        return self.applied


class PreserveGlobalFlags:
    def __init__(self, binding):
        self.binding = binding
        self.original = descriptor(binding.owner.stream.fileno())
        self.sys_stdout = __import__("sys").stdout
        self.sys_stderr = __import__("sys").stderr

    def observe(self):
        current = descriptor(self.original["fd"])
        if current != self.original:
            raise AssertionError("product_changed_original_global_FD_flags")
        import sys
        if sys.stdout is not self.sys_stdout or sys.stderr is not self.sys_stderr:
            raise AssertionError("driver_or_product_swapped_current_stdio")
        return {"descriptor": current,
                "stdout": object_key(sys.stdout),
                "stderr": object_key(sys.stderr)}
