"""Bind one original caller clock to its actual qualified native entry frame.

No operation, artifact, callback, trace shard or historical receipt may reset it.
Current whole-caller orchestration is NOT_DONE: Root must supply/qualify the
actual producer and outer site covering ALL activity/metadata/DB/UI tail.
"""
from __future__ import annotations
import os
import threading
import time
from .qualification import Unbound, code_digest, proc_birth
from .native import object_key

_ISSUER = object()


class WholeCallerWitness:
    def __init__(self, issuer, *, frame, qualification, agent,
                 approved_original_seconds, absolute_parent_deadline, observed_at):
        if issuer is not _ISSUER:
            raise Unbound("no_actual_original_entry_issuer")
        code = frame.f_code
        if (code_digest(code) != qualification["code_sha256"]
                or code.co_filename != qualification["filename"]
                or code.co_qualname != qualification["qualname"]
                or frame.f_locals.get(qualification["agent_argument"]) is not agent):
            raise Unbound("whole_caller_entry_not_exact")
        if approved_original_seconds != 30:
            raise Unbound("whole_caller_original30_required")
        self.frame = frame
        self.agent = agent
        self.qualification = dict(qualification)
        self.actor_tid = threading.get_native_id()
        self.host = proc_birth(os.getpid())
        self.entered = observed_at
        if type(observed_at) not in (int, float) or observed_at > time.monotonic():
            raise Unbound("original_entry_observation_clock_invalid")
        self.deadline = min(self.entered+30.0, absolute_parent_deadline)
        if self.deadline <= self.entered:
            raise Unbound("caller_parent_clock_already_exhausted")
        self.returned = None
        self.stop_intent = None
        self.operation_refs = []
        self.failures = []
        self.unknown = []
        self.tail = []

    def require_original(self, binding):
        if (binding.agent is not self.agent or binding.host_birth["pid"] != self.host["pid"]
                or binding.host_birth["starttime"] != self.host["starttime"]
                or self.returned is not None):
            raise Unbound("caller_identity_or_lifetime_lost")
        if time.monotonic() >= self.deadline:
            raise Unbound("original_whole_caller_deadline")
        return self

    def observe_tail(self, site, *, monotonic, actor, observation):
        # Consumer receives raw native trace data, never a case success boolean.
        if self.returned is not None:
            self.unknown.append("tail_observed_after_closed_caller")
            raise Unbound("tail_after_caller_closed")
        self.tail.append({"site": site, "monotonic": monotonic,
                          "actor": actor, "observation": observation})

    def observe_original_return(self, *, event, frame):
        if event != "return" or frame is not self.frame or self.returned is not None:
            raise Unbound("not_exact_original_return")
        self.returned = time.monotonic()
        required = {"activity_join_return", "metadata_callback_return",
                    "session_flush_return", "ui_completed_return"}
        present = {row["site"] for row in self.tail}
        if not required <= present:
            self.unknown.append("missing_actual_native_tail")
        if self.returned > self.deadline:
            self.failures.append("whole_original_caller_exceeded30")
        if any(row["monotonic"] > self.returned for row in self.tail):
            self.failures.append("tail_after_original_return")
        # Joining the heartbeat must prove the actual created thread ended.
        joins = [row for row in self.tail if row["site"] == "activity_join_return"]
        if not joins or any(row["observation"].get("alive_after_join") is not False
                            for row in joins):
            self.unknown.append("activity_thread_join_not_proven")

    def observation(self):
        return {"host": self.host, "agent": object_key(self.agent),
                "actor_tid": self.actor_tid, "entered": self.entered,
                "deadline": self.deadline, "returned": self.returned,
                "elapsed": None if self.returned is None else self.returned-self.entered,
                "stop_intent": self.stop_intent, "tail": list(self.tail),
                "failures": list(self.failures), "unknown": list(self.unknown),
                "qualified_site": self.qualification,
                "postreturn_owned_tree_absence": "ROOT_REQUIRED"}


def witness_actual_entry(*, event, frame, qualification, agent,
                         approved_original_seconds, absolute_parent_deadline):
    observed_at = time.monotonic()
    if event != "call":
        raise Unbound("original_entry_not_call_event")
    return WholeCallerWitness(_ISSUER, frame=frame, qualification=qualification,
                              agent=agent,
                              approved_original_seconds=approved_original_seconds,
                              absolute_parent_deadline=absolute_parent_deadline,
                              observed_at=observed_at)
