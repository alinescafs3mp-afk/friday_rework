"""Finite partial Source dispatcher, not an execution grant or full150 gate.

There is deliberately no command-line/import-time execution entry point.
Root separately owns real original frontend/caller construction and acceptance.
"""
from __future__ import annotations
import asyncio
import hashlib
import json
import time
from .qualification import Unbound, exact_file, exact_json, verify_go, verify_loaded
from .scenarios import component_case
from .control import original_readonly_status, original_release


def load_collection(reference):
    index, custody = exact_json(reference)
    if index["schema"] != "CLI150-EXACT-COLLECTION-1" or index["count"] != 150:
        raise Unbound("wrong_exact150_collection")
    rows = []
    for part in index["parts"]:
        body, _ = exact_json(part)
        if body["schema"] != "CLI150-EXACT-COLLECTION-PART-1":
            raise Unbound("wrong_collection_part")
        rows.extend(body["controls"])
    ids = [row["id"] for row in rows]
    if len(ids) != 150 or len(set(ids)) != 150:
        raise Unbound("missing_or_duplicate_control")
    if sum(x["origin"] == "LEGACY106_EXACT_REQUIRED" for x in rows) != 106:
        raise Unbound("legacy106_collection_changed")
    if sum(x["origin"] == "NEW44_EXACT_REQUIRED" for x in rows) != 44:
        raise Unbound("new44_collection_changed")
    return index, {row["id"]: row for row in rows}


class PartialDriver:
    def __init__(self, *, go_reference, assignment, generation,
                 collection_reference, driver_reference):
        self.go, self.go_custody = verify_go(
            go_reference, assignment=assignment, generation=generation,
            collection_ref=collection_reference, source_ref=driver_reference)
        self.index, self.cases = load_collection(collection_reference)
        self.used = set()
        self.stop_intent = None
        self.observations = {}
        self.not_run = {key: "NO_CURRENT_EXECUTION" for key in self.cases}
        # Whole150 execution is refused while required producers are missing.
        # Author Source presence is not independent review or runtime acceptance.
        self.whole_driver_done = False

    def begin_case(self, case_id):
        if self.stop_intent is not None:
            raise Unbound("original_stop_cleanup_intent")
        if time.monotonic() >= self.go["useful_deadline_monotonic"]:
            raise Unbound("original_useful_boundary")
        if case_id not in self.cases or case_id not in self.go["allowed_cases"]:
            raise Unbound("case_not_explicitly_granted")
        if case_id in self.used:
            raise Unbound("control_replay_prohibited")
        self.used.add(case_id)
        return self.cases[case_id]

    async def observe_component(self, case_id, *, binding, capture, foreign=None):
        row = self.begin_case(case_id)
        if row["component_source_path"] != "driver.scenarios.component_case":
            self.not_run[case_id] = "NOT_DONE_COMPONENT_CASE_PRODUCER"
            raise Unbound("NOT_DONE_component_path")
        verify_loaded(binding.whole_layout)
        try:
            value = await component_case(case_id, binding, capture, foreign=foreign)
        except BaseException as exc:
            self.stop_intent = "fault_or_uncertain_component"
            self.not_run[case_id] = "FAILED_OR_UNKNOWN:" + type(exc).__name__
            raise
        else:
            self.observations[case_id] = value
            self.not_run[case_id] = "WHOLE_NATIVE_CALLER_AND_REQUIRED_SUBCONTROLS_NOT_DONE"
            return value

    def observe_status(self, case_id, *, binding, reference, collector):
        if case_id not in {"N33", "L052", "L064"}:
            raise Unbound("not_original_status_case")
        self.begin_case(case_id)
        value = original_readonly_status(binding, reference=reference, collector=collector)
        self.observations[case_id] = value
        self.not_run[case_id] = "FULL_NAMED_STATUS_SUBCONTROLS_NOT_DONE"
        return value

    def require_full150_complete(self):
        if not self.whole_driver_done:
            raise Unbound("NOT_DONE_genuine_original_caller_orchestration_all150")
        # No static or component boolean path can currently reach full PASS.
        raise Unbound("NOT_DONE_final_independent_full150_qualification")

    def close_slice(self):
        """No waiting/polling, sender, rerun, new task or clock change."""
        return {"whole_driver": "NOT_DONE", "all150": "NOT_ACCEPTED",
                "observed_components": self.observations,
                "missing_or_unrun": self.not_run,
                "original_useful": self.go["useful_deadline_monotonic"],
                "original_final": self.go["final_deadline_monotonic"],
                "stop_intent": self.stop_intent,
                "owned_helpers_afterreturn": "ROOT_REQUIRED_UNKNOWN_NOT_ZERO"}
