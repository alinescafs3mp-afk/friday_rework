"""Actual original authenticated slash/status Source, without send or remint.

Only real constructor-owned original CLI/context/host are accepted. This module
has NOT_RUN status and does not complete the native full150 orchestration.
"""
from __future__ import annotations
import hashlib
import os
from pathlib import Path
import time
from .qualification import Unbound, exact_file
from .native import object_key


def state_reference(binding):
    path = Path(binding.state.path)
    if not path.is_absolute() or not path.is_relative_to(binding.cap.home):
        raise Unbound("state_outside_actual_authenticated_private_home")
    data = path.read_bytes()
    return {"path": str(path), "bytes": len(data),
            "sha256": hashlib.sha256(data).hexdigest()}


def current_state_custody(binding):
    ref = state_reference(binding)
    data, custody = exact_file(ref, limit=65536)
    return ref, data, custody


def original_readonly_status(binding, *, reference, collector):
    if type(reference) is not str or not reference or len(reference.encode()) > 512:
        raise Unbound("invalid_actual_history_reference")
    binding.check(input_required=False, output_required=False)
    before_ref, before_bytes, before_custody = current_state_custody(binding)
    before_ordinal = binding.prompt.ordinal if binding.prompt is not None else None
    start_event = len(collector.events)
    started = time.monotonic()
    # Original registry -> original cli_command -> original PluginContext/host
    # handler. No direct synthetic control lease, task, tool ID or output route.
    binding.cli._run_plugin_slash_command("/friday-status", reference)
    after_ref, after_bytes, after_custody = current_state_custody(binding)
    if (after_bytes != before_bytes or after_ref != before_ref
            or after_custody != before_custody):
        raise AssertionError("read_only_status_changed_or_replaced_state")
    after_ordinal = binding.prompt.ordinal if binding.prompt is not None else None
    if after_ordinal != before_ordinal:
        raise AssertionError("historical_status_created_new_output_operation")
    events = collector.events[start_event:]
    returned = [x for x in events if x["site"] == "authenticated_status_return"]
    if len(returned) != 1:
        raise Unbound("actual_original_status_return_UNBOUND")
    # The current command may read a truthful historical DELIVERED/UNKNOWN row.
    # It cannot reuse its earlier live send/operation capability.
    if any(x["site"] in {"prompt_begin", "physical_write_return",
                         "result_deliver_enter"} for x in events):
        raise AssertionError("status_caused_send_or_replay")
    return {"reference": reference, "events": events,
            "state_before": before_ref, "state_after": after_ref,
            "custody": before_custody,
            "elapsed": time.monotonic()-started,
            "input_scope": "GENUINE_CURRENT_COMMAND_NOT_CURRENT_SEND",
            "historical_capability_reused": False,
            "qualification": "UNBOUND_NOT_ACCEPTED"}


def original_generic_command_refusal(binding, *, command, arguments):
    binding.check(input_required=False, output_required=False)
    if command.startswith("/friday-"):
        raise Unbound("this_adverse_control_requires_nonfriday_command")
    before, bytes_before, custody_before = current_state_custody(binding)
    try:
        binding.cli._run_plugin_slash_command(command, arguments)
    except Exception as exc:
        kind = type(exc).__name__
    else:
        raise AssertionError("unowned_generic_plugin_command_not_refused")
    after, bytes_after, custody_after = current_state_custody(binding)
    if (before != after or bytes_before != bytes_after
            or custody_before != custody_after):
        raise AssertionError("refused_generic_command_changed_state")
    return {"refusal": kind, "state": before,
            "actual_original_cli": object_key(binding.cli),
            "qualification": "UNBOUND_NOT_ACCEPTED"}


def original_release(binding):
    binding.check(input_required=False, output_required=False)
    output, lease = binding.owner, binding.cli._active_session_lease
    binding.cli._release_active_session()
    if not output.closed.is_set() or binding.cli._active_session_lease is not None:
        raise AssertionError("original_constructor_output_lifetime_not_closed")
    # This irreversible private fixture closure is not reset or reminted.
    return {"original_output": object_key(output), "original_lease": object_key(lease),
            "closed": True, "recreation": "PROHIBITED_IN_THIS_FIXTURE",
            "unknown_native_task_tree": "ROOT_REQUIRED"}
