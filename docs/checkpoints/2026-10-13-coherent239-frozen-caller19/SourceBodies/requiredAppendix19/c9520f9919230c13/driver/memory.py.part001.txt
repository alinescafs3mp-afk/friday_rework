"""Observe the exact original C StringIO without substituting its write method."""
from __future__ import annotations
import hashlib
import io
import time
from .qualification import Unbound


class OriginalMemoryCapture:
    def __init__(self, stream, *, deadline):
        if type(stream) is not io.StringIO or stream.closed:
            raise Unbound("not_original_exact_StringIO")
        initial = stream.getvalue()
        if stream.tell() != len(initial) or len(initial) > 393216:
            raise Unbound("not_original_bounded_append_position")
        self.stream = stream
        self.start_position = stream.tell()
        self.initial = initial
        self.deadline = deadline
        self.closed = False
        self.failures = []

    def read_available(self):
        if self.closed or self.stream.closed:
            raise Unbound("memory_observer_closed")
        if time.monotonic() >= self.deadline:
            raise Unbound("memory_observer_original_deadline")
        text = self.stream.getvalue()
        if not text.startswith(self.initial) or len(text) > 2*393216:
            raise Unbound("memory_prefix_changed_or_unbounded")
        return len(text)-self.start_position

    def snapshot(self):
        self.read_available()
        body = self.stream.getvalue()[self.start_position:].encode("ascii")
        return {"wire": body, "bytes": len(body),
                "sha256": hashlib.sha256(body).hexdigest(),
                "chunks": [{"at": time.monotonic(), "bytes": len(body),
                            "sha256": hashlib.sha256(body).hexdigest()}],
                "unknown": [], "failures": list(self.failures),
                "stream_identity": id(self.stream),
                "original_cursor": self.start_position}

    def close(self):
        # The observer owns no product stream and never closes it.
        self.closed = True
