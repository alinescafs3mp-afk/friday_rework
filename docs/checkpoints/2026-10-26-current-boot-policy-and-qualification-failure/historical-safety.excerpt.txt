# Exact source span: validate_saved
def validate_saved(saved,binding,generation,prepared,runtime,claim,checked_safety,specimen,
                   *, historical_dsh_raw=None,current_dsh_raw=None,policy_successor=None):
    """Preserve real old wrapper/clock/IDs; an A0 input change alone is irrelevant."""
    need(saved['schema']=='friday.user-worker-stage-observation.v1'
         and saved['home']==str(HOME) and saved['profile']==binding['runtime_profile']
         and saved['generation']==generation and saved['kind']=='dsh_safety'
         and saved['effects']=='OBSERVATION_ONLY' and saved['per_job_authority']=='NOT_GRANTED'
         and set(saved['preparations'])==set(prepared)=={'dsh','a0'}, 'saved_safety_ownership_changed')
    old_a0=saved['preparations']['a0']; reference(old_a0)
    need(old_a0['path']==str(HOME/'workers/a0/runtime-input.json'), 'historical_a0_preparation_identity_changed')
    if saved['preparations']['dsh'] != prepared['dsh'] or policy_successor is not None:
        need(historical_dsh_raw is not None and current_dsh_raw is not None,'historical_dsh_metadata_proof_required')
        dsh_preparation_unchanged(saved,historical_dsh_raw,current_dsh_raw,prepared['dsh'],policy_successor=policy_successor)
    old=saved['original_attempt']; observation=saved['observation']
    need(type(old) is dict and type(old.get('boot_id')) is str
         and type(old.get('budget_seconds')) is int and old['budget_seconds']==600
         and all(type(old.get(k)) in (int,float) and math.isfinite(old[k]) for k in ('started_mono','deadline_mono'))
         and 0 < old['started_mono'] and old['deadline_mono']-old['started_mono']==600
         and type(claim) is dict and type(claim.get('boot_id')) is str
         and all(type(claim.get(k)) in (int,float) and math.isfinite(claim[k]) for k in ('started_mono','deadline_mono'))
         and 0 < claim['started_mono'] < claim['deadline_mono'],'historical_attempt_not_separate')
    same_boot=old['boot_id']==claim['boot_id']
    if same_boot:
        need(old['deadline_mono']<claim['started_mono'],'historical_attempt_not_separate')
    else:
        need(policy_successor is not None,'s384_reconciled_crossboot_safety_required')
        evidence=_s384_scope_evidence(policy_successor)
        need(old['boot_id']==evidence['historical_boot_id'] and claim['boot_id']==evidence['scope']['boot_id']
             and evidence['scope']['observed_monotonic']<=claim['started_mono'],
             's384_historical_safety_current_claim_join_changed')
    checked_safety(observation['report'],runtime,HOME)
    need(observation['specimen_sha256']==sha(specimen.encode()),'safety_specimen_changed')
    execution=observation['execution']; envelope=execution['resource_envelope']
    expected_unit='friday-qualify-dsh-'+sha(str(HOME/'workers/qualification/dsh-safety').encode())[:24]+'-1.scope'
    need(type(execution['returncode']) is int and execution['returncode']==0 and execution['reaped'] is True and execution['timeout'] is False
         and set(envelope)=={'unit','memory_bytes','cpu_percent','tasks','boot_id','deadline'}
         and envelope['unit']==expected_unit and envelope['boot_id']==old['boot_id']
         and type(envelope['deadline']) in (int,float) and math.isfinite(envelope['deadline'])
         and old['started_mono']<=envelope['deadline']<=old['deadline_mono']
         and (not same_boot or envelope['deadline']<=claim['deadline_mono'])
         and all(typed_equal(envelope[k],runtime['dsh'][k]) for k in ('memory_bytes','cpu_percent','tasks')),
         'saved_safety_resources_changed')
    return copy.deepcopy(observation)
