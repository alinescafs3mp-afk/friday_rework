# Exact source span: _secondary_policy_same
def _secondary_policy_same(left, right):
    if type(left) is not type(right): return False
    if type(left) is dict:
        return set(left) == set(right) and all(_secondary_policy_same(left[k], right[k]) for k in left)
    if type(left) is list:
        return len(left) == len(right) and all(_secondary_policy_same(a,b) for a,b in zip(left,right))
    return left == right

        and value['schema'] == 'frw.Root.current-policy-native-scope.v1'

def _secondary_policy_plan(profile, identity, attachment, preparations, old_policy, new_policy,
                           *, now, issued_unix, validity_seconds, current_scope=None):
    """Exact expiry-only or captured current-boot+expiry data, never runtime authority."""
    require(profile.name == SECONDARY_PROFILE and identity['generation'] == 2
        and attachment['generation'] == identity['generation']
        and set(attachment['runtimes']) == set(preparations) == {'dsh','a0'}
        and type(now) in (int,float) and math.isfinite(now)
        and type(issued_unix) in (int,float) and math.isfinite(issued_unix)
        and type(validity_seconds) is int and 1 <= validity_seconds <= 86400
        and issued_unix <= now < issued_unix + validity_seconds,
        'worker_policy_successor_exact_finite_scope_required')
    allowed = {'expires_unix'}
    if current_scope is not None:
        _secondary_policy_scope(current_scope, now=now, home=str(profile.parent.parent), profile=profile.name)
        allowed |= {'boot_id','net_namespace'}
        require(new_policy['boot_id'] == current_scope['boot_id']
            and new_policy['net_namespace'] == current_scope['net_namespace']
            and type(new_policy['boot_id']) is str
            and type(new_policy['net_namespace']) is list
            and all(type(v) is int and v > 0 for v in new_policy['net_namespace']),
            'worker_policy_successor_captured_boot_namespace_required')
    require(set(old_policy) == set(new_policy)
        and all(_secondary_policy_same(old_policy[k],new_policy[k])
                for k in old_policy if k not in allowed)
        and type(new_policy['expires_unix']) is type(old_policy['expires_unix'])
        and type(new_policy['expires_unix']) in (int,float)
        and math.isfinite(new_policy['expires_unix'])
        and old_policy['expires_unix'] <= now < new_policy['expires_unix']
        and new_policy['expires_unix'] == issued_unix + validity_seconds,
        'worker_policy_successor_expiry_only_required')
    previous = attachment['runtimes']['dsh']['dsh']['web']['egress_evidence']
    require(previous == SECONDARY_POLICY_SUCCESSOR_BINDING['policy']
        and previous['path'] == str(profile/'workers/dsh/inputs/web-policy.json'), 'worker_policy_successor_exact_expired_policy_required')
    # Match the installed native policy rebase serializer used by the qualifier.
    raw = (json.dumps(new_policy,sort_keys=True,allow_nan=False)+'\n').encode()
    reference = {'path':str(profile/'workers/dsh/inputs'/('web-policy-successor-'+hashlib.sha256(raw).hexdigest()+'.json')),
        'sha256':hashlib.sha256(raw).hexdigest()}
    updated = copy.deepcopy(attachment)
    updated['runtimes']['dsh']['dsh']['web']['egress_evidence'] = reference
    attachment_raw = _encoded(updated)
    attachment_ref = {'path':str(profile/'workers/attachment.json'),
        'sha256':hashlib.sha256(attachment_raw).hexdigest()}
    data = {str(Path(reference['path']).relative_to(profile)):raw,'workers/attachment.json':attachment_raw}
    for kind in ('dsh','a0'):
        value = preparations[kind]
        require(value['runtime'] == attachment['runtimes'][kind]
            and value['generation'] == identity['generation']
            and value['principal_binding_sha256'] == attachment['binding_sha256']
            and value['state'] == 'PREPARED_RUNTIME_UNOBSERVED'
            and value['attachment']['path'] == str(profile/'workers/attachment.json'),
            'worker_policy_successor_original_preparation_changed')
        prepared = copy.deepcopy(value)
        prepared.update(runtime=updated['runtimes'][kind], attachment=attachment_ref)
        if kind == 'dsh':
            require(value['inputs'].count(previous) == 1,
                'worker_policy_successor_exact_prepared_policy_pin_required')
            prepared['inputs'] = [reference if p == previous else p for p in value['inputs']]
        data['workers/'+kind+'/runtime-input.json'] = _encoded(prepared)
    return data
