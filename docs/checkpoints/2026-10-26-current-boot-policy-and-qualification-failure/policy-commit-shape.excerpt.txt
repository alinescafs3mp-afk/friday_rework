def _s383_policy_preparation_result(value, result_ref, binding, generation, prepared,
                                    policy_successor, current_preparations, accepted_monotonic, boot_id):
    """Exact R454 policy commit; data validation grants no worker authority."""
    need(policy_successor is not None, 's383_policy_reuse_required')
    old, new = policy_envelope(policy_successor)
    current=policy_successor['operator_policy']['schema']=='friday.worker-web.current-boot-expiry-successor.v1'
    fields = {'schema','state','attachment','preparations','preserved_policy',
        'preserved_policy_archive','beforeimages','reuse_binding','policy',
        'prior_derivative','clock','ready','runtime_acceptance','workers_enabled',
        'qualification_started','consumers_started','resume_allowed'}
    if current: fields.add('current_scope')
    need(type(value) is dict and set(value) == fields
        and value['schema'] == (CURRENT_POLICY_SCHEMA if current else 'friday.worker-declarations.secondary-policy-successor.v1')
        and value['state'] == 'POLICY_ONLY_SUCCESSOR_COMMITTED'
        and typed_equal(value['preparations'], prepared)
        and typed_equal(value['reuse_binding'], policy_successor)
        and typed_equal(value['policy'], new)
        and typed_equal(value['prior_derivative'], S383_POLICY_PRIOR_DERIVATIVE),
        's383_exact_policy_commit_required')
    need(all(value[key] is False for key in ('ready','workers_enabled',
        'qualification_started','consumers_started','resume_allowed'))
        and value['runtime_acceptance'] == 'NOT_GRANTED',
        's383_policy_no_authority_required')

# Exact source span: current scope and original parent clock join
    clock = value['clock']
    need(type(clock) is dict and set(clock) == {'schema','seconds','started_mono','deadline_mono','boot_id'}
        and clock['schema'] == 'friday.worker-declarations.maintenance-clock.v1'
        and type(clock['seconds']) is int and clock['seconds'] == 1200
        and all(type(clock[key]) in (int,float) and math.isfinite(clock[key])
            for key in ('started_mono','deadline_mono'))
        and type(accepted_monotonic) in (int,float) and math.isfinite(accepted_monotonic)
        and 0 <= clock['started_mono'] <= accepted_monotonic
        and clock['started_mono'] < clock['deadline_mono'] <= clock['started_mono']+1200
        and type(clock['boot_id']) is str and clock['boot_id'] == boot_id,
        's383_policy_immutable_clock_required')
    if current:
        need(typed_equal(reference(value['current_scope']),policy_successor['operator_policy']['current_scope']),
             's384_policy_commit_operator_scope_join_changed')
        evidence=_s384_scope_evidence(policy_successor); scope=evidence['scope']; parent=evidence['parent']
        need(scope['boot_id']==boot_id and scope['observed_monotonic']<=clock['started_mono']
             and clock['deadline_mono']<=parent['deadline_monotonic']-parent['reserve_seconds'],
             's384_current_policy_clock_parent_join_changed')
    return True
