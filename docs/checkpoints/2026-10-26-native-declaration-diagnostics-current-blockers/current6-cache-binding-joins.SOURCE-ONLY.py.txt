# Exact original lines 25-26
def encoded(value):
    return (json.dumps(value,ensure_ascii=False,sort_keys=True,indent=2,allow_nan=False)+'\n').encode()


# Exact original lines 28-32
def reference(ref):
    assert type(ref) is dict and set(ref)=={'path','sha256'}
    assert Path(ref['path']).is_absolute() and '..' not in Path(ref['path']).parts
    assert re.fullmatch('[0-9a-f]{64}',ref['sha256']) and ref['sha256']!='0'*64
    return ref


# Exact original lines 34-41
def read_cache(ref):
    reference(ref)
    path=Path(ref['path'])
    assert path.is_relative_to(CACHE), 'product/H read forbidden'
    assert not any(p.is_symlink() for p in [path,*path.parents])
    raw=path.read_bytes()
    assert hashlib.sha256(raw).hexdigest()==ref['sha256']
    return json.loads(raw)


# Exact original lines 43-46
def pin_cache(path):
    path=Path(path)
    assert path.is_absolute() and path.is_relative_to(CACHE)
    return {'path':str(path),'sha256':hashlib.sha256(path.read_bytes()).hexdigest()}

def write(name,value):
    target=HERE/name
    with target.open('xb') as stream: stream.write(encoded(value))
    target.chmod(0o600)
