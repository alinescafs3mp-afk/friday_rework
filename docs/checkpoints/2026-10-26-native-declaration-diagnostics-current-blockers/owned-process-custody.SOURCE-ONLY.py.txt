def process_identity(pid):
    raw=(Path('/proc')/str(pid)/'stat').read_text()
    fields=raw[raw.rfind(')')+2:].split()
    return {'birth':fields[19],'pgid':int(fields[2]),'state':fields[0]}

def wait_unreaped(process, seconds):
    """Observe leader completion without releasing its PID/group identity."""
    limit=time.monotonic()+seconds
    while True:
        event=os.waitid(os.P_PID,process.pid,os.WEXITED|os.WNOHANG|os.WNOWAIT)
        if event is not None:return event
        if time.monotonic()>=limit:return None
        time.sleep(min(0.05,max(0,limit-time.monotonic())))


# Exact original lines 164-171
def live_group_members(pgid):
    result=[]
    for path in Path('/proc').iterdir():
        if not path.name.isdecimal():continue
        try:row=process_identity(int(path.name))
        except (FileNotFoundError,ProcessLookupError):continue
        if row['pgid']==pgid and row['state']not in ('Z','X'):result.append(int(path.name))
    return result

def reap_owned(process, birth, *, deadline=None):
    # No Popen poll/wait occurs until the owned group has ceased. Even after
    # normal leader exit its WNOWAIT zombie retains the PID, preventing reuse.
    identity=process_identity(process.pid)
    need(identity['birth']==birth and identity['pgid']==process.pid,
         'owned_native_unreaped_identity_changed')
    limit_all=time.monotonic()+10 if deadline is None else deadline
    for sig in (signal.SIGTERM,signal.SIGKILL):
        if not live_group_members(process.pid):break
        need(process_identity(process.pid)['birth']==birth,'owned_native_unreaped_identity_changed')
        try:os.killpg(process.pid,sig)
        except ProcessLookupError:pass
        limit=min(limit_all,time.monotonic()+5)
        while live_group_members(process.pid) and time.monotonic()<limit:time.sleep(0.05)
    need(not live_group_members(process.pid),'STOP_UNCONFIRMED_owned_native_descendants')
    remaining=max(0,limit_all-time.monotonic())
    need(wait_unreaped(process,remaining)is not None,'STOP_UNCONFIRMED_owned_native_leader')
    return process.wait(timeout=max(0.001,limit_all-time.monotonic()))

def run_once(root):
    d,cut,end=preflight(root)
    args=[root['native_python'],'-I','-B',root['caller']['path']]
    put(d,'ONCE-INTENT.json',{'state':'UNKNOWN_BEFORE_SINGLE_NATIVE_PROCESS','argv':args,
        'root_binding':root,'no_retry':True,'phase_wait_cutoff':cut,'phase_hard_deadline':end})
    child=None; birth=None; code=None; timed_out=False; reaped=False; error=None; cleanup_attempted=False
    try:
        # Recheck ORIGINAL phase boundary after intent preparation, before spawn.
        need(time.monotonic()<cut,'phase_expired_before_spawn')
        with os.fdopen(os.open(d/'native.stdout.private',os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o600),'wb') as out, \
             os.fdopen(os.open(d/'native.stderr.private',os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o600),'wb') as err:
            syncdir(d)
            child=subprocess.Popen(args,cwd=d,env=root['factory_env'],stdout=out,stderr=err,start_new_session=True)
            birth=process_identity(child.pid)['birth']
            put(d,'STARTED.json',{'pid':child.pid,'birth':birth,'owned_pgid':child.pid,
                'started_observed_monotonic':time.monotonic(),'phase_clock':root['phase_clock']})
            event=wait_unreaped(child,max(0,cut-time.monotonic()))
            timed_out=event is None
            cleanup_attempted=True
            code=reap_owned(child,birth,deadline=end); reaped=True
    except BaseException as exc:
        # No exception message, traceback, stdout, stderr or user content in diagnostics.
        error=type(exc).__name__
    finally:
        if child is not None and not reaped and not cleanup_attempted:
            try:
                need(birth is not None,'STOP_UNCONFIRMED_birth_not_captured')
                cleanup_attempted=True
                code=reap_owned(child,birth,deadline=end); reaped=True
            except BaseException as exc: error='STOP_UNCONFIRMED_'+type(exc).__name__
    put(d,'RETURNED.json',{'state':'RETURNED' if reaped else 'UNKNOWN','pid':None if child is None else child.pid,
        'birth':birth,'exit_code':code,'reaped':reaped,'timed_out':timed_out,'error_class':error,
        'ended_monotonic':time.monotonic(),'phase_clock':root['phase_clock'],
        'caller_result_path':root['caller_result_path'],'genuine_inspect_accepted':False,'no_retry':True})
    need(reaped and error is None,'native_process_not_cleanly_returned')
    return code
