                        'control_dir_fd':context['control_dir_fd'],'trace_dir_fd':context['trace_dir_fd'],
                        'output_parent_identity':parent['identity']}
                    aux={'record':row['record'],'qualification_output_identity':row['output_identity'],
                         'qualification_output_mount_id':row['output_mount_id'],
                         'qualification_output_parent_capability':parent,
                         'existing_owner':{'group_fd':context['actual_clean_group_fd'],
                                          'slot_fd':context['actual_existing_slot_fd']}}
                    kernel120_verify_output_inventory(aux,temporary,custody)
                    require(Path(row['record']).parent.name=='friday-native-'+row['action_id'],
                            'actual action-private auxiliary output path')
                    for fd,identity in inventory.items():rows.append((fd,identity,kind,row))
                else:
                    fd=row['fd'];q=os.fstat(fd);identity=[q.st_dev,q.st_ino,stat.S_IFMT(q.st_mode)]
                    require(identity==row['identity'] if kind=='auxiliary_borrowed'
                            else identity[:2]==row['identity'],'exact auxiliary actual identity')
                    kernel120_auxiliary_actual_owner(context,row['ownership'],identity,kind)
                    if kind=='auxiliary_group':
                        group=Path(row['path']);parent=row['parent_capability']
                        root_fd_custody.directory_capability(parent,False)
                        root_fd_custody.directory_capability({'fd':fd,'path':str(group),
                            'identity':row['identity'],'policy':row['directory_policy']},False)
                        require(group.parent==Path(parent['path'])
                                and group.name=='friday-native-'+row['action_id']
                                and group.is_relative_to('/sys/fs/cgroup')
                                and group!=Path('/sys/fs/cgroup'),'actual action-private auxiliary group capability')
                        for writer in context['protected_writer_groups']:
                            protected=Path(writer['path'])
                            require(group!=protected and not group.is_relative_to(protected)
                                    and not protected.is_relative_to(group)
                                    and identity[:2]!=writer['identity'],
                                    'never remove protected/shared auxiliary scope or ancestor')
                        require((group/'pids.current').read_text().strip()=='0'
                                and 'populated 0' in (group/'cgroup.events').read_text()
                                and not (group/'cgroup.procs').read_text().strip(),'actual auxiliary kernel0')
                    else:
                        policy=row['owner_policy']
                        require(type(policy)is dict and set(policy)=={'uid','gid','mode'}
                                and all(type(v)is int and v>=0 for v in policy.values())
                                and (q.st_uid,q.st_gid,stat.S_IMODE(q.st_mode))
                                ==(policy['uid'],policy['gid'],policy['mode']),
                                'auxiliary borrowed actual admitted UID/mode')
                        require(row['role']==kind and (stat.S_ISDIR(q.st_mode) or stat.S_ISREG(q.st_mode)),
                                'borrowed auxiliary explicit directory/file role; never pidfd/GO')
                    rows.append((fd,identity,kind,row))
        require(len(rows)<=64 and len({r[0] for r in rows})==len(rows),'all auxiliary role FDs unique')
        # Semantic removable-node uniqueness also holds for direct cleanup callers.
        removable_ids=set();removable_paths=set();removable_mounts=set();removable_callers=set()
        seen_objects=set()
        for fd,identity,kind,row in rows:
            if kind=='auxiliary_borrowed':continue
            marker=id(row)
            if marker in seen_objects:continue # one output inventory, multiple FDs
            seen_objects.add(marker)
            key=tuple(row['identity'] if kind=='auxiliary_group' else row['output_identity'])
            path=row['path'] if kind=='auxiliary_group' else str(Path(row['record']).parent)
            mount=None if kind=='auxiliary_group' else row['output_mount_id']
            caller=row['ownership']['caller_fd'] if kind=='auxiliary_group' else None
            require(key not in removable_ids and path not in removable_paths
                    and (mount is None or mount not in removable_mounts)
                    and (caller is None or caller not in removable_callers),
                    'auxiliary semantic owned duplicate ZERO total prefix effects')
            removable_ids.add(key);removable_paths.add(path);removable_mounts.add(mount)
            if caller is not None:removable_callers.add(caller)
        for fd,identity,kind,row in rows:
            if kind=='auxiliary_borrowed':
                require(tuple(identity[:2]) not in removable_ids,
                        'borrowed inode never aliases owned removal')
            if custody.get('phase_order') is not None:
                planned=[r for r in custody['fds'] if r['fd']==fd and not r['closed']]
                require(len(planned)==1 and planned[0].get('planned')
                        and planned[0]['lifetime']=='phase'
                        and planned[0]['phase']==custody['active_phase']
                        and root_fd_custody.canonical_role(planned[0]['label'])==kind,
                        'actual auxiliary exact CURRENT planned lifetime; future is reserved')
        reserved.update(r['fd'] for r in custody['fds'] if not r['closed']
                        and root_fd_custody.canonical_role(r['label']) not in
                        ('auxiliary_group','auxiliary_output','auxiliary_borrowed'))
        require(not {r[0] for r in rows}&reserved,
                'auxiliary disjoint slot/group/control/trace/guardian/pidfd/GO/output/caller')
        reserved_ids={tuple(x) for x in context['all_phase_reserved_identities'].values()}
        for fd in reserved:
            try:q=os.fstat(fd)
            except OSError as error:
                closed_rows=[r for r in custody['fds'] if r['fd']==fd and r['closed']]
                require(error.errno==9 and closed_rows,'unknown reserved FD absence is not safe closure')
            else:reserved_ids.add((q.st_dev,q.st_ino,stat.S_IFMT(q.st_mode)))
        require(all(tuple(identity) not in reserved_ids for _,identity,_,_ in rows),
                'auxiliary actual inode disjoint reserved capabilities')
        for fd,identity,kind,row in rows:
            root_fd_custody.preflight_existing_role(custody,fd,kind)
        return rows
    finally:
        if capture is not None and scope[2]==1:
            capture.mark('auxiliary_consumer',scope[0],scope[1],2)



def kernel120_remove_owned_group(spec,fd,parent_cap=None,identity=None,path=None,policy=None):
    """Original caller's owned empty node; parent-FD removal, actual retained deletion."""
    group=Path(spec['native']['group'] if path is None else path)
    identity=spec['native']['first_birth_group_identity'] if identity is None else identity
    parent_cap=spec['native']['group_parent_capability'] if parent_cap is None else parent_cap
    policy=spec['native']['group_directory_policy'] if policy is None else policy
    root_fd_custody.directory_capability(parent_cap,False)
    root_fd_custody.directory_capability({'fd':fd,'path':str(group),'identity':identity,'policy':policy},False)
    require(group.parent==Path(parent_cap['path']),'same actual owned group parent before rmdir')
    require((group/'pids.current').read_text().strip()=='0'
            and 'populated 0' in (group/'cgroup.events').read_text()
            and not (group/'cgroup.procs').read_text().strip(),'fresh same owned empty group before rmdir')
    os.rmdir(group.name,dir_fd=parent_cap['fd'])
    try:os.stat(group.name,dir_fd=parent_cap['fd'],follow_symlinks=False)
    except FileNotFoundError:pass
    else:raise ValueError('original exact parent child still present after group removal')
    q=os.fstat(fd);link=os.readlink('/proc/self/fd/'+str(fd))
    require([q.st_dev,q.st_ino]==identity and (q.st_nlink==0 or link==str(group)+' (deleted)'),
            'retained original group kernel node deleted; renamed-path absence is not removal')
    return {'identity':identity,'retained_fd_deleted_link':link,'actual_nlink':q.st_nlink,
            'parent_fd_identity':parent_cap['identity'],'parent_relative_rmdir':True}

def kernel120_revalidate_output_before_effect(spec):
    """Parent/path/mount revalidated immediately BEFORE close or normal umount."""
    output=Path(spec['record']).parent
    parent=spec['qualification_output_parent_capability']
    root_fd_custody.validate_private_parent(str(output),parent)
    q=output.lstat()
    require(stat.S_ISDIR(q.st_mode) and stat.S_IMODE(q.st_mode)==0o700
            and q.st_uid==os.getuid() and q.st_gid==os.getgid()
            and [q.st_dev,q.st_ino]==spec['qualification_output_identity'],
            'same actual private output before path/close effect')
    mount=[r.split() for r in Path('/proc/self/mountinfo').read_text().splitlines()
           if r.split()[4]==str(output)]
    require(len(mount)==1 and mount[0][mount[0].index('-')+1]=='tmpfs'
            and int(mount[0][0])==spec['qualification_output_mount_id'],
            'same actual retained output mount immediately before effect')
    return output
