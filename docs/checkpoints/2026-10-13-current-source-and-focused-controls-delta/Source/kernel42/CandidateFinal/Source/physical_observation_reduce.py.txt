def _bounds(run,name,pid=None):
    return tuple(r for r in run.boundaries if r.name==name and (pid is None or r.pid==pid))
def _marks(run,name,pid=None):
    return tuple(r for r in run.markers if r.event==name and (pid is None or r.pid==pid))
def _code(run):
    observed=_marks(run,'refusal')
    physical.need(observed,'GENUINE_CAUGHT_REFUSAL_MISSING')
    codes=[]
    for event in observed:
        number=event.values[0]
        physical.need(0<number<=len(markers.REFUSALS),'UNKNOWN_REFUSAL_NOT_INTENDED')
        codes.append(markers.REFUSALS[number-1])
    physical.need(len(set(codes))==1,'BOUNDARIES_REFUSED_DIFFERENT_CAUSES')
    return codes[0]
def _proof(run,name,predicate,selectors=()):
    physical.need(predicate,'INDEPENDENT_PHYSICAL_PROOF_MISSING:'+name)
    return dict(run.raw_ref,proof=name,selectors=list(selectors),
                observer_pid=run.observer_pid,observer_birth=run.observer_birth)

def _interval(run,pid,late=False):
    if late:
        physical.need(pid==run.case.root_pid and run.case.mutation=='late_holder',
            'EXACT_LATE_CONSUMER_ROOT_ROLE')
        scope=(run.case.phase_order.index(run.case.last_phase)+1,current_control_word(run.case),1)
        begin=tuple(mark for mark in _marks(run,'auxiliary_consumer',pid)
            if mark.birth==run.known_subjects[pid][0] and mark.values==scope)
        end_scope=scope[:2]+(2,)
        end=tuple(mark for mark in _marks(run,'auxiliary_consumer',pid)
            if mark.birth==run.known_subjects[pid][0] and mark.values==end_scope)
    else:
        begin=_marks(run,'attempt_begin',pid)
        end=_marks(run,'attempt_end',pid)
    physical.need(len(begin)==1 and len(end)==1 and begin[0].kernel_sequence<end[0].kernel_sequence,
                  'EXACT_RAW_INTERVAL_UNBOUND_OR_RESET')
    return begin[0].kernel_sequence,end[0].kernel_sequence
def _scope_events(run,late=False):
    result=[]
    for pid in ((run.case.root_pid,) if late else run.registration['required_attempt_pids']):
        lo,hi=_interval(run,pid,late)
        result.extend(r for r in run.kernel if r.pid==pid and lo<r.sequence<hi)
    return tuple(sorted(result,key=lambda r:r.sequence))
def _count_marks(run,name,late=False):
    count=0
    for pid in ((run.case.root_pid,) if late else run.registration['required_attempt_pids']):
        lo,hi=_interval(run,pid,late)
        count+=sum(r.event==name and r.pid==pid and lo<r.kernel_sequence<hi for r in run.markers)
    return count
def _syscall_entries(run,number,late=False):
    return tuple(r for r in _scope_events(run,late) if r.stage=='entry' and r.syscall==number)
def _real_close_calls(run,late=False):
    # Count all original/future/batch capabilities and source-marked closes.
    # Observational source/proc reads are retained in raw transcript, and their
    # successful O_RDONLY acquisitions are checked. Nothing unknown is adopted.
    entries=_syscall_entries(run,3,late)
    actual=[]
    for row in entries:
        before=_bounds(run,'attempt_begin',row.pid)
        physical.need(len(before)==1,'COMPLETE_ORIGINAL_FD_SNAPSHOT_MISSING')
        original={r.fd for r in before[0].all_fd_reads}
        fd=row.arguments[0]
        if fd in original:actual.append(row)
        else:
            # Find the exact preceding real successful open of this fd and
            # classify it from source-qualified allowed read path and flags.
            opens=[e for e in run.kernel if e.pid==row.pid and e.stage=='exit'
                   and e.syscall in (2,257) and e.result==fd and e.sequence<row.sequence]
            physical.need(opens,'UNKNOWN_CLOSED_FD_STOP_NO_ZERO')
            opened=opens[-1];flags=opened.arguments[1] if opened.syscall==2 else opened.arguments[2]
            permitted={p['path'] for p in run.registration['complete_current_source_pins']}
            path=opened.path
            allowed=path in permitted or (path and path.startswith('/proc/'))
            allowed=allowed or any(path and path.startswith(g['path']+'/') and
                os.path.basename(path) in ('memory.max','memory.swap.max','pids.max','cpu.max',
                'cgroup.events','cgroup.procs','pids.current') for g in run.registration['admitted_groups'])
            physical.need(allowed and flags&os.O_ACCMODE==os.O_RDONLY and
                not flags&(os.O_CREAT|os.O_TRUNC),'UNKNOWN_OR_NONOBSERVATIONAL_CLOSE_NO_ZERO')
    return tuple(actual)

def prefix_counts(run,late=False):
    kernel=_scope_events(run,late)
    caller=run.case.caller_pid;root=run.case.root_pid
    def bind(pid):
        lo,hi=_interval(run,pid,late)
        return sum(m.event=='bind_existing' and m.pid==pid and
                   lo<m.kernel_sequence<hi for m in run.markers)
    pids=(run.case.root_pid,) if late else run.registration['required_attempt_pids']
    return {'caller_role_bind':bind(caller) if caller in pids else 0,
        'Root_role_bind':bind(root) if root in pids else 0,
        'derived_validated_record':_count_marks(run,'record',late),
        'close':len(_real_close_calls(run,late)),
        'rmdir':sum(e.stage=='entry' and
            (e.syscall==84 or e.syscall==263 and e.arguments[2]&0x200) for e in kernel),
        'umount':sum(e.stage=='entry' and e.syscall==166 for e in kernel),
        'first_birth':sum(e.stage=='entry' and e.syscall in (56,57,58,435) for e in kernel)}

def _late_original_guardian_difference(run,before,after):
    """One declared new owned pidfd, established by independent raw acquisition.

    Original capability equality remains strict; no FD-count allowance or
    baseline rewrite. This difference is ONLY the original guardian on the
    retained C83 late STOP and lasts to the actual abnormal Root terminal.
    """
    physical.need(run.case.mutation=='late_holder' and before.pid==after.pid==run.case.root_pid and
        before.birth==after.birth,'EXACT_LATE_ROOT_GUARDIAN_DOMAIN')
    marker_by_sequence={m.sequence:m.kernel_sequence for m in run.markers}
    lo=marker_by_sequence[before.marker_sequence];hi=marker_by_sequence[after.marker_sequence]
    stops=[b for b in run.boundaries if b.pid==before.pid and b.birth==before.birth and
        b.semantic_read and b.semantic_read['kind']==19 and
        lo<marker_by_sequence[b.marker_sequence]<hi]
    physical.need(len(stops)==1,'ONE_REAL_LATE_ROOT_STOP_CUSTODY_REQUIRED')
    state=stops[0].semantic_read['state'];custody=state['custody'];clock=state['clock']
    declaration=custody.get('original_guardian_ownership')
    physical.need(type(declaration)is dict and custody.get('STOP_UNCONFIRMED') is True and
        custody.get('capacity_release') is False,'DECLARED_ORIGINAL_GUARDIAN_STOP_OWNERSHIP')
    fd=declaration.get('fd');guard=run.birth_reads[run.registration['subjects']['actual_GNU']['pid']]
    physical.need(type(fd)is int and fd>2 and fd==custody.get('actual_original_guardian_pidfd') and
        all(declaration.get('holder_birth',{}).get(k)==v for k,v in before.process.birth_dict().items()) and
        all(declaration.get('guardian_birth',{}).get(k)==v for k,v in guard.birth_dict().items()) and
        declaration.get('generation')==[before.birth,guard.birth] and
        declaration.get('lifetime')=='original_kernel120_until_Root_terminal_on_STOP' and
        clock==custody.get('actual_root_clock_before_record') and
        all(clock['actual_guardian_birth'].get(k)==v for k,v in guard.birth_dict().items()) and clock.get('clock_reset') is False and
        clock['original_seconds']==120 and clock['cleanup_seconds']==30 and
        declaration.get('original_final_m')==clock['started_at_monotonic']+120==run.registration['original_final_monotonic'] and
        declaration.get('original_final_u')==clock['started_at_unix']+120==run.registration['original_final_unix'],
        'EXACT_ORIGINAL_GUARDIAN_BIRTH_GENERATION_LIFETIME_CLOCK')
    old={r.fd:r for r in before.all_fd_reads};new={r.fd:r for r in after.all_fd_reads}
    physical.need(fd not in old and set(new)==set(old)|{fd},'ONLY_QUALIFIED_GUARDIAN_RESOURCE_DIFFERENCE')
    actual=new[fd];rows=[r for r in custody['fds'] if r['fd']==fd]
    physical.need(len(rows)==1 and rows[0]['label']=='original_guardian_pidfd' and
        tuple(rows[0]['identity'])==actual.identity==tuple(declaration['identity']) and
        not rows[0]['closed'] and not rows[0]['close_attempted'] and not rows[0]['close_ambiguous'] and
        actual.pid==before.pid and actual.birth==before.birth and actual.link=='anon_inode:[pidfd]' and
        not actual.locks and actual.flags&os.O_CLOEXEC and
        all(r.fd==fd or r.identity!=actual.identity for r in after.all_fd_reads),
        'EXACT_RETAINED_ORIGINAL_GUARDIAN_ROW_AND_NO_ALIAS')
    namespace=lambda p:tuple(tuple(line.split()[1:]) for line in p.status_raw.splitlines() if line.startswith('NSpid:'))
    root_ns=namespace(before.process);guard_ns=namespace(guard)
    physical.need(len(root_ns)==len(guard_ns)==1 and root_ns[0] and
        len(guard_ns[0])>=len(root_ns[0]),'GUARDIAN_PIDFD_NAMESPACE_UNBOUND')
    raw_pid=int(guard_ns[0][len(root_ns[0])-1])
    captures=[e for e in run.kernel if e.stage=='exit' and e.syscall==434 and
        e.pid==before.pid and e.birth==before.birth and lo<e.sequence<hi and
        e.arguments[:2]==(raw_pid,0) and e.result==fd and type(e.fd_after)is physical.FDRead]
    physical.need(len(captures)==1,'ONE_EXACT_RAW_ORIGINAL_GUARDIAN_PIDFD_OPEN')
    acquired=captures[0]
    entries=[e for e in run.kernel if e.stage=='entry' and e.syscall==434 and
        e.pid==before.pid and e.birth==before.birth and lo<e.sequence<acquired.sequence and
        e.arguments==acquired.arguments]
    physical.need(len(entries)==1 and acquired.actual_process_sample is not None and
        namespace(acquired.actual_process_sample['process'])==root_ns,
        'EXACT_ORIGINAL_GUARDIAN_RAW_ENTRY_NAMESPACE')
    physical.untouched((acquired.fd_after,),(actual,))
    targets=[line.split(':',1)[1].strip() for line in actual.info_raw.splitlines() if line.startswith('Pid:')]
    physical.need(targets==[str(guard.pid)] and guard.pid==before.process.ppid,
        'ORIGINAL_GUARDIAN_PIDFD_ACTUAL_TARGET')
    mutations=[e for e in run.kernel if e.pid==before.pid and e.birth==before.birth and
        acquired.sequence<e.sequence<hi and e.stage=='exit' and e.result>=0 and (
        e.syscall==3 and e.arguments[0]==fd or e.syscall==436 and e.arguments[0]<=fd<=e.arguments[1] or
        e.syscall in (33,292) and e.arguments[1]==fd or
        e.syscall==32 and e.arguments[0]==fd or
        e.syscall==72 and e.arguments[0]==fd and e.arguments[1] in (0,2,1030))]
    physical.need(not mutations,'ORIGINAL_GUARDIAN_HANDLE_MUTATION_OR_DUPLICATION')
    return tuple(r for r in after.all_fd_reads if r.fd!=fd)

def _preserved(run,declared_original_guardian=False):
    # Every original known FD is compared physically at the two stopped
    # boundaries, including all original caller FLOCK data. No absence guessed.
    checks=[]
    for pid in run.registration['required_attempt_pids']:
        before=_bounds(run,'attempt_begin',pid);after=_bounds(run,'attempt_end',pid)
        physical.need(len(before)==len(after)==1,'BEFORE_AFTER_ACTUAL_CAPABILITIES_UNBOUND')
        actual_after=after[0].all_fd_reads
        if declared_original_guardian and pid==run.case.root_pid:
            actual_after=_late_original_guardian_difference(run,before[0],after[0])
        physical.untouched(before[0].all_fd_reads,actual_after)
        physical.untouched(before[0].caller_fd_reads,after[0].caller_fd_reads)
        physical.need(before[0].original_slot==after[0].original_slot,
                      'ORIGINAL_CALLER_SLOT_FLOCK_DRIFT')
        physical.need(before[0].protected_reads==after[0].protected_reads,
                      'PROTECTED_WRITER_CONTROLLER_CHANGE')
        checks.extend((before[0].marker_sequence,after[0].marker_sequence))
    return _proof(run,'every_original_capability_and_slot_preserved',True,checks)

def _actual_caps(run):
    # Genuine subject limits, not observer limits or declared expected caps.
    names={'Max address space':'AS','Max cpu time':'CPU','Max file size':'FS','Max core file size':'CORE'}
    root=_bounds(run,'attempt_begin',run.case.root_pid)
    physical.need(len(root)==1,'ACTUAL_ROOT_LIMITS_BOUNDARY_UNBOUND')
    rows=root[0].process.limits_raw.splitlines()
    result={}
    for prefix,name in names.items():
        found=[r[len(prefix):].split() for r in rows if r.startswith(prefix)]
        physical.need(len(found)==1 and len(found[0])>=3,'ACTUAL_ROOT_LIMITS_UNKNOWN')
        physical.need(found[0][0].isdecimal() and found[0][1].isdecimal()
                      and found[0][0]==found[0][1],'ACTUAL_ROOT_LIMITS_NOT_HARD_FINITE')
        result[name]=int(found[0][0])
    return result

def _good_preflight(run):
    good=_bounds(run,'whole_preflight_complete')
    binds=_marks(run,'bind_existing')
    physical.need(good and binds,'GENUINE_PREFLIGHT_AND_BIND_OBSERVATION_UNBOUND')
    for pid in run.registration['required_attempt_pids']:
        local_good=[g for g in good if g.pid==pid]
        local_bind=[b for b in binds if b.pid==pid]
        physical.need(local_good and local_bind and
            min(g.marker_sequence for g in local_good)<min(b.sequence for b in local_bind),
            'WHOLE_PREFLIGHT_AFTER_LOCAL_FIRST_BIND')
    physical.need(all(len(b.phase_inventory['phases'])==23 for b in good),
                  'GENUINE_ALL23_PREFLIGHT_INVENTORY')
    return _proof(run,'all23_preflight_before_first_bind',True,
                  tuple(b.marker_sequence for b in good))

def _full_scope_proof(run):
    # Do not promote an observer binding, scope declaration, single positive or
    # copied historical result into actual current full32/current-gate execution.
    full=run.registration.get('current_whole_mandatory_observation')
    physical.need(type(full)is CurrentMandatoryObservation,
                  'CURRENT_FULL32_C75_C55_H991_FD8_OBSERVER_REDUCERS_UNBOUND')
    full.verify_same_final_bytes_and_original_scope(run)
    return full.raw_ref

class CurrentMandatoryObservation:
    """Intrinsic witness over actual current captured original32/current82 runs.
    It proves execution coverage/clock/source identity, never successful physical
    fit, Source Safety, runtime acceptance or capacity release.
    """
    def __init__(self,actual_runs,raw_ref,source_pins,clock_identity,mandatory_scopes=()):
        self.runs=dict(actual_runs);self.raw_ref=raw_ref
        self.source_pins=source_pins;self.clock_identity=clock_identity
        self.mandatory_scopes=tuple(mandatory_scopes)
    def verify_same_final_bytes_and_original_scope(self,run):
        from qualification_batch import CASES
        from physical_legacy_controls import CONTROLS
        expected={r[0] for r in CASES}|set(CONTROLS)|{r[0] for r in C83}
        physical.need(set(self.runs)==expected and len(expected)==114,
                      'CURRENT_EXACT114_CAUSE_CAPTURE_SCOPE_MISSING')
        physical.need(self.source_pins==run.registration['complete_current_source_pins'] and
            self.clock_identity==run.registration['original_clock_identity'],
            'NO_OLD_SOURCE_OR_CLOCK_COVERAGE_TRANSFER')
        for control_id,observed in self.runs.items():
            for member in capture.validate_current_capture(observed):
                physical.need(member.registration['complete_current_source_pins']==self.source_pins
                    and member.registration['original_clock_identity']==self.clock_identity,
                    'CURRENT_WHOLE_FINAL_SOURCE_OR_CLOCK_DRIFT')
                if control_id.startswith('K120_'):
                    from physical_legacy_reduce import LiveFacts
                    LiveFacts(member).scope(control_id)
                else:physical.need(member.case.control_id==control_id,'CURRENT_CAUSE_CAPTURE_SUBSTITUTION')
        from physical_mandatory_producer import validate_mandatory_scope
        physical.need({r.scope for r in self.mandatory_scopes}=={'H991','FD8','C45_MEMORY18','C45_ACTUAL32_CONSUMER','C59_28'},
                      'CURRENT_H991_FD8_C45_C59_28_EXECUTION_WITNESS_UNBOUND')
        for scope in self.mandatory_scopes:validate_mandatory_scope(scope,run.registration)
        physical.need(self.raw_ref is not None,'CURRENT_WHOLE_RAW_WITNESS_NOT_SEALED')
        return self.raw_ref

class C83Observation(Mapping):
    def __init__(self,run):
        self.run=capture.validate_raw_run(run)
        self.fields={}
        self._derive()
    def __iter__(self):return iter(self.fields)
    def __len__(self):return len(self.fields)
    def __getitem__(self,name):
        physical.need(name in self.fields,'PHYSICAL_FIELD_UNBOUND:'+name)
        value=self.fields[name]
        return value() if callable(value) else value

    def _derive(self):
        run=self.run;case=run.case;reg=run.registration;delta=case.declared_delta
        prefix=case.kind=='zero';late=case.kind=='late'
        boundaries=(['fixed_Root_entry'] if case.mutation in
            ('gnu_holder','empty_holder','source','slot','main_role') else
            ['original_caller_prepare','fixed_Root_entry'])
        caller_m=_marks(run,'caller_prepare');root_m=_marks(run,'Root_entry')
        actual_boundaries=(['original_caller_prepare'] if caller_m else [])+ \
                          (['fixed_Root_entry'] if root_m else [])
        # A fixture cannot request two boundaries and receive credit for one.
        if prefix:
            physical.need(actual_boundaries==boundaries,'EXACT_BOUNDARIES_NOT_OBSERVED')
            required=reg['required_attempt_pids']
            physical.need(required==({case.root_pid} if len(boundaries)==1 else
                {case.root_pid,case.caller_pid}),'EXACT_ATTEMPT_PID_SCOPE')
            if case.mutation in ('uid','gid','mode','identity','path'):
                for pid in required:
                    begins=_marks(run,'parent_row_begin',pid)
                    valid=_marks(run,'parent_row_valid',pid)
                    bad_fd=delta['last_bad']['fd']
                    physical.need(begins and begins[-1].values[0]==bad_fd and
                        begins[-1].values[1]+1==begins[-1].values[2],
                        'ACTUAL_BAD_PARENT_ROW_NOT_LAST')
                    prior=[v for v in valid if v.kernel_sequence<begins[-1].kernel_sequence]
                    physical.need(len({v.values[0] for v in prior})>=3,
                        'THREE_ACTUALLY_VALID_DISTINCT_ROWS_NOT_OBSERVED')
            else:
                for pid in required:
                    prior=_marks(run,'phase_inventory_valid',pid)
                    physical.need(len(prior)>=3,'THREE_ACTUALLY_VALID_PHASE_ROWS_NOT_OBSERVED')
        raw=run.raw_ref
        caps_before=_bounds(run,'attempt_begin')
        protected_before=tuple(b.protected_reads for b in caps_before)
        protected_after=tuple(b.protected_reads for b in _bounds(run,'attempt_end'))
        f=self.fields
        f.update(control_id=case.control_id,
            independent_final_source_acceptance_ref=reg['source_safety_ref'],
            specific_current_action_admission_ref=reg['specific_action_ref'],
            independent_raw_root_trace_ref=raw,
            exact_current_complete_source_pins_ref=reg['complete_source_pins_ref'],
            simulated=False,unknown_as_zero=False,clock_reset=False,
            original_seconds=reg['original_seconds'],cleanup_seconds=reg['cleanup_seconds'],
            actual_hard_caps=lambda:_actual_caps(run),
            original_protected_writers_and_controllers_unchanged_raw_ref=
                lambda:_proof(run,'protected_writers_and_controllers',protected_before==protected_after),
            fallback_or_retry_count=0,capacity_release=False,
            expected_refusal=case.expected_refusal,
            actual_code_matches_expected_exact=lambda:_code(run)==case.expected_refusal,
            unrelated_error_as_pass=False,
            caller_context_holder_mutations=_count_marks(run,'holder_mutation'),
            clock_catalog_and_phase_record_counts=[_count_marks(run,n) for n in ('clock_once','catalog','record')],
            counter_origin='before_FIRST_caller_or_Root_bind',
            counter_origin_raw_ref=lambda:_proof(run,'one_raw_origin_before_first_bind',
                all(len(_marks(run,'attempt_begin',pid))==1 for pid in reg['required_attempt_pids'])),
            counter_reset_count=run.counter_resets,
            observation_boundaries=actual_boundaries,
            holder_clock_catalog_counters_raw_ref=lambda:_proof(run,'actual_hook_counts',True),
            all_original_capabilities_and_slot_FLOCK_after_refusal_raw_ref=lambda:_preserved(run),
            actual_original_caller_slot_and_caps_unchanged_raw_ref=lambda:_preserved(run),
            TOTAL_prefix_effect_counts=lambda:prefix_counts(run),
            unknown_original_FD_adopted_or_closed=lambda:len(_real_close_calls(run)),
            bad_row_position='LAST',
            earlier_distinct_valid_rows=len(delta.get('earlier_valid',())) if 'earlier_valid' in delta else len(case.phase_order)-1)
        physical.need(case.phase_order[-1]==case.last_phase,'ACTUAL_BAD_DECLARATION_NOT_LAST')
        if case.mutation in ('uid','gid','mode','identity','path'):
            f['declared_parent_field_changed']=delta['field']
            root_bound=_bounds(run,'Root_entry',case.root_pid)
            caller_bound=_bounds(run,'caller_prepare',case.caller_pid)
            def duplicate_proof():
                physical.need(delta['good']['fd']==delta['last_bad']['fd'],'SAME_FD_CONTRADICTION_NOT_PRESENT')
                physical.need(delta['good']!=delta['last_bad'],'CONTRADICTORY_DECLARATION_NOT_PRESENT')
                physical.need(len({r['fd'] for r in delta['earlier_valid']})>=3,'EARLIER_DISTINCT_VALID_ROWS_MISSING')
                # The LAST duplicate exists in both actual attempted plans.
                for caps in (case.caller_caps,case.root_caps):
                    physical.need(caps[case.last_phase]['auxiliary_parent_capabilities'][-1]==delta['last_bad'],

    def _distinct_slot_read(self):
        run=self.run;delta=run.case.declared_delta;declared=delta['actual_alternate']
        physical.need(type(declared)is physical.FDRead and declared.pid==run.case.caller_pid and
            declared.birth==run.known_subjects[run.case.caller_pid][0] and
            declared.fd==delta['supplied_slot_fd']!=delta['original_slot_fd'],
            'EXACT_TYPED_ALTERNATE_SLOT_DECLARATION')
        bounds=_bounds(run,'attempt_begin',run.case.root_pid)
        physical.need(len(bounds)==1,'EXACT_ROOT_ALTERNATE_SLOT_BOUNDARY')
        reads=[row for row in bounds[0].caller_fd_reads if row.fd==declared.fd and
            row.pid==declared.pid and row.birth==declared.birth]
        physical.need(len(reads)==1 and reads[0].identity==declared.identity and
            reads[0].policy()==declared.policy(),'ALTERNATE_SLOT_INDEPENDENT_KERNEL_READ')
        return True

    def _parent_once(self,event):
        run=self.run;pid=run.case.caller_pid
        parents={c['fd'] for cap in run.case.caller_caps.values()
                 for c in (cap['group_parent_capability'],
                    *cap['private_directory_capabilities'].values(),
                    *cap.get('auxiliary_parent_capabilities',()))}
        if event=='bind_existing':
            counts={fd:sum(m.values[0]==fd for m in _marks(run,event,pid)) for fd in parents}
        else:
            counts={fd:sum(e.pid==pid and e.arguments[0]==fd for e in _syscall_entries(run,3)) for fd in parents}
        physical.need(counts and len(set(counts.values()))==1,'PARENT_ONCE_COUNTS_NOT_UNIFORM')
        return next(iter(counts.values()))
    def _extra_inventory(self):
        run=self.run;extras=run.case.declared_delta['extra_parents']
        good=_bounds(run,'whole_preflight_complete')
        physical.need(good,'EXTRA_PARENT_GENUINE_PREFLIGHT_MISSING')
        for b in good:
            byfd={r.fd:r for r in b.all_fd_reads}
            for cap in extras:
                row=byfd[cap['fd']]
                physical.need(row.identity[:2]==tuple(cap['identity']) and
                    row.policy()==cap['policy'] and row.link==cap['path'],'EXTRA_PARENT_ACTUAL_DRIFT')
        return _proof(run,'all_extra_original_parent_kernel_reads',True)
    def _early_extra_closes(self):
        run=self.run;extras={r['fd'] for r in run.case.declared_delta['extra_parents']}
        terminal=_marks(run,'original_caller_terminal',run.case.caller_pid)
        physical.need(len(terminal)==1,'ACTUAL_CALLER_TERMINAL_MISSING')
        return sum(e.pid==run.case.caller_pid and e.stage=='entry' and e.syscall==3
                   and e.arguments[0] in extras and e.sequence<terminal[0].kernel_sequence for e in run.kernel)
    def _distinct_parent_proof(self):
        run=self.run;a,b=run.case.resources['distinct_parent_pair']
        physical.need(a!=b,'DISTINCT_PARENT_FDS_REQUIRED')
        before=_bounds(run,'attempt_begin',run.case.caller_pid)
        rows={r.fd:r for r in before[0].all_fd_reads}
        physical.need(rows[a].identity==rows[b].identity and rows[a].policy()==rows[b].policy()
                      and rows[a].link==rows[b].link,'DISTINCT_PARENT_ACTUAL_IDENTITY_POLICY_PATH')
        closes=[e for e in _syscall_entries(run,3) if e.pid==run.case.caller_pid]
        physical.need(sum(e.arguments[0]==a for e in closes)==
                      sum(e.arguments[0]==b for e in closes)==1,'BOTH_ORIGINAL_PARENTS_ONCE_CLOSE_UNPROVEN')
        return _proof(run,'both_distinct_original_parent_capabilities_closed_once',True)
    def _outer_holder(self):

    def _other22(self):
