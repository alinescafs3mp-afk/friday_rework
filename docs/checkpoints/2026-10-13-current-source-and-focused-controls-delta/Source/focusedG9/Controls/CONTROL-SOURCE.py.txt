"""Focused executable Source controls; never run by CLI4 author.

FD controls use real OS descriptors. Connected controls require Root's real
qualified original HermesCLI frontend/caller producer, which is NOT_DONE in
this package. No fake AppSession, worker-completion Boolean or old PASS is a
substitute. These definitions do not replace the canonical original150.
"""
import io
import os
import unittest
from hermes_cli.friday_cli_fd import OriginalDescriptor, DescriptorUnconfirmed

class DescriptorControls(unittest.TestCase):
    def pipe_stream(self):
        read_fd, write_fd = os.pipe()
        stream = io.TextIOWrapper(io.FileIO(write_fd, "wb", closefd=True),
                                  encoding="utf-8", write_through=True)
        os.set_blocking(read_fd, False)
        self.addCleanup(os.close, read_fd)
        self.addCleanup(stream.close)
        return read_fd, stream

    def pin(self, stream):
        pin = OriginalDescriptor(stream)
        self.addCleanup(pin.close)
        return pin

    def assert_sticky_refusal_after_restore(self, pin):
        # Restoring the numeric FD cannot reauthorize an observed failed pin.
        # No fresh frontend/native generation is minted by these controls.
        observed_failure = pin.failure
        self.assertEqual(observed_failure, "DescriptorUnconfirmed")
        self.assertFalse(pin.readers)
        for operation in (pin.validate, pin.reopen):
            with self.assertRaisesRegex(
                    DescriptorUnconfirmed,
                    "^original_descriptor_prior_observed_failure$"):
                operation()
            self.assertEqual(pin.failure, observed_failure)
            self.assertFalse(pin.readers)

    def test_K01_original_real_pipe_positive_and_flags(self):
        import fcntl
        reader, stream = self.pipe_stream()
        flags = fcntl.fcntl(stream.fileno(), fcntl.F_GETFL)
        pin = self.pin(stream)
        fd = pin.reopen()
        try:
            self.assertEqual(os.write(fd, b'{"original":true}\n'), 18)
            pin.validate()
        finally:
            pin.release(fd)
        self.assertEqual(os.read(reader, 4096), b'{"original":true}\n')
        self.assertEqual(fcntl.fcntl(stream.fileno(), fcntl.F_GETFL), flags)
        self.assertFalse(pin.readers)

    def test_K02_swap_before_first_acquisition_no_foreign_bytes(self):
        reader, stream = self.pipe_stream()
        foreign_reader, foreign = self.pipe_stream()
        pin = self.pin(stream)
        saved = os.dup(stream.fileno())
        try:
            os.dup2(foreign.fileno(), stream.fileno())
            with self.assertRaises(DescriptorUnconfirmed):
                pin.reopen()
            with self.assertRaises(BlockingIOError):
                os.read(foreign_reader, 4096)
        finally:
            os.dup2(saved, stream.fileno())
            os.close(saved)
        self.assert_sticky_refusal_after_restore(pin)
        with self.assertRaises(BlockingIOError):
            os.read(reader, 4096)
        with self.assertRaises(BlockingIOError):
            os.read(foreign_reader, 4096)

    def test_K03_swap_between_completed_operations(self):
        reader, stream = self.pipe_stream()
        foreign_reader, foreign = self.pipe_stream()
        pin = self.pin(stream)
        fd = pin.reopen()
        try:
            os.write(fd, b"first\n")
        finally:
            pin.release(fd)
        self.assertEqual(os.read(reader, 4096), b"first\n")
        saved = os.dup(stream.fileno())
        try:
            os.dup2(foreign.fileno(), stream.fileno())
            with self.assertRaises(DescriptorUnconfirmed):
                pin.validate()
            with self.assertRaises(BlockingIOError):
                os.read(foreign_reader, 4096)
        finally:
            os.dup2(saved, stream.fileno())
            os.close(saved)

    def test_K04_same_inode_reopen_is_not_original_description(self):
        reader, stream = self.pipe_stream()
        pin = self.pin(stream)
        saved = os.dup(stream.fileno())
        reopened = os.open("/proc/self/fd/" + str(pin.anchor), os.O_WRONLY | os.O_CLOEXEC)
        try:
            os.dup2(reopened, stream.fileno())
            # fstat tuple alone still matches. Open-description query must not.
            self.assertEqual(os.fstat(stream.fileno()).st_ino, os.fstat(pin.anchor).st_ino)
            with self.assertRaises(DescriptorUnconfirmed):
                pin.validate()
        finally:
            os.dup2(saved, stream.fileno())
            os.close(saved)
            os.close(reopened)
        self.assert_sticky_refusal_after_restore(pin)
        with self.assertRaises(BlockingIOError):
            os.read(reader, 4096)

    def test_K05_anchor_routes_during_transient_numeric_swap(self):
        reader, stream = self.pipe_stream()
        foreign_reader, foreign = self.pipe_stream()
        pin = self.pin(stream)
        fd = pin.reopen()
        saved = os.dup(stream.fileno())
        try:
            os.dup2(foreign.fileno(), stream.fileno())
            # Even in a validate/write race, the actual descriptor writes only
            # to the admitted original object. This is not a claim to observe
            # every invisible swap-and-restore transition.
            os.write(fd, b"anchored\n")
            self.assertEqual(os.read(reader, 4096), b"anchored\n")
            with self.assertRaises(BlockingIOError):
                os.read(foreign_reader, 4096)
            with self.assertRaises(DescriptorUnconfirmed):
                pin.validate()
        finally:
            os.dup2(saved, stream.fileno())
            os.close(saved)
            pin.release(fd)

    def test_K06_closed_lifecycle_cannot_remint_from_wrapper(self):
        reader, stream = self.pipe_stream()
        pin = self.pin(stream)
        pin.close()
        with self.assertRaises(DescriptorUnconfirmed):
            pin.reopen()
        fresh = self.pin(stream)
        fd = fresh.reopen()
        fresh.release(fd)
        self.assertFalse(fresh.readers)

    def test_K07_live_private_description_prevents_anchor_close(self):
        reader, stream = self.pipe_stream()
        pin = self.pin(stream)
        fd = pin.reopen()
        try:
            with self.assertRaises(DescriptorUnconfirmed):
                pin.close()
            self.assertFalse(pin.closed)
        finally:
            pin.release(fd)
        pin.close()
        self.assertTrue(pin.closed)

    def test_K08_exact_append_memory_regression(self):
        stream = io.StringIO()
        self.addCleanup(stream.close)
        pin = self.pin(stream)
        pin.validate()
        stream.write("normal\n")
        pin.validate()
        pin.close()

def require_original_fixture(live):
    """Concrete original instances and actual native witnesses, not a mock."""
    from cli import HermesCLI
    from prompt_toolkit.application import Application
    from prompt_toolkit.patch_stdout import StdoutProxy
    from hermes_cli.friday_cli_output import ClassicConsoleOutput
    if (type(live.cli) is not HermesCLI or type(live.app) is not Application
            or type(live.proxy) is not StdoutProxy
            or type(live.owner) is not ClassicConsoleOutput
            or live.cli._friday_cli_output is not live.owner
            or live.cli._app is not live.app
            or live.proxy.app_session is not live.session
            or live.session.app is not live.app or not live.app.is_running
            or live.proxy._finite_state is not live.owner.frontend):
        raise AssertionError("not_genuine_original_native_frontend")
    live.qualification.require_exact_loaded_source_and_guard()
    return live

def actual_connected_control(live, case):
    """Executable caller/control coupling after genuine factory qualification.

    The immutable factory and its significant adverse controls are still an
    UNBOUND external dependency. Each call below requires raw actual entry,
    physical bytes, whole native tail and exact known owned reap observations.
    """
    live = require_original_fixture(live)
    before = live.observe_native_frontend()
    history = live.read_original_delivery_history()
    live.arm_actual_fault(case)
    result = live.invoke_original_result_tool_executor()
    observation = live.observe_original_caller_return()
    after = live.observe_native_frontend()
    live.check_recorded_bytes_and_boundaries()
    if observation.elapsed is None or observation.elapsed > 30.0:
        raise AssertionError("original_whole30_not_met")
    if observation.missing_tail or observation.unknown_owned_births:
        raise AssertionError("native_tail_or_reap_UNCONFIRMED")
    if before.session is not after.session or before.app is not after.app:
        raise AssertionError("native_AppSession_or_app_replaced")
    live.require_original_stop_and_deadline_preserved(observation)
    live.require_no_implicit_resend(history)
    if case in {"normal_prompt", "queued_stdout", "cursor_input", "two_distinct_records"}:
        live.require_exact_intact_records(result)
        live.require_actual_physical_repaint_and_proxy_barrier()
        live.require_current_durable_delivery_after_original_native_tail()
    elif case in {"partial_then_other_artifact", "second_cancel_cleanup", "metadata_flush_late",
                  "UI_tail_late", "proxy_close_error", "staging_IO_stall", "activity_join_late"}:
        live.require_exact_UNKNOWN_no_replay(result)
        if case == "partial_then_other_artifact":
            live.invoke_other_owned_artifact_same_frontend()
            live.require_new_document_refused_and_exact_old_prefix_retained()
    elif case in {"FD_before_acquire", "FD_between_operations", "FD_same_inode_reopen",
                  "FD_between_chunks", "foreign_AppSession"}:
        live.require_zero_foreign_sink_bytes()
    else:
        raise AssertionError("unknown_focused_control")
    live.require_all_normal_or_adverse_known_owned_participants_reaped()
    return observation
