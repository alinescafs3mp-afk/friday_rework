def strict_number(value, name, low, high):
    return number(value, name, low, high)

def resolve_attribute(row):
    # Entire row has already passed pure exact JSON validation.
    module = sys.modules.get(text(row["module"], "module", 256))
    if module is None:
        raise Unbound("qualified_module_not_loaded")
    value = module
    for part in text(row["attribute"], "attribute", 1024).split("."):
        if not part.isidentifier() or part.startswith("__"):
            raise Unbound("qualified_attribute_shape")
        value = getattr(value, part)
    return value

def nested_code(function, qualname):
    function = function.__func__ if type(function) is types.MethodType else function
    if type(function) is not types.FunctionType:
        raise Unbound("site_not_original_python_function")
    pending, found = [function.__code__], []
    while pending:
        code = pending.pop()
        if code.co_qualname == qualname:
            found.append(code)
        pending.extend(x for x in code.co_consts if type(x) is types.CodeType)
    if len(found) != 1:
        raise Unbound("exact_nested_site_ambiguous")
    return found[0]

@dataclass(frozen=True)
class FrontendFunctionIdentity:
    """Exact original callable, decorator cells and body; code is not custody."""
    function: object
    body: object
    factory: object | None
    activity: object | None
    activity_target: object | None
    cell: object | None
    activity_cell: object | None
    row: dict
    codes: tuple
    globals: object
    body_globals: object
    recovery: tuple | None = None

    def verify(self):
        function,body = self.function,self.body
        if (resolve_attribute(self.row) is not function or function.__code__ is not self.codes[0]
                or function.__globals__ is not self.globals or body.__code__ is not self.codes[1]
                or body.__globals__ is not self.body_globals):
            raise Unbound("original_frontend_function_body_identity_lost")
        if self.factory is not None:
            spec = self.row["decorator"]
            if (resolve_attribute(spec["factory"]) is not self.factory
                    or resolve_attribute(spec["activity"]) is not self.activity
                    or self.factory.__code__ is not self.codes[2] or self.activity.__code__ is not self.codes[3]
                    or self.activity_target.__code__ is not self.codes[4]
                    or self.factory.__globals__ is not self.globals or self.globals.get("cli_activity") is not self.activity
                    or function.__code__.co_freevars != ("method",) or type(function.__closure__) is not tuple or len(function.__closure__) != 1 or function.__closure__[0] is not self.cell
                    or self.cell.cell_contents is not body or function.__wrapped__ is not body
                    or self.activity.__code__.co_freevars != ("func",) or type(self.activity.__closure__) is not tuple or len(self.activity.__closure__) != 1 or self.activity.__closure__[0] is not self.activity_cell
                    or self.activity_cell.cell_contents is not self.activity_target
                    or self.activity.__wrapped__ is not self.activity_target
                    or nested_code(self.factory,function.__code__.co_qualname) is not function.__code__):
                raise Unbound("original_cli_bound_and_activity_closure_identity_lost")
        if self.recovery is not None:
            function,row,code,globals,body_span,call_span = self.recovery
            if (resolve_attribute(row) is not function or function.__code__ is not code
                    or function.__globals__ is not globals):
                raise Unbound("original_recovery_helper_callable_identity_lost")
        return self

    def entry_matches(self, frame, cli):
        self.verify()
        argument = "self" if self.factory is not None else "cli"
        if (frame.f_code is not self.codes[0] or frame.f_globals is not self.globals
                or frame.f_locals.get(argument) is not cli):
            return False
        return self.factory is None or frame.f_locals.get("method") is self.body

    def body_matches(self, frame, cli, *, entry_frame):
        # The raw invocation belongs to the one prepared dispatch, never to
        # another wrapper that shares the same code, globals and closure.
        if entry_frame is None or not self.entry_matches(entry_frame,cli):
            return False
        if self.factory is None:
            return frame is entry_frame
        self.verify()
        return (frame.f_code is self.codes[1] and frame.f_globals is self.body_globals
            and frame.f_locals.get("self") is cli and frame.f_back is entry_frame)

    def site_matches(self, frame, cli, *, wrapper, entry_frame, body_frame):
        if entry_frame is None:
            return False
        if wrapper:
            return frame is entry_frame and self.entry_matches(frame,cli)
        if body_frame is None or not self.body_matches(body_frame,cli,entry_frame=entry_frame):
            return False
        if frame is body_frame:
            return True
        if frame.f_code is self.codes[1] or frame.f_globals is not self.body_globals:
            return False
        functions = [v for v in body_frame.f_locals.values() if type(v) is types.FunctionType
            and v.__code__ is frame.f_code and v.__globals__ is frame.f_globals
            and all(c.cell_contents is frame.f_locals.get(n)
                for n,c in zip(v.__code__.co_freevars,v.__closure__ or ()))]
        if len(functions) != 1:
            return False
        callback = functions[0]
        if frame.f_code.co_name == "_nonquiet_recover_turn":
            # The original raw body PASSES its actual local callable to the
            # pinned helper. Its immediate parent is that helper, not raw F0.
            if self.recovery is None:
                raise Unbound("original_recovery_helper_body_and_invocation_chain_UNBOUND")
            function,row,code,globals,body_span,call_span = self.recovery
            helper = frame.f_back
            return (helper is not None and helper.f_code is code and helper.f_globals is globals
                and helper.f_back is body_frame and body_span[0] <= body_frame.f_lineno <= body_span[1]
                and call_span[0] <= helper.f_lineno <= call_span[1]
                and body_frame.f_locals.get("recover_failed_kanban_turns") is function
                and body_frame.f_locals.get("_nonquiet_recover_turn") is callback
                and helper.f_locals.get("turn_fn") is callback)
        # Other direct local calls retain the original immediate raw parent;
        # neither copied code nor an older raw/wrapper ancestor is a route.
        return frame.f_back is body_frame


def source_call_span(qualification, function, row, name):
    """One exact call statement in the already qualified immutable body."""
    qualification.real_function(function,row)
    data,_ = qualification.exact_file(reference(row["source"],"invocation_source"))
    tree = ast.parse(data,filename=row["filename"])
    nodes = [n for n in ast.walk(tree) if isinstance(n,(ast.FunctionDef,ast.AsyncFunctionDef))
        and n.name == function.__code__.co_name
        and min([n.lineno]+[d.lineno for d in n.decorator_list]) <= function.__code__.co_firstlineno <= n.lineno]
    if len(nodes) != 1:
        raise Unbound("original_invocation_body_AST_UNBOUND")
    calls = [n for n in ast.walk(nodes[0]) if isinstance(n,ast.Call)
        and (isinstance(n.func,ast.Name) and n.func.id == name
             or isinstance(n.func,ast.Attribute) and n.func.attr == name)]
    if len(calls) != 1:
        raise Unbound("original_single_invocation_callsite_AST_UNBOUND:"+name)
    return calls[0].lineno,calls[0].end_lineno


@dataclass(frozen=True)
class ContextManagerFunctionIdentity:
    function: object
    body: object
    factory: object
    enter: object
    constructor: object
    cell: object
    row: dict
    codes: tuple
    globals: object
    body_globals: object
    root_span: tuple
    constructor_span: tuple
    constructor_name: str
    class_key: str
    root_lookup: str

    def verify(self):
        spec = self.row["contextmanager"]
        if (resolve_attribute(self.row) is not self.function
                or resolve_attribute(spec["factory"]) is not self.factory
                or self.function.__code__ is not self.codes[0] or self.body.__code__ is not self.codes[1]
                or self.factory.__code__ is not self.codes[2] or self.enter.__code__ is not self.codes[3]
                or self.constructor.__code__ is not self.codes[4]
                or self.function.__globals__ is not self.globals or self.body.__globals__ is not self.body_globals
                or self.factory.__globals__ is not self.globals or self.enter.__globals__ is not self.globals
                or self.constructor.__globals__ is not self.globals
                or self.function.__code__.co_freevars != ("func",)
                or type(self.function.__closure__) is not tuple or len(self.function.__closure__) != 1
                or self.function.__closure__[0] is not self.cell or self.cell.cell_contents is not self.body
                or self.function.__wrapped__ is not self.body
                or nested_code(self.factory,self.function.__code__.co_qualname) is not self.codes[0]
                or self.globals.get("_GeneratorContextManager") is not _GeneratorContextManager
                or _GeneratorContextManager.__init__ is not self.constructor
                or _GeneratorContextManager.__enter__ is not self.enter):
            raise Unbound("original_contextmanager_callable_cells_or_class_lost")
        return self


def qualify_contextmanager_function(qualification,function,row,frontend,*,class_key,constructor_name,root_lookup):
    function = qualification.real_function(function,row)
    spec = row.get("contextmanager")
    if type(spec) is not dict or set(spec) != {"factory","target","enter","manager_constructor"}:
        raise Unbound("original_contextmanager_target_factory_enter_constructor_rows_REQUIRED")
    factory = qualification.real_function(resolve_attribute(spec["factory"]),spec["factory"])
    if (function.__code__.co_freevars != ("func",) or type(function.__closure__) is not tuple
            or len(function.__closure__) != 1):
        raise Unbound("original_contextmanager_closure_shape_UNBOUND")
    body = qualification.real_function(function.__closure__[0].cell_contents,spec["target"])
    # Target and special methods are qualified as the actual retained objects,
    # never through forbidden __wrapped__/__init__/__enter__ traversal.
    enter = qualification.real_function(_GeneratorContextManager.__enter__,spec["enter"])
    constructor = qualification.real_function(_GeneratorContextManager.__init__,spec["manager_constructor"])
    if (factory.__code__.co_qualname != "contextmanager" or factory.__module__ != "contextlib"
            or body.__code__.co_name != function.__name__
            or function.__code__.co_freevars != ("func",) or type(function.__closure__) is not tuple
            or len(function.__closure__) != 1):
        raise Unbound("original_contextmanager_shape_UNBOUND")
    root_row = frontend.row["decorator"]["target"]
    root_span = source_call_span(qualification,frontend.body,root_row,function.__name__)
    constructor_span = source_call_span(qualification,body,spec["target"],constructor_name)
    witness = ContextManagerFunctionIdentity(function,body,factory,enter,constructor,function.__closure__[0],dict(row),
        (function.__code__,body.__code__,factory.__code__,enter.__code__,constructor.__code__),function.__globals__,body.__globals__,
        root_span,constructor_span,constructor_name,class_key,root_lookup)
    return witness.verify()



def qualify_frontend_function(qualification, function, row, *, decorated):
    function = qualification.real_function(function,row)
    if not decorated:
        if getattr(function,"__wrapped__",None) is not None:
            raise Unbound("unqualified_direct_frontend_decorator")
        helper_row = row.get("recovery_helper")
        if (type(helper_row) is not dict or helper_row.get("module") != "agent.kanban_turn_recovery"
                or helper_row.get("attribute") != "recover_failed_kanban_turns"
                or helper_row.get("source",{}).get("sha256") != "c654aed03c88d45ca99ff0376e9e01bf2254033cf0bfb42f7187c68bd28b9e9d"):
            raise Unbound("exact_frozen_original_recovery_helper_row_UNBOUND")
        helper = qualification.real_function(resolve_attribute(helper_row),helper_row)
        recovery = (helper,dict(helper_row),helper.__code__,helper.__globals__,
            source_call_span(qualification,function,row,"recover_failed_kanban_turns"),
            source_call_span(qualification,helper,helper_row,"turn_fn"))
        return FrontendFunctionIdentity(function,function,None,None,None,None,None,dict(row),
            (function.__code__,function.__code__),function.__globals__,function.__globals__,recovery)
    spec = row.get("decorator")
    if type(spec) is not dict or set(spec) != {"factory","activity","activity_target","target"}:
        raise Unbound("exact_original_cli_bound_activity_and_target_rows_REQUIRED")
    factory = qualification.real_function(resolve_attribute(spec["factory"]),spec["factory"])
    activity = qualification.real_function(resolve_attribute(spec["activity"]),spec["activity"])
    if (factory.__name__ != "cli_bound" or activity.__name__ != "cli_activity"
            or function.__code__.co_freevars != ("method",) or type(function.__closure__) is not tuple or len(function.__closure__) != 1
            or activity.__code__.co_freevars != ("func",) or type(activity.__closure__) is not tuple or len(activity.__closure__) != 1):
        raise Unbound("original_frontend_decorator_cell_shape")
    cell,activity_cell = function.__closure__[0],activity.__closure__[0]
    body = qualification.real_function(cell.cell_contents,spec["target"])
    activity_target = qualification.real_function(activity_cell.cell_contents,spec["activity_target"])
    if (function.__wrapped__ is not body or body.__qualname__ != "HermesCLI.run"
            or body.__module__ != function.__module__ or function.__globals__ is not factory.__globals__
            or function.__globals__.get("cli_activity") is not activity
            or activity.__wrapped__ is not activity_target or activity_target.__qualname__ != "cli_activity"
            or activity_target.__globals__ is not factory.__globals__
            or nested_code(factory,function.__code__.co_qualname) is not function.__code__):
        raise Unbound("decorator_does_not_own_original_run_and_activity_bodies")
    witness = FrontendFunctionIdentity(function,body,factory,activity,activity_target,cell,activity_cell,dict(row),
        (function.__code__,body.__code__,factory.__code__,activity.__code__,activity_target.__code__),function.__globals__,body.__globals__)
    return witness.verify()

class NativeRuntime:
    def __setattr__(self, name, value):
        if self.__dict__.get("_sealed",False):
            raise Unbound("runtime_binding_sealed_against_scope_reassignment")
        object.__setattr__(self,name,value)

    def __init__(self, *, grant_reference, authorization_reference, safety_reference,
                 protocol_reference, guardian_reference, expected_scope_sha256, assignment, generation,
                 collection_reference, producer_reference):
        # All externally supplied fields are exact data BEFORE supplied methods,
        # canonical/hash, file readers, qualification or any native effect.
        refs = {name: reference(value, name) for name, value in (
            ("grant", grant_reference), ("authorization", authorization_reference), ("safety", safety_reference),
            ("protocol", protocol_reference), ("guardian", guardian_reference), ("collection", collection_reference), ("producer", producer_reference))}
        text(assignment, "assignment", 256)
        integer(generation, "generation", 1, 10**9)
        digest(expected_scope_sha256, "expected_scope")
        if "SOURCE" in assignment or "PREPARATION" in assignment:
            raise Unbound("Source_assignment_has_no_runtime_grant")
        self._fd_budget = SourceFDAccount(32)
        self._qualification = MeteredQualification(self._fd_budget)
        grant, self.grant_custody = self._qualification.exact_json(refs["grant"])
        grant = pure(grant, "grant")
        required = {"schema", "status", "assignment", "generation", "root_event", "root_authorization", "collection", "producer_source", "product_source",
            "independent_safety", "whole_layout", "caller_protocol", "native_guardian", "host", "start_monotonic", "useful_deadline_monotonic", "final_deadline_monotonic",
            "original_caller_seconds", "effects", "constructor_effects", "resources", "allowed_cases", "isolated_home", "class_rows", "function_rows", "site_rows", "constructor_kwargs"}
        if type(grant) is not dict or set(grant) != required or grant["schema"] != "CLI150-NATIVE-PRODUCER-GRANT-PROPOSAL-2":
            raise Unbound("native_grant_exact_v2_schema_required")
        for key in ("root_authorization", "collection", "producer_source", "product_source", "independent_safety", "whole_layout", "caller_protocol", "native_guardian"):
            reference(grant[key], key)
        if (grant["status"] != "ROOT_AUTHORIZED_NEW_EXACT_NATIVE_EXECUTION" or grant["assignment"] != assignment
                or type(grant["generation"]) is not int or grant["generation"] != generation
                or grant["collection"] != refs["collection"] or grant["producer_source"] != refs["producer"]
                or grant["root_authorization"] != refs["authorization"] or grant["independent_safety"] != refs["safety"]
                or grant["caller_protocol"] != refs["protocol"] or grant["native_guardian"] != refs["guardian"] or type(grant["original_caller_seconds"]) is not int
                or grant["original_caller_seconds"] != 30):
            raise Unbound("native_grant_not_exact_assignment_or_pinned_scope")
        if type(grant["resources"]) is not dict:
            raise Unbound("resources_not_exact_mapping")
        self._fd_budget.narrow(integer(grant["resources"].get("owned_fds"),"owned_fds",8,32))
        text(grant["root_event"], "root_event", 256)
        # Only the two cross-reference backlinks are excluded to avoid recursive
        # hashes; their EXACT independent refs are caller-pinned above and below.
        scope = {k: v for k, v in grant.items() if k not in ("root_authorization", "independent_safety")}
        scope_sha = hashlib.sha256(canonical(scope)).hexdigest()
        if scope_sha != expected_scope_sha256:
            raise Unbound("whole_native_scope_differs_from_Root_pin")
        root, _ = self._qualification.exact_json(refs["authorization"])
        safety, _ = self._qualification.exact_json(refs["safety"])
        root, safety = pure(root, "root"), pure(safety, "safety")
        root_keys = {"schema", "decision", "event", "assignment", "generation", "grant_scope_sha256", "caller_protocol", "independent_safety"}
        if (type(root) is not dict or set(root) != root_keys or root["schema"] != "CLI150-NATIVE-ROOT-AUTHORIZATION-2"
                or root["decision"] != "AUTHORIZE_EXACT_NATIVE_PRODUCER" or root["event"] != grant["root_event"]
                or root["assignment"] != assignment or type(root["generation"]) is not int or root["generation"] != generation
                or root["grant_scope_sha256"] != scope_sha or root["caller_protocol"] != refs["protocol"]
                or root["independent_safety"] != refs["safety"]):
            raise Unbound("fresh_exact_whole_Root_authorization_UNBOUND")
        safety_keys = {"schema", "decision", "reviewer_role", "grant_scope_sha256", "collection", "producer_source", "product_source", "caller_protocol"}
        if (type(safety) is not dict or set(safety) != safety_keys or safety["schema"] != "CLI150-INDEPENDENT-NATIVE-SAFETY-2"
                or safety["decision"] != "ACCEPT_EXACT_NATIVE_PRODUCER_SOURCE" or safety["reviewer_role"] != "INDEPENDENT"
                or safety["grant_scope_sha256"] != scope_sha
                or any(safety[k] != grant[k] for k in ("collection", "producer_source", "product_source", "caller_protocol"))):
            raise Unbound("independent_whole_native_Safety_UNBOUND")
        self.grant, self.scope_sha256 = grant, scope_sha
        self.birth = self._qualification.proc_birth(os.getpid())
        if type(grant["host"]) is not dict or set(grant["host"]) != {"pid", "starttime", "boot_id"}:
            raise Unbound("host_birth_schema")
        integer(grant["host"]["pid"], "host_pid", 1, 10**9)
        integer(grant["host"]["starttime"], "host_starttime", 1, 10**15)
        text(grant["host"]["boot_id"], "boot_id", 64)
        if any(self.birth[k] != grant["host"][k] for k in ("pid", "starttime", "boot_id")):
            raise Unbound("native_grant_host_birth_lost")
        a, b, c = (number(grant[k], k, 0, 1e12) for k in ("start_monotonic", "useful_deadline_monotonic", "final_deadline_monotonic"))
        if not a < b < c or time.monotonic() >= b:
            raise Unbound("original_parent_clock_expired")
        self._original_clock = (a,b,c)
        for key in ("effects", "constructor_effects", "allowed_cases"):
            if type(grant[key]) is not list or not 1 <= len(grant[key]) <= 256 or any(type(x) is not str for x in grant[key]) or len(set(grant[key])) != len(grant[key]):
                raise Unbound(key + "_exact_unique_sequence")
        if not set(grant["effects"]) <= EFFECTS or set(grant["constructor_effects"]) != CONSTRUCTOR_EFFECTS:
            raise Unbound("unsupported_or_incomplete_native_effect_scope")
        guardian,_ = self._qualification.exact_json(refs["guardian"])
        guardian = pure(guardian,"guardian")
        guardian_keys = {"schema","status","assignment","generation","root_event","host","supervisor_birth","source","independent_safety","useful_deadline_monotonic","final_deadline_monotonic"}
        if (type(guardian) is not dict or set(guardian) != guardian_keys or guardian["schema"] != "CLI150-EXACT-NATIVE-GUARDIAN-RECEIPT-1"
                or guardian["status"] != "ROOT_VERIFIED_ARMED_EXACT_NATIVE_GUARDIAN" or guardian["assignment"] != assignment
                or type(guardian["generation"]) is not int or guardian["generation"] != generation
                or guardian["root_event"] != grant["root_event"] or guardian["host"] != grant["host"]
                or guardian["useful_deadline_monotonic"] != b or guardian["final_deadline_monotonic"] != c):
            raise Unbound("fresh_exact_native_guardian_UNBOUND_mandatory_before_effects")
        reference(guardian["source"],"guardian_source")
        guardian_safety,_ = self._qualification.exact_json(reference(guardian["independent_safety"],"guardian_safety"))
        guardian_safety = pure(guardian_safety,"guardian_safety")
        if (type(guardian_safety) is not dict or guardian_safety.get("decision") != "ACCEPT_EXACT_NATIVE_GUARDIAN_CONTRACT"
                or guardian_safety.get("reviewer_role") != "INDEPENDENT" or guardian_safety.get("source") != guardian["source"]):
            raise Unbound("independent_native_guardian_Safety_UNBOUND")
        supervisor = guardian["supervisor_birth"]
        if type(supervisor) is not dict or set(supervisor) != {"pid","starttime","boot_id"}:
            raise Unbound("guardian_birth_shape")
        integer(supervisor["pid"],"guardian_pid",1,10**9)
        integer(supervisor["starttime"],"guardian_starttime",1,10**15)
        text(supervisor["boot_id"],"guardian_boot",64)
        actual_guardian = self._qualification.proc_birth(supervisor["pid"])
        if actual_guardian["state"] not in {"R","S","D","I"} or supervisor["pid"] == self.birth["pid"] or any(actual_guardian[k] != supervisor[k] for k in supervisor):
