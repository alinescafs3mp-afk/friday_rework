def kernel120_record(custody,directory,name,document):
    """Already-owned outside control directory only, bounded O_EXCL evidence."""
    if root_fd_custody._physical_capture is not None:root_fd_custody._physical_capture.mark('record')
    require(type(name)is str and name and '/' not in name and name not in ('.','..'),'fixed component')
    raw=json.dumps(document,sort_keys=True,separators=(',',':'),allow_nan=False).encode()
    require(len(raw)<=65536,'bounded original-clock/control record')
    private=custody['private_directory_capabilities']
    matching=[role for role in ('control','trace') if private[role]['fd']==directory]
    require(len(matching)==1,'record uses exact original admitted control/trace role')
    role=matching[0]
    root_fd_custody.directory_capability(private[role],True)
    root_fd_custody.validate_private_parent(private[role]['path'],private[role+'_parent'])
    q=os.fstat(directory)
    require(stat.S_ISDIR(q.st_mode) and q.st_uid==os.getuid() and q.st_gid==os.getgid()
            and stat.S_IMODE(q.st_mode)==0o700,'private retained record directory before write')
    fd=root_fd_custody.opened(custody,'control_record:'+name,name,
            os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o400,dir_fd=directory)
    try:
        view=memoryview(raw)
        while view:
            n=os.write(fd,view);require(n>0,'bounded record progress');view=view[n:]
        os.fsync(fd)
    except BaseException as error:root_fd_custody.stopped(custody,error);raise
    else:root_fd_custody.closed(custody,fd)
    os.fsync(directory)
    path=str(Path('/proc/self/fd/'+str(directory)).resolve(strict=True)/name)
    return {'path':path,'bytes':len(raw),'sha256':hashlib.sha256(raw).hexdigest()}

class Kernel120Clone3Refusal(OSError):

def kernel120_atomic_named_child(custody,label):
    """Only stock/finalizer fixed roles, atomic CLONE_PIDFD; no fork handle gap."""
    if root_fd_custody._physical_capture is not None:root_fd_custody._physical_capture.mark('first_birth')
    import ctypes,signal,platform
    require(label in ('kernel120_stock','kernel120_post_terminal_helper'),'two fixed existing-owner roles')
    require(platform.machine()=='x86_64','reviewed clone3 ABI, no fallback')
    class CloneArgs(ctypes.Structure):
        _fields_=[(key,ctypes.c_uint64) for key in
            ('flags','pidfd','child_tid','parent_tid','exit_signal','stack','stack_size','tls',
             'set_tid','set_tid_size','cgroup')]
    fd=ctypes.c_int(-1);args=CloneArgs();args.flags=0x1000
    args.pidfd=ctypes.addressof(fd);args.exit_signal=signal.SIGCHLD
    libc=ctypes.CDLL(None,use_errno=True);libc.syscall.restype=ctypes.c_long
    pid=libc.syscall(ctypes.c_long(435),ctypes.byref(args),ctypes.c_size_t(88))
    if pid<0:
        error=ctypes.get_errno()
        custody.setdefault('actual_clone3_refusals',[]).append(
            {'syscall_result':int(pid),'kernel_errno':error,'site':'candidate_clone3_namespace',
             'actual_role':label,'at_m':time.monotonic(),'at_u':time.time(),'retry_count':0})
        raise Kernel120Clone3Refusal(int(pid),error,'candidate_clone3_namespace')
    if pid>0:
        root_fd_custody.own(custody,fd.value,{'kernel120_stock':'stock_pidfd','kernel120_post_terminal_helper':'post_terminal_helper_pidfd'}[label]) # FIRST after successful acquisition.
        custody.setdefault('birth_attempts',[]).append({'pid':int(pid),'pidfd':fd.value,'role':label})
    return int(pid),fd.value

def kernel120_verify_output_inventory(spec,context,custody=None):
    """Actual Root inventory before effects AND before original output closes."""
    output=kernel120_revalidate_output_before_effect(spec)
    require(output.is_absolute() and output.resolve(strict=True)==output,'exact output path')
    q=output.lstat();fd=context['output_fd'];rows=context['output_owned_fds']
    require(type(rows)is dict and 0<len(rows)<=32 and fd in rows
            and all(type(k)is int and k>2 for k in rows),'complete actual Root output inventory')
    require(stat.S_ISDIR(q.st_mode) and stat.S_IMODE(q.st_mode)==0o700
            and q.st_uid==os.getuid() and q.st_gid==os.getgid()
            and [q.st_dev,q.st_ino]==spec['qualification_output_identity'],
            'actual Root output path identity')
    mounts=[r.split() for r in Path('/proc/self/mountinfo').read_text().splitlines()
            if r.split()[4]==str(output)]
    require(len(mounts)==1 and mounts[0][mounts[0].index('-')+1]=='tmpfs'
            and int(mounts[0][0])==spec['qualification_output_mount_id'],'actual Root exact tmpfs mount')
    reserved={spec['existing_owner']['group_fd'],spec['existing_owner']['slot_fd'],
              context['control_dir_fd'],context['trace_dir_fd']}
    if custody is not None:
        reserved.update(r['fd'] for r in custody['fds'] if not r['closed']
                        and r['fd'] not in rows and root_fd_custody.canonical_role(r['label'])!='output')

def kernel120_terminal_failure_hold(context,custody,clock,error):
    """No voluntary normal exit, release, retry or refork after STOP."""
    import select
    if root_fd_custody._physical_capture is not None:root_fd_custody._physical_capture.refusal(error)
    root_fd_custody.stopped(custody,error)
    custody['terminal_failure_hold_entered']={'m':time.monotonic(),'u':time.time()}
    if root_fd_custody._physical_capture is not None:
        root_fd_custody._physical_capture.mark('target_owner_stop')
        root_fd_custody._physical_capture.observe_state('target_owner_stop',{'custody':custody,'clock':clock})
    while True:
        left=min(clock['started_at_monotonic']+120-time.monotonic(),
                 clock['final_deadline_unix']-time.time())
        if left<=0:raise error # abnormal exit; actual independent holder retains its originals.
        select.select([],[],[],left) # unchanged finite GNU original boundary.


def kernel120_wait_original_age(clock,age):
    import select
    require(age in (19,45.001,90.001),'fixed original temporal boundaries')
    while True:
        m=time.monotonic();u=time.time()
        require(m<clock['started_at_monotonic']+120 and u<clock['final_deadline_unix']
                and abs((m-clock['started_at_monotonic'])-(u-clock['started_at_unix']))<=2,
                'unchanged actual dualclock temporal boundary')
        left=max(clock['started_at_monotonic']+age-m,clock['started_at_unix']+age-u)
        if left<=0:return
        select.select([],[],[],left)

        expected['pre_entry_role_birth']=exact_birth(os.getpid())
        expected['controls_module']=modules['kernel120_control_inputs']
        batch.produce_original32_phase_inputs(owner,base,expected,clock,catalog,
            modules['kernel120_control_inputs'],modules['physical_control_producer'])
        # Complete scope preflight is BEFORE any stock/actor/clone/opaque GO.
        plan=batch.make_original32_plan(catalog)
        batch.before_any_batch_GO(expected,plan,clock_pin,
            adapter.original_root_clock(clock,clock['boot_id'],time.monotonic(),time.time()),time.monotonic(),time.time())
        return batch.execute_original32_serial(owner,adapter,base,expected,clock,catalog)
    except BaseException as error:kernel120_terminal_failure_hold(context,custody,clock,error)



def kernel120_validate_private_directories(spec,context):
    """Exact original pre-admitted private parents and scratch handles BEFORE effects."""
    caps=context['private_directory_capabilities']
    require(type(caps)is dict and set(caps)=={'output_parent','control','control_parent','trace','trace_parent'},
            'complete exact private directory capability roles')
    for cap in caps.values():root_fd_custody.directory_capability(cap,True)
    require(caps['output_parent']['identity']==context['output_parent_identity'],
            'same original actual output parent identity')
    root_fd_custody.validate_private_parent(str(Path(spec['record']).parent),caps['output_parent'])
    for role in ('control','trace'):
        cap=caps[role];parent=caps[role+'_parent']
        root_fd_custody.validate_private_parent(cap['path'],parent)
        require(cap['fd']==context[role+'_dir_fd'] and cap['identity']==context[role+'_dir_fd_identity'],
                'same original actual '+role+' capability')
    protected={x['pid'] for x in spec['existing_owner']['protected_writers']}
    require(os.getpid() not in protected,'private directories exclude protected writer role')
    require(len({caps['control']['fd'],caps['trace']['fd'],context['output_fd'],
                 spec['existing_owner']['group_fd'],spec['existing_owner']['slot_fd']})==5,
            'private directory roles disjoint retained handles')
    identities=[tuple(caps[x]['identity']) for x in ('control','trace')]
    require(len(set(identities))==2
            and all(tuple(context['output_owned_fds'][fd][:2]) not in identities
                    for fd in context['output_owned_fds']),
            'private scratch identities disjoint output')
    group=Path(spec['native']['group']);parent=spec['native']['group_parent_capability']
    root_fd_custody.directory_capability(parent,False)
    require(group.parent==Path(parent['path']) and group!=Path('/sys/fs/cgroup')
            and group.is_relative_to('/sys/fs/cgroup'),'exact admitted cgroup parent')
    group_cap={'fd':spec['existing_owner']['group_fd'],'path':str(group),
               'identity':spec['native']['first_birth_group_identity'],
               'policy':spec['native']['group_directory_policy']}
    root_fd_custody.directory_capability(group_cap,False)
    for writer in spec['existing_owner']['protected_writers']:
        c=Path('/proc/'+str(writer['pid'])+'/cgroup').read_text().strip()
        require(c.startswith('0::/'),'private parent protected unified scope')
        p=Path('/sys/fs/cgroup')/c[4:]
        require(group!=p and not group.is_relative_to(p) and not p.is_relative_to(group),
                'private group disjoint protected scope before effects')
    return caps

def kernel120_auxiliary_actual_owner(context,ownership,identity,role):
    """Bind an original caller-owned capability to actual remote inode/birth/role."""
    holder=context['retained_existing_caller'];birth=holder['birth']
    root_fd_custody.preflight_retained_original_caller(holder,context.get('retained_holder_verified'))
    require(existing_root_same_identity(exact_birth(birth['pid']),birth),
            'fresh original caller before auxiliary ownership')
    require(type(ownership)is dict and set(ownership)=={'caller_fd','role','identity'}
            and ownership['role']==role and ownership['identity']==identity,
            'exact auxiliary requested ownership role')
    rows=holder['auxiliary_capabilities']
    matches=[r for r in rows if r['fd']==ownership['caller_fd'] and r['role']==role]
    require(len(matches)==1 and matches[0]['identity']==identity,
            'original caller admitted auxiliary role capability')
    fd=ownership['caller_fd'];require(type(fd)is int and fd>2,'actual original auxiliary handle')
    a=os.stat('/proc/'+str(birth['pid'])+'/fd/'+str(fd))
    require([a.st_dev,a.st_ino,stat.S_IFMT(a.st_mode)]==identity,
            'actual original caller auxiliary capability identity')
    require(existing_root_same_identity(exact_birth(birth['pid']),birth),
            'fresh original caller after auxiliary ownership')
    root_fd_custody.preflight_retained_original_caller(holder,context.get('retained_holder_verified'))

def kernel120_preflight_auxiliary_control(actual,context,clock,custody):
    """Validate EVERY auxiliary row before ANY bind, close, umount or removal."""
    capture=root_fd_custody._physical_capture
    if capture is not None:
        scope=context.get('physical_current_auxiliary_scope',(0,0,0))
        require(type(scope)is tuple and len(scope)==3 and all(type(value)is int and value>=0 for value in scope),
            'exact current auxiliary marker scope')
        capture.mark('auxiliary_consumer',*scope)
    try:
        root_fd_custody.require_custody(custody)
        root_fd_custody.preflight_retained_original_caller(context['retained_existing_caller'],context.get('retained_holder_verified'))
        require(type(actual)is dict,'bounded actual auxiliary control object')
        require(time.monotonic()<clock['started_at_monotonic']+120
                and time.time()<clock['final_deadline_unix'],'original final auxiliary preflight')
        reserved={context['actual_clean_group_fd'],context['actual_existing_slot_fd'],
                  context['control_dir_fd'],context['trace_dir_fd']}
        reserved.update(context['output_owned_fds'])
        reserved.update(context['all_phase_reserved_fds'])
        reserved.update(cap['fd'] for cap in context['private_directory_capabilities'].values())
        for cap in context.get('auxiliary_parent_capabilities',[]):
            root_fd_custody.directory_capability(cap,not root_fd_custody.retained_parent_is_cgroup(cap))
            reserved.add(cap['fd'])
        reserved.add(context['group_parent_capability']['fd'])
        rows=[]
        for kind,key in (('auxiliary_group','auxiliary_groups'),('auxiliary_output','auxiliary_outputs'),
                         ('auxiliary_borrowed','auxiliary_fds')):
            collection=actual.get(key,[])
            require(type(collection)is list and len(collection)<=32,'bounded auxiliary role rows')
            for row in collection:
                require(type(row)is dict,'exact auxiliary row type')
                if kind!='auxiliary_borrowed':
                    require(type(row['action_id'])is str and len(row['action_id'])==36
                            and set(row['action_id'])<=set('0123456789abcdef-'),'finite exact auxiliary action UUID')
                if kind=='auxiliary_output':
                    handles=row['output_owned_fds']
                    require(type(handles)is list and 0<len(handles)<=32
                            and all(type(r)is list and len(r)==2 for r in handles),
                            'exact bounded auxiliary output inventory')
                    inventory={r[0]:r[1] for r in handles}
                    require(len(inventory)==len(handles) and row['output_fd'] in inventory,
                            'unique auxiliary output role FDs')
                    parent=row['parent_capability']
                    root_fd_custody.validate_private_parent(str(Path(row['record']).parent),parent)
                    require(parent['fd'] in reserved,'auxiliary output uses original admitted retained parent')
                    kernel120_auxiliary_actual_owner(context,row['ownership'],
                        [*parent['identity'],stat.S_IFDIR],'original_caller_output_parent')
                    temporary={'output_fd':row['output_fd'],'output_owned_fds':inventory,
                        'control_dir_fd':context['control_dir_fd'],'trace_dir_fd':context['trace_dir_fd'],
                        'output_parent_identity':parent['identity']}
                    aux={'record':row['record'],'qualification_output_identity':row['output_identity'],
                         'qualification_output_mount_id':row['output_mount_id'],
                         'qualification_output_parent_capability':parent,
                         'existing_owner':{'group_fd':context['actual_clean_group_fd'],
                                          'slot_fd':context['actual_existing_slot_fd']}}
                    kernel120_verify_output_inventory(aux,temporary,custody)
                    require(Path(row['record']).parent.name=='friday-native-'+row['action_id'],
                            'actual action-private auxiliary output path')
                    for fd,identity in inventory.items():rows.append((fd,identity,kind,row))
                else:
                    fd=row['fd'];q=os.fstat(fd);identity=[q.st_dev,q.st_ino,stat.S_IFMT(q.st_mode)]
                    require(identity==row['identity'] if kind=='auxiliary_borrowed'
                            else identity[:2]==row['identity'],'exact auxiliary actual identity')
                    kernel120_auxiliary_actual_owner(context,row['ownership'],identity,kind)
                    if kind=='auxiliary_group':
                        group=Path(row['path']);parent=row['parent_capability']
                        root_fd_custody.directory_capability(parent,False)
                        root_fd_custody.directory_capability({'fd':fd,'path':str(group),
                            'identity':row['identity'],'policy':row['directory_policy']},False)
                        require(group.parent==Path(parent['path'])
                                and group.name=='friday-native-'+row['action_id']
                                and group.is_relative_to('/sys/fs/cgroup')
                                and group!=Path('/sys/fs/cgroup'),'actual action-private auxiliary group capability')
                        for writer in context['protected_writer_groups']:
                            protected=Path(writer['path'])
                            require(group!=protected and not group.is_relative_to(protected)
                                    and not protected.is_relative_to(group)
                                    and identity[:2]!=writer['identity'],
                                    'never remove protected/shared auxiliary scope or ancestor')
                        require((group/'pids.current').read_text().strip()=='0'
                                and 'populated 0' in (group/'cgroup.events').read_text()
                                and not (group/'cgroup.procs').read_text().strip(),'actual auxiliary kernel0')
                    else:
                        policy=row['owner_policy']
                        require(type(policy)is dict and set(policy)=={'uid','gid','mode'}
                                and all(type(v)is int and v>=0 for v in policy.values())
                                and (q.st_uid,q.st_gid,stat.S_IMODE(q.st_mode))
                                ==(policy['uid'],policy['gid'],policy['mode']),
                                'auxiliary borrowed actual admitted UID/mode')
                        require(row['role']==kind and (stat.S_ISDIR(q.st_mode) or stat.S_ISREG(q.st_mode)),
                                'borrowed auxiliary explicit directory/file role; never pidfd/GO')
                    rows.append((fd,identity,kind,row))
        require(len(rows)<=64 and len({r[0] for r in rows})==len(rows),'all auxiliary role FDs unique')
        # Semantic removable-node uniqueness also holds for direct cleanup callers.
        removable_ids=set();removable_paths=set();removable_mounts=set();removable_callers=set()
        seen_objects=set()
        for fd,identity,kind,row in rows:
            if kind=='auxiliary_borrowed':continue
            marker=id(row)
            if marker in seen_objects:continue # one output inventory, multiple FDs
            seen_objects.add(marker)
            key=tuple(row['identity'] if kind=='auxiliary_group' else row['output_identity'])
            path=row['path'] if kind=='auxiliary_group' else str(Path(row['record']).parent)
            mount=None if kind=='auxiliary_group' else row['output_mount_id']
            caller=row['ownership']['caller_fd'] if kind=='auxiliary_group' else None
            require(key not in removable_ids and path not in removable_paths
                    and (mount is None or mount not in removable_mounts)
                    and (caller is None or caller not in removable_callers),
                    'auxiliary semantic owned duplicate ZERO total prefix effects')
            removable_ids.add(key);removable_paths.add(path);removable_mounts.add(mount)
            if caller is not None:removable_callers.add(caller)
        for fd,identity,kind,row in rows:
            if kind=='auxiliary_borrowed':
                require(tuple(identity[:2]) not in removable_ids,
                        'borrowed inode never aliases owned removal')
            if custody.get('phase_order') is not None:
                planned=[r for r in custody['fds'] if r['fd']==fd and not r['closed']]
                require(len(planned)==1 and planned[0].get('planned')
                        and planned[0]['lifetime']=='phase'
                        and planned[0]['phase']==custody['active_phase']
                        and root_fd_custody.canonical_role(planned[0]['label'])==kind,
                        'actual auxiliary exact CURRENT planned lifetime; future is reserved')
        reserved.update(r['fd'] for r in custody['fds'] if not r['closed']
                        and root_fd_custody.canonical_role(r['label']) not in
                        ('auxiliary_group','auxiliary_output','auxiliary_borrowed'))
        require(not {r[0] for r in rows}&reserved,
                'auxiliary disjoint slot/group/control/trace/guardian/pidfd/GO/output/caller')
        reserved_ids={tuple(x) for x in context['all_phase_reserved_identities'].values()}
        for fd in reserved:
            try:q=os.fstat(fd)
            except OSError as error:
                closed_rows=[r for r in custody['fds'] if r['fd']==fd and r['closed']]
                require(error.errno==9 and closed_rows,'unknown reserved FD absence is not safe closure')
            else:reserved_ids.add((q.st_dev,q.st_ino,stat.S_IFMT(q.st_mode)))
        require(all(tuple(identity) not in reserved_ids for _,identity,_,_ in rows),
                'auxiliary actual inode disjoint reserved capabilities')
        for fd,identity,kind,row in rows:
            root_fd_custody.preflight_existing_role(custody,fd,kind)
        return rows
    finally:
        if capture is not None and scope[2]==1:
            capture.mark('auxiliary_consumer',scope[0],scope[1],2)



def kernel120_remove_owned_group(spec,fd,parent_cap=None,identity=None,path=None,policy=None):
    """Original caller's owned empty node; parent-FD removal, actual retained deletion."""
    group=Path(spec['native']['group'] if path is None else path)
    identity=spec['native']['first_birth_group_identity'] if identity is None else identity
    parent_cap=spec['native']['group_parent_capability'] if parent_cap is None else parent_cap
    policy=spec['native']['group_directory_policy'] if policy is None else policy
    root_fd_custody.directory_capability(parent_cap,False)
    root_fd_custody.directory_capability({'fd':fd,'path':str(group),'identity':identity,'policy':policy},False)
    require(group.parent==Path(parent_cap['path']),'same actual owned group parent before rmdir')
    require((group/'pids.current').read_text().strip()=='0'
            and 'populated 0' in (group/'cgroup.events').read_text()
            and not (group/'cgroup.procs').read_text().strip(),'fresh same owned empty group before rmdir')
    os.rmdir(group.name,dir_fd=parent_cap['fd'])
    try:os.stat(group.name,dir_fd=parent_cap['fd'],follow_symlinks=False)
    except FileNotFoundError:pass
    else:raise ValueError('original exact parent child still present after group removal')
