_causal_control=None
_physical_capture=None # Only a current genuinely traced source-qualified role.
"""Finite bookkeeping of already-owned Root handles; no FD transfer or runner.
Exterior Root retains this exact mutable registry on every uncertain outcome.
"""
import os
import stat

def require_custody(custody):
    if type(custody) is not dict or type(custody.get('fds')) is not list:
        raise ValueError('ROOT_FD_CUSTODY_UNBOUND: exterior-retained registry required')
    if custody.get('capacity_release') is not False or custody.get('STOP_UNCONFIRMED') is True:
        raise ValueError('ROOT_FD_CUSTODY_STOP: no retry after uncertain operation')
    return custody

def own(custody,fd,label):
    # FIRST operation after open/dup returns, before any fstat/identity checks.
    phase=custody.get('active_phase')
    row={'fd':fd,'label':label,'close_attempted':False,'closed':False,
         'close_ambiguous':False,'lifetime':'phase' if phase is not None else 'local',
         'phase':phase,'planned':False,'identity':None}
    custody['fds'].append(row) # FIRST after acquisition; identity failure retains custody.
    # Legacy standalone acquisition/trace controls retain their original first
    # observation point. Connected full-batch custody snapshots EVERY acquisition.
    if custody.get('phase_order') is not None:
        row['identity']=handle_identity(fd)
        if _physical_capture is not None:_physical_capture.observe_state('custody_after',custody)
    return row

def bind_existing(custody,fd,label):
    if _causal_control is not None:_causal_control.before('bind',custody,fd=fd,label=label)
    if _physical_capture is not None:_physical_capture.mark('bind_existing',fd)
    try:row=preflight_existing_role(custody,fd,label)
    except BaseException:
        custody['STOP_UNCONFIRMED']=True
        custody['capacity_release']=False
        raise
    return row if row is not None else own(custody,fd,label)

def opened(custody,label,*args,**kwargs):
    require_custody(custody)
    fd=os.open(*args,**kwargs)
    own(custody,fd,label)
    return fd

def duplicated(custody,label,fd):
    require_custody(custody)
    duplicate=os.dup(fd)
    own(custody,duplicate,label)
    return duplicate

def closed(custody,fd):
    if _causal_control is not None and _causal_control.before('close',custody,fd=fd):return None
    try:
        if _physical_capture is not None:_physical_capture.mark('close',fd)
        matches=[r for r in custody['fds'] if r['fd']==fd and not r['closed']]
        if len(matches)!=1 or matches[0]['close_attempted']:
            custody['STOP_UNCONFIRMED']=True
            raise ValueError('ROOT_FD_CLOSE_RETRY_FORBIDDEN: retained exact custody')
        row=matches[0]
        if row.get('identity') is not None and handle_identity(fd)!=row['identity']:
            stopped(custody,ValueError('ROOT_FD_IDENTITY_BEFORE_CLOSE'))
            raise ValueError('ROOT_FD_IDENTITY_BEFORE_CLOSE')
        row['close_attempted']=True
        try:os.close(fd)
        except BaseException as error:
            row['close_ambiguous']=True
            row['close_error_type']=type(error).__name__
            custody['STOP_UNCONFIRMED']=True
            custody['capacity_release']=False
            raise
        row['closed']=True
    finally:
        if _causal_control is not None:_causal_control.after_close(custody,fd)

def stopped(custody,error):
    custody['STOP_UNCONFIRMED']=True
    custody['capacity_release']=False
    custody['error_type']=type(error).__name__
    if _physical_capture is not None:
        _physical_capture.mark('STOP')
        _physical_capture.observe_state('STOP_custody',custody)
    if hasattr(error,'scan_custody'):
        custody.setdefault('scan_failure_custody',[]).append(dict(error.scan_custody))
    # Retain all live/ambiguous handles and partial evidence; no guessed retry.


class ExistingCallerCustody:
    """LIVE actual retained handles in the already-existing outside caller.
    A serialized descriptor inventory cannot construct this capability object.
    Caller owns original slot/group/outside-output-parent capabilities throughout
    the finite child CLI, and MUST keep this object on STOP rather than exiting.
    No FD transfer, new controller, acquisition of a new slot, or second launch.
    """
    def __init__(self,owner_birth,slot_fd,slot_identity,groups,output_parents,retained_roles=None):
        import resource,stat,time
        from pathlib import Path
        if resource.getrlimit(resource.RLIMIT_AS)!=(268435456,268435456) \
            or resource.getrlimit(resource.RLIMIT_CPU)!=(30,30) \
            or resource.getrlimit(resource.RLIMIT_FSIZE)!=(67108864,67108864) \
            or resource.getrlimit(resource.RLIMIT_CORE)!=(0,0) \
            or not 0<len(os.sched_getaffinity(0))<=2:
            raise ValueError('CALLER_HARD_CAPS: before capability reads')
        self.owner_birth=owner_birth;self.slot_fd=slot_fd;self.slot_identity=slot_identity
        self.groups=groups;self.output_parents=output_parents;self.state='ACTUAL_RETAINED'
        self.registry={'fds':[],'capacity_release':False,'STOP_UNCONFIRMED':False}
        raw=Path('/proc/self/stat').read_text();fields=raw[raw.rfind(')')+2:].split()
        if owner_birth['pid']!=os.getpid() or str(owner_birth['birth'])!=fields[19]:
            raise ValueError('CALLER_ACTUAL_OWN_BIRTH')
        if type(slot_fd)is not int or slot_fd<=2:raise ValueError('CALLER_SLOT_SHAPE')
        q=os.fstat(slot_fd)
        if not stat.S_ISREG(q.st_mode) or q.st_nlink!=1 or [q.st_dev,q.st_ino]!=slot_identity:
            raise ValueError('CALLER_SLOT_IDENTITY')
        roles=[{'fd':slot_fd,'identity':[q.st_dev,q.st_ino,stat.S_IFMT(q.st_mode)],'role':'original_caller_slot'}]
        for row in groups:
            cap={k:row[k] for k in ('fd','path','identity','policy')}
            directory_capability(cap,False);directory_capability(row['parent_capability'],False)
            if Path(row['path']).parent!=Path(row['parent_capability']['path']):
                raise ValueError('CALLER_EXACT_GROUP_PARENT')
            q=os.fstat(row['fd'])
            roles.append({'fd':row['fd'],'identity':[q.st_dev,q.st_ino,stat.S_IFMT(q.st_mode)],
                          'role':row['role']})
        for row in output_parents:
            cap={k:row[k] for k in ('fd','path','identity','policy')}
            directory_capability(cap,True)
            for path in row['output_paths']:validate_private_parent(path,cap)
            q=os.fstat(row['fd'])
            roles.append({'fd':row['fd'],'identity':[q.st_dev,q.st_ino,stat.S_IFMT(q.st_mode)],
                          'role':'original_caller_output_parent'})
        for row in retained_roles or []:
            q=os.fstat(row['fd'])
            if [q.st_dev,q.st_ino,stat.S_IFMT(q.st_mode)]!=row['identity']:
                raise ValueError('CALLER_ACTUAL_RETAINED_ROLE_IDENTITY')
            if row['role']=='auxiliary_borrowed' and not (stat.S_ISDIR(q.st_mode) or stat.S_ISREG(q.st_mode)):
                raise ValueError('CALLER_AUXILIARY_BORROWED_NOT_PIDFD_GO')
            roles.append(row)
        # Complete semantic removal inventory before FIRST role-binding mutation.
        removal_ids=set();removal_paths=set()
        for group in groups:
            identity=tuple(group['identity']);path=group['path']
            if identity in removal_ids or path in removal_paths:
                raise ValueError('CALLER_GROUP_SEMANTIC_DUPLICATE_ZERO_PREFIX')
            removal_ids.add(identity);removal_paths.add(path)
        # Duplicate private-parent refs have one caller lifetime; removable roles
        # may never alias parents/borrowed refs by inode through a distinct FD.
        for row in roles:
            if tuple(row['identity'][:2]) in removal_ids and row['role'] not in ('original_caller_group','auxiliary_group'):
                raise ValueError('CALLER_REMOVAL_ROLE_INODE_ALIAS')
        # Full role/identity collection checked before FIRST role-binding mutation.
        seen={}
        for row in roles:
            if type(row['fd'])is not int or row['fd']<=2:raise ValueError('CALLER_ROLE_FD_SHAPE')
            if row['fd'] in seen and (row['identity']!=seen[row['fd']]['identity']
                or canonical_role(row['role'])!=canonical_role(seen[row['fd']]['role'])):
                raise ValueError('CALLER_ROLE_ALIAS')
            if row['fd'] not in seen:seen[row['fd']]=row
        self.actual_roles=list(seen.values());self.actual_handles=set(seen)
        for row in self.actual_roles:
            retained=bind_existing(self.registry,row['fd'],row['role'])
            retained['identity']=list(row['identity']) # actual full fstat collection validated before FIRST bind.
        self.initial_binding_m=time.monotonic()

    def binding_for_phase(self,source_pin,group_fd,output_parent_fd):
        import stat
        roles={r['fd']:r for r in self.actual_roles}
        if group_fd not in roles or roles[group_fd]['role']!='original_caller_group' \
           or output_parent_fd not in roles or roles[output_parent_fd]['role']!='original_caller_output_parent':
            raise ValueError('CALLER_PHASE_CAPABILITY_ROLE_UNBOUND')
        for fd,row in roles.items():
            q=os.fstat(fd)
            if [q.st_dev,q.st_ino,stat.S_IFMT(q.st_mode)]!=row['identity']:
                raise ValueError('CALLER_PHASE_ACTUAL_ROLE_IDENTITY_DRIFT')
        return {'birth':self.owner_birth,'slot_fd':self.slot_fd,'group_fd':group_fd,
                'output_parent_fd':output_parent_fd,'source_pin':source_pin,
                'auxiliary_capabilities':[dict(row) for row in self.actual_roles]}

    def stop_retaining(self,reason):
        # Live object and live FD originals stay in EXISTING caller context.
        # Return a failure marker to the existing owner; never close/LOCK_UN.
        self.state='STOP_UNCONFIRMED';stopped(self.registry,reason)
        return {'state':self.state,'capacity_release':False,'actual_handles_retained':sorted(self.actual_handles)}

    def after_exact_cli_wait(self,guard_birth,guard_pidfd,clock):
        """Existing caller has actually waited its ONE exact direct GNU child.
        Only actual kernel/group/mount/FD closure admits LAST original close.
        Source records/results from the child do not establish these predicates.
        """
        import time,stat,select
        from pathlib import Path
        if self.state!='ACTUAL_RETAINED':
            return self.stop_retaining(ValueError('CALLER_ALREADY_STOPPED_NO_RETRY'))
        try:
            raw=Path('/proc/'+str(guard_birth['pid'])+'/stat').read_text()
            fields=raw[raw.rfind(')')+2:].split()
            if fields[19]!=str(guard_birth['birth']) or int(fields[1])!=os.getpid():
                raise ValueError('CALLER_ACTUAL_DIRECT_GUARD_BIRTH')
            info=Path('/proc/self/fdinfo/'+str(guard_pidfd)).read_text()
            if not any(r.split()==['Pid:',str(guard_birth['pid'])] for r in info.splitlines()):
                raise ValueError('CALLER_ACTUAL_RETAINED_GUARD_PIDFD')
            left=max(0,min(clock['started_at_monotonic']+119-time.monotonic(),
                           clock['started_at_unix']+119-time.time()))
            if select.select([guard_pidfd],[],[],left)[0]!=[guard_pidfd]:
                return self.stop_retaining(ValueError('CLI_NOT_TERMINAL_ORIGINAL_RESERVE'))
            got,raw_wait_status=os.waitpid(guard_birth['pid'],0)
            if got!=guard_birth['pid']:raise ValueError('CALLER_ACTUAL_EXACT_DIRECT_WAIT')
            if os.waitstatus_to_exitcode(raw_wait_status)!=0:
                return self.stop_retaining(ValueError('CLI_FAILED_KEEP_REAL_CUSTODY'))
            stat_path=Path('/proc/'+str(guard_birth['pid'])+'/stat')
            if stat_path.exists():
                raw=stat_path.read_text();fields=raw[raw.rfind(')')+2:].split()
                if fields[19]==guard_birth['birth']:raise ValueError('CLI_FRESH_ABSENCE_REQUIRED')
            if Path('/proc/self/task/'+str(os.getpid())+'/children').read_text().strip():
                raise ValueError('CALLER_UNKNOWN_ADOPTED_CHILD_STOP')
            mounts=Path('/proc/self/mountinfo').read_text().splitlines()
            for group in self.groups:
                if Path(group['path']).exists():raise ValueError('CALLER_ACTUAL_GROUP_REMOVAL_REQUIRED')
                removed=os.fstat(group['fd'])
                if removed.st_nlink!=0 and os.readlink('/proc/self/fd/'+str(group['fd']))!=group['path']+' (deleted)':
                    raise ValueError('CALLER_RETAINED_ORIGINAL_GROUP_NOT_DELETED')
                q=os.fstat(group['fd'])
                if [q.st_dev,q.st_ino]!=group['identity']:raise ValueError('CALLER_GROUP_CAPABILITY_DRIFT')
            for parent in self.output_parents:
                q=os.fstat(parent['fd'])
                if [q.st_dev,q.st_ino]!=parent['identity']:raise ValueError('CALLER_PARENT_CAPABILITY_DRIFT')
                for output in parent['output_paths']:
                    if any(r.split()[4]==output for r in mounts):raise ValueError('CALLER_NORMAL_MOUNT_ABSENCE_REQUIRED')
            if not time.monotonic()<clock['started_at_monotonic']+120 \
                or not time.time()<clock['final_deadline_unix']:
                raise ValueError('CALLER_ORIGINAL_FINAL_EXHAUSTED_NO_RESET')
            if guard_pidfd in self.actual_handles:raise ValueError('CALLER_GUARD_RESERVED_ALIAS')
            for row in self.actual_roles:
                q=os.fstat(row['fd'])
                if [q.st_dev,q.st_ino,stat.S_IFMT(q.st_mode)]!=row['identity']:
                    raise ValueError('CALLER_ACTUAL_FINAL_ROLE_DRIFT')
            bind_existing(self.registry,guard_pidfd,'original_guard_pidfd')
            closed(self.registry,guard_pidfd)
            for fd in sorted(self.actual_handles-{self.slot_fd}):closed(self.registry,fd)
            closed(self.registry,self.slot_fd) # actual last original close, once, after kernel cleanup.
            self.registry['capacity_release']=True;self.state='ACTUAL_CLOSED'
            if _physical_capture is not None:
                _physical_capture.mark('caller_closed',self.slot_fd)
                _physical_capture.observe_state('caller_closed',self.registry)
            return {'state':self.state,'capacity_release':True,'clock_reset':False}
        except BaseException as error:return self.stop_retaining(error)


def retain_in_existing_caller(existing_owner_context,holder):
    """Bind this LIVE object into the existing owner's persistent lifetime.
    Explicit code connection; neither a JSON receipt nor an in-CLI registry.
    Existing caller setup/terminal path must be separately admitted at real Go.
    """
    if _physical_capture is not None:_physical_capture.mark('holder_mutation')
    if type(holder)is not ExistingCallerCustody or holder.owner_birth['pid']!=os.getpid():
        raise ValueError('EXISTING_CALLER_LIVE_OBJECT_REQUIRED')
    if existing_owner_context.get('kernel120_holder') is not None:
        raise ValueError('EXISTING_CALLER_NO_REBIND_OR_SECOND_LAUNCH')
    for fd in holder.actual_handles:os.fstat(fd)
    existing_owner_context['kernel120_holder']=holder
    return holder

def canonical_role(label):
    # Explicit aliases of ONE semantic role; auxiliary roles never alias reserved roles.
    aliases={'existing_output':'output','Root_output':'output','helper_output':'output',
             'helper_slot_duplicate':'slot','original_exterior_slot':'slot',
             'exterior_evidence':'trace_directory',
             'kernel120_stock:pidfd':'stock_pidfd',
             'kernel120_post_terminal_helper:pidfd':'post_terminal_helper_pidfd',
             'existing_final_helper_pidfd':'post_terminal_helper_pidfd',
             'root_ready':'actor_ready:read','root_GO':'actor_GO:write',
             'output_parent':'private_parent','original_caller_output_parent':'private_parent',
             'original_caller_private_parent':'private_parent',
             'original_caller_group_parent':'group_parent'}
    return aliases.get(label,label)

def preflight_existing_role(custody,fd,label):
    require_custody(custody)
    if type(fd)is not int or fd<=2 or type(label)is not str or not label:
        raise ValueError('ROOT_FD_ROLE_SHAPE')
    rows=[r for r in custody['fds'] if r['fd']==fd and not r['closed']]
    if len(rows)>1 or any(r['close_attempted'] or r.get('close_ambiguous') for r in rows):
        raise ValueError('ROOT_FD_ROLE_AMBIGUOUS')
    if rows and rows[0].get('identity') is not None and handle_identity(fd)!=rows[0]['identity']:
        raise ValueError('ROOT_FD_REBIND_IDENTITY_DRIFT')
    if rows and canonical_role(rows[0]['label'])!=canonical_role(label):
        raise ValueError('ROOT_FD_ROLE_MISMATCH: '+label)
    return rows[0] if rows else None

def directory_capability(cap,private=True,process_pid=None):
    """Observe an actual pre-existing exact directory capability; never chmod/open."""
    import stat
    from pathlib import Path
    if type(cap)is not dict or set(cap)!={'fd','path','identity','policy'}:
        raise ValueError('DIRECTORY_CAPABILITY_SHAPE')
    fd=cap['fd'];path=Path(cap['path']);policy=cap['policy']
    if type(fd)is not int or fd<=2 or type(policy)is not dict \
       or set(policy)!={'uid','gid','mode'} or any(type(v)is not int or v<0 for v in policy.values()):
        raise ValueError('DIRECTORY_CAPABILITY_POLICY')
    if not path.is_absolute() or '..' in path.parts or str(path)!=os.path.normpath(str(path)):
        raise ValueError('DIRECTORY_CAPABILITY_PATH')
    for ancestor in reversed(path.parents):
        q=ancestor.lstat()
        if not stat.S_ISDIR(q.st_mode):raise ValueError('DIRECTORY_CAPABILITY_SYMLINK_ANCESTOR')
    q=path.lstat()
    proc=os.getpid() if process_pid is None else process_pid
    a=os.stat('/proc/'+str(proc)+'/fd/'+str(fd))
    link=os.readlink('/proc/'+str(proc)+'/fd/'+str(fd))
    if not stat.S_ISDIR(q.st_mode) or not stat.S_ISDIR(a.st_mode) \
       or link!=str(path) or [a.st_dev,a.st_ino]!=cap['identity'] \
       or [q.st_dev,q.st_ino]!=cap['identity']:
        raise ValueError('DIRECTORY_CAPABILITY_ACTUAL_IDENTITY')
    expected=(policy['uid'],policy['gid'],policy['mode'])
    if (q.st_uid,q.st_gid,stat.S_IMODE(q.st_mode))!=expected \
       or (a.st_uid,a.st_gid,stat.S_IMODE(a.st_mode))!=expected:
        raise ValueError('DIRECTORY_CAPABILITY_UID_MODE')
    if private and (policy['uid']!=os.getuid() or policy['gid']!=os.getgid() or policy['mode']!=0o700):
        raise ValueError('DIRECTORY_CAPABILITY_PRIVATE_0700_OWNER')
    # Existing cgroup ancestors use their admitted actual policy, never 0700.
    return {'fd':fd,'path':str(path),'identity':cap['identity'],'policy':policy}

def validate_private_parent(child_path,parent_cap,process_pid=None):
    from pathlib import Path
    path=Path(child_path);parent=directory_capability(parent_cap,True,process_pid)
    if not path.is_absolute() or '..' in path.parts or path.parent!=Path(parent['path']):
        raise ValueError('PRIVATE_PARENT_EXACT_CHILD')
    return parent

def handle_identity(fd):
    q=os.fstat(fd)
    return [q.st_dev,q.st_ino,stat.S_IFMT(q.st_mode)]


def actual_process_birth(birth,code='AUX_OWNER_ACTUAL_BIRTH_ZERO_PREFIX'):
    """Fresh kernel identity; a supplied PID or live inherited FD is insufficient."""
    from pathlib import Path
    if type(birth)is not dict or type(birth.get('pid'))is not int or birth['pid']<=0:
        raise ValueError(code)
    fields=Path('/proc/'+str(birth['pid'])+'/stat').read_text().rsplit(')',1)[1].split()
    current={'pid':birth['pid'],'birth':fields[19],'ppid':int(fields[1]),
             'pgid':int(fields[2]),'sid':int(fields[3])}
    if fields[0] in ('Z','X') or any(current[k]!=birth.get(k) for k in current):
        raise ValueError(code)
    return current


def actual_process_uids(pid):
    from pathlib import Path
    rows=[r.split()[1:] for r in Path('/proc/'+str(pid)+'/status').read_text().splitlines()
          if r.startswith('Uid:')]
    if len(rows)!=1 or len(rows[0])!=4 or any(not x.isdecimal() for x in rows[0]):
        raise ValueError('AUX_ORIGINAL_CALLER_UID_ZERO_PREFIX')
    return tuple(int(x) for x in rows[0])


def observe_original_caller_association(admission):
    """Reobserve admitted ORIGINAL caller -> actual GNU parent -> current Root.
    This observation is not Source/action authorization and creates no capability.
    """
    if type(admission)is not dict:
        raise ValueError('AUX_ORIGINAL_CALLER_ADMISSION_REQUIRED_ZERO_PREFIX')
    caller=actual_process_birth(admission['actual_holder_birth'])
    guard=actual_process_birth(admission['guard_birth'])
    root=actual_process_birth(admission['actual_Root_birth'])
    if root['pid']!=os.getpid() or root['ppid']!=os.getppid() or guard['pid']!=root['ppid'] \
       or guard['ppid']!=caller['pid'] or caller['pid'] in (root['pid'],guard['pid']) \
       or caller['pgid']==guard['pgid']:
        raise ValueError('AUX_ORIGINAL_CALLER_ASSOCIATION_ZERO_PREFIX')
    expected=(os.getuid(),)*4
    if admission.get('actual_holder_uid')!=os.getuid() \
       or any(actual_process_uids(p)!=expected for p in (caller['pid'],guard['pid'],root['pid'])):
        raise ValueError('AUX_ORIGINAL_CALLER_UID_ZERO_PREFIX')
    return caller


def preflight_retained_original_caller(holder,admission):
    """Every phase holder uses the independently observed outer original caller."""
    caller=observe_original_caller_association(admission)
    if type(holder)is not dict or any(holder.get('birth',{}).get(k)!=caller[k] for k in caller):
        raise ValueError('AUX_ORIGINAL_CALLER_ASSOCIATION_ZERO_PREFIX')
    actual_process_birth(holder['birth'])
    original=admission['actual_holder_binding']
    if holder.get('source_pin')!=original['source_pin'] or holder.get('slot_fd')!=original['slot_fd']:
        raise ValueError('AUX_ORIGINAL_CALLER_BINDING_ZERO_PREFIX')
    if type(holder.get('auxiliary_capabilities'))is not list:
        raise ValueError('AUX_CALLER_INVENTORY_ZERO_PREFIX')
    return caller['pid']


def retained_parent_is_cgroup(cap):
    from pathlib import Path
    # Component-aware boundary: /sys/fs/cgroup-other is a private parent.
    return Path(cap['path']).is_relative_to('/sys/fs/cgroup')

def preflight_auxiliary_phase_plans(phase_caps,strict_existing_root=False,original_caller_birth=None,
                                    admitted_original_caller=None):
    """WHOLE semantic inventory; observation only, before bind or any record.
    Removable nodes never share. Borrowed duplicates require an explicit finite
    purpose and distinct phase-local/caller handles, each closed exactly once.
    """
    from pathlib import Path
    import stat

def preflight_role_plan(custody,plans,phase_order):
    """Exact finite role/lifetime connection, with no registry mutation."""
    require_custody(custody)
    if type(plans)is not list or len(plans)>8192 or len(phase_order)!=23 or len(set(phase_order))!=23:
        raise ValueError('ROOT_ROLE_PLAN_SHAPE')
    table={};phase_inodes={};phase_paths={}
    for plan in plans:
        fd=plan['fd'];role=canonical_role(plan['role']);life=plan['lifetime'];phase=plan['phase']
        if life not in ('batch','phase') or (phase not in phase_order if life=='phase' else phase is not None):
            raise ValueError('ROOT_ROLE_PLAN_LIFETIME')
        if not plan.get('purpose') or handle_identity(fd)!=plan['identity']:
            raise ValueError('ROOT_ROLE_PLAN_ACTUAL_IDENTITY')
        if role in ('auxiliary_group','auxiliary_output','auxiliary_borrowed') and (life!='phase' or phase not in AUXILIARY_PHASE_CONSUMERS or plan['purpose']!='current_auxiliary_once_only_cleanup/'+AUXILIARY_PHASE_CONSUMERS[phase]):
            raise ValueError('ROOT_AUXILIARY_CONSUMER_LIFETIME_ZERO_PREFIX')
        preflight_existing_role(custody,fd,role)
        if fd in table:
            previous=table[fd]
            if any(previous[k]!=plan[k] for k in ('identity','lifetime','phase','purpose')) or previous['role']!=role:
                raise ValueError('ROOT_ROLE_PLAN_FD_ALIAS')
            continue
        if life=='phase' and role in ('group','trace_directory','output'):
            key=tuple(plan['identity'][:2])
            previous=phase_inodes.get(key)
            if previous is not None and (previous['phase']!=phase or previous['role']!=role):
                raise ValueError('ROOT_ROLE_PLAN_CROSS_PHASE_INODE')
            phase_inodes[key]=dict(plan,role=role)
            path=plan.get('path')
            if path:
                previous=phase_paths.get(path)
                if previous is not None and (previous['phase']!=phase or previous['role']!=role):
                    raise ValueError('ROOT_ROLE_PLAN_CROSS_PHASE_PATH')
                phase_paths[path]=dict(plan,role=role)
        table[fd]=dict(plan,role=role)
    for row in custody['fds']:
        if not row['closed'] and row['fd'] not in table:
            raise ValueError('ROOT_ROLE_PLAN_UNKNOWN_LIVE_HANDLE')
    if actual_open_fd_set()!=set(table):
        raise ValueError('ROOT_ROLE_PLAN_ACTUAL_FD_INVENTORY')
    return list(table.values())


def install_role_plan(custody,plans,phase_order):
    complete=preflight_role_plan(custody,plans,phase_order)
    if custody.get('phase_order') is not None:raise ValueError('ROOT_ROLE_PLAN_NO_REINSTALL')
    # FULL preflight precedes the FIRST mutation.
    for plan in complete:
        row=bind_existing(custody,plan['fd'],plan['role'])
        row.update(identity=plan['identity'],lifetime=plan['lifetime'],phase=plan['phase'],
                   purpose=plan['purpose'],planned=True)
    custody['phase_order']=list(phase_order);custody['completed_phases']=[]
    custody['active_phase']=None;custody['role_plan']=complete
    return complete


def verify_retained_plan(custody):
    if _causal_control is not None:_causal_control.before('retained_plan',custody)
    require_custody(custody)
    completed=set(custody['completed_phases'])
    for plan in custody['role_plan']:
        rows=[r for r in custody['fds'] if r['fd']==plan['fd'] and r.get('planned')]
        if len(rows)!=1:raise ValueError('ROOT_PLAN_ROW_LOST_OR_REUSED')
        row=rows[0]
        if plan['lifetime']=='phase' and plan['phase'] in completed:
            if not row['closed'] or row.get('close_ambiguous'):raise ValueError('ROOT_COMPLETED_PHASE_NOT_CLOSED')
        else:
            if row['closed'] or row['close_attempted'] or row.get('close_ambiguous'):
                raise ValueError('ROOT_FUTURE_OR_BATCH_HANDLE_LOST')
            try:identity=handle_identity(row['fd'])
            except OSError as error:
                raise ValueError('ROOT_FUTURE_OR_BATCH_HANDLE_LOST') from error
            if identity!=plan['identity']:raise ValueError('ROOT_FUTURE_OR_BATCH_HANDLE_LOST')
    verify_live_registry(custody)
    return True


def begin_phase(custody,phase):
    if _causal_control is not None:_causal_control.before('begin_phase',custody,phase=phase)
    if _physical_capture is not None:
        _physical_capture.mark('phase_begin',custody.get('phase_order',[]).index(phase))
        _physical_capture.observe_state('phase_begin',custody)
    verify_retained_plan(custody)
    index=len(custody['completed_phases'])
    if custody['active_phase'] is not None or index>=len(custody['phase_order']) or custody['phase_order'][index]!=phase:
        raise ValueError('ROOT_PHASE_ORDER_NO_REPLAY')
    if any(not r['closed'] and not r.get('planned') for r in custody['fds']):
        raise ValueError('ROOT_UNKNOWN_LIVE_DYNAMIC_HANDLE')
    custody['active_phase']=phase


def complete_phase(custody,phase):
    if _causal_control is not None:_causal_control.before('complete_phase',custody,phase=phase)
    if _physical_capture is not None:_physical_capture.mark('phase_complete')
    require_custody(custody)
    if custody['active_phase']!=phase:raise ValueError('ROOT_PHASE_NOT_ACTIVE')
    verify_live_registry(custody)
    current=[r for r in custody['fds'] if r.get('lifetime')=='phase' and r.get('phase')==phase]
    # Trace capability is the sole intentional live CURRENT handle here;
    # successful normal cleanup/raw capture must already have closed all others.
    live=[r for r in current if not r['closed']]
    if any(canonical_role(r['label'])!='trace_directory' or not r.get('planned') for r in live):
        raise ValueError('ROOT_CURRENT_PHASE_EFFECT_HANDLE_NOT_CLOSED')
    for row in live:closed(custody,row['fd'])
    if any(not r['closed'] or r.get('close_ambiguous') for r in current):
        raise ValueError('ROOT_CURRENT_PHASE_AMBIGUOUS_CLOSE')
    if any(not r['closed'] and r.get('lifetime') not in ('phase','batch') for r in custody['fds']):
        raise ValueError('ROOT_PHASE_UNKNOWN_LIFETIME')
    custody['completed_phases'].append(phase);custody['active_phase']=None
    verify_retained_plan(custody)
    if _physical_capture is not None:
        _physical_capture.mark('phase_closed',custody['phase_order'].index(phase))
        _physical_capture.observe_state('phase_closed',custody)
    return {'phase':phase,'actual_phase_fd_closure':True,'capacity_release':False}


def verify_current_effect_closure(custody,phase,slot_fd):
    """Current effect closure plus exact preservation of known future/batch."""
    require_custody(custody)
    if custody.get('active_phase')!=phase:raise ValueError('ROOT_CURRENT_PHASE_UNBOUND')
    verify_live_registry(custody)
    for row in custody['fds']:
        if row.get('close_ambiguous'):raise ValueError('ROOT_PHASE_CLOSE_AMBIGUOUS')
        if row['closed']:continue
        life=row.get('lifetime')
        if life=='phase' and row.get('phase')==phase:
            if canonical_role(row['label'])!='trace_directory' or not row.get('planned'):
                raise ValueError('ROOT_CURRENT_PHASE_EFFECT_HANDLE_NOT_CLOSED')
        elif life=='phase':
            if not row.get('planned') or row.get('phase') in custody['completed_phases']:
                raise ValueError('ROOT_UNKNOWN_FUTURE_HANDLE')
        elif life=='batch':
            if not row.get('planned'):raise ValueError('ROOT_UNKNOWN_BATCH_HANDLE')
        else:raise ValueError('ROOT_UNKNOWN_LIVE_LIFETIME')
        if handle_identity(row['fd'])!=row['identity']:raise ValueError('ROOT_RETAINED_HANDLE_IDENTITY_DRIFT')
    # Last original is in the external caller; this live FD is the batch duplicate.
    rows=[r for r in custody['fds'] if r['fd']==slot_fd and not r['closed']]
    if len(rows)!=1 or rows[0].get('lifetime')!='batch' or canonical_role(rows[0]['label'])!='slot':
        raise ValueError('ROOT_ORIGINAL_BATCH_SLOT_LOST')
    return True


def finalize_batch_roles(custody,slot_fd):
    """After ALL actual phases: settle known batch refs, CLI slot duplicate LAST.
    This NEVER releases the external caller's original slot or grants capacity.
    """
    if _causal_control is not None:_causal_control.before('batch_finalize',custody,slot_fd=slot_fd)
    if _physical_capture is not None:_physical_capture.mark('batch_finalize',slot_fd)
    require_custody(custody)
    if custody['active_phase'] is not None or custody['completed_phases']!=custody['phase_order']:
        raise ValueError('ROOT_BATCH_PHASES_NOT_ACTUALLY_COMPLETE')
    verify_retained_plan(custody)
    live=[r for r in custody['fds'] if not r['closed']]
    if any(not r.get('planned') or r.get('lifetime')!='batch' for r in live):
        raise ValueError('ROOT_BATCH_UNKNOWN_OR_PHASE_HANDLE')
    if len([r for r in live if r['fd']==slot_fd and canonical_role(r['label'])=='slot'])!=1:
        raise ValueError('ROOT_BATCH_SLOT_NOT_EXACT')
    for row in live:
        if row['fd']!=slot_fd:closed(custody,row['fd'])
    closed(custody,slot_fd)
    if any(not r['closed'] or r.get('close_ambiguous') for r in custody['fds']):
        raise ValueError('ROOT_BATCH_FINAL_CLOSURE_UNCONFIRMED')
    custody['batch_cli_closed']=True
    if _physical_capture is not None:
        _physical_capture.mark('batch_closed',slot_fd)
        _physical_capture.observe_state('batch_closed',custody)
    return {'batch_cli_closed':True,'capacity_release':False,'existing_caller_last_slot_close_required':True}


def actual_open_fd_identities():
    """One exact Linux/glibc directory stream; its OWN dirfd is never inherited.
    No EBADF suppression: every enumerated non-scan handle must still be real.
    """
    if _physical_capture is not None:_physical_capture.mark('scan_enter')
    import ctypes,platform
    if platform.system()!='Linux' or platform.machine()!='x86_64':
        raise ValueError('ROOT_FD_SCAN_REVIEWED_ABI_REQUIRED')
    class Dirent(ctypes.Structure):
        _fields_=[('d_ino',ctypes.c_ulong),('d_off',ctypes.c_long),
                  ('d_reclen',ctypes.c_ushort),('d_type',ctypes.c_ubyte),
                  ('d_name',ctypes.c_char*256)]
    if Dirent.d_name.offset!=19 or ctypes.sizeof(Dirent)!=280:
        raise ValueError('ROOT_FD_SCAN_DIRENT_ABI')
    libc=ctypes.CDLL(None,use_errno=True)
    libc.opendir.argtypes=[ctypes.c_char_p];libc.opendir.restype=ctypes.c_void_p
    libc.dirfd.argtypes=[ctypes.c_void_p];libc.dirfd.restype=ctypes.c_int
    libc.readdir.argtypes=[ctypes.c_void_p];libc.readdir.restype=ctypes.POINTER(Dirent)
    libc.closedir.argtypes=[ctypes.c_void_p];libc.closedir.restype=ctypes.c_int
    stream=libc.opendir(b'/proc/self/fd')
    if not stream:raise OSError(ctypes.get_errno(),'ROOT_FD_SCAN_OPEN')
    # The returned stream owns its exact descriptor immediately. closedir is
    # its sole close; no fallback, guessed transient, retry or inherited close.
    scan_custody={'role':'self_owned_proc_fd_scan','stream_address':int(stream),'fd':None,
                  'identity':None,'close_attempted':False,'closed':False,'close_ambiguous':False}
    result={};scan_seen=False
    try:
        scan_fd=libc.dirfd(stream)
        scan_custody['fd']=scan_fd if scan_fd>=0 else None
        if scan_fd<0:raise OSError(ctypes.get_errno(),'ROOT_FD_SCAN_DIRFD')
        scan_identity=handle_identity(scan_fd);scan_custody['identity']=scan_identity
        if _physical_capture is not None:
            _physical_capture.mark('scan_dirfd',scan_fd)
            _physical_capture.observe_state('scan_dirfd',scan_custody)
        for _ in range(65536):
            ctypes.set_errno(0);entry=libc.readdir(stream)
            if not entry:
                error=ctypes.get_errno()
                if error:raise OSError(error,'ROOT_FD_SCAN_READ')
                break
            name=bytes(entry.contents.d_name)
            if name in (b'.',b'..'):continue
            if not name or not name.isdigit():raise ValueError('ROOT_FD_SCAN_ENTRY')
            fd=int(name)
            if fd==scan_fd:
                if scan_seen or handle_identity(fd)!=scan_identity:
                    raise ValueError('ROOT_FD_SCAN_OWN_IDENTITY')
                scan_seen=True
                continue
            if fd in result:raise ValueError('ROOT_FD_SCAN_DUPLICATE')
            result[fd]=handle_identity(fd) # Includes genuine EBADF: ALWAYS stop.
        else:raise ValueError('ROOT_FD_SCAN_BOUND')
        if not scan_seen:raise ValueError('ROOT_FD_SCAN_OWN_ENTRY_MISSING')
    except BaseException as error:
        error.scan_custody=scan_custody
        raise
    finally:
        scan_custody['close_attempted']=True
        if libc.closedir(stream)!=0:
            scan_custody['close_ambiguous']=True
            error=OSError(ctypes.get_errno(),'ROOT_FD_SCAN_CLOSE_UNCONFIRMED')
            error.scan_custody=scan_custody
            raise error
        scan_custody['closed']=True
        if _physical_capture is not None:
            _physical_capture.mark('scan_leave',scan_fd)
            _physical_capture.observe_state('scan_closed',scan_custody)
    return result


def actual_open_fd_set():
    return {fd for fd in actual_open_fd_identities() if fd>2}


def child_descriptor_hygiene(custody,keep):
    """Close only exact inherited LIVE custody rows in this child's COW copy.
    Retained keep/owned EBADF or unknown actual handles stop BEFORE any close.
    """
    if _causal_control is not None:_causal_control.before('child_hygiene',custody,keep=keep)
    require_custody(custody)
    if type(keep)is not set or any(type(fd)is not int or fd<=2 for fd in keep):
        raise ValueError('CHILD_RETAINED_KEEP_SHAPE')
    live=[r for r in custody['fds'] if not r['closed']]
    table={r['fd']:r for r in live}
    if _physical_capture is not None:_physical_capture.observe_state('child_hygiene',
        {'custody':custody,'keep':sorted(keep)})
    if len(table)!=len(live) or not keep<=set(table):
        raise ValueError('CHILD_INHERITED_CUSTODY_SHAPE')
    try:observed=actual_open_fd_identities()
    except BaseException as error:
        stopped(custody,error)
        raise
    if {fd for fd in observed if fd>2}!=set(table):
        stopped(custody,ValueError('CHILD_UNKNOWN_OR_MISSING_INHERITED_FD'))
        raise ValueError('CHILD_UNKNOWN_OR_MISSING_INHERITED_FD')
    for fd,row in table.items():
        if row.get('identity') is None or row.get('close_attempted') or row.get('close_ambiguous') \
           or observed[fd]!=row['identity']:
            stopped(custody,ValueError('CHILD_INHERITED_IDENTITY_UNCONFIRMED'))
            raise ValueError('CHILD_INHERITED_IDENTITY_UNCONFIRMED')
    # Complete validation precedes FIRST inherited descriptor effect. No
    # range-close/whitelist adoption: closed() revalidates one actual owned row.
    for fd in sorted(set(table)-keep):closed(custody,fd)
    for fd in keep:
        if handle_identity(fd)!=table[fd]['identity']:
            stopped(custody,ValueError('CHILD_RETAINED_KEEP_LOST'))
            raise ValueError('CHILD_RETAINED_KEEP_LOST')
    if _physical_capture is not None:_physical_capture.observe_state('child_hygiene_closed',
        {'custody':custody,'keep':sorted(keep)})
    return {fd:table[fd]['identity'] for fd in keep}


def verify_live_registry(custody):
    live=[r for r in custody['fds'] if not r['closed']]
    if len({r['fd'] for r in live})!=len(live) or actual_open_fd_set()!={r['fd'] for r in live}:
        raise ValueError('ROOT_ACTUAL_UNKNOWN_OR_MISSING_LIVE_FD')
    for row in live:
        if row.get('identity') is None or row.get('close_ambiguous') or handle_identity(row['fd'])!=row['identity']:
            raise ValueError('ROOT_ACTUAL_LIVE_FD_IDENTITY_UNCONFIRMED')
    return True
