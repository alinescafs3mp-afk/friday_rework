        child=any(e.syscall in (57,58) or e.syscall==56 and not e.arguments[0]&0x10000 or
            e.syscall==435 and e.raw_input and not int.from_bytes(e.raw_input[:8],'little')&0x10000 for e in births)
        physical.need(thread and child,'EXACT_SUBJECT_LOCAL_FORK_THREAD_REQUIRED')
        _exact_task_operation(f,operation['operation'])
        return f.proof('exact_original_operation/'+case_id,True,
                       (operation['entry_sequence'],operation['return_sequence']))
    if case_id=='K120_TASKS128_REAL_LIMIT':
        return f.proof('real_libc_mallopt_minus8_one',bool(f.allocator()))
    predicates={
      'K120_FIRST_BIRTH_RESERVATION':any(e.syscall==435 and e.result>0 and e.raw_input is not None
          and int.from_bytes(e.raw_input[:8],'little')&(1<<33) and type(e.fd_before)is physical.FDRead for e in events),
      'K120_CONTROLLER_WRITE_REFUSAL':any(e.syscall in (2,257) and e.result in (-13,-30) and e.path is not None and e.path.startswith('/sys/fs/cgroup/') for e in events),
      'K120_OUTPUT_BYTE_INODE_LIMIT':any(e.syscall==1 and e.result==-28 for e in events) and any(e.syscall in (2,257) and e.result==-28 for e in events),
      'K120_ZOMBIE_ADOPTED_WAIT':any(e.syscall==61 and any(p.state=='Z' for p in e.actual_wait_targets) for e in entries),
      'K120_ROOT_WAIT_KERNEL_ZERO':bool(f.raw_waits(f.root)),
      'K120_ROOT_NORMAL_UMOUNT':any(e.syscall==166 and e.result==0 and e.arguments[1]==0 for e in events),
      'K120_PROTECTED_WRITERS_UNCHANGED':f.protected()}
    physical.need(case_id in predicates,'EXACT_ORIGINAL_POSITIVE_CAUSE_BODY')
    return f.proof('real_original_operation/'+case_id,predicates[case_id])

def _exact_task_operation(f,number):
    row=f.operation(number);lo=row['entry_sequence'];hi=row['return_sequence']
    births=tuple(event for event in f.events(pid=row['pid'],lo=lo,hi=hi,stage='actual_birth')
        if event.birth==row['birth'])
    physical.need(len(births)>=2,'EXACT_OPERATION_REAL_FORK_AND_THREAD_BIRTHS')
    returned=[event for event in f.events(pid=row['pid'],lo=lo,hi=hi) if
        event.birth==row['birth'] and event.syscall in (56,57,58,435) and event.result>0]
    def status_ids(process,key):
        values=[line.split()[1:] for line in process.status_raw.splitlines() if line.startswith(key+':')]
        physical.need(len(values)==1 and values[0] and all(value.isdecimal() for value in values[0]),
            'EXACT_OPERATION_RAW_NAMESPACE_OR_TGID')
        return tuple(map(int,values[0]))
    process_children=[];thread_children=[];used_returns=set()
    for event in births:
        pid=event.arguments[0];child=f.run.birth_reads[pid]
        physical.need(f.run.known_subjects[pid][0]==child.birth,'EXACT_OPERATION_CHILD_BIRTH')
        candidates=[]
        for created in returned:
            physical.need(created.actual_process_sample is not None,'EXACT_BIRTH_PARENT_PROCESS_READ')
            parent=created.actual_process_sample['process']
            physical.need(parent.pid==row['pid'] and parent.birth==row['birth'],'EXACT_BIRTH_PARENT_IDENTITY')
            level=len(status_ids(parent,'NSpid'))-1;ids=status_ids(child,'NSpid')
            if len(ids)>level and ids[level]==created.result and event.sequence<created.sequence:
                candidates.append(created)
        physical.need(len(candidates)==1 and candidates[0].sequence not in used_returns,
            'ONE_EXACT_NAMESPACE_CORRECT_OPERATION_BIRTH_RETURN')
        created=candidates[0];used_returns.add(created.sequence)
        if created.syscall==435:
            physical.need(created.raw_input is not None and len(created.raw_input)>=8,
                'EXACT_CLONE3_RAW_FLAGS')
            thread=bool(int.from_bytes(created.raw_input[:8],'little')&0x10000)
        else:thread=created.syscall==56 and bool(created.arguments[0]&0x10000)
        tgid=status_ids(child,'Tgid')
        physical.need(len(tgid)==1,'EXACT_OPERATION_CHILD_TGID')
        if thread:
            physical.need(tgid[0]==row['pid'],'EXACT_OPERATION_THREAD_OWNER')
            thread_children.append(child)
        else:
            physical.need(tgid[0]==pid and child.ppid==row['pid'],'EXACT_OPERATION_PROCESS_PARENT')
            process_children.append(child)
    physical.need(process_children and thread_children and len(used_returns)==len(returned),
        'COMPLETE_EXACT_OPERATION_PROCESS_THREAD_BIRTH_SET')
    waits=f.raw_waits(row['pid'],lo,hi);selected_waits=[]
    for child in process_children:
        matching=[]
        for wait in waits:
            if wait.birth!=row['birth'] or wait.actual_process_sample is None:continue
            parent=wait.actual_process_sample['process']
            physical.need(parent.pid==row['pid'] and parent.birth==row['birth'],'EXACT_OPERATION_RAW_WAIT_OWNER')
            level=len(status_ids(parent,'NSpid'))-1
            for target in wait.actual_wait_targets:
                ids=status_ids(target,'NSpid')
                if target.pid==child.pid and target.birth==child.birth and target.ppid==row['pid'] and len(ids)>level and ids[level]==wait.result:
                    matching.append(wait)
        physical.need(len(matching)==1,'ONE_EXACT_RAW_WAIT_PER_OPERATION_PROCESS_CHILD')
        wait=matching[0]
        physical.need(any(terminal[0]==child.pid and terminal[1]==child.birth and
            terminal[2]==f.status(wait) for terminal in f.run.terminals),'EXACT_OPERATION_RAW_WAIT_STATUS_CAUSALITY')
        selected_waits.append(wait.sequence)
    physical.need(len(selected_waits)==len(set(selected_waits)),'NO_OPERATION_CHILD_WAIT_REUSE')
    descendants=process_children+thread_children
    physical.need(all(any(terminal[0]==child.pid and terminal[1]==child.birth for terminal in f.run.terminals)
        and not physical.exact_presence_read(child.pid,child.birth).same_birth for child in descendants),
        'EXACT_OPERATION_RAW_TERMINALS_AND_FRESH_BIRTH_ABSENCE')
    return f.proof('exact_task_operation/'+str(number),True,
        (lo,hi,*[event.sequence for event in births],*selected_waits))

class GroupedLegacyObservation(Mapping):
    def __init__(self,group):
        runs=capture.validate_current_capture(group)
        physical.need(type(group)is capture.RawRunGroup,'ACTUAL_GROUP_OBJECT_REQUIRED')
        self.group=group;self.views=tuple(_extend_c75(LegacyObservation(run)) for run in runs)
        self.runs=runs
    def __iter__(self):return iter(self.views[0])
    def __len__(self):return len(self.views[0])
    def __getitem__(self,name):
        keys=[tuple(r.case.declared_delta['member_key']) for r in self.runs]
        if name=='control_id':return self.group.control_id
        if name=='independent_raw_root_trace_ref':return self.group.raw_ref
        if name=='tested_phase_ids':return list(dict.fromkeys(p for p,k in keys))
        if name=='tested_auxiliary_kinds':return list(dict.fromkeys(k for p,k in keys))
        if name=='exact_phase_kind_attempts':return len(keys)
        if name in ('actual_independent_phase_inventories_ref','actual_independent_nonempty_all_safe_inventories_ref'):
            for run,(phase,kind) in zip(self.runs,keys):
                facts=LiveFacts(run);rows=[r for r in facts.aux_rows() if r[0]==phase and
                    {'auxiliary_groups':'auxiliary_group','auxiliary_outputs':'auxiliary_output','auxiliary_fds':'auxiliary_borrowed'}[r[1]]==kind]
                physical.need(rows,'REAL_INDEPENDENT_NONEMPTY_GROUP_MEMBER')
            return self.group.raw_ref
        values=[view[name] for view in self.views]
        if name=='TOTAL_prefix_effect_counts':
            return {key:sum(value[key] for value in values) for key in values[0]}
        if name in ('actual_original_caller_slot_and_caps_unchanged_raw_ref',
                    'original_protected_writers_and_controllers_unchanged_raw_ref'):
            physical.need(all(value is not None for value in values),'EVERY_GROUP_MEMBER_PHYSICAL_PRESERVATION')
            return self.group.raw_ref
        physical.need(all(value==values[0] for value in values),'GROUP_MEMBER_CONSUMER_DISAGREEMENT:'+name)
        return values[0]

def _narrow_current_fields(observation):
    f=observation.facts;case=f.run.case
    def fault_events(nr=None):
        lo,hi,before,after=f.fault_span()
        return f.events(nr,before.pid,lo,hi)
    def unclosed():
        lo,hi,before,after,stopped=f.stop_span();fd=case.resources['actual_unclosed_owned_fd']
        state=before.semantic_read['state'];physical.need(state['point']=='close' and state['actual']['fd']==fd,
            'EXACT_OMITTED_CURRENT_CLOSE_POINT')
        row=next(r for r in f.registry(before)['fds'] if r['fd']==fd and not r['closed'])
        physical.need(row.get('phase')==f.registry(before)['active_phase'] and row.get('lifetime')=='phase',
            'EXACT_CURRENT_EFFECT_RESOURCE')
        physical.need(f.row_physically_live(before,row) and f.row_physically_live(after,row) and
            f.row_physically_live(stopped,row) and not f.closes(fd,before.pid,row['identity'],lo,f._seq(stopped)),
            'REAL_SAME_OWNED_RESOURCE_LIVE_THROUGH_STOP')
        return f.proof('exact_current_unclosed_effect_to_STOP',True,(lo,f._seq(stopped)))
    def inode_drift():
        lo,hi,before,after=f.fault_span();fd=case.resources['actual_fault_owned_fd']
        old=next(r for r in before.all_fd_reads if r.fd==fd)
        new=next(r for r in after.all_fd_reads if r.fd==fd)
        physical.need(old.identity!=new.identity and any(e.result==fd and
            ((e.syscall==33 and e.arguments[1]==fd) or (e.syscall==292 and e.arguments[1]==fd))
            for e in fault_events()),'EXACT_REAL_DUP2_TARGET_INODE_DRIFT')
        return f.proof('exact_original_changed_target_inode',(old.identity!=new.identity),(lo,hi))
    def replay():
        lo,hi,before,after=f.fault_span();completed=f.registry(before)['completed_phases']
        physical.need(completed and before.semantic_read['state']['point']=='begin_phase','EXACT_CURRENT_REPLAY_REQUEST')
        index=case.phase_order.index(completed[0])
        physical.need(any(m.values[0]==index and lo<m.kernel_sequence<hi for m in f.mark('phase_begin',f.root)),
                      'SECOND_ACTUAL_BEGIN_OF_SAME_COMPLETED_PHASE')
        return True
    def exact_scan(role):
        candidates=[]
        for start,end,close,dents in observation.scan():
            hygiene=[b for b in f.states(8,start.pid) if f._seq(b)<=start.kernel_sequence]
            if not hygiene:continue
            declared=hygiene[-1].semantic_read['state'].get('role')
            # Existing child hygiene has a canonical role parameter; if absent,
            # exact inherited pidfd role+raw owned wait is required, never guessed.
            if declared==role:candidates.append((start,end,close,dents))
        if not candidates:
            target=f.role_wait('post_terminal_helper_pidfd' if role=='helper' else 'stock_pidfd')['target_pid']
            candidates=list(observation.scan(target))
        physical.need(len(candidates)==1,'ONE_EXACT_ROLE_SCAN_INTERVAL:'+role)
        return tuple(candidates)
    fields={
      'actual_current_effect_fd_still_live_raw_ref':unclosed,
      'future_original_and_changed_inode_raw_refs':inode_drift,
      'actual_original_guardian_pidfd_drift_raw_ref':inode_drift,
      'actual_close_error_raw_ref':lambda:f.proof('exact_control_close_error',
          any(e.arguments[0]==case.resources['actual_fault_owned_fd'] and e.result==-9 for e in fault_events(3))),
      'second_begin_same_phase_requested':replay,
      'exact_direct_helper_raw_wait_ref':lambda:f.proof('exact_helper_role_raw_direct_wait',f.role_wait('post_terminal_helper_pidfd')['kind']=='direct'),
      'fresh_helper_birth_absence_ref':lambda:f.proof('exact_helper_role_wait_fresh_absence',f.role_wait('post_terminal_helper_pidfd')['fresh_exact_absent']),
      'exact_helper_direct_wait_status':lambda:int.from_bytes(f.role_wait('post_terminal_helper_pidfd')['raw_status_bytes'],'little'),
      'fresh_exact_helper_birth_absence_ref':lambda:f.proof('same_helper_raw_wait_fresh_absence',f.role_wait('post_terminal_helper_pidfd')['fresh_exact_absent']),
      'completed_phase_appends':lambda:sum(m.kernel_sequence>f.fault_span()[0] for m in f.mark('phase_closed')),
      'normal_current_phase_credit':False,
    }
    if case.control_id in ('C75_STOCK_EXACT_SCAN_FD','C75_HELPER_EXACT_SCAN_FD'):
        role='stock' if case.control_id=='C75_STOCK_EXACT_SCAN_FD' else 'helper'
        fields.update(actual_opendir_dirfd_readdir_closedir_raw_ref=lambda:f.proof('exact_role_scan_raw',bool(exact_scan(role))),
            scan_closedir_calls=lambda:len(exact_scan(role)),
            exact_scan_owned_fd_excluded=lambda:exact_scan(role)[0][0].values[0],
            actual_dirfd_return=lambda:exact_scan(role)[0][0].values[0])
    relation=case.declared_delta.get('exact_alias_relation')
    if relation is not None:
        def alias_proof():
            b=f.aux_inventory();a=next(r for r in b.all_fd_reads if r.fd==relation['reference_fd'])
            c=next(r for r in b.all_fd_reads if r.fd==relation['alias_fd'])
            physical.need(a.identity==tuple(relation['reference_identity']) and c.identity==tuple(relation['alias_identity']),
                          'EXACT_ALIAS_REFERENCE_CURRENT_KERNEL_READ')
            physical.need(a.mount_id==c.mount_id if case.mutation=='SAME_AUXILIARY_OUTPUT_MOUNT' else a.identity[:2]==c.identity[:2],
                          'EXACT_SEMANTIC_ALIAS_RELATION')
            return f.proof('exact_semantic_alias/'+case.mutation,True,(relation['reference_fd'],relation['alias_fd']))
        for key in ('distinct_output_FDs_same_mount_identity_raw_ref','distinct_fds_same_borrowed_owned_inode_raw_ref',
                    'borrowed_fd_same_output_file_inode_raw_ref','actual_duplicate_inode_ref','different_FD_same_reserved_inode_raw_ref',
                    'actual_distinct_FDs_same_clean_group_inode_ref'):fields[key]=alias_proof
    observation.fields.update(fields)
    return observation

def _exact_four_cut_operation(f,case_id):
    number={'K120_PIDFD_FSTAT':7,'K120_MAP_GO':8,'K120_INIT_WAIT':9,'K120_PIPE_CLOSE':10}.get(case_id)
    physical.need(number is not None,'EXACT_ORIGINAL_FOUR_CUT_CAUSE')
    row=f.operation(number,case_id);lo=row['entry_sequence'];hi=row['return_sequence']
    selected=f.events(pid=row['pid'],lo=lo,hi=hi)
    if number==7:
        actual=[e for e in selected if e.syscall==5 and e.result==0 and
            type(e.fd_before)is physical.FDRead and 'pidfd' in e.fd_before.link and e.raw_output]
    elif number==8:
        actual=[e for e in selected if e.syscall==1 and e.result==1 and e.raw_input==b'G']
        physical.need(any(e.syscall==1 and type(e.fd_before)is physical.FDRead and
            e.fd_before.link.endswith('/uid_map') for e in selected) and
            any(e.syscall==1 and type(e.fd_before)is physical.FDRead and e.fd_before.link.endswith('/gid_map')
                for e in selected),'EXACT_REAL_UID_GID_MAP_BEFORE_NAMESPACE_GO')
    elif number==9:
        actual=list(f.raw_waits(row['pid'],lo,hi))
    else:
        actual=[e for e in selected if e.syscall==3 and e.result==0 and
            type(e.fd_before)is physical.FDRead and e.fd_before.link.startswith('pipe:')]
    physical.need(len(actual)==1,'ONE_EXACT_CURRENT_ORIGINAL_OPERATION_BEFORE_SOURCE_UNCERTAINTY')
    return f.proof('exact_four_cut_real_operation/'+case_id,True,(lo,actual[0].sequence,hi))
