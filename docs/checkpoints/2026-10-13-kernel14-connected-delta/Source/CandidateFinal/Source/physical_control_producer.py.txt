"""Connected C83 control drivers; effects are supplied genuine owned fixtures.

All physical preparation, existing-GNU/Root invocation, source loading and
resource topology are the original outside owner's separately admitted job.
This Source does NOT allocate substitute capabilities or manufacture authority.
Each named attempt has one raw origin before any attempted caller/Root bind.
"""
import copy
import os
from dataclasses import dataclass
import physical_observation_primitives as physical
import physical_observation_markers as markers

C83=(
 ('C83_PARENT_UID_LAST_ZERO','zero','DIRECTORY_CAPABILITY_UID_MODE','uid'),
 ('C83_PARENT_GID_LAST_ZERO','zero','DIRECTORY_CAPABILITY_UID_MODE','gid'),
 ('C83_PARENT_MODE_LAST_ZERO','zero','DIRECTORY_CAPABILITY_UID_MODE','mode'),
 ('C83_PARENT_IDENTITY_LAST_ZERO','zero','DIRECTORY_CAPABILITY_ACTUAL_IDENTITY','identity'),
 ('C83_PARENT_PATH_LAST_ZERO','zero','DIRECTORY_CAPABILITY_PATH','path'),
 ('C83_IDENTICAL_PARENT_DUPLICATES_POSITIVE','positive',None,'identical'),
 ('C83_ADDITIONAL_PARENT_CAPABILITIES_POSITIVE','positive',None,'extras'),
 ('C83_ALL23_ORIGINAL_CALLER_ASSOCIATIONS_POSITIVE','positive',None,'original23'),
 ('C83_WRONG_GENUINE_GNU_HOLDER_LAST_ROOT_ZERO','zero',
     'AUX_ORIGINAL_CALLER_ASSOCIATION_ZERO_PREFIX','gnu_holder'),
 ('C83_WRONG_HOLDER_IN_EMPTY_PHASE_LAST_ROOT_ZERO','zero',
     'AUX_ORIGINAL_CALLER_ASSOCIATION_ZERO_PREFIX','empty_holder'),
 ('C83_CALLER_SOURCE_BINDING_LAST_ROOT_ZERO','zero',
     'AUX_ORIGINAL_CALLER_BINDING_ZERO_PREFIX','source'),
 ('C83_CALLER_SLOT_BINDING_LAST_ROOT_ZERO','zero',
     'AUX_ORIGINAL_CALLER_BINDING_ZERO_PREFIX','slot'),
 ('C83_REMOTE_MAIN_ROLE_LAST_ROOT_ZERO','zero',
     'AUX_CALLER_SEMANTIC_BINDING_ZERO_PREFIX','main_role'),
 ('C83_MAIN_PARENT_POLICY_LAST_ZERO','zero','DIRECTORY_CAPABILITY_UID_MODE','main_policy'),
 ('C83_WHOLE_MAIN_OUTPUT_INVENTORY_LAST_ZERO','zero',
     'AUX_MAIN_OUTPUT_INVENTORY_ZERO_PREFIX','main_inventory'),
 ('C83_LATE_CONSUMER_RECHECKS_ORIGINAL_ASSOCIATION','late',
     'AUX_ORIGINAL_CALLER_ASSOCIATION_ZERO_PREFIX','late_holder'))
EMPTY=('ALL_SAFE','K120_PIDFD_FSTAT','K120_MAP_GO','K120_INIT_WAIT','K120_PIPE_CLOSE')
ROOT_ONLY=('gnu_holder','empty_holder','source','slot','main_role')
@dataclass(frozen=True)
class CaseDefinition:
    control_id: str
    kind: str
    expected_refusal: object
    mutation: str
    last_phase: str
    phase_order: tuple
    caller_caps: object
    root_caps: object
    declarations_before: object
    declared_delta: object
    caller_pid: int
    root_pid: int
    true_original_holder: object
    resources: object
    def phase_caps_for_pid(self,pid):
        if pid==self.caller_pid:return self.caller_caps
        if pid==self.root_pid:return self.root_caps
        raise physical.ObservationMissing('NO_DECLARATION_FOR_UNKNOWN_SUBJECT')

def _ordered_last(caps,phase):
    return {p:copy.deepcopy(c) for p,c in caps.items() if p!=phase}|{phase:copy.deepcopy(caps[phase])}

def _all_parents(caps):
    out={}
    for cap in caps.values():
        for row in (cap['group_parent_capability'],
                    *cap['private_directory_capabilities'].values(),
                    *cap.get('auxiliary_parent_capabilities',())):
            # Only batch-lifetime parents, never control/trace child directories.
            path=row['path']
            if row in cap['private_directory_capabilities'].values():
                if row not in (cap['private_directory_capabilities'][k]
                               for k in ('output_parent','control_parent','trace_parent')):
                    continue
            out.setdefault(row['fd'],copy.deepcopy(row))
    return tuple(out.values())

def define_c83_case(control_id,actual_caps,actual_context,resources):
    """Only DECLARED metadata changes here. No actual inode/policy/birth changes.

    Real alternate slots, extra parents, different Source pins, and GNU inherited
    handles must already exist and have their separately admitted outside owner.
    Caller/root identities are read from the kernel, never fixture constants.
    """
    physical.hard_caps()
    rows=[r for r in C83 if r[0]==control_id]
    physical.need(len(rows)==1,'EXACT_C83_CONTROL_ID')
    _,kind,refusal,mutation=rows[0]
    physical.need(type(actual_caps)is dict and len(actual_caps)==23,'COMPLETE_CURRENT23_CAPABILITIES')
    original=copy.deepcopy(actual_caps)
    caller=physical.process_read(actual_context['actual_original_caller_pid'])
    root=physical.process_read(actual_context['actual_Root_pid'])
    physical.need(caller.pid!=root.pid,'DISTINCT_GENUINE_CALLER_ROOT_ROLES')
    last=next(reversed(actual_caps))
    if mutation=='empty_holder':
        last=resources['empty_phase_id']
        physical.need(last in EMPTY,'GENUINE_EMPTY_PHASE_REQUIRED')
        a=actual_caps[last].get('actual_control_binding',{})
        physical.need(not any(a.get(k,()) for k in
            ('auxiliary_groups','auxiliary_outputs','auxiliary_fds')),
            'EMPTY_PHASE_CANNOT_BE_FABRICATED')
    caps=_ordered_last(actual_caps,last)
    before=copy.deepcopy(caps)
    phase=caps[last]
    delta={}
    if mutation in ('uid','gid','mode','identity','path','identical'):
        main=copy.deepcopy(phase['private_directory_capabilities']['output_parent'])
        parents=_all_parents(caps)
        distinct=[r for r in parents if r['fd']!=main['fd']]
        physical.need(len(distinct)>=3,'THREE_DISTINCT_GENUINE_VALID_PARENT_ROWS_UNBOUND')
        phase['auxiliary_parent_capabilities']=[
            *phase.get('auxiliary_parent_capabilities',()),*distinct[:3],copy.deepcopy(main)]
        bad=copy.deepcopy(main)
        if mutation in ('uid','gid'):
            bad['policy'][mutation]+=1
        elif mutation=='mode':bad['policy']['mode']=0o755
        elif mutation=='identity':
            actual_other=resources['distinct_actual_parent_capability']
            physical.need(actual_other['identity']!=main['identity'],'REAL_DIFFERENT_INODE_REQUIRED')
            bad['identity']=copy.deepcopy(actual_other['identity'])
        elif mutation=='path':bad['path']=main['path']+'/..'
        phase['auxiliary_parent_capabilities'].append(bad)
        delta={'field':mutation,'good':main,'last_bad':bad,'earlier_valid':distinct[:3]}
    elif mutation=='extras':
        extras=resources['valid_extra_parent_capabilities']
        physical.need(len(extras)>=2 and any(PathCap['path'].startswith('/sys/fs/cgroup/')
            for PathCap in extras) and any(not PathCap['path'].startswith('/sys/fs/cgroup/')
            for PathCap in extras),'REAL_GROUP_AND_PRIVATE_EXTRA_PARENTS_REQUIRED')
        phase['auxiliary_parent_capabilities']=[*phase.get('auxiliary_parent_capabilities',()),
                                                *copy.deepcopy(extras)]
        delta={'extra_parents':copy.deepcopy(extras)}
    elif mutation in ('gnu_holder','empty_holder','late_holder'):
        # Actual GNU is live, has the SAME real inherited FDs, and is still the
        # original caller's child. Only its declaration as original holder is bad.
        wrong=physical.process_read(actual_context['actual_GNU_pid'])
        physical.need(wrong.pid!=caller.pid and wrong.ppid==caller.pid,'WRONG_GENUINE_GNU_RELATION')
        holder=copy.deepcopy(phase['retained_existing_caller'])
        inventory=holder['auxiliary_capabilities']
        for row in inventory:
            observed=physical.fd_read(wrong,row['fd'])
            physical.need(observed.identity==tuple(row['identity']),'GNU_REAL_INHERITED_ORIGINAL_FD_REQUIRED')
        if mutation!='late_holder':phase['retained_existing_caller']=dict(holder,birth=wrong.birth_dict())
        delta={'wrong_birth':wrong.birth_dict(),'bad_phase':last,'wrong_inventory':inventory}
    elif mutation=='source':
        supplied=resources['different_actual_source_pin']
        physical.same_read_source(supplied)
        holder=phase['retained_existing_caller']
        physical.need(supplied!=holder['source_pin'],'GENUINELY_DISTINCT_SOURCE_PIN_REQUIRED')
        delta={'supplied_source_pin':supplied,'original_source_pin':holder['source_pin']}
        holder['source_pin']=copy.deepcopy(supplied)
    elif mutation=='slot':
        alternate=resources['genuine_distinct_slot_fd']
        actual=physical.fd_read(caller,alternate)
        holder=phase['retained_existing_caller']
        physical.need(alternate!=holder['slot_fd'] and actual.identity[2]!=0,
                      'GENUINE_DIFFERENT_LIVE_SLOT_HANDLE_REQUIRED')
        delta={'supplied_slot_fd':alternate,'original_slot_fd':holder['slot_fd'],'actual_alternate':actual}
        holder['slot_fd']=alternate
    elif mutation=='main_role':
        holder=phase['retained_existing_caller']
        rows=[r for r in holder['auxiliary_capabilities'] if r['fd']==holder['group_fd']]
        physical.need(len(rows)==1 and rows[0]['role']=='original_caller_group','GENUINE_MAIN_ROLE_INVENTORY')
        delta={'fd':rows[0]['fd'],'actual_identity':rows[0]['identity'],
               'original_role':rows[0]['role'],'supplied_role':'auxiliary_borrowed'}
        rows[0]['role']='auxiliary_borrowed'
    elif mutation=='main_policy':
        parent=phase['private_directory_capabilities']['output_parent']
        delta={'good':copy.deepcopy(parent),'field':'mode'}
        parent['policy']['mode']=0o755
        delta['last_bad']=copy.deepcopy(parent)
    elif mutation=='main_inventory':
        inventory=phase['output_owned_fds']
        physical.need(len(inventory)>=2,'GENUINE_MULTIPLE_ORIGINAL_MAIN_MOUNT_FDS_REQUIRED')
        removed=next(r for r in inventory if r[0]!=phase['output_fd'])
        physical.fd_read(root,removed[0])
        phase['output_owned_fds']=[r for r in inventory if r is not removed]
        delta={'omitted_original':copy.deepcopy(removed),'whole_original':copy.deepcopy(inventory)}
    elif mutation=='original23':
        delta={'original':True}
    else:raise physical.ObservationMissing('C83_CAUSAL_PRODUCER_MISSING')
    # Caller path is deliberately unchanged for direct-Root-only malformed input:
    # the genuine connected caller correctly overwrites all returned holders.
    caller_caps=before if mutation in ROOT_ONLY or mutation=='late_holder' else copy.deepcopy(caps)
    return CaseDefinition(control_id,kind,refusal,mutation,last,tuple(caps),
        caller_caps,caps,original,delta,caller.pid,root.pid,
        copy.deepcopy(actual_context['original_outer_holder_binding']),resources)

def run_original_caller_prepare(case,actual_context,adapter,custody_pin,observer_binding):
    physical.need(os.getpid()==case.caller_pid,'EXACT_EXISTING_ORIGINAL_CALLER_ROLE')
    # Actual same-read accepted adapter/module identity comes from real outside
    # loader; no callback/provider is substituted here.
    physical.need(actual_context['actual_accepted_same_read_adapter_module'] is adapter,
                  'EXACT_CONNECTED_CALLER_SOURCE_OBJECT_UNBOUND')
    markers.configure(observer_binding)
    adapter.root_fd_custody._physical_capture=markers
    markers.mark('attempt_begin')
    code=None;result=None
    try:
        result=adapter.kernel120_existing_caller_prepare(actual_context,
            physical.process_read(os.getpid()).birth_dict(),case.caller_caps,custody_pin)
        markers.mark('caller_prepare_complete',len(result))
        route=actual_context.get('current_private_control_spec_route')
        physical.need(type(route)is dict,'CURRENT_EXISTING_SPEC_ENCODER_BINDING_UNBOUND')
        actual_context['current_private_control_spec_ref']=encode_current_control(case,route['spec'],
            route['actual_same_read_owner_module'],route['custody'],route['control_fd'],route['component'])
    except BaseException as error:
        code=markers.refusal(error)
        holder=actual_context.get('kernel120_holder')
        if holder is not None:holder.stop_retaining(error)
        markers.mark('STOP')
    finally:
        if case.kind=='zero' or code is not None:markers.mark('attempt_end')
    actual_context['physical_positive_capture_open']=case.kind!='zero' and code is None
    return {'actual_exception_code':code,'prepared_actual_caps':result,
            'actual_context':actual_context,'observation_is_not_acceptance':True}

def run_fixed_root_entry(case,spec,ref,modules,observer_binding):
    physical.need(os.getpid()==case.root_pid,'EXACT_EXISTING_GNU_CHILD_ROOT_ROLE')
    owner=modules['stock_install_owner']
    physical.need(owner is modules['actual_accepted_same_read_owner_object'],
                  'EXACT_CURRENT_ROOT_SOURCE_OBJECT_UNBOUND')
    # Outer independently observed original holder remains correct. The bad
    # phase holder affects ONLY one LAST row of direct Root input.
    candidate=copy.deepcopy(spec)
    candidate['kernel120_named_root_binding']['actual_phase_capabilities']=case.root_caps
    physical.need(candidate['kernel120_named_root_binding']['retained_existing_caller']
                  ==case.true_original_holder,'OUTER_ORIGINAL_CALLER_CANNOT_BE_CHANGED')
    markers.configure(observer_binding)
    install_current_case(case)
    if case.kind=='stop':
        modules['physical_legacy_controls'].install_current_fault(case)
    modules['root_fd_custody']._physical_capture=markers
    markers.mark('attempt_begin')
    code=None;result=None
    try:
        result=owner.kernel120_named_root_entry(candidate,ref,modules)
    except BaseException as error:
        code=markers.refusal(error)
        # No guessed close/retry/normal terminal; outer original caller stays
        # alive and retains every capability. The observer independently checks.
        markers.mark('STOP')
    finally:markers.mark('attempt_end')
    return {'actual_exception_code':code,'actual_result':result,
            'observation_is_not_acceptance':True}

def run_late_exact_consumer(case,owner,actual,context,clock,custody):
    """Separate late interval; it can NEVER earn whole-prefix zero credit."""
    physical.need(case.mutation=='late_holder' and os.getpid()==case.root_pid,'EXACT_LATE_ROLE')
    physical.need(context['actual_complete_whole23_preflight_observation_ref'] is not None,
                  'PREVIOUS_REAL_WHOLE23_PREFLIGHT_UNBOUND')
    candidate=copy.deepcopy(context)
    candidate['retained_existing_caller']=dict(
        candidate['retained_existing_caller'],birth=case.declared_delta['wrong_birth'])
    candidate['physical_current_auxiliary_scope']=(case.phase_order.index(case.last_phase)+1,
        current_control_word(case),1)
    try:
        return owner.kernel120_preflight_auxiliary_control(actual,candidate,clock,custody)
    except BaseException as error:
        markers.refusal(error)
        owner.root_fd_custody.stopped(custody,error)
        markers.mark('STOP')
        raise

def run_current_assertions(actual_observer_module,observed_runs,test_modules):
    """Full exact consumers, no t supplied by a fixture."""
    expected={r[0] for r in C83}
    physical.need(set(observed_runs)==expected,'ALL16_ACTUAL_RUNS_REQUIRED')
    results=[]
    for control_id,kind,refusal,mutation in C83:
        run=observed_runs[control_id]
        observation=actual_observer_module.project_c83(run)
        name=actual_observer_module.C83_ASSERTIONS[control_id]
        test=getattr(test_modules['C83'],name)
        try:test(observation)
        except BaseException as error:
            results.append({'id':control_id,'state':'FAIL','error':type(error).__name__})
        else:results.append({'id':control_id,'state':'PASS'})
    return {'results':results,'all16':len(results),'source_author_acceptance':False,
            'runtime_acceptance':False,'capacity_release':False}

_active_current_case=None

def install_current_case(case):
    global _active_current_case
    physical.need(_active_current_case is None,'ONE_CURRENT_CASE_NO_REPLAY')
    physical.need(os.getpid()==case.root_pid,'EXACT_CURRENT_ROOT_CASE_ROLE')
    _active_current_case=case

def current_auxiliary_context(phase_id,context):
    case=_active_current_case
    if case is None:return context
    physical.need(phase_id in case.phase_order,'EXACT_CURRENT_AUXILIARY_PHASE')
    candidate=copy.copy(context)
    late=case.mutation=='late_holder' and phase_id==case.last_phase
    candidate['physical_current_auxiliary_scope']=(case.phase_order.index(phase_id)+1,
        current_control_word(case),int(late))
    if not late:return candidate
    # Only the LAST exact late consumer declaration changes. Earlier whole23
    # preflight and the independently admitted outer caller remain untouched.
    candidate['retained_existing_caller']=dict(context['retained_existing_caller'],
        birth=case.declared_delta['wrong_birth'])
    return candidate

def current_control_word(case):
    from physical_legacy_controls import CONTROLS
    identifiers=tuple(row[0] for row in C83)+tuple(CONTROLS)
    physical.need(case.control_id in identifiers,'EXACT_CURRENT_AUXILIARY_CONTROL')
    return identifiers.index(case.control_id)+1

def is_current_late_phase(phase_id):
    case=_active_current_case
    return case is not None and case.mutation=='late_holder' and phase_id==case.last_phase

def run_current_caller_terminal(case,actual_context,adapter,guard_birth,guard_pidfd,clock):
    physical.need(os.getpid()==case.caller_pid,'REAL_ORIGINAL_CALLER_TERMINAL_ROLE')
    physical.need(actual_context.get('physical_positive_capture_open') is True,
                  'CURRENT_PREPARE_NOT_ACTUALLY_OPEN_FOR_TERMINAL')
    try:
        result=adapter.kernel120_existing_caller_terminal(actual_context,guard_birth,guard_pidfd,clock)
        holder=actual_context['kernel120_holder']
        event='target_owner_terminal' if result['capacity_release'] is True else 'target_owner_stop'
        markers.mark(event)
        markers.observe_state(event,{'custody':holder.registry,'guard_birth':guard_birth,'clock':clock})
        return result
    finally:
        markers.mark('attempt_end')
        actual_context['physical_positive_capture_open']=False

CONTROL_SPEC_KEY='kernel120_current_control'
CONTROL_SPEC_SCHEMA='C100.current.private.control.v1'
CONTROL_FIELDS=tuple(CaseDefinition.__dataclass_fields__)
CONTROL_FDREAD_TAG='__frw_physical_fdread_v1__'

def _fdread_fields(value):
    """Typed declaration only; the observer must re-read its actual capability."""
    fields=tuple(physical.FDRead.__dataclass_fields__)
    physical.need(type(value)is dict and set(value)==set(fields),'EXACT_FDREAD_CODEC_FIELDS')
    integer_fields=('pid','fd','uid','gid','mode','nlink','mount_id','flags')
    physical.need(all(type(value[k])is int and value[k]>=0 for k in integer_fields) and
        value['pid']>0 and value['fd']>2 and value['mode']<=0o7777,
        'EXACT_FDREAD_CODEC_INTEGERS')
    physical.need(type(value['birth'])is str and value['birth'].isdecimal() and
        int(value['birth'])>0 and type(value['link'])is str and type(value['info_raw'])is str,
        'EXACT_FDREAD_CODEC_TEXT')
    physical.need(type(value['identity'])in (tuple,list) and len(value['identity'])==3 and
        all(type(x)is int and x>=0 for x in value['identity']) and
        type(value['locks'])in (tuple,list) and all(type(x)is str for x in value['locks']),
        'EXACT_FDREAD_CODEC_IDENTITY_LOCKS')
    return dict(value,identity=tuple(value['identity']),locks=tuple(value['locks']))

def _encode_control_value(value,depth=0):
    physical.need(depth<=64,'FINITE_EXISTING_CONTROL_VALUE_DEPTH')
    if type(value)is physical.FDRead:
        fields=_fdread_fields({name:getattr(value,name) for name in physical.FDRead.__dataclass_fields__})
        return {CONTROL_FDREAD_TAG:{name:_encode_control_value(item,depth+1) for name,item in fields.items()}}
    if type(value)is dict:
        physical.need(all(type(key)is str for key in value) and CONTROL_FDREAD_TAG not in value,
            'EXACT_EXISTING_CONTROL_KEYS')
        return {key:_encode_control_value(item,depth+1) for key,item in value.items()}
    if type(value)in (list,tuple):return [_encode_control_value(item,depth+1) for item in value]
    physical.need(value is None or type(value)in (str,int,float,bool),'UNSUPPORTED_EXISTING_CONTROL_VALUE')
    return value

def _decode_control_value(value,depth=0):
    physical.need(depth<=64,'FINITE_EXISTING_CONTROL_VALUE_DEPTH')
    if type(value)is dict:
        if CONTROL_FDREAD_TAG in value:
            physical.need(set(value)=={CONTROL_FDREAD_TAG},'EXACT_FDREAD_CODEC_TAG')
            return physical.FDRead(**_fdread_fields(value[CONTROL_FDREAD_TAG]))
        return {key:_decode_control_value(item,depth+1) for key,item in value.items()}
    if type(value)is list:return [_decode_control_value(item,depth+1) for item in value]
    physical.need(value is None or type(value)in (str,int,float,bool),'UNSUPPORTED_EXISTING_CONTROL_VALUE')
    return value

def encode_current_control(case,spec,owner,custody,control_fd,component):
    """Extend the existing private JSON/ref spec; no new transport or authority.
    Existing original caller supplies its exact declaration after preparation.
    The same owner.kernel120_record encoder and owner.main same_read decoder
    remain the only file/ref route. The encoded object has no live grant.
    """
    import json
    physical.need(os.getpid()==case.caller_pid,'EXACT_CALLER_CONTROL_ENCODER')
    physical.qualify_current_function(owner,spec['kernel120_root_guard']['owner_source'],'kernel120_record')
    physical.need(type(component)is str and component and '/' not in component,'EXACT_PRIVATE_CONTROL_COMPONENT')
    payload={name:_encode_control_value(getattr(case,name)) for name in CONTROL_FIELDS}
    # Validate every complete leaf before the first existing record write.
    documents=[]
    for field in ('caller_caps','root_caps','declarations_before'):
        physical.need(set(payload[field])==set(case.phase_order),'ALL23_CONTROL_SOURCE_MAP')
        documents.extend({'phase':phase,'capability':payload[field][phase]} for phase in case.phase_order)
    documents.extend(payload[field] for field in ('resources','declared_delta','true_original_holder'))
    documents.append({key:value for key,value in payload.items() if key not in
        ('caller_caps','root_caps','declarations_before','resources','declared_delta','true_original_holder')})
    for document in documents:
        physical.need(len(json.dumps(document,sort_keys=True,separators=(',',':'),allow_nan=False).encode())<=65536,
            'BOUNDED_EXISTING_CONTROL_LEAF_BEFORE_WRITE')
    # Reuse the existing private JSON/ref record route for bounded leaves. The
    # three full all23 maps are not embedded repeatedly into one64KiB record.
    leaves={}
    for field in ('caller_caps','root_caps','declarations_before'):
        mapping=payload.pop(field)
        physical.need(set(mapping)==set(case.phase_order),'ALL23_CONTROL_SOURCE_MAP')
        leaves[field]=[]
        for index,phase in enumerate(case.phase_order):
            name=component+'.'+field+'.'+str(index)+'.json'
            pin=owner.kernel120_record(custody,control_fd,name,{'phase':phase,'capability':mapping[phase]})
            leaves[field].append(pin)
    for field in ('resources','declared_delta','true_original_holder'):
        leaves[field]=owner.kernel120_record(custody,control_fd,component+'.'+field+'.json',payload.pop(field))
    raw=json.dumps(payload,sort_keys=True,separators=(',',':'),allow_nan=False).encode()
    physical.need(len(raw)<=65536,'BOUNDED_EXISTING_CONTROL_SPEC')
    candidate=copy.deepcopy(spec)
    physical.need(candidate['entry_role']=='kernel120_exterior_root','EXISTING_FIXED_ROOT_SPEC')
    physical.need(CONTROL_SPEC_KEY not in candidate,'NO_SECOND_CONTROL_SPEC_OR_REPLAY')
    candidate[CONTROL_SPEC_KEY]={'schema':CONTROL_SPEC_SCHEMA,'declaration':json.loads(raw),'private_leaves':leaves}
    candidate['kernel120_named_root_binding']['actual_phase_capabilities']={}
    # Strict decoder restores all23 before the existing installer sees the spec.

    # Exact accepted Source encoder, original control directory and finite clock
    # are the existing owner's already-admitted resources; JSON grants nothing.
    return owner.kernel120_record(custody,control_fd,component,candidate)

def decode_current_control(spec):
    payload=spec.get(CONTROL_SPEC_KEY)
    physical.need(type(payload)is dict and set(payload)=={'schema','declaration','private_leaves'} and
        payload['schema']==CONTROL_SPEC_SCHEMA,'EXACT_EXISTING_PRIVATE_CONTROL_SPEC')
    import json
    row=dict(payload['declaration']);leaves=payload['private_leaves']
    expected_leaves={'caller_caps','root_caps','declarations_before','resources','declared_delta','true_original_holder'}
    physical.need(type(leaves)is dict and set(leaves)==expected_leaves,'EXACT_PRIVATE_CONTROL_LEAF_FIELDS')
