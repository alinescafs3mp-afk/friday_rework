sys.dont_write_bytecode = True
def load(source, name):
    raw = read(source)
    assert name not in sys.modules, 'dedicated source module name already used'
    spec = importlib.util.spec_from_loader(name, loader=None, origin=source['path'])
    module = importlib.util.module_from_spec(spec)
    module.__file__ = source['path']
    sys.modules[name] = module
    exec(compile(raw, source['path'], 'exec'), module.__dict__)
    return module
