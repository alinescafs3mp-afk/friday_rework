def need(value, reason):
    if not value: raise PermissionError(reason)

def function(raw,name,owner=None):
    tree=ast.parse(raw)
    if owner: tree=next(n for n in tree.body if isinstance(n,ast.ClassDef) and n.name==owner)
    return copy.deepcopy(next(n for n in tree.body if isinstance(n,ast.FunctionDef) and n.name==name))

def expected_tree(context):
    """Keep every current6 authenticated entry; advance exactly its three outputs."""
    request=context['request']; published=context['published_refs']
    home=Path(request['home'])/'profiles'/request['profile']; folder=home/'workers'
    old=copy.deepcopy(request['old_tree'])
    need(set(published)=={'dsh','a0','attachment'},'current6_exact_publications_required')
    for key,rel in (('dsh','dsh/runtime-input.json'),('a0','a0/runtime-input.json'),('attachment','attachment.json')):
        need(published[key]['path']==str(folder/rel) and old[rel]['kind']=='file',
             'current6_tree_publication_path_changed')
        old[rel]['sha256']=published[key]['sha256']
    need(all(not any(part in {'qualification','bootstrap','runtime-receipt.json','probe-plan.json','current'}
             for part in Path(name).parts) for name in old),'current6_observed_tree_forbidden')
    files={name:{'path':str(folder/name),'sha256':row['sha256']} for name,row in old.items() if row['kind']=='file'}
    directories={name for name,row in old.items() if row['kind']=='directory'}
    need(len(files)+len(directories)==len(old) and '.' in directories,'current6_tree_kind_changed')
    return home,files,directories

def patch_namespace(namespace,onboarding_raw,current6_context,verify_preparation_inputs):
    need(hashlib.sha256(onboarding_raw).hexdigest()==NATIVE_SHA,'current6_native_source_changed')
    need(type(current6_context) is dict and callable(verify_preparation_inputs),'current6_authenticated_context_required')
    context=copy.deepcopy(current6_context)
    home,expected_files,expected_directories=expected_tree(context)
    scope=namespace['scope']; request=context['request']; service=context['service']
    original_location=namespace['_worker_preparation_location']
    original_successor=namespace['_worker_successor_location']
    prepare=function(onboarding_raw,'_prepare_workers','Onboarding')
    replace=function(onboarding_raw,'_replace_inactive_worker_inputs')
    def existing(root,actual_home,binding,generation,checked):
        if actual_home!=home: return False
        need(root==Path(request['home']) and binding==request['identity']['binding']
             and type(generation) is int and generation==request['identity']['generation']
             and checked['receipt']==service,'current6_exact_secondary_identity_required')
        target={'home':str(home),'profile':binding['runtime_profile'],'binding_sha256':scope._fingerprint(binding)}
        _,secondary=original_location(root,actual_home,binding,generation,checked,current_main=target)
        need(secondary is True,'current6_authenticated_secondary_required')
        return True
    def location(root,actual_home,binding,generation,checked):
        if not existing(root,actual_home,binding,generation,checked):
            return original_successor(root,actual_home,binding,generation,checked)
        verify_preparation_inputs()
        target={'home':str(home),'profile':binding['runtime_profile'],'binding_sha256':scope._fingerprint(binding)}
        tx,secondary=original_location(root,actual_home,binding,generation,checked,current_main=target)
        need(secondary is True,'current6_authenticated_secondary_required')
        return tx,context['published_refs']['attachment']
    def stale(old,declaration,actual_home):
        if actual_home!=home:
            return old['installation_config_sha256']==declaration['installation_config_sha256']
        need(old==context['attachment']
             and declaration['installation_config_sha256']==service['after']['config.yaml']
             and old['installation_config_sha256']==declaration['installation_config_sha256'],
             'current6_existing_attachment_changed')
        return False
    def retained(checked,actual_home):
        if actual_home!=home: return checked['request']['retained']
        need(checked['receipt']==service,'current6_checked_service_changed')
        return {str(Path(ref['path']).relative_to(Path(request['home']))):ref for ref in expected_files.values()}
    def inventory(actual_home,names,directories):
        if actual_home!=home:return names,directories
        need(names<=set(expected_files) and directories<=expected_directories,'current6_required_inputs_missing')
        return set(expected_files),set(expected_directories)
    # The original local clone remains byte-for-byte AST-identical outside these
    # exact native boundary substitutions. Original imports and effects stay native.
    changes={'fresh':0,'location':0,'stale':0,'retained':0,'inventory':0}
    for node in ast.walk(prepare):
        if isinstance(node,ast.If) and isinstance(node.test,ast.Name) and node.test.id=='fresh_secondary' and len(node.body)==1 and isinstance(node.body[0],ast.Raise) and any(
            isinstance(x,ast.Constant) and x.value=='worker_successor_fresh_secondary_requires_absent_state' for x in ast.walk(node.body[0])):
            # Only the early existing-tree branch has a bare fresh_secondary test.
            node.test=ast.parse('fresh_secondary and not _S356_existing(root,home,binding,generation,checked)',mode='eval').body
            changes['fresh']+=1
    class Transform(ast.NodeTransformer):
        def visit_Call(self,node):
            self.generic_visit(node)
            if isinstance(node.func,ast.Name) and node.func.id=='_worker_successor_location':
                node.func.id='_S356_location';changes['location']+=1
            return node
        def visit_Compare(self,node):
            if ast.unparse(node)=="old['installation_config_sha256'] == declaration['installation_config_sha256']":
                changes['stale']+=1
                return ast.parse('_S356_stale(old,declaration,home)',mode='eval').body
            return self.generic_visit(node)
        def visit_Subscript(self,node):
            if ast.unparse(node)=="checked['request']['retained']":
                changes['retained']+=1
                return ast.parse('_S356_retained(checked,home)',mode='eval').body
            return self.generic_visit(node)
    replace=Transform().visit(replace)
    for i,node in enumerate(replace.body):
        if isinstance(node,ast.Assign) and any(isinstance(t,ast.Name) and t.id=='before' for t in node.targets):
            replace.body.insert(i,ast.parse('names,directories=_S356_inventory(home,names,directories)').body[0]);changes['inventory']+=1;break
    need(changes=={'fresh':1,'location':1,'stale':1,'retained':2,'inventory':1},'current6_exact_native_transform_changed')
    inverse_prepare=copy.deepcopy(prepare)
    for node in ast.walk(inverse_prepare):
        if isinstance(node,ast.If) and isinstance(node.test,ast.BoolOp) and ast.unparse(node.test)=='fresh_secondary and (not _S356_existing(root, home, binding, generation, checked))':
            node.test=ast.Name(id='fresh_secondary',ctx=ast.Load())
    class Inverse(ast.NodeTransformer):
        def visit_Call(self,node):
            if isinstance(node.func,ast.Name):
                if node.func.id=='_S356_location':node.func.id='_worker_successor_location'
                elif node.func.id=='_S356_stale':return ast.parse("old['installation_config_sha256'] == declaration['installation_config_sha256']",mode='eval').body
                elif node.func.id=='_S356_retained':return ast.parse("checked['request']['retained']",mode='eval').body
            return self.generic_visit(node)
    inverse_replace=Inverse().visit(copy.deepcopy(replace))
    inverse_replace.body=[n for n in inverse_replace.body if not (isinstance(n,ast.Assign) and isinstance(n.value,ast.Call) and isinstance(n.value.func,ast.Name) and n.value.func.id=='_S356_inventory')]
    normalized=lambda n:ast.dump(n,include_attributes=False)
    need(normalized(inverse_prepare)==normalized(function(onboarding_raw,'_prepare_workers','Onboarding'))
         and normalized(inverse_replace)==normalized(function(onboarding_raw,'_replace_inactive_worker_inputs')),
         'current6_exact_source_inverse_failed')
    namespace['_S356_NATIVE_TRANSFORM_PROOF']={'changes':changes,'inverse_exact':True,'native_sha256':NATIVE_SHA}
    namespace.update(_S356_existing=existing,_S356_location=location,_S356_stale=stale,
                     _S356_retained=retained,_S356_inventory=inventory)
    exec(compile(ast.fix_missing_locations(ast.Module(body=[replace],type_ignores=[])),
         '<S356-exact-local-native-replacement>','exec'),namespace)
    return prepare
