def dockerd_child_environment():
    # RootlessKit documents these three child variables (since v0.3/v0.8).
    # Preserve genuine values; identity/map guards already require UID == GID.
    rootlesskit = {k: os.environ.get(k) for k in
                   ('ROOTLESSKIT_STATE_DIR', 'ROOTLESSKIT_PARENT_EUID', 'ROOTLESSKIT_PARENT_EGID')}
    require(rootlesskit['ROOTLESSKIT_STATE_DIR'] == str(STATE / 'rootlesskit')
            and rootlesskit['ROOTLESSKIT_PARENT_EUID'] == str(UID)
            and rootlesskit['ROOTLESSKIT_PARENT_EGID'] == str(UID), 'route_rootlesskit_environment_changed')
    env = {k: os.environ[k] for k in ('PATH', 'HOME', 'XDG_RUNTIME_DIR', 'DBUS_SESSION_BUS_ADDRESS', 'DOCKER_CONFIG') if k in os.environ}
    if os.environ.get('NOTIFY_SOCKET'): env['NOTIFY_SOCKET'] = os.environ['NOTIFY_SOCKET']
    env.update(rootlesskit)
    return env

    def wait_ready(self):
        # Only bounded harmless native GETs may be repeated. Create/start,
        # message/model work and uncertain control calls are never replayed.
        script = self.m.HEALTH_SCRIPT.replace('\n raise SystemExit(1)', '')
        if script == self.m.HEALTH_SCRIPT:
            raise HostUnavailable('a0_bootstrap_health_source_changed')
        for _ in range(self.original['seconds'] * 10 + 1):
            self.current()
            with self.runtime.locked():
                r = self.runtime.receipt()
                # Classify the exact owned object's startup state before a
                # live endpoint check can mask an asynchronous native failure.
                obj = self.runtime.inspect(r, stop_owned=True, timeout=min(3, self.left()))
                unit = self.runtime.supervisor.observe(self.runtime.association(r))
                if (unit.missing or unit.quiescent or unit.invocation_id != r['invocation_id']):
                    raise HostUnavailable('a0_bootstrap_native_start_lost')
                if obj['State']['Running'] and obj['State']['Pid'] > 0:
                    # Readiness and exec still require the full current runtime,
                    # resource and allocated-network boundary.
                    obj = self.runtime.inspect(r, timeout=min(3, self.left()))
                    if not obj['State']['Running'] or obj['State']['Pid'] <= 0:
                        raise HostUnavailable('a0_bootstrap_native_start_terminal')
                    sample = self.runtime.snapshot_container(obj)
                    r['observations'].append(sample)
                    self.runtime.known = r
                    self.m.write_json(self.runtime.receipt_path, r, replace=True)
                    reply = json.loads(self.runtime.docker('exec', '--workdir=/a0', r['container_id'],
                        '/opt/venv-a0/bin/python', '-B', '-c', script, timeout=min(4, self.left())))
                else:
                    # A healthy created object can still be awaiting native
                    # start. Reuse the exact never-started phase validator;
                    # failed-created and incomplete state remain failures.
                    self.m.container_checked(self.runtime.p, obj, r['container_id'], pre_start=True)
                    self.runtime.check_network(container_id=r['container_id'])
                    reply = {'http': None, 'status': 'NOT_READY'}
            self.current()
            if (set(reply) == {'http', 'gitinfo_present', 'native_error_present'}
                    and reply['http'] == 200 and reply['gitinfo_present'] is True
                    and reply['native_error_present'] is False): return
            if reply != {'http': None, 'status': 'NOT_READY'}:
                raise HostUnavailable('a0_bootstrap_native_health_refused')
            time.sleep(min(.1, self.left()))
        raise HostUnavailable('a0_bootstrap_startup_observation_limit')
