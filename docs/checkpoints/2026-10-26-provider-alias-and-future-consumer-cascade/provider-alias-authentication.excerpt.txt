                        alias_relative = _native_provider_alias_relative(
                            name, hasattr(module,'__path__'), receipt['files'])
                        if alias_relative is not None:
                            discovery = receipt['files'].get('providers/__init__.py', {})
                            need(discovery.get('sha256') == _NATIVE_PROVIDER_DISCOVERY_SHA256
                                and discovery.get('mode') == '100644',
                                'owned_native_provider_discovery_not_pinned')
                            discovery_path = native/'providers/__init__.py'
                            discovery_mode = stat.S_IMODE(discovery_path.lstat().st_mode)
                            need(discovery_mode in (0o644,0o664),
                                'owned_native_provider_discovery_mode_refused')
                            discovery_raw = owned_read({'path':str(discovery_path),
                                'sha256':_NATIVE_PROVIDER_DISCOVERY_SHA256}, mode=discovery_mode)
                            need(len(discovery_raw) == discovery.get('bytes'),
                                'owned_native_provider_discovery_size_changed')
                            need(getattr(module,'__name__',None) == name
                                and getattr(getattr(module,'__spec__',None),'name',None) == name,
                                'owned_native_provider_alias_spec_name_changed')
                            relative = alias_relative
                            # Pin all alias and discovery parent directory identities
                            # alongside the existing held-FD source custody.
                            for parent in ((native/relative).parent, discovery_path.parent):
                                while parent != native:
                                    info = parent.lstat()
                                    need(parent.resolve() == parent and stat.S_ISDIR(info.st_mode)
                                        and info.st_uid == os.getuid() and not info.st_mode & 0o022,
                                        'owned_native_provider_parent_identity_refused')
                                    before = identity(info)
                                    need(parent not in directories or directories[parent] == before,
                                        'owned_native_provider_parent_changed')
                                    directories.setdefault(parent, before)
                                    parent = parent.parent
                        need(relative in receipt['files'],'owned_native_module_not_pinned')
                        row=receipt['files'][relative]
                        need(row.get('mode')=='100644','owned_native_module_mode_required')
                        expected=native/relative
                        physical_mode=stat.S_IMODE(expected.lstat().st_mode)
                        need(physical_mode in (0o644,0o664),'owned_native_physical_mode_refused')
                        raw=owned_read({'path':str(expected),'sha256':row['sha256']},mode=physical_mode)
                        need(len(raw)==row['bytes'],'owned_native_module_size_changed')
            need(filename==str(expected) and origin==str(expected),
                'owned_module_origin_changed')
            if hasattr(module,'__path__'):
                allowed=[str(expected.parent)]
                if name in ('tools','plugins'):allowed.append(str(canonical/name))
                need(list(module.__path__) in (allowed,allowed[:1]),'owned_package_path_changed')
