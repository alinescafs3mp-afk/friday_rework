};
struct frwk_task { struct task_struct *ref; u64 id, table, parent, exec;
 bool live, born_complete, executing, adopted; };

struct frwk_retained { struct file *held; u64 id, task, table, slot, ofd; u32 fd;
 bool cleanup_only, live; };
struct frwk_ring { struct file *ref; /* separate owned memory pin, NOT file.f_ref */ u64 id, task, table, ofd; bool live; };
struct frwk_delivery { u64 id, hi, lo, first, last; u32 count; };
/* Original native custody only. OPEN is zero for the still-missing bootstrap.
 * Closing never restores loss, useful time, adoption, producer or ACK authority.
 * Every slot below represents one independent acquired hold, even for equal
 * pointer values. owner_ref is an EXTRA task hold; scope_ref is a cgroup hold.
 * Bootstrap acquisition, charge and genuine shutdown/lifetime qualification
 * are missing. No caller boolean, user handle or transport can supply them. */
enum frwk_close_state {
 FRWK_EPOCH_OPEN=0, FRWK_EPOCH_CLOSING, FRWK_EPOCH_TRANSFERRED,
 FRWK_EPOCH_PUTS_ISSUED, FRWK_EPOCH_PARTIAL
};
enum frwk_batch_state {
 FRWK_BATCH_EMPTY=0, FRWK_BATCH_READY, FRWK_BATCH_DRAINING,
 FRWK_BATCH_PUTS_ISSUED, FRWK_BATCH_PARTIAL
};
enum frwk_batch_kind { FRWK_BATCH_RETIRED=1, FRWK_BATCH_FINAL=2 };
enum frwk_put_kind {
 FRWK_PUT_TASK=1, FRWK_PUT_TABLE, FRWK_PUT_FILE,
 FRWK_PUT_OWNER, FRWK_PUT_SCOPE
};
struct frwk_epoch;
struct frwk_reference_batch {
 /* Storage must be precharged and zeroed before the original bootstrap.
  * This binding is created only by the native internal transfer bodies.
  * owner_epoch is borrowed: the missing native issuer must pin its lifetime. */
 struct frwk_epoch *owner_epoch;
 u64 hi, lo, final_ns;
 enum frwk_batch_state state;
 enum frwk_batch_kind kind;
 struct task_struct *owner; struct cgroup *scope;
 struct task_struct *tasks[FRWK_TASKS]; struct files_struct *tables[FRWK_TABLES];
 struct file *files[FRWK_OFDS+FRWK_RINGS+FRWK_RETAINS];
 u8 file_kinds[FRWK_OFDS+FRWK_RINGS+FRWK_RETAINS];
 u32 native_files_n; /* independently held retained native refs precede all memory pins */
 u32 tasks_n, tables_n, files_n, tasks_i, tables_i, files_i;
 bool owner_issued, scope_issued, put_inflight;
 enum frwk_put_kind put_kind; u32 put_index;
 /* put_inflight means one detached hold was handed to a void kernel put.
  * An interrupted/nonreturning put is uncertain, never replayable. Remaining
  * non-NULL slots retain ownership; this batch is never reset/rearmed. */
};
struct frwk_epoch {
 raw_spinlock_t lock;
 enum frwk_close_state close_state;
 u64 close_hi, close_lo, close_final_ns;
 struct frwk_reference_batch final_refs;
 struct frwk_reference_batch *retired_batch; /* one borrowed, precharged in-flight batch */
 struct task_struct *owner_ref; struct cgroup *scope_ref;
 u64 hi, lo, seq, generation, loss, final_ns, useful_ns, coverage;
 u32 tasks_n, tables_n, ofds_n, slots_n, retained_n, records_n, rings_n;
 u64 fence_seq;
 bool copy_active, delivery_pending;
 struct frwk_delivery delivery;
 u32 producer, consumer, events_n;
 bool prearmed, armed, final_only;
 struct frwk_task tasks[FRWK_TASKS];
 struct frwk_table tables[FRWK_TABLES];
 struct frwk_ofd ofds[FRWK_OFDS];
 struct frwk_slot slots[FRWK_SLOTS];
 struct frwk_retained retained[FRWK_RETAINS];
 struct frwk_ring rings[FRWK_RINGS];
 struct frwk_event events[FRWK_EVENTS];
};

 * a proven native serialization mechanism. Genuine exclusion stays MISSING. Private copies
