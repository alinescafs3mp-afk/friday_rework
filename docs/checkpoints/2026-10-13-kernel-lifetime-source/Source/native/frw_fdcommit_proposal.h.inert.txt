/* NEW PROPOSED ABI v1. NOT an installed Linux UAPI, module, BPF program,
 * hook map, syscall or ioctl assignment. No change to friday_cli_birth.c.
 * Frames use explicit little endian integers and no kernel/user pointers.
 * Capture is allowed only through complete, independently qualified commit
 * sites. BTF declarations alone cannot set FRW_REQUIRED_COVERAGE. */
#ifndef FRW_FDCOMMIT_PROPOSAL_H
#define FRW_FDCOMMIT_PROPOSAL_H
#include <stdint.h>
#define FRW_MAGIC UINT32_C(0x46525731)
#define FRW_VERSION 1
#define FRW_FRAME_BYTES 176
#define FRW_TASKS 64
#define FRW_TABLES 64
#define FRW_OFDS 1024
#define FRW_SLOTS 4096
#define FRW_RINGS 64
#define FRW_EVENTS 1024
#define FRW_RECORDS 8192
#define FRW_RETAINS 32 /* lifetime, not simultaneous live entries */
#define FRW_MAX_FD 1048575
#define FRW_STATUS_MASK (3ULL|0x400|0x800|0x1000|0x2000|0x4000|0x8000|0x10000|0x20000|0x40000|0x100000|0x200000|0x400000)
#define FRW_REQUIRED_COVERAGE UINT64_C(0x0000ffff)
/* Native/direct operations, transfers, clone/CLONE_FILES/copy/unshare,
 * exec/CLOEXEC, exit, flags, io_uring fixed files, owner/cgroup/delegation,
 * loss detection and atomic birth/arm are ALL mandatory; not alternatives. */
enum frw_kind { FRW_ARM=1, FRW_OFD_BIRTH, FRW_TABLE_BIRTH, FRW_TASK_BIRTH,
 FRW_INHERIT, FRW_INSTALL, FRW_REPLACE, FRW_CLOSE, FRW_FLAGS, FRW_SHARE,
 FRW_COPY_BEGIN, FRW_EXEC_BEGIN, FRW_EXEC_END, FRW_TASK_EXIT, FRW_OFD_END,
 FRW_TABLE_END, FRW_FENCE, FRW_LOSS, FRW_ADOPT, FRW_STATUS_FLAGS,
 FRW_RING_BIRTH, FRW_RING_INSTALL, FRW_RING_CLOSE, FRW_RETAIN, FRW_SCOPE };
/* SOURCE revision: native memory-pin custody, ctx-bound teardown, exact copy extent.
 * COPY_BEGIN.fd=actual native copied prefix extent (0 denotes legacy complete copy).
 * NATIVE_TEARDOWN is emitted only from genuine native final-owner/ctx boundaries;
 * no caller or data flag creates that authority. Wire176/version1 and caps stay.
 * A new qualified producer/decoder binding is required; no provider is supplied. */
/* TABLE_BIRTH flags=immutable copy purpose, aux=one reserved child generation
 * or original actor for switch. Wire176 and resource count caps are unchanged. */
enum frw_copy_operation { FRW_COPY_PRIVATE_BIRTH=1, FRW_COPY_UNSHARE=2, FRW_COPY_EXEC=3 };
enum frw_loss { FRW_GAP=1, FRW_OVERFLOW=2, FRW_UNSUPPORTED=4,
 FRW_LATE_ARM=8, FRW_SATURATED=16, FRW_SCOPE_DRIFT=32,
 FRW_MALFORMED=64, FRW_HISTORY_CHANGED=128, FRW_TRANSPORT_LOSS=256 };
enum frw_path { FRW_NATIVE=1, FRW_SCM_RIGHTS=2, FRW_PIDFD_GETFD=4,
 FRW_IO_URING=8, FRW_INHERITED=16, FRW_DUP=32, FRW_CLOEXEC=64,
 FRW_DIRECT=128, FRW_ROOT_SNAPSHOT=256, FRW_COPY_DONE=512,
 FRW_BIRTH_DONE=1024, FRW_ARM_DONE=2048, FRW_EXEC_DONE=4096,
 FRW_RETAIN_RELEASE=8192, FRW_RING_RETIRE=16384, FRW_NATIVE_TEARDOWN=32768 };
/* Packing is a NEW proposal; encoder/decoder must not depend on C padding.
 * Wire offsets: magic0/version4/kind6; fifteen identity/state Q fields after
 * four epoch/ordering Q fields at8; fd160/src_fd164/reason168/reserved172. */
struct frw_frame {
 uint32_t magic; uint16_t version, kind;
 uint64_t epoch_hi, epoch_lo, seq, mono_ns;
 uint64_t task, table, ofd, slot, exec, parent_task, src_table, src_ofd,
          src_slot, before_ofd, before_slot, aux, flags, coverage, loss;
 uint32_t fd, src_fd, reason, reserved;
};
/* Internal native authority API: never caller-supplied vtables/callbacks.
 * These opaque types can only originate in the separately qualified native
 * bridge. They are not serialized tokens and expose no public constructors.
 * The actual OS transport allocation/installation is intentionally UNPROVEN.
 */
struct frw_native_session;
struct frw_native_epoch;
struct frw_native_qualification;
struct frw_native_retained;
#endif
