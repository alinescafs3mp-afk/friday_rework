def cheap_reference(ref):
    """No observer or memo proof: authenticate actual inputs before history.
    Historical 0664 is admitted only under exact owned 0700 H, matching the
    existing guarded history leaf contract; never chmod or rewrite it.
    """
    assert type(ref) is dict and set(ref)=={'path','sha256'}
    p=Path(ref['path']);assert p.is_absolute() and '..' not in p.parts and p.resolve()==p
    assert type(ref['sha256']) is str and len(ref['sha256'])==64 and all(c in '0123456789abcdef' for c in ref['sha256'])
    historical=p.is_relative_to(H)
    parents=[]
    for parent in p.parents:
        st=parent.lstat();assert stat.S_ISDIR(st.st_mode)
        private_parent=historical and (parent==H or parent.is_relative_to(H))
        if private_parent:
            assert st.st_uid==os.getuid() and st.st_gid==os.getgid() and not st.st_mode&0o7002
        else:
            assert st.st_uid in (0,os.getuid()) and not st.st_mode&0o022
        parents.append((parent,tuple(getattr(st,k) for k in META[:5])))
    if historical:
        anchor=H.lstat();assert anchor.st_uid==os.getuid() and anchor.st_gid==os.getgid() and stat.S_IMODE(anchor.st_mode)==0o700
    fd=os.open(p,os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK)
    try:
        before=os.fstat(fd);mode=stat.S_IMODE(before.st_mode)
        assert stat.S_ISREG(before.st_mode) and before.st_uid==os.getuid() and before.st_gid==os.getgid() and before.st_nlink==1 and not before.st_mode&0o7002
        assert not before.st_mode&0o020 or (historical and mode==0o664)
        assert before.st_size<=128*1024**2
        digest=hashlib.sha256()
        while data:=os.read(fd,1024**2):digest.update(data)
        assert digest.hexdigest()==ref['sha256'] and ident(before)==ident(os.fstat(fd))==ident(p.lstat())
        assert all(tuple(getattr(parent.lstat(),k) for k in META[:5])==saved for parent,saved in parents)
    finally:os.close(fd)
