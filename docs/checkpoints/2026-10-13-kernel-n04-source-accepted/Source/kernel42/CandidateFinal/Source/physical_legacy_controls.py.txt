def _read_resource(resources,key,process):
    value=resources[key]
    physical.need(type(value)is dict and set(value)>={'fd','identity'},'ACTUAL_CAUSAL_RESOURCE_SHAPE')
    actual=physical.fd_read(process,value['fd'])
    declared=tuple(value['identity'])
    if len(declared)==2:
        physical.need(set(value)>={'path','policy'} and actual.identity[2]==0o040000 and
            actual.identity[:2]==declared and actual.link==value['path'] and actual.policy()==value['policy'],
            'ACTUAL_CAUSAL_DIRECTORY_RESOURCE_IDENTITY_POLICY')
    else:physical.need(len(declared)==3 and actual.identity==declared,'ACTUAL_CAUSAL_RESOURCE_IDENTITY')
    return value

def define_legacy_case(control_id,actual_caps,actual_context,resources):
    physical.hard_caps()
    physical.need(control_id in CONTROLS,'EXACT_CURRENT_LEGACY_CONTROL')
    caller=physical.process_read(actual_context['actual_original_caller_pid'])
    root=physical.process_read(actual_context['actual_Root_pid'])
    physical.need(len(actual_caps)==23,'ALL23_CURRENT_CAPS')
    kind=CONTROLS[control_id]['kind'];suffix=control_id[4:]
    last=next(reversed(actual_caps));caps=_ordered_last(actual_caps,last)
    before=copy.deepcopy(caps);delta={'requested_mutation':suffix}
    target=caps[last];actual=target.get('actual_control_binding',{})
    # The outside fixture freezes real fd/path/inode capabilities before tracing.
    # Changed declaration rows are retained alongside all actual kernel reads.
    if kind=='zero':
        if suffix=='CONTROL_AS_TRACE_REFUSED':
            target['trace_dir_fd']=target['control_dir_fd']
        elif suffix in ('CROSS_PHASE_SAME_FD','CROSS_PHASE_TRACE_INODE'):
            first=next(iter(caps.values()))
            cap=(first['private_directory_capabilities']['trace'] if suffix=='CROSS_PHASE_SAME_FD'
                 else _read_resource(resources,'real_duplicate_trace_capability',root))
            target['trace_dir_fd']=cap['fd'];target['private_directory_capabilities']['trace']=copy.deepcopy(cap)
        elif suffix in ('GROUP_ROLE_LAST_ZERO','GROUP_OWNER_IDENTITY_LAST_ZERO','GROUP_POLICY_LAST_ZERO'):
            row=actual['auxiliary_groups'][-1];delta['before']=copy.deepcopy(row)
            if suffix=='GROUP_ROLE_LAST_ZERO':row['ownership']['role']='auxiliary_borrowed'
            elif suffix=='GROUP_OWNER_IDENTITY_LAST_ZERO':row['ownership']['identity'][1]+=1
            else:row['directory_policy']['mode']^=0o020
            delta['after']=copy.deepcopy(row)
        elif suffix in ('OUTPUT_PARENT_ROLE_LAST_ZERO','OUTPUT_PARENT_IDENTITY_LAST_ZERO','OUTPUT_PARENT_POLICY_LAST_ZERO'):
            row=actual['auxiliary_outputs'][-1];delta['before']=copy.deepcopy(row)
            if suffix=='OUTPUT_PARENT_ROLE_LAST_ZERO':row['ownership']['role']='auxiliary_group'
            elif suffix=='OUTPUT_PARENT_IDENTITY_LAST_ZERO':row['ownership']['identity'][1]+=1
            else:row['parent_capability']['policy']['mode']=0o755
            delta['after']=copy.deepcopy(row)
        elif suffix in ('BORROWED_ROLE_LAST_ZERO','BORROWED_POLICY_LAST_ZERO'):
            row=actual['auxiliary_fds'][-1];delta['before']=copy.deepcopy(row)
            if suffix=='BORROWED_ROLE_LAST_ZERO':row['ownership']['role']='auxiliary_group'
            else:row['owner_policy']['mode']^=0o020

        elif suffix in ('REMOTE_CALLER_INODE_LAST_ZERO','REMOTE_CALLER_ROLE_LAST_ZERO'):
            row=actual['auxiliary_groups'][-1];delta['before']=copy.deepcopy(row)
            inventory=target['retained_existing_caller']['auxiliary_capabilities']
            declared=next(r for r in inventory if r['fd']==row['ownership']['caller_fd'])
            if suffix=='REMOTE_CALLER_INODE_LAST_ZERO':
                other=_read_resource(resources,'real_different_caller_capability',caller)
                row['ownership']['caller_fd']=other['fd']
                declared['fd']=other['fd'] # declared old inode intentionally contradicts real remote fd.
            else:declared['role']='auxiliary_borrowed'
            delta['after']=copy.deepcopy(row);delta['holder_row']=copy.deepcopy(declared)
        elif suffix in ('ALL_SAFE_AUXILIARY_ZERO','FOUR_CUT_AUXILIARY_ZERO'):
            physical.need(resources.get('member_key') is not None,'EMPTY_CONTROL_REQUIRES_COMPLETE_ATTEMPT_GROUP')
            phase=resources['actual_empty_phase_id']
            physical.need(phase in ('ALL_SAFE','K120_PIDFD_FSTAT','K120_MAP_GO','K120_INIT_WAIT','K120_PIPE_CLOSE'),
                          'EXACT_ORIGINAL_EMPTY_PHASE')
            caps=_ordered_last(caps,phase);last=phase;target=caps[last]
            real=resources['actual_nonempty_empty_phase_inventory']
            for key in ('auxiliary_groups','auxiliary_outputs','auxiliary_fds'):
                for row in real.get(key,()):
                    fd=row['output_fd'] if key=='auxiliary_outputs' else row['fd']
                    physical.fd_read(root,fd)
            target['actual_control_binding']=copy.deepcopy(real)
            expected_kind=resources['actual_empty_phase_auxiliary_kind']
            key={'auxiliary_group':'auxiliary_groups','auxiliary_output':'auxiliary_outputs','auxiliary_borrowed':'auxiliary_fds'}[expected_kind]

            local=physical.fd_read(root,fd);remote=physical.fd_read(caller,row['ownership']['caller_fd'])

        else:raise physical.ObservationMissing('LEGACY_ZERO_CAUSAL_TRANSFORM_MISSING:'+suffix)
    elif kind=='stop':
        # This executable fault is installed in actual consumers, never run here.
        delta['fault']=suffix
    expected=ZERO_CODES.get(suffix) if kind=='zero' else STOP_CODES.get(suffix) if kind=='stop' else None
    return CaseDefinition(control_id,kind,expected,suffix,last,tuple(caps),caps,copy.deepcopy(caps),
        before,delta,caller.pid,root.pid,copy.deepcopy(actual_context['original_outer_holder_binding']),resources)

class CurrentFault:
    def __init__(self,case):
        physical.need(case.control_id in CONTROLS and case.kind=='stop','EXACT_FINITE_STOP_CONTROL')
        self.case=case;self.fired=False;self.pending_close=None;self.duplicate_operation=None
    def before(self,point,custody,**actual):
        was_fired=self.fired
        try:return self._before(point,custody,**actual)
        finally:
            if not was_fired and self.fired and self.pending_close is None:
                markers.observe_state('fault_after',{'mutation':self.case.mutation,'point':point,
                    'actual':actual,'custody':custody,
                    'created_unknown_fd':self.case.resources.get('actual_created_unknown_fd'),
                    'duplicate_fd_operation':self.duplicate_operation})
                markers.mark('fault_end',list(CONTROLS).index(self.case.control_id))
    def after_close(self,custody,fd):
        pending=self.pending_close
        if pending is None:return
        physical.need(pending['actual']['fd']==fd and os.getpid()==self.case.root_pid,
            'EXACT_AMBIGUOUS_OUTER_CLOSE_COMPLETION')

        markers.observe_state('fault_after',{'mutation':self.case.mutation,'point':'close',
            'actual':pending['actual'],'custody':custody,'created_unknown_fd':None})
        markers.mark('fault_end',list(CONTROLS).index(self.case.control_id))
    def _capture_fault_before(self,point,custody,actual):

    def _before(self,point,custody,**actual):
        suffix=self.case.mutation
        child=point=='child_hygiene';root=os.getpid()==self.case.root_pid
        if self.fired or not (child or root):return
        eligible={
          'FUTURE_FD_LOST':'retained_plan','FUTURE_INODE_DRIFT':'retained_plan',
          'FUTURE_CLOSED_PREMATURELY':'retained_plan','GUARDIAN_DRIFT':'retained_plan',
          'COMPLETED_PLAN_ROW_LOST':'retained_plan','UNKNOWN_LIVE_DYNAMIC_FD':'begin_phase',
          'CURRENT_UNCLOSED_EFFECT':'close','EXTERNAL_CALLER_HOLD_ON_STOP':'close',
          'CURRENT_AMBIGUOUS_CLOSE':'close','BATCH_SLOT_PREMATURE_RELEASE':'close',
          'PHASE_REPLAY':'begin_phase','INCOMPLETE_BATCH_TERMINAL':'complete_phase',
          'UNKNOWN_BATCH_TERMINAL_FD':'batch_finalize',
          'RECORDED_KEEP_MISSING_STOP':'child_hygiene','MISSING_FD_REUSED_BY_SCAN_STOP':'child_hygiene',
          'UNKNOWN_INHERITED_FD_STOP':'child_hygiene','INHERITED_AMBIGUOUS_CLOSE_STOP':'child_hygiene',
          'STOCK_AS_ACTOR_REFUSED':'bind','HELPER_AS_STOCK_REFUSED':'bind','READY_AS_GO_REFUSED':'bind'}
        if eligible.get(suffix)!=point:return
        if suffix in ('PHASE_REPLAY','COMPLETED_PLAN_ROW_LOST','INCOMPLETE_BATCH_TERMINAL') and not custody.get('completed_phases'):return
        if point=='close':
            wanted=self.case.resources.get('actual_unclosed_owned_fd') if suffix in ('CURRENT_UNCLOSED_EFFECT','EXTERNAL_CALLER_HOLD_ON_STOP') else self.case.resources.get('actual_fault_owned_fd')
            if suffix=='BATCH_SLOT_PREMATURE_RELEASE':wanted=self.case.resources['actual_batch_slot_duplicate']
            if actual['fd']!=wanted:return
        rows=custody['fds'];resources=self.case.resources
        if suffix in ('STOCK_AS_ACTOR_REFUSED','HELPER_AS_STOCK_REFUSED','READY_AS_GO_REFUSED'):
            roles={'STOCK_AS_ACTOR_REFUSED':('stock_pidfd','actor_pidfd'),
                   'HELPER_AS_STOCK_REFUSED':('post_terminal_helper_pidfd','stock_pidfd'),
                   'READY_AS_GO_REFUSED':('actor_ready:read','actor_GO:write')}
            original,requested=roles[suffix]
            from root_fd_custody import canonical_role
            if canonical_role(actual['label'])!=original:return
            self.fired=True
            self._capture_fault_before(point,custody,actual)
            from root_fd_custody import preflight_existing_role
            preflight_existing_role(custody,actual['fd'],requested)
            raise physical.ObservationMissing('EXPECTED_REAL_ROLE_REFUSAL_DID_NOT_OCCUR')
        target_fd=resources.get('actual_fault_owned_fd')
        if target_fd is not None:
            matches=[r for r in rows if r['fd']==target_fd and not r['closed']]
            physical.need(len(matches)==1,'CAUSAL_FAULT_EXACT_OWNED_ROW')
            observed=physical.fd_read(physical.process_read(os.getpid()),target_fd)
            physical.need(tuple(matches[0]['identity'])==observed.identity,'CAUSAL_FAULT_ACTUAL_IDENTITY')
        if suffix=='UNKNOWN_BATCH_TERMINAL_FD':self._require_original_batch_fault_call(custody,actual)
        self.fired=True
        self._capture_fault_before(point,custody,actual)
        if suffix in ('FUTURE_FD_LOST','FUTURE_CLOSED_PREMATURELY','RECORDED_KEEP_MISSING_STOP','MISSING_FD_REUSED_BY_SCAN_STOP'):
            os.close(target_fd) # genuine subject duplicate; original caller retains originals.
        elif suffix in ('FUTURE_INODE_DRIFT','GUARDIAN_DRIFT'):
            donor=_read_resource(resources,'real_distinct_fault_donor',physical.process_read(os.getpid()))
            physical.need(donor['fd']!=target_fd,'CAUSAL_DUP2_DISTINCT_DONOR')
            os.dup2(donor['fd'],target_fd) # one genuine changed live fd; no retry.
        elif suffix=='COMPLETED_PLAN_ROW_LOST':
            completed=set(custody['completed_phases'])
            row=next(r for r in rows if r.get('planned') and r.get('phase') in completed)
            rows.remove(row) # genuine consumer plan defect; physical fd facts stay raw.
        elif suffix=='PHASE_REPLAY':
            from root_fd_custody import begin_phase
            begin_phase(custody,custody['completed_phases'][0])
        elif suffix=='INCOMPLETE_BATCH_TERMINAL':
            from root_fd_custody import finalize_batch_roles
            finalize_batch_roles(custody,resources['actual_batch_slot_duplicate'])
        elif suffix in ('UNKNOWN_LIVE_DYNAMIC_FD','UNKNOWN_BATCH_TERMINAL_FD','UNKNOWN_INHERITED_FD_STOP'):
            donor=_read_resource(resources,'real_distinct_fault_donor',physical.process_read(os.getpid()))
            # Only N03's useful child/dynamic paths use operation_begin4.
            # The original fully qualified late batch adverse closure already
            # has fault17/18 raw custody and syscall capture. It gains no useful
            # operation, late-marker exception or observer authority.
            useful_bracket=suffix!='UNKNOWN_BATCH_TERMINAL_FD'
            if useful_bracket:markers.mark('operation_begin',4,donor['fd'])
            fd=os.dup(donor['fd']);resources['actual_created_unknown_fd']=fd
            self.duplicate_operation={'function':'os.dup','operation':4,'donor_fd':donor['fd'],'result_fd':fd}
            if useful_bracket:markers.mark('operation_end',4,fd,donor['fd'])
            if suffix in ('UNKNOWN_LIVE_DYNAMIC_FD','UNKNOWN_BATCH_TERMINAL_FD'):
                from root_fd_custody import own
                own(custody,fd,'unknown_lifetime_control_owned')
            # Child unknown fd is deliberately absent from inherited registry.
        elif suffix in ('CURRENT_AMBIGUOUS_CLOSE','INHERITED_AMBIGUOUS_CLOSE_STOP'):
            os.close(target_fd)
            matches[0]['identity']=None # the next real close obtains kernel EBADF.
            if child:
                from root_fd_custody import closed
                closed(custody,target_fd)
            else:
                # The outer original closed() must perform the real EBADF
                # before this control interval is sealed.
                self.pending_close={'actual':dict(actual)}
        elif suffix in ('CURRENT_UNCLOSED_EFFECT','EXTERNAL_CALLER_HOLD_ON_STOP'):
            row=next(r for r in rows if r['fd']==resources['actual_unclosed_owned_fd'])
            physical.need(not row['closed'] and physical.fd_read(physical.process_read(os.getpid()),row['fd']).identity==tuple(row['identity']),
                          'REAL_OMITTED_CLOSE_CONTROL_REQUIRES_LIVE_OWNED_FD')
            return True # deliberately omit this ONE actual close; row stays genuinely live.
        elif suffix=='BATCH_SLOT_PREMATURE_RELEASE':
            physical.need(actual['fd']==resources['actual_batch_slot_duplicate'],'EXACT_BATCH_SLOT_CONTROL')
            raise ValueError('ROOT_PHASE_PREMATURE_SLOT_RELEASE_FORBIDDEN')
        else:raise physical.ObservationMissing('CURRENT_FAULT_IMPLEMENTATION_MISSING:'+suffix)

def define_legacy_group(control_id,actual_caps,actual_context,member_resources):
    physical.need(control_id in ('C75_FOUR_CUT_AUXILIARY_ZERO','C75_ALL_SAFE_AUXILIARY_ZERO'),
                  'EXACT_FINITE_EMPTY_CONTROL_GROUP')
    phases=FOUR_CUT_PHASES if control_id=='C75_FOUR_CUT_AUXILIARY_ZERO' else ('ALL_SAFE',)
    expected=tuple((phase,kind) for phase in phases for kind in EMPTY_KINDS)
    physical.need(set(member_resources)==set(expected),'ALL_ORIGINAL_PHASE_KIND_MEMBERS_NO_CUTS')
    cases=[]
    for phase,kind in expected:
        resources=dict(member_resources[(phase,kind)],member_key=(phase,kind),
            actual_empty_phase_id=phase,actual_empty_phase_auxiliary_kind=kind)
        # Resources belong to twelve independent already-owned attempts; the
        # group never creates launches, shares a run or fabricates raw events.
        cases.append(define_legacy_case(control_id,actual_caps[(phase,kind)],
                         actual_context[(phase,kind)],resources))
    return tuple(cases)

def qualify_alias_resource(suffix,caps,row,local,remote,root,caller,resources):
    """Classify the exact real causal relationship before malformed declaration.
    Independent capture rechecks these physical rows; the declaration is no proof.
    """
    group_rows=[(p,r) for p,c in caps.items() for r in c.get('actual_control_binding',{}).get('auxiliary_groups',())]
    output_rows=[(p,r) for p,c in caps.items() for r in c.get('actual_control_binding',{}).get('auxiliary_outputs',())]
    borrowed_rows=[(p,r) for p,c in caps.items() for r in c.get('actual_control_binding',{}).get('auxiliary_fds',())]
    if suffix=='SAME_AUXILIARY_OUTPUT_MOUNT':
        candidates=[(p,r,physical.fd_read(root,r['output_fd'])) for p,r in output_rows]
        matches=[(p,r,a) for p,r,a in candidates if a.fd!=local.fd and a.mount_id==local.mount_id]
        physical.need(matches,'ACTUAL_DISTINCT_FDS_SAME_OWNED_OUTPUT_MOUNT')
    elif suffix in ('BORROWED_OWNED_INODE_ALIAS','AUXILIARY_CROSS_ROLE_FD','CALLER_CLEAN_GROUP_INODE_DUP'):
        candidates=[(p,r,physical.fd_read(root,r['fd'])) for p,r in group_rows]
        matches=[(p,r,a) for p,r,a in candidates if a.identity[:2]==local.identity[:2] and
            (a.fd==local.fd if suffix=='AUXILIARY_CROSS_ROLE_FD' else a.fd!=local.fd)]
        if suffix=='CALLER_CLEAN_GROUP_INODE_DUP':
            for p,c in caps.items():
                a=physical.fd_read(root,c['group_fd'])
                if a.identity[:2]==local.identity[:2] and a.fd!=local.fd:matches.append((p,{'fd':a.fd},a))
        physical.need(matches,'ACTUAL_EXACT_OWNED_BORROWED_OR_CLEAN_GROUP_RELATION')
    elif suffix=='BORROWED_OWNED_OUTPUT_FILE_ALIAS':
        expected=resources['actual_owned_output_file_fd']
        matches=[]
        for p,c in caps.items():
            for fd,identity in c['output_owned_fds']:
                if fd==expected:
                    a=physical.fd_read(root,fd)
                    physical.need(a.identity==tuple(identity),'REAL_ORIGINAL_OWNED_OUTPUT_FILE')
                    if a.fd!=local.fd and a.identity==local.identity:matches.append((p,{'fd':fd},a))
        physical.need(matches,'ACTUAL_BORROWED_DUPLICATE_OWNED_OUTPUT_FILE')
    elif suffix=='BORROWED_SHARE_UNDECLARED':
        matches=[(p,r,physical.fd_read(root,r['fd'])) for p,r in borrowed_rows
            if r['fd']!=local.fd and physical.fd_read(root,r['fd']).identity==local.identity]
        contract=('read_only_original_caller_capability','distinct_phase_local_duplicates')
        physical.need(matches and all((r.get('shared_purpose'),r.get('shared_lifetime'))==contract
            for p,r,a in matches) and (row.get('shared_purpose'),row.get('shared_lifetime'))==contract,
                      'REAL_EXPLICIT_BORROWED_SHARE_BEFORE_DECLARATION_REMOVAL')
    elif suffix=='RESERVED_INODE_DUP':
        matches=[]
        for p,c in caps.items():
            for key in ('group_fd','slot_fd','control_dir_fd','trace_dir_fd'):
                a=physical.fd_read(root,c[key])
                if a.fd!=local.fd and a.identity==local.identity:matches.append((p,{'fd':a.fd,'reserved_role':key},a))
        physical.need(matches,'ACTUAL_DISTINCT_FD_SAME_CURRENT_RESERVED_INODE')
    else:raise physical.ObservationMissing('EXACT_ALIAS_CLASSIFICATION_MISSING:'+suffix)
    phase,original,actual=matches[0]
    return {'phase':phase,'reference_fd':actual.fd,'alias_fd':local.fd,
            'reference_identity':list(actual.identity),'alias_identity':list(local.identity),
            'reference_mount_id':actual.mount_id,'alias_mount_id':local.mount_id,
            'caller_fd':remote.fd,'caller_identity':list(remote.identity),'semantic_relation':suffix}
