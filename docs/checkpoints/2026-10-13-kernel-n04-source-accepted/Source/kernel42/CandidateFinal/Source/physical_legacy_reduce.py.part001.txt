"""Intrinsic real-observation reducers. No external reducer body or t fixture.

Consumer state read through ptrace is semantic declaration, never kernel truth.
Every physical conclusion uses actual syscall results or stopped /proc reads.
Unsupported/missing facts raise ObservationMissing; no unknown-to-zero credit.
"""
import os
import stat
import struct
from collections.abc import Mapping
import physical_observation_primitives as physical
import physical_observation_capture as capture
import physical_observation_reduce as base
import physical_observation_markers as markers
import root_fd_custody as custody_source
from qualification_batch import CASES

def _missing_physical(code):
    raise physical.ObservationMissing(code)

class LiveFacts:
    def __init__(self,run):
        self.run=capture.validate_raw_run(run)
        self.root=run.case.root_pid;self.caller=run.case.caller_pid
    def events(self,nr=None,pid=None,lo=0,hi=None,stage='exit'):
        return tuple(e for e in self.run.kernel if (nr is None or e.syscall==nr)
            and (pid is None or e.pid==pid) and e.stage==stage and
            lo<e.sequence<(hi if hi is not None else 1<<63))
    def states(self,kind=None,pid=None):
        return tuple(b for b in self.run.boundaries if b.semantic_read is not None
            and (kind is None or b.semantic_read['kind']==kind)
            and (pid is None or b.pid==pid))
    def state(self,kind,pid=None,last=True):
        rows=self.states(kind,pid)
        physical.need(rows,'ACTUAL_CONSUMER_STATE_MISSING:'+str(kind))
        return rows[-1 if last else 0]
    def registry(self,b):
        state=b.semantic_read['state']
        return state.get('custody',state)
    def closes(self,fd,pid=None,identity=None,lo=0,hi=None):
        rows=self.events(3,pid,lo,hi)
        return tuple(e for e in rows if e.arguments[0]==fd and e.result==0
            and (identity is None or type(e.fd_before)is physical.FDRead
                 and e.fd_before.identity==tuple(identity)))
    def closed_once(self,row,pid,lo=0,hi=None):
        return len(self.closes(row['fd'],pid,row.get('identity'),lo,hi))==1
    def row_physically_live(self,b,row):
        return any(r.fd==row['fd'] and r.identity==tuple(row['identity']) for r in b.all_fd_reads)
    def verified_registry(self,b,allow_stopped=False):
        state=self.registry(b);rows=state['fds'];live=[r for r in rows if not r['closed']]
        physical.need(len({r['fd'] for r in live})==len(live),'ACTUAL_REGISTRY_DUPLICATE')
        for row in rows:
            if row.get('close_ambiguous'):
                physical.need(allow_stopped,'AMBIGUOUS_ROW_CANNOT_BE_NORMAL')
            elif row['closed']:
                physical.need(self.closed_once(row,b.pid,hi=self._seq(b)),
                              'DECLARED_CLOSED_WITHOUT_EXACT_RAW_CLOSE')
            else:physical.need(self.row_physically_live(b,row),'DECLARED_LIVE_WITHOUT_ACTUAL_FD')
        return state
    def _seq(self,b):
        return next(m.kernel_sequence for m in self.run.markers if m.sequence==b.marker_sequence)
    def mark(self,name,pid=None):return base._marks(self.run,name,pid)
    def proof(self,name,predicate,selectors=()):return base._proof(self.run,name,predicate,selectors)
    def scope(self,case_id):
        phase='ALL_SAFE' if case_id in {r[0] for r in CASES if r[1]=='positive'} else case_id
        order=self.run.case.phase_order
        physical.need(phase in order,'CURRENT_ORIGINAL_PHASE_MISSING')
        index=order.index(phase)
        begins=[m for m in self.mark('phase_begin',self.root) if m.values[0]==index]
        ends=[m for m in self.mark('phase_closed',self.root) if m.values[0]==index]
        physical.need(len(begins)==len(ends)==1 and begins[0].kernel_sequence<ends[0].kernel_sequence,
                      'CURRENT_EXACT_PHASE_INTERVAL_MISSING')
        return begins[0].kernel_sequence,ends[0].kernel_sequence
    def operation(self,number,phase_id='ALL_SAFE'):
        lo,hi=self.scope(phase_id)
        rows=[r for r in self.run.operations if r['operation']==number and
              lo<r['entry_sequence']<r['return_sequence']<hi]
        physical.need(len(rows)==1,'ONE_EXACT_INDEPENDENT_OPERATION_INTERVAL:'+str(number))
        row=rows[0]
        physical.need(row['pid'] in self.run.known_subjects and
            self.run.known_subjects[row['pid']][0]==row['birth'],'EXACT_OPERATION_SUBJECT_BIRTH')
        return row
    def allocator(self):
        lo,hi=self.scope('ALL_SAFE')
        intervals=[r for r in self.run.operations if r['operation']==3 and lo<r['entry_sequence']<r['return_sequence']<hi]
        physical.need(intervals,'ACTUAL_ALLOCATOR_INTERVAL_MISSING')
        interval=min(intervals,key=lambda r:r['entry_sequence'])
        rows=[r for r in self.run.function_reads if r['pid']==interval['pid'] and
            r['birth']==interval['birth'] and r['operation']==interval['entry_sequence']]
        physical.need(len(rows)==1,'ONE_ACTUAL_LIBC_CALL_ENTRY_RETURN')
        row=rows[0]
        physical.need(row['entry'] is not None and row['return'] is not None and
            interval['entry_sequence']<=row['entry']['kernel_sequence']<=row['return']['kernel_sequence']<interval['return_sequence']
            and row['return']['result']==1,'ACTUAL_MALLOPT_MINUS8_ONE_RETURN_ONE')
        first=[e for e in self.events(stage='actual_birth') if e.pid==interval['pid']]
        physical.need(first and row['return']['kernel_sequence']<min(e.sequence for e in first),
                      'REAL_ALLOCATOR_RETURN_BEFORE_FIRST_TASK_BIRTH')
        return row
    def fault_span(self):
        starts=self.mark('fault_begin');ends=self.mark('fault_end')
        physical.need(len(starts)==len(ends)==1 and starts[0].pid==ends[0].pid and
            starts[0].kernel_sequence<ends[0].kernel_sequence,'EXACT_CURRENT_CONTROL_EFFECT_INTERVAL')
        lo,hi=starts[0].kernel_sequence,ends[0].kernel_sequence
        before=[b for b in self.states(17,starts[0].pid) if lo<self._seq(b)<hi]
        after=[b for b in self.states(18,starts[0].pid) if lo<self._seq(b)<hi]
        physical.need(len(before)==len(after)==1 and
            before[0].semantic_read['state']['mutation']==self.run.case.mutation and
            after[0].semantic_read['state']['mutation']==self.run.case.mutation,'ACTUAL_CAUSAL_CONTROL_NOT_ALIAS')
        return lo,hi,before[0],after[0]
    def stop_span(self):
        lo,hi,before,after=self.fault_span()
        stopped=[b for b in self.states(16,self.root) if self._seq(b)>hi]
        physical.need(stopped,'REAL_STOP_AFTER_EXACT_CURRENT_FAULT')
        terminal=[m.kernel_sequence for m in self.mark('attempt_end') if m.kernel_sequence>hi]
        return lo,(max(terminal) if terminal else max(e.sequence for e in self.run.kernel)+1),before,after,stopped[0]
    def role_wait(self,role):
        phase=self.run.case.resources.get('observed_role_phase_id','ALL_SAFE')
        lo,hi=self.scope(phase)
        rows=[(b,r) for b,r in self.role_rows() if custody_source.canonical_role(r['label'])==role and
            r.get('identity') is not None and lo<self._seq(b)<hi]
        physical.need(rows,'EXACT_OWNED_ROLE_WAITER_MISSING:'+role)
        candidates=[]
        for settlement in self.run.target_settlement:
            for wait in settlement['raw_owned_waits']:
                if wait['owner_pid']!=self.root or not lo<wait['sequence']<hi:continue
                for b,row in rows:
                    handle=next((r for r in b.all_fd_reads if r.fd==row['fd'] and r.identity==tuple(row['identity'])),None)
                    if handle is None:continue
                    pids=[s.split(':',1)[1].strip() for s in handle.info_raw.splitlines() if s.startswith('Pid:')]
                    if pids==[str(wait['target_pid'])] and self._seq(b)<wait['sequence']:
                        candidates.append(wait)
        unique={w['sequence']:w for w in candidates}
        physical.need(len(unique)==1,'ONE_EXACT_ROLE_RAW_WAIT:'+role)
        return next(iter(unique.values()))
    def raw_waits(self,pid=None,lo=0,hi=None):
        rows=self.events(61,pid,lo,hi)
        return tuple(e for e in rows if e.result>0 and e.raw_output is not None and len(e.raw_output)==4)
    def status(self,event):return int.from_bytes(event.raw_output,'little')
    def physical_group_reads(self,name,lo=0,hi=None):
        rows=self.events(0,lo=lo,hi=hi)
        return tuple(e for e in rows if type(e.fd_before)is physical.FDRead and
            e.fd_before.link.startswith('/sys/fs/cgroup/') and
            os.path.basename(e.fd_before.link)==name and e.result>0 and e.raw_output is not None)
    def group_values(self,name,lo=0,hi=None):
        rows=self.physical_group_reads(name,lo,hi)
        physical.need(rows,'REAL_GROUP_READ_MISSING:'+name)
        return tuple(e.raw_output.decode('ascii').strip() for e in rows)
    def protected(self):
        first=base._bounds(self.run,'attempt_begin');last=base._bounds(self.run,'attempt_end')
        physical.need(first and last,'ACTUAL_PROTECTED_BOUNDARIES_MISSING')
        return first[0].protected_reads==last[-1].protected_reads
    def final_root_state(self):return self.verified_registry(self.state(5,self.root))
    def completed(self):
        return self.final_root_state()['completed_phases']
    def aux_inventory(self):
        bounds=base._bounds(self.run,'attempt_begin',self.root)
        physical.need(len(bounds)==1 and bounds[0].phase_inventory is not None,'REAL23_INVENTORY_MISSING')
        return bounds[0]
    def aux_rows(self):
        b=self.aux_inventory();result=[]
        byfd={r.fd:r for r in b.all_fd_reads};remote={r.fd:r for r in b.caller_fd_reads}
        for p in b.phase_inventory['phases']:
            cap=p['declared'];a=cap.get('actual_control_binding',{})
            for key,role in [('auxiliary_groups','auxiliary_group'),('auxiliary_outputs','original_caller_output_parent'),('auxiliary_fds','auxiliary_borrowed')]:
                for row in a.get(key,()):
                    fd=row['output_fd'] if key=='auxiliary_outputs' else row['fd']
                    own=row['ownership'];caller_fd=own['caller_fd']
                    physical.need(fd in byfd and caller_fd in remote,'REAL_LOCAL_REMOTE_AUX_FD_MISSING')
                    result.append((p['phase'],key,row,byfd[fd],remote[caller_fd]))
        return tuple(result)
    def trace_rows(self):
        boundary=self.aux_inventory();byfd={row.fd:row for row in boundary.all_fd_reads}
        rows=[]
        for phase in boundary.phase_inventory['phases']:
            cap=phase['declared'];fd=cap['trace_dir_fd']
            physical.need(type(fd)is int and fd in byfd,'EXACT_PHASE_TRACE_FD_READ_MISSING')
            rows.append((phase['phase'],cap,byfd[fd]))
        physical.need(len(rows)==23,'COMPLETE_CURRENT_TRACE_ROLE_DOMAIN')
        return tuple(rows)
    def created_unknown_batch_fd(self):
        """Original late adverse closure, ONLY exact fault17/18 raw evidence.
        No operation_begin exemption and no normal/terminal/capacity credit.
        """
        lo,hi,before,after=self.fault_span();case=self.run.case
        physical.need(case.control_id=='C55_UNKNOWN_BATCH_TERMINAL_FD' and case.kind=='stop' and
            case.mutation=='UNKNOWN_BATCH_TERMINAL_FD' and
            case.expected_refusal=='ROOT_BATCH_UNKNOWN_OR_PHASE_HANDLE' and
            before.pid==after.pid==self.root and before.birth==after.birth and
            self.run.known_subjects.get(self.root)==(before.birth,'current_Root'),
            'EXACT_RETAINED_BATCH_CONTROL_SUBJECT_BIRTH')
        state=before.semantic_read['state'];final=after.semantic_read['state']
        slot=self.run.registration['root_slot_duplicate_fd']
        physical.need(state.get('point')==final.get('point')=='batch_finalize' and
            state.get('actual')==final.get('actual')=={'slot_fd':slot} and
            slot==case.resources['actual_batch_slot_duplicate'],
            'EXACT_RETAINED_BATCH_FINALIZE_CALL_ARGUMENT')
        temporal={'K120_USEFUL_RESERVE_SHORT','K120_FINAL_RESERVE_SHORT'}
        order=('ALL_SAFE',)+tuple(r[0] for r in CASES if r[1]=='negative' and r[0] not in temporal)+tuple(
            r[0] for r in CASES if r[1]=='source_uncertainty_after_actual_operation')+(
            'K120_USEFUL_RESERVE_SHORT','K120_FINAL_RESERVE_SHORT')
        registry=self.verified_registry(before)
        physical.need(case.phase_order==tuple(registry['phase_order'])==tuple(registry['completed_phases'])==order and
            registry['active_phase'] is None and registry['STOP_UNCONFIRMED'] is False and len(order)==23,
            'RETAINED_BATCH_FAULT_AFTER_FULL_ORIGINAL32_ONLY')
        closed=self.states(4,self.root)
        physical.need(len(closed)==len(order),'EVERY_REAL_ORIGINAL_PHASE_CLOSURE_REQUIRED')
        for index,phase in enumerate(order):
            begin,end=self.scope(phase);boundary=closed[index]
            observed=self.verified_registry(boundary)
            physical.need(end<self._seq(boundary)<lo and before.birth==boundary.birth and
                tuple(observed['completed_phases'])==order[:index+1] and observed['active_phase'] is None,
                'REAL_FULL_SERIAL_PHASE_AND_RAW_FD_CLOSES_BEFORE_BATCH_FAULT')
        final_lo,final_hi=self.scope('K120_FINAL_RESERVE_SHORT')
        clock_rows=[b for b in self.states(13,self.root) if
            b.semantic_read['state'].get('case_id')=='K120_FINAL_RESERVE_SHORT']
        physical.need(len(clock_rows)==1,'EXACT_ORIGINAL_FINAL_SHORT_RAW_CLOCK_ENTRY')
        original=clock_rows[0].semantic_read['state'];raw=next(e for e in self.run.kernel if
            e.sequence==self._seq(clock_rows[0]))
        deadline_m=self.run.registration['original_final_monotonic']
        deadline_u=self.run.registration['original_final_unix']
        physical.need(final_lo<raw.sequence<final_hi<lo and raw.birth==before.birth and
            original['original_seconds']==120 and original['cleanup_seconds']==30 and
            original['original_final_m']==deadline_m and original['original_final_u']==deadline_u and
            original['clock_reset'] is False and
            original['entry_m']-original['clock_start_m']>=90.001 and
            original['entry_u']-original['clock_start_u']>=90.001 and
            raw.monotonic>=deadline_m-120+90.001 and raw.unix>=deadline_u-120+90.001,
            'SAME_LATE_RAW_FINAL_SHORT_WITHOUT_EARLY_CONTROL_OR_RENEWAL')
        physical.need(not [m for m in self.run.markers if m.pid==self.root and
            m.event in ('operation_begin','operation_end') and m.values[0]==4 and lo<m.kernel_sequence<hi],
            'NO_LATE_USEFUL_OPERATION_MARKER_AUTHORITY')
        batches=self.mark('batch_finalize',self.root)
        physical.need(len(batches)==1 and batches[0].birth==before.birth and
            hi<batches[0].kernel_sequence and batches[0].values[0]==slot,
            'REAL_ORIGINAL_FINALIZER_CONTINUES_AFTER_REAL_FAULT')
        declared=final.get('created_unknown_fd');operation=final.get('duplicate_fd_operation')
        physical.need(type(declared)is int and declared>2 and type(operation)is dict and
            operation.get('function')=='os.dup' and operation.get('operation')==4 and
            operation.get('result_fd')==declared and
            operation.get('donor_fd')==case.resources['real_distinct_fault_donor']['fd'],
            'EXACT_ORIGINAL_BATCH_DUPLICATE_DECLARATION_NOT_AUTHORITY')
        dup_lo=self._seq(before);dup_hi=self._seq(after)
        physical.need(lo<dup_lo<dup_hi<hi,'EXACT_ORIGINAL_FAULT17_18_RAW_DUP_INTERVAL')
        abi=self.run.registration.get('duplicate_fd_native_abi')
        physical.need(type(abi)is dict and abi.get('machine')=='x86_64' and
            abi.get('function')=='os.dup' and abi.get('dup_syscall')==32 and
            abi.get('fcntl_syscall')==72 and abi.get('F_DUPFD_CLOEXEC')==1030 and
            abi.get('minimum_fd')==0,'PINNED_NATIVE_OS_DUP_ABI_UNBOUND')
        pins=[abi.get(k) for k in ('python_pin','implementation_pin','fcntl_header_pin')]
        physical.need(all(type(p)is dict and p in self.run.registration['exact_abi_pins'] for p in pins),
            'EXACT_DUPLICATE_ABI_SOURCE_PINS_REQUIRED')
        for pin in pins:physical.same_read_source(pin)
        selected=self.events(pid=before.pid,lo=dup_lo,hi=dup_hi)
        duplicates=[e for e in selected if e.birth==before.birth and
            (e.syscall in (32,33,292) or e.syscall==72 and e.arguments[1] in (0,1030))]
        physical.need(len(duplicates)==1,'ONE_EXACT_DUPLICATE_FD_SYSCALL_REQUIRED')
        event=duplicates[0]
        physical.need((event.syscall==32 or event.syscall==72 and
            event.arguments[1:3]==(abi['F_DUPFD_CLOEXEC'],abi['minimum_fd'])) and
            event.arguments[0]==operation['donor_fd'] and event.result==declared and
            type(event.fd_before)is physical.FDRead and type(event.fd_after)is physical.FDRead,
            'EXACT_SUBJECT_LOCAL_UNKNOWN_DUP_RESULT_MISSING')
        entry=[e for e in self.events(pid=before.pid,lo=dup_lo,
            hi=event.sequence,stage='entry') if e.birth==before.birth and
            e.syscall==event.syscall and e.arguments==event.arguments]
        physical.need(len(entry)==1 and event.actual_process_sample is not None,
            'EXACT_RAW_DUP_ENTRY_EXIT_AND_SUBJECT_REQUIRED')
        process=event.actual_process_sample['process'];actual=event.fd_after
        from pathlib import Path
        executable=Path(abi['python_pin']['path']).stat()
        physical.need(event.actual_process_sample.get('native_executable_identity')==
            (executable.st_dev,executable.st_ino,executable.st_size,executable.st_mtime_ns),
            'ACTUAL_DUP_NATIVE_EXECUTABLE_INODE_PIN')
        namespace=lambda p:tuple(tuple(line.split()[1:]) for line in p.status_raw.splitlines() if line.startswith('NSpid:'))
        physical.need(process.pid==before.pid and process.birth==before.birth and
            process.exe==abi['python_pin']['path'] and namespace(process)==namespace(before.process)==namespace(after.process) and
            len(namespace(process))==1 and namespace(process)[0] and
            actual.pid==before.pid and actual.birth==before.birth and actual.fd==declared and
            event.fd_before.pid==before.pid and event.fd_before.birth==before.birth and
            event.fd_before.fd==operation['donor_fd'] and event.fd_before.identity==actual.identity and
            any(read.fd==operation['donor_fd'] and read.identity==actual.identity for read in before.all_fd_reads) and
            all(read.fd!=declared for read in before.all_fd_reads) and
            any(read.fd==declared and read.identity==actual.identity and read.flags&os.O_CLOEXEC for read in after.all_fd_reads) and
            all(row['fd']!=declared for row in self.registry(before)['fds']),
            'INDEPENDENT_EXACT_UNKNOWN_FD_BIRTH_IDENTITY_NAMESPACE')
        # The one actual duplicate is still unplanned and physically live at
        # the real refusal/STOP. A declared outcome, alias or leak earns no credit.
        changed=self.registry(after)
        physical.need(tuple(changed['completed_phases'])==order and changed['active_phase'] is None and
            {(r.fd,r.identity,r.flags,r.mount_id) for r in after.all_fd_reads if r.fd!=declared}==
                {(r.fd,r.identity,r.flags,r.mount_id) for r in before.all_fd_reads},
            'ONLY_EXACT_ORIGINAL_BATCH_DUPLICATE_FD_DIFFERENCE')
        rows=[r for r in changed['fds'] if r['fd']==declared]
        physical.need(len(rows)==1 and not rows[0].get('planned') and not rows[0]['closed'] and
            tuple(rows[0]['identity'])==actual.identity,
            'REAL_ORIGINAL_BATCH_UNPLANNED_LIVE_OWNED_ROW')
        stopped=self.states(16,self.root)
        physical.need(stopped,'REAL_BATCH_REFUSAL_STOP_CUSTODY_REQUIRED')
        stopped=next(b for b in stopped if self._seq(b)>hi)
        physical.need(self.row_physically_live(stopped,rows[0]) and
            not self.closes(declared,self.root,actual.identity,event.sequence,self._seq(stopped)) and
            any(m.birth==before.birth and m.values[0]==markers.REFUSALS.index(
                'ROOT_BATCH_UNKNOWN_OR_PHASE_HANDLE')+1 and batches[0].kernel_sequence<m.kernel_sequence<self._seq(stopped)
                for m in self.mark('refusal',self.root)) and
            event.monotonic<deadline_m-2 and event.unix<deadline_u-2,
            'ACTUAL_BATCH_DUPLICATE_REFUSAL_STOP_BEFORE_OWN_SETTLEMENT_TAIL')
        return event,actual

    def created_unknown_fd(self):
        lo,hi,before,after=self.fault_span()
        physical.need(before.pid==after.pid and before.birth==after.birth and
            self.run.case.mutation=='UNKNOWN_INHERITED_FD_STOP','EXACT_UNKNOWN_CHILD_CONTROL')
        declared=after.semantic_read['state'].get('created_unknown_fd')
        operation=after.semantic_read['state'].get('duplicate_fd_operation')
        physical.need(type(declared)is int and declared>2 and type(operation)is dict and
            operation.get('function')=='os.dup' and operation.get('operation')==4 and
            operation.get('result_fd')==declared,'UNKNOWN_CREATED_FD_DECLARATION_MISSING')
        starts=[m for m in self.mark('operation_begin',before.pid) if m.birth==before.birth and
            m.values[0]==4 and lo<m.kernel_sequence<hi]
        ends=[m for m in self.mark('operation_end',before.pid) if m.birth==before.birth and
            m.values[0]==4 and lo<m.kernel_sequence<hi]
        physical.need(len(starts)==len(ends)==1 and
            lo<self._seq(before)<starts[0].kernel_sequence<ends[0].kernel_sequence<self._seq(after)<hi and
            starts[0].values[1]==operation.get('donor_fd') and
            ends[0].values[1:]==(declared,operation['donor_fd']),
            'EXACT_OS_DUP_OPERATION_ARGUMENT_INTERVAL')
        intervals=[r for r in self.run.operations if r['pid']==before.pid and r['birth']==before.birth and
            r['operation']==4 and r['entry_sequence']==starts[0].kernel_sequence and
            r['return_sequence']==ends[0].kernel_sequence]
        physical.need(len(intervals)==1,'EXACT_INDEPENDENT_DUPLICATE_FD_INTERVAL_MISSING')
        abi=self.run.registration.get('duplicate_fd_native_abi')
        physical.need(type(abi)is dict and abi.get('machine')=='x86_64' and
            abi.get('function')=='os.dup' and abi.get('dup_syscall')==32 and
            abi.get('fcntl_syscall')==72 and abi.get('F_DUPFD_CLOEXEC')==1030 and
            abi.get('minimum_fd')==0,'PINNED_NATIVE_OS_DUP_ABI_UNBOUND')
        pins=[abi.get(k) for k in ('python_pin','implementation_pin','fcntl_header_pin')]
        physical.need(all(type(p)is dict and p in self.run.registration['exact_abi_pins'] for p in pins),
            'EXACT_DUPLICATE_ABI_SOURCE_PINS_REQUIRED')
        for pin in pins:physical.same_read_source(pin)
        selected=self.events(pid=before.pid,lo=starts[0].kernel_sequence,hi=ends[0].kernel_sequence)
        duplicates=[e for e in selected if e.birth==before.birth and
            (e.syscall in (32,33,292) or e.syscall==72 and e.arguments[1] in (0,1030))]
        physical.need(len(duplicates)==1,'ONE_EXACT_DUPLICATE_FD_SYSCALL_REQUIRED')
        event=duplicates[0]
        physical.need((event.syscall==32 or event.syscall==72 and
            event.arguments[1:3]==(abi['F_DUPFD_CLOEXEC'],abi['minimum_fd'])) and
            event.arguments[0]==operation['donor_fd'] and event.result==declared and
            type(event.fd_before)is physical.FDRead and type(event.fd_after)is physical.FDRead,
            'EXACT_SUBJECT_LOCAL_UNKNOWN_DUP_RESULT_MISSING')
        entry=[e for e in self.events(pid=before.pid,lo=starts[0].kernel_sequence,
            hi=event.sequence,stage='entry') if e.birth==before.birth and
