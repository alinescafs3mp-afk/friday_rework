ORIGINAL150 = frozenset(["L"+str(i).zfill(3) for i in range(1,107)]+["P"+str(i).zfill(2) for i in range(1,9)]+["N"+str(i).zfill(2) for i in range(1,37)])

@dataclass(frozen=True)
class NativePlan:
    case_id: str
    profile: str
    fixture: FixtureSpec
    actions: tuple
    direct_text: str | None = None

    def validate_shape(self):
        if type(self) is not NativePlan or type(self.fixture) is not FixtureSpec:
            raise Unbound("plan_or_fixture_not_exact_Source_type")
        if type(self.case_id) is not str or self.case_id not in ORIGINAL150:
            raise Unbound("case_not_original106_plus44_identity")
        if type(self.profile) is not str or self.profile not in {"prompt","direct"}:
            raise Unbound("plan_profile_type")
        if type(self.actions) not in (tuple,list) or len(self.actions) > 64:
            raise Unbound("plan_action_sequence_type_or_bound")
        # Reject ALL nested custom records BEFORE any supplied method/hash.
        if any(type(x) is not Action or type(x.trigger) is not Trigger for x in self.actions):
            raise Unbound("nested_action_trigger_not_exact_Source")
        FixtureSpec.validate(self.fixture)
        for action in self.actions:
            Action.validate(action)
        if self.profile == "prompt":
            if self.direct_text is not None or self.fixture.kind != "pty" or not self.actions:
                raise Unbound("classic_prompt_requires_real_PTY_and_finite_actions")
            last = self.actions[-1]
            if last.kind not in {"request_exit","native_app_exit","submit"}:
                raise Unbound("prompt_requires_finite_original_exit_path")
            if last.kind == "submit":
                if (len(self.actions) < 2 or self.actions[-2].kind != "type_text" or self.actions[-2].data != b"/exit"
                        or last.trigger.text_sha256 != hashlib.sha256(b"/exit").hexdigest()):
                    raise Unbound("final_submit_not_separate_observed_original_exit")
        else:
            text(self.direct_text,"direct_original_query",65536)
            if not self.direct_text.strip() or any(x.kind in {"type_text","submit","request_exit"} for x in self.actions):
                raise Unbound("direct_query_not_plain_original_text")
        return self

    def canonical(self):
        NativePlan.validate_shape(self)
        fixture = {k:getattr(self.fixture,k) for k in ("kind","reader_mode","prefix_limit","capture_limit","contract","pipe_capacity","rows","columns")}
        actions = []
        for action in self.actions:
            trigger = {k:getattr(action.trigger,k) for k in ("kind","occurrence","caller","minimum_bytes","text_sha256")}
            actions.append({"trigger":trigger,"kind":action.kind,"actual_input_bytes":len(action.data),"actual_input_sha256":hashlib.sha256(action.data).hexdigest(),"delay_seconds":action.delay_seconds})
        return canonical({"case_id":self.case_id,"profile":self.profile,"fixture":fixture,"actions":actions,
            "direct_text_bytes":None if self.direct_text is None else len(self.direct_text.encode()),
            "direct_text_sha256":None if self.direct_text is None else hashlib.sha256(self.direct_text.encode()).hexdigest()})

    def validate(self, runtime):
        if type(runtime) is not NativeRuntime:
            raise Unbound("plan_runtime_not_exact_Source")
        NativePlan.validate_shape(self)
        if self.case_id not in runtime.grant["allowed_cases"]:
            raise Unbound("plan_case_not_explicitly_allowed")
        body = NativePlan.canonical(self)
        if hashlib.sha256(body).hexdigest() != runtime.protocol["case_plan_sha256"][self.case_id]:
            raise Unbound("whole_exact_Root_accepted_fixture_plan_UNBOUND")
        return self

class NativeProducer:
    def __init__(self, runtime):
        if type(runtime) is not NativeRuntime:
            raise Unbound("producer_runtime_not_exact_Source_type")
        self.runtime,self.result,self.primary_exception,self.used = runtime,None,None,False
        self.cleanup_errors,self.cleanup_stages = [],[]

    def _stage_error(self, stage, exc):
        row = {"stage":stage,"type":type(exc).__name__,"at":time.monotonic()}
        self.cleanup_errors.append(row)
        return {"unknown":"cleanup_stage_failed","error":row}

    def run(self, plan, directory):
        if type(self) is not NativeProducer or type(self.runtime) is not NativeRuntime:
            raise Unbound("producer_not_exact_Source_type")
        if self.used:
            raise Unbound("native_producer_not_replayable")
        # Consume this Source instance once, including malformed/refused attempts.
        self.used = True
        runtime,entered = self.runtime,time.monotonic()
        fixture = observer = bus = controller = reaper = custody = None
        native_input = native_output = session_cm = stdio_cm = None
        session_entered = stdio_entered = native_call_entered = native_call_returned = False
        exit_code = original_return = plan_body = None
        stage = "strict_preflight"
        try:
            if type(plan) is not NativePlan or type(plan.fixture) is not FixtureSpec or type(plan.actions) not in (tuple,list):
                raise Unbound("public_plan_boundary_not_exact_Source_data")
            if any(type(x) is not Action or type(x.trigger) is not Trigger for x in plan.actions):
                raise Unbound("public_nested_action_trigger_boundary")
            NativePlan.validate(plan,runtime)
            target = canonical_path(directory)
            if target != runtime.protocol["case_output_directories"][plan.case_id]:
                raise Unbound("output_directory_not_exact_Root_protocol_pin")
            if threading.current_thread() is not threading.main_thread():
                raise Unbound("intact_frontend_requires_original_main_thread")
            plan_body = NativePlan.canonical(plan)
            runtime.begin(plan.case_id)
            runtime.require_effects(CONSTRUCTOR_EFFECTS)
            stage = "private_case_custody"
            custody = runtime.home_custody.create_child(target.rsplit("/",1)[1])
            fixture = NativeFixture(runtime,plan.fixture,custody)
            bus,observer = EventBus(),None
            observer = OriginalObserver(runtime,bus)
            fixture.observer = observer
            controller = NativeController(runtime,fixture,bus,plan.actions)
            reaper = OriginalClosureAdapter(runtime,observer,bus)
            stage = "fixture_open"
            fixture.open()
            stage = "observer_install"
            observer.install()
            stage = "fixture_readers"
            fixture.start_readers(bus)
            stage = "stdio_install"
            stdio_cm = fixture.stdio()
            stdio_cm.__enter__()
            stdio_entered = True
            # Mandatory DB/plugin/provider/discovery/network/model/worker/
            # stdio/pipe scope is checked AGAIN before EACH native constructor.
            stage = "native_input_constructor"
            runtime.require_effects(CONSTRUCTOR_EFFECTS)
            fixture.native_borrow_started = True
            native_input = runtime.qualified_function("create_input")(stdin=fixture.stdin)
            stage = "native_output_constructor"
            runtime.require_effects(CONSTRUCTOR_EFFECTS)
            native_output = runtime.qualified_function("create_output")(stdout=fixture.stdout)
            stage = "native_session_constructor"
            runtime.require_effects(CONSTRUCTOR_EFFECTS)
            session_cm = runtime.qualified_function("create_app_session")(input=native_input,output=native_output)
            if type(session_cm) is not _GeneratorContextManager:
                raise Unbound("native_AppSession_context_manager_type_not_exact")
            session_cm.__enter__()
            session_entered = True
            stage = "native_CLI_constructor"
            runtime.require_effects(CONSTRUCTOR_EFFECTS)
            cli = runtime.qualified_class("cli")(**runtime.constructor_parameters())
            observer.require_admission_open()
            if observer.registry.original_cli is not cli:
                raise Unbound("real_frontend_constructor_not_observed")
            reaper.start()
            controller.start()
            stage = "intact_original_frontend_call"
            frontend_function = observer.prepare_frontend_dispatch(self,cli,plan.profile)
            native_call_entered = True
            try:
                if plan.profile == "prompt":
                    original_return = frontend_function(cli)
                else:
                    runtime.require_effects(("model_inference","network","owned_worker"))
                    original_return = frontend_function(cli,plan.direct_text,None,False,True,stream_json=False)
            finally:
                # Indicates the ORIGINAL call unwound; never implies success.
                native_call_returned = True
                fixture.native_returned = True
        except SystemExit as exc:
            exit_code = exc.code
            if type(exit_code) is not int or exit_code != 0:
                self.primary_exception = exc
        except BaseException as exc:
            self.primary_exception = exc
        finally:
            state,final = {},None
            try:
                final = runtime.original_deadline(final=True)
                if observer is not None and observer.case_deadline is not None:
                    final = min(final,number(observer.case_deadline,"retained_case_cleanup_clock",0,1e12))
                state["cleanup_clock"] = {"status":"RETAINED_VALIDATED_ORIGINAL","final":final,"new_deadline_created":False}
            except BaseException as exc:
                state["cleanup_clock"] = self._stage_error("cleanup_setup_original_clock",exc)
                state["cleanup_clock"]["new_deadline_created"] = False
            # Each stage has its OWN try/failure record. No early failure skips
            # later stages; no cleanup error replaces the original exception.
            for name in ("controller","reaper","session_context","native_input","stdio","native_closure","fixture","observer_detach","history_custody","case_custody"):
                try:
                    if final is None and name in {"controller","reaper","session_context","native_input","native_closure","fixture"}:
                        raise Unbound("cleanup_original_clock_unknown_no_bounded_effect_no_new_deadline")
                    if name == "controller":
                        state[name] = None if controller is None else controller.close(final)
                    elif name == "reaper":
                        state[name] = None if reaper is None else reaper.close(final)
                    elif name == "session_context":
                        if session_entered:
                            exc = self.primary_exception
                            returned = session_cm.__exit__(None if exc is None else type(exc),exc,None if exc is None else exc.__traceback__)
                            state[name] = {"actual_exit_return_type":type(returned).__name__,"suppression_does_not_erase_primary":returned is True}
                        else:
                            state[name] = {"entered":False,"partial_enter_or_constructor":"NOT_CREDITED_AS_CLOSED"}
                    elif name == "native_input":
                        if native_input is None:
                            state[name] = {"constructed":False}
                        else:
                            close_input = runtime.original_function("native_input_close")
                            returned = close_input(native_input)
                            state[name] = {"actual_return_type":type(returned).__name__}
                    elif name == "stdio":
                        if stdio_entered:
                            exc = self.primary_exception
                            stdio_cm.__exit__(None if exc is None else type(exc),exc,None if exc is None else exc.__traceback__)
                        elif fixture is not None:
                            fixture.restore_stdio()
                        state[name] = {"restoration":"PER_BINDING_OWNERSHIP_ONLY"}
                    elif name == "native_closure":
                        own = tuple(x for x in ((None if controller is None else controller.thread),(None if reaper is None else reaper.thread),*([] if fixture is None else fixture.readers)) if x is not None)
                        state[name] = None if observer is None else final_native_closure(runtime,observer,own_threads=own)
                    elif name == "fixture":
                        actor_closed = all(x is None or x.thread is None or not x.thread.is_alive() for x in (controller,reaper))
                        qualified = actor_closed and not self.cleanup_errors and type(state.get("native_closure")) is dict and state["native_closure"].get("writers_qualified_quiescent") is True
                        state[name] = None if fixture is None else fixture.close(final,writers_quiescent=qualified)
                    elif name == "observer_detach":
                        if observer is not None:
                            observer.detach()
                        state[name] = {"attempted":observer is not None}
                    elif name == "history_custody":
                        rows = []
                        if observer is not None and (reaper is None or reaper.thread is None or not reaper.thread.is_alive()):
                            for caller in observer.callers:
                                binding = caller.binding
                                if binding is not None and hasattr(binding,"original_state_directory_custody"):
                                    try:
                                        rows.append(binding.original_state_directory_custody.close())
                                    except BaseException as exc:
                                        rows.append(self._stage_error("history_custody_one",exc))
                        elif observer is not None:
                            rows.append({"unknown":"reaper_unjoined_history_custody_retained"})
                        state[name] = rows
                    elif name == "case_custody":
                        state[name] = None if custody is None else custody.close()
                    self.cleanup_stages.append({"stage":name,"returned":time.monotonic()})
                except BaseException as exc:
                    state[name] = self._stage_error(name,exc)
            try:
                observed = None if observer is None else observer.snapshot()
            except BaseException as exc:
                observed = self._stage_error("final_observer_snapshot",exc)
            try:
                Source_fds = runtime.source_FD_state()
            except BaseException as exc:
                Source_fds = self._stage_error("final_Source_FD_snapshot",exc)
            # Product catches cannot erase the fatal Source before-effect guard.
            if observer is not None and observer.admission_failure is not None and self.primary_exception is None:
                self.primary_exception = observer.admission_failure
            self.result = {"case_id":None if type(plan) is not NativePlan else plan.case_id,"profile":None if type(plan) is not NativePlan else plan.profile,
                "plan_sha256":None if plan_body is None else hashlib.sha256(plan_body).hexdigest(),"factory_entered":entered,"factory_returned_after_closure":time.monotonic(),
                "primary_stage":stage,"primary_exception_type":None if self.primary_exception is None else type(self.primary_exception).__name__,
                "native_exit_code":exit_code,"native_return_type":type(original_return).__name__,"native_call_entered":native_call_entered,"native_call_unwound":native_call_returned,
                "whole_native_case_entered":None if observer is None else observer.case_entered,"whole_native_case_original_deadline":None if observer is None else observer.case_deadline,
                "whole_native_case_closed":time.monotonic(),"observer":observed,"controller":state.get("controller"),"reaper":state.get("reaper"),
                "native_closure":state.get("native_closure"),"fixture_closure":state.get("fixture"),"cleanup_stages":list(self.cleanup_stages),"cleanup_errors":list(self.cleanup_errors),
                "Source_FD_account":Source_fds,"cleanup_state":state,"captured_original_bytes":None if fixture is None or not fixture.capture_observed else bytes(fixture.capture),
                "captured_foreign_bytes":None if fixture is None or not fixture.foreign_capture_observed else bytes(fixture.foreign_capture),"captured_actual_memory":None if fixture is None else fixture.memory_value,
                "all150":"NOT_RUN_BY_SINGLE_GENERAL_PRODUCER","full150_complete_paths":0,"runtime_acceptance":"INDEPENDENT_ROOT_REQUIRED",
                "guardian":"UNBOUND_MANDATORY_NO_NATIVE_RETURN_SUBSTITUTION"}
        if self.primary_exception is not None:
            raise self.primary_exception
        if self.cleanup_errors:
            # Source failure is explicit after evidence; does NOT overwrite a
            # native primary (already raised above) or claim safe closure.
            raise Unbound("producer_cleanup_unconfirmed_result_retained")
        return self.result

def ordinary_prompt_plan(case_id, text_value, *, reader_mode="drain", prefix_limit=0):
    text(text_value,"ordinary_original_text",65536)
    if not text_value.strip():
        raise Unbound("empty_original_text")
    actions = (Action(Trigger("frontend_ready"),"type_text",text_value.encode()),
        Action(Trigger("prompt_text_changed",text_sha256=hashlib.sha256(text_value.encode()).hexdigest()),"submit",delay_seconds=.06),
        Action(Trigger("caller_closed",caller=1),"type_text",b"/exit"),
        Action(Trigger("prompt_text_changed",text_sha256=hashlib.sha256(b"/exit").hexdigest()),"submit",delay_seconds=.06))
    plan = NativePlan(case_id,"prompt",FixtureSpec("pty",reader_mode,prefix_limit),actions)
    NativePlan.validate_shape(plan)
    return plan

def direct_plan(case_id, text_value, *, kind="pipe", contract="exact", actions=()):
    plan = NativePlan(case_id,"direct",FixtureSpec(kind,contract=contract),actions,text_value)
    NativePlan.validate_shape(plan)
    return plan
