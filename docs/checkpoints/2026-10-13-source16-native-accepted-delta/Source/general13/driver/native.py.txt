"""Genuine native observations; this Source has not been imported or executed.

Instrumentation observes original frames and real descriptors. It never
replaces os.write/flush, a native function, a return count, a thread or a scope.
All object references remain live opaque references; JSON cannot remint them.
"""
from __future__ import annotations
import asyncio
import hashlib
import io
import os
from pathlib import Path
import sys
import threading
import time
import types
from .qualification import Unbound, code_digest, exact_file, proc_birth

EVENT_LIMIT = 2048
BODY_LIMIT = 65536


def object_key(value):
    return None if value is None else {
        "id": id(value), "type": type(value).__module__ + "." + type(value).__qualname__
    }


def descriptor(fd):
    if type(fd) is not int or fd < 0:
        raise Unbound("descriptor_not_integer")
    st = os.fstat(fd)
    import fcntl
    return {"fd": fd, "identity": (st.st_dev, st.st_ino, st.st_mode, st.st_rdev),
            "flags": fcntl.fcntl(fd, fcntl.F_GETFL),
            "fdflags": fcntl.fcntl(fd, fcntl.F_GETFD),
            "target": os.readlink(f"/proc/self/fd/{fd}")}


def native_thread(thread):
    # Sample only an actually started original thread, never its pre-start object.
    if type(thread) is not threading.Thread or thread.native_id is None:
        raise Unbound("thread_not_created")
    tid = thread.native_id
    raw = Path(f"/proc/self/task/{tid}/stat").read_text()
    closing = raw.rfind(")")
    return {"object": object_key(thread), "native_id": tid,
            "starttime": int(raw[closing + 2:].split()[19]), "raw_stat": raw,
            "alive": thread.is_alive()}


class LiveBinding:
    def __init__(self, *, cli, app_session, context, host, whole_layout,
                 classes, function_qualifications, source_qualification=None):
        """Accept actual constructor-owned objects from the qualified host.

        The caller may not create SimpleNamespace/adapters or restore a lease
        from a durable row. Whole interpreter/import qualification is mandatory.
        Root's current composition must be bound separately before construction.
        """
        from .qualification import verify_loaded, real_function
        # The genuine Source producer passes its same original FD account.
        # Existing driver consumers retain their original unqualified boundary.
        if source_qualification is not None:
            from native_producer.qualification_io import MeteredQualification
            if type(source_qualification) is not MeteredQualification:
                raise Unbound("native_binding_Source_qualification_not_exact")
            verify_loaded = source_qualification.verify_loaded
            real_function = source_qualification.real_function
        verify_loaded(whole_layout)
        for name, value in (("cli", cli), ("session", app_session),
                            ("context", context), ("host", host)):
            if type(value) is not classes[name]:
                raise Unbound("not_original_native_" + name)
        self.cli, self.session, self.ctx, self.host = cli, app_session, context, host
        self.agent = cli.agent
        self.cap = cli._friday_cli_scope
        self.owner = cli._friday_cli_output
        self.database = self.agent._session_db
        self.console = cli.console
        self.app = cli._app
        self.prompt = self.owner.prompt
        self.manager = context._manager
        self.state = context.state
        self.manifest = context.manifest
        self.whole_layout = whole_layout
        self.classes = classes
        self.functions = {}
        for name, row in function_qualifications.items():
            module = sys.modules[row["module"]]
            value = module
            for part in row["attribute"].split("."):
                value = getattr(value, part)
            self.functions[name] = real_function(value, row)
        self.host_birth = (proc_birth(os.getpid()) if source_qualification is None
                           else source_qualification.proc_birth(os.getpid()))
        self.operation_counter = 0
        self.check()

    def check(self, *, input_required=True, output_required=True):
        from hermes_cli import friday_cli_work as work
        from hermes_cli.friday_cli_principal import current_cli
        from prompt_toolkit.application.current import get_app_session
        if (self.cli.agent is not self.agent or self.cli._app is not self.app
                or self.cli._friday_cli_scope is not self.cap
                or self.cli._friday_cli_output is not self.owner
                or self.agent._session_db is not self.database
                or self.cli.console is not self.console
                or self.owner.console is not self.console
                or self.ctx._manager is not self.manager
                or self.ctx.state is not self.state
                or self.ctx.manifest is not self.manifest
                or self.host.ctx is not self.ctx
                or self.manager._cli_ref is not self.cli
                or current_cli(required=True) is not self.cap):
            raise Unbound("original_live_binding_lost")
        if output_required:
            self.owner.check()
        else:
            self.owner.check_lifetime()
        self.cap.check()
        if output_required and self.prompt is not None:
            if (self.owner.prompt is not self.prompt
                    or self.prompt.session is not self.session
                    or self.prompt.app is not self.app):
                raise Unbound("original_app_session_lost")
            self.prompt.check()
        # Current authenticated status uses the genuine command producer,
        # without inventing a current send/input turn from historical state.
        if not input_required:
            return None
        # Only an actual original frontend input turn can drive local work.
        turn = work.current_turn()
        if (turn["cli"] is not self.cli or turn["agent"] is not self.agent
                or turn["cap"] is not self.cap or turn["output"] is not self.owner):
            raise Unbound("no_original_frontend_input_turn")
        return turn

    def next_operation(self):
        self.check()
        self.operation_counter += 1
        return OperationCustody(self, self.operation_counter)


class OperationCustody:
    def __init__(self, binding, ordinal):
        self.binding, self.ordinal = binding, ordinal
        witness = getattr(binding, "caller_witness", None)
        if witness is None:
            raise Unbound("original_whole_caller_entry_witness_UNBOUND")
        witness.require_original(binding)
        self.caller_witness = witness
        self.started = witness.entered
        self.deadline = witness.deadline
        self.stop_intent = None
        self.caller_task = self.caller_loop = None
        self.actor = threading.current_thread()
        self.actor_tid = threading.get_native_id()
        actor_raw = Path(f"/proc/self/task/{self.actor_tid}/stat").read_text()
        closing = actor_raw.rfind(")")
        self.actor_birth = {"native_id": self.actor_tid,
                            "starttime": int(actor_raw[closing+2:].split()[19]),
                            "raw_stat": actor_raw}
        self.native_operation = None
        self.closed = False
        self.complete_after_tail = False

    def retain_original_task(self):
        loop = asyncio.get_running_loop()
        task = asyncio.current_task()
        if task is None:
            raise Unbound("no_original_owned_task")
        if self.caller_task is not None and (
                task is not self.caller_task or loop is not self.caller_loop):
            raise Unbound("caller_task_or_loop_changed")
        self.caller_task, self.caller_loop = task, loop

    def check(self, *, cleanup=False):
        self.caller_witness.require_original(self.binding)
        if self.closed:
            raise Unbound("operation_closed")
        if not cleanup and self.stop_intent is not None:
            raise Unbound("original_stop_intent")
        if time.monotonic() >= self.deadline:
            raise Unbound("original_whole_caller_30_exhausted")
        return self.binding.check()

    def end_after_original_tail(self, observations):
        witness = self.caller_witness
        if witness.returned is None or witness.returned > witness.deadline:
            raise Unbound("actual_outer_caller_not_returned_on_time")
        if witness.unknown or witness.failures:
            raise Unbound("original_whole_caller_tail_unconfirmed")
        required = {"original_caller_return", "activity_join_return",
                    "metadata_callback_return", "session_flush_return",
                    "ui_completed_return"}
        seen = {x["site"] for x in observations}
        if not required <= seen:
            raise Unbound("native_risky_tail_not_observed")
        returned = [x for x in observations if x["site"] == "original_caller_return"]
        if len(returned) != 1 or returned[0]["monotonic"] >= self.deadline:
            raise Unbound("caller_return_not_exact_on_time")
        for item in observations:
            if item["site"] in required and item["monotonic"] > returned[0]["monotonic"]:
                raise Unbound("tail_after_caller_completion")
        self.complete_after_tail = True
        self.closed = True


class NativeTrace:
    def __init__(self, custody, sites):
        """sites are separately qualified exact code hashes and events.

        Each entry identifies an existing function/line, not a label supplied
        by the operation. A native site never writes document bytes for us.
        """
        self.custody = custody
        self.sites = sites
        self.events = []
        self.overflow = False
        self.failure = None
        self.installed = False
        self.original_sys = self.original_threads = None
        self.refs = []  # exact references retained until real joined cleanup
        self.authorized_tids = {threading.get_native_id()}
        self.ledger_witnesses = []
        self.code_cache = {}

    def append(self, site, frame, event, result=None):
        if len(self.events) >= EVENT_LIMIT:
            self.overflow = True
            raise Unbound("observer_capacity_exhausted")
        tid = threading.get_native_id()
        raw = Path(f"/proc/self/task/{tid}/stat").read_text()
        closing = raw.rfind(")")
        entry = {"site": site, "event": event, "monotonic": time.monotonic(),
                 "actor_tid": tid, "actor_birth": int(raw[closing + 2:].split()[19]),
                 "code": code_digest(frame.f_code), "frame": id(frame),
                 "caller_frame": id(frame.f_back) if frame.f_back else None}
        loc = frame.f_locals
        if site == "ledger_task_created":
            from .ledger import witness_actual_return
            binding = self.custody.binding
            qualification = self.sites[site]
            witness = witness_actual_return(event=event, frame=frame,
                returned=result, qualification=qualification, binding=binding)
            self.ledger_witnesses.append(witness)
            binding.ledger_witness = witness
            entry["ledger"] = witness.observation()
        value = loc.get("self")
        if value is not None:
            self.refs.append(value)
            entry["self"] = object_key(value)
        if site == "physical_write_return":
            chunk = loc.get("chunk")
            entry["count_type"] = type(result).__name__
            entry["accepted_count"] = result if type(result) is int else None
            if type(chunk) is bytes:
                entry["requested_bytes"] = len(chunk)
                entry["requested_sha256"] = hashlib.sha256(chunk).hexdigest()
                if type(result) is int and 0 < result <= len(chunk):
                    entry["accepted_sha256"] = hashlib.sha256(chunk[:result]).hexdigest()
            lease = value
            if lease is not None:
                entry["descriptor"] = lease.fd
                entry["lease_accepted"] = lease.accepted
                entry["original_identity"] = lease.original_identity
        if site == "heartbeat_started":
            thread = loc.get("thread")
            self.refs.append(thread)
            entry["thread"] = native_thread(thread)
            self.authorized_tids.add(thread.native_id)
        if site == "activity_join_return":
            thread = loc.get("thread")
            self.refs.append(thread)
            entry["thread_object"] = object_key(thread)
            entry["thread_native_id"] = getattr(thread, "native_id", None)
            entry["alive_after_join"] = thread.is_alive() if thread is not None else None
        if site == "authenticated_status_return":
            if type(result) is not str or len(result.encode()) > BODY_LIMIT:
                raise Unbound("actual_status_return_not_bounded_native_text")
            entry["return_text"] = result
            entry["return_sha256"] = hashlib.sha256(result.encode()).hexdigest()
        if site.startswith("prompt_"):
            op = loc.get("op")
            if op is not None:
                self.refs.append(op)
                entry["operation"] = object_key(op)
                entry["ordinal"] = op.ordinal
                entry["phase"] = op.phase
                entry["descriptor_closed"] = op.descriptor_closed
                entry["body_accepted"] = op.body_accepted
                entry["finished_within_deadline"] = op.finished_within_deadline
        self.events.append(entry)

    def trace(self, frame, event, result):
        # Never alter a native return, inject a success or swallow a guard.
        try:
            digest = self.code_cache.get(frame.f_code)
            if digest is None:
                digest = code_digest(frame.f_code)
                self.code_cache[frame.f_code] = digest
            for name, row in self.sites.items():
                if (digest == row["code_sha256"]
                        and frame.f_code.co_filename == row["filename"]
                        and frame.f_code.co_qualname == row["qualname"]
                        and event == row["event"]
                        and (row.get("line") is None or frame.f_lineno == row["line"])):
                    self.append(name, frame, event, result)
        except BaseException as exc:
            self.failure = type(exc).__name__
            raise
        return self.trace

    def install_current_and_future_native_threads(self):
        # A preexisting trace/guard cannot be displaced.
        self.original_sys = sys.gettrace()
        self.original_threads = threading.gettrace()
        if self.original_sys is not None or self.original_threads is not None:
            raise Unbound("existing_trace_or_guard_must_not_be_replaced")
        if self.installed:
            raise Unbound("duplicate_observer_install")
        threading.settrace(self.trace)
        sys.settrace(self.trace)
        self.installed = True

    def detach(self):
        if self.installed:
            if (sys.gettrace() != self.trace or threading.gettrace() != self.trace):
                raise Unbound("observer_ownership_lost")
            sys.settrace(self.original_sys)
            threading.settrace(self.original_threads)
            self.installed = False
        # Existing native threads must finish/detach separately. A changed
        # global hook does not prove that a preexisting native loop lost its hook.

    def result(self):
        if self.overflow or self.failure:
            raise Unbound("observer_unknown_or_failed")
        if self.installed:
            raise Unbound("observer_not_detached")
        return {"events": self.events, "overflow": False,
                "unknown_native_threads": "REQUIRES_EXACT_ROOT_POSTRETURN_REAP",
                "complete_after_tail": self.custody.complete_after_tail}
