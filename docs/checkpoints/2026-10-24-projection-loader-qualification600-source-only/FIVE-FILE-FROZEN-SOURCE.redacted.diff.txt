+def _consumed_tree(root):
+    """Complete descriptor-relative CAS, including retained 0400 stop sources."""
+    def identity(i):

+            ('dev','ino','mode','uid','gid','nlink','size','mtime_ns','ctime_ns')]

+            and stat.S_IMODE(first.st_mode) == 0o700, 'worker_derivative_private_directory_required')

+        for leaf in sorted(os.listdir(fd)):

+            info = os.stat(leaf, dir_fd=fd, follow_symlinks=False)
+            require(not stat.S_ISLNK(info.st_mode), 'worker_derivative_tree_link_refused')
+            flags = os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK

+            child = os.open(leaf, flags, dir_fd=fd)
