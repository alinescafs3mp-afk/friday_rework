def checked_absence(parsed,inventory,target):
    result=inventory(parsed,target)
    need(result['target_persistent_metadata']=='ABSENT' and not result['target_matches']
         and result['endpoint_record_count']==0 and result['unclassified_records']==0
         and not any(n['id']==target.get('id') for n in result['networks'])
         and not any(b['bridge_name']==target['bridge'] for b in result['bridge_driver_records']),
         'target_network_not_absent')
def network_evidence(acceptance,closure,original_state,retirement,targets):
    """Historical wrapper proves original deletion, never current DB identity."""
    if acceptance.get('schema')=='frw.Root.actual-R421-network-closure-premises.v1':
        route=original_state['route'];network=route['network']
        need(acceptance.get('accepted') is True
             and acceptance.get('actual_original_stop_and_network_removed') is True
             and acceptance.get('no_fresh_network_API_list_claim') is True
             and acceptance.get('original_stopped_state')==closure['source_evidence']['original_bootstrap_state']
             and acceptance.get('accepted_maintenance')==RETIREMENT_REF==closure['source_evidence']['container']
             and acceptance.get('archive_closed')==closure['source_evidence']['archive']
             and acceptance.get('network_id')==network['id']
             and acceptance.get('network_name')==network['name']
             and route['network_removed'] is True and route['pending'] is None
             and route['settlement']['status']=='STOP_CONFIRMED'
             and route['settlement']['network_removed'] is True
             and retirement.get('accepted') is True and retirement.get('native_stop_confirmed') is True,
             'actual_network_wrapper_provenance_changed')
        archive=json.loads(checked(acceptance['archive_closed']))
        need(archive.get('accepted') is True and archive.get('archive_mapping')==closure['archive_mapping'],
             'actual_network_archive_join_changed')
        need(any(t.get('id')==acceptance['network_id'] and t['name']==acceptance['network_name']
                 for t in targets),'actual_network_wrapper_target_changed')
        # No historical DB receipt exists for this exact R421 target. Its
        # persistent DB was preserved by R434, so fresh complete inventory is
        # mandatory below. R420's different-target DB receipt is never used.
        return None
    need(acceptance.get('target_absent') is True and type(acceptance.get('database')) is dict,
         'actual_database_absence_required')
    need(any(acceptance.get('target_id')==target.get('id') and 'id' in target
             and acceptance.get('target_name')==target['name']
             and acceptance.get('target_nonce')==target['nonce']
             and acceptance.get('target_owner')==target['owner'] for target in targets),
         'actual_database_target_join_required')
    return acceptance['database']
def database_path_checked(path):
    path=Path(path)
    need(path==NATIVE_HOME.parents[1]/'.runtime/rootless-docker/data/network/files/local-kv.db'
         and path.is_absolute() and path.resolve()==path,'database_scope_changed')
    for parent in path.parents:
        info=parent.lstat()
        need(stat.S_ISDIR(info.st_mode) and info.st_uid in (0,os.getuid())
             and not info.st_mode&0o022,'database_parent_changed')
    return path
def database_observation(stream,dbpath):
    before=os.fstat(stream.fileno())
    need(stat.S_ISREG(before.st_mode) and before.st_uid==os.getuid()
         and before.st_nlink==1 and not before.st_mode&0o022,'database_custody_changed')
    stream.seek(0);raw=stream.read(64*1024*1024+1)
    need(len(raw)<=64*1024*1024
         and identity(before)==identity(os.fstat(stream.fileno()))==identity(dbpath.lstat()),
         'database_changed')
    return raw,{'path':str(dbpath),'sha256':sha(raw),
        'identity':[before.st_dev,before.st_ino,before.st_size,before.st_mtime_ns,before.st_ctime_ns]}
