_DIAGNOSTIC_NAME = 'DECLARATION-DIAGNOSTIC.json'
_DIAGNOSTIC_CLASSES = frozenset(('Refused','DeclarationError','FridayInstallError',
    'StopUnconfirmed','ValueError','TypeError','KeyError','AttributeError',
    'NameError','ImportError','ModuleNotFoundError','FileNotFoundError',
    'PermissionError','OSError','RuntimeError','TimeoutError','AssertionError'))

def _diagnostic_class(error):
    name=type(error).__name__
    return name if name in _DIAGNOSTIC_CLASSES else 'UNLISTED_EXCEPTION'

def _diagnostic_source(state, reference_value, label):
    # Caller adds only sources already authenticated by the ordinary guards or
    # held-FD loader. No reads, imports or traceback pathname are emitted here.
    reference(reference_value)
    state['sources'][reference_value['path']]={'source':label,
        'source_sha256':reference_value['sha256']}

def _diagnostic_phase(state, phase):
    state['phase']=phase

_DIAGNOSTIC_META = ('st_dev','st_ino','st_uid','st_gid','st_mode',
    'st_nlink','st_size','st_mtime_ns','st_ctime_ns')
_DIAGNOSTIC_MAXIMUM_BYTES = 32768

def _diagnostic_encoded(value):
    # Compact only this doubled diagnostic; business/proof encoding is unchanged.
    return (json.dumps(value,sort_keys=True,ensure_ascii=False,allow_nan=False,
        separators=(',',':'))+'\n').encode()

def _diagnostic_path_label(home, value):
    # Lexical only. No resolve/stat/read and no content-derived values.
    require(type(value) is str and len(value)<=16384,'diagnostic_path_shape')
    path=Path(value);root=Path(home)
    if path.is_absolute() and '..' not in path.parts and path.is_relative_to(root):
        relative=str(path.relative_to(root))
        # Bound the JSON representation too: control characters can expand sixfold.
        if len(json.dumps(relative,ensure_ascii=False).encode())<=130:
            return {'relative_H':relative}
    return {'absolute_path_sha256':digest(os.fsencode(value))}

def _diagnostic_field_differences(left, right, names):
    require(type(left) is tuple and type(right) is tuple
        and len(left)==len(right)==len(names)
        and all(type(value) is int for value in left+right),'diagnostic_identity_shape')
    return [name for name,a,b in zip(names,left,right) if a!=b]

def _diagnostic_parent_proof(value):
    require(type(value) is tuple and len(value)<=64,'diagnostic_parent_bound')
    result={}
    for row in value:
        require(type(row) is tuple and len(row)==2 and type(row[0]) is str
            and len(row[0])<=16384 and row[0] not in result,'diagnostic_parent_shape')
        _diagnostic_field_differences(row[1],row[1],_DIAGNOSTIC_META[:5])
        result[row[0]]=row[1]
    return result

def _diagnostic_merge_observation(state, frame):
    # This helper is called ONLY for the code object held from the genuine
    # authenticated private ordinary module, never for a name/path lookalike.
    values=frame.f_locals
    require(type(values.get('self')) is state['_merge_class'],'diagnostic_merge_receiver')
    home=state['_merge_home']
    require(type(home) is str and Path(home).is_absolute(),'diagnostic_home_shape')
    section=values.get('section');old=values.get('old');new=values.get('value')
    require(type(section) is str and section in ('files','directories')
        and type(old) is tuple and type(new) is tuple
        and len(old)==len(new)==2,'diagnostic_merge_shape')
    path=_diagnostic_path_label(home,values.get('path'))
    if section=='directories':
        old_identity,old_names=old;new_identity,new_names=new
        require(type(old_identity) is tuple and len(old_identity) in (5,9),
            'diagnostic_directory_identity_shape')
        names=_DIAGNOSTIC_META[:len(old_identity)]
        differing=_diagnostic_field_differences(old_identity,new_identity,names)
        for entries in (old_names,new_names):
            require(entries is None or (type(entries) is tuple and len(entries)<=4096
                and all(type(name) is str and len(name)<=255 for name in entries)),
                'diagnostic_directory_membership_shape')
        return {'status':'OBSERVED','path':path,'section':section,
            'identity_differing_fieldnames':differing,
            'directory_membership_equal':old_names==new_names,
            'directory_entry_names_included':False}
    old_before,old_sha=old;new_before,new_sha=new
    require(type(old_before) is tuple and type(new_before) is tuple
        and len(old_before)==len(new_before)==2,'diagnostic_file_before_shape')
    require(type(old_sha) is str and type(new_sha) is str
        and re.fullmatch('[0-9a-f]{64}',old_sha) is not None
        and re.fullmatch('[0-9a-f]{64}',new_sha) is not None,'diagnostic_content_hash_shape')
    old_identity,old_parents=old_before;new_identity,new_parents=new_before
    differing=_diagnostic_field_differences(old_identity,new_identity,_DIAGNOSTIC_META)
    left=_diagnostic_parent_proof(old_parents);right=_diagnostic_parent_proof(new_parents)
    parents=[];changed=0
    for parent in sorted(set(left)|set(right)):
        membership=parent not in left or parent not in right
        if membership or left[parent]!=right[parent]:
            changed+=1
            if len(parents)<4:
                row={'path':_diagnostic_path_label(home,parent),'membership_changed':membership}
                if not membership:
                    row['differing_fieldnames']=_diagnostic_field_differences(
                        left[parent],right[parent],_DIAGNOSTIC_META[:5])
                parents.append(row)
    return {'status':'OBSERVED','path':path,'section':section,
        'old_new_content_hash_equal':old_sha==new_sha,
        'identity9_differing_fieldnames':differing,
        'parent_proof_equal':old_parents==new_parents,
        'parent_proof_order_changed':list(left)!=list(right),
        'parent_proof_differences':parents,'parent_differences_observed':changed,
        'parent_differences_omitted':max(0,changed-len(parents))}

def _diagnostic_payload(state, error):
    chain=[];seen=set();current=error
    while current is not None and id(current) not in seen and len(chain)<4:
        seen.add(id(current));frames=[];tb=current.__traceback__;walked=0
        merge=None;merge_frames=0
        while tb is not None and walked<64:
            walked+=1
            source=state['sources'].get(tb.tb_frame.f_code.co_filename)
            name=tb.tb_frame.f_code.co_name
            if source is not None and re.fullmatch(r'[A-Za-z_][A-Za-z0-9_]{0,99}|<module>',name):
                frames.append({**source,'function':name,'line':tb.tb_lineno})
            if tb.tb_frame.f_code is state.get('_merge_code'):
                merge_frames+=1
                if merge is None:
                    try:merge=_diagnostic_merge_observation(state,tb.tb_frame)
                    except Exception as observation_error:
                        merge={'status':'UNAVAILABLE','error_class':_diagnostic_class(observation_error)}
            tb=tb.tb_next
        count=len(frames)
        chain.append({'exception_class':_diagnostic_class(current),'frames':frames[-8:],
            'outer_frames':frames[:8],'trace_truncated':tb is not None,
            'trace_frames_walked':walked,'authenticated_frames_walked':count,
            'authenticated_frames_retained':min(count,16),
            'authenticated_frames_omitted':max(0,count-16),
            'inner_frames_sliced':count>8,'traversal_cap_reached':tb is not None,
            'outer_inner_overlap':max(0,2*min(8,count)-count),
            'merge_frames_matched':merge_frames,'merge_frames_omitted':max(0,merge_frames-1),
            'merge_observation':merge})
        current=current.__cause__ if current.__cause__ is not None else (
            None if current.__suppress_context__ else current.__context__)
    return {'schema':'frw.Root.protected-declaration-failure.v1',
        'correlation':state['correlation'],'request_sha256':state['request_sha256'],
        'phase':state['phase'],'stable_unwind_entered':state.get('stable_unwind_entered',False),
        'exceptions':chain,'message_or_content_included':False,
        'earlier_refusal_cause_proven':False,'no_retry':True}

def _diagnostic_save(binding, state, error):
    # Request directory is an owned cache destination; never follow H refs or
    # replace an earlier diagnostic. Publication is atomic and mode0600.
    request=cache_binding_reference(binding['request']);parent=Path(request['path']).parent
    parents={}
    for p in reversed((parent,*parent.parents)):
        info=p.lstat()
        require(stat.S_ISDIR(info.st_mode) and not stat.S_ISLNK(info.st_mode)
            and info.st_uid in (0,os.getuid()) and info.st_mode&0o022==0,'diagnostic_parent_identity')
        parents[p]=(info.st_dev,info.st_ino,info.st_uid,info.st_gid,info.st_mode)
    before=parents[parent]
    require(before[2]==os.getuid() and stat.S_IMODE(before[4])==0o700,'diagnostic_private_cache_directory')
    raw=_diagnostic_encoded(_diagnostic_payload(state,error));require(len(raw)<=_DIAGNOSTIC_MAXIMUM_BYTES,'diagnostic_bound')
    directory=os.open(parent,os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW)
    temporary='.declaration-diagnostic-'+os.urandom(16).hex()+'.tmp';created=False
    try:
        info=os.fstat(directory)
        require((info.st_dev,info.st_ino,info.st_uid,info.st_gid,info.st_mode)==before,'diagnostic_directory_changed')
        fd=os.open(temporary,os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o600,dir_fd=directory);created=True
        with os.fdopen(fd,'wb') as stream:
            stream.write(raw);stream.flush();os.fsync(stream.fileno())
        for p,identity in parents.items():
            info=p.lstat()
            require((info.st_dev,info.st_ino,info.st_uid,info.st_gid,info.st_mode)==identity,'diagnostic_parent_changed')
        os.link(temporary,_DIAGNOSTIC_NAME,src_dir_fd=directory,dst_dir_fd=directory,follow_symlinks=False)
        os.unlink(temporary,dir_fd=directory);created=False;os.fsync(directory)
        return digest(raw)
    finally:
        if created:os.unlink(temporary,dir_fd=directory)
        os.close(directory)
