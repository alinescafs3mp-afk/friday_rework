def _accepted_maintenance_history(home,receipt,request,historical,current_history):
    """Reuse the exact previously authenticated maintenance role, not ordinary ancestry.

    The accepted current semantic execution validated this pinned latest ordinary
    request through the genuine HistoryWalk. Its bridge is the exact prior
    maintenance history, including original registration and recovery journals.
    Current physical custody for every reused digest is refreshed by the caller.
    The unchanged final consumer independently reconstructs this same role.
    """
    prior=historical.get('worker_source_maintenance')
    need(type(prior)is dict and set(prior)=={'schema','receipt','request','committed','history'}
        and prior['schema']=='friday.ordinary-worker-declaration-successor-bridge.v1'
        and receipt.get('previous_source_successor')==request.get('previous_source_successor')==prior['receipt']
        and 'recovery' not in receipt
        and historical.get('home')==str(home)
        and request['source_update_ancestry']['latest_request']==LATEST_REQUEST
        and receipt['source_update_ancestry']==request['source_update_ancestry'],
        'exact_accepted_prior_maintenance_role_required')
    previous=dict(prior['history']);ancestry=request['source_update_ancestry']['history']
    need(previous and all(type(n)is str and pathlib.Path(n).parts
        and not pathlib.Path(n).is_absolute() and pathlib.Path(n).as_posix()==n
        and all(p not in ('.','..') for p in pathlib.Path(n).parts)
        and type(h)is str and len(h)==64 and set(h)<=set('0123456789abcdef')
        and ancestry.get(n)==h for n,h in previous.items()),
        'accepted_prior_maintenance_ancestry_changed')
    prior_tx=pathlib.Path(prior['receipt']['path']).parent
    need(prior_tx.parent==home/'preparation' and prior_tx.name.startswith('worker-declarations-successor-')
        and prior['receipt']['path']==str(prior_tx/'service-source-receipt.json')
        and prior['committed']['path']==str(prior_tx/'committed.json')
        and all(previous.get(str((prior_tx/n).relative_to(home)))==prior[k]['sha256']
            for n,k in (('request.json','request'),('committed.json','committed'),('service-source-receipt.json','receipt'))),
        'accepted_prior_maintenance_journal_join_changed')
    predecessor=request['predecessor'];prefix=pathlib.Path(predecessor['prefix'])
    need(predecessor==receipt['predecessor'] and prefix.parent==home/'preparation'
        and prefix.name.startswith('ordinary-a0-registration-'),
        'accepted_registration_prefix_role_changed')
    registration={str((prefix/n).relative_to(home)):row['sha256']
        for n,row in predecessor['inventory'].items() if n!='.' and row.get('kind')=='file'}
    prefix_rel=prefix.relative_to(home)
    need(registration and {n:h for n,h in previous.items()
        if pathlib.Path(n).is_relative_to(prefix_rel)}==registration,
        'accepted_registration_prefix_inventory_changed')
    tx=pathlib.Path(receipt['transaction'])
    need(type(current_history)is dict and current_history
        and all(pathlib.Path(n).is_relative_to(tx.relative_to(home)) for n in current_history)
        and not set(previous)&set(current_history),
        'accepted_current_maintenance_journal_role_changed')
    previous.update(current_history)
    return previous
