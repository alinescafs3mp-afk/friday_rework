class _HistoryWalk:
    """Exact completed graph facts retained only by one finite invocation.

    A successful fact owns the bytes and identities observed by its checker,
    including external pinned reads and immutable directory membership. Every
    reuse re-authenticates that custody; publication authenticates it again.
    """
    _identity_fields = ('st_dev','st_ino','st_uid','st_gid','st_mode','st_nlink',
        'st_size','st_mtime_ns','st_ctime_ns')
    _parent_fields = ('st_dev','st_ino','st_uid','st_gid','st_mode')

    def __init__(self, *, budget=None):
        self._budget = budget
        self._completed = {}
        self._active = set()
        self._frames = []
        self._home = None
        self._hits = 0
        self._misses = 0
        self._anchor = None
        self._anchor_identity = None
        self._scope_parents = None

    @staticmethod
    def _key(kind, arguments):
        def plain(value):
            if value is None or type(value) in (str, bool, int):
                return True
            if type(value) is list:
                return all(plain(item) for item in value)
            if type(value) is dict:
                return all(type(key) is str and plain(item) for key, item in value.items())
            return False
        if not plain(arguments):
            return None
        return json.dumps([kind, arguments], sort_keys=True, separators=(',', ':'), allow_nan=False)

    def _check_budget(self):
        if self._budget is not None:
            self._budget.check()

    @classmethod
    def _identity(cls, info, fields=None):
        return tuple(getattr(info, key) for key in (cls._identity_fields if fields is None else fields))

    def _custody_identity(self, path, info):
        private_tree = self._home is not None and (path == self._home or self._home in path.parents)
        return self._identity(info, None if private_tree else self._parent_fields)

    @staticmethod
    @contextlib.contextmanager
    def _fd_ownership():
        # The existing deadline raises asynchronously from SIGALRM. Defer it
        # (and KeyboardInterrupt) only until descriptor ownership is recorded
        # or all owned closes are attempted; never change the timer or clock.
        signal = __import__('signal')
        previous = signal.pthread_sigmask(signal.SIG_BLOCK, {signal.SIGALRM, signal.SIGINT})
        try:
            yield
        finally:
            signal.pthread_sigmask(signal.SIG_SETMASK, previous)

    @contextlib.contextmanager
    def _directory_custody(self, path):
        """Hold each named directory without ever following a parent link."""
        path = Path(path)
        if not path.is_absolute() or '..' in path.parts:
            raise ValueError('ordinary_history_canonical_leaf_required')
        held = []
        try:
            for part in path.parts:
                named = Path('/') if not held else held[-1][0]/part
                with self._fd_ownership():
                    fd = (os.open('/', os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW) if not held else
                        os.open(part, os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW, dir_fd=held[-1][1]))
                    # Register before fstat/identity or a pending deadline can
                    # raise; failed acquisition validation still owns cleanup.
                    held.append((named, fd, None))
                held[-1] = (named, fd, self._custody_identity(named, os.fstat(fd)))
            self._prove_directories(held)
            yield held
            self._prove_directories(held)
        finally:
            primary = __import__('sys').exc_info()[1]
            failures = []
            with self._fd_ownership():
                for _, fd, _ in reversed(held):
                    try:
                        os.close(fd)
                    except BaseException as exc:
                        failures.append(exc)
            if failures:
                if primary is None:
                    raise failures[0]
                # Preserve cancellation/the read failure while retaining every
                # cleanup failure; one failed close never skips the others.
                primary.friday_history_cleanup_failures = tuple(failures)

    def _prove_directories(self, held):
        for index, (path, fd, identity) in enumerate(held):
            named = (path.lstat() if not index else
                os.stat(path.name, dir_fd=held[index-1][1], follow_symlinks=False))
            # The full leaf-parent identity is stable over a read, including
            # ctime; the existing memo parent proof deliberately stays five
            # fields because H/preparation gain this invocation's transaction.
            if (self._custody_identity(path, os.fstat(fd)) != identity
                    or self._custody_identity(path, named) != identity
                    or self._custody_identity(path, path.lstat()) != identity):
                raise ValueError('ordinary_history_parent_changed')

    @contextlib.contextmanager
    def _anchor_scope(self):
        if self._anchor is not None or self._home is None:
            yield
            return
        with self._directory_custody(self._home) as held:
            identity = self._identity(os.fstat(held[-1][1]), self._parent_fields)
            if self._anchor_identity is not None and identity != self._anchor_identity:
                raise ValueError('ordinary_history_private_anchor_changed')
            self._anchor_identity = identity
            self._anchor = held
            self._scope_parents = {}
            try:
                yield
                self._prove_directories(held)
            finally:
                self._anchor = None
                self._scope_parents = None

    def _private_leaf(self, path, parents):
        if self._anchor is None or self._home not in path.parents:
            return False
        self._prove_directories(self._anchor)
        anchor = os.fstat(self._anchor[-1][1])
        if not (anchor.st_uid == os.getuid() and anchor.st_gid == os.getgid()
                and stat.S_IMODE(anchor.st_mode) == 0o700):
            return False
        # No other user can newly traverse the held canonical 0700 anchor.
        # Privileged, same-UID and previously held foreign descriptors are not
        # excluded by this local discretionary-access guarantee.
        for parent, info in parents:
            if parent == self._home or self._home in parent.parents:
                if not (info.st_uid == os.getuid() and info.st_gid == os.getgid()
                        and not info.st_mode & 0o7002):
                    return False
        return True

    def before_read(self, path):
        self._check_budget()
        path = Path(path)
        if not path.is_absolute() or '..' in path.parts or '.' in path.parts:
            raise ValueError('ordinary_history_canonical_leaf_required')
        if self._anchor is not None:
            self._prove_directories(self._anchor)
        parents = []
        for parent in path.parents:
            info = parent.lstat()
            if not stat.S_ISDIR(info.st_mode):
                raise ValueError('ordinary_history_parent_changed')
            if self._scope_parents is not None:
                identity = self._custody_identity(parent, info)
                prior = self._scope_parents.setdefault(str(parent), identity)
                if prior != identity:
                    raise ValueError('ordinary_history_parent_changed')
            parents.append((parent, info))
        info = path.lstat()
        private_groupwrite = (stat.S_IMODE(info.st_mode) == 0o664
            and self._private_leaf(path, parents))
        if not (stat.S_ISREG(info.st_mode) and info.st_uid == os.getuid()
                and info.st_gid == os.getgid() and info.st_nlink == 1
                and not info.st_mode & 0o7002
                and (not info.st_mode & 0o020 or private_groupwrite)):
            raise ValueError('ordinary_history_owned_leaf_required')
        return self._identity(info), tuple((str(parent), self._identity(st, self._parent_fields))
            for parent, st in parents)

    def read_leaf(self, path):
        path = Path(path)
        with self._anchor_scope(), self._directory_custody(path.parent) as held:
            before = self.before_read(path)
            fd = None
            try:
                with self._fd_ownership():
                    fd = os.open(path.name, os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK, dir_fd=held[-1][1])
                if self._identity(os.fstat(fd)) != before[0]:
                    raise ValueError('ordinary_history_leaf_identity_changed')
                with os.fdopen(fd, 'rb', closefd=False) as stream:
                    data = stream.read(64*1024**2+1)
                if (len(data) > 64*1024**2 or len(data) != before[0][6]
                        or self._identity(os.fstat(fd)) != before[0]
                        or self.before_read(path) != before):
                    raise ValueError('ordinary_history_leaf_changed')
                return data
            finally:
                if fd is not None:
                    with self._fd_ownership():
                        os.close(fd)

    def _merge(self, section, path, value):
        for frame in self._frames:
            old = frame[section].get(path)
            if old is not None and old != value:
                raise ValueError('ordinary_history_observation_changed')
            frame[section][path] = value

    def record_read(self, path, raw, before):
        if type(raw) is not bytes or self.before_read(path) != before:
            raise ValueError('ordinary_history_read_changed')
        proof = (before, __import__('hashlib').sha256(raw).hexdigest())
        self._merge('files', str(path), proof)

    def history_directory(self, path):
        path = Path(path)
        return (self._home is not None and path != self._home/'preparation'
            and (self._home/'preparation') in path.parents)

    def directory_state(self, path):
        self._check_budget()
        path = Path(path); info = path.lstat()
        if not stat.S_ISDIR(info.st_mode):
            raise ValueError('ordinary_history_directory_changed')
        # These two shared containers legitimately gain this invocation's new
        # transaction. Their identities remain fixed; old graph roots do not.
        mutable = self._home is not None and path in (self._home, self._home/'preparation')
        identity = self._identity(info, self._parent_fields if mutable else None)
        names = None if mutable else tuple(sorted(item.name for item in path.iterdir()))
        if self._identity(path.lstat(), self._parent_fields if mutable else None) != identity:
            raise ValueError('ordinary_history_directory_changed')
        return identity, names

    def record_directory(self, path, before):
        if self.directory_state(path) != before:
            raise ValueError('ordinary_history_directory_changed')
        self._merge('directories', str(path), before)

    def _prove(self, proof):
        with self._anchor_scope():
            return self._prove_scoped(proof)

    def _prove_scoped(self, proof):
        for path, (before, sha) in proof['files'].items():
            self._check_budget()
            if (self.before_read(path) != before
                    or __import__('hashlib').sha256(self.read_leaf(path)).hexdigest() != sha):
                raise ValueError('ordinary_history_reused_leaf_changed')
        for path, before in proof['directories'].items():
            if self.directory_state(path) != before:
                raise ValueError('ordinary_history_reused_directory_changed')

    def prove(self):
        """Fresh exact graph custody immediately before publication/commit."""
        self._check_budget()
        if self._active:
            raise ValueError('ordinary_history_incomplete_scope')
        proof = {'files': {}, 'directories': {}}
        for _, completed in self._completed.values():
            for section in proof:
                for path, value in completed[section].items():
                    if path in proof[section] and proof[section][path] != value:
                        raise ValueError('ordinary_history_observation_changed')
                    proof[section][path] = value
        self._prove(proof)

    def _seed(self, kind, arguments):
        if not self._frames or len(self._frames) != 1:
            return
        if kind in ('update_history', 'legacy_update_history'):
            expected = arguments[1]
            if type(expected) is not dict:
                return
            for name, sha in expected.items():
                rel = Path(name)
                if rel.is_absolute() or '..' in rel.parts or not rel.parts:
                    raise ValueError('ordinary_history_relative_leaf_required')
                path = self._home/rel; before = self.before_read(path)
                raw = self.read_leaf(path)
                if __import__('hashlib').sha256(raw).hexdigest() != sha:
                    raise ValueError('ordinary_history_expected_leaf_changed')
                self.record_read(path, raw, before)

    def _checked(self, kind, arguments, checker):
        if self._home is None:
            self._home = Path(arguments[0])
        elif self._home != Path(arguments[0]):
            raise ValueError('ordinary_history_scope_home_changed')
        try:
            with self._anchor_scope():
                return self._checked_scoped(kind, arguments, checker)
        except BaseException:
            self._completed.clear()
            raise

    def _checked_scoped(self, kind, arguments, checker):
        self._check_budget()
        key = self._key(kind, arguments)
        if key is None:
            checked = checker()
            self._check_budget()
            return checked
        if key in self._active:
            self._completed.clear()
            raise ValueError('ordinary_preflight_history_cycle')
        if self._home is None:
            self._home = Path(arguments[0])
        elif self._home != Path(arguments[0]):
            raise ValueError('ordinary_history_scope_home_changed')
        if key in self._completed:
            try:
                checked, proof = self._completed[key]
                self._prove(proof)
                for section in proof:
                    for path, value in proof[section].items():
                        self._merge(section, path, value)
                self._check_budget()
                self._hits += 1
                return copy.deepcopy(checked)
            except BaseException:
                self._completed.clear()
                raise
        self._misses += 1
        self._active.add(key)
        frame = {'files': {}, 'directories': {}}
        self._frames.append(frame)
        observer = api()._ORDINARY_HISTORY_OBSERVER
        token = observer.set(self)
        try:
            self._seed(kind, arguments)
            checked = checker()
            self._check_budget()
            # Nested acquisitions are joined to this outer proof; only its
            # boundary seals the full union, avoiding another recursive walk.
            if len(self._frames) == 1: self._prove(frame)
            self._completed[key] = (copy.deepcopy(checked), frame)
            return copy.deepcopy(checked)
        except BaseException:
            # A failed enclosing walk never leaves its successful children as
            # reusable partial evidence, including cycle/budget/read failures.
            self._completed.clear()
            raise
        finally:
            observer.reset(token)
            self._frames.pop()
            self._active.remove(key)
