import ast

import re

import pytest

ROOT = Path(__file__).resolve().parents[2]


def exact(relative, name, namespace):
    tree = ast.parse((ROOT / relative).read_text())
    node = next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == name)
    exec(compile(ast.Module(body=[node], type_ignores=[]), str(ROOT / relative), 'exec'), namespace)
    return namespace[name]


class Response:
    status_code = 200

    def __init__(self, text, *, encoding='identity', length=None):
        self.body = json.dumps({'jsonrpc': '2.0', 'id': 1, 'result': {'content': [{'type': 'text', 'text': text}]}}).encode()
        self.headers = {'Content-Encoding': encoding, 'Content-Length': str(len(self.body) if length is None else length)}
        self.pos = 0
        self.raw = self
        self._fp = types.SimpleNamespace(fp=types.SimpleNamespace(raw=types.SimpleNamespace(_sock=self)))

    def settimeout(self, seconds):
        assert seconds > 0

    def read1(self, size):
        chunk = self.body[self.pos:self.pos + size]
        self.pos += len(chunk)
        return chunk


@pytest.fixture
def reader():
    return exact('plugins/web/keyless_mcp_bounded.py', 'read_response', dict(json=json, re=re, time=time))


@pytest.mark.parametrize('size', [14999, 15000])
def test_inline_ceiling_unchanged(reader, size):
    assert reader(Response('x' * size), time.monotonic() + 5, RuntimeError) == 'x' * size


@pytest.mark.parametrize('text', ['x' * 15001, 7])
def test_raw_oversized_or_nonstr_refused(reader, text):
    with pytest.raises(RuntimeError, match='^keyless text bound$'):
        reader(Response(text), time.monotonic() + 5, RuntimeError)


def test_only_bounded_handler_result_can_leave_transport(reader):
    seen = []
    def handler(text, deadline):
        seen.append((len(text), deadline))
        return text[:15000]
    assert len(reader(Response('x' * 30000), time.monotonic() + 5, RuntimeError, text_handler=handler)) == 15000
    assert seen[0][0] == 30000


@pytest.mark.parametrize('result', ['x' * 15001, 7])

@pytest.mark.parametrize('mode', ['failure', 'stopped', 'foreign'])
def test_failed_publication_never_reacquires_native_cache(monkeypatch, mode):
    cap = types.SimpleNamespace(principal=('telegram',))
    authority = {'principal': 'synthetic'}
    scope = types.ModuleType('hermes_cli.friday_user_scope')
    scope.current = lambda: cap
    scope.check_agent = lambda agent: None
    scope.delegation_authority = lambda: authority
    approval = types.ModuleType('tools.approval_context')
    approval.get_current_session_key = lambda default='': 'session-key'
    monkeypatch.setitem(sys.modules, scope.__name__, scope)
    monkeypatch.setitem(sys.modules, approval.__name__, approval)
    agent = types.SimpleNamespace(session_id='session', _current_turn_id='turn', _web_run_deadline=123, _interrupt_requested=False)
    flag = types.SimpleNamespace(is_set=lambda: mode == 'stopped')
    call = types.SimpleNamespace(agent=agent, task_id='task', call_id='call', session_id='session', session_key='session-key', turn_id='turn', run_deadline=123, user_authority=authority, stopped=flag, turn=types.SimpleNamespace(stopped=flag, failure=object() if mode == 'failure' else None), deadline=lambda: time.monotonic() + 5)
    agent._tool_result_calls = {('task', 'call'): call}
    if mode == 'foreign':
        call.session_key = 'foreign-session'
    def cache(*args):
        raise AssertionError('native cache must not be reacquired')
    publication = exact('agent/tool_executor.py', '_tool_result_environment', dict(contextlib=contextlib, time=time, get_active_env=cache))
    if mode == 'foreign':
        with pytest.raises(RuntimeError, match='^tool_result_foreign_original_caller$'):
            with publication(agent, 'task', 'call'):
                pass
    else:
        with publication(agent, 'task', 'call') as env:
            assert env is None
