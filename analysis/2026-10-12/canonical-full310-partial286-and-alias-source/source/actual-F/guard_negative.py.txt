def _j_bind_public(module):
    ns=module if isinstance(module,dict) else module.__dict__
    original_deliver=ns['deliver_in_worker'];original_send=ns['send_prepared']
    metrics={'original_deliver_calls':0,'original_send_calls':0}

        _j_admit('nested_deliver')
        fault=[(name,kind) for name,kind in [('AS',resource.RLIMIT_AS),
            ('CPU',resource.RLIMIT_CPU),('FSIZE',resource.RLIMIT_FSIZE)]
            if resource.getrlimit(kind)!=_J_KERNEL_LIMIT(0,kind)]
        if fault:
            assert len(fault)==1 and baseline,'RESOURCE_VALID_FULL_GRANT_BASELINE_MISSING'
            name,kind=fault[0];bp,bs,root,loader=baseline[0]
            oldroot=ns['R'];oldloader=ns['DEPENDENCY_LOADER'];oldlimit=resource.getrlimit
            ns['R']=root;ns['DEPENDENCY_LOADER']=loader
            try:
                # Positive full grant reaches the unchanged preparation endpoint.
                resource.getrlimit=lambda k:_J_KERNEL_LIMIT(0,k)
                _j_before_effect()
                ns['prepare_delivery'](bp,bs,min(custody.operation_end,_j_clock['useful_deadline_monotonic']))
                resource.getrlimit=oldlimit
                try:ns['prepare_delivery'](bp,bs,min(custody.operation_end,_j_clock['useful_deadline_monotonic']))
                except ns['Refusal'] as exc:
                    assert str(exc)=='finite helper resource boundary required',str(exc)
                    ns['_j_resource_rows'].append({'resource':name,'PASS':True,
                        'positive_full_grant_baseline':True,'specific_refusal':str(exc),
                        'kernel_caps_unchanged':all(_J_KERNEL_LIMIT(0,k)==v for k,v in
                            [(resource.RLIMIT_AS,(256<<20,256<<20)),(resource.RLIMIT_CPU,(30,30)),
                             (resource.RLIMIT_FSIZE,(64<<20,64<<20)),(resource.RLIMIT_CORE,(0,0))]),
                        'actual_fork_or_native_effects':0})
                    raise

            finally:
                resource.getrlimit=oldlimit;ns['R']=oldroot;ns['DEPENDENCY_LOADER']=oldloader
        metrics['original_deliver_calls']+=1

        if isinstance(value,dict) and value.get('state')=='ENQUEUED' and not baseline:
            baseline.append((path,sha,ns['R'],ns['DEPENDENCY_LOADER']))

def _j_read_private(path, expected=None):
    import stat
    p = Path(path)
    for part in [*reversed(p.parents), p]:
        assert not stat.S_ISLNK(part.lstat().st_mode), 'NOT_RUN_SYMLINK_INPUT'
    before = p.lstat()
    assert stat.S_ISREG(before.st_mode) and before.st_uid == os.getuid()
    assert not before.st_mode & 0o077 and before.st_nlink == 1
    assert 0 < before.st_size <= (1 << 20)
    fd = os.open(p, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK)
    try:
        opened = os.fstat(fd)
        assert (before.st_dev, before.st_ino) == (opened.st_dev, opened.st_ino)
        data = os.read(fd, (1 << 20) + 1)
        assert len(data) == opened.st_size
        assert (p.lstat().st_dev, p.lstat().st_ino) == (opened.st_dev, opened.st_ino)
    finally:
        os.close(fd)
    if expected is not None:
        assert set(expected) == {'path', 'bytes', 'sha256'} and str(p) == expected['path']
        assert type(expected['bytes']) is int and len(data) == expected['bytes']
        assert hashlib.sha256(data).hexdigest() == expected['sha256']
    return data

def _j_check_clock(c, unix, monotonic, boot, useful=0, final=0, active=True, cleanup=180):
    import math
    keys = ('started_at_unix','started_at_monotonic','final_deadline_unix',
            'final_deadline_monotonic','useful_deadline_unix','useful_deadline_monotonic')
    assert all(type(c[k]) in (int,float) and math.isfinite(c[k]) and c[k]>0 for k in keys), 'NOT_RUN_CLOCK_NUMBER'
    assert type(c['original_seconds']) is int and c['original_seconds']>cleanup, 'NOT_RUN_ORIGINAL_BUDGET'
    assert type(c['cleanup_seconds']) is int and c['cleanup_seconds']==cleanup, 'NOT_RUN_CLEANUP_RESERVE'
    for suffix in ('unix','monotonic'):
        assert c['final_deadline_'+suffix]==c['started_at_'+suffix]+c['original_seconds'], 'NOT_RUN_FINAL_EXTENSION'
        assert c['useful_deadline_'+suffix]==c['final_deadline_'+suffix]-cleanup, 'NOT_RUN_USEFUL_EXTENSION'
    if active:
        assert c['boot_id']==boot, 'NOT_RUN_WRONG_BOOT'
        assert 0 <= unix-c['started_at_unix'] and 0 <= monotonic-c['started_at_monotonic'], 'NOT_RUN_FUTURE_START'
        assert abs((unix-c['started_at_unix'])-(monotonic-c['started_at_monotonic']))<2, 'NOT_RUN_CLOCK_RELATION'
        assert min(c['useful_deadline_unix']-unix,c['useful_deadline_monotonic']-monotonic)>useful, 'NOT_RUN_USEFUL_EXHAUSTED'
        assert min(c['final_deadline_unix']-unix,c['final_deadline_monotonic']-monotonic)>final, 'NOT_RUN_FINAL_EXHAUSTED'
