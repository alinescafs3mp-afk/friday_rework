"""Genuine kernel reads for C83; inert until a separately admitted invocation.

Declared policy/role is kept beside, never substituted for, kernel observation.
No capabilities are opened, adopted, closed, repaired or released here.
"""
import hashlib
import os
import resource
import stat
from dataclasses import dataclass
from pathlib import Path

class ObservationMissing(ValueError):
    pass

def need(value, code):
    if not value:
        raise ObservationMissing(code)

def hard_caps():
    caps = {name: resource.getrlimit(limit) for name, limit in (
        ('AS',resource.RLIMIT_AS),('CPU',resource.RLIMIT_CPU),
        ('FS',resource.RLIMIT_FSIZE),('CORE',resource.RLIMIT_CORE))}
    need(caps=={'AS':(268435456,268435456),'CPU':(30,30),
                'FS':(67108864,67108864),'CORE':(0,0)},'OBSERVATION_HARD_CAPS')
    affinity=tuple(sorted(os.sched_getaffinity(0)))
    need(affinity==(0,1),'OBSERVATION_AFFINITY_0_1')
    return {'limits':caps,'affinity':affinity}

def same_read_source(pin):
    # Consistency only; this function never produces Source/action admission.
    need(type(pin)is dict and set(pin)>={'path','bytes','sha256'},'SOURCE_PIN_SHAPE')
    p=Path(pin['path'])
    a=p.lstat();need(stat.S_ISREG(a.st_mode) and a.st_nlink==1,'SOURCE_REGULAR_EXACT')
    b=p.read_bytes();z=p.lstat()
    need((a.st_dev,a.st_ino,a.st_size,a.st_mtime_ns)==
         (z.st_dev,z.st_ino,z.st_size,z.st_mtime_ns),'SOURCE_PIN_READ_RACE')
    need(len(b)==pin['bytes'] and hashlib.sha256(b).hexdigest()==pin['sha256'],
         'SOURCE_PIN_BYTES')
    return b

@dataclass(frozen=True)
class ProcessRead:
    pid: int
    birth: str
    ppid: int
    pgid: int
    sid: int
    state: str
    uids: tuple
    gids: tuple
    tracer: int
    exe: str
    cmdline: bytes
    stat_raw: str
    status_raw: str
    limits_raw: str
    def birth_dict(self):
        return {'pid':self.pid,'birth':self.birth,'ppid':self.ppid,
                'pgid':self.pgid,'sid':self.sid}

def process_read(pid, allow_terminal=False):
    need(type(pid)is int and pid>0,'ACTUAL_PROCESS_PID')
    p=Path('/proc')/str(pid)
    raw=(p/'stat').read_text();fields=raw.rsplit(')',1)[1].split()
    status=(p/'status').read_text()
    table={}
    for line in status.splitlines():
        if ':' in line:
            key,value=line.split(':',1);table[key]=value.strip().split()
    need(len(table.get('Uid',()))==4 and len(table.get('Gid',()))==4,
         'ACTUAL_PROCESS_UID_GID')
    state=fields[0]
    need(allow_terminal or state not in ('Z','X'),'ACTUAL_PROCESS_LIVE')
    exe=os.readlink(p/'exe')
    cmdline=(p/'cmdline').read_bytes()
    end=(p/'stat').read_text().rsplit(')',1)[1].split()
    need(fields[19]==end[19] and fields[1:4]==end[1:4],
         'ACTUAL_PROCESS_BIRTH_READ_RACE')
    return ProcessRead(pid,fields[19],int(fields[1]),int(fields[2]),int(fields[3]),
        state,tuple(map(int,table['Uid'])),tuple(map(int,table['Gid'])),
        int(table['TracerPid'][0]),exe,cmdline,raw,status,(p/'limits').read_text())

@dataclass(frozen=True)
class FDRead:
    pid: int
    birth: str
    fd: int
    identity: tuple
    uid: int
    gid: int
    mode: int
    nlink: int
    link: str
    info_raw: str
    mount_id: int
    flags: int
    locks: tuple
    def policy(self):
        return {'uid':self.uid,'gid':self.gid,'mode':self.mode}

def fd_read(process,fd):
    need(type(process)is ProcessRead and type(fd)is int and fd>2,'ACTUAL_FD_READ_SHAPE')
    p=Path('/proc')/str(process.pid)
    first=(p/'stat').read_text().rsplit(')',1)[1].split()
    need(first[19]==process.birth,'ACTUAL_FD_BEFORE_BIRTH')
    target=p/'fd'/str(fd);q=target.stat();link=os.readlink(target)
    info=(p/'fdinfo'/str(fd)).read_text();rows={}
    locks=[]
    for line in info.splitlines():
        key,_,value=line.partition(':')
        if key=='lock':locks.append(value.strip())
        else:rows[key]=value.strip()
    need('mnt_id' in rows and 'flags' in rows,'ACTUAL_FDINFO_COMPLETE')
    end=target.stat();last=(p/'stat').read_text().rsplit(')',1)[1].split()
    need(last[19]==process.birth and
        (q.st_dev,q.st_ino,q.st_mode,q.st_uid,q.st_gid)==
        (end.st_dev,end.st_ino,end.st_mode,end.st_uid,end.st_gid),
        'ACTUAL_FD_IDENTITY_READ_RACE')
    return FDRead(process.pid,process.birth,fd,(q.st_dev,q.st_ino,stat.S_IFMT(q.st_mode)),
        q.st_uid,q.st_gid,stat.S_IMODE(q.st_mode),q.st_nlink,link,info,
        int(rows['mnt_id']),int(rows['flags'],8),tuple(locks))

def all_remote_fds(process):
    # Remote scan does not insert its own dirfd into the subject's table.
    # The observer must be a DIFFERENT already-existing process.
    need(process.pid!=os.getpid(),'INDEPENDENT_REMOTE_OBSERVER_REQUIRED')
    first=process_read(process.pid)
    need(first.birth==process.birth,'REMOTE_SCAN_ACTUAL_BIRTH')
    names=tuple(sorted(int(x.name) for x in (Path('/proc')/str(process.pid)/'fd').iterdir()
                       if x.name.isdecimal() and int(x.name)>2))
    observed=tuple(fd_read(process,fd) for fd in names)
    last=tuple(sorted(int(x.name) for x in (Path('/proc')/str(process.pid)/'fd').iterdir()
                      if x.name.isdecimal() and int(x.name)>2))
    need(names==last and process_read(process.pid).birth==process.birth,
         'REMOTE_SCAN_COMPLETE_STABLE')
    return observed

def directory_read(process,cap):
    need(type(cap)is dict and set(cap)=={'fd','path','identity','policy'},
         'DIRECTORY_DECLARATION_SHAPE')
    f=fd_read(process,cap['fd']);p=Path(cap['path'])
    # Return malformed declarations as declarations, with actual FD facts.
    # No expected metadata is copied into the actual-policy fields.
    ancestors=[]
    for parent in reversed(p.parents):
        q=parent.lstat()
        ancestors.append((str(parent),q.st_dev,q.st_ino,q.st_mode,q.st_uid,q.st_gid))
    q=p.lstat()
    return {'declared':cap,'fd_actual':f,'path_actual':(
        str(p),q.st_dev,q.st_ino,q.st_mode,q.st_uid,q.st_gid),
        'ancestors_actual':tuple(ancestors)}

def ancestry_read(caller_pid,gnu_pid,root_pid,gnu_pin):
    caller,gnu,root=map(process_read,(caller_pid,gnu_pid,root_pid))
    same_read_source(gnu_pin)
    actual=Path('/proc')/str(gnu_pid)/'exe'
    raw=actual.read_bytes()
    need(len(raw)==gnu_pin['bytes'] and hashlib.sha256(raw).hexdigest()==gnu_pin['sha256'],
         'ACTUAL_GNU_EXECUTABLE_BYTES')
    need(root.ppid==gnu.pid and gnu.ppid==caller.pid and
         caller.pid not in (gnu.pid,root.pid) and caller.pgid!=gnu.pgid,
         'ACTUAL_ORIGINAL_CALLER_GNU_ROOT_ANCESTRY')
    need(caller.uids==gnu.uids==root.uids==(os.getuid(),)*4,
         'ACTUAL_ORIGINAL_CALLER_GNU_ROOT_UIDS')
    return caller,gnu,root

def mount_rows(pid):
    raw=(Path('/proc')/str(pid)/'mountinfo').read_text()
    rows=[]
    for line in raw.splitlines():
        row=line.split();separator=row.index('-')
        rows.append({'id':int(row[0]),'parent':int(row[1]),'root':row[3],
            'path':row[4],'options':row[5],'fs':row[separator+1],
            'source':row[separator+2],'super_options':row[separator+3],
            'raw':line})
    return tuple(rows)

def original_slot_read(caller,slot_fd,slot_identity,lock_owner_pid):
    row=fd_read(caller,slot_fd)
    need(row.identity[:2]==tuple(slot_identity) and row.identity[2]==stat.S_IFREG
         and row.nlink==1,'ORIGINAL_SLOT_PHYSICAL_IDENTITY')
    need(any('FLOCK' in lock and 'WRITE' in lock and
             str(lock_owner_pid) in lock.split() for lock in row.locks),
         'ORIGINAL_SLOT_ACTUAL_UNINTERRUPTED_FLOCK')
    return row

def phase_inventory_read(subject,phase_caps,allow_known_closed=False):
    need(type(phase_caps)is dict and len(phase_caps)==23,'ALL23_DECLARATIONS_REQUIRED')
    live=all_remote_fds(subject);byfd={r.fd:r for r in live};mounts=mount_rows(subject.pid)
    phases=[]
    for phase,cap in phase_caps.items():
        parents=(cap['group_parent_capability'],
                 *cap['private_directory_capabilities'].values(),
                 *cap.get('auxiliary_parent_capabilities',()))
        local_fds={cap['group_fd'],cap['slot_fd'],cap['control_dir_fd'],cap['trace_dir_fd']}
        local_fds.update(row[0] for row in cap['output_owned_fds'])
        local_fds.update(row['fd'] for row in parents)
        actual=cap.get('actual_control_binding',{})
        for key in ('auxiliary_groups','auxiliary_outputs','auxiliary_fds'):
            for row in actual.get(key,()):
                if key=='auxiliary_outputs':local_fds.update(r[0] for r in row['output_owned_fds'])
                else:local_fds.add(row['fd'])
        missing=tuple(sorted(local_fds-set(byfd)))
        need(allow_known_closed or not missing,'PHASE_ORIGINAL_LIVE_FDS_MISSING')
        declared_mount=cap['output_mount_id']
        actual_mount_fds=tuple(r for r in live if r.mount_id==declared_mount)
        phases.append({'phase':phase,'declared':cap,'actual_fds':tuple(byfd[i] for i in sorted(local_fds) if i in byfd),
            'missing_declared_fds':missing,
            'actual_main_mount_fd_inventory':actual_mount_fds,
            'actual_main_mounts':tuple(r for r in mounts if r['id']==declared_mount),
            'parent_reads':tuple(directory_read(subject,row) for row in parents if row['fd'] in byfd)})
    return {'subject':subject,'all_live_fds':live,'all_mounts':mounts,'phases':tuple(phases)}

def untouched(before,after):
    need(type(before)is tuple and type(after)is tuple,'ACTUAL_PRESERVATION_INPUT')
    def capability(row):
        return (row.pid,row.birth,row.fd,row.identity,row.uid,row.gid,row.mode,
                row.nlink,row.link,row.mount_id,row.flags,row.locks)
    need({r.fd:capability(r) for r in before}=={r.fd:capability(r) for r in after},
         'ORIGINAL_CAPABILITY_OR_FLOCK_DRIFT')
    return True
