def _bounds(run,name,pid=None):
    return tuple(r for r in run.boundaries if r.name==name and (pid is None or r.pid==pid))
def _marks(run,name,pid=None):
    return tuple(r for r in run.markers if r.event==name and (pid is None or r.pid==pid))
def _code(run):
    observed=_marks(run,'refusal')
    physical.need(observed,'GENUINE_CAUGHT_REFUSAL_MISSING')
    codes=[]
    for event in observed:
        number=event.values[0]
        physical.need(0<number<=len(markers.REFUSALS),'UNKNOWN_REFUSAL_NOT_INTENDED')
        codes.append(markers.REFUSALS[number-1])
    physical.need(len(set(codes))==1,'BOUNDARIES_REFUSED_DIFFERENT_CAUSES')
    return codes[0]
def _proof(run,name,predicate,selectors=()):
    physical.need(predicate,'INDEPENDENT_PHYSICAL_PROOF_MISSING:'+name)
    return dict(run.raw_ref,proof=name,selectors=list(selectors),
                observer_pid=run.observer_pid,observer_birth=run.observer_birth)

def _interval(run,pid,late=False):
    if late:
        begin=_marks(run,'auxiliary_consumer',pid)
    else:
        begin=_marks(run,'attempt_begin',pid)
    end=_marks(run,'attempt_end',pid)
    physical.need(len(begin)==1 and len(end)==1 and begin[0].kernel_sequence<end[0].kernel_sequence,
                  'EXACT_RAW_INTERVAL_UNBOUND_OR_RESET')
    return begin[0].kernel_sequence,end[0].kernel_sequence
def _scope_events(run,late=False):
    result=[]
    for pid in run.registration['required_attempt_pids']:
        lo,hi=_interval(run,pid,late)
        result.extend(r for r in run.kernel if r.pid==pid and lo<r.sequence<hi)
    return tuple(sorted(result,key=lambda r:r.sequence))
def _count_marks(run,name,late=False):
    count=0
    for pid in run.registration['required_attempt_pids']:
        lo,hi=_interval(run,pid,late)
        count+=sum(r.event==name and r.pid==pid and lo<r.kernel_sequence<hi for r in run.markers)
    return count
def _syscall_entries(run,number,late=False):
    return tuple(r for r in _scope_events(run,late) if r.stage=='entry' and r.syscall==number)
def _real_close_calls(run,late=False):
    # Count all original/future/batch capabilities and source-marked closes.
    # Observational source/proc reads are retained in raw transcript, and their
    # successful O_RDONLY acquisitions are checked. Nothing unknown is adopted.
    entries=_syscall_entries(run,3,late)
    actual=[]
    for row in entries:
        before=_bounds(run,'attempt_begin',row.pid)
        physical.need(len(before)==1,'COMPLETE_ORIGINAL_FD_SNAPSHOT_MISSING')
        original={r.fd for r in before[0].all_fd_reads}
        fd=row.arguments[0]
        if fd in original:actual.append(row)
        else:
            # Find the exact preceding real successful open of this fd and
            # classify it from source-qualified allowed read path and flags.
            opens=[e for e in run.kernel if e.pid==row.pid and e.stage=='exit'
                   and e.syscall in (2,257) and e.result==fd and e.sequence<row.sequence]
            physical.need(opens,'UNKNOWN_CLOSED_FD_STOP_NO_ZERO')
            opened=opens[-1];flags=opened.arguments[1] if opened.syscall==2 else opened.arguments[2]
            permitted={p['path'] for p in run.registration['complete_current_source_pins']}
            path=opened.path
            allowed=path in permitted or (path and path.startswith('/proc/'))
            allowed=allowed or any(path and path.startswith(g['path']+'/') and
                os.path.basename(path) in ('memory.max','memory.swap.max','pids.max','cpu.max',
                'cgroup.events','cgroup.procs','pids.current') for g in run.registration['admitted_groups'])
            physical.need(allowed and flags&os.O_ACCMODE==os.O_RDONLY and
                not flags&(os.O_CREAT|os.O_TRUNC),'UNKNOWN_OR_NONOBSERVATIONAL_CLOSE_NO_ZERO')
    return tuple(actual)

def prefix_counts(run,late=False):
    kernel=_scope_events(run,late)
    caller=run.case.caller_pid;root=run.case.root_pid
    def bind(pid):
        lo,hi=_interval(run,pid,late)
        return sum(m.event=='bind_existing' and m.pid==pid and
                   lo<m.kernel_sequence<hi for m in run.markers)
    pids=run.registration['required_attempt_pids']
    return {'caller_role_bind':bind(caller) if caller in pids else 0,
        'Root_role_bind':bind(root) if root in pids else 0,
        'derived_validated_record':_count_marks(run,'record',late),
        'close':len(_real_close_calls(run,late)),
        'rmdir':sum(e.stage=='entry' and
            (e.syscall==84 or e.syscall==263 and e.arguments[2]&0x200) for e in kernel),
        'umount':sum(e.stage=='entry' and e.syscall==166 for e in kernel),
        'first_birth':sum(e.stage=='entry' and e.syscall in (56,57,58,435) for e in kernel)}

def _preserved(run):
    # Every original known FD is compared physically at the two stopped
    # boundaries, including all original caller FLOCK data. No absence guessed.
    checks=[]
    for pid in run.registration['required_attempt_pids']:
        before=_bounds(run,'attempt_begin',pid);after=_bounds(run,'attempt_end',pid)
        physical.need(len(before)==len(after)==1,'BEFORE_AFTER_ACTUAL_CAPABILITIES_UNBOUND')
        physical.untouched(before[0].all_fd_reads,after[0].all_fd_reads)
        physical.untouched(before[0].caller_fd_reads,after[0].caller_fd_reads)
        physical.need(before[0].original_slot==after[0].original_slot,
                      'ORIGINAL_CALLER_SLOT_FLOCK_DRIFT')
        physical.need(before[0].protected_reads==after[0].protected_reads,
                      'PROTECTED_WRITER_CONTROLLER_CHANGE')
        checks.extend((before[0].marker_sequence,after[0].marker_sequence))
    return _proof(run,'every_original_capability_and_slot_preserved',True,checks)

def _actual_caps(run):
    # Genuine subject limits, not observer limits or declared expected caps.
    names={'Max address space':'AS','Max cpu time':'CPU','Max file size':'FS','Max core file size':'CORE'}
    root=_bounds(run,'attempt_begin',run.case.root_pid)
    physical.need(len(root)==1,'ACTUAL_ROOT_LIMITS_BOUNDARY_UNBOUND')
    rows=root[0].process.limits_raw.splitlines()
    result={}
    for prefix,name in names.items():
        found=[r[len(prefix):].split() for r in rows if r.startswith(prefix)]
        physical.need(len(found)==1 and len(found[0])>=3,'ACTUAL_ROOT_LIMITS_UNKNOWN')
        physical.need(found[0][0].isdecimal() and found[0][1].isdecimal()
                      and found[0][0]==found[0][1],'ACTUAL_ROOT_LIMITS_NOT_HARD_FINITE')
        result[name]=int(found[0][0])
    return result

def _good_preflight(run):
    good=_bounds(run,'whole_preflight_complete')
    binds=_marks(run,'bind_existing')
    physical.need(good and binds,'GENUINE_PREFLIGHT_AND_BIND_OBSERVATION_UNBOUND')
    for pid in run.registration['required_attempt_pids']:
        local_good=[g for g in good if g.pid==pid]
        local_bind=[b for b in binds if b.pid==pid]
        physical.need(local_good and local_bind and
            min(g.marker_sequence for g in local_good)<min(b.sequence for b in local_bind),
            'WHOLE_PREFLIGHT_AFTER_LOCAL_FIRST_BIND')
    physical.need(all(len(b.phase_inventory['phases'])==23 for b in good),
                  'GENUINE_ALL23_PREFLIGHT_INVENTORY')
    return _proof(run,'all23_preflight_before_first_bind',True,
                  tuple(b.marker_sequence for b in good))

def _full_scope_proof(run):
    # Do not promote an observer binding, scope declaration, single positive or
    # copied historical result into actual current full32/current-gate execution.
    full=run.registration.get('current_whole_mandatory_observation')
    physical.need(type(full)is CurrentMandatoryObservation,
                  'CURRENT_FULL32_C75_C55_H991_FD8_OBSERVER_REDUCERS_UNBOUND')
    full.verify_same_final_bytes_and_original_scope(run)
    return full.raw_ref

class CurrentMandatoryObservation:
    """External genuine current scope result; this package cannot self-mint it.

    Requires all exact current control reducers and independently captured RawRun
    objects. C83 Source preparation does not provide old/full-gate execution.
    """
    def __init__(self,qualified_reducer_module,actual_runs,raw_ref,source_pins,clock_identity):
        self.reducer=qualified_reducer_module;self.runs=actual_runs
        self.raw_ref=raw_ref;self.source_pins=source_pins;self.clock_identity=clock_identity
    def verify_same_final_bytes_and_original_scope(self,run):
        registration=run.registration
        pin=registration['external_current_mandatory_reducer_source_pin']
        physical.same_read_source(pin)
        method=self.reducer.verify_current_all_mandatory_actual_observations
        physical.need(method.__code__.co_filename==pin['path'],'GENUINE_FULL_SCOPE_REDUCER_SOURCE_OBJECT')
        physical.need(self.source_pins==registration['complete_current_source_pins'] and
            self.clock_identity==registration['original_clock_identity'],
            'NO_OLD_CURRENT_SCOPE_TRANSFER')
        # All raw objects are checked before the admitted independent reducer.
        for observed in self.runs.values():capture.validate_raw_run(observed)
        method(self.runs,registration)
        physical.need(self.raw_ref is not None,'FULL_SCOPE_RAW_REF_UNBOUND')

    def _parent_once(self,event):
        run=self.run;pid=run.case.caller_pid
        parents={c['fd'] for cap in run.case.caller_caps.values()
                 for c in (cap['group_parent_capability'],
                    *cap['private_directory_capabilities'].values(),
                    *cap.get('auxiliary_parent_capabilities',()))}
        if event=='bind_existing':
            counts={fd:sum(m.values[0]==fd for m in _marks(run,event,pid)) for fd in parents}
        else:
            counts={fd:sum(e.pid==pid and e.arguments[0]==fd for e in _syscall_entries(run,3)) for fd in parents}
        physical.need(counts and len(set(counts.values()))==1,'PARENT_ONCE_COUNTS_NOT_UNIFORM')
        return next(iter(counts.values()))
    def _extra_inventory(self):
        run=self.run;extras=run.case.declared_delta['extra_parents']
        good=_bounds(run,'whole_preflight_complete')
        physical.need(good,'EXTRA_PARENT_GENUINE_PREFLIGHT_MISSING')
        for b in good:
            byfd={r.fd:r for r in b.all_fd_reads}
            for cap in extras:
                row=byfd[cap['fd']]
                physical.need(row.identity[:2]==tuple(cap['identity']) and
                    row.policy()==cap['policy'] and row.link==cap['path'],'EXTRA_PARENT_ACTUAL_DRIFT')
        return _proof(run,'all_extra_original_parent_kernel_reads',True)
    def _early_extra_closes(self):
        run=self.run;extras={r['fd'] for r in run.case.declared_delta['extra_parents']}
        terminal=_marks(run,'original_caller_terminal',run.case.caller_pid)
        physical.need(len(terminal)==1,'ACTUAL_CALLER_TERMINAL_MISSING')
        return sum(e.pid==run.case.caller_pid and e.stage=='entry' and e.syscall==3
                   and e.arguments[0] in extras and e.sequence<terminal[0].kernel_sequence for e in run.kernel)
    def _distinct_parent_proof(self):
        run=self.run;a,b=run.case.resources['distinct_parent_pair']
        physical.need(a!=b,'DISTINCT_PARENT_FDS_REQUIRED')
        before=_bounds(run,'attempt_begin',run.case.caller_pid)
        rows={r.fd:r for r in before[0].all_fd_reads}
        physical.need(rows[a].identity==rows[b].identity and rows[a].policy()==rows[b].policy()
                      and rows[a].link==rows[b].link,'DISTINCT_PARENT_ACTUAL_IDENTITY_POLICY_PATH')
        closes=[e for e in _syscall_entries(run,3) if e.pid==run.case.caller_pid]
        physical.need(sum(e.arguments[0]==a for e in closes)==
                      sum(e.arguments[0]==b for e in closes)==1,'BOTH_ORIGINAL_PARENTS_ONCE_CLOSE_UNPROVEN')
        return _proof(run,'both_distinct_original_parent_capabilities_closed_once',True)
    def _outer_holder(self):

    def _wrong_gnu(self):

    def _main_policy(self):

            row.link==d['good']['path'] and row.policy()==d['good']['policy']
            and row.policy()!=d['last_bad']['policy'],'ACTUAL_GOOD_PARENT_BAD_DECLARED_POLICY')

    def _omitted_inventory(self):

                      'OMITTED_FD_NOT_REAL_ORIGINAL_MAIN_MOUNT')
        supplied={r[0] for r in run.case.root_caps[run.case.last_phase]['output_owned_fds']}
        physical.need({r.fd for r in actual}-supplied=={missing[0]},'EXACT_ONE_REAL_MOUNT_FD_OMISSION')
        return _proof(run,'all_actual_last_mount_FDs_and_exact_declared_omission',True)
    def _late_stop(self):

    """Executable connection to old C75_25/C55_41/original32 consumers.

                  'LEGACY_FIXTURE_DICTIONARY_CANNOT_GRANT_OBSERVATION')
