"""C50 Source definitions for INDEPENDENT Root actual-evidence review.
NOT RUN. No test constructs a physical provider, calls a launcher, or promotes
JSON/Source checks to admission. Each actual-trace ref must be independently
pinned and produced under NEW final Source/action acceptance; unrelated failures
never pass. Original32/all63/kernel120/full19/normal3600 gates are unchanged.
"""


def _actual(t,case):
    assert t['control_id']==case and t['actual_independent_Root_trace_ref'] is not None
    assert t['independent_exact_final_Source_acceptance_ref'] is not None
    assert t['actual_specific_action_admission_ref'] is not None
    assert t['simulated'] is False and t['unknown_as_zero'] is False
    assert t['native_install_or_receipt_or_RPC_calls']==0
    assert t['protected_writer_changes']==0 and t['no_replay'] is True


def _refused(t,case,cause):
    _actual(t,case)
    assert t['state']=='REFUSE' and t['intended_cause']==cause
    assert t['actual_refusal_site_ref'] is not None and t['unrelated_error_as_pass'] is False
    assert t['capacity_release'] is False and t['opaque_GO_calls']==0
    assert t['actual_firstbirth_group_pids_current']==0

    assert t['actual_timer_arm_m']<=t['actual_opaque_GO_m']<t['actual_firstbirth_attempt_m']
    assert t['stock_pidfd_fstat_before_GO_ref'] is not None and t['actor_pidfd_fstat_before_GO_ref'] is not None
    assert t['actual_group0_immediately_before_GO_ref'] is not None
    assert t['old_source_or_bool_as_wait_proof'] is False
