def causal_edit(case_id,sources):
    assert case_id in EDITS
    name,old,new=EDITS[case_id];result=dict(sources);text=result[name].decode()
    # Two close sites exist (Source and native), so constrain to native method
    # interval and never modify immutable Source-domain custody behavior.
    if case_id in ('PIDFD_CLOSE_AMBIGUOUS','CAPTURE_SEALED_EXTENSION_CLOSE_EIO','CALLER_NORMAL_RESOURCE_FD_CLOSE_EIO'):
        start=text.index('    def native_close_fd(');end=text.index('    def native_children_closed(',start)
        region=text[start:end];assert region.count(old)==1
        text=text[:start]+region.replace(old,new,1)+text[end:]
    else:
        assert text.count(old)==1,'changed final causal anchor: Root new review required'
        text=text.replace(old,new,1)
    result[name]=text.encode()
    return result
