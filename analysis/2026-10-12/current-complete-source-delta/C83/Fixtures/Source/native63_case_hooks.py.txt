NINE=('base64','binascii','http','jsonschema','secrets','selectors','shutil','tarfile','urllib')


def missing_nine(descriptor):
    # Nine distinct exact pin/review/admission cases; not one base64 predicate.
    result=[]
    for name in NINE:
        value=copy.deepcopy(descriptor)
        assert name in value['native_capture']['modules']
        del value['native_capture']['modules'][name]
        result.append({'missing':name,'descriptor':value})
    return result


def missing_parent(descriptor,parent_name):
    catalog=descriptor['native_capture'];modules=catalog['modules']
    assert parent_name not in NINE and parent_name in modules
    assert any(name.startswith(parent_name+'.') for name in modules)
    value=copy.deepcopy(descriptor)
    del value['native_capture']['modules'][parent_name]
    assert all(name in value['native_capture']['modules'] for name in NINE)
    return value


def missing_resource_terms(spec):
    values=[];limits=spec['native']['resource_limits']
    assert len(limits)==20
    for name in limits:
        value=copy.deepcopy(spec);del value['native']['resource_limits'][name]
        values.append({'missing':name,'spec':value})
    return values


def input_change(spec,descriptor,case_id,fixture):
    # Called BEFORE Root seals/reviews/adopts the distinct case input. No
    # controller/clock/receipt/config/shared-file mutation is performed here.
    s,d=copy.deepcopy(spec),copy.deepcopy(descriptor)
    n=s['native']
    if case_id=='MISSING_ROLE_LIMIT':return missing_resource_terms(s)
    if case_id=='CAPTURE_NINE_ROOTS_MISSING':return missing_nine(d)
    if case_id=='CAPTURE_PARENT_NAMESPACE_MISSING':
        d=missing_parent(d,fixture['nonmandatory_parent_name'])
    elif case_id=='CAPTURE_SAME_READ_DIGEST_DRIFT':
        module=fixture['disposable_capture_module'];d['native_capture']['modules'][module]['pin']['sha256']='0'*64
    elif case_id=='JSON_SOURCE21_AS_AUTHORITY':n['request']['native_admission']='Source21'
    elif case_id=='SHARED_TERMINAL_SCOPE':n['group']=fixture['actual_shared_group_path']
    elif case_id=='UPDATE_LOCK_MISSING':n['current_pins'].append(fixture['actual_update_lock_negative_pin'])
    elif case_id=='CANONICAL_CONSUMER_DRIFT':
        name=fixture['one_actual_consumer_name'];n['installed_targets'][name]['sha256']='0'*64
    elif case_id=='THREE_TARGET_INODE_MODE_DRIFT':
        n['installed_targets'][fixture['one_actual_consumer_name']]['sha256']='0'*64
    elif case_id=='FULL_TREE_OR_VERSION_DRIFT':n['version']=fixture['actual_negative_version']
    elif case_id in ('OLD_OR_RENEWED_CLOCK','USEFUL_RESERVE_SHORT','FINAL_RESERVE_SHORT'):
        # A previously pinned actual expired/other-actor clock is a NEGATIVE
        # input, never a reset of the admitted actual original Root clock.
        s['clock']=fixture['negative_actual_clock_pin']
    elif case_id=='SOURCE_AUDIT_REMOVAL':s['helper']=fixture['negative_audit_pin']
    elif case_id=='CAPSULE_SEALS_OR_BYTES_DRIFT':
        n['fixture']['capsule_expected_digest']='0'*64
    return {'spec':s,'descriptor':d,'no_effects_or_authority':True}


def _private_mutation(owner,fixture):
    # Root creates a DISPOSABLE original-byte-exact private copy before pinning.
    # Never edit C39, installed modules/resources, account files or shared data.
    path=Path(fixture['disposable_capture_path'])
    assert path.is_relative_to(owner.temp_root) and path.resolve(strict=True).is_relative_to(owner.temp_root)
    before=path.lstat();assert stat.S_ISREG(before.st_mode) and before.st_nlink==1
    assert before.st_mode&0o022==0 and before.st_mode&0o200
    original=owner.capture_raw[str(path)]
    assert len(original)==before.st_size and 0<len(original)<=16777216
    raw=bytes([original[0]^1])+original[1:]
    label='c40_disposable_mutation'
    fd=owner.native_track(label,os.open(path,os.O_WRONLY|os.O_NOFOLLOW))
    pos=0
    while pos<len(raw):n=os.write(fd,raw[pos:]);assert n>0;pos+=n
    os.fsync(fd);owner.native_close_fd(label)
    after=path.lstat()
    assert before.st_mtime_ns!=after.st_mtime_ns or before.st_ctime_ns!=after.st_ctime_ns
    assert owner.capture_raw[str(path)]==original
    return path,original,raw


def same_read_mutation(owner,path,fd,Root_bound_disposable_root):
    # Injection hook immediately AFTER an ACTUAL first capture os.read and
    # BEFORE the production pre/post fstat/digest comparison. Actual private
    # path changes, not a false condition standing in for an inode change.
    path=Path(path)
    private_root=Path(Root_bound_disposable_root)
    assert private_root.is_absolute() and path.is_relative_to(private_root) and path.resolve(strict=True).is_relative_to(private_root)
    s=os.fstat(fd);assert stat.S_ISREG(s.st_mode) and s.st_nlink==1 and s.st_size>0
    writer=owner.native_track('c40_same_read_writer',os.open(path,os.O_WRONLY|os.O_NOFOLLOW))
    old=os.pread(fd,1,0);assert len(old)==1
    assert os.pwrite(writer,bytes([old[0]^1]),0)==1
    os.fsync(writer);owner.native_close_fd('c40_same_read_writer')


def oversized_capsule():
    # Bounded expansion crosses the real272MiB ceiling by one byte. This is
    # futureRoot AS1GiB Sourcefixture, never author AS256 allocation/execution.
    raw=b'0'*285212673
    compressed=zlib.compress(raw,level=6)
    assert len(compressed)<=67108864
    return compressed


def dispatch(owner,fixture,observe):
    case_id=fixture['id'];owner.native_check()
    if case_id=='CAPTURE_UNKNOWN_IMPORT_NO_DISK_FALLBACK':
        observe.capture_unknown(owner);return
    if case_id=='CAPTURE_PACKAGE_RESOURCE_NO_REOPEN':
        path=Path(fixture['disposable_capture_path'])
        original=owner.capture_raw[str(path)]
        # Actual Root/host changes only its disposable private Source copy
        # AFTER capture, before GO. Caller Landlock writes remain temp-only.
        fd=owner.native_track('c40_resource_after_capture_read',os.open(path,os.O_RDONLY|os.O_NOFOLLOW))
        changed=os.pread(fd,len(original)+1,0);owner.native_close_fd('c40_resource_after_capture_read')
        assert len(changed)==len(original) and changed!=original
        observe.captured_resource(owner,str(path));return
    if case_id=='CAPTURE_CYCLIC_CANONICAL_NAMESPACES':
        # Root binds ACTUAL genuine modules and their canonical observed
        # relationships. No fixture mini-library replaces genuine dependencies.
        modules=fixture['genuine_cyclic_fromlist_dataclass_modules'];assert len(modules)>=3
        results=[]
        for name in modules:
            module=owner.native_module(name)
            assert module is sys.modules[name] and module.__spec__.name==name
            assert module.__spec__.loader is owner or name in owner.capture_extensions
            results.append({'name':name,'module_id':id(module),'spec_name':module.__spec__.name})
        owner._c40_add({'operation':'canonical_genuine_imports','phase':'actual','modules':results})
        return
    modes=fixture['real_workload_modes']
    for mode in modes:
        result=observe.real_workload(owner,fixture['python_path'],fixture['payload_path'],mode)
        # EOF positivity requires actual raw count and exact stdout/stderr
        # producer pins; CompletedProcess text decoding alone is insufficient.
        if mode=='positive_eof':
            assert owner.output_bytes-fixture['output_bytes_before']==524288
            assert result.returncode==0
    if modes:return
    # Remaining controls operate at actual existing host/stock or exact Source
    # fault sites. A caller-only invocation must not fabricate their outcomes.
    raise RuntimeError('case requires its separately Root-bound existing host/stock causal site: '+case_id)
