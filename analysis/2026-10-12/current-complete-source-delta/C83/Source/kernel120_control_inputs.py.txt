def bind_preflight_control(base,case_id,Root_actual_control_bindings):
    spec=copy.deepcopy(base);actual=Root_actual_control_bindings[case_id]
    if actual['original_clock_ref']!=base['clock']:
        raise ValueError('ACTUAL_CONTROL_COMMON_CLOCK')
    # These are pinned independently admitted fixture provenance, never grants.
    for key in ('fixture_source_acceptance_ref','fixture_specific_action_ref'):
        if actual.get(key) is None:raise ValueError('ACTUAL_CONTROL_FIXTURE_ADMISSION_UNBOUND')
    if case_id in ('K120_UNBOUNDED_MEMORY','K120_NONZERO_SWAP','K120_UNBOUNDED_TASKS',
                   'K120_UNBOUNDED_CPU','K120_SHARED_WRITER_SCOPE'):
        for key in ('group','group_relative','first_birth_group_identity','group_parent_capability','group_directory_policy'):spec['native'][key]=actual[key]
        spec['existing_owner']['group_fd']=actual['group_fd']
    elif case_id=='K120_FOREIGN_GROUP_FD':spec['existing_owner']['group_fd']=actual['foreign_group_fd']
    elif case_id in ('K120_FOREIGN_SLOT_FD','K120_UNHELD_SLOT'):
        spec['existing_owner']['slot_fd']=actual['alternate_slot_fd']
    elif case_id=='K120_RELATIVE1800_GUARD':spec['native']['stock_guard_seconds']=1800
    elif case_id=='K120_OUTPUT_NOT_BOUNDED':spec['record']=actual['private_unbounded_output_record']
    elif case_id=='K120_CLOCK_RENEWAL':spec['clock']=actual['different_immutable_clock_ref']
    elif case_id=='K120_NO_CURRENT_SOURCE_ACCEPTANCE':spec['existing_owner']['independent_source_acceptance']=None
    elif case_id=='K120_NO_ACTUAL_ROOT_ADMISSION':spec['existing_owner']['actual_Root_admission']=None
    elif case_id in ('K120_FINAL_RESERVE_SHORT','K120_USEFUL_RESERVE_SHORT',
                     'K120_ALREADY_ELAPSED_GUARDIAN','K120_ROLE_HARD_CAP_MISSING',
                     'K120_PIDFD_OR_NAMESPACE_UNSUPPORTED'):
        pass # Actual clock/role/kernel predicates, never monkeypatch/injected outcome.
    else:raise ValueError('FIXED_NEGATIVE_CASE_ONLY')
    return spec


def validate_existing_Root_before_entry(spec,Root_expected):
    if Root_expected.get('actual_source_safety_acceptance') is None \
       or Root_expected.get('actual_specific_action_admission') is None:
        raise ValueError('ACTUAL_ROOT_ADMISSION_UNBOUND')
    if spec['clock']!=Root_expected['original_clock_pin'] \
       or spec['owner_host']!=Root_expected['accepted_current_host_pin']:
        raise ValueError('ACTUAL_CURRENT_SOURCE_OR_CLOCK_BINDING')
    if spec['existing_owner']['group_fd']!=Root_expected['actual_private_group_fd'] \
       or spec['existing_owner']['slot_fd']!=Root_expected['actual_existing_slot_fd']:
        raise ValueError('ACTUAL_RETAINED_ROOT_CAPABILITIES')
    return spec


def dispatch_negative(owner,adapter,spec,clean_spec,expected,clock,catalog,case_id):
    """Only exact typed intended cause/site can produce REFUSED_INTENDED.
    This is candidate output; original32 acceptance still uses independent raw
    Root traces. A different refusal is a contained FAIL, never a PASS.
    """
    code,site=NEGATIVE_CONTRACTS[case_id];custody=expected['retained_fd_custody']
    actual=expected['actual_control_bindings'][case_id]
    custody['private_directory_capabilities']=expected['private_directory_capabilities']
    owner.require(set(NEGATIVE_CONTRACTS)==set(owner.root_fd_custody.AUXILIARY_PHASE_CONSUMERS),
                  'exact original18 actual auxiliary consumers')
    if case_id=='K120_PIDFD_OR_NAMESPACE_UNSUPPORTED':
        # These independent private auxiliaries do not feed clone3/namespace.
        # Consume them FIRST on both syscall-refusal and successful contained
        # branches; snapshot follows their exact once-only normal closure.
        owner.kernel120_cleanup_auxiliary_control(actual,expected,clock)
    before=owner.kernel120_no_birth_snapshot(clean_spec,expected,custody)
    if case_id=='K120_PIDFD_OR_NAMESPACE_UNSUPPORTED':
        # Real syscall path. ONLY typed exact no-child clone failure is caught.
        before_rows=len(custody['fds']);before_births=len(custody.get('birth_attempts',[]))
        try:result=adapter.kernel120_named_root_case_adapter(owner,spec,expected,clock,catalog,case_id)
        except owner.Kernel120Clone3Refusal as error:
            owner.require(error.site==site and error.syscall_result<0 and error.errno in (1,22,38,95)
                          and len(custody.get('birth_attempts',[]))==before_births,
                          'exact intended actual no-child clone3 failure only')
            for row in custody['fds'][before_rows:]:
                if not row['closed']:owner.root_fd_custody.closed(custody,row['fd'])
            after=owner.kernel120_no_birth_snapshot(clean_spec,expected,custody)
            owner.require(before['actual_fd_identities']==after['actual_fd_identities'],
                          'actual failed-clone acquired handles closed once')
            raw=owner.kernel120_record(custody,expected['trace_dir_fd'],'kernel-refusal.json',
                {'case_id':case_id,'site':site,'actual_clone3_syscall_result':error.syscall_result,
                 'actual_kernel_errno':error.errno,'raw_clone3':custody['actual_clone3_refusals'][-1],
                 'actual_before':before,'actual_after':after,'fallback_or_retry_count':0,
                 'unrelated_error_as_pass':False,'original_clock_ref':catalog.clock_pin})
            result=owner.kernel120_no_birth_cleanup(clean_spec,expected,custody)
            result.update(control_state='REFUSED_INTENDED',actual_refusal_trace_ref=raw,
                          actual_clone3_syscall_result=error.syscall_result,actual_kernel_errno=error.errno)
        else:
            # A successful kernel capability is an honest contained negative
            # FAIL unless independent raw evidence establishes a later exact
            # unsupported namespace syscall; NEVER a broad exception PASS.
            result.update(control_state='FAIL_CONTAINED_REQUIRES_RAW_KERNEL_REVIEW',
                          required_raw_file='kernel120-clone3-refusal.json')
        result.update(expected_refusal={'code':code,'site':site},unrelated_error_as_pass=False,
                      candidate_output_is_not_negative_acceptance=True)
        return result
    error=None
    try:
        if case_id=='K120_ROLE_HARD_CAP_MISSING':
            role_context=dict(expected,pre_entry_role_birth=actual['actual_role_birth'])
            owner.kernel120_first_birth_preflight(spec,role_context,clock)
        elif case_id=='K120_ALREADY_ELAPSED_GUARDIAN':
            guard_context=dict(expected,before_entry_guardian=actual['actual_guardian'])
            owner.kernel120_first_birth_preflight(spec,guard_context,clock)
        else:owner.kernel120_first_birth_preflight(spec,expected,clock)
    except owner.Kernel120Refusal as caught:error=caught
    after=owner.kernel120_no_birth_snapshot(clean_spec,expected,custody)
    owner.require(before['actual_fd_identities']==after['actual_fd_identities']
                  and all(before['actual_kernel'][key]==after['actual_kernel'][key]
                          for key in ('group_identity','first_raw','last_raw','owned_task_rows','actual_kernel_zero')),
                  'actual no-firstbirth FD/kernel observations unchanged')
    intended=error is not None and error.code==code and error.site==site
    state='REFUSED_INTENDED' if intended else 'FAIL_CONTAINED'
    refusal=owner.kernel120_record(custody,expected['trace_dir_fd'],'negative-refusal.json',
        {'case_id':case_id,'state':state,'expected_code':code,'expected_site':site,
         'actual_code':None if error is None else error.code,
         'actual_site':None if error is None else error.site,
         'actual_no_birth_before':before,'actual_no_birth_after':after,
         'fallback_or_retry_count':0,'unrelated_error_as_pass':False,
         'actual_first_birth_call_count':0,'original_clock_ref':expected['original_clock_pin'],
         'candidate_output_is_not_negative_acceptance':True})
    owner.kernel120_cleanup_auxiliary_control(actual,expected,clock)
    result=owner.kernel120_no_birth_cleanup(clean_spec,expected,custody)
    result.update({'control_state':state,'actual_refusal_trace_ref':refusal,'actual_first_birth_call_count':0,
                   'expected_refusal':{'code':code,'site':site},'unrelated_error_as_pass':False})
    return result
