import root_fd_custody
"""Source-only counterpart for the EXISTING finite outside Root context.
No launcher, timer/broker/service, subreaper setter, FD transfer protocol or
queue. Missing supported before-firstbirth facility is a concrete refusal.
Data/provenance is consistency; only independent Root can admit real effects.
"""
import ctypes
import hashlib
import math
import os
import resource
import stat
import time

AS=268435456
FS=67108864

class RootFacilityRefusal(ValueError):
    def __init__(self,code,detail):
        self.code=code
        super().__init__(code+': '+detail)


def need(ok,code,detail):
    if not ok:raise RootFacilityRefusal(code,detail)


def finite(value):
    return type(value) in (int,float) and math.isfinite(value)


def existing_root_caps():
    need(resource.getrlimit(resource.RLIMIT_AS)==(AS,AS)
         and resource.getrlimit(resource.RLIMIT_CPU)==(30,30)
         and resource.getrlimit(resource.RLIMIT_FSIZE)==(FS,FS)
         and resource.getrlimit(resource.RLIMIT_CORE)==(0,0)
         and 0<len(os.sched_getaffinity(0))<=2,'ROOT_HARD_CAPS','before Source/clock/kernel reads')


def original_root_clock(clock,boot_id,now_m,now_u):
    need(type(clock)is dict and clock.get('boot_id')==boot_id,'ORIGINAL_CLOCK','same kernel boot')
    fields=('started_at_monotonic','started_at_unix','useful_deadline_unix','final_deadline_unix')
    need(all(finite(clock.get(k)) for k in fields) and finite(now_m) and finite(now_u),
         'ORIGINAL_CLOCK','exact finite dualclock fields; bool is not a number')
    need(clock.get('original_seconds')==120 and type(clock['original_seconds'])is int
         and clock.get('cleanup_seconds')==30 and type(clock['cleanup_seconds'])is int,
         'ORIGINAL_CLOCK','distinct proposed120including30, never Source1800/native1800')
    start_m,start_u=clock['started_at_monotonic'],clock['started_at_unix']
    need(clock['final_deadline_unix']==start_u+120 and clock['useful_deadline_unix']==start_u+90
         and 0<=now_m-start_m<120 and 0<=now_u-start_u<120
         and abs((now_m-start_m)-(now_u-start_u))<=2,
         'ORIGINAL_CLOCK','fixed original start/useful/final and actual elapsed; no renewal')
    return {'start_m':start_m,'start_u':start_u,'useful_m':start_m+90,'useful_u':start_u+90,
            'final_m':start_m+120,'final_u':start_u+120,
            'actor_stop_m':start_m+100,'actor_stop_u':start_u+100,
            'stock_stop_m':start_m+110,'stock_stop_u':start_u+110}


def verify_existing_root_facility(spec,Root_expected,clock,now_m,now_u,boot_id):
    """Root_expected is the already trusted Root execution context, not spec JSON.
    It must contain actual independently reviewed supported facility evidence.
    This function does NOT implement a missing provider or mint authority.
    """
    need(type(Root_expected)is dict,'MISSING_EXISTING_ROOT_FACILITY','no trusted existing context supplied')
    required=('outside_wait_provider_source','before_firstbirth_absolute_wait_ref',
              'direct_and_adopted_wait_provider_source','final_helper_wait_fresh_release_source',
              'independent_current_source_acceptance','actual_specific_action_admission',
              'original_clock_pin','original_final_helper_guard_binding_ref')
    missing=[k for k in required if Root_expected.get(k) is None]
    need(not missing,'MISSING_EXISTING_ROOT_FACILITY',','.join(missing))
    need(spec.get('domain')=='kernel120_qualification','HARMLESS_DOMAIN','native admission is separate')
    need(spec.get('clock')==Root_expected['original_clock_pin'],'ORIGINAL_CLOCK_PIN','no renewed spec clock')
    absolute=original_root_clock(clock,boot_id,now_m,now_u)
    proof=Root_expected.get('actual_wait_observation')
    need(type(proof)is dict,'MISSING_EXISTING_ROOT_FACILITY','actual exterior wait-arm evidence unbound')
    need(proof.get('action_id')==spec['native']['action_id']
         and proof.get('original_clock_pin')==spec['clock']
         and proof.get('actual_provider_source')==Root_expected['outside_wait_provider_source']
         and proof.get('Root_independently_verified_before_firstbirth') is True,
         'WAIT_PROVIDER_NOT_VERIFIED','candidate hashes/JSON alone never prove a wait or grant')
    need(all(finite(proof.get(k)) for k in ('wait_entered_m','wait_entered_u','first_birth_m','first_birth_u')),
         'WAIT_ARM_ORDER','actual independently collected dualclock before-firstbirth timestamps')
    need(absolute['start_m']<=proof['wait_entered_m']<proof['first_birth_m']<=now_m
         and absolute['start_u']<=proof['wait_entered_u']<proof['first_birth_u']<=now_u
         and proof.get('actor_stop_m')==absolute['actor_stop_m']
         and proof.get('actor_stop_u')==absolute['actor_stop_u']
         and proof.get('stock_stop_m')==absolute['stock_stop_m']
         and proof.get('stock_stop_u')==absolute['stock_stop_u']
         and proof.get('final_m')==absolute['final_m'] and proof.get('final_u')==absolute['final_u'],
         'WAIT_ARM_ORDER','one original absolute wait final-20/-10, shortening only')
    need(proof.get('retained_owned_stock_actor_pidfds_before_firstbirth') is True
         and proof.get('already_admitted_dedicated_context_subreaper') is True
         and proof.get('protected_or_TUI_context') is False,
         'WAIT_PROVIDER_NOT_VERIFIED','existing exact owned handles and supported dedicated subreaper context')
    return absolute


def existing_root_adapter(owner,spec,Root_expected,stock_birth,stock_pidfd,actor_birth,actor_pidfd,
                          group_fd,slot_fd,output_fd,output_owned_fds,evidence_fd,evidence_identity):
    """Called only by the existing independently admitted finite Root owner.
    The exterior wait already entered BEFORE firstbirth via a verified EXISTING
    provider. This counterpart cannot manufacture that provider from G8 history.
    Its existing owner finalizer waits exact children, captures raw events, then
    performs normal kernel0/group/FD/umount closure. Original slot stays held
    until the exterior Root waits THIS helper, observes fresh absence/terminal.
    """
    existing_root_caps()
    need(type(Root_expected)is dict and Root_expected.get('outside_wait_provider_source') is not None,
         'MISSING_EXISTING_ROOT_FACILITY','outside wait provider unbound; no launch/new facility')
    need(Root_expected.get('original_case_clock_mapping_ref') is not None
         and Root_expected.get('original_case_clock_producer_source_refs') is not None
         and type(Root_expected.get('original_case_clock_bindings')) is dict,
         'CLOCK_CASE_MAPPING_UNBOUND','exact original case-clock producer/bindings missing; before Source/FD effects')
    # Use the already accepted same-read owner object supplied by trusted Root.
    # Independent Root verifies that object against the exact final Source pins.
    need(Root_expected.get('actual_accepted_same_read_owner_object') is owner,
         'OWNER_SOURCE_OBJECT','actual same-read accepted object, not cached import or metadata label')
    custody=Root_expected.get('retained_fd_custody')
    root_fd_custody.require_custody(custody)
    owner.kernel120_validate_private_directories(spec,Root_expected)
    custody['private_directory_capabilities']=Root_expected['private_directory_capabilities']
    for handle,label in ((stock_pidfd,'stock_pidfd'),(actor_pidfd,'actor_pidfd'),(group_fd,'group'),
                         (slot_fd,'helper_slot_duplicate'),(output_fd,'output'),(evidence_fd,'trace_directory')):
        root_fd_custody.bind_existing(custody,handle,label)
    for handle in output_owned_fds:root_fd_custody.bind_existing(custody,handle,'existing_output')
    clock=__import__('json').loads(owner.same_read_pin(spec['clock'],65536))
    fd=root_fd_custody.opened(custody,'boot_id','/proc/sys/kernel/random/boot_id',os.O_RDONLY|os.O_NOFOLLOW)
    try:boot=os.read(fd,128).decode().strip()
    except BaseException as error:
        root_fd_custody.stopped(custody,error)
        raise
    else:root_fd_custody.closed(custody,fd)
    verify_existing_root_facility(spec,Root_expected,clock,time.monotonic(),time.time(),boot)
    proof=Root_expected['actual_wait_observation']
    need(proof.get('stock_identity')==[stock_birth['pid'],stock_birth['birth']]
         and proof.get('actor_identity')==[actor_birth['pid'],actor_birth['birth']],
         'WAIT_HANDLE_IDENTITY','same owned births as the actual before-firstbirth wait')
    for fd,label in ((stock_pidfd,'stock_pidfd_identity'),(actor_pidfd,'actor_pidfd_identity')):
        actual=os.fstat(fd)
        need(proof.get(label)==[actual.st_dev,actual.st_ino],
             'WAIT_HANDLE_IDENTITY','same retained pidfd inode as original wait; no recycled FD')
    value=ctypes.c_int(-1);libc=ctypes.CDLL(None,use_errno=True)
    need(libc.prctl(37,ctypes.byref(value),0,0,0)==0 and value.value==1,
         'EXISTING_SUBREAPER_REQUIRED','GET only: never enable/modify TUI or create new facility')
    need(len({stock_pidfd,actor_pidfd,group_fd,slot_fd,output_fd,evidence_fd})==6,
         'ROOT_HANDLE_ALIAS','distinct actual retained capabilities')
    result=owner.existing_root_finalize(spec,stock_birth,stock_pidfd,actor_birth,actor_pidfd,
             group_fd,slot_fd,output_fd,output_owned_fds,evidence_fd,evidence_identity,custody)
    need(result['capacity_release'] is False and result['Root_raw_trace_capture']['actual_wait_before_copy'] is True,
         'ROOT_FINAL_CUSTODY','helper cannot release original exterior heavy flock while alive')
    return result



def existing_root_before_opaque_provider(owner,spec,Root_expected,stock_birth,stock_pidfd,
        actor_birth,actor_pidfd,group_fd,original_slot_fd,ready_fd,go_fd,retained_fd_custody):
    """Actual named finite Root arm/GO/wait counterpart, not a JSON verifier.
    Must run in the exact admitted already dedicated exterior Root. The caller
    must have obtained/retained all actual stock/actor/gate handles before entry.
    Does not launch a helper/stock, modify prctl, pass FDs, mint a grant, or release
    the slot. Missing actual supported caller is preserved as a specific blocker.
    """
    import json
    existing_root_caps()
    need(type(Root_expected)is dict and Root_expected.get('actual_accepted_same_read_owner_object') is owner,
         'OWNER_SOURCE_OBJECT','Root independent same-read accepted owner object required')
    need(Root_expected.get('independent_current_source_acceptance') is not None
         and Root_expected.get('actual_specific_action_admission') is not None,
         'ROOT_NOT_ADMITTED','Source-only code never creates admission')
    need(spec['domain']=='kernel120_qualification' and spec['clock']==Root_expected['original_clock_pin'],
         'ORIGINAL_CLOCK_PIN','same original qualification clock')
    need(Root_expected.get('actual_existing_stock_launch_entrypoint_ref') is not None
         and Root_expected.get('actual_original_case_catalog_before_batch_ref') is not None,
         'EXISTING_ROOT_ENTRYPOINT_UNBOUND','existing launcher must establish inherited gate/handles and original catalog')
    clock=json.loads(owner.same_read_pin(spec['clock'],65536))
    state=owner.existing_root_arm_wait(spec,clock,retained_fd_custody,stock_birth,stock_pidfd,
        actor_birth,actor_pidfd,group_fd,original_slot_fd,ready_fd,go_fd)
    owner.existing_root_release_opaque_GO(state)
    return owner.existing_root_wait_owned(state)

def kernel120_existing_caller_prepare(actual_context,owner_birth,phase_caps,custody_source_pin):
    """Original caller validates coherent actual roles/parents, then retains originals."""
    from pathlib import Path
    existing_root_caps()
    need(actual_context.get('independent_current_source_acceptance') is not None
         and actual_context.get('actual_specific_action_admission') is not None,
         'ROOT_NOT_ADMITTED','actual existing caller Source/action admission')
    need(actual_context.get('actual_accepted_same_read_custody_module') is root_fd_custody,
         'OWNER_SOURCE_OBJECT','accepted exact live custody Source object')
    need(type(phase_caps)is dict and len(phase_caps)==23,
         'ACTUAL_INPUT_BINDINGS','complete existing actual phase fixture capabilities')
    # ALL phase local/remote semantic duplicates refuse before holder/role mutation.
    root_fd_custody.preflight_auxiliary_phase_plans(phase_caps,original_caller_birth=owner_birth)
    slot_fd=actual_context['original_slot_fd'];slot_identity=actual_context['original_slot_identity']
    groups={};parents={};retained={}
    def remember(table,row):
        fd=row['fd']
        need(fd not in table or table[fd]==row,'ACTUAL_CAPABILITY_ROLE_COLLISION','no silent last-row relabel')
        table[fd]=row
    def retain_directory(cap,role,private):
        root_fd_custody.directory_capability(cap,private)
        q=os.fstat(cap['fd'])
        remember(retained,{'fd':cap['fd'],'identity':[q.st_dev,q.st_ino,stat.S_IFMT(q.st_mode)],'role':role})
    def output_parent(cap,path):
        root_fd_custody.validate_private_parent(path,cap)
        fd=cap['fd']
        if fd in parents:
            need(all(parents[fd][k]==cap[k] for k in ('fd','path','identity','policy')),
                 'ACTUAL_PARENT_CAPABILITY_COLLISION','same retained private parent')
        else:parents[fd]=dict(cap,output_paths=[])
        if path not in parents[fd]['output_paths']:parents[fd]['output_paths'].append(path)
    for cap in phase_caps.values():
        need(cap['slot_fd']==slot_fd,'SERIAL_OWNER','one same original retained existing slot')
        private=cap['private_directory_capabilities']
        need(set(private)=={'output_parent','control','control_parent','trace','trace_parent'},
             'ACTUAL_PRIVATE_PARENTS','complete original admitted private directory roles')
        output_parent(private['output_parent'],str(Path(cap['record']).parent))
        need(private['output_parent']['fd']==cap['output_parent_fd']
             and private['output_parent']['identity']==cap['output_parent_identity'],
             'ACTUAL_OUTPUT_PARENT','exact original private parent role')
        for role in ('control','trace'):
            root_fd_custody.validate_private_parent(private[role]['path'],private[role+'_parent'])
            retain_directory(private[role],role+'_directory',True)
            retain_directory(private[role+'_parent'],'original_caller_private_parent',True)
        group={'fd':cap['group_fd'],'identity':cap['first_birth_group_identity'],'path':cap['group'],
               'policy':cap['group_directory_policy'],'parent_capability':cap['group_parent_capability'],
               'role':'original_caller_group'}
        remember(groups,group)
        retain_directory(cap['group_parent_capability'],'original_caller_group_parent',False)
        for directory in cap.get('auxiliary_parent_capabilities',[]):
            group_parent=root_fd_custody.retained_parent_is_cgroup(directory)
            retain_directory(directory,'original_caller_group_parent' if group_parent
                             else 'original_caller_private_parent',not group_parent)
        actual=cap.get('actual_control_binding',{})
        for aux in actual.get('auxiliary_groups',[]):
            remember(groups,{'fd':aux['ownership']['caller_fd'],'identity':aux['identity'],
                'path':aux['path'],'policy':aux['directory_policy'],'parent_capability':aux['parent_capability'],
                'role':'auxiliary_group'})
            retain_directory(aux['parent_capability'],'original_caller_group_parent',False)
        for aux in actual.get('auxiliary_outputs',[]):
            output_parent(aux['parent_capability'],str(Path(aux['record']).parent))
            # The declared original caller owner may be a distinct duplicate
            # of the same private parent. Retain BOTH exact capabilities;
            # physical/role/policy preflight above has already validated them.
            output_parent(dict(aux['parent_capability'],fd=aux['ownership']['caller_fd']),
                          str(Path(aux['record']).parent))
        for aux in actual.get('auxiliary_fds',[]):
            need(aux['role']=='auxiliary_borrowed' and aux['ownership']['role']=='auxiliary_borrowed',
                 'ACTUAL_AUXILIARY_ROLE','borrowed never reserved role')
            remember(retained,{'fd':aux['ownership']['caller_fd'],'identity':aux['identity'],'role':'auxiliary_borrowed'})
    holder=root_fd_custody.ExistingCallerCustody(owner_birth,slot_fd,slot_identity,
             list(groups.values()),list(parents.values()),list(retained.values()))
    root_fd_custody.retain_in_existing_caller(actual_context,holder)
    return {phase_id:dict(cap,retained_existing_caller=holder.binding_for_phase(
                  custody_source_pin,cap['group_fd'],cap['output_parent_fd']))
            for phase_id,cap in phase_caps.items()}


def kernel120_existing_caller_terminal(actual_context,guard_birth,guard_pidfd,clock):
    """Fixed actual caller terminal hook, ONE exact direct wait then actual
    closure. Failure returns STOP to existing owner with its LIVE holder retained;
    caller must not normally exit/close originals, retry or launch another cohort.
    """
    existing_root_caps()
    holder=actual_context.get('kernel120_holder')
    need(type(holder)is root_fd_custody.ExistingCallerCustody,
         'EXISTING_CALLER_CUSTODY_UNBOUND','actual persistent caller capability object')
    result=holder.after_exact_cli_wait(guard_birth,guard_pidfd,clock)
    if result['capacity_release'] is not True:
        actual_context['kernel120_STOP_UNCONFIRMED']=True
        actual_context['kernel120_no_more_launches']=True
    return result
