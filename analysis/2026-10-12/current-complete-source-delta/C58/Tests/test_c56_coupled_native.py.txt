@pytest.mark.parametrize('case_id',NEGATIVE)
def test_negative_native_cause_no_false_credit(c56_live_native,case_id):
    o=c56_live_native.run(case_id)
    assert o.intended_cause_actually_reached
    assert o.delivery_observation!='DELIVERED'
    assert o.no_late_promotion_or_delivered_persistence
    assert o.retry_or_budget_reset_count==0
    assert o.signaled_births<=o.actual_product_owned_births
    assert not (o.signaled_births & o.protected_writer_births_before)
    assert o.protected_writer_births_before==o.protected_writer_births_after
    if 'blocked_' in case_id:
        assert o.real_native_block_observed
        assert o.guard_armed_before_block_and_retained_to_terminal
    if 'observer_' in case_id:
        assert o.direct_pid_recorded_before_first_parent_failure
        assert o.normal_or_adverse_raw_wait_recorded or o.exact_unreaped_custody_retained
        assert o.no_GOrerun_or_second_fork
        if o.cleanup_unconfirmed:
            assert o.gate_acceptance is False
            assert o.stop_status=='STOP_UNCONFIRMED'
            assert o.exact_unreaped_custody_retained
    if 'state_replace' in case_id:
        assert o.real_named_inode_changed
        assert o.same_content_replacement_is_not_same_generation
        assert o.changed_generation_did_not_earn_certificate
