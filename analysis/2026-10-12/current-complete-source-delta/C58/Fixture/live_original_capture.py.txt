"""C57 original native capture. Observation only: no call, scope, ledger,
UI, task ID or process is constructed. Physical fault production is not granted.
Missing raw/native facts fail explicitly; pre-ARM refusals may have no lease.
"""
import os,time
from dataclasses import dataclass
from types import MappingProxyType
from collections.abc import Mapping
class MissingWitness(RuntimeError): pass

@dataclass(frozen=True)
class OriginalObjects:
    AppSession: object
    Application: object
    Console: object
    stream: object
    Input: object
    PluginContext: object
    CLIUserScope: object
    manager: object
    SessionDB: object

def original_objects(cli,context):
    from prompt_toolkit.application.current import get_app_session
    from hermes_cli.plugins import PluginContext
    from cli import HermesCLI
    from hermes_state import SessionDB
    if not isinstance(cli,HermesCLI) or not isinstance(context,PluginContext):
        raise MissingWitness('actual_original_HermesCLI_PluginContext_required')
    # Observation must also retain genuinely changed identities in negatives.
    # Admission is checked separately before effects by the original product.
    session=get_app_session()
    out=cli._friday_cli_output
    db=cli.agent._session_db
    if not isinstance(db,SessionDB):
        raise MissingWitness('original_SessionDB_unbound')
    return OriginalObjects(session,cli._app if hasattr(cli,'_app') else None,
        cli.console,out.stream,session.input,context,cli._friday_cli_scope,
        context._manager,db)

def same_objects(a,b):
    return all(getattr(a,k) is getattr(b,k) for k in a.__dataclass_fields__)

class NativeCase:
    def __init__(self,key,cli,context,authority):
        from hermes_cli.friday_cli_work import _TOOL,_TURN
        if (_TOOL.get() is not authority or authority['turn'] is not _TURN.get()
                or not authority['active'] or authority['turn']['cli'] is not cli):
            raise MissingWitness('genuine_original_native_call_not_current')
        self.case_id=key.rsplit('::',1)[-1];self.key=key
        self.cli=cli;self.context=context;self.authority=authority
        self.before=original_objects(cli,context)
        self.ids_before=dict(authority['ids']);self.birth_before=self._birth()
        self.events=[('before_original_handler',time.monotonic())]
        self.lease=None;self.result=None;self.primary=None
        self.certificates={};self.sealed=False;self.facts={}
        self.raw_evidence=None;self.historical_evidence={}

    def _birth(self):
        from hermes_cli.friday_cli_finite import _birth
        return _birth(os.getpid())

    def observe_after_handler(self,result):
        from hermes_cli.friday_cli_finite import OriginalDeliveryLease
        self.result=result
        lease=self.authority.get('finite_lease')
        if lease is not None:
            if type(lease) is not OriginalDeliveryLease or lease.authority is not self.authority:
                raise MissingWitness('foreign_original_lease')
            self.lease=lease
            # Only this operation's certificates; no historical A-to-B mixing.
            self.certificates={h:lease.port.certificates[h]
                for h in lease.proposals if h in lease.port.certificates}
            if not lease.proposals and lease.historical_reads:
                import json
                from hermes_cli.friday_cli_finite import AuthenticatedHistoricalRead,canonical
                response=json.loads(result)
                reference=response.get('reference') if type(response) is dict else None
                for read in lease.historical_reads:
                    if (type(read) is not AuthenticatedHistoricalRead
                            or read.current_lease is not lease
                            or not lease.started<=read.reader_m<lease.end):
                        raise MissingWitness('foreign_current_historical_reader')
                    old=read.original_lease;h=read.proposal_hash
                    proposal=old.proposals[h]
                    if proposal['reference']!=reference:continue
                    if (not old.closed or old.port is not lease.port
                            or old.completed_certificate_bytes[h]!=read.certificate_bytes
                            or canonical(old.port.certificates[h])!=read.certificate_bytes):
                        raise MissingWitness('changed_immutable_historical_certificate_custody')
                    self.historical_evidence[h]=read
                    self.certificates[h]=json.loads(read.certificate_bytes)

        self.events.append(('after_original_handler',time.monotonic()))

    def observe_primary(self,exception):
        self.primary=exception
        self.events.append(('actual_original_primary',time.monotonic()))

    def prepare_actual_terminal(self):
        if self.sealed or hasattr(self,'after'):
            raise MissingWitness('terminal_capture_no_replay')
        self.after=original_objects(self.cli,self.context)
        self.birth_after=self._birth();self.ids_after=dict(self.authority['ids'])
        self.events.append(('after_actual_caller_tail_before_final_packet',time.monotonic()))
        self.raw_evidence=RawOriginalWitness(self)

    def observe_actual_terminal(self):
        # No proc/read/import/clock/callback after the final completion packet.
        if self.sealed or not hasattr(self,'after'):
            raise MissingWitness('actual_under_guard_terminal_capture_missing')
        if self.lease is not None and not self.lease.closed:
            raise MissingWitness('original_caller_completion_unconfirmed')
        if self.lease is not None and self.lease.original_projection is not None:
            published=self.lease.original_projection['published']
            if published is not None:self.result=published
        self.sealed=True

    def measured(self):
        if not self.sealed:raise MissingWitness('actual_caller_terminal_not_observed')
        keys=('AppSession','Application','Console','stream','PluginContext','CLIUserScope')
        values={
            'original_objects_before':{k:getattr(self.before,k) for k in keys},
            'original_objects_after':{k:getattr(self.after,k) for k in keys},
            'original_AppSession_is_same':self.before.AppSession is self.after.AppSession,
            'original_Input_is_same':self.before.Input is self.after.Input,
            'original_objects_preserved':same_objects(self.before,self.after),
            'original_native_objects_preserved':same_objects(self.before,self.after),
            'native_five_ids_before':self.ids_before,'native_five_ids_after':self.ids_after,
            'actual_original_ids_preserved':self.ids_before==self.ids_after,
            'frontend_birth_before':self.birth_before,'frontend_birth_after':self.birth_after,
            'original_frontend_still_alive':self.birth_before==self.birth_after,
            'actual_result':self.result,'certificates':MappingProxyType(self.certificates),
            'actual_terminal_capture_m':self.events[-1][1],
            'primary_exception':self.primary,
        }
        if self.lease is not None:
            values['original_started_m']=self.lease.started
            values['original_deadline_m']=self.lease.end
            values['original_deadline_seconds']=self.lease.end-self.lease.started
            if self.lease.closed and hasattr(self.lease,'actual_caller_completed_m'):
                values['normal_closed_m']=self.lease.actual_caller_completed_m
                values['native_terminal_m']=self.lease.actual_caller_completed_m
        return MappingProxyType(values)

class RawOriginalWitness:
    """Derive fields from actual original references and authenticated raw
    observer certificates. No caller boolean/JSON is accepted as a witness.
    Unknown physical faults, targets or negative syscalls remain unavailable."""
    def __init__(self,case):
        self.case=case;self.retained_native_original_objects=case.before
        self.raw={}
        lease=case.lease
        if lease is None:
            # A genuine pre-ARM refusal is recorded without inventing a lease.
            self.raw['actual_no_original_lease']=True
            return
        self.raw['actual_activity_thread_joined']=(
            case.authority['activity_thread'] is not None
            and not case.authority['activity_thread'].is_alive()
            and case.authority['activity_joined'])
        if case.historical_evidence:
            from hermes_cli.friday_cli_finite_owner import current_record_generation
            from hermes_cli.friday_cli_finite import canonical
            fresh_generation=canonical(current_record_generation(case.context,lease.end))
            for read in case.historical_evidence.values():
                if read.current_generation_bytes!=fresh_generation:
                    raise MissingWitness('historical_reader_generation_changed_before_capture')
            self.raw['actual_reader_current_generation_matches_A_certificate']=True
            self.raw['actual_retained_A_certificate_bytes_unchanged']=all(
                read.original_lease.completed_certificate_bytes[read.proposal_hash]==read.certificate_bytes
                for read in case.historical_evidence.values())
        observed_leases=([read.original_lease for read in case.historical_evidence.values()]
                         if case.historical_evidence else [lease])
        self.raw['same_native_registration_released_and_removed']=all(
            observed.registration is not None and not observed.registration.active
            and observed.registration is getattr(observed.task,'_friday_cli_ledger_cleanup',None)
            and not any(observed.registration is h
                for hs in case.context._manager._ownership_ledger.values() for h in hs)
            for observed in observed_leases)
        self.observed_delivery_leases=tuple(observed_leases)
        if case.certificates:
            from hermes_cli.friday_cli_finite import validate_certificate
            certs=[validate_certificate(c,(case.historical_evidence[h].original_lease
                if h in case.historical_evidence else lease).proposals[h])
                for h,c in case.certificates.items()]
            raw={c['observer']['raw_status'] for c in certs}
            if len(raw)!=1:raise MissingWitness('incoherent_actual_observer_raw_status')
            self.raw['actual_finite_observer_raw_wait_status']=next(iter(raw))
            self.raw['actual_normal_observer_raw_wait_status']=next(iter(raw))
            self.raw['observer_raw_status']=next(iter(raw))
            self.raw['actual_certified_m']=max(c['certified_m'] for c in certs)
            self.raw['actual_proposal_durable_m']=max(c['proposal_durable_m'] for c in certs)
            self.raw['actual_output_and_ledger_join_m']=max(c['frontend_joined_m'] for c in certs)
            direct=[DirectChildWaitEvidence.from_actual_native(case,c) for c in certs]
            self.direct_wait_evidence=tuple(direct)
            self.raw['raw_wait_proves_observer_only']=all(
                w.child_pid!=case.birth_before['pid'] and w.raw_status==0 for w in direct)
            self.raw['observer_raw_wait_source']=direct[0].discriminator
            self.raw['actual_named_state_generation_observer']=[
                c['record_generation'] for c in certs]

    @property
    def original_contract_mapping(self):
        return RetainedOriginalContractMapping(self.case,self)

    def field(self,name):
        lease=self.case.lease
        if lease is not None and self.case.sealed:
            if name=='current_lease_proposal_hashes':return frozenset(lease.proposals)
            if name=='terminal_validated_proposal_hashes':
                return frozenset(lease.terminal_validated_proposal_hashes)
            if name=='actual_named_state_generation_terminal':
                return lease.terminal_record_generation
            if name=='original_guardian_accepted_completed_scope':
                ack=lease.guardian_caller_acceptance
                return (lease.closed and ack is not None and ack['scope']==lease.scope
                        and ack['completed_m']==lease.actual_caller_completed_m
                        and ack['accepted_m']<lease.end)
        if name not in self.raw:
            raise MissingWitness('UNBOUND_PHYSICAL_OR_NATIVE_FIELD:'+name)
        return self.raw[name]

class LiveOriginalCapture:
    def __init__(self,expected_keys):
        self.expected=frozenset(expected_keys);self.cases={};self.consumed=set()
    def record(self,key,case):
        if (key not in self.expected or key in self.cases
                or type(case) is not NativeCase or not case.sealed):
            raise MissingWitness('unexpected_duplicate_or_unsealed_native_case')
        self.cases[key]=case
    def consume(self,key):
        if key not in self.cases:raise MissingWitness('UNBOUND_NATIVE_CASE:'+key)
        if key in self.consumed:raise MissingWitness('native_case_no_replay')
        self.consumed.add(key);return self.cases[key].measured()
    def complete_inventory(self):
        if set(self.cases)!=self.expected:
            raise MissingWitness('actual_case_missing_no_silent_skip')
        return tuple(sorted(self.cases))

class OriginalNativeProducer:
    """Connected original-call producer; does not drive or manufacture calls.
    The existing admitted native Harness must supply the real case journey and
    external fault/custody facilities. That integration remains NOT_DONE."""
    def __init__(self,cli,context,capture):
        if type(capture) is not LiveOriginalCapture:
            raise MissingWitness('exact_original_capture_required')
        self.cli=cli;self.context=context;self.capture=capture
        self.selected=None;self.selected_tool=None;self.open={}
    def select_original_case(self,key,tool_name='friday_result'):
        if key not in self.capture.expected or self.selected is not None:
            raise MissingWitness('unadmitted_or_overlapping_case')
        if tool_name not in ('friday_result','friday_work'):
            raise MissingWitness('unsupported_original_case_tool')
        self.selected=key;self.selected_tool=tool_name
    def begin(self,authority):
        if self.selected is None or authority['tool_name']!=self.selected_tool:
            return None
        key=self.selected;self.selected=None;self.selected_tool=None
        case=NativeCase(key,self.cli,self.context,authority)
        ident=authority['ids']['tool_call_id']
        if ident in self.open:raise MissingWitness('original_call_replay')
        self.open[ident]=case
        return case
    def prepare_terminal(self,agent,call_id):
        if call_id not in self.open:return
        case=self.open[call_id]
        if case.authority['agent'] is not agent:
            raise MissingWitness('original_terminal_agent_changed')
        case.prepare_actual_terminal()

    def finish(self,agent,call_id):
        if call_id not in self.open:return
        case=self.open[call_id]
        if case.authority['agent'] is not agent:
            raise MissingWitness('original_terminal_agent_changed')
        case.observe_actual_terminal()
        self.capture.record(case.key,case);del self.open[call_id]


@dataclass(frozen=True)
class DirectChildWaitEvidence:
    parent_pid:int
    parent_birth:str
    child_pid:int
    child_birth:str
    raw_status:int
    wait_m:float
    cleanup_m:float
    @classmethod
    def from_actual_native(cls,case,certificate):
        from hermes_cli.friday_cli_finite import validate_certificate,digest
        lease=case.lease
        if lease is None or lease.authority is not case.authority:
            raise MissingWitness('actual_original_raw_wait_lease_missing')
        h=certificate['proposal_sha256']
        original=lease
        if h in case.historical_evidence:
            read=case.historical_evidence[h]
            from hermes_cli.friday_cli_finite import AuthenticatedHistoricalRead,canonical
            if (type(read) is not AuthenticatedHistoricalRead or read.current_lease is not lease
                    or read.original_lease is lease or not read.original_lease.closed
                    or read.certificate_bytes!=canonical(certificate)):
                raise MissingWitness('actual_historical_reader_raw_custody_unbound')
            original=read.original_lease
        if (h not in original.proposals or h not in case.certificates
                or original.port is not lease.port
                or lease.port.certificates[h]!=case.certificates[h]):
            raise MissingWitness('copied_or_foreign_raw_wait_not_original_channel')
        c=validate_certificate(certificate,original.proposals[h]);w=c['observer'];d=w['direct_wait']
        if (c['guardian_pid']!=lease.port.outside['pid']
                or c['guardian_birth']!=lease.port.outside['birth']
                or c['frontend_pid']!=case.birth_before['pid']
                or c['frontend_birth']!=case.birth_before['birth']
                or d['wait_pid']!=w['pid'] or d['retained_child_birth']!=w['birth']
                or d['parent_pid']!=c['guardian_pid']
                or d['parent_birth']!=c['guardian_birth']
                or d['observed_child_ppid']!=d['parent_pid']
                or w['pid'] in (d['parent_pid'],c['frontend_pid'])):
            raise MissingWitness('foreign_wrapper_or_frontend_raw_wait_cannot_qualify')
        return cls(d['parent_pid'],d['parent_birth'],w['pid'],w['birth'],
                   w['raw_status'],w['wait_m'],w['cleanup_m'])
    @property
    def discriminator(self):
        # This value is projected only from the above actual retained guardian
        # certificate chain. A wrapper/frontend exit or missing parent/birth
        # cannot construct the qualified field through the native producer.
        return 'DIRECT_CHILD_WAITPID_EXACT_BIRTH'


class RetainedOriginalContractMapping(Mapping):
    """Actual original object references plus typed raw fields. Incomplete
    physical provenance cannot become a dictionary of invented True markers."""
    KEYS=('actual_all_effect_observer_cleanup_joined_m', 'actual_case_id', 'actual_certificate_reader_admission_preserved', 'actual_finite_observer_raw_wait_status', 'actual_first_effect_m', 'actual_frontend_alive_after_return', 'actual_frontend_pid_birth_after', 'actual_frontend_pid_birth_before', 'actual_guard_armed_m', 'actual_input_restored_m', 'actual_last_effect_m', 'actual_native_renderer_and_ledger_joined', 'actual_no_owned_orphan_writer_or_observer', 'actual_original_end_m', 'actual_original_input_modes_restored', 'actual_original_start_m', 'actual_proposal_durable_m', 'actual_raw_wait_producer_custody_verified', 'actual_reader_return_m', 'actual_witness_m', 'artifact_bytes_exactly_once', 'cleanup_unconfirmed', 'exact_owned_custody_retained', 'exact_source_bootstrap_dependency_binding_verified', 'finite_effect_actor_is_original_frontend', 'fixed_original_native_record_handle_transfer_only', 'full_gate_acceptance', 'genuine_joined_finite_effect_observer_cleanup_witness', 'independent_fixture_provenance_verified', 'native_five_ids_after', 'native_five_ids_before', 'native_uid_home_generation_database_inode_preserved', 'no_delivery_replay', 'no_developer_or_foreign_writer_effect', 'no_failure_promoted_from_json_or_task_done', 'no_late_delivered_persistence_or_presentation', 'no_new_engine_pump_service_hidden_tui_or_unqualified_fd_transfer', 'no_original_attempt_budget_renewal', 'observer_raw_wait_proves_observer_only', 'original_effect_and_cleanup_completed_while_frontend_lives', 'original_objects_after', 'original_objects_before', 'physical_native_run', 'primary_exception', 'public_delivery_observation', 'reported_primary_exception', 'same_native_registration_release_and_removal_observed', 'whole_original_frontend_wait_status')
    def __init__(self,case,raw):self.case=case;self.raw=raw
    def __iter__(self):return iter(self.KEYS)
    def __len__(self):return len(self.KEYS)
    def __getitem__(self,key):
        measured=self.case.measured();lease=self.case.lease
        if key=='actual_case_id':return self.case.case_id
        if key in ('original_objects_before','original_objects_after'):
            source=self.case.before if key.endswith('before')else self.case.after
            names={'app_session':'AppSession','application':'Application',
                'console':'Console','stream':'stream','input':'Input',
                'context':'PluginContext','manager':'manager',
                'native_principal':'CLIUserScope','session_database':'SessionDB'}
            return {name:getattr(source,field)for name,field in names.items()}
        if key=='native_five_ids_before':return self.case.ids_before
        if key=='native_five_ids_after':return self.case.ids_after
        if key=='actual_frontend_pid_birth_before':return self.case.birth_before
        if key=='actual_frontend_pid_birth_after':return self.case.birth_after
        if key=='actual_original_start_m'and lease is not None:return lease.started
        if key=='actual_original_end_m'and lease is not None:return lease.end
        if key=='primary_exception':return self.case.primary
        if key=='same_native_registration_release_and_removal_observed':
            return self.raw.field('same_native_registration_released_and_removed')
        if key=='actual_finite_observer_raw_wait_status':
            return self.raw.field('actual_finite_observer_raw_wait_status')
        if key=='observer_raw_wait_proves_observer_only':
            return self.raw.field('raw_wait_proves_observer_only')
        # actual physical run, independent fixture qualification, outside
        # frontend raw custody, stop/fault observations are never synthesized.
        return self.raw.field(key)
