"""Current pytest adapters over actual same-process original native capture.
Binding via pytest config is installed only by the independently qualified intact
Harness. Missing any witness fails; no skip, fake case or invocation is created."""
import pytest
from live_original_capture import LiveOriginalCapture,MissingWitness
from types import SimpleNamespace

class ActualCaseView:
    def __init__(self,case):
        self.case=case;self.measured=case.measured()
    def __getattr__(self,name):
        if name in ('presentation','delivery','delivery_observation'):
            import json
            result=self.case.result
            if type(result) is not str:
                raise MissingWitness('actual_original_result_string_unbound')
            row=json.loads(result)
            if 'delivery_observation' in row:state=row['delivery_observation']
            elif 'delivery' in row:state=row['delivery']
            else:raise MissingWitness('actual_original_delivery_field_missing')
            if state not in ('DELIVERED','UNKNOWN','FAILED','PARTIAL','NOT_RUN'):
                raise MissingWitness('actual_original_delivery_field_invalid')
            if state=='DELIVERED':
                if (not self.case.certificates or not self.measured['original_objects_preserved']
                        or not self.measured['original_frontend_still_alive']
                        or not self.case.lease.closed
                        or self.case.lease.actual_caller_completed_m>=self.case.lease.end):
                    raise MissingWitness('actual_DELIVERED_conjunction_unconfirmed')
            return state
        if name=='original_objects_preserved':
            return self.measured['original_objects_preserved']
        if name in self.measured:
            return self.measured[name]
        # Retained actual native witness access is qualified as part of the
        # intact Harness source, not accepted from arbitrary JSON/booleans.
        producer=self.case.raw_evidence
        if not getattr(producer,'retained_native_original_objects',None) is self.case.before:
            raise MissingWitness('UNBOUND_REAL_NATIVE_WITNESS:'+name)
        from live_original_capture import RawOriginalWitness
        if type(producer) is not RawOriginalWitness:
            raise MissingWitness('actual_raw_original_witness_type_unbound')
        return producer.field(name)

class CaseAdapter:
    def __init__(self,capture,module):
        if type(capture) is not LiveOriginalCapture:
            raise MissingWitness('intact_native_capture_not_bound')
        self.capture=capture;self.module=module
    def run(self,case_id):
        key=self.module+'::'+case_id
        self.capture.consume(key)
        return ActualCaseView(self.capture.cases[key])
    def run_exact_current_source(self,case_id):
        return self.run(case_id)
    def __getitem__(self,case_id):
        # C54 contract requires real Python "is" object references, so its
        # qualified native projection remains a retained same-process mapping.
        key=self.module+'::'+case_id
        self.capture.consume(key)
        case=self.capture.cases[key]
        producer=case.raw_evidence
        if getattr(producer,'retained_native_original_objects',None) is not case.before:
            raise MissingWitness('genuine_native_mapping_unbound')
        result=producer.original_contract_mapping
        if result['actual_case_id']!=case_id:
            raise MissingWitness('wrong_actual_native_case')
        return result

def _capture(request):
    capture=getattr(request.config,'_friday_actual_native_capture',None)
    if type(capture) is not LiveOriginalCapture:
        raise MissingWitness('UNBOUND:intact_Harness_native_capture_fixture')
    return capture
