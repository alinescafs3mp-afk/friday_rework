"""Optional original native observation seam; no job/authority/clock created."""
from contextvars import ContextVar
from contextlib import contextmanager
_CAPTURE=ContextVar('friday_original_native_capture',default=None)

@contextmanager
def installed_original_capture(producer):
    from live_original_capture import OriginalNativeProducer
    from hermes_cli.friday_cli_principal import current_cli
    from hermes_cli.friday_cli_finite import RetainedOwnerChannel
    from cli import HermesCLI
    # The actual ordinary bootstrap and constructor already own these objects.
    # Installation observes them; it cannot create a principal/UI/native call.
    cap=current_cli(required=True)
    port=getattr(producer.cli,'_friday_cli_finite_port',None)
    if (type(producer) is not OriginalNativeProducer
            or not isinstance(producer.cli,HermesCLI)
            or producer.cli._friday_cli_scope is not cap
            or type(port) is not RetainedOwnerChannel
            or port.cli is not producer.cli or port.cap is not cap):
        raise RuntimeError('actual_original_capture_scope_required')
    port._check()
    if _CAPTURE.get() is not None:
        raise RuntimeError('original_capture_cannot_replace_existing')
    token=_CAPTURE.set(producer)
    try:yield producer.capture
    finally:_CAPTURE.reset(token)

def begin_original_call(authority):
    producer=_CAPTURE.get()
    if producer is None:return None
    case=producer.begin(authority)
    if case is None:return None
    authority['_native_capture_case']=case
    authority['agent']._friday_cli_native_capture_producer=producer
    return case

def after_original_handler(result):
    from hermes_cli.friday_cli_work import _TOOL
    authority=_TOOL.get()
    if authority is None:return
    case=authority.get('_native_capture_case')
    if case is not None:case.observe_after_handler(result)

def primary_original_failure(authority,failure):
    case=authority.get('_native_capture_case') if authority is not None else None
    if case is not None:case.observe_primary(failure)

def finish_original_caller(agent,call_id):
    producer=getattr(agent,'_friday_cli_native_capture_producer',None)
    if producer is not None:producer.finish(agent,call_id)

def prepare_original_terminal(agent,call_id):
    producer=getattr(agent,'_friday_cli_native_capture_producer',None)
    if producer is not None:producer.prepare_terminal(agent,call_id)
