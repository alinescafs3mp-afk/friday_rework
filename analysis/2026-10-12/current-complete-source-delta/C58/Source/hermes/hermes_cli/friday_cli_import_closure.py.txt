"""Closed original Python/plugin namespace restriction. SOURCE ONLY.
Installation restricts source dependencies; it grants no RootGo or effects.
Unknown native ELF loader custody refuses before initializer execution."""
import hashlib,importlib.abc,importlib.machinery,importlib.util,importlib.metadata
import os,stat,sys,threading,io
from importlib.resources.abc import Traversable,TraversableResources
from pathlib import Path
_POLICY=None
_LOCK=threading.RLock()
class ClosedImportError(ImportError):pass

def exact_bytes(row):
    fd=os.open(row['path'],os.O_RDONLY|os.O_NOFOLLOW|os.O_CLOEXEC)
    try:
        a=os.fstat(fd)
        sig=lambda s:(s.st_dev,s.st_ino,s.st_mode,s.st_uid,s.st_nlink,s.st_size,s.st_mtime_ns,s.st_ctime_ns)
        if (not stat.S_ISREG(a.st_mode) or a.st_nlink!=1 or a.st_size!=row['bytes']
                or a.st_uid not in (0,os.getuid()) or a.st_mode&0o022):
            raise ClosedImportError('unqualified_singlelink_import_file')
        b=bytearray()
        while len(b)<a.st_size:
            part=os.read(fd,min(65536,a.st_size-len(b)))
            if not part:raise ClosedImportError('short_import_file')
            b.extend(part)
        named=os.stat(row['path'],follow_symlinks=False)
        if (sig(os.fstat(fd))!=sig(a) or sig(named)!=sig(a)
                or hashlib.sha256(b).hexdigest()!=row['sha256']):
            raise ClosedImportError('import_generation_or_hash_changed')
        return bytes(b)
    finally:os.close(fd)

class ExactSourceLoader(importlib.abc.Loader):
    def __init__(self,row,package,resources,logical_resources,logical_path):
        self.row=row;self.package=package;self.resources=resources
        self.logical_resources=logical_resources;self.logical_path=logical_path
    def create_module(self,spec):return None
    def exec_module(self,module):
        b=exact_bytes(self.row)
        module.__file__=self.row['path']
        if self.package:module.__path__=list(module.__spec__.submodule_search_locations)
        # Future independently admitted execution only; no C58 compile/import.
        exec(compile(b,self.row['path'],'exec',dont_inherit=True),module.__dict__)
    def get_data(self,path):
        row=self.resources.get(os.path.abspath(path))
        if row is None:raise ClosedImportError('UNBOUND_PACKAGE_RESOURCE:'+path)
        return exact_bytes(row)
    def get_resource_reader(self,fullname):
        return ExactResourceReader(self.logical_path.rsplit('/',1)[0],self.logical_resources)

class ExactDistribution(importlib.metadata.Distribution):
    def __init__(self,root,resources):self.root=Path(root);self.resources=resources
    def read_text(self,filename):
        p=str(self.root/filename);row=self.resources.get(p)
        return None if row is None else exact_bytes(row).decode('utf-8')
    def locate_file(self,path):
        p=os.path.abspath(self.root.parent/path)
        if p not in self.resources:raise ClosedImportError('unqualified_distribution_file')
        exact_bytes(self.resources[p]);return Path(p)

class ClosedSourceFinder(importlib.abc.MetaPathFinder):
    def __init__(self,modules,resources,namespaces,logical_resources):
        self.modules=dict(modules);self.resources=dict(resources)
        self.logical_resources=dict(logical_resources)
        self.namespaces={k:tuple(v) for k,v in namespaces.items()}
        self.namespaces['hermes_plugins']=()
    def _spec(self,name,item):
        if item['kind']=='extension':
            # DT_NEEDED byte inventory does not authenticate the actual loader,
            # mapped SONAME objects or late dlopen. Fail before native effects.
            raise ClosedImportError('UNBOUND_RETAINED_ELF_LOADER_CUSTODY:'+name)
        row=item['effective'];exact_bytes(row)
        loader=ExactSourceLoader(row,item['package'],self.resources,self.logical_resources,item['logical_path'])
        spec=importlib.util.spec_from_loader(name,loader,is_package=item['package'])
        spec.origin=row['path'];spec.has_location=True
        if item['package']:spec.submodule_search_locations=[str(Path(row['path']).parent)]
        return spec
    def find_spec(self,fullname,path=None,target=None):
        if fullname in ('site','sitecustomize','usercustomize','_virtualenv'):
            raise ClosedImportError('ambient_startup_forbidden')
        with _LOCK:
            if fullname in self.modules:return self._spec(fullname,self.modules[fullname])
            if fullname in self.namespaces:
                spec=importlib.machinery.ModuleSpec(fullname,None,is_package=True)
                spec.submodule_search_locations=list(self.namespaces[fullname]);return spec
        raise ClosedImportError('UNBOUND_IMPORT:'+fullname)
    def find_distributions(self,context=importlib.metadata.DistributionFinder.Context()):
        roots=sorted({str(Path(p).parent)for p in self.resources
            if p.endswith('/METADATA') and Path(p).parent.name.endswith('.dist-info')})
        for root in roots:
            dist=ExactDistribution(root,self.resources)
            if context.name is None or dist.metadata['Name'].lower().replace('-','_')==context.name.lower().replace('-','_'):
                yield dist
    def dynamic_spec(self,name,file,locations):
        file=os.path.abspath(file)
        matches=[(n,r)for n,r in self.modules.items()if r['effective']['path']==file]
        if not matches:raise ClosedImportError('unqualified_dynamic_plugin_file:'+file)
        _,item=matches[0]
        if name in self.modules and self.modules[name]['effective']!=item['effective']:
            raise ClosedImportError('dynamic_alias_cannot_shadow_qualified_module')
        # Only the original plugin directory loaders use this alias path; it
        # remains a dependency restriction, never native operation authority.
        if name not in self.modules and not name.startswith(('hermes_plugins.','plugins.')):
            raise ClosedImportError('unqualified_dynamic_alias:'+name)
        prefix=item['logical_path'].removesuffix('/__init__.py')
        if item['package']:
            additions={}
            for old,child in tuple(self.modules.items()):
                logical=child['logical_path']
                if logical.startswith(prefix+'/') and logical.endswith('.py'):
                    rel=logical[len(prefix)+1:]
                    suffix=rel.removesuffix('.py').replace('/','.')
                    if suffix=='__init__':continue
                    if suffix.endswith('.__init__'):suffix=suffix[:-9]
                    additions[name+'.'+suffix]=child
            for alias,child in additions.items():
                if alias in self.modules and self.modules[alias]['effective']!=child['effective']:
                    raise ClosedImportError('dynamic_relative_alias_collision')
            self.modules.update(additions)
        self.modules[name]=item
        return self._spec(name,item)

def restricted_namespace(name,locations):
    if _POLICY is None:return
    wanted=tuple(map(os.path.abspath,locations))
    if name not in _POLICY.namespaces or wanted!=_POLICY.namespaces[name]:
        raise ClosedImportError('unqualified_synthetic_namespace:'+name)

def restricted_file_spec(name,file,locations=None):
    if _POLICY is None:
        return importlib.util.spec_from_file_location(name,str(file),submodule_search_locations=locations)
    with _LOCK:return _POLICY.dynamic_spec(name,file,locations)

def install_prequalified_map(modules,resources,namespaces,logical_resources):
    global _POLICY
    if not sys.flags.isolated or not sys.flags.no_site or not sys.dont_write_bytecode:
        raise ClosedImportError('fixed_I_S_B_required')
    # This API is a restriction only. Its input map cannot authorize a native
    # caller or replace independently accepted Source/fixture/Harness gates.
    with _LOCK:
        if _POLICY is not None:raise ClosedImportError('closure_cannot_be_replaced')
        candidate=ClosedSourceFinder(modules,resources,namespaces,logical_resources)
        for name,module in tuple(sys.modules.items()):
            origin=getattr(getattr(module,'__spec__',None),'origin',None)
            if origin in ('built-in','frozen'):continue
            file=getattr(module,'__file__',None)
            if not file:
                if name not in candidate.namespaces and name not in ('__main__',):
                    raise ClosedImportError('unqualified_loaded_namespace:'+name)
                continue
            item=modules.get(name)
            if item is None or os.path.abspath(file)!=item['effective']['path']:
                raise ClosedImportError('already_loaded_ambient_or_parent_module:'+name)
            exact_bytes(item['effective'])
        _POLICY=candidate
        sys.meta_path=[importlib.machinery.BuiltinImporter,importlib.machinery.FrozenImporter,candidate]
        sys.path=[]
        return candidate


class ExactResourceNode(Traversable):
    def __init__(self,logical,resources):self.logical=logical;self.resources=resources
    @property
    def name(self):return self.logical.rsplit('/',1)[-1]
    def is_file(self):return self.logical in self.resources
    def is_dir(self):return any(p.startswith(self.logical+'/')for p in self.resources)
    def iterdir(self):
        if not self.is_dir():raise NotADirectoryError(self.logical)
        prefix=self.logical+'/'
        names=sorted({p[len(prefix):].split('/',1)[0]for p in self.resources if p.startswith(prefix)})
        return iter(ExactResourceNode(prefix+n,self.resources)for n in names)
    def joinpath(self,*descendants):
        parts=[]
        for descendant in descendants:
            value=os.fspath(descendant)
            if value.startswith('/')or '\\' in value:raise ClosedImportError('resource_escape')
            for part in value.split('/'):
                if part in ('','.', '..'):raise ClosedImportError('resource_escape')
                parts.append(part)
        return ExactResourceNode(self.logical+'/'+ '/'.join(parts),self.resources)
    def open(self,mode='r',*args,**kwargs):
        if mode not in ('r','rt','rb') or not self.is_file():
            raise ClosedImportError('qualified_readonly_resource_required')
        data=exact_bytes(self.resources[self.logical])
        if mode=='rb':return io.BytesIO(data)
        encoding=kwargs.pop('encoding',args[0]if args else 'utf-8')
        errors=kwargs.pop('errors','strict')
        if kwargs or len(args)>1:raise ClosedImportError('resource_open_options_unqualified')
        return io.StringIO(data.decode(encoding or 'utf-8',errors))

class ExactResourceReader(TraversableResources):
    def __init__(self,logical,resources):self.logical=logical;self.resources=resources
    def files(self):return ExactResourceNode(self.logical,self.resources)
