import pytest

CASES = (('direct_positive', 'DELIVERED'), ('wrapped_positive', 'DELIVERED'), ('whole_set_multiple_artifacts', 'DELIVERED'), ('caller_shorter_original_budget', 'UNKNOWN'), ('slow_metadata_before_reservation', 'NO_SEND'), ('slow_metadata_terminal_publication', 'UNKNOWN'), ('slow_native_renderer_cpu', 'UNKNOWN'), ('slow_input_detach', 'UNKNOWN'), ('slow_input_cooked_exit', 'UNKNOWN'), ('slow_native_redraw', 'UNKNOWN'), ('stop_before_write', 'NO_SEND'), ('stop_during_partial_write', 'UNKNOWN'), ('revoke_before_dispatch', 'NO_SEND'), ('revoke_after_partial_output', 'UNKNOWN'), ('generation_change_before_terminal_save', 'UNKNOWN'), ('cancel_during_owned_read', 'NO_SEND'), ('cancel_during_renderer_write', 'UNKNOWN'), ('cancel_during_native_join', 'UNKNOWN'), ('late_native_done_callback', 'UNKNOWN'), ('output_close_ambiguous', 'UNKNOWN'), ('input_close_ambiguous', 'NO_SEND'), ('primary_with_output_cleanup_failure', 'UNKNOWN'), ('primary_with_input_cleanup_failure', 'NO_SEND'), ('primary_with_metadata_cleanup_failure', 'NO_SEND'), ('explicit_retry_unknown_forbidden', 'NO_SEND'), ('automatic_replay_unknown_forbidden', 'NO_SEND'), ('late_atomic_replace', 'UNKNOWN'), ('late_directory_fsync', 'UNKNOWN'))

@pytest.mark.parametrize('case,expected', CASES)
def test_original_whole_cli_contract(existing_cli_case, case, expected):
    trace = existing_cli_case(case, whole_seconds=30, renew_deadline=False,
                              automatic_retry=False, same_original_owner=True)
    assert trace['case'] == case
    assert trace['source_is_final_current20'] is True
    assert trace['independent_final_source_safety'] is True
    assert trace['fresh_specific_root_go'] is True
    assert trace['native_execution_is_actual'] is True
    assert trace['before_first_effect'] <= trace['started_monotonic']
    assert trace['deadline_monotonic'] <= trace['before_first_effect'] + 30
    assert trace['deadline_monotonic'] <= trace['caller_original_deadline']
    assert trace['closed_monotonic'] <= trace['deadline_monotonic']
    assert trace['full_physical_elapsed'] <= 30
    assert trace['clock_renewals'] == 0
    assert trace['one_original_task_and_original_appsession'] is True
    assert trace['principal_home_uid_sessiondb_inode_unchanged'] is True
    assert trace['native_five_ids_and_original_call_ledger_unchanged'] is True
    assert trace['capabilities_reminted'] == 0
    assert trace['new_writers_pumps_executors_threads'] == 0
    assert trace['input_read_and_fd_cleanup_owned_by_original_task'] is True
    assert trace['native_task_join_is_actual'] is True
    assert trace['owned_output_fds_closed_or_exact_uncertainty_retained'] is True
    assert trace['owned_input_fds_closed_or_exact_uncertainty_retained'] is True
    assert trace['original_output_ofd_flags_unchanged'] is True
    assert trace['sent_after_stop_revoke_generation_or_deadline'] == 0
    assert trace['uncertain_automatic_or_explicit_replays'] == 0
    assert trace['late_persisted_delivered'] == 0
    if expected == 'DELIVERED':
        assert trace['returned_state'] == 'DELIVERED'
        assert trace['durable_state'] == 'DELIVERED'
        assert trace['kernel_accepted_exact_original_envelope'] is True
        assert trace['human_read'] == 'UNCONFIRMED'
        assert trace['normal_native_input_renderer_restored'] is True
    elif expected == 'NO_SEND':
        assert trace['document_bytes_accepted'] == 0
        assert trace['returned_state'] != 'DELIVERED'
    else:
        assert trace['returned_state'] != 'DELIVERED'
        assert trace['durable_state'] == 'UNKNOWN'
    if case.startswith('primary_with_') or case.startswith('cancel_'):
        assert trace['original_primary_exception_object_preserved'] is True
    if 'ambiguous' in case:
        assert trace['uncertain_close_retries'] == 0
        assert trace['frontend_revoked_before_future_effect'] is True
