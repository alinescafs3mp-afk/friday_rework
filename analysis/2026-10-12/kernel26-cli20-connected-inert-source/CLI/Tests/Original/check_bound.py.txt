class CanaryRaw(io.RawIOBase):
    """Inert FileIO type/FD source seam; not an actual native descriptor."""
    def __init__(self):
        self.writes = 0; self.flushes = 0; self.monitor = True
    def writable(self): return True
    def fileno(self): return 91
    def write(self, value):
        if self.monitor:
            self.writes += 1
            raise AssertionError('original buffered write must not run')
        return len(value)
    def flush(self):
        if self.monitor:
            self.flushes += 1
            # Reproduces a genuinely slow legacy flush if it is ever invoked.
            threading.Event().wait(.04)


class FDCanary:
    """Actual nonblocking write contract simulated without opening an OS FD."""
    def __init__(self, mode='ok', cap=None, cli=None, ctx=None, kind='pipe'):
        self.mode = mode; self.cap = cap; self.cli = cli; self.ctx = ctx
        self.flags = {91: os.O_WRONLY}; self.fdflags = {91: 0}; self.live = set()
        self.stats = {91: types.SimpleNamespace(st_dev=100, st_ino=200,
                     st_mode=(stat.S_IFIFO if kind == 'pipe' else stat.S_IFCHR if kind == 'tty' else stat.S_IFREG)|0o600,
                     st_rdev=0)}
        self.written = bytearray(); self.calls = []; self.opens = 0; self.closes = 0
        self.write_calls = 0; self.blocked_until = time.monotonic()+1
        self.kind = kind; self.next_fd = 92
        self.os = types.SimpleNamespace(O_ACCMODE=os.O_ACCMODE, O_RDONLY=os.O_RDONLY,
                  O_WRONLY=os.O_WRONLY, O_NONBLOCK=os.O_NONBLOCK, O_CLOEXEC=os.O_CLOEXEC,
                  O_NOCTTY=os.O_NOCTTY, open=self.open, close=self.close, fstat=self.fstat,
                  isatty=lambda fd:self.kind=='tty', write=self.write)
        self.fcntl_api = types.SimpleNamespace(F_GETFL=OLD_FCNTL.F_GETFL, F_GETFD=OLD_FCNTL.F_GETFD,
                           FD_CLOEXEC=OLD_FCNTL.FD_CLOEXEC, fcntl=self.fcntl)
        CANARIES.append(self)
    def fstat(self, fd):
        if fd not in self.stats: raise OSError('inert missing descriptor')
        return self.stats[fd]
    def fcntl(self, fd, command):
        self.calls.append(('fcntl', fd, command))
        if command == OLD_FCNTL.F_GETFL: return self.flags[fd]
        if command == OLD_FCNTL.F_GETFD: return self.fdflags[fd]
        raise AssertionError('No F_SETFL or mutation of original descriptor allowed')
    def open(self, path, flags):
        assert path == '/proc/self/fd/91' and flags & os.O_NONBLOCK and flags & os.O_CLOEXEC and flags & os.O_NOCTTY
        assert self.flags[91] == os.O_WRONLY
        fd = self.next_fd; self.next_fd += 1
        self.flags[fd] = os.O_WRONLY|os.O_NONBLOCK; self.fdflags[fd] = OLD_FCNTL.FD_CLOEXEC
        self.stats[fd] = copy.copy(self.stats[91]); self.live.add(fd); self.opens += 1
        self.calls.append(('open', path, flags, fd))
        if self.mode == 'open_identity_drift': self.stats[fd].st_ino += 1
        if self.mode == 'missing_nonblock': self.flags[fd] &= ~os.O_NONBLOCK
        if self.mode == 'missing_cloexec': self.fdflags[fd] = 0
        if self.mode == 'cancel_on_open':
            task = asyncio.current_task(); asyncio.get_running_loop().call_soon(task.cancel)
        return fd
    def close(self, fd):
        assert fd in self.live, 'must close only exact owned canary descriptor once'
        self.live.remove(fd); self.closes += 1; self.calls.append(('close',fd))
        if self.mode=='close_error_after_close':raise OSError('inert uncertain close result')
    def write(self, fd, chunk):
        assert fd in self.live and fd != 91 and self.flags[fd] & os.O_NONBLOCK
        assert type(chunk) is bytes and 0 < len(chunk) <= output.WRITE_SLICE
        self.write_calls += 1; self.calls.append(('write',fd,len(chunk),id(asyncio.current_task())))
        if self.mode in {'slow_write','slow_flush','blocked'} or (self.mode=='partial_timeout' and self.written):
            raise BlockingIOError(errno.EAGAIN, 'inert native output backpressure')
        if self.mode=='error': raise OSError('inert native write error')
        if self.mode=='zero': return 0
        if self.mode=='bool_count': return True
        count = min(17,len(chunk)) if self.mode in {'partial_complete','partial_timeout'} else len(chunk)
        self.written.extend(chunk[:count])
        if self.mode=='revoke': self.cap.revoked=True
        if self.mode=='replace_stream': self.cli.console.file=io.StringIO()
        if self.mode=='replace_console': self.cli.console=types.SimpleNamespace(file=self.cli.console.file,_lock=threading.RLock())
        if self.mode=='replace_lifetime': self.cli._friday_cli_output=copy.copy(self.cli._friday_cli_output)
        if self.mode=='frontend_close': self.cli._friday_cli_output.close()
        if self.mode=='original_fd_drift': self.stats[91].st_ino += 1
        if self.mode=='interrupt': self.cli.agent._interrupt_requested=True
        if self.mode=='unload': self.ctx._manager._plugins.clear()
        return count
    def observation(self):
        return {'mode':self.mode,'opens':self.opens,'closes':self.closes,'live_owned_fds':sorted(self.live),
                'write_calls':self.write_calls,'accepted_bytes':len(self.written),'original_flags':self.flags[91],
                'global_flag_mutations':0,'writer_task_ids':sorted({v[3] for v in self.calls if v[0]=='write'})}


@contextlib.contextmanager
def sink(cli,cap,ctx,mode='ok',kind='pipe'):
    raw = CanaryRaw(); stream=io.TextIOWrapper(io.BufferedWriter(raw),encoding='utf-8')
    canary=FDCanary(mode,cap,cli,ctx,kind)
    original_stream=cli.console.file
    cli.console.file=stream
    output.os=canary.os;output.fcntl=canary.fcntl_api
    output.io=types.SimpleNamespace(TextIOWrapper=io.TextIOWrapper,BufferedWriter=io.BufferedWriter,
                                  FileIO=CanaryRaw,StringIO=io.StringIO)
    try:
        yield canary, raw
    finally:
        assert not canary.live, canary.observation()
        output.os=OLD_OS;output.fcntl=OLD_FCNTL;output.io=OLD_IO
        raw.monitor=False;stream.close()
        cli.console.file=original_stream

def timed(mode,outer=None,cancel=False):
    f,cap,cli,ctx,adm,host=setup()
    with sink(cli,cap,ctx,mode) as (c,raw),input_turn(cli):
        async def run():
            start=time.monotonic();events=[]
            task=asyncio.create_task(direct(cli,ctx,timeout=.12))
            def control():
                events.append({'at':time.monotonic()-start,'done':task.done()})
                if cancel:task.cancel()
            timer=asyncio.get_running_loop().call_later(.035,control)
            try:
                try:
                    receipt=await (asyncio.wait_for(task,outer) if outer is not None else task)
                    assert not cancel and outer is None
                    assert receipt['state']=='UNKNOWN'
                    state=receipt['state']
                except (asyncio.CancelledError,TimeoutError):
                    assert cancel or outer is not None
                    state='CANCELLED_AND_JOINED'
                elapsed=time.monotonic()-start
                await asyncio.sleep(0)
                assert task.done() and not c.live and c.opens==c.closes==1
                assert raw.writes==raw.flushes==0 and not cli._friday_cli_output.busy.locked()
                assert events and events[0]['done'] is False
                assert elapsed < .35, (mode,elapsed)
                assert elapsed < .16 if (cancel or outer is not None) else elapsed < .18
                TIMINGS.append({'mode':mode,'output_timeout':.12,'outer_wait_for':outer,'cancel':cancel,
                                'elapsed':elapsed,'state':state,'control':events,'accepted_bytes':len(c.written)})
                return TIMINGS[-1]
            finally:
                timer.cancel()
                if not task.done():task.cancel()
                with contextlib.suppress(asyncio.CancelledError):await task
        return asyncio.run(run())


test('slow_nonblocking_write_expires_without_loop_starvation_or_late_DELIVERED',lambda:timed('slow_write'))
