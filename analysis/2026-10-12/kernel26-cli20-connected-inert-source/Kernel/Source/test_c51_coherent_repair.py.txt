"""Coherent C51 repair independent control definitions. ALL NOT_RUN.
Actual-trace controls require final-byte Source/action admission and raw kernel
evidence. AST/metadata controls do not replace them or original32 assertions.
"""

def _actual(t,case):
    assert t['control_id']==case
    assert t['independent_final_source_acceptance_ref'] is not None
    assert t['specific_current_action_admission_ref'] is not None
    assert t['independent_raw_root_trace_ref'] is not None
    assert t['simulated'] is False and t['unknown_as_zero'] is False
    assert t['clock_reset'] is False and t['original_seconds']==120 and t['cleanup_seconds']==30

def test_c51_external_caller_survives_trace_overflow(t):
    _actual(t,'C51_TRACE_OVERFLOW_REAL_CUSTODY')
    assert t['actual_raw_trace_size']>2097152 and t['refusal_site']=='all raw files finite/owned/singlelink'
    assert t['STOP_UNCONFIRMED'] is True and t['CLI_normal_exit_calls']==0
    assert t['actual_external_caller_birth_after_CLI_terminal_ref'] is not None
    assert t['actual_caller_slot_FLOCK_after_CLI_terminal_ref'] is not None
    assert t['actual_caller_group_and_output_parent_fstat_ref'] is not None
    assert t['capacity_release'] is False and t['close_retry_count']==0

def test_c51_original_output_alias_refused_before_effect(t):
    _actual(t,'C51_OUTPUT_SLOT_ALIAS')
    assert t['intended_refusal_site']=='output disjoint slot/group/control/trace/pidfd/GO'
    assert t['actual_output_inventory_aliases_original_slot'] is True
    assert t['actual_first_birth_call_count']==0 and t['actual_original_slot_close_count']==0

def test_c51_all18_exact_negatives(t):
    _actual(t,'C51_TYPED_ALL18_NEGATIVES')
    assert len(t['negative_rows'])==18
    for row in t['negative_rows']:
        assert row['actual_code']==row['expected_code'] and row['actual_site']==row['expected_site']
        assert row['actual_refusal_raw_ref'] is not None
        assert row['unrelated_error_as_pass'] is False and row['fallback_or_retry_count']==0
        assert row['actual_owned_FD_and_private_kernel_closure_ref'] is not None
        if row['case_id']!='K120_PIDFD_OR_NAMESPACE_UNSUPPORTED':
            assert row['actual_first_birth_call_count']==0

def test_c51_four_actual_serial_cycles_fit_or_honest_fail(t):
    _actual(t,'C51_FOUR_REAL_CYCLES')
    assert t['unchanged_original_assertion_count']==32 and t['serial_owner_count']==1
    assert t['declared244_305_used_as_physical_minimum'] is False
    if t['state']=='PASS_ACTUAL_FULL32':
        assert len(t['actual_cycles'])==4
        previous=None
        for row in t['actual_cycles']:
            assert 19<=row['actual_entry_age']<=21
            assert row['actual_wait_and_normal_cleanup_raw_ref'] is not None
            if previous is not None:assert previous<=row['actual_entry_m']
            previous=row['actual_cleanup_terminal_m']
    else:assert t['state'] in ('FAIL','NOT_RUN','STOP_UNCONFIRMED')
