def same_read_pin(ref,maximum=2097152):
    require(type(ref) is dict and set(ref)=={'path','bytes','sha256'},'exact pin')
    p=Path(ref['path']);require(p.is_absolute() and '..' not in p.parts
        and str(p)==os.path.normpath(str(p)),'absolute normalized pin')
    require(type(ref['bytes']) is int and 0<ref['bytes']<=maximum,'bounded pin bytes')
    for parent in p.parents:require(stat.S_ISDIR(parent.lstat().st_mode),'pin ancestor no symlink')
    fd=os.open(p,os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK)
    try:
        before=os.fstat(fd);require(stat.S_ISREG(before.st_mode) and before.st_nlink==1
            and before.st_size==ref['bytes'],'exact regular single-link pin')
        raw=bytearray()
        while len(raw)<=ref['bytes']:
            b=os.read(fd,min(65536,ref['bytes']+1-len(raw)))
            if not b:break
            raw.extend(b)
        after=os.fstat(fd)
        require((before.st_dev,before.st_ino,before.st_size,before.st_mtime_ns,before.st_ctime_ns)
                ==(after.st_dev,after.st_ino,after.st_size,after.st_mtime_ns,after.st_ctime_ns)
                and len(raw)==ref['bytes'] and hashlib.sha256(raw).hexdigest()==ref['sha256'],
                'same-read digest and kernel identity')
        return bytes(raw)
    finally:os.close(fd)

def existing_owner_kernel_observation(spec,group_fd):
    """ONE bounded Root checkpoint, not a polling worker or authority receipt.
    Exact private kernel group is the membership source; count Linux TIDs,
    including threads, descendants and Z. Never infer unknown as zero.
    """
    group=Path(spec['native']['group']);q=os.fstat(group_fd);g=group.stat()
    require([q.st_dev,q.st_ino]==[g.st_dev,g.st_ino]==spec['native']['first_birth_group_identity'],
            'actual retained group inode at observation')
    def small(path):
        fd=os.open(path,os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK)
        try:
            b=os.read(fd,65537);require(len(b)<=65536,'bounded actual kernel observation');return b.decode()
        finally:os.close(fd)
    first={k:small(group/k) for k in ('pids.current','cgroup.events','cgroup.procs','cgroup.threads',
                                    'memory.max','memory.swap.max','pids.max','cpu.max','memory.events','pids.events','cpu.stat')}
    ids=[int(x) for x in first['cgroup.threads'].split()];require(len(ids)<=128,'actual bounded Linux TID inventory')
    require(not ({r['pid'] for r in spec['existing_owner']['protected_writers']} & set(ids)), 'protected writer excluded')
    rows=[]
    for tid in ids:
        try:
            identity=exact_birth(tid);c=small('/proc/'+str(tid)+'/cgroup').strip()
            require(c=='0::'+spec['native']['group_relative'],'only current private owned cohort')
            status=small('/proc/'+str(tid)+'/status')
            row={'tid':tid,'identity':identity,'cgroup':c,'status_raw':status}
            if identity['state']=='Z':row['zombie_no_live_probe_or_reap_credit']=True
            else:row['limits_raw']=small('/proc/'+str(tid)+'/limits')
            rows.append(row)
        except FileNotFoundError:rows.append({'tid':tid,'state':'UNKNOWN_EXITED_DURING_SAME_CHECKPOINT_NOT_ZERO'})
    last={k:small(group/k) for k in ('pids.current','cgroup.events','cgroup.threads')}
    empty=first['pids.current'].strip()==last['pids.current'].strip()=='0'
    require(not empty or (not ids and not last['cgroup.threads'].strip()
            and 'populated 0' in first['cgroup.events'] and 'populated 0' in last['cgroup.events']),
            'actual fresh kernel0 agreement, no existence/wait surrogate')
    return {'group_identity':[q.st_dev,q.st_ino],'first_raw':first,'last_raw':last,'owned_task_rows':rows,
        'actual_kernel_zero':empty,'capacity_release':False,'at_unix':time.time(),'at_monotonic':time.monotonic(),
        'unknown_as_zero':False,'Root_must_independently_pin_and_accept_this_actual_checkpoint':True}

            and existing_root_same_identity(w.root,exact_birth(os.getpid())),'one actual existing Root wait')

        while stage<3:

                     w.clock['started_at_unix']+115-time.time())

                    (time.time()-w.clock['started_at_unix']))<=2,'original dualclock/no suspend/renewal')
            events=dict(poller.poll(max(0,int(left*1000))))
            require(all(not flags&(select.POLLERR|select.POLLNVAL) for flags in events.values()),
                    'exact retained owned poll capabilities')

                    notice=json.loads(data)

                    require(selected in ('AFTER_REAL_PIDFD_FSTAT','AFTER_REAL_MAP_GO',
                                         'AFTER_REAL_INIT_WAIT','AFTER_REAL_PIPE_CLOSE')
                            and notice['site']==selected and notice['original_clock_pin']==w.spec['clock']
                            and notice['operation_already_attempted'] is True

                            'fixed site/commonclock AFTER-real-operation correlation, no EIO credit')

                            and existing_root_same_identity(exact_birth(w.stock['pid']),w.stock),

                        time.clock_gettime(time.CLOCK_BOOTTIME)-time.monotonic())

                            and abs(notice['guardian_expiry_u']-(w.clock['started_at_unix']+
                                expiry-w.clock['started_at_monotonic']))<=0.05,

                            'owned private init notification excludes protected writers')

                    except FileNotFoundError:init=None

                                =='0::'+w.spec['native']['group_relative'],'actual private owned namespace init before stop')
                        fd=os.pidfd_open(init_pid);root_fd_custody.own(w.custody,fd,'early_owned_private_init_pidfd')

                        w.private_init=init;w.private_init_fd=fd
                    else:

                                'live owned init required before earlier fstat/map cut')
                    w.events.append({'kind':'CORRELATED_REAL_OPERATION_BEFORE_EARLY_STOP','notice':notice,
                        'actual_private_kernel_sample':existing_owner_kernel_observation(w.spec,w.group_fd),

                    # retained before GO. Signal only those exact pidfds.
                    for fd in (w.private_init_fd,w.actor_fd,w.stock_fd):
                        if fd is None:continue
                        try:signal.pidfd_send_signal(fd,signal.SIGKILL)
                        except ProcessLookupError:pass
                    w.events.append({'kind':'EARLY_EXACT_OWNED_STOP','m':time.monotonic(),'u':time.time(),
                                     'original_guardian_birth':w.stock,'original_guardian_expiry_m':expiry})
                poller.unregister(w.ready_fd);notice_open=False
                root_fd_custody.closed(w.custody,w.ready_fd)

                require(len(os.read(w.timer_fd,8))==8,'actual original timer consumed once')
                if stage==0:
                    try:signal.pidfd_send_signal(w.actor_fd,signal.SIGKILL)
                    except ProcessLookupError:pass

                    except ProcessLookupError:pass
                    w.events.append({'kind':'STOCK_STOP_FINAL_MINUS10','m':time.monotonic(),'u':time.time()})
                    existing_root_set_timer(w,w.clock['started_at_monotonic']+115)

        got,status=os.waitpid(w.stock['pid'],0);require(got==w.stock['pid'],'exact direct stock wait')
        w.wait_statuses.append({'pid':got,'status':status,'kind':'direct_stock'})
        rows=[(w.actor,w.actor_fd,'adopted_actor')]
        if w.private_init is not None:rows.append((w.private_init,w.private_init_fd,'adopted_private_init'))
        for row,fd,kind in rows:
            try:live=exact_birth(row['pid'])
            except FileNotFoundError:live=None
            if live is not None and live['birth']==row['birth']:
                require(live['ppid']==os.getpid(),'exact adopted owned child belongs to Root')
                left=max(0,min(w.clock['started_at_monotonic']+115-time.monotonic(),
                               w.clock['started_at_unix']+115-time.time()))
                require(select.select([fd],[],[],left)[0]==[fd],'actual adopted owned child terminal')
                got,status=os.waitpid(row['pid'],0);require(got==row['pid'],'one exact adopted wait')
                w.wait_statuses.append({'pid':got,'status':status,'kind':kind})
            try:present=exact_birth(row['pid'])['birth']==row['birth']
            except FileNotFoundError:present=False
            require(not present,'fresh exact birth absence AFTER actual wait')
        try:present=exact_birth(w.stock['pid'])['birth']==w.stock['birth']
        except FileNotFoundError:present=False
        require(not present and not Path('/proc/self/task/'+str(os.getpid())+'/children').read_text().strip(),
                'actual fresh all direct/adopted children absent; unknown never zero')
        if w.private_init_fd is not None:root_fd_custody.closed(w.custody,w.private_init_fd)
        if notice_open:root_fd_custody.closed(w.custody,w.ready_fd)
        root_fd_custody.closed(w.custody,w.timer_fd)
        w.state='ACTUAL_WAIT_FRESH_COMPLETE';return w
    except BaseException as error:root_fd_custody.stopped(w.custody,error);raise


def existing_root_final_helper_release(w,helper,helper_pidfd,release_original_slot=True):
    """Exterior original flock release only after actual direct helper wait.
    The finalizer runs in the already-owned dedicated helper, not in Root/TUI.
    Same helper Source result is consistency; outside Root independently checks
    actual group/mount absence and all retained custody BEFORE slot close.
    """
    import select
    require(type(w)is ExistingKernel120Wait and w.state=='ACTUAL_WAIT_FRESH_COMPLETE'
            and existing_root_same_identity(w.root,exact_birth(os.getpid())), 'outside Root final release only')
    root_fd_custody.bind_existing(w.custody,helper_pidfd,'existing_final_helper_pidfd')
    try:
        h=exact_birth(helper['pid']);require(existing_root_same_identity(h,helper) and h['ppid']==os.getpid(), 'actual owned direct helper')
        info=Path('/proc/self/fdinfo/'+str(helper_pidfd)).read_text()
        require(any(x.split()==['Pid:',str(helper['pid'])] for x in info.splitlines()),'exact retained helper pidfd')
        left=max(0,min(w.clock['started_at_monotonic']+118-time.monotonic(),
                       w.clock['started_at_unix']+118-time.time()))
        if select.select([helper_pidfd],[],[],left)[0]!=[helper_pidfd]:
            import signal
            signal.pidfd_send_signal(helper_pidfd,signal.SIGKILL)
            left=max(0,min(w.clock['started_at_monotonic']+119-time.monotonic(),w.clock['started_at_unix']+119-time.time()))
            require(select.select([helper_pidfd],[],[],left)[0]==[helper_pidfd],'actual exact helper terminal before original final')
        got,status=os.waitpid(helper['pid'],0);require(got==helper['pid'] and os.waitstatus_to_exitcode(status)==0,
                                                    'actual direct helper wait success, not candidate boolean')
        try:present=exact_birth(helper['pid'])['birth']==helper['birth']
        except FileNotFoundError:present=False
        require(not present,'fresh exact helper absent after wait')
        require(not Path(w.spec['native']['group']).exists(), 'Root actual normal group removal fresh')
        output=str(Path(w.spec['record']).parent)
        require(not any(x.split()[4]==output for x in Path('/proc/self/mountinfo').read_text().splitlines()),
                'Root actual normal mount absence fresh')
        require(not Path('/proc/self/task/'+str(os.getpid())+'/children').read_text().strip(),'no remaining adopted helpers')
        root_fd_custody.closed(w.custody,helper_pidfd)
        root_fd_custody.closed(w.custody,w.group_fd)
        require(all(r['closed'] or r['fd']==w.original_slot_fd or r['label']=='exterior_evidence'
                    for r in w.custody['fds']), 'all owned effect handles closed, ambiguity retained/no release')
        require(time.monotonic()<w.clock['started_at_monotonic']+120
                and time.time()<w.clock['final_deadline_unix'],'unchanged original final before capacity release')
        # No LOCK_UN on a shared duplicate. LAST original Root handle closes once.
        if release_original_slot:
            root_fd_custody.closed(w.custody,w.original_slot_fd)
            w.capacity_release=True;w.custody['capacity_release']=True
            w.state='ROOT_OUTSIDE_HELPER_WAIT_FRESH_SLOT_CLOSED'
        else:
            require(Path('/proc/self/fdinfo/'+str(w.original_slot_fd)).read_text().find('FLOCK')>=0,'same original batch slot STILL held')
            w.state='ROOT_OUTSIDE_HELPER_WAIT_FRESH_ORIGINAL_SLOT_RETAINED'
        return {'capacity_release':release_original_slot,'actual_cohort_closure':True,'original_slot_retained':not release_original_slot,'actual_helper_wait_status':status,'clock_reset':False,
                'Root_must_independently_accept_actual_evidence':True}
    except BaseException as error:root_fd_custody.stopped(w.custody,error);raise



def existing_root_same_identity(a,b):
    """Kernel execution state is volatile, never part of birth identity."""
    fields=('pid','birth','ppid','pgid','sid')
    return type(a)is dict and type(b)is dict and all(k in a and k in b and a[k]==b[k] for k in fields)

def kernel120_record(custody,directory,name,document):
    """Already-owned outside control directory only, bounded O_EXCL evidence."""
    require(type(name)is str and name and '/' not in name and name not in ('.','..'),'fixed component')
    raw=json.dumps(document,sort_keys=True,separators=(',',':'),allow_nan=False).encode()
    require(len(raw)<=65536,'bounded original-clock/control record')
    fd=root_fd_custody.opened(custody,'control_record:'+name,name,
            os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o400,dir_fd=directory)
    try:
        view=memoryview(raw)
        while view:
            n=os.write(fd,view);require(n>0,'bounded record progress');view=view[n:]
        os.fsync(fd)
    except BaseException as error:root_fd_custody.stopped(custody,error);raise
    else:root_fd_custody.closed(custody,fd)
    os.fsync(directory)
    path=str(Path('/proc/self/fd/'+str(directory)).resolve(strict=True)/name)
    return {'path':path,'bytes':len(raw),'sha256':hashlib.sha256(raw).hexdigest()}


def kernel120_original_clock_once(spec,ref,custody,control_fd):

            and type(guard['seconds'])is int,'one original GNU120 guardian')
    parent=exact_birth(os.getppid())

            and parent['state'] not in ('Z','X'),'actual existing GNU guardian excludes protected writers')
    command=[b'/usr/bin/gnutimeout',b'-s',b'KILL',b'120s',PYTHON.encode(),b'-I',b'-B',
             guard['owner_source']['path'].encode(),ref['path'].encode(),
             str(ref['bytes']).encode(),ref['sha256'].encode()]
    require(Path('/proc/'+str(parent['pid'])+'/cmdline').read_bytes().split(b'\0')[:-1]==command,
            'fixed exterior GNU parent exact command; no invented host timer')

    guardian_fd=os.pidfd_open(parent['pid']);root_fd_custody.own(custody,guardian_fd,'exterior_evidence')

    custody['actual_root_clock_before_record']=clock

class Kernel120Clone3Refusal(OSError):
    def __init__(self,result,error,site):

def kernel120_atomic_named_child(custody,label):
    """Only stock/finalizer fixed roles, atomic CLONE_PIDFD; no fork handle gap."""
    import ctypes,signal,platform
    require(label in ('kernel120_stock','kernel120_post_terminal_helper'),'two fixed existing-owner roles')
    require(platform.machine()=='x86_64','reviewed clone3 ABI, no fallback')
    class CloneArgs(ctypes.Structure):
        _fields_=[(key,ctypes.c_uint64) for key in
            ('flags','pidfd','child_tid','parent_tid','exit_signal','stack','stack_size','tls',
             'set_tid','set_tid_size','cgroup')]
    fd=ctypes.c_int(-1);args=CloneArgs();args.flags=0x1000
    args.pidfd=ctypes.addressof(fd);args.exit_signal=signal.SIGCHLD
    libc=ctypes.CDLL(None,use_errno=True);libc.syscall.restype=ctypes.c_long
    pid=libc.syscall(ctypes.c_long(435),ctypes.byref(args),ctypes.c_size_t(88))
    if pid<0:
        error=ctypes.get_errno()
        custody.setdefault('actual_clone3_refusals',[]).append(
            {'syscall_result':int(pid),'kernel_errno':error,'site':'candidate_clone3_namespace',
             'actual_role':label,'at_m':time.monotonic(),'at_u':time.time(),'retry_count':0})
        raise Kernel120Clone3Refusal(int(pid),error,'candidate_clone3_namespace')
    if pid>0:
        root_fd_custody.own(custody,fd.value,label+':pidfd') # FIRST after successful acquisition.
        custody.setdefault('birth_attempts',[]).append({'pid':int(pid),'pidfd':fd.value,'role':label})
    return int(pid),fd.value


def kernel120_child_bootstrap(root,root_pidfd,boot_read,keep,clock):
    """Inherited descriptors only; no FD-transfer framework or new topology."""
    import ctypes,signal,select
    require(os.getppid()==root['pid'] and existing_root_same_identity(exact_birth(root['pid']),root),
            'same actual exterior parent before child bootstrap')
    libc=ctypes.CDLL(None,use_errno=True)
    require(libc.prctl(1,signal.SIGKILL,0,0,0)==0 and not select.select([root_pidfd],[],[],0)[0],
            'actual retained parent pidfd/PDEATH before child work')
    left=max(0,min(clock['started_at_monotonic']+90-time.monotonic(),clock['useful_deadline_unix']-time.time()))
    require(select.select([boot_read,root_pidfd],[],[],left)[0]==[boot_read]
            and os.read(boot_read,1)==b'G','one existing inherited bootstrap gate')
    for entry in list(Path('/proc/self/fd').iterdir()):
        fd=int(entry.name)
        if fd>2 and fd not in keep:
            os.close(fd) # once; any uncertainty aborts child, Root retains originals.
    for fd in keep:os.set_inheritable(fd,True)
    require(os.getppid()==root['pid'],'same parent after descriptor hygiene')

def kernel120_launch_stock_before_opaque(spec,clock,custody,control_fd,timer_fd):

    actor=kernel120_read_actor_birth(ready_r,stock_fd,timer_fd,clock)

           spec['existing_owner']['group_fd'],spec['existing_owner']['slot_fd'],ready_r,go_w,timer_fd=timer_fd)

def kernel120_post_terminal_helper(w,output_fd,output_fds,evidence_fd,evidence_identity):
    """Fork-inherited post-terminal helper: NEVER waits already reaped stock.
    Root alone owns stock/adopted waits. Helper normal cleanup alone owns its
    inherited output refs; Root closes its output refs BEFORE helper GO.
    """
    import ctypes
    require(w.state=='ACTUAL_WAIT_FRESH_COMPLETE' and w.wait_statuses[0]['kind']=='direct_stock',
            'actual in-process Root wait state; no candidate JSON substitute')
    existing_root_provider_caps();custody={'fds':[],'capacity_release':False,'STOP_UNCONFIRMED':False}
    for fd,label in ((w.group_fd,'group'),(w.original_slot_fd,'helper_slot_duplicate'),
                     (evidence_fd,'exterior_evidence')):root_fd_custody.bind_existing(custody,fd,label)
    for fd in output_fds:root_fd_custody.bind_existing(custody,fd,'helper_output')
    for row in (w.stock,w.actor):
        try:present=exact_birth(row['pid'])['birth']==row['birth']
        except FileNotFoundError:present=False
        require(not present,'actual stock/actor absent BEFORE post-terminal cleanup; no second wait')
    require(not Path('/proc/self/task/'+str(os.getpid())+'/children').read_text().strip(),'helper childless')
    group=Path(w.spec['native']['group']);q=os.fstat(w.group_fd);g=group.stat()
    require([q.st_dev,q.st_ino]==[g.st_dev,g.st_ino]==w.spec['native']['first_birth_group_identity'],'same retained group')
    before=protected_snapshot(w.spec,group)
    sample=existing_owner_kernel_observation(w.spec,w.group_fd);require(sample['actual_kernel_zero'],'fresh after-wait kernel0')
    require(time.monotonic()+5<w.clock['started_at_monotonic']+120 and time.time()+5<w.clock['final_deadline_unix'],
            'original cleanup reserve; no helper timer renewal')
    output=Path(w.spec['record']).parent;q=os.fstat(output_fd);g=output.stat()
    require(stat.S_ISDIR(q.st_mode) and [q.st_dev,q.st_ino]==[g.st_dev,g.st_ino]
            ==w.spec['qualification_output_identity'],'same retained private output')
    mounts=[x for x in Path('/proc/self/mountinfo').read_text().splitlines() if x.split()[4]==str(output)]
    require(len(mounts)==1 and ' - tmpfs ' in mounts[0]
            and int(mounts[0].split()[0])==w.spec['qualification_output_mount_id'],'same exact tmpfs mount')
    actual=set()
    for entry in Path('/proc/self/fd').iterdir():
        try:p=Path(os.readlink(entry))
        except FileNotFoundError:continue
        if p==output or p.is_relative_to(output):actual.add(int(entry.name))
    require(actual==set(output_fds),'all helper output refs known; Root originals closed before helper GO')
    for fd,identity in output_fds.items():
        q=os.fstat(fd);require([q.st_dev,q.st_ino,stat.S_IFMT(q.st_mode)]==identity,'exact inherited output handle')
        root_fd_custody.closed(custody,fd)
    require(ctypes.CDLL(None,use_errno=True).umount2(os.fsencode(output),0)==0,'normal umount0; no lazy detach')
    require(not any(x.split()[4]==str(output) for x in Path('/proc/self/mountinfo').read_text().splitlines()),'fresh normal mount absence')
    group.rmdir();require(not group.exists(),'normal empty same group removal')
    require(protected_snapshot(w.spec,group)==before,'same four protected writers/controllers')
    root_fd_custody.closed(custody,w.group_fd)
    kernel120_record(custody,evidence_fd,'post-terminal-helper.json',
        {'actual_kernel':sample,'Root_direct_and_adopted_waits':w.wait_statuses,'normal_umount_flags':0,
         'capacity_release':False,'original_slot_retained_by_Root':True,'clock_pin':w.spec['clock']})
    root_fd_custody.closed(custody,evidence_fd)
    root_fd_custody.closed(custody,w.original_slot_fd) # child duplicate only, no LOCK_UN.
    require(time.monotonic()<w.clock['started_at_monotonic']+118 and time.time()<w.clock['started_at_unix']+118,
            'fixed helper closure before original final reserve')
    return 0


def kernel120_launch_post_terminal_helper(w,output_fd,output_fds,evidence_fd,evidence_identity):
    import select
    require(w.state=='ACTUAL_WAIT_FRESH_COMPLETE','Root actual wait BEFORE helper creation')
    kernel120_verify_output_inventory(w.spec,w.phase_context,w.custody)
    root_wait={'stock':w.stock,'actor':w.actor,'actual_wait_statuses':w.wait_statuses,'fresh_stock_actor_absent':True,
        'original_entry_timer_arm':w.custody['original_entry_timer_arm'],'before_opaque_events':w.events,'actual_original_case_entry':w.actual_case_entry}
    w.raw_trace=existing_root_capture_private_trace(w.spec,output_fd,evidence_fd,evidence_identity,
           root_wait,w.clock['started_at_monotonic']+120,w.clock['final_deadline_unix'],w.custody)
    for fd in (w.stock_fd,w.actor_fd):root_fd_custody.closed(w.custody,fd)
    boot_r,boot_w=kernel120_pipe(w.custody,'helper_bootstrap')
    root=exact_birth(os.getpid());parent_fd=os.pidfd_open(root['pid'])
    root_fd_custody.own(w.custody,parent_fd,'helper_parent_pidfd')
    pid,pidfd=kernel120_atomic_named_child(w.custody,'kernel120_post_terminal_helper')
    if pid==0:
        try:
            keep=set(output_fds)|{w.group_fd,w.original_slot_fd,evidence_fd}
            # Unlike stock bootstrap, helper uses original final, never useful+90.
            left=max(0,min(w.clock['started_at_monotonic']+118-time.monotonic(),w.clock['started_at_unix']+118-time.time()))
            import ctypes,signal
            require(ctypes.CDLL(None,use_errno=True).prctl(1,signal.SIGKILL,0,0,0)==0
                    and not select.select([parent_fd],[],[],0)[0],'helper actual parent death')
            require(select.select([boot_r,parent_fd],[],[],left)[0]==[boot_r] and os.read(boot_r,1)==b'G',
                    'helper GO after Root output originals closed')
            for entry in list(Path('/proc/self/fd').iterdir()):
                fd=int(entry.name)
                if fd>2 and fd not in keep:os.close(fd)
            os._exit(kernel120_post_terminal_helper(w,output_fd,output_fds,evidence_fd,evidence_identity))
        except BaseException:os._exit(125)
    helper=exact_birth(pid);require(helper['ppid']==root['pid'],'actual direct post-terminal helper')
    for fd in (boot_r,parent_fd):root_fd_custody.closed(w.custody,fd)
    kernel120_verify_output_inventory(w.spec,w.phase_context,w.custody)
    for fd in output_fds:root_fd_custody.closed(w.custody,fd) # Root refs BEFORE helper GO/normal umount.
    require(os.write(boot_w,b'G')==1,'one helper GO after actual Root output closes')
    root_fd_custody.closed(w.custody,boot_w)
    w.helper=helper
    return existing_root_final_helper_release(w,helper,pidfd,release_original_slot=False)


def kernel120_verify_output_inventory(spec,context,custody=None):
    """Actual Root inventory before effects AND before original output closes."""
    output=Path(spec['record']).parent
    require(output.is_absolute() and output.resolve(strict=True)==output,'exact output path')
    q=output.lstat();fd=context['output_fd'];rows=context['output_owned_fds']
    require(type(rows)is dict and 0<len(rows)<=32 and fd in rows
            and all(type(k)is int and k>2 for k in rows),'complete actual Root output inventory')
    require(stat.S_ISDIR(q.st_mode) and [q.st_dev,q.st_ino]==spec['qualification_output_identity'],
            'actual Root output path identity')
    mounts=[r.split() for r in Path('/proc/self/mountinfo').read_text().splitlines()
            if r.split()[4]==str(output)]
    require(len(mounts)==1 and mounts[0][mounts[0].index('-')+1]=='tmpfs'
            and int(mounts[0][0])==spec['qualification_output_mount_id'],'actual Root exact tmpfs mount')
    reserved={spec['existing_owner']['group_fd'],spec['existing_owner']['slot_fd'],
              context['control_dir_fd'],context['trace_dir_fd']}
    if custody is not None:
        reserved.update(r['fd'] for r in custody['fds'] if not r['closed']
                        and r['fd'] not in rows and r['label']!='Root_output')
    require(not set(rows)&reserved,'output disjoint slot/group/control/trace/pidfd/GO')
    actual=set()
    for entry in Path('/proc/self/fd').iterdir():
        try:info=Path('/proc/self/fdinfo/'+entry.name).read_text()
        except FileNotFoundError:continue
        ids=[r.split()[1] for r in info.splitlines() if r.startswith('mnt_id:')]
        require(len(ids)==1,'exact actual Root fd mount identity')
        if int(ids[0])==spec['qualification_output_mount_id']:actual.add(int(entry.name))
    require(actual==set(rows),'complete actual Root output mount FD inventory')
    for handle,identity in rows.items():
        a=os.fstat(handle);path=Path(os.readlink('/proc/self/fd/'+str(handle)));b=path.lstat()
        require(path==output or path.is_relative_to(output),'Root output handle exact path')
        require([a.st_dev,a.st_ino,stat.S_IFMT(a.st_mode)]==identity
                and [a.st_dev,a.st_ino,stat.S_IFMT(a.st_mode)]
                ==[b.st_dev,b.st_ino,stat.S_IFMT(b.st_mode)]
                and a.st_dev==q.st_dev and not stat.S_ISLNK(b.st_mode),'same Root output FD/path identity')
    require([os.fstat(fd).st_dev,os.fstat(fd).st_ino]==[q.st_dev,q.st_ino],'Root output directory identity')
    return {'output_identity':[q.st_dev,q.st_ino],'mount_id':spec['qualification_output_mount_id'],
            'exact_fds':sorted(actual),'verified_m':time.monotonic(),'verified_u':time.time()}

def kernel120_terminal_failure_hold(context,custody,clock,error):
    """No voluntary normal exit, release, retry or refork after STOP."""
    import select
    root_fd_custody.stopped(custody,error)
    custody['terminal_failure_hold_entered']={'m':time.monotonic(),'u':time.time()}
    while True:
        left=min(clock['started_at_monotonic']+120-time.monotonic(),
                 clock['final_deadline_unix']-time.time())
        if left<=0:raise error # abnormal exit; actual independent holder retains its originals.
        select.select([],[],[],left) # unchanged finite GNU original boundary.


def kernel120_wait_original_age(clock,age):
    import select
    require(age in (19,45.001,90.001),'fixed original temporal boundaries')
    while True:
        m=time.monotonic();u=time.time()
        require(m<clock['started_at_monotonic']+120 and u<clock['final_deadline_unix']
                and abs((m-clock['started_at_monotonic'])-(u-clock['started_at_unix']))<=2,
                'unchanged actual dualclock temporal boundary')
        left=max(clock['started_at_monotonic']+age-m,clock['started_at_unix']+age-u)
        if left<=0:return
        select.select([],[],[],left)


class Kernel120Refusal(ValueError):
    def __init__(self,code,site,detail):
        self.code=code;self.site=site
        super().__init__(code+': '+site+': '+detail)


def kernel120_need(ok,code,site,detail):
    if not ok:raise Kernel120Refusal(code,site,detail)


def kernel120_first_birth_preflight(spec,expected,clock):

    check=kernel120_need

          'NO_CURRENT_SOURCE_ACCEPTANCE','candidate_source_admission','exact Source before effects')

          'NO_ACTUAL_ROOT_ADMISSION','candidate_action_admission','actual action before effects')

          'CLOCK_RENEWAL','candidate_original_clock','one original common pin')

          'CLOCK_RENEWAL','candidate_original_clock','exact same-read common clock')

          and time.time()+30<clock['final_deadline_unix'],
          'FINAL_RESERVE_SHORT','candidate_final_reserve','original30 final remains')
    check(time.monotonic()+45<clock['started_at_monotonic']+90
          and time.time()+45<clock['useful_deadline_unix'],
          'USEFUL_RESERVE_SHORT','candidate_useful_reserve','original45 BEFORE actual first birth')
    check(type(spec['native']['stock_guard_seconds'])is int
          and 45<spec['native']['stock_guard_seconds']<=90,
          'RELATIVE1800_GUARD','candidate_stock_duration','finite original child guard')
    try:existing_root_provider_caps()
    except ValueError as error:
        if str(error)!='existing Root hard caps before reads':raise
        raise Kernel120Refusal('ROLE_HARD_CAP_MISSING','candidate_role_caps',str(error)) from error
    kernel120_validate_actual_pre_entry_role(expected['pre_entry_role_birth'],spec)
    kernel120_validate_actual_guardian_remaining(expected['before_entry_guardian'],spec,clock)
    try:return existing_owner_admission(spec)
    except ValueError as error:
        codes={'retained Root private group FD/inode':('FOREIGN_GROUP_FD','candidate_group_inode'),
          'whole controllers effective BEFORE birth':('CONTROLLER_BOUND','candidate_controllers'),
          'outside actor is excluded from whole native group':('SHARED_WRITER_SCOPE','candidate_scope'),

          'effective bounded outside filesystem before candidate load':('OUTPUT_NOT_BOUNDED','candidate_output_vfs'),
          'actual private output mount':('OUTPUT_NOT_BOUNDED','candidate_output_mount')}
        if str(error) not in codes:raise
        code,site=codes[str(error)]
        if code=='CONTROLLER_BOUND':
            values={k:(Path(spec['native']['group'])/k).read_text().strip() for k in CONTROLLERS}
            bad=[k for k in CONTROLLERS if values[k]!=CONTROLLERS[k]]
            table={'memory.max':'UNBOUNDED_MEMORY','memory.swap.max':'NONZERO_SWAP',
                   'pids.max':'UNBOUNDED_TASKS','cpu.max':'UNBOUNDED_CPU'}
            if len(bad)!=1:raise
            code=table[bad[0]];site='candidate_controller:'+bad[0]
        raise Kernel120Refusal(code,site,str(error)) from error
