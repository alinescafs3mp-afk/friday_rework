"""Trusted private stock-guard negative; no Source import, fork or connector."""
import json
import os
from pathlib import Path
import resource
import select
import signal

W = Path(__file__).resolve().parent
assert resource.getrlimit(resource.RLIMIT_AS) == (256 << 20, 256 << 20)
assert resource.getrlimit(resource.RLIMIT_CPU) == (30, 30)
assert resource.getrlimit(resource.RLIMIT_FSIZE) == (64 << 20, 64 << 20)
assert resource.getrlimit(resource.RLIMIT_CORE) == (0, 0)
assert sorted(os.sched_getaffinity(0)) == [0, 1]
signal.signal(signal.SIGTERM, signal.SIG_IGN)
signal.signal(signal.SIGALRM, signal.SIG_IGN)
data = {'pid': os.getpid(), 'birth': Path('/proc/self/stat').read_text().rsplit(')', 1)[1].split()[19],
        'pgid': os.getpgrp(), 'ppid': os.getppid(), 'TERM': 'IGNORED', 'ALRM': 'IGNORED',
        'Source_import': False, 'fork': False, 'caps': {'AS': list(resource.getrlimit(resource.RLIMIT_AS)),
        'CPU': list(resource.getrlimit(resource.RLIMIT_CPU)), 'FSIZE': list(resource.getrlimit(resource.RLIMIT_FSIZE)),
        'CORE': list(resource.getrlimit(resource.RLIMIT_CORE)), 'affinity': sorted(os.sched_getaffinity(0))}}
fd = os.open(W/'GUARD-NEGATIVE-TRACED-CALLER.json', os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o400)
with os.fdopen(fd, 'wb') as f:
    f.write((json.dumps(data, sort_keys=True) + '\n').encode()); f.flush(); os.fsync(f.fileno())
while True:
    select.select([], [], [], .05)
