def canonical_variant(block=False, alias_fault=None, grant_fault=None):
    """Exact candidate bytes, scoped private path mapping and explicit offline hook.
    Selector and transport compiled original bytes; no rewritten sender variant.
    The delivery hook is synthetic and grants no actual transport/native credit.
    Blocking hook runs at exact transport exec import site before its code executes.
    """
    import sys
    import builtins
    label=('block' if block else 'positive') if alias_fault is None and grant_fault is None else str(alias_fault or grant_fault)
    d=OUT/('canonical-'+label);d.mkdir(mode=0o700)
    custody=d/'custody';custody.mkdir(mode=0o700)
    sp=_C26_BIND['selector'];tp=_C26_BIND['transport']
    mappings={sp['path']:HERE/'result_selector.py',tp['path']:HERE/'result_rpc.py'}
    trace=d/'ALIAS-TRACE.jsonl'
    def note(row):
        fd=os.open(trace,os.O_WRONLY|os.O_CREAT|os.O_APPEND|os.O_NOFOLLOW,0o600)
        with os.fdopen(fd,'wb') as f:f.write((json.dumps(row,sort_keys=True)+'\n').encode());f.flush();os.fsync(f.fileno())
    if alias_fault:
        side,kind=alias_fault.split(':',1);literal=sp['path'] if side=='selector' else tp['path']
        expected=HERE/('result_selector.py' if side=='selector' else 'result_rpc.py')
        target=d/(side+'-alias.py')
        if kind=='missing':mappings[literal]=target
        elif kind=='unmapped':del mappings[literal]
        elif kind=='wrong':mappings[literal]=HERE/('result_rpc.py' if side=='selector' else 'result_selector.py')
        elif kind=='outside':mappings[literal]=Path('/tmp/C26-OUTSIDE-SCOPE-MUST-NOT-BE-READ')
        else:
            raw=expected.read_bytes();target.write_bytes(raw);target.chmod(0o600)
            if kind=='drift':target.write_bytes(raw+b'\n# intentional private fixture drift\n')
            elif kind=='unsafe':target.chmod(0o604)
            elif kind=='symlink':target.unlink();target.symlink_to(expected)
            elif kind=='hardlink':os.link(target,d/(side+'-extra-link.py'))
            else:raise AssertionError('unknown alias negative')
            mappings[literal]=target
    Base=type(Path())
    class AliasPath(Base):
        def __new__(cls,*parts):
            p=Base(*parts)
            if str(p) in (sp['path'],tp['path']):
                if str(p) not in mappings:raise AssertionError('fixture runtime alias mapping missing')
                mapped=Base(mappings[str(p)])
                assert mapped.is_relative_to(W),'fixture alias outside owned package refused'
                note({'kind':'mapped_alias','canonical':str(p),'private':str(mapped)})
                p=mapped
            return super().__new__(cls,p)
    real_exec=builtins.exec

    def fixture_exec(code,g=None,l=None):
        assert isinstance(g,dict) and l is None
        name=source_by_path.get(g.get('__file__'))
        assert name in ('result_selector.py','result_rpc.py'),'unexpected fixture exec target'
        assert code==compile(_C29_BYTES[name],g['__file__'],'exec'),'compiled source differs from exact reviewed bytes'
        note({'kind':'exact_source_exec_site','source':name,'sha256':hashlib.sha256(_C29_BYTES[name]).hexdigest(),'original_compiled_code':True})
        if name=='result_rpc.py' and block:
            note({'kind':'explicit_synthetic_import_site_flock_hook','transport_source_rewritten':False})
            a=os.open(d/'lock',os.O_RDWR|os.O_CREAT,0o600);b=os.open(d/'lock',os.O_RDWR)
            fcntl.flock(a,fcntl.LOCK_EX);fcntl.flock(b,fcntl.LOCK_EX)
            raise AssertionError('blocking fixture lock unexpectedly returned')
        real_exec(code,g)
        if name=='result_selector.py':g['Path']=AliasPath;g['exec']=fixture_exec
        else:

            g['deliver_in_worker']=offline_delivery
    g={'source':copy.deepcopy(tp),'event':'SOL-CANONICAL','assignment':'OFFLINE','generation':'1','custody_dir':str(custody),
       'notice':{'path':str(d/'notice')},'dependencies':{'sender':{'path':str(d/'sender')},'recipient':{'path':str(d/'recipient')}}}

    elif grant_fault=='source_sha':g['source']['sha256']='0'*64

    elif grant_fault=='notice_path':g['notice']['path']=str(d/'wrong-notice')

    ns={'__name__':'private_offline_canonical','__file__':str(candidate)}
    real_exec(compile(built,str(candidate),'exec'),ns)
    ns['Path']=AliasPath;ns['exec']=fixture_exec
