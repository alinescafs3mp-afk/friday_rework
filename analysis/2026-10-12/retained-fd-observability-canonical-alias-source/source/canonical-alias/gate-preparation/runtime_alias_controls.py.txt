def host_owned_cleanup(label, end):
    # Restore fault injections before touching exact own custody. A failed
    # Source assertion remains FAIL even if this external host cleanup succeeds.
    os.waitpid=REAL_WAIT;os.close=REAL_CLOSE;socket.socketpair=REAL_PAIR
    socket.socket.close=REAL_SOCKET_CLOSE;signal.pidfd_send_signal=REAL_PIDFD_SIGNAL
    os.fork=REAL_FORK;S.birth=REAL_BIRTH
    observations=[]
    for pid,held in list(S.PENDING.items()):
        if not isinstance(held,dict) or not held.get('birth'):
            observations.append({'pid':pid,'status':'UNKNOWN_MISSING_EXACT_BIRTH'});continue
        ticks=held['birth'];fd=held.get('pidfd');reaped=held.get('reaped',False)
        status=held.get('wait_status');errors=[]
        if not reaped:
            try:
                if S.birth(pid)==ticks and fd is not None:
                    REAL_PIDFD_SIGNAL(fd,signal.SIGKILL)
            except FileNotFoundError:pass
            except ProcessLookupError:pass
            except BaseException as exc:errors.append({'phase':'signal','class':type(exc).__name__})
            until=min(time.monotonic()+1,end)
            while time.monotonic()<until:
                try:got,st=REAL_WAIT(pid,os.WNOHANG)
                except BaseException as exc:
                    errors.append({'phase':'direct_wait','class':type(exc).__name__,'errno':getattr(exc,'errno',None)});break
                if got==pid:reaped=True;status=st;break
                select.select([],[],[],.01)
        try:absent=S.birth(pid)!=ticks
        except FileNotFoundError:absent=True
        except BaseException as exc:absent=False;errors.append({'phase':'birth','class':type(exc).__name__})
        closed=False
        if reaped and absent:
            held.update(reaped=True,wait_status=status)
            try:closed=S.reconcile_pending(pid)['status']=='STOP_CONFIRMED'
            except BaseException as exc:errors.append({'phase':'release','class':type(exc).__name__})
        observations.append({'pid':pid,'birth':ticks,'actual_direct_wait':reaped,
            'fresh_exact_birth_absent':absent,'closed':closed,'direct_wait_returncode':os.waitstatus_to_exitcode(status) if status is not None else None,
            'errors':errors,'Source_failure_not_promoted':True})
    children=Path('/proc/self/task/'+str(os.getpid())+'/children').read_text().strip()
    row={'case':label,'exact_owned':observations,'pending_empty':not S.PENDING,
         'children_at_end':children,'actual_host_closure':not S.PENDING and not children}
