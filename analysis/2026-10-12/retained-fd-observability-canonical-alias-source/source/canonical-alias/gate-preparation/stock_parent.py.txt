def save(name, value):
    fd = os.open(W/name, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o400)
    with os.fdopen(fd, 'wb') as f:
        f.write((json.dumps(value, sort_keys=True, indent=2) + '\n').encode()); f.flush(); os.fsync(f.fileno())

def birth(pid):
    try: return Path('/proc/'+str(pid)+'/stat').read_text().rsplit(')', 1)[1].split()[19]
    except FileNotFoundError: return None

def fd_count(root):
    try: return len(list((root/'fd').iterdir()))
    except (PermissionError, FileNotFoundError): return 'UNKNOWN_PROC_ACCESS_OR_EXIT'

def info(pid):
    ticks = birth(pid)
    if ticks is None: return None
    root = Path('/proc')/str(pid)
    try:
        limits = {line.split('  ')[0]: line.strip() for line in (root/'limits').read_text().splitlines()
                  if line.startswith(('Max address space', 'Max cpu time', 'Max file size', 'Max core file size'))}
        status = (root/'status').read_text().splitlines()
        return {'pid': pid, 'birth': ticks, 'ppid': int(next(x.split(':')[1] for x in status if x.startswith('PPid:'))),
                'pgrp': os.getpgid(pid), 'session': os.getsid(pid), 'affinity': sorted(os.sched_getaffinity(pid)),
                'limits': limits, 'RSS': next((x for x in status if x.startswith('VmRSS:')), 'UNKNOWN'),
                'fd_count': fd_count(root)}
    except (FileNotFoundError, ProcessLookupError): return None
    except PermissionError: return {'pid': pid, 'birth': ticks, 'metadata': 'UNKNOWN_PERMISSION_DENIED'}

def children(pid):
    try: return [int(x) for x in Path('/proc/'+str(pid)+'/task/'+str(pid)+'/children').read_text().split()]
    except (FileNotFoundError, PermissionError): return []

def fd_set(): return sorted(os.listdir('/proc/self/fd'))

def pinned(key):
    p = READY[key]; raw = Path(p['path']).read_bytes()
    assert len(raw) == p['bytes'] and hashlib.sha256(raw).hexdigest() == p['sha256']
    return p

def run(name, host_key, guard_seconds, source):
    host = pinned(host_key)
    remaining = C['useful_deadline_monotonic'] - time.monotonic()
    assert remaining > guard_seconds+2 and (not source or remaining > 120)
    for p in list(READY['source_pins'].values())+list(READY['full_input_readiness'].values()):
        raw = Path(p['path']).read_bytes()
        assert len(raw) == p['bytes'] and hashlib.sha256(raw).hexdigest() == p['sha256']
    baseline = fd_set(); start = time.monotonic()
    cmd = ['prlimit', '--as=268435456:268435456', '--cpu=30:30', '--fsize=67108864:67108864', '--core=0:0',
           'taskset', '-c', '0,1', 'timeout', '--signal=TERM', '--kill-after=2s', str(guard_seconds)+'s', PY, '-B', host['path']]
    env = os.environ.copy()
    env.update(C29_ROOT_GO_REF=READY['authority']['path'], C29_ROOT_GO_SHA256=READY['authority']['sha256'],
               C26_WHOLE_OUT=('alias-controls' if name=='alias22' else 'execution'), C26_COMPAT_OUT='execution-compat81', PYTHONDONTWRITEBYTECODE='1')
    known = {}; adopted = []; unexpected = []
    with open(W/(name+'.stdout'), 'xb', buffering=0) as out, open(W/(name+'.stderr'), 'xb', buffering=0) as err:
        p = subprocess.Popen(cmd, cwd=W, env=env, stdin=subprocess.DEVNULL, stdout=out, stderr=err, start_new_session=True)
        timeout_birth = birth(p.pid)
        assert timeout_birth is not None
        save(name+'-START.json', {'command': cmd, 'timeout_pid': p.pid, 'timeout_birth': timeout_birth,
             'host': host, 'remaining_useful': remaining, 'Source_imported': False, 'caps_before_stock_timeout': True})
        until = min(start+guard_seconds+5, C['useful_deadline_monotonic'])
        while p.poll() is None and time.monotonic() < until:
            todo = [p.pid] + children(os.getpid()); visited = set()
            while todo:
                pid = todo.pop()
                if pid in visited: continue
                visited.add(pid); row = info(pid)
                if row:
                    known[(pid, row['birth'])] = row
                    todo.extend(children(pid))
            select.select([], [], [], .01)
        if p.poll() is None:
            if birth(p.pid) == timeout_birth and os.getpgid(p.pid) == p.pid:
                os.killpg(p.pid, signal.SIGKILL)
            unexpected.append('STOCK_GUARD_DID_NOT_FINISH_IN_BOUND')
        returncode = p.wait(timeout=2)
    # GNU timeout's KILL negative kills the whole fresh group, including itself.
    # The finite subreaper directly waits the exact adopted caller, never broad kills.
    for pid in children(os.getpid()):
        ticks = birth(pid)
        assert (pid, ticks) in known, ('UNKNOWN_ADOPTED_CHILD', pid, ticks)
        deadline = min(time.monotonic()+2, C['final_deadline_monotonic'])
        waited = False
        while time.monotonic() < deadline:
            got, status = os.waitpid(pid, os.WNOHANG)
            if got:
                waited = True; adopted.append({'pid': pid, 'birth': ticks, 'actual_direct_wait': True,
                    'returncode': os.waitstatus_to_exitcode(status), 'fresh_birth_absent': birth(pid) != ticks}); break
            select.select([], [], [], .01)
        assert waited, ('STOP_UNCONFIRMED', pid, ticks)
    closure = [{'pid': pid, 'birth': ticks, 'fresh_birth_absent': birth(pid) != ticks, 'last_observation': v}
               for (pid, ticks), v in known.items()]
    assert all(x['fresh_birth_absent'] for x in closure)
    assert not children(os.getpid())
    assert baseline == fd_set(), ('PARENT_FD_LEAK', baseline, fd_set())
    value = {'phase': name, 'command': cmd, 'host': host, 'returncode': returncode,
             'elapsed_seconds': time.monotonic()-start, 'stock_timeout_actual_direct_wait': True,
             'timeout_pid': p.pid, 'timeout_birth': timeout_birth, 'timeout_fresh_absent': birth(p.pid) != timeout_birth,
             'known_cohort_fresh_closed': closure, 'adopted_directwait': adopted, 'unexpected': unexpected,
             'children': children(os.getpid()), 'parent_FD_baseline_restored': True,
             'original_clock_not_reset': True, 'remaining_useful': C['useful_deadline_monotonic']-time.monotonic()}
    save(name+'-RETURN.json', value)
    if source:
        assert returncode == 0 and not unexpected, value
    else:
        assert returncode in (137,-signal.SIGKILL) and not unexpected, value # existing stock uutils wrapper137; exact adopted caller must still be SIGKILL
        caller = json.loads((W/'GUARD-NEGATIVE-TRACED-CALLER.json').read_bytes())
        assert any(x['pid'] == caller['pid'] and x['birth'] == caller['birth'] and x['returncode'] == -signal.SIGKILL for x in adopted)
        assert 2.8 <= value['elapsed_seconds'] < 6
    return value
