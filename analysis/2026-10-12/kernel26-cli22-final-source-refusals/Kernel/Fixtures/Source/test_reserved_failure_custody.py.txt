"""Producer-shaped reserved failure controls Source, NOT_RUN/NOT_ADMITTED."""
import copy
import json

def reserved_owned():
    return {'known':[],'direct_pids':[91],'unknown_spawn':True,'ambiguous':{},
            'output_bytes':0,'live_labels':['go_r','go_w','ready_r','ready_w'],
            'pending_receipt':False}

def rejected(fn,value):
    try:fn(value)
    except AssertionError:return
    raise AssertionError('uncertain reserved observation must not gain normal credit')

def test_reserved_direct_pidfd_open_failure(evidence):
    value=reserved_owned()
    assert evidence.failure_owned_state(value)==[]
    rejected(evidence.owned_state,value)
    assert value['known']==[] and value['direct_pids']==[91] and value['unknown_spawn'] is True

def test_reserved_unbound_direct_requires_unknown(evidence):
    value=reserved_owned();value['unknown_spawn']=False
    rejected(evidence.failure_owned_state,value)

def test_reserved_strict_read_charge_overshoot(evidence):
    value=reserved_owned();value['direct_pids']=[];value['unknown_spawn']=False
    value['output_bytes']=33554432+131072
    assert evidence.failure_owned_state(value)==[]
    rejected(evidence.owned_state,value)
    value['output_bytes']+=1;rejected(evidence.failure_owned_state,value)

def test_reserved_consumer_preserves_unknown_and_no_release(evidence):
    value={'schema':'friday.causal.reserved-failure.v1','normal_evidence_complete':False,
           'capacity_release':False,'actual_primary_exception':{'class':'OSError','detail':'pidfd_open fake'},
           'kernel':{'actual':'FAKE_ONLY'},'owned':reserved_owned()}
    old=evidence.same_read
    evidence.same_read=lambda fd,pin:json.dumps(value).encode()
    try:
        result=evidence.failure_evidence(3,{'memory':'FAKE_ONLY'})
        assert result['owned']==value['owned'] and result['capacity_release'] is False
        value['capacity_release']=True
        try:evidence.failure_evidence(3,{'memory':'FAKE_ONLY'})
        except AssertionError:pass
        else:raise AssertionError('reserved failure cannot release')
    finally:evidence.same_read=old

def test_reserved_fake_wait_credit_rejected(evidence):
    value=reserved_owned();value['known']=[{'pid':91,'birth':'12','state':None,'ppid':1,
            'waited':False,'absent':True,'returncode':None}]
    rejected(evidence.failure_owned_state,value)
    value['known'][0]['absent']=False;value['known'][0]['returncode']=0
    rejected(evidence.failure_owned_state,value)

def run_reserved_failure_controls(evidence):
    test_reserved_direct_pidfd_open_failure(evidence)
    test_reserved_unbound_direct_requires_unknown(evidence)
    test_reserved_strict_read_charge_overshoot(evidence)
    test_reserved_consumer_preserves_unknown_and_no_release(evidence)
    test_reserved_fake_wait_credit_rejected(evidence)
    return {'controls':5,'kernel_calls':0,'Source_only':True,'normal_output_limit_unchanged':33554432}
