          'compressed_hex':compressed.hex()}

                                               'sha256':hashlib.sha256(leaf_raw).hexdigest()}

           'compressed_bytes':len(compressed),'compressed_sha256':hashlib.sha256(compressed).hexdigest(),
           'raw_bytes':len(raw),'raw_sha256':hashlib.sha256(raw).hexdigest(),'event_count':len(events),
           'owned_delta_initial':{} if initial is None else initial,'kernel':{},'owned':final,
           'observation_failed':False,'failure_count':0,'first_failure':None,'last_failure':None,
           'capacity_release':False,'root_outside_wait':'REQUIRED_NOT_OBSERVED_INSIDE'}

def test_c44_stack_reservation_arithmetic(probe,payload):
    for module in (probe,payload):
        reserve=module.THREAD_BASE_MAX+module.THREAD_MAX_NEW*(module.THREAD_STACK_BYTES+
                module.THREAD_GUARD_RESERVE+module.THREAD_RUNTIME_RESERVE)+module.THREAD_SHARED_RESERVE
        assert module.THREAD_STACK_BYTES==262144 and module.THREAD_MAX_NEW==126
        assert reserve==257294336 and reserve<=268435456
    # Arithmetic only, no pthread allocation or actual pids.max credit.


def test_c44_quota_complete_and_positive_partial(probe):
    for replies,expected in [([8],(8,0,1,None)),([3,5],(8,1,2,None)),([2,3,3],(8,2,3,None))]:
        seen=[];iterator=iter(replies)
        def writer(fd,raw):seen.append(bytes(raw));return next(iterator)
        with patch.object(probe.os,'write',side_effect=writer):
            assert probe.quota_chunk(123,b'12345678',0)==expected
        assert b''.join(raw[:n] for raw,n in zip(seen,replies))==b'12345678'


def test_c44_quota_partial_then_enospc(probe):
    with patch.object(probe.os,'write',side_effect=[3,OSError(errno.ENOSPC,'MEMORY_ONLY')]):
        assert probe.quota_chunk(123,b'12345678',0)==(3,1,2,28)
    with patch.object(probe.os,'write',side_effect=OSError(errno.ENOSPC,'MEMORY_ONLY')):
        assert probe.quota_chunk(123,b'12345678',100)==(100,0,1,28)


def test_c44_quota_invalid_progress_and_unrelated_errno(probe):
    for reply in (0,-1,9,True):
        with patch.object(probe.os,'write',return_value=reply):
            _refused(lambda:probe.quota_chunk(123,b'12345678',0))
    with patch.object(probe.os,'write',side_effect=OSError(errno.EIO,'MEMORY_ONLY')):
        _refused(lambda:probe.quota_chunk(123,b'12345678',0))


def test_c44_quota_budget_refuses_before_write(probe):
    with patch.object(probe.os,'write') as write:
        _refused(lambda:probe.quota_chunk(123,b'12345678',probe.MAX_WRITE-7))
        assert write.call_count==0
    with patch.object(probe.os,'write',return_value=1) as write:
        _refused(lambda:probe.quota_chunk(123,b'x'*4097,0))
        assert write.call_count==4096
    with patch.object(probe.os,'write',return_value=1) as write:
        _refused(lambda:probe.quota_chunk(123,b'ab',0,call_budget=1))
        assert write.call_count==1
    with patch.object(probe.os,'write') as write:
        _refused(lambda:probe.quota_chunk(123,b'ab',0,call_budget=0))
        assert write.call_count==0


def _adapter(observe,primary,latched=False,already_reserved=False,broken_logger=False):
    writes=[];calls=[]
    owner=SimpleNamespace(temp_root=Path('/MEMORY_ONLY'),group=Path('/MEMORY_ONLY_GROUP'),known={},direct={},
        unknown_spawn=False,ambiguous={},output_bytes=0,
        fds={'lock_update':1,'lock_install':2,'native_host_pidfd':3},pending_receipt=None)
    owner.native_check=lambda:None
    for name in observe.METHODS:setattr(owner,name,lambda *args,**kwargs:None)
    def run(*args,**kwargs):
        calls.append('native_run')
        if latched:owner._c40_observation_failed=True
        else:owner._c40_raw_bytes=observe.TRACE_RAW_MAX
        raise primary
    owner.native_run=run;owner.native_put=lambda path,value:writes.append(value)
    with patch.object(observe,'kernel_snapshot',return_value={}):
        observe.attach(owner,'MEMORY_ONLY_C44')
        owner._c40_reserved_attempted=already_reserved
        context=patch.object(observe,'observation_failure',side_effect=RuntimeError('MEMORY_ONLY logger')) if broken_logger else patch.object(observe,'observation_failure',wraps=observe.observation_failure)
        with context:
            try:observe.real_workload(owner,'/MEMORY_ONLY/python','/MEMORY_ONLY/script','MEMORY_ONLY')
            except BaseException as caught:assert caught is primary
            else:raise AssertionError('original native exception lost through actual wrapper')
    assert calls==['native_run'] and owner._c40_observation_failed is True
    assert owner._c40_first_failure is not None and owner._c40_last_failure is not None
    assert len(writes)==(0 if already_reserved else 1)
    if writes:
        assert writes[0]['actual_primary_exception']['class']==type(primary).__name__
        assert writes[0]['capacity_release'] is False and writes[0]['normal_evidence_complete'] is False

def test_c44_mismatched_noncanonical_and_incomplete_birth_negative(evidence):
    original=dict(_owned(),known=[_row()],direct_pids=[7]);key='[7,"100"]'
    variants=[('pid',8),('birth','101'),('pid',True),('birth',100),('birth','0100'),
              ('birth',''),('state','ABSENT'),('ppid',True),('waited',1),('absent',1),('returncode',True)]
    for field,value in variants:
        event=_first(copy.deepcopy(original));event['known_birth_delta'][key][field]=value
        _refused(lambda:_caller(evidence,[event],original))
    for wrong_key in ('[7, "100"]','[7,"101"]','[8,"100"]'):
        event=_first(copy.deepcopy(original));row=event['known_birth_delta'].pop(key)
        event['known_birth_delta'][wrong_key]=row;event['known_birth_order']=[wrong_key]
        _refused(lambda:_caller(evidence,[event],original))
    for missing in ('pid','birth','waited'):
        event=_first(copy.deepcopy(original));del event['known_birth_delta'][key][missing]
        _refused(lambda:_caller(evidence,[event],original))
    event=_first(copy.deepcopy(original));event['known_birth_delta'][key]['unexpected']=1
    _refused(lambda:_caller(evidence,[event],original))
