"""Original CLI finite delivery protocol; serialized records grant no authority.

C54 uses an actual fixed product guardian and direct finite observer raw wait;
the original frontend/AppSession stays alive on positive completion.

The original product bootstrap installs an authenticated retained channel
BEFORE creating the original CLI. No environment FD, worker JSON, SDK callback,
new principal or session reconstruction can install one. The durable association
stays UNKNOWN. Only its original outside owner can supply a certified observation.
Missing custody refuses delivery before validation or filesystem effects.
"""
from contextlib import contextmanager
from contextvars import ContextVar
import asyncio
import copy
import hashlib
import json
import math
import os
import select
import socket
import stat
import time

CALL_FIELDS = ('task_id', 'session_id', 'turn_id', 'api_request_id', 'tool_call_id')
MAX_PACKET = 65536
_SCOPE = ContextVar('friday_original_finite_cli_scope', default=None)
_PORT = ContextVar('friday_original_finite_cli_port', default=None)
_READER = ContextVar('friday_original_cli_certification_reader', default=None)


class FiniteOwnerUnavailable(RuntimeError):
    pass


def canonical(value):
    return json.dumps(value, sort_keys=True, ensure_ascii=True, allow_nan=False,
                      separators=(',', ':')).encode('ascii')


def _number(value):
    if type(value) not in (int, float) or not math.isfinite(value) or value <= 0:
        raise FiniteOwnerUnavailable('invalid_finite_clock')
    return value


def _ids(value):
    if (type(value) is not dict or set(value) != set(CALL_FIELDS)
            or any(type(v) is not str or not 0 < len(v.encode('utf-8')) <= 512
                   or '\x00' in v for v in value.values())):
        raise FiniteOwnerUnavailable('invalid_original_five_ids')
    return copy.deepcopy(value)


def digest(value):
    return hashlib.sha256(canonical(value)).hexdigest()


def validate_proposal(value):
    fields = {'schema', 'scope', 'original_ids', 'started_m', 'deadline_m',
              'reference', 'number', 'admission_hash', 'owner_sha256',
              'artifact_sha256', 'artifact_bytes', 'output_joined_m'}
    if (type(value) is not dict or set(value) != fields
            or value['schema'] != 'friday.cli-delivery-proposal.v1'
            or type(value['scope']) is not str or len(value['scope']) != 64
            or any(c not in '0123456789abcdef' for c in value['scope'])
            or type(value['reference']) is not str or not 0 < len(value['reference']) <= 128
            or type(value['number']) is not int or not 1 <= value['number'] <= 16
            or type(value['artifact_bytes']) is not int or not 0 <= value['artifact_bytes'] <= 262144):
        raise FiniteOwnerUnavailable('invalid_delivery_proposal')
    _ids(value['original_ids'])
    for field in ('admission_hash', 'owner_sha256', 'artifact_sha256'):
        v = value[field]
        if type(v) is not str or len(v) != 64 or any(c not in '0123456789abcdef' for c in v):
            raise FiniteOwnerUnavailable('invalid_delivery_proposal_hash')
    start, end, joined = (_number(value[k]) for k in ('started_m','deadline_m','output_joined_m'))
    if not start < joined < end <= start + 30:
        raise FiniteOwnerUnavailable('expired_delivery_proposal')
    return copy.deepcopy(value)


def validate_certificate(value, proposal):
    p = validate_proposal(proposal)
    fields = {'schema', 'proposal_sha256', 'scope', 'frontend_pid', 'frontend_birth',
              'guardian_pid', 'guardian_birth', 'sampler', 'observer',
              'frontend_joined_m', 'input_restored_m', 'proposal_durable_m', 'certified_m'}
    if (type(value) is not dict or set(value) != fields
            or value['schema'] != 'friday.cli-delivery-certificate.v2'
            or value['scope'] != p['scope'] or value['proposal_sha256'] != digest(p)):
        raise FiniteOwnerUnavailable('uncertified_finite_operation')
    for who in ('frontend', 'guardian'):
        if (type(value[who+'_pid']) is not int or value[who+'_pid'] <= 1
                or type(value[who+'_birth']) is not str or not value[who+'_birth'].isdecimal()):
            raise FiniteOwnerUnavailable('invalid_native_birth')
    if value['frontend_pid'] == value['guardian_pid']:
        raise FiniteOwnerUnavailable('collapsed_native_roles')
    for who in ('sampler', 'observer'):
        w = value[who]
        if (type(w) is not dict or set(w) != {'pid','birth','raw_status','wait_m','cleanup_m'}
                or type(w['pid']) is not int or w['pid'] <= 1
                or type(w['birth']) is not str or not w['birth'].isdecimal()
                or type(w['raw_status']) is not int or w['raw_status'] != 0
                or w['pid'] in (value['frontend_pid'], value['guardian_pid'])):
            raise FiniteOwnerUnavailable('actual_direct_observer_raw_wait_required')
    if value['sampler']['pid'] == value['observer']['pid']:
        raise FiniteOwnerUnavailable('distinct_native_observers_required')
    ordered = [_number(value['sampler'][k]) for k in ('cleanup_m','wait_m')]
    ordered += [p['output_joined_m'], _number(value['frontend_joined_m']),
                _number(value['input_restored_m']), _number(value['proposal_durable_m']),
                _number(value['observer']['cleanup_m']), _number(value['observer']['wait_m']),
                _number(value['certified_m'])]
    if not p['started_m'] <= ordered[0] or ordered != sorted(ordered) or ordered[-1] >= p['deadline_m']:
        raise FiniteOwnerUnavailable('late_or_unjoined_finite_certificate')
    return copy.deepcopy(value)


def _close_primary(fd, primary=None):
    try:os.close(fd)
    except BaseException as cleanup:
        if primary is None:raise
        BaseException.add_note(primary,'Finite owner FD close unconfirmed:'+type(cleanup).__name__)


def _birth(pid):
    fd = os.open('/proc/'+str(pid)+'/stat', os.O_RDONLY | os.O_CLOEXEC)
    primary=None
    try:b=os.read(fd,8192).decode('ascii')
    except BaseException as failure:primary=failure;raise
    finally:_close_primary(fd,primary)
    tail = b[b.rfind(')')+2:].split()
    return {'pid':pid,'birth':tail[19],'ppid':int(tail[1]),'state':tail[0]}


def _pidfd(fd, expected):
    st = os.fstat(fd)
    if stat.S_ISREG(st.st_mode) or stat.S_ISDIR(st.st_mode):
        raise FiniteOwnerUnavailable('not_retained_pidfd')
    info_fd = os.open('/proc/self/fdinfo/'+str(fd), os.O_RDONLY | os.O_CLOEXEC)
    primary=None
    try:info=os.read(info_fd,8192).decode('ascii')
    except BaseException as failure:primary=failure;raise
    finally:_close_primary(info_fd,primary)
    if not any(line.split() == ['Pid:',str(expected['pid'])] for line in info.splitlines()):
        raise FiniteOwnerUnavailable('foreign_pidfd')
    now = _birth(expected['pid'])
    if now['birth'] != expected['birth']:
        raise FiniteOwnerUnavailable('native_birth_changed')
    return (st.st_dev, st.st_ino)


class RetainedOwnerChannel:
    """Exact existing stock-owned SOCK_SEQPACKET; no path/env/JSON reconstruction.

    Constructed only at the trusted existing stock bootstrap with its native
    socket and already-retained outside owner pidfd. Lifetime/peer identity is
    checked on every packet. No detached reader, new loop, service or retry.
    """
    def __init__(self, native_socket, outside_birth, outside_pidfd):
        if type(native_socket) is not socket.socket or native_socket.type != socket.SOCK_SEQPACKET:
            raise FiniteOwnerUnavailable('native_owner_channel_missing')
        if native_socket.family != socket.AF_UNIX or native_socket.gettimeout() != 0:
            raise FiniteOwnerUnavailable('native_owner_channel_not_nonblocking')
        self.socket = native_socket
        self.outside = copy.deepcopy(outside_birth)
        self.pidfd = outside_pidfd
        self.pidfd_identity = _pidfd(outside_pidfd, outside_birth)
        st = os.fstat(native_socket.fileno())
        self.socket_identity = (st.st_dev, st.st_ino)
        self.closed = False
        self.cli = self.cap = self.output = None
        self.active = None
        self.certificates = {}
        self._check()

    def _check(self):
        import struct
        if self.closed:
            raise FiniteOwnerUnavailable('original_finite_channel_closed')
        st = os.fstat(self.socket.fileno())
        if (st.st_dev,st.st_ino) != self.socket_identity or _pidfd(self.pidfd,self.outside) != self.pidfd_identity:
            raise FiniteOwnerUnavailable('original_finite_handle_changed')
        if self.socket.getsockopt(socket.SOL_SOCKET,socket.SO_PASSCRED) != 1:
            raise FiniteOwnerUnavailable('native_reply_credentials_required')
        if self.cli is not None:
            if (getattr(self.cli,'_friday_cli_scope',None) is not self.cap
                    or getattr(self.cli,'_friday_cli_output',None) is not self.output
                    or getattr(self.cli,'_friday_cli_finite_port',None) is not self):
                raise FiniteOwnerUnavailable('original_finite_binding_changed')

    def receive(self, end):
        self._check()
        remaining = end - time.monotonic()
        if remaining <= 0 or not select.select([self.socket,self.pidfd],[],[],remaining)[0]:
            raise FiniteOwnerUnavailable('finite_owner_reply_unconfirmed')
        self._check()
        data, ancillary, flags, address = self.socket.recvmsg(
            MAX_PACKET, socket.CMSG_SPACE(12), socket.MSG_DONTWAIT)
        import struct
        creds = [struct.unpack('3i', v) for level, kind, v in ancillary
                 if level == socket.SOL_SOCKET and kind == socket.SCM_CREDENTIALS and len(v) == 12]
        if (not data or flags & (socket.MSG_TRUNC | socket.MSG_CTRUNC)
                or len(ancillary) != len(creds) or len(creds) != 1
                or creds[0] != (self.outside['pid'], os.getuid(), os.getgid())):
            raise FiniteOwnerUnavailable('unauthenticated_live_native_owner_reply')
        reply = json.loads(data)
        if time.monotonic() >= end or type(reply) is not dict:
            raise FiniteOwnerUnavailable('late_finite_owner_reply')
        return reply

    def exchange(self, packet, end, *, retained_fd=None):
        self._check()
        data = canonical(packet)
        if len(data) > MAX_PACKET or time.monotonic() >= end:
            raise FiniteOwnerUnavailable('finite_packet_expired_or_large')
        try:
            if retained_fd is None:
                n = self.socket.send(data, socket.MSG_DONTWAIT | socket.MSG_NOSIGNAL)
            else:
                import array
                if packet.get('op') != 'BIND_RECORD' or type(retained_fd) is not int:
                    raise FiniteOwnerUnavailable('foreign_record_handle_transfer')
                n = self.socket.sendmsg([data], [(socket.SOL_SOCKET, socket.SCM_RIGHTS,
                        array.array('i', [retained_fd]))],
                        socket.MSG_DONTWAIT | socket.MSG_NOSIGNAL)
            if n != len(data):
                raise FiniteOwnerUnavailable('ambiguous_finite_packet')
            return self.receive(end)
        except BaseException as primary:
            self.closed = True
            if self.output is not None:
                try:
                    self.output.close()
                except BaseException as cleanup:
                    BaseException.add_note(primary, 'Finite frontend stop unconfirmed:'+type(cleanup).__name__)
            # Closing the authenticated channel wakes the native guardian. If
            # an operation is active, it stops only its exact product frontend.
            try:
                self.socket.close()
            except BaseException as cleanup:
                BaseException.add_note(primary, 'Finite channel close unconfirmed:'+type(cleanup).__name__)
            raise


@contextmanager
def existing_product_cli_port(channel):
    if type(channel) is not RetainedOwnerChannel or _PORT.get() is not None:
        raise FiniteOwnerUnavailable('original_finite_port_not_native')
    channel._check()
    token=_PORT.set(channel)
    try:yield channel
    finally:_PORT.reset(token)


def attach_original_cli_port(cli):
    port=_PORT.get()
    if port is None:return
    if type(port) is not RetainedOwnerChannel or port.cli is not None:
        raise FiniteOwnerUnavailable('finite_cli_already_bound')
    port._check()
    port.cli=cli;port.cap=cli._friday_cli_scope;port.output=cli._friday_cli_output
    cli._friday_cli_finite_port=port
    cli._friday_cli_certificate_reader=NativeCertificateReader(port)
    port._check()


class OriginalDeliveryLease:
    def __init__(self, port, started, end, ids):
        self.port=port;self.started=_number(started);self.end=_number(end);self.ids=_ids(ids)
        self.task=None;self.output_observations={};self.proposals={};self.joined=False;self.closed=False;self.expected_artifact=None;self.context=None;self.registration=None;self.native_rows={}
        if not started < end <= started+30 or port.active is not None:
            raise FiniteOwnerUnavailable('original_finite_budget_or_ownership')
        # Existing outside guardian ACK precedes validation, reads or a native
        # output effect. Receipt JSON never constructs or installs this port.
        ack=port.exchange({'op':'ARM','schema':'friday.cli-finite-owner.v1',
                          'original_ids':self.ids,'started_m':started,'deadline_m':end},end)
        if set(ack)!={'op','scope','deadline_m','native_handles'} or ack['op']!='ARMED' or not started<ack['deadline_m']<=end:
            raise FiniteOwnerUnavailable('original_guardian_not_armed')
        if type(ack['scope']) is not str or len(ack['scope'])!=64:
            raise FiniteOwnerUnavailable('invalid_original_scope')
        self.scope=ack['scope'];self.end=ack['deadline_m'];port.active=self
        try:
            self._bind_original_native_handles(ack['native_handles'])
        except BaseException:
            self.port.closed=True
            self.port.output.close()
            self.port.socket.close()
            raise

    def _bind_original_native_handles(self, sample):
        import fcntl
        import io
        if type(sample) is not dict or set(sample) != {
                'input_identity','output_identity','input_flags','output_flags','input_state'}:
            raise FiniteOwnerUnavailable('actual_native_handle_sample_required')
        stream = self.port.output.stream
        if type(stream) is io.StringIO:
            # Preserve the already-owned exact memory sink. Its immutable C
            # writes and task-owned cleanup are observed by output_completed;
            # observer raw wait certifies record observation, not this effect.
            from hermes_cli.friday_cli_output import _check_memory_methods, _memory_bounds
            _check_memory_methods(stream)
            position,end=_memory_bounds(stream)
            if position != end:
                raise FiniteOwnerUnavailable('original_memory_output_not_at_end')
            return
        if type(stream) is not io.TextIOWrapper:
            raise FiniteOwnerUnavailable('actual_original_native_stream_unbound')
        raw = stream.buffer.raw if type(stream.buffer) is io.BufferedWriter else stream.buffer
        if type(raw) is not io.FileIO:
            raise FiniteOwnerUnavailable('actual_original_raw_output_unbound')
        fd = io.FileIO.fileno(raw)
        def identity(value):
            st = os.fstat(value)
            return [st.st_dev,st.st_ino,st.st_mode,st.st_rdev]
        if (identity(fd) != sample['output_identity'] or identity(0) != sample['input_identity']
                or fcntl.fcntl(fd,fcntl.F_GETFL) != sample['output_flags']
                or fcntl.fcntl(0,fcntl.F_GETFL) != sample['input_flags']):
            raise FiniteOwnerUnavailable('guardian_not_bound_to_original_native_IO')
        app = self.port.output.renderer_app
        if app is not None and app.input.fileno() != 0:
            raise FiniteOwnerUnavailable('original_AppSession_input_unbound')

    def check(self):
        if self.closed or self.port.active is not self or time.monotonic()>=self.end:
            raise FiniteOwnerUnavailable('finite_delivery_expired')
        self.port._check()
        from hermes_cli.friday_cli_work import admitted_call
        admitted_call(self.ids)
        self.port.output.check_lifetime()
        if self.context is not None:
            from hermes_cli.friday_cli_work import check_store
            check_store(self.port.cap,self.context.state)
            manager=self.context._manager
            loaded=manager._plugins.get(self.context.plugin_id)
            if (self.context._load_abandoned or getattr(manager,'_gateway_unloading',False)
                    or loaded is None or not loaded.enabled or loaded.manifest is not self.context.manifest
                    or self.context.get_config('results') != {'enabled':True}):
                raise FiniteOwnerUnavailable('actual_native_effect_admission_lost')

    def shorten(self, end):
        self.check();end=min(self.end,_number(end))
        if end!=self.end:
            ack=self.port.exchange({'op':'SHORTEN','scope':self.scope,'deadline_m':end},end)
            if ack!={'op':'SHORTENED','scope':self.scope,'deadline_m':end}:
                raise FiniteOwnerUnavailable('finite_budget_shortening_unconfirmed')
            self.end=end
        self.check();return self.end

    def bind_native_task(self, task):
        self.check()
        if self.task is not None or type(task) is not asyncio.Task:
            raise FiniteOwnerUnavailable('foreign_finite_ledger_task')
        from hermes_cli.plugins_ledger import PluginRegistration
        handle = getattr(task, '_friday_cli_ledger_registration', None)
        manager = self.context._manager
        if (type(handle) is not PluginRegistration or handle.kind != 'background_task'
                or not handle.active or not any(handle is h for hs in manager._ownership_ledger.values() for h in hs)):
            raise FiniteOwnerUnavailable('actual_original_native_ledger_owner_required')
        self.registration=handle
        self.task=task

    def expect_artifact(self, artifact, number):
        self.check()
        if asyncio.current_task() is not self.task:
            raise FiniteOwnerUnavailable('foreign_original_artifact_task')
        self.expected_artifact=(artifact['reference'],number,artifact['sha256'])

    def output_completed(self, body):
        self.check()
        if asyncio.current_task() is not self.task:
            raise FiniteOwnerUnavailable('foreign_finite_output_task')
        if self.port.output.renderer_requests:
            raise FiniteOwnerUnavailable('native_renderer_join_unconfirmed')
        if self.expected_artifact is None or self.expected_artifact[2]!=body['sha256']:
            raise FiniteOwnerUnavailable('foreign_original_artifact_output')
        self.output_observations[self.expected_artifact[:2]]=(body['bytes'],time.monotonic())
        self.check()

    def proposal(self, row, artifact, number):
        self.check()
        size,joined=self.output_observations[(artifact['reference'],number)]
        if size!=artifact['size_bytes'] or asyncio.current_task() is not self.task:
            raise FiniteOwnerUnavailable('original_output_observation_missing')
        from hermes_cli.friday_user_scope import check_retained_job
        check_retained_job(row)
        p=validate_proposal({'schema':'friday.cli-delivery-proposal.v1','scope':self.scope,
            'original_ids':self.ids,'started_m':self.started,'deadline_m':self.end,
            'reference':artifact['reference'],'number':number,'admission_hash':row['admission_hash'],
            'owner_sha256':digest(row['owner']),'artifact_sha256':artifact['sha256'],
            'artifact_bytes':size,'output_joined_m':joined})
        self.proposals[digest(p)]=p
        self.native_rows[digest(p)]=copy.deepcopy(row)
        return p

    def native_task_joined(self, task):
        self.check()
        if task is not self.task or not task.done() or task.cancelled() or task.exception() is not None:
            raise FiniteOwnerUnavailable('original_native_ledger_join_unconfirmed')
        self._normal_frontend_cleanup()
        self.joined = True
        # This is a necessary local join, never a delivery certificate by itself.

    def _normal_frontend_cleanup(self):
        self.port.output.check()
        if self.task is not None and self.task.done():
            handle=self.registration;manager=self.context._manager
            if (getattr(self.task,'_friday_cli_ledger_cleanup',None) is not handle
                    or handle is None or handle.active
                    or any(handle is h for hs in manager._ownership_ledger.values() for h in hs)):
                raise FiniteOwnerUnavailable('actual_native_ledger_disposal_unconfirmed')
        if (self.port.output.renderer_requests or self.port.output.busy.locked()
                or self.port.output.uncertain_descriptor is not None):
            raise FiniteOwnerUnavailable('original_output_cleanup_unconfirmed')

    def bind_record(self, context):
        from hermes_cli.friday_cli_work import check_store
        from hermes_cli.friday_cli_principal import current_cli
        check_store(current_cli(required=True), context.state)
        self.context=context
        # Only the actual native PluginState directory; no argument/path/env
        # can supply this handle. Guardian is already armed before open/fstat.
        fd = os.open(os.fspath(context.state.data_dir),
                     os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW | os.O_CLOEXEC)
        primary = None
        try:
            s = os.fstat(fd)
            if not stat.S_ISDIR(s.st_mode) or s.st_uid != os.getuid() or stat.S_IMODE(s.st_mode) != 0o700:
                raise FiniteOwnerUnavailable('original_private_record_directory')
            ack = self.port.exchange({'op':'BIND_RECORD','scope':self.scope},self.end,retained_fd=fd)
            if ack != {'op':'RECORD_BOUND','scope':self.scope}:
                raise FiniteOwnerUnavailable('native_record_custody_unconfirmed')
        except BaseException as exc:
            primary = exc
            raise
        finally:
            _close_primary(fd,primary)
        self.check()

    def certify_after_bridge(self):
        # Called AFTER native ledger join AND _run_async's temporary loop and
        # renderer finally blocks return. Original AppSession is still alive.
        self.check()
        if not self.joined:
            raise FiniteOwnerUnavailable('native_finite_task_not_joined')
        self._normal_frontend_cleanup()
        if not self.proposals:
            return
        from hermes_cli.friday_user_scope import check_retained_job
        from hermes_cli.friday_cli_work import should_stop
        for h,p in self.proposals.items():
            check_retained_job(self.native_rows[h])
            if should_stop():
                raise FiniteOwnerUnavailable('native_effect_stop_before_certification')
            ack=self.port.exchange({'op':'PROPOSE','scope':self.scope,'proposals':[p]},self.end)
            if ack != {'op':'PROPOSALS_RETAINED','scope':self.scope,'proposal_hashes':[h]}:
                raise FiniteOwnerUnavailable('original_proposal_handoff_unconfirmed')
        ack = self.port.exchange({'op':'SETTLE','scope':self.scope,
                                 'frontend_joined_m':time.monotonic()},self.end)
        if (set(ack) != {'op','scope','certificates'} or ack['op'] != 'CERTIFIED'
                or ack['scope'] != self.scope or type(ack['certificates']) is not dict
                or set(ack['certificates']) != set(self.proposals)):
            raise FiniteOwnerUnavailable('actual_live_finite_certificate_missing')
        for h,p in self.proposals.items():
            c = validate_certificate(ack['certificates'][h],p)
            if (c['frontend_pid'] != os.getpid() or c['frontend_birth'] != _birth(os.getpid())['birth']
                    or c['guardian_pid'] != self.port.outside['pid']
                    or c['guardian_birth'] != self.port.outside['birth']):
                raise FiniteOwnerUnavailable('foreign_native_finite_certificate')
            if len(self.port.certificates) >= 512 and h not in self.port.certificates:
                del self.port.certificates[next(iter(self.port.certificates))]
            self.port.certificates[h] = c
        self.check()

    def retire(self, primary=None):
        # Actual output/record/observer cleanup completed while the hard guard
        # still covers final caller projection and JSON encoding.
        try:
            if primary is not None:
                self.port.exchange({'op':'ABORT','scope':self.scope},self.end)
                raise FiniteOwnerUnavailable('finite_abort_not_clean_completion')
            self.check()
            self._normal_frontend_cleanup()
            if self.task is not None and not self.joined:
                raise FiniteOwnerUnavailable('native_task_cleanup_unconfirmed')
            ack = self.port.exchange({'op':'FINISH','scope':self.scope},self.end)
            if (set(ack) != {'op','scope','closed_m'} or ack['op'] != 'CLOSED'
                    or ack['scope'] != self.scope or not self.started < ack['closed_m'] < self.end):
                raise FiniteOwnerUnavailable('native_operation_normal_close_unconfirmed')
            if time.monotonic() >= self.end:
                raise FiniteOwnerUnavailable('late_frontend_operation_return')
        except BaseException as cleanup:
            self.port.closed = True
            self.port.output.close()
            try:
                self.port.socket.close()
            except BaseException as later:
                BaseException.add_note(cleanup, 'native_channel_close_unconfirmed:'+type(later).__name__)
            if primary is None:
                raise
            BaseException.add_note(primary,'Finite frontend stop unconfirmed:'+type(cleanup).__name__)
        finally:
            self.closed = True
            if self.port.active is self:
                self.port.active = None


@contextmanager
def original_result_boundary(context, started, end):
    from hermes_cli.friday_cli_work import _TURN, _TOOL
    turn=_TURN.get()
    if turn is None:
        yield None
        return
    port=getattr(turn['cli'],'_friday_cli_finite_port',None);tool=_TOOL.get()
    if (type(port) is not RetainedOwnerChannel or port.cli is not turn['cli']
            or port.cap is not turn['cap'] or port.output is not turn['output']
            or type(tool) is not dict or tool['turn'] is not turn):
        raise FiniteOwnerUnavailable('existing_original_outside_owner_not_connected')

class NativeCertificateReader:
    """Original live reader bound once to the authenticated native bootstrap."""
    def __init__(self, channel):
        if type(channel) is not RetainedOwnerChannel:
            raise FiniteOwnerUnavailable('actual_native_channel_required')
        channel._check()
        self.channel = channel

    def observe(self, proposal, end):
        p = validate_proposal(proposal)
        if time.monotonic() >= end:
            raise FiniteOwnerUnavailable('original_receipt_read_expired')
        from hermes_cli.friday_cli_work import current_turn
        turn=current_turn()
        if (turn['cli'] is not self.channel.cli or turn['cap'] is not self.channel.cap
                or turn['output'] is not self.channel.output
                or getattr(turn['cli'],'_friday_cli_finite_port',None) is not self.channel):
            raise FiniteOwnerUnavailable('reader_native_bootstrap_binding_lost')
        self.channel._check()
        c = self.channel.certificates.get(digest(p))
        if c is None:
            raise FiniteOwnerUnavailable('actual_observer_certificate_not_retained')
        c = validate_certificate(c,p)
        if (c['frontend_pid'] != os.getpid() or c['frontend_birth'] != _birth(os.getpid())['birth']
                or c['guardian_pid'] != self.channel.outside['pid']
                or c['guardian_birth'] != self.channel.outside['birth']):
            raise FiniteOwnerUnavailable('foreign_live_certificate')
        if time.monotonic() >= end:
            raise FiniteOwnerUnavailable('late_original_receipt_read')
        return c

def observed_attempt(row, artifact, attempt, *, read_end=None):
    if row['owner'].get('platform')!='cli':return attempt['state']
    receipt=attempt.get('receipt') or {}
    if attempt['state']=='FAILED' and receipt=={'state':'FAILED','error':'cli_document_preflight_rejected'}:
        return 'FAILED'
    p=receipt.get('proposal');reader=_READER.get()
    if reader is None:
        try:
            from hermes_cli.friday_cli_work import current_turn
            reader=getattr(current_turn()['cli'],'_friday_cli_certificate_reader',None)
        except Exception:
            return 'UNKNOWN'
    if type(reader) is not NativeCertificateReader or p is None:return 'UNKNOWN'
    try:
        p=validate_proposal(p)
        if (p['reference']!=artifact['reference'] or p['number']!=attempt['number']
                or p['admission_hash']!=row['admission_hash'] or p['owner_sha256']!=digest(row['owner'])
                or p['artifact_sha256']!=artifact['sha256'] or p['artifact_bytes']!=artifact['size_bytes']):
            return 'UNKNOWN'
        # Current owner/access generation is checked through retained native
        # admission, not a timestamp field or boolean in the proposal.
        from hermes_cli.friday_user_scope import check_retained_job
        check_retained_job(row)
        if read_end is None:return 'UNKNOWN'  # no reader +30 renewal
        reader.observe(p,read_end)
        return 'DELIVERED'
    except Exception:return 'UNKNOWN'
