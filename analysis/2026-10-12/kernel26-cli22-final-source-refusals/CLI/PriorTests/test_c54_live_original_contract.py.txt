import pytest

POSITIVE = ['direct_live_positive', 'wrapped_live_positive', 'direct_two_distinct_live_operations', 'wrapped_two_distinct_live_operations', 'sixteen_artifacts_unicode_original_ids', 'shorter_original_caller_positive', 'original106_exact_current_binding', 'real_normal_finite_actor_raw_wait_positive']
NEGATIVE = ['missing_actual_stock_bootstrap', 'late_arm_after_first_validation', 'env_fd_bootstrap_substitution', 'new_principal_or_home_substitution', 'whole_frontend_exit_positive_substitution', 'copied_appsession_or_console', 'source_unknown_only_positive_substitution', 'outside_reader_reminted_admission', 'original_task_presentation_before_propose', 'second_arm_original_owner_busy', 'gnu_raw_actor_wait_not_supplied', 'stock_zero_actor_status_inferred', 'recycled_actor_birth', 'foreign_raw_wait_producer', 'copied_serialized_raw_wait', 'wrong_native_actor_ancestry', 'protected_developer_writer_target', 'slow_initial_validation', 'blocked_reservation_fsync', 'late_atomic_proposal_rename', 'late_proposal_directory_fsync', 'blocked_outside_record_read', 'blocked_outside_record_fsync', 'blocked_outside_close', 'sync_renderer_before_first_yield', 'blocked_input_detach', 'blocked_cooked_mode_exit', 'blocked_reset_redraw_flush', 'preyield_primary_cleanup_failure', 'body_primary_cleanup_failure', 'native_ledger_cancelled', 'native_renderer_cancelled', 'stop_before_first_effect', 'stop_after_partial_output', 'uninterruptible_finite_actor_stop', 'late_native_callback', 'access_generation_changed', 'five_native_ids_changed', 'foreign_channel_credentials', 'forged_proposal_or_certificate', 'auto_unknown_retry', 'explicit_unknown_retry']
CASES = POSITIVE + NEGATIVE

@pytest.mark.parametrize('case', CASES)
def test_c54_actual_live_original_contract(c54_live_cli_observation, case):
    # Independent existing harness creates this observation from the named real
    # case. No new run_exact/current owner API is proposed by these definitions.
    o = c54_live_cli_observation[case]
    assert o['actual_case_id'] == case
    assert o['exact_source_bootstrap_dependency_binding_verified'] is True
    assert o['physical_native_run'] is True
    assert o['independent_fixture_provenance_verified'] is True
    assert o['no_developer_or_foreign_writer_effect'] is True
    assert o['no_new_engine_pump_service_hidden_tui_or_fd_transfer'] is True
    assert o['no_delivery_replay'] is True
    assert o['no_original_attempt_budget_renewal'] is True
    start, end = o['actual_original_start_m'], o['actual_original_end_m']
    assert start < end <= start + 30
    # Refusal is insufficient for hard30 if a writer/observer/cleanup continues.
    assert o['actual_all_effect_observer_cleanup_joined_m'] <= end
    assert o['actual_no_owned_orphan_writer_or_observer'] is True
    if case in POSITIVE:
        assert o['public_delivery_observation'] == 'DELIVERED'
        assert o['actual_frontend_alive_after_return'] is True
        assert o['whole_original_frontend_wait_status'] is None
        assert o['actual_frontend_pid_birth_before'] == o['actual_frontend_pid_birth_after']
        before, after = o['original_objects_before'], o['original_objects_after']
        fields = ('app_session', 'application', 'console', 'stream', 'input',
                  'context', 'manager', 'native_principal', 'session_database')
        assert set(before) == set(after) == set(fields)
        assert all(before[k] is after[k] for k in fields)
        assert o['native_five_ids_before'] == o['native_five_ids_after']
        assert o['native_uid_home_generation_database_inode_preserved'] is True
        assert o['actual_original_input_modes_restored'] is True
        assert o['actual_native_renderer_and_ledger_joined'] is True
        assert o['artifact_bytes_exactly_once'] is True
        assert o['genuine_joined_finite_effect_observer_cleanup_witness'] is True
        assert o['finite_effect_actor_is_original_frontend'] is False
        assert o['actual_stock_raw_wait_status'] == 0
        assert o['actual_finite_actor_raw_wait_status'] == 0
        assert o['actual_raw_wait_producer_custody_verified'] is True
        assert o['actual_certificate_reader_admission_preserved'] is True
        assert start <= o['actual_guard_armed_m'] < o['actual_first_effect_m']
        assert o['actual_last_effect_m'] <= o['actual_input_restored_m']
        assert o['actual_proposal_durable_m'] <= o['actual_witness_m'] < end
        assert o['actual_reader_return_m'] < end
    else:
        assert o['public_delivery_observation'] in ('UNKNOWN', 'FAILED')
        assert o['no_late_delivered_persistence_or_presentation'] is True
        assert o['no_failure_promoted_from_json_or_task_done'] is True
    if o['cleanup_unconfirmed']:
        assert o['public_delivery_observation'] != 'DELIVERED'
        assert o['exact_owned_custody_retained'] is True
        # Such a run remains a red full gate; do not count correct refusal as
        # whole physical cleanup success merely because handles were retained.
        assert o['full_gate_acceptance'] is False
    if o['primary_exception'] is not None:
        assert o['reported_primary_exception'] is o['primary_exception']
