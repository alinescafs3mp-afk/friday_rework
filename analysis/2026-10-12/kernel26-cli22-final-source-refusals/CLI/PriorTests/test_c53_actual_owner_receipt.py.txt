@pytest.mark.parametrize('case',CASES)
def test_c53_actual_original_owner_receipt(existing_finite_cli_case,case):
    observation=existing_finite_cli_case.run_exact_current_source(case)
    assert observation.source_manifest_matches is True
    assert observation.actual_original_ids_preserved is True
    assert observation.no_developer_or_foreign_writer_effect is True
    assert observation.original_deadline_seconds<=30
    assert observation.no_attempt_or_clock_renewal is True
    assert observation.no_delivery_replay is True
    assert observation.durable_attempt_state=='UNKNOWN'
    if observation.presentation=='DELIVERED':
        assert observation.authenticated_original_outside_reader is True
        assert observation.actual_stock_raw_wait_status==0
        assert observation.actual_actor_raw_wait_status==0
        assert observation.actual_input_state_restored is True
        assert observation.actual_original_group_empty is True
        assert observation.actual_output_and_ledger_join_m<observation.original_deadline_m

        assert observation.actual_certified_m<observation.original_deadline_m
    else:
        assert observation.presentation in ('UNKNOWN','FAILED','NOT_RUN','PARTIAL')
    if observation.cleanup_unconfirmed:
        assert observation.presentation!='DELIVERED'
        assert observation.exact_owned_custody_retained is True
    if observation.primary_exception is not None:
        assert observation.reported_primary is observation.primary_exception
