"""Selected actual native API source with explicit inert UI/input/FD seams.

These are source controls, not installed input/renderer/PTY or kernel acceptance.
The whole synchronous consumer deadline remains unestablished.
"""
import ast
import asyncio
import contextlib
import hashlib
import io
import json
import pathlib
import sys
import time

ROOT = pathlib.Path(__file__).parent
SELECTED = pathlib.Path(sys.argv[1])
assert SELECTED in (ROOT / 'renderer-base', ROOT / 'renderer-source')
candidate = SELECTED.name == 'renderer-source'
path = SELECTED / 'check_owner_renderer.py'
tree = ast.parse(path.read_text())
last = next(i for i, node in enumerate(tree.body)
            if isinstance(node, ast.FunctionDef) and node.name == 'interleave')
nodes = [node for node in tree.body[:last + 1]
         if not isinstance(node, ast.For)
         and not (isinstance(node, ast.Expr) and isinstance(node.value, ast.Call)
                 and isinstance(node.value.func, ast.Name) and node.value.func.id == 'test')]
ns = {'__file__': str(path), '__name__': 'renderer_cleanup_selected_source'}
exec(compile(ast.Module(body=nodes, type_ignores=[]), str(path), 'exec'), ns)
observations = []


def record(name, fn):
    before = time.monotonic()
    value = fn()
    observations.append({'name': name, 'assertion': 'PASS_SOURCE',
                         'elapsed_seconds': time.monotonic() - before,
                         'observation': value})


def fault(stage, slow=False):
    f, cap, cli, ctx, adm, host = ns['setup']()
    with ns['frontend'](cli) as app, ns['input_turn'](cli):
        owner = cli._friday_cli_output
        counts = {'detach': 0, 'cooked': 0}
        slow_calls = []

        @contextlib.contextmanager
        def mode(name):
            app.events.append(name + ':enter')
            if stage == name + '_enter':
                raise RuntimeError('inert native input entry failure')
            counts[name] += 1
            try:
                yield
            finally:
                app.events.append(name + ':exit')
                if stage == name + '_exit':
                    raise RuntimeError('inert native input cleanup failure')
                counts[name] -= 1

        app.input.detach = lambda: mode('detach')
        app.input.cooked_mode = lambda: mode('cooked')
        if stage in ('reset', 'cursor', 'redraw'):
            def fail(*args, **kwargs):
                app.events.append(stage + ':failure')
                if slow:
                    # Deliberately blocks the loop. An asyncio deadline cannot
                    # preempt this native synchronous callback; preserve red.
                    began = time.monotonic()
                    slow_calls.append(True)
                    while time.monotonic() - began < .75:
                        pass
                raise RuntimeError('inert native renderer cleanup failure')
            if stage == 'reset': app.renderer.reset = fail
            elif stage == 'cursor': app._request_absolute_cursor_position = fail
            else: app._redraw = fail

        async def run():
            app.loop = asyncio.get_running_loop()
            original_route = ns['work'].route_for(ns['work'].current_turn())
            before = time.monotonic()
            receipt = await ns['document'](cli, ctx, timeout=.5)
            elapsed = time.monotonic() - before
            assert receipt['state'] == 'UNKNOWN', receipt
            if not slow:
                assert owner.closed.is_set() is candidate, (stage, candidate, owner.closed.is_set())
            elif candidate:
                assert owner.closed.is_set()
            assert not app._background_tasks and not owner.renderer_requests
            assert not owner.busy.locked() and not app._running_in_terminal
            assert app._running_in_terminal_f is None or app._running_in_terminal_f.done()
            assert not app.output._buffer
            original_text = io.StringIO.getvalue(owner.stream)
            second = await ctx.deliver_cli_document(route=original_route, data=b'new task document',
                                                    file_name='second.txt', timeout=.5)
            after_text = io.StringIO.getvalue(owner.stream)
            if candidate:
                assert second['state'] == 'FAILED', second
                assert original_text == after_text, 'unsafe frontend emitted another frame'
            if slow:
                assert slow_calls and elapsed > .5, (slow_calls, elapsed)
            return {'state': receipt['state'], 'owner_closed': owner.closed.is_set(),
                    'source_native_task_joined': True,
                    'input_seam_counts': counts, 'native_events': app.events,
                    'second_state': second['state'],
                    'second_bytes': len(after_text) - len(original_text),
                    'elapsed_seconds': elapsed,
                    'scaled_fixture_deadline_seconds': .5,
                    'synchronous_CPU_seam_seconds': .75 if slow else None,
                    'strict_deadline': 'SOURCE_FAIL_REPRODUCED' if slow else 'NOT_QUALIFIED',
                    'installed_native_input_cleanup': 'NOT_RUN'}
        return asyncio.run(run())


for stage in ('detach_enter', 'cooked_enter', 'cooked_exit', 'detach_exit',
              'reset', 'cursor', 'redraw'):
    record('native_' + stage + '_failure_retires_unsafe_frontend', lambda stage=stage: fault(stage))
record('synchronous_native_renderer_CPU_callback_overruns_asyncio_deadline',
       lambda: fault('redraw', True))
if candidate:
    record('on_time_native_renderer_memory_delivery_preserved', ns['case'])
    record('on_time_inert_nonblocking_FD_delivery_preserved', lambda: ns['case'](fd=True))
    record('native_predecessor_completion', ns['preceding'])
    record('cancelled_predecessor_not_cancelled_by_request', lambda: ns['preceding']('cancel'))
    record('native_owner_two_documents_serialize', ns['interleave'])
    record('unavailable_owner_loop_revokes_queued_start', ns['queued_start_cancel'])
    for stage in ('blocked', 'cancel', 'partial', 'error', 'revoke', 'replace_stream',
                  'flags', 'exit', 'replace_renderer', 'unload', 'call_ids',
                  'generation', 'original_app'):
        record('existing_' + stage + '_negative_preserved', lambda stage=stage: ns['case'](stage, True))

assert not ns['NATIVE_ERRORS'], ns['NATIVE_ERRORS']
assert all(not h.active for ctx in ns['bound']['n']['CONTEXTS']
           for h in ctx.handles if h.kind == 'task')
report = {'schema': 'friday.source.renderer-cleanup-controls.v1',
          'candidate': candidate, 'controls': observations,
          'source_partial_fix': 'UNSAFE_NATIVE_TEARDOWN_FRONTEND_RETIRED' if candidate else 'BASELINE_DEFECT_REPRODUCED',
          'whole_consumer_status': 'SOURCE_FAIL_STRICT_WHOLE_CONSUMER_BOUND_UNESTABLISHED',
          'pins': ns['PINS'], 'source_sha256': hashlib.sha256((SELECTED / 'successor/hermes/hermes_cli/friday_cli_output.py').read_bytes()).hexdigest(),
          'runtime_admission': False, 'normal_journeys': 'NOT_RUN'}
destination = ROOT / ('renderer-candidate-controls.json' if candidate else 'renderer-baseline-controls.json')
destination.write_text(json.dumps(report, indent=2) + '\n')
print(json.dumps({'report': str(destination), 'count': len(observations),
                  'whole_consumer_status': report['whole_consumer_status']}))
