def same_read_pin(ref,maximum=2097152):
    require(type(ref) is dict and set(ref)=={'path','bytes','sha256'},'exact pin')
    p=Path(ref['path']);require(p.is_absolute() and '..' not in p.parts
        and str(p)==os.path.normpath(str(p)),'absolute normalized pin')
    require(type(ref['bytes']) is int and 0<ref['bytes']<=maximum,'bounded pin bytes')
    for parent in p.parents:require(stat.S_ISDIR(parent.lstat().st_mode),'pin ancestor no symlink')
    fd=os.open(p,os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK)
    try:
        before=os.fstat(fd);require(stat.S_ISREG(before.st_mode) and before.st_nlink==1
            and before.st_size==ref['bytes'],'exact regular single-link pin')
        raw=bytearray()
        while len(raw)<=ref['bytes']:
            b=os.read(fd,min(65536,ref['bytes']+1-len(raw)))
            if not b:break
            raw.extend(b)
        after=os.fstat(fd)
        require((before.st_dev,before.st_ino,before.st_size,before.st_mtime_ns,before.st_ctime_ns)
                ==(after.st_dev,after.st_ino,after.st_size,after.st_mtime_ns,after.st_ctime_ns)
                and len(raw)==ref['bytes'] and hashlib.sha256(raw).hexdigest()==ref['sha256'],
                'same-read digest and kernel identity')
        return bytes(raw)
    finally:os.close(fd)

def existing_root_wait_owned(w):

            and existing_root_same_identity(w.root,exact_birth(os.getpid())), 'one actual existing Root wait')

        stage=0;terminal=False

                     w.clock['started_at_unix']+115-time.time())

                    except ProcessLookupError:pass

                    existing_root_set_timer(w,w.clock['started_at_monotonic']+115)
                stage+=1

        got,status=os.waitpid(w.stock['pid'],0);require(got==w.stock['pid'],'actual exact direct stock wait')
        w.wait_statuses.append({'pid':got,'status':status,'kind':'direct_stock'})

def existing_root_final_helper_release(w,helper,helper_pidfd,release_original_slot=True):

def existing_root_same_identity(a,b):

def kernel120_record(custody,directory,name,document):
    """Already-owned outside control directory only, bounded O_EXCL evidence."""
    require(type(name)is str and name and '/' not in name and name not in ('.','..'),'fixed component')
    raw=json.dumps(document,sort_keys=True,separators=(',',':'),allow_nan=False).encode()
    require(len(raw)<=65536,'bounded original-clock/control record')
    fd=root_fd_custody.opened(custody,'control_record:'+name,name,
            os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o400,dir_fd=directory)
    try:
        view=memoryview(raw)
        while view:
            n=os.write(fd,view);require(n>0,'bounded record progress');view=view[n:]
        os.fsync(fd)
    except BaseException as error:root_fd_custody.stopped(custody,error);raise
    else:root_fd_custody.closed(custody,fd)
    os.fsync(directory)
    path=str(Path('/proc/self/fd/'+str(directory)).resolve(strict=True)/name)
    return {'path':path,'bytes':len(raw),'sha256':hashlib.sha256(raw).hexdigest()}


def kernel120_original_clock_once(spec,ref,custody,control_fd):
    """Current never-run kernel120 clock, derived ONCE from real GNU parent birth.
    Not G7/G8/history, not percase, not a fresh now+120 after work has started.
    Absolute timer later uses this same origin; all32 use the same immutable pin.

            and type(guard['seconds'])is int,'one original GNU120 guardian')
    parent=exact_birth(os.getppid())
    require(parent['pid'] not in {r['pid'] for r in spec['existing_owner']['protected_writers']}
            and parent['state'] not in ('Z','X'),'actual existing GNU guardian excludes protected writers')
    command=[b'/usr/bin/gnutimeout',b'-s',b'KILL',b'120s',PYTHON.encode(),b'-I',b'-B',
             guard['owner_source']['path'].encode(),ref['path'].encode(),
             str(ref['bytes']).encode(),ref['sha256'].encode()]
    require(Path('/proc/'+str(parent['pid'])+'/cmdline').read_bytes().split(b'\0')[:-1]==command,
            'fixed exterior GNU parent exact command; no invented host timer')

    now_m=time.monotonic();now_u=time.time()

    start_u=now_u-(now_m-start_m)
    require(0<=now_m-start_m<5 and 0<=now_u-start_u<5,'new original guardian, no old replay/reset')
    clock={'original_seconds':120,'cleanup_seconds':30,'started_at_monotonic':start_m,
           'started_at_unix':start_u,'useful_deadline_unix':start_u+90,'final_deadline_unix':start_u+120,
           'boot_id':Path('/proc/sys/kernel/random/boot_id').read_text().strip(),
           'origin':'kernel120_original_clock_once','actual_guardian_birth':parent,'clock_reset':False}

def kernel120_atomic_named_child(custody,label):
    """Only stock/finalizer fixed roles, atomic CLONE_PIDFD; no fork handle gap."""
    import ctypes,signal,platform
    require(label in ('kernel120_stock','kernel120_post_terminal_helper'),'two fixed existing-owner roles')
    require(platform.machine()=='x86_64','reviewed clone3 ABI, no fallback')
    class CloneArgs(ctypes.Structure):
        _fields_=[(key,ctypes.c_uint64) for key in
            ('flags','pidfd','child_tid','parent_tid','exit_signal','stack','stack_size','tls',
             'set_tid','set_tid_size','cgroup')]
    fd=ctypes.c_int(-1);args=CloneArgs();args.flags=0x1000
    args.pidfd=ctypes.addressof(fd);args.exit_signal=signal.SIGCHLD
    libc=ctypes.CDLL(None,use_errno=True);libc.syscall.restype=ctypes.c_long
    pid=libc.syscall(ctypes.c_long(435),ctypes.byref(args),ctypes.c_size_t(88))
    if pid<0:raise OSError(ctypes.get_errno(),'fixed clone3 refused; no retry/fallback')
    if pid>0:
        root_fd_custody.own(custody,fd.value,label+':pidfd') # FIRST after successful acquisition.
        custody.setdefault('birth_attempts',[]).append({'pid':int(pid),'pidfd':fd.value,'role':label})
    return int(pid),fd.value

        except BaseException:os._exit(125)
    helper=exact_birth(pid);require(helper['ppid']==root['pid'],'actual direct post-terminal helper')

    require(os.write(boot_w,b'G')==1,'one helper GO after actual Root output closes')
