import copy


def refusal(batch,code,call):
    try:call()
    except batch.BatchRefusal as error:assert error.code==code
    else:raise AssertionError('intended refusal absent')


def test_c51_full_scope_metadata(batch,catalog):
    plan=batch.make_original32_plan(catalog)
    assert plan['case_ids']==[r[0] for r in batch.CASES] and len(plan['case_ids'])==32
    assert len([r for r in batch.CASES if r[1]=='positive'])==10
    assert len([r for r in batch.CASES if r[1]=='negative'])==18
    assert len([r for r in batch.CASES if r[1]=='source_uncertainty_after_actual_operation'])==4
    assert plan['provider_max_simultaneous_cohorts']==1 and plan['actual_complete_physical_fit']=='UNPROVEN'


def test_c51_planned_catalog_is_not_actual(batch,catalog):
    plan=batch.make_original32_plan(catalog)
    assert not catalog.actual and all(r['actual_entry'] is None for r in catalog.planned.values())
    assert all(r['original_clock_ref']==catalog.clock_pin for r in catalog.planned.values())
    refusal(batch,'CLOCK_CASE_MAPPING_UNBOUND',lambda:batch.original_case_clock_bindings({'original_catalog_object':catalog},catalog.clock_pin))


def test_c51_exact_serial_cut_refusal(batch,owner,catalog):
