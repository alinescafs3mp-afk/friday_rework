class BatchRefusal(ValueError):
    def __init__(self,code,detail):
        self.code=code
        super().__init__(code+': '+detail)


def need(ok,code,detail):
    if not ok:raise BatchRefusal(code,detail)


def finite(v):return type(v) in (int,float) and math.isfinite(v)








def drive_all32_actual_assertions(owner,tests,Root_expected,trace_refs,clock_pin):
    """Existing Root actual trace interface, not an executor/provider substitute.
    Verify all real pinned inputs first; call every unchanged intended assertion
    on its own complete actual Root trace. No partial total or unrelated-error
    pass. A shared positive cohort still has ten distinct cause observations.
    Source author cannot use this as independent final acceptance.
    """
    import json
    ids=[x[0] for x in CASES]
    mapping=original_case_clock_bindings(Root_expected,clock_pin)
    need(type(Root_expected)is dict and Root_expected.get('independent_final_Source_safety_acceptance') is not None
         and Root_expected.get('actual_specific_kernel120_admission') is not None,
         'BATCH_NOT_ADMITTED','outside Root actual final body/action admission required')
    need(Root_expected.get('actual_complete_batch_finished_under_original_clock_ref') is not None,
         'BATCH_PHYSICAL_FIT_UNBOUND','actual full32 oneclock completion prerequisite missing')
    need(type(trace_refs)is dict and set(trace_refs)==set(ids),'BATCH_SCOPE','exact all32 Root raw trace pins')
    traces={};functions={}
    for case_id,kind,name,cause in CASES:
        trace=json.loads(owner.same_read_pin(trace_refs[case_id],65536))
        need(trace.get('case_id')==case_id and trace.get('intended_cause')==cause
             and trace.get('original_clock_ref')==mapping[case_id]['original_clock_ref']
             and trace.get('Root_expected_original_clock_ref')==mapping[case_id]['Root_expected_original_clock_ref']
             and trace.get('simulated') is False and trace.get('unknown_as_zero') is False
             and trace.get('old_clock_or_Source_credit_transferred') is False,
             'BATCH_TRACE_CAUSE','actual exact case/clock/cause, no Source history transfer')
        if kind=='source_uncertainty_after_actual_operation':
            need(trace.get('accepted_original_clock_elapsed_at_entry')
                 ==mapping[case_id]['accepted_original_clock_elapsed_at_entry'],
                 'BATCH_TRACE_CAUSE','same actual original producer entry elapsed, no substituted batch anchor')
        function=getattr(tests,name,None)
        need(callable(function),'BATCH_ASSERTION_MISSING',name)
        traces[case_id]=trace;functions[case_id]=function
    result=[]
    for case_id,kind,name,cause in CASES:
        try:functions[case_id](traces[case_id])
        except Exception as error:
            result.append({'id':case_id,'kind':kind,'state':'FAIL','error_type':type(error).__name__})
        else:result.append({'id':case_id,'kind':kind,'state':'PASS'})
    need(len(result)==32,'BATCH_SCOPE','full mandatory assertions, no early green partial return')
    return {'results':result,'passed':sum(r['state']=='PASS' for r in result),
            'failed':sum(r['state']=='FAIL' for r in result),'total':32,'clock_reset':False,
            'independent_acceptance':'ROOT_SEPARATE_REVIEW_REQUIRED','capacity_release':False,
            'native_genuine_or_installed_runtime_acceptance':False}





class Original32Catalog:
    """Original shared clock with separately frozen plans and actual observations.
    No future timestamp is an actual entry and no per-case clock is constructed.
    """
    def __init__(self,clock,clock_pin,producer_pin,host_pin,planned,frozen_ref):
        self.clock=dict(clock);self.clock_pin=dict(clock_pin);self.producer_pin=dict(producer_pin)
        self.host_pin=dict(host_pin);self.planned=planned;self.frozen_ref=frozen_ref
        self.actual={};self.phase_intervals=[];self.active_phase=None


def produce_original32_catalog(owner,spec,clock,clock_pin,custody,control_fd):
    import json
    need(clock==json.loads(owner.same_read_pin(clock_pin,65536)),'ORIGINAL_CLOCK','same-read current original guardian clock')
    need(clock['original_seconds']==120 and clock['cleanup_seconds']==30
         and clock['final_deadline_unix']==clock['started_at_unix']+120
         and clock['useful_deadline_unix']==clock['started_at_unix']+90,
         'ORIGINAL_CLOCK','one kernel120INCLUDING30, no per-case120')
    producer=spec['kernel120_dependency_pins']['qualification_batch'];owner.same_read_pin(producer)
    planned={}
    for case_id,kind,name,cause in CASES:
        low,high=(19,21) if kind=='source_uncertainty_after_actual_operation' else (0,45 if kind=='positive' else 90)
        planned[case_id]={'case_id':case_id,'kind':kind,'cause':cause,'original_clock_ref':clock_pin,
          'Root_expected_original_clock_ref':clock_pin,'original_clock_producer_ref':producer,
          'planned_age_window':[low,high],'planned_entry_window_m':[clock['started_at_monotonic']+low,clock['started_at_monotonic']+high],
          'planned_entry_window_u':[clock['started_at_unix']+low,clock['started_at_unix']+high],
          'actual_entry':None,'clock_reset':False}
    document={'schema':'kernel120-original32-planned-catalog.v1','clock_pin':clock_pin,
       'clock_origin':clock['origin'],'producer_pin':producer,'case_ids':[r[0] for r in CASES],
       'planned':planned,'actual_entries_exist':False,'historical_common_anchor_claimed':False,
       'provider_max_simultaneous_cohorts':1,'RootGO':None}
    frozen=owner.kernel120_record(custody,control_fd,'kernel120-original32-catalog.json',document)
    return Original32Catalog(clock,clock_pin,producer,spec['owner_host'],planned,frozen)


def make_original32_plan(catalog):
    need(type(catalog)is Original32Catalog and not catalog.actual,'ORIGINAL_CATALOG','new frozen planned catalog only')
    positive=[r[0] for r in CASES if r[1]=='positive'];negative=[r[0] for r in CASES if r[1]=='negative']
    cuts=[r[0] for r in CASES if r[1]=='source_uncertainty_after_actual_operation']
    phases=[{'case_ids':positive,'first_birth_required':True,'probe_mode':'ALL_SAFE','phase_id':'ALL_SAFE'}]
    phases.extend({'case_ids':[i],'first_birth_required':i=='K120_PIDFD_OR_NAMESPACE_UNSUPPORTED',
                   'phase_id':i} for i in negative)
    phases.extend({'case_ids':[i],'first_birth_required':True,'phase_id':i} for i in cuts)
    return {'case_ids':[r[0] for r in CASES],'original_clock_pin':catalog.clock_pin,'catalog_ref':catalog.frozen_ref,
      'catalog_object':catalog,'phases':phases,'provider_max_simultaneous_cohorts':1,
      'whole_memory':2147483648,'swap':0,'Linux_tasks':128,'cpu_max':'200000 100000','AS_per_outside_role':268435456,
      'whole_lifetime_IO_upper':356515840,'whole_lifetime_inode_upper':3072,
      'declared_worst_time_bounds':[244,305],'declared_worst_is_measured_minimum':False,
      'actual_complete_physical_fit':'UNPROVEN','reset_clock':False}


def assert_complete_plan(plan,original_clock_pin,absolute,now_m,now_u,Root_expected=None):
    need(type(plan)is dict and plan.get('original_clock_pin')==original_clock_pin
         and plan.get('reset_clock') is False,'BATCH_ORIGINAL_CLOCK','one current clock, no renewal')
    catalog=plan.get('catalog_object')
    need(type(catalog)is Original32Catalog and catalog.clock_pin==original_clock_pin
         and Root_expected.get('original_catalog_object') is catalog,
         'ORIGINAL_CATALOG','actual current producer object, not a JSON/boolean grant')
    ids=[r[0] for r in CASES]
    need(plan['case_ids']==ids and len(set(ids))==32 and set(catalog.planned)==set(ids),
         'BATCH_SCOPE','original32/10+18+4 retained')
    phases=plan['phases'];seen=[i for p in phases for i in p['case_ids']]
    need(len(seen)==len(set(seen))==32 and set(seen)==set(ids),'BATCH_SCOPE','each original cause once')
    need(phases[0]['case_ids']==[r[0] for r in CASES if r[1]=='positive']
         and phases[0]['probe_mode']=='ALL_SAFE','BATCH_POSITIVE_COHORT','all10 true intended observations')
    cuts=[r[0] for r in CASES if r[1]=='source_uncertainty_after_actual_operation']
    need(all(any(p['case_ids']==[i] and p['first_birth_required'] for p in phases) for i in cuts),
         'BATCH_OWNER_CUT','four distinct real attempts, no four labels on one owner')
    need(plan['provider_max_simultaneous_cohorts']==1 and type(plan['provider_max_simultaneous_cohorts'])is int,
         'SERIAL_OWNER','one actual active owner, no topology change')
    need(plan['whole_memory']==2147483648 and plan['swap']==0 and plan['Linux_tasks']==128
         and plan['cpu_max']=='200000 100000' and plan['AS_per_outside_role']==268435456,
         'BATCH_RESOURCES','unchanged resource domains')
    need(now_m<absolute['useful_m'] and now_u<absolute['useful_u'],'BATCH_TIME','original useful remains')
    for row in catalog.planned.values():
        need(row['actual_entry'] is None and row['original_clock_ref']==original_clock_pin,
             'PLANNED_IS_NOT_ACTUAL','frozen plans cannot substitute future observation')
    # Reservations are declarations, not predicted actual durations. Serial
    # ownership is checked from actual closure before each new attempted phase.
    return {'total':32,'positive':10,'negative':18,'owner_cuts':4,'physical_fit':'UNPROVEN',
            'clock_reset':False,'capacity_release':False,'worst_bounds_are_not_minima':True}


def original32_serial_cut_conflict(owner,catalog):
    """Concrete source/spec contradiction, independent of declared244/305.
    After every injected uncertainty the unchanged stock host retains itself
    until its real guardian. That expiry must be strictly > actualentry+50.
    A second serial cut cannot enter the same original [19,21] window.
    """
    import ast
    need(type(catalog)is Original32Catalog,'ORIGINAL_CATALOG','same actual original producer')
    raw=owner.same_read_pin(catalog.host_pin);text=raw.decode();tree=ast.parse(text)
    functions={n.name:n for n in tree.body if isinstance(n,ast.FunctionDef)}
    hold=functions['retain_until_stock_guard'];host=functions['kernel120_stock_first_birth']
    need(any(isinstance(n,ast.While) and isinstance(n.test,ast.Constant) and n.test.value is True
             for n in ast.walk(hold))
         and any(isinstance(n,ast.Call) and isinstance(n.func,ast.Attribute) and n.func.attr=='pause'
                 for n in ast.walk(hold)),'CUT_CONTRACT_DRIFT','original guard-retaining Source changed')
    body=ast.get_source_segment(text,host)
    need("45<n['stock_guard_seconds']" in body
         and "n['stock_guard_seconds']+5<=root_guard['seconds']" in body
         and "time.monotonic()+n['stock_guard_seconds']+5<=guardian_expiry_m" in body
         and 'retain_until_stock_guard()' in body,'CUT_CONTRACT_DRIFT','exact unchanged original child/guardian/hold predicates')
    cuts=[r[0] for r in CASES if r[1]=='source_uncertainty_after_actual_operation']
    need(len(cuts)==4 and all(catalog.planned[i]['planned_age_window']==[19,21] for i in cuts),
         'CUT_CONTRACT_DRIFT','four original one-clock windows retained')
    return {'code':'BATCH_ONE_CLOCK_SERIAL_CUT_CONFLICT','clock_pin':catalog.clock_pin,
      'host_pin':catalog.host_pin,'cut_count':4,'window_seconds':2,
      'first_cut_actual_terminal_lower_bound_after_entry_seconds':50,
      'lower_bound_strict':True,'why':'actual source-retained stock cannot voluntarily finish before original guardian expiry; expiry>entry+50',
      'serial_owner_count':1,'worst244_305_used_as_minimum':False,'first_birth_attempted':False,
      'necessary_owner_decision':'Resolve the four same-original19..21 entry windows versus serial guard-retained attempts. No threshold/clock/topology changes applied.'}


def before_any_batch_GO(Root_expected,plan,clock_pin,absolute,now_m,now_u):
    need(Root_expected.get('independent_current_source_acceptance') is not None
         and Root_expected.get('actual_specific_action_admission') is not None,
         'BATCH_NOT_ADMITTED','external exact safety/action approval, never produced here')
    assert_complete_plan(plan,clock_pin,absolute,now_m,now_u,Root_expected)
    owner=Root_expected['actual_accepted_same_read_owner_object'];catalog=Root_expected['original_catalog_object']
    conflict=original32_serial_cut_conflict(owner,catalog)
    raise BatchRefusal(conflict['code'],conflict['why']) # BEFORE stock/helper/opaque/kernel GO.


def capture_original_case_clock(owner,case_id,spec,original_case_catalog,original_global_clock):
    import json,time
    c=original_case_catalog
    need(type(c)is Original32Catalog and case_id in c.planned and case_id not in c.actual,
         'ORIGINAL_CATALOG','one actual entry per cause, no replay or guessed historical row')
    need(spec['clock']==c.clock_pin and original_global_clock==c.clock
         and json.loads(owner.same_read_pin(spec['clock'],65536))==c.clock,
         'ORIGINAL_CLOCK','same immutable current original clock argument')
    clock=c.clock;m=time.monotonic();u=time.time();age=m-clock['started_at_monotonic']
    need(0<=age<120 and abs(age-(u-clock['started_at_unix']))<=2,'ORIGINAL_CLOCK','actual captured dualclock')
    kind=next(r[1] for r in CASES if r[0]==case_id)
    if kind=='source_uncertainty_after_actual_operation':
        need(19<=age<=21 and 19<=u-clock['started_at_unix']<=21,'BATCH_OWNER_CUT_WINDOW','original four19..21 preserved')
    need(c.active_phase is None,'SERIAL_OWNER','previous phase must have actual fresh closure before next entry')
    row={'original_clock_ref':c.clock_pin,'Root_expected_original_clock_ref':c.clock_pin,
      'original_clock_producer_ref':c.producer_pin,'catalog_frozen_before_batch_ref':c.frozen_ref,
      'original_seconds':120,'cleanup_seconds':30,'clock_start_m':clock['started_at_monotonic'],
      'clock_start_u':clock['started_at_unix'],'entry_m':m,'entry_u':u,'accepted_original_clock_elapsed_at_entry':age,
      'original_final_m':clock['started_at_monotonic']+120,'original_final_u':clock['final_deadline_unix'],
      'original_useful_m':clock['started_at_monotonic']+90,'original_useful_u':clock['useful_deadline_unix'],
      'clock_reset':False,'new_actual_entry_producer':'capture_original_case_clock','capacity_release':False}
    c.actual[case_id]=row;c.active_phase=case_id
    return row


def original_case_clock_bindings(Root_expected,clock_pin):
    catalog=Root_expected.get('original_catalog_object')
    need(type(catalog)is Original32Catalog and catalog.clock_pin==clock_pin
         and set(catalog.actual)=={r[0] for r in CASES} and catalog.active_phase is None,
         'CLOCK_CASE_MAPPING_UNBOUND','actual all32 entries are a final observation prerequisite, never a pre-GO plan')
    return catalog.actual


def execute_original32_serial(owner,adapter,base,Root_expected,clock,catalog):
    """Fixed named caller with actual serial closure; no generic launch callback.
    Currently blocked BEFORE reaching this by exact original source conflict.
    Actual fixture specs/raw traces still require Root-specific outside admission.
    No Source-only or candidate-return acceptance is manufactured here.
    """
    import time
    plan=make_original32_plan(catalog)
    need(set(Root_expected['actual_phase_specs'])=={p['phase_id'] for p in plan['phases']},
         'ACTUAL_INPUT_BINDINGS','all Root-owned real private fixtures required, no mocks')
    outcomes=[]
    for phase in plan['phases']:
        need(catalog.active_phase is None,'SERIAL_OWNER','one actual owner until outside closure')
        need(time.monotonic()+45<clock['started_at_monotonic']+90
             and time.time()+45<clock['useful_deadline_unix'],
             'BATCH_REAL_BUDGET_EXHAUSTED','original remaining45 before next attempt; no +120/reset')
        spec=Root_expected['actual_phase_specs'][phase['phase_id']]
        phase_context=dict(Root_expected,**Root_expected['actual_phase_capabilities'][phase['phase_id']])
        need(phase_context['retained_fd_custody'] is Root_expected['retained_fd_custody'],'SERIAL_OWNER','same exterior custody/slot across phases')
        need(spec['clock']==catalog.clock_pin,'ORIGINAL_CLOCK','same global original argument for every phase')
        result=adapter.kernel120_named_root_case_adapter(owner,spec,phase_context,clock,catalog,phase['case_ids'][0])
        need(result['actual_cohort_closure'] is True and result['capacity_release'] is False
             and result['original_slot_retained'] is True,'SERIAL_OWNER','actual cohort cleanup before next phase; same original batch slot retained')
        terminal_m=time.monotonic();terminal_u=time.time()
        need(terminal_m<clock['started_at_monotonic']+120 and terminal_u<clock['final_deadline_unix'],
             'BATCH_REAL_BUDGET_EXHAUSTED','truthful original final; no fit promise')
        original=catalog.actual[phase['case_ids'][0]]
        for case_id in phase['case_ids'][1:]:catalog.actual[case_id]=dict(original)
        catalog.phase_intervals.append({'case_ids':phase['case_ids'],'entry_m':original['entry_m'],
             'entry_u':original['entry_u'],'terminal_m':terminal_m,'terminal_u':terminal_u})
        catalog.active_phase=None;outcomes.append(result)
    # All attempted cohorts/helper waits are terminal/fresh before LAST original batch slot close.
    need(catalog.active_phase is None,'SERIAL_OWNER','no remaining current cohort')
    owner.root_fd_custody.require_custody(Root_expected['retained_fd_custody'])
    slot=base['existing_owner']['slot_fd']
    need(all(r['closed'] or r['fd']==slot or r['label']=='exterior_evidence'
             for r in Root_expected['retained_fd_custody']['fds']),'SERIAL_OWNER','all effect handles closed once; ambiguity never releases')
    owner.root_fd_custody.closed(Root_expected['retained_fd_custody'],slot)
    return {'outcomes':outcomes,'case_clock_bindings':catalog.actual,'actual_original_scope':32,
       'physical_fit':'ROOT_MUST_INDEPENDENTLY_VERIFY','all32_assertion_acceptance':'PENDING_ROOT_RAW_TRACE_REVIEW',
       'clock_reset':False,'native_genuine_or_installed_runtime_acceptance':False}
