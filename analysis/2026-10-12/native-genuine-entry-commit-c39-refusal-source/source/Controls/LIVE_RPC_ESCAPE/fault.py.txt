def identity(pid):
    v = Path('/proc/' + str(pid) + '/stat').read_text().rsplit(')', 1)[1].split()
    return {'pid': pid, 'birth': v[19], 'ppid': int(v[1]), 'pgid': int(v[2]), 'sid': int(v[3])}


def read_pin(ref, maximum=1048576):
    require(type(ref) is dict and set(ref) == {'path', 'bytes', 'sha256'}, 'pin shape')
    p = Path(ref['path'])
    require(p.is_absolute() and '..' not in p.parts and str(p) == os.path.normpath(str(p)), 'pin path')
    require(type(ref['bytes']) is int and 0 <= ref['bytes'] <= maximum, 'pin size')
    for parent in reversed(p.parents):
        require(stat.S_ISDIR(parent.lstat().st_mode), 'pin ancestor')
    fd = os.open(p, os.O_RDONLY | os.O_NONBLOCK | os.O_NOFOLLOW)
    try:
        a = os.fstat(fd)
        require(stat.S_ISREG(a.st_mode) and a.st_nlink == 1 and a.st_uid == os.getuid()
                and not a.st_mode & 0o077 and a.st_size <= maximum, 'private regular pin')
        chunks, size = [], 0
        while True:
            b = os.read(fd, min(8192, maximum + 1 - size))
            if not b:
                break
            chunks.append(b)
            size += len(b)
            require(size <= maximum, 'pin read bound')
        raw = b''.join(chunks)
        key = lambda s: (s.st_dev, s.st_ino, s.st_size, s.st_mode, s.st_mtime_ns, s.st_ctime_ns)
        require(key(a) == key(os.fstat(fd)) == key(p.lstat()), 'pin drift')
        require(len(raw) == ref['bytes'] and hashlib.sha256(raw).hexdigest() == ref['sha256'], 'pin mismatch')
        return raw
    finally:
        os.close(fd)


def put(path, value):
    raw = (json.dumps(value, sort_keys=True, allow_nan=False) + '\n').encode()
    require(len(raw) <= LIMIT, 'evidence bound')
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o400)
    with os.fdopen(fd, 'wb') as f:
        f.write(raw)
        f.flush()
        os.fsync(f.fileno())


def caller_limits():
    for kind, cap in ((resource.RLIMIT_AS, AS), (resource.RLIMIT_CPU, 60), (resource.RLIMIT_FSIZE, MAX)):
        require(resource.getrlimit(kind) == (cap, cap), 'hard caller resources before Source')
    require(resource.getrlimit(resource.RLIMIT_CORE) == (0, 0)
            and 0 < len(os.sched_getaffinity(0)) <= 2, 'caller core/affinity')
    return {'pid': os.getpid(), 'birth': identity(os.getpid())['birth'],
            'AS': [AS, AS], 'CPU': [60, 60], 'FSIZE': [MAX, MAX], 'CORE': [0, 0],
            'affinity': sorted(os.sched_getaffinity(0))}

    def _observe(self, phase):

                    'same actual caller and host births in observation')

                                    'direct_wait_observed': c.waited, 'fresh_birth_absent': c.child_absent,
                                    'process_state_is_never_reap_or_release_authority': True}
                require(len(c.fds) <= 24, 'bounded caller observation handles')
                for name, fd in c.fds.items():
                    if fd is None:
                        continue
                    row = {'fd': fd, 'cached_identity': c.fd_ids.get(name), 'current_identity': 'UNKNOWN',
                           'target': 'UNKNOWN', 'fdinfo': 'UNKNOWN'}
                    for key in ('current_identity', 'target', 'fdinfo'):
                        try:
                            if key == 'current_identity':
                                s = os.fstat(fd)
                                row[key] = [s.st_dev, s.st_ino, stat.S_IFMT(s.st_mode)]
                                if c.fd_ids.get(name) is not None:
                                    require(row[key] == list(c.fd_ids[name]), 'retained caller FD identity drift')
                            elif key == 'target':
                                row[key] = os.readlink('/proc/self/fd/' + str(fd))
                            else:

                                    raw = f.read(4097)
                                require(len(raw) <= 4096, 'bounded caller fdinfo')
                                row[key] = raw.decode('utf-8', 'replace')
                        except BaseException as exc:
                            frame['errors'].append({'label': name, 'field': key, 'class': type(exc).__name__})
                    frame['custody']['retained_fds'][name] = row
            raw = (json.dumps(frame, sort_keys=True, allow_nan=False) + '\n').encode()
            require(len(raw) <= 16384 and self.observation_bytes + len(raw) <= LIMIT, 'whole caller observation output')
            self.observation_bytes += len(raw)
            offset = 0
            while offset < len(raw):
                n = os.write(1, raw[offset:])
                require(n > 0, 'zero caller observation write')
                offset += n
            if frame['errors']:
                self.observation_errors.extend(frame['errors'][:16])
                return False
            return True
        except BaseException as exc:
            self.observation_errors.append({'phase': phase, 'class': type(exc).__name__})
            return False

    def _observe_or_retain(self, phase):

            # already admitted host deadline/parent-death/stock guard remain live.
            while True:
                try:
                    signal.pause()
                except BaseException:
                    pass  # Existing host/stock stop bounds the SAME caller.

    def native_scan(self,reap=True):

def native_sandbox(temp_root):
    # Kernel confinement, no fabricated Seccomp/namespace JSON authority.
    import platform
    require(platform.machine()=='x86_64','audited syscall table only x86_64')
    libc=ctypes.CDLL(None,use_errno=True)
    require(libc.prctl(38,1,0,0,0)==0,'actual no_new_privs')
    abi=libc.syscall(444,0,0,1);require(abi>=3,'Landlock truncate/refer support required')
    class Ruleset(ctypes.Structure):_fields_=[('handled_access_fs',ctypes.c_uint64)]
    class PathRule(ctypes.Structure):
        _pack_=1
        _fields_=[('allowed_access',ctypes.c_uint64),('parent_fd',ctypes.c_int)]
    mask=(1<<1)|sum(1<<i for i in range(4,15));rs=Ruleset(mask)
    rules_fd=libc.syscall(444,ctypes.byref(rs),ctypes.sizeof(rs),0);require(rules_fd>=0,'Landlock ruleset actual create')
    root_fd=os.open(temp_root,os.O_PATH|os.O_DIRECTORY|os.O_NOFOLLOW)
    rule=PathRule(mask&~((1<<6)|(1<<11)),root_fd)
    require(libc.syscall(445,rules_fd,1,ctypes.byref(rule),0)==0,'private temp write rule')
    require(libc.syscall(446,rules_fd,0)==0,'actual inherited Landlock restriction')
    os.close(root_fd);os.close(rules_fd)
    class Filter(ctypes.Structure):_fields_=[('code',ctypes.c_ushort),('jt',ctypes.c_ubyte),('jf',ctypes.c_ubyte),('k',ctypes.c_uint)]
    class Prog(ctypes.Structure):_fields_=[('len',ctypes.c_ushort),('filter',ctypes.POINTER(Filter))]
    code=[(0x20,0,0,4),(0x15,1,0,0xc000003e),(0x06,0,0,0x80000000),(0x20,0,0,0)]
    for nr in [101,155,161,165,166,169,175,176,246,272,303,304,308,311,313,321]:
        code.extend([(0x15,0,1,nr),(0x06,0,0,0x00050001)])
    code.extend([(0x15,0,1,435),(0x06,0,0,0x00050026)]) # clone3 ENOSYS, preserve libc fallback.
    code.extend([(0x15,0,4,56),(0x20,0,0,16),(0x45,0,1,0x7e020000),(0x06,0,0,0x00050001),(0x20,0,0,0)])
    code.extend([(0x15,0,3,41),(0x20,0,0,16),(0x15,1,0,1),(0x06,0,0,0x00050001),(0x06,0,0,0x7fff0000)])
    arr=(Filter*len(code))(*(Filter(*x) for x in code));prog=Prog(len(code),arr)
    require(libc.prctl(22,2,ctypes.byref(prog),0,0)==0,'actual inherited namespace/network escape filter')

def native_caller(spec,launch):

    # Bootstrap is already finite and Root's host owns this exact Popen before
    # native GO; no updater/dependency import or clone has happened.

    os.close(go);os.close(ready)
    resource.setrlimit(resource.RLIMIT_AS,(1073741824,1073741824));resource.setrlimit(resource.RLIMIT_CPU,(120,120))
    libc=ctypes.CDLL(None,use_errno=True)

    require(fcntl.fcntl(capfd,fcntl.F_GET_SEALS)==seals,'inherited sealed capture snapshot')

            'same sealed capture bytes')
