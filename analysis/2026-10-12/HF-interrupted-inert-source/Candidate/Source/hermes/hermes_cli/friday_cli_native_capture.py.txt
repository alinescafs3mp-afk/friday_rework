"""Optional original native observation seam; no job/authority/clock created."""
from contextvars import ContextVar
from contextlib import contextmanager
_CAPTURE=ContextVar('friday_original_native_capture',default=None)

@contextmanager
def installed_original_capture(producer):
    from live_original_capture import OriginalNativeProducer
    from hermes_cli.friday_cli_principal import current_cli
    from hermes_cli.friday_cli_finite import RetainedOwnerChannel
    from cli import HermesCLI
    # The actual ordinary bootstrap and constructor already own these objects.
    # Installation observes them; it cannot create a principal/UI/native call.
    cap=current_cli(required=True)
    port=getattr(producer.cli,'_friday_cli_finite_port',None)
    if (type(producer) is not OriginalNativeProducer
            or not isinstance(producer.cli,HermesCLI)
            or producer.cli._friday_cli_scope is not cap
            or type(port) is not RetainedOwnerChannel
            or port.cli is not producer.cli or port.cap is not cap):
        raise RuntimeError('actual_original_capture_scope_required')
    port._check()
    if _CAPTURE.get() is not None:
        raise RuntimeError('original_capture_cannot_replace_existing')
    token=_CAPTURE.set(producer)
    try:yield producer.capture
    finally:_CAPTURE.reset(token)

def begin_original_call(authority):
    producer=_CAPTURE.get()
    if producer is None:return None
    case=producer.begin(authority)
    if case is None:return None
    authority['_native_capture_case']=case
    authority['agent']._friday_cli_native_capture_producer=producer
    return case

def after_original_handler(result):
    from hermes_cli.friday_cli_work import _TOOL
    authority=_TOOL.get()
    if authority is None:return
    case=authority.get('_native_capture_case')
    if case is not None:case.observe_after_handler(result)

def primary_original_failure(authority,failure):
    case=authority.get('_native_capture_case') if authority is not None else None
    if case is not None:case.observe_primary(failure)

def finish_original_caller(agent,call_id):
    producer=getattr(agent,'_friday_cli_native_capture_producer',None)
    if producer is not None:producer.finish(agent,call_id)

def prepare_original_terminal(agent,call_id):
    producer=getattr(agent,'_friday_cli_native_capture_producer',None)
    if producer is not None:producer.prepare_terminal(agent,call_id)


from enum import Enum
from dataclasses import dataclass
import os,time,sys,inspect

class NativeIOStage(Enum):
    RECORD_DIRECTORY_OPENED='record_directory_opened'
    STATE_FILE_FSYNC='state_file_fsync'
    STATE_DIRECTORY_FSYNC='state_directory_fsync'
    OUTPUT_WRITE='output_write'
    METADATA_CALLBACK='metadata_callback'
    SESSION_DB_FLUSH='session_db_flush'
    UI_COMPLETED='ui_completed'

@dataclass(frozen=True)
class OriginalIOEvent:
    stage: NativeIOStage
    caller_code: object
    monotonic: float
    pid: int
    thread_id: int
    owner: object
    fd: object
    fd_stat: object

_STAGE_CALLERS={
    NativeIOStage.RECORD_DIRECTORY_OPENED:('hermes_cli.friday_cli_finite_owner','current_record_generation'),
    NativeIOStage.STATE_FILE_FSYNC:('hermes_cli.friday_cli_finite_owner','_read_private_at'),
    NativeIOStage.STATE_DIRECTORY_FSYNC:('hermes_cli.friday_cli_finite_owner','_read_private_at'),
    NativeIOStage.OUTPUT_WRITE:('hermes_cli.friday_cli_output','_ConsoleWrite.write'),
    NativeIOStage.METADATA_CALLBACK:('agent.tool_executor','_commit_tool_result'),
    NativeIOStage.SESSION_DB_FLUSH:('agent.tool_executor','_commit_tool_result'),
    NativeIOStage.UI_COMPLETED:('agent.tool_executor','_commit_tool_result'),
}

def observe_original_io(stage,owner=None,fd=None,*,agent=None,call_id=None):
    producer=_CAPTURE.get()
    if producer is None and agent is not None:
        producer=getattr(agent,'_friday_cli_native_capture_producer',None)
    if producer is None:return
    if type(stage) is not NativeIOStage:raise RuntimeError('typed_native_stage_required')
    module,qualname=_STAGE_CALLERS[stage]
    caller=sys._getframe(1)
    actual=sys.modules.get(module)
    if actual is None or caller.f_globals is not actual.__dict__:
        raise RuntimeError('foreign_native_stage_caller')
    method=actual
    for name in qualname.split('.'):method=inspect.getattr_static(method,name)
    if caller.f_code is not method.__code__:
        raise RuntimeError('substitute_native_stage_code')
    # These are real source-site observations, not test-name providers.
    st=os.fstat(fd) if fd is not None else None
    import threading
    event=OriginalIOEvent(stage,caller.f_code,time.monotonic(),os.getpid(),
        threading.get_native_id(),owner,fd,
        (st.st_dev,st.st_ino,st.st_mode,st.st_rdev) if st is not None else None)
    producer.observe_io(event,agent,call_id)
