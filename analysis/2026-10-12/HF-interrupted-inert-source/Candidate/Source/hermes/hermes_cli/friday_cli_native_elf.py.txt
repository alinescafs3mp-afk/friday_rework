"""Actual retained ELF input custody; SOURCE ONLY, no dlopen or initializer.
This reader recomputes ELF dependencies from retained bytes. Python's original
ExtensionFileLoader has no accepted pre-initializer audit tap here. Consequently
native admission is DENIED even with a green byte/DT_NEEDED inventory.
"""
import os,time,stat,struct,hashlib
class NativeELFCustodyUnavailable(RuntimeError):pass

def _need(ok,why):
    if not ok:raise NativeELFCustodyUnavailable(why)
def _generation(st):
    return (st.st_dev,st.st_ino,st.st_mode,st.st_uid,st.st_nlink,
            st.st_size,st.st_mtime_ns,st.st_ctime_ns)
def _pread(fd,size,offset,end):
    _need(time.monotonic()<end,'original_loader_read_deadline')
    data=os.pread(fd,size,offset)
    _need(len(data)==size,'short_retained_ELF_read')
    return data

def retained_elf64_dependencies(fd,end):
    header=_pread(fd,64,0,end)
    _need(header[:6]==b'\x7fELF\x02\x01','qualified_ELF64_little_endian_required')
    h=struct.unpack('<16sHHIQQQIHHHHHH',header)
    phoff,phsize,phcount=h[5],h[9],h[10]
    _need(phsize==56 and 0<phcount<=256,'qualified_ELF_program_header_bounds')
    segments=[];dynamic=None
    for i in range(phcount):
        p=struct.unpack('<IIQQQQQQ',_pread(fd,56,phoff+i*phsize,end))
        if p[0]==1:segments.append(p)
        if p[0]==2:
            _need(dynamic is None,'ambiguous_ELF_dynamic_segment');dynamic=p
    _need(dynamic is not None and dynamic[5]%16==0 and dynamic[5]//16<=65536,
          'bounded_actual_ELF_dynamic_table')
    values=[]
    for i in range(dynamic[5]//16):
        tag,value=struct.unpack('<qQ',_pread(fd,16,dynamic[2]+16*i,end))
        if tag==0:break
        values.append((tag,value))
    def unique(tag):
        found=[value for t,value in values if t==tag]
        _need(len(found)==1,'ambiguous_missing_ELF_dynamic_value');return found[0]
    address,size=unique(5),unique(10)
    _need(0<size<=16*1024*1024,'bounded_ELF_string_table')
    owners=[p for p in segments if p[3]<=address and address+size<=p[3]+p[5]]
    _need(len(owners)==1,'actual_ELF_string_address_not_file_backed')
    p=owners[0];strings=_pread(fd,size,p[2]+address-p[3],end)
    def text(offset):
        _need(offset<size,'ELF_string_offset_outside_table')
        stop=strings.find(b'\0',offset)
        _need(stop>=offset,'unterminated_ELF_string')
        return strings[offset:stop].decode('utf-8','strict')
    needed=tuple(text(value)for tag,value in values if tag==1)
    sonames=[text(value)for tag,value in values if tag==14]
    _need(len(sonames)<=1,'ambiguous_ELF_SONAME')
    return {'needed':needed,'soname':sonames[0]if sonames else None,
            'rpath':tuple(text(v)for t,v in values if t in (15,29))}

class RetainedELFClosure:
    def __init__(self,pins,end):
        self.end=end;self.retained={};self.unknown=[];self.close_failures=[]
        try:
            for pin in pins:
                path=pin['path'];_need(os.path.isabs(path) and path not in self.retained,
                    'unique_absolute_original_ELF_path_required')
                fd=os.open(path,os.O_RDONLY|os.O_NOFOLLOW|os.O_CLOEXEC)
                row={'fd':fd,'pin':dict(pin),'generation':None,'ELF':None};self.retained[path]=row
                st=os.fstat(fd);row['generation']=_generation(st)
                _need(stat.S_ISREG(st.st_mode) and not st.st_mode&0o022
                      and st.st_size==pin['bytes'],'unsafe_or_changed_ELF_file')
                sha=hashlib.sha256();used=0
                while used<st.st_size:
                    _need(time.monotonic()<end,'original_ELF_custody_clock_expired')
                    part=os.read(fd,min(65536,st.st_size-used))
                    _need(bool(part),'short_actual_ELF_input');sha.update(part);used+=len(part)
                _need(sha.hexdigest()==pin['sha256'] and _generation(os.fstat(fd))==row['generation']
                      and _generation(os.stat(path,follow_symlinks=False))==row['generation'],
                      'actual_ELF_byte_or_named_generation_changed')
                row['ELF']=retained_elf64_dependencies(fd,end)
        except BaseException as primary:
            try:self.close()
            except BaseException as failure:BaseException.add_note(primary,str(failure))
            raise
    def verify_retained_named_generations(self):
        for path,row in self.retained.items():
            _need(row['fd'] is not None and _generation(os.fstat(row['fd']))==row['generation']
                  and _generation(os.stat(path,follow_symlinks=False))==row['generation'],
                  'retained_original_ELF_substituted')
        _need(time.monotonic()<self.end,'actual_ELF_original_clock_expired')
    def require_actual_original_preinitializer_admission(self,name):
        self.verify_retained_named_generations()
        # Holding a source pin or FD does not prove the glibc object actually
        # mapped before its constructors, the interpreter's PyInit ordering,
        # SONAME resolution or future dlopen. Never substitute a JSON receipt,
        # ctypes preload, copied loader or alternate authority for that tap.
        raise NativeELFCustodyUnavailable('UNBOUND_EXISTING_NATIVE_LOADER_PREINITIALIZER_TAP:'+name)
    def close(self):
        for path,row in self.retained.items():
            if row['fd'] is None:continue
            fd=row['fd'];row['fd']=None
            try:os.close(fd)
            except BaseException as failure:self.close_failures.append((path,type(failure).__name__))
        if self.close_failures:raise NativeELFCustodyUnavailable('ELF_CLOSE_UNCONFIRMED')
