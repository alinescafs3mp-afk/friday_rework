"""Exact future driver resolver. No imports/effects from candidate while preparing."""
import hashlib,json,os,stat
from pathlib import Path
class Unbound(RuntimeError):pass

def stable_bytes(path, cap=64*1024*1024):
    path=Path(path)
    if not path.is_absolute():raise Unbound('absolute_input_required')
    fd=os.open(path,os.O_RDONLY|os.O_NOFOLLOW|os.O_CLOEXEC)
    try:
        a=os.fstat(fd)
        if not stat.S_ISREG(a.st_mode) or a.st_nlink!=1 or a.st_size>cap:
            raise Unbound('finite_singlelink_regular_input_required')
        data=b''
        while len(data)<a.st_size:
            part=os.read(fd,min(65536,a.st_size-len(data)))
            if not part:raise Unbound('short_input')
            data+=part
        b=os.fstat(fd)
        fields=lambda s:(s.st_dev,s.st_ino,s.st_mode,s.st_uid,s.st_nlink,
                         s.st_size,s.st_mtime_ns,s.st_ctime_ns)
        if fields(a)!=fields(b):raise Unbound('input_changed')
        return data
    finally:os.close(fd)

def verify(pin):
    data=stable_bytes(pin['path'])
    if len(data)!=pin['bytes'] or hashlib.sha256(data).hexdigest()!=pin['sha256']:
        raise Unbound('exact_input_hash_changed')
    return data

def resolve(package, qualification=None):
    package=Path(package)
    effective=json.loads(stable_bytes(package/'Driver/HF-WHOLE-CURRENT-EFFECTIVE-INPUTS.json'))
    seen={}
    for family in ('hermes','project','stdlib','site','fixture','driver','tests'):
        for row in effective[family]:
            logical=row['logical_path']
            if logical in seen:raise Unbound('duplicate_effective_logical_input')
            verify(row['effective']);seen[logical]=row['effective']
    # Source/dependency pins are constraints only. A presence-only RootGo field
    # or JSON declaration cannot appoint a native authority or grant effects.
    if qualification is not None:
        raise Unbound('JSON_qualification_cannot_grant_original_native_effects')
    return seen
