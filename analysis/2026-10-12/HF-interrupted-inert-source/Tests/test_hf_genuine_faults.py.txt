import pytest
POSITIVE=('historical_task_reference_artifact_reference_distinct', 'unchanged_named_directory_both_native_caller_routes', 'shared_original_loader_current_sibling_and_relative_import', 'main_original_loader_current_sibling_and_relative_import', 'wrapped_native_payload_same_current_guarded_projection', 'outside_parent_normal_live_frontend_helpers_raw_join')
NEGATIVE=('same_content_named_directory_replace_after_open_before_fsync', 'same_content_named_directory_replace_during_close', 'foreign_historical_task_with_equal_artifact_reference', 'ambiguous_original_file_lineage', 'unqualified_original_sibling_before_initializer', 'fake_native_stage_caller_before_fault', 'native_pipe_kernel_EAGAIN_at_original_output', 'original_OFD_flags_changed_at_native_write', 'outside_parent_wrong_child_birth_or_parent', 'wrapper_exit0_not_actual_frontend_raw_wait', 'native_extension_without_preinitializer_tap', 'late_dlopen_unknown_before_initializer')

@pytest.mark.parametrize('case',POSITIVE)
def test_hf_genuine_positive(hf_genuine_native,case):
    o=hf_genuine_native.run(case)
    assert o.original_objects_preserved
    assert o.actual_original_ids_preserved
    assert o.original_frontend_still_alive
    assert o.delivery_observation=='DELIVERED'
    assert o.native_terminal_m<o.original_deadline_m<=o.original_started_m+30
    if case=='historical_task_reference_artifact_reference_distinct':
        reads=tuple(o.case.historical_evidence.values())
        assert reads and all(r.task_reference!=r.artifact_reference for r in reads)
        assert all(r.original_lease.closed and r.current_lease is o.case.lease for r in reads)
        assert not o.case.lease.proposals
        assert o.same_native_registration_released_and_removed
    elif case=='wrapped_native_payload_same_current_guarded_projection':
        pending=o.case.lease.original_projection
        assert pending['wire_before']!=pending['wire_after']
        assert pending['message']['content']==pending['wire_after']
        assert pending['published']==o.case.result
    elif case.startswith('outside_parent'):
        assert o.actual_original_parent_wait_subjects_match_retained_births
        assert o.actual_normal_helpers_waitpid_raw_closed
    elif 'loader' in case:
        assert o.actual_original_loader_initialized_current_sibling_bytes
        assert o.actual_relative_imports_used_same_effective_lineage

@pytest.mark.parametrize('case',NEGATIVE)
def test_hf_genuine_negative(hf_genuine_native,case):
    o=hf_genuine_native.run(case)
    assert o.delivery_observation!='DELIVERED'
    assert o.intended_cause_actually_reached
    assert o.no_late_promotion_or_delivered_persistence
    assert o.no_protected_writer_effect
    assert o.retry_or_budget_reset_count==0
    if case.startswith('same_content_named_directory'):
        assert o.actual_named_directory_dev_inode_before!=o.actual_named_directory_dev_inode_after
        assert o.actual_caller_completion_latch_count==0
    if case=='native_pipe_kernel_EAGAIN_at_original_output':
        raw=o.actual_real_fault_syscalls
        assert raw and raw[-1].name=='write' and raw[-1].result[0]=='EAGAIN'
    if case=='original_OFD_flags_changed_at_native_write':
        raw=o.actual_real_fault_syscalls
        assert raw[-1].result[0]!=raw[-1].result[1]
    if 'initializer' in case:
        assert o.actual_unadmitted_initializer_effect_count==0
        assert o.actual_refusal_precedes_import_or_entry
