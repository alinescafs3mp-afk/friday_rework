@pytest.mark.parametrize('case_id', POSITIVE_CASES)
def test_original_live_positive_requires_DELIVERED(c54_live_original, case_id):
    case = c54_live_original.run(case_id)
    # This fixture must retain actual native caller/actor/observer handles.
    # Its qualification is external; an unbound or JSON-only fixture fails.
    assert case.fixture_qualification == 'ACTUAL_RETAINED_NATIVE_INTACT_HARNESS'
    assert case.delivery_observation == 'DELIVERED'
    assert case.delivery == 'DELIVERED'
    assert case.original_frontend_still_alive
    assert case.frontend_birth_before == case.frontend_birth_after
    assert case.original_objects_before == case.original_objects_after
    assert set(case.original_objects_before) == {
        'AppSession','Application','Console','stream','PluginContext','CLIUserScope'}
    assert case.native_five_ids_before == case.native_five_ids_after
    assert set(case.native_five_ids_before) == {
        'task_id','session_id','turn_id','api_request_id','tool_call_id'}
    assert case.actual_output_envelopes_match_retained_artifacts
    assert case.original_output_and_renderer_cleanup_observed
    assert case.same_native_registration_released_and_removed
    assert case.native_job_admission_and_stop_checked
    assert case.actual_observer_read_and_fsync_observed
    assert case.observer_raw_wait_source == 'DIRECT_CHILD_WAITPID_EXACT_BIRTH'
    assert case.observer_raw_status == 0
    assert case.observer_owned_handles_closed_before_raw_wait
    assert case.no_pending_operation_children_or_output_leases
    assert case.normal_FINISH_authenticated
    assert case.original_started_m < case.normal_closed_m < case.original_deadline_m
    assert case.original_deadline_m <= case.original_started_m + 30
    assert case.durable_attempt_states and set(case.durable_attempt_states) == {'UNKNOWN'}
    assert case.document_sends == case.expected_artifact_count
    assert case.retry_or_budget_reset_count == 0
    if case_id in ('same_frontend_second_distinct_operation',
                   'first_operation_after_frontend_idle_over30'):
        assert case.distinct_scopes == len(set(case.operation_scopes))
        assert all(end <= start + 30 for start,end in case.operation_clocks)
        assert case.original_bootstrap_guardian_birth_preserved

@pytest.mark.parametrize('case_id', NEGATIVE_CASES)
def test_original_finite_negative_no_false_delivery(c54_live_original, case_id):
    case = c54_live_original.run(case_id)
    assert case.fixture_qualification == 'ACTUAL_RETAINED_NATIVE_INTACT_HARNESS'
    assert case.delivery_observation != 'DELIVERED'
    assert case.delivery != 'DELIVERED'
    assert case.new_published_DELIVERED_attempts == 0
    assert case.retry_or_budget_reset_count == 0
    assert case.no_late_document_effect
    assert case.protected_writer_births_before == case.protected_writer_births_after
    assert case.signaled_births <= case.actual_product_owned_births
    assert not (case.signaled_births & case.protected_writer_births_before)
    assert case.durable_unknown_never_authorizes_resend
    if case_id.startswith('synchronous_') or case_id == 'final_caller_projection_block':
        # Requires real blocking syscall/UI path, never cancellable mocks.
        assert case.actual_blocked_native_path_observed
        assert case.absolute_guard_was_armed_before_block
        assert case.guard_targets_exact_product_pidfds
        assert case.stop_status in ('JOINED_ADVERSE_NO_CERTIFICATE', 'STOP_UNCONFIRMED')
        assert case.no_late_promotion
    if case_id in ('unrelated_observer_exit0_without_output_join',
                   'Task_done_without_native_disposal',
                   'GNU_exit0_without_actor_raw'):
        assert case.observer_or_wrapper_exit0_did_not_grant_effects_closure
