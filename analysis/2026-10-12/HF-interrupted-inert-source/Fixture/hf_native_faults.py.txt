"""Real native fault primitives. SOURCE ONLY; no runtime or qualification grant.
All facts come from syscalls at typed original source sites. Unsupported stages,
objects, clocks, descriptors or missing Root admission refuse; no fake provider.
"""
import os,time,stat,fcntl,hashlib
from dataclasses import dataclass
from live_original_capture import MissingWitness
from hermes_cli.friday_cli_native_capture import NativeIOStage

@dataclass(frozen=True)
class FaultSyscall:
    name:str
    before_m:float
    after_m:float
    subject:tuple
    result:object

class OriginalFaultOperation:
    def __init__(self,stage):
        if type(stage) is not NativeIOStage:raise MissingWitness('actual_typed_stage_required')
        self.stage=stage;self.reached=False;self.case=None;self.records=[]
    def reach(self,case,event):
        if event.stage is not self.stage:return
        if self.reached:return # fixed operation has already consumed its one effect
        lease=case.authority.get('finite_lease')
        from hermes_cli.friday_cli_finite import OriginalDeliveryLease
        if (type(lease) is not OriginalDeliveryLease or lease.authority is not case.authority
                or not lease.started<event.monotonic<lease.end
                or event.pid!=case.birth_before['pid']):
            raise MissingWitness('fault_outside_actual_original_guard')
        lease.port._check()
        if time.monotonic()>=lease.end:raise MissingWitness('fault_original_clock_expired')
        self.case=case;self.reached=True;self.apply(case,event,lease)
    def apply(self,case,event,lease):raise MissingWitness('abstract_fault_has_no_effect')

class NativePipeBackpressure(OriginalFaultOperation):
    def __init__(self):super().__init__(NativeIOStage.OUTPUT_WRITE)
    def apply(self,case,event,lease):
        from hermes_cli.friday_cli_output import _ConsoleWrite
        owner=event.owner
        if (type(owner) is not _ConsoleWrite or owner.owner is not lease.port.output
                or event.fd is None or event.fd!=owner.fd
                or event.fd_stat!=tuple(owner.original_identity)):
            raise MissingWitness('actual_owned_original_output_required')
        st=os.fstat(event.fd);flags=fcntl.fcntl(event.fd,fcntl.F_GETFL)
        if not stat.S_ISFIFO(st.st_mode) or not flags&os.O_NONBLOCK:
            raise MissingWitness('real_nonblocking_native_pipe_required')
        used=0
        while used<=1048576 and time.monotonic()<lease.end:
            before=time.monotonic()
            try:n=os.write(event.fd,b'\0'*4096)
            except BlockingIOError as failure:
                self.records.append(FaultSyscall('write',before,time.monotonic(),event.fd_stat,
                    ('EAGAIN',failure.errno,used)));return
            self.records.append(FaultSyscall('write',before,time.monotonic(),event.fd_stat,n));used+=n
            if n<=0:raise MissingWitness('native_pipe_fill_unknown_write')
        raise MissingWitness('native_pipe_backpressure_not_reached_within_original_bound')

class NamedDirectorySubstitution(OriginalFaultOperation):
    def __init__(self):
        super().__init__(NativeIOStage.RECORD_DIRECTORY_OPENED)
        self.retained=None;self.replacement=None;self.path=None;self.backup=None
    def apply(self,case,event,lease):
        if event.owner is not case.context or event.fd is None:
            raise MissingWitness('actual_current_record_directory_required')
        path=os.fspath(case.context.state.data_dir);named=os.stat(path,follow_symlinks=False)
        old=os.fstat(event.fd)
        if (named.st_dev,named.st_ino)!=(old.st_dev,old.st_ino) or not stat.S_ISDIR(old.st_mode):
            raise MissingWitness('already_replaced_native_directory')
        if old.st_uid!=os.getuid() or stat.S_IMODE(old.st_mode)!=0o700:
            raise MissingWitness('owned_private_state_only')
        f=os.open('state.json',os.O_RDONLY|os.O_NOFOLLOW,dir_fd=event.fd)
        try:
            st=os.fstat(f)
            if not stat.S_ISREG(st.st_mode) or st.st_nlink!=1 or st.st_size>10485760:
                raise MissingWitness('actual_owned_state_required')
            raw=b''
            while len(raw)<=10485760:
                part=os.read(f,min(65536,10485761-len(raw)))
                if not part:break
                raw+=part
            if len(raw)!=st.st_size:raise MissingWitness('state_clone_read_incomplete')
        finally:os.close(f)
        backup=path+'.hf-retained-'+lease.scope[:16]
        if os.path.lexists(backup):raise MissingWitness('owned_custody_path_already_exists')
        before=time.monotonic();os.rename(path,backup)
        self.path=path;self.backup=backup;self.retained=(old.st_dev,old.st_ino)
        os.mkdir(path,0o700)
        d=os.open(path,os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW)
        try:
            f=os.open('state.json',os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o600,dir_fd=d)
            try:
                off=0
                while off<len(raw):off+=os.write(f,raw[off:])
                os.fsync(f)
            finally:os.close(f)
            os.fsync(d);new=os.fstat(d);self.replacement=(new.st_dev,new.st_ino)
        finally:os.close(d)
        self.records.append(FaultSyscall('rename/mkdir/write/fsync',before,time.monotonic(),
            self.retained,(self.replacement,hashlib.sha256(raw).hexdigest())))
        if self.retained==self.replacement:raise MissingWitness('real_substitution_not_reached')
    def restore_known_owned(self,end):
        if self.retained is None:return
        if time.monotonic()>=end:raise MissingWitness('STOP_UNCONFIRMED_original_cleanup_expired')
        current=os.stat(self.path,follow_symlinks=False)
        if (current.st_dev,current.st_ino)!=self.replacement:
            raise MissingWitness('STOP_UNCONFIRMED_unknown_current_directory')
        failed=self.path+'.hf-failed'
        if os.path.lexists(failed):raise MissingWitness('STOP_UNCONFIRMED_cleanup_destination_exists')
        os.rename(self.path,failed);os.rename(self.backup,self.path)
        # Retain substituted directory as owned evidence; no unknown unlink.
        restored=os.stat(self.path,follow_symlinks=False)
        if (restored.st_dev,restored.st_ino)!=self.retained:
            raise MissingWitness('STOP_UNCONFIRMED_original_directory_not_restored')
        self.records.append(FaultSyscall('restore_original_directory',time.monotonic(),
            time.monotonic(),self.retained,failed))


class NativeOriginalFDFlagChange(OriginalFaultOperation):
    def __init__(self):super().__init__(NativeIOStage.OUTPUT_WRITE)
    def apply(self,case,event,lease):
        from hermes_cli.friday_cli_output import _ConsoleWrite
        owner=event.owner
        if type(owner) is not _ConsoleWrite or owner.owner is not lease.port.output or owner.original_fd is None:
            raise MissingWitness('actual_original_owned_output_descriptor_required')
        fd=owner.original_fd;before=time.monotonic()
        flags=fcntl.fcntl(fd,fcntl.F_GETFL)
        changed=flags^os.O_NONBLOCK
        fcntl.fcntl(fd,fcntl.F_SETFL,changed)
        observed=fcntl.fcntl(fd,fcntl.F_GETFL)
        self.records.append(FaultSyscall('F_SETFL/F_GETFL',before,time.monotonic(),
            tuple(owner.original_identity),(flags,observed)))
        if observed==flags:raise MissingWitness('actual_OFD_flag_change_not_reached')
        self.fd=fd;self.flags=flags;self.identity=tuple(owner.original_identity)
    def restore_known_owned(self,end):
        if time.monotonic()>=end:raise MissingWitness('STOP_UNCONFIRMED_original_cleanup_expired')
        st=os.fstat(self.fd)
        if (st.st_dev,st.st_ino,st.st_mode,st.st_rdev)!=self.identity:
            raise MissingWitness('STOP_UNCONFIRMED_original_fd_substituted')
        fcntl.fcntl(self.fd,fcntl.F_SETFL,self.flags)
        if fcntl.fcntl(self.fd,fcntl.F_GETFL)!=self.flags:
            raise MissingWitness('STOP_UNCONFIRMED_original_flags_not_restored')
