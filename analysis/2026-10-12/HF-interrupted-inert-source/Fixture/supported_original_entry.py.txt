"""Supported existing HermesCLI.chat observation entry. SOURCE ONLY, NOT RUN.
The existing independently admitted intact Harness supplies the already live
original CLI/context and genuine local input. No frontend/guardian/IDs/model
response/engine/ledger or process is constructed by this entry. This does not
supply RootGo, a slot, fixture qualification or physical fault facilities."""
import inspect
from live_original_capture import (LiveOriginalCapture,OriginalNativeProducer,
                                    MissingWitness,original_objects,same_objects)

def run_existing_original_chat(cli,context,key,original_user_message,*,capture,fault=None,tool_name="friday_result"):
    from cli import HermesCLI
    from hermes_cli.cli_chat_turn_mixin import CLIChatTurnMixin
    from hermes_cli.friday_cli_native_capture import installed_original_capture
    from hermes_cli.friday_cli_principal import current_cli
    from hermes_cli.friday_cli_finite import RetainedOwnerChannel
    if (not isinstance(cli,HermesCLI) or type(capture) is not LiveOriginalCapture
            or key not in capture.expected or type(original_user_message) is not str
            or not original_user_message or cli.agent is None):
        raise MissingWitness('already_live_original_Harness_entry_required')
    # Reject a replaced/monkeypatched driver callback before original entry.
    method=inspect.getattr_static(type(cli),'chat')
    original=inspect.getattr_static(CLIChatTurnMixin,'chat')
    if method is not original:
        raise MissingWitness('actual_supported_original_chat_method_changed')
    cap=current_cli(required=True);port=getattr(cli,'_friday_cli_finite_port',None)
    if (cli._friday_cli_scope is not cap or type(port) is not RetainedOwnerChannel
            or port.cli is not cli or port.cap is not cap or port.active is not None):
        raise MissingWitness('actual_native_bootstrap_before_case_effect_required')
    port._check()
    before=original_objects(cli,context)
    if fault is not None:
        from hf_native_faults import NativePipeBackpressure,NamedDirectorySubstitution,NativeOriginalFDFlagChange
        if type(fault) not in (NativePipeBackpressure,NamedDirectorySubstitution,NativeOriginalFDFlagChange):
            raise MissingWitness("qualified_concrete_fault_primitive_required")
    producer=OriginalNativeProducer(cli,context,capture,fault)
    producer.select_original_case(key,tool_name)
    with installed_original_capture(producer):
        # Supported user input creates the actual native turn/API/tool IDs in
        # the original intact agent. No direct ResultTool/handler invocation.
        cli.chat(original_user_message)
    if producer.selected is not None or producer.open or key not in capture.cases:
        raise MissingWitness('actual_original_native_case_not_completed')
    if not same_objects(before,original_objects(cli,context)):
        raise MissingWitness('original_objects_changed_over_native_case_entry')
    return capture.cases[key]
