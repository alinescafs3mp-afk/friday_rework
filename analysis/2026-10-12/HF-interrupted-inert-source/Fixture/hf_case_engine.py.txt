"""Connected existing-Harness/actual pytest-item producer. SOURCE ONLY.
Requires a complete separately qualified registry of original inputs before any
case entry. This module does not fill missing106 scenarios, create a frontend,
replace local inference, mint native IDs, accept JSON RootGo or run itself.
"""
from dataclasses import dataclass
from pathlib import Path
import hashlib,os
from live_original_capture import LiveOriginalCapture,MissingWitness
from supported_original_entry import run_existing_original_chat

@dataclass(frozen=True)
class OriginalInputJourney:
    original_user_message:str
    tool_name:str='friday_result'
    fault:object=None
    def __post_init__(self):
        if (type(self.original_user_message) is not str or not self.original_user_message
                or self.tool_name not in ('friday_result','friday_work')):
            raise MissingWitness('actual_supported_original_user_input_required')

class WholeOriginalCaseEngine:
    def __init__(self,cli,context,capture,current209_keys,original106_keys,journeys,body_pins):
        if type(capture) is not LiveOriginalCapture:
            raise MissingWitness('actual_original_capture_required')
        self.cli=cli;self.context=context;self.capture=capture
        current=frozenset(current209_keys);original=frozenset(original106_keys)
        if len(current)!=209 or len(original)!=106 or current&original:
            raise MissingWitness('mandatory_full209_plus_original106_not_bound')
        self.required=current|original;self.journeys=dict(journeys);self.body_pins=dict(body_pins)
        if set(self.journeys)!=self.required or any(type(v) is not OriginalInputJourney
                                                   for v in self.journeys.values()):
            raise MissingWitness('complete_genuine315_input_registry_required_before_effects')
        if self.capture.expected!=self.required:
            raise MissingWitness('capture_collection_and_producer_scope_mismatch')
        self.items=None;self.finished=set()
    @staticmethod
    def item_key(item):
        params=item.callspec.params if hasattr(item,'callspec') else {}
        case=params.get('case',params.get('case_id'))
        return Path(str(item.path)).name+'::'+(case if type(case) is str else item.name)
    def _verify_original_body(self,item):
        path=os.path.abspath(str(item.path));pin=self.body_pins.get(path)
        if pin is None:raise MissingWitness('unqualified_actual_collected_test_program')
        fd=os.open(path,os.O_RDONLY|os.O_NOFOLLOW)
        try:
            st=os.fstat(fd);sha=hashlib.sha256();used=0
            while used<st.st_size:
                b=os.read(fd,min(65536,st.st_size-used))
                if not b:raise MissingWitness('short_actual_test_program')
                sha.update(b);used+=len(b)
            after=os.fstat(fd)
            fields=lambda x:(x.st_dev,x.st_ino,x.st_size,x.st_mtime_ns,x.st_ctime_ns)
            if (fields(st)!=fields(after) or st.st_size!=pin['bytes']
                    or sha.hexdigest()!=pin['sha256']):
                raise MissingWitness('original_mandatory_test_body_changed')
        finally:os.close(fd)
        function=getattr(item,'function',None)
        if function is None or os.path.abspath(function.__code__.co_filename)!=path:
            raise MissingWitness('collected_original_body_is_a_substitute')
    def pytest_collection_finish(self,session):
        import pytest
        items={}
        for item in session.items:
            if not isinstance(item,pytest.Item):raise MissingWitness('actual_pytest_item_required')
            key=self.item_key(item)
            if key not in self.required or key in items:
                raise MissingWitness('missing_extra_duplicate_mandatory_collection')
            self._verify_original_body(item);items[key]=item
        if set(items)!=self.required or len(items)!=315:
            raise MissingWitness('actual315_collection_not_complete_no_effects')
        self.items=items
    def pytest_runtest_setup(self,item):
        key=self.item_key(item)
        if self.items is None or self.items.get(key) is not item or key in self.finished:
            raise MissingWitness('uncollected_replaced_or_replayed_native_case')
        self._verify_original_body(item)
        journey=self.journeys[key]
        case=run_existing_original_chat(self.cli,self.context,key,
            journey.original_user_message,capture=self.capture,
            fault=journey.fault,tool_name=journey.tool_name)
        # The actual supported original chat must create/return the native case;
        # a prompt that did not reach the intended tool never becomes a PASS.
        if self.capture.cases.get(key) is not case or not case.sealed:
            raise MissingWitness('real_original_case_journey_not_completed')
        self.finished.add(key)
    def pytest_sessionfinish(self,session,exitstatus):
        if self.finished!=self.required or exitstatus!=0:
            raise MissingWitness('full_mandatory_native_process_not_accepted')
        self.capture.complete_inventory()
