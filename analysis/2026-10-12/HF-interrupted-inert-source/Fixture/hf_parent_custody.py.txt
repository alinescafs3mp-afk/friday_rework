"""Outside original-parent custody primitives. SOURCE ONLY, no process created.
The already admitted original Harness parent supplies its real existing Popen
child, actual subreaper libc and original deadline. No JSON RootGo/provenance.
Whole integration/late unknown children remain UNBOUND until independent gates.
"""
import os,time,stat,select,resource,subprocess,ctypes
from dataclasses import dataclass
from live_original_capture import MissingWitness

@dataclass(frozen=True)
class ProcessBirth:
    pid:int
    ticks:int
    ppid:int
    uid:int
    @classmethod
    def read(cls,pid):
        fd=os.open('/proc/'+str(pid)+'/stat',os.O_RDONLY|os.O_NOFOLLOW)
        try:raw=os.read(fd,8192).decode().rsplit(')',1)[1].split()
        finally:os.close(fd)
        uid=os.stat('/proc/'+str(pid),follow_symlinks=False).st_uid
        return cls(pid,int(raw[19]),int(raw[1]),uid)

@dataclass(frozen=True)
class RawOriginalParentWait:
    parent:ProcessBirth
    subject:ProcessBirth
    role:str
    wait_pid:int
    raw_status:int
    before_m:float
    after_m:float
    sample_before:object

class OriginalParentCustody:
    def __init__(self,original_frontend,original_end,actual_libc,protected_births):
        if type(original_frontend) is not subprocess.Popen:
            raise MissingWitness('retained_actual_original_Harness_child_required')
        if original_frontend.returncode is not None or type(actual_libc) is not ctypes.CDLL:
            raise MissingWitness('already_live_original_frontend_and_native_libc_required')
        self.parent=ProcessBirth.read(os.getpid())
        child=ProcessBirth.read(original_frontend.pid)
        self.protected=frozenset((x.pid,x.ticks)for x in protected_births)
        if child.ppid!=self.parent.pid or child.uid!=self.parent.uid:
            raise MissingWitness('outside_waiter_is_not_actual_original_parent')
        if (child.pid,child.ticks) in self.protected:
            raise MissingWitness('protected_writer_cannot_be_product_target')
        flag=ctypes.c_int()
        # Observe the existing parent setting. Never set/change subreaper here.
        if actual_libc.prctl(37,ctypes.byref(flag),0,0,0)!=0 or flag.value!=1:
            raise MissingWitness('actual_original_parent_subreaper_not_installed')
        if not time.monotonic()<original_end:
            raise MissingWitness('original_frontend_clock_expired')
        self.frontend=original_frontend;self.end=original_end
        self.known={};self.waits=[];self.close_failures=[];self.unknown=[]
        self.retain_owned(child,'original_frontend',direct=True)
    def retain_owned(self,birth,role,*,direct=False):
        if type(birth) is not ProcessBirth or role not in (
                'original_frontend','original_guardian','original_observer'):
            raise MissingWitness('exact_original_native_role_required')
        current=ProcessBirth.read(birth.pid)
        if current!=birth or birth.uid!=self.parent.uid or (birth.pid,birth.ticks) in self.protected:
            raise MissingWitness('foreign_or_changed_owned_birth')
        if direct and birth.ppid!=self.parent.pid:
            raise MissingWitness('actual_original_direct_parent_required')
        if role=='original_guardian' and birth.ppid!=self.frontend.pid:
            raise MissingWitness('guardian_not_actual_original_frontend_child')
        if role=='original_observer':
            guardians=[r['birth'] for r in self.known.values() if r['role']=='original_guardian']
            if len(guardians)!=1 or birth.ppid!=guardians[0].pid:
                raise MissingWitness('observer_not_actual_original_guardian_child')
        if birth.pid in self.known:raise MissingWitness('duplicate_owned_process_custody')
        fd=os.pidfd_open(birth.pid,0)
        try:
            if ProcessBirth.read(birth.pid)!=birth:raise MissingWitness('birth_changed_during_pidfd_open')
            self.known[birth.pid]={'birth':birth,'role':role,'fd':fd,
                'sample':self.sample(birth),'raw':None,'close_confirmed':False}
        except BaseException:
            os.close(fd);raise
    def sample(self,birth):
        if ProcessBirth.read(birth.pid)!=birth:raise MissingWitness('sample_birth_changed')
        fields={str(k):resource.prlimit(birth.pid,k)for k in (
            resource.RLIMIT_AS,resource.RLIMIT_CPU,resource.RLIMIT_FSIZE,resource.RLIMIT_CORE)}
        fd=os.open('/proc/'+str(birth.pid)+'/stat',os.O_RDONLY|os.O_NOFOLLOW)
        try:v=os.read(fd,8192).decode().rsplit(')',1)[1].split()
        finally:os.close(fd)
        return {'limits':fields,'affinity':tuple(sorted(os.sched_getaffinity(birth.pid))),
                'utime_ticks':int(v[11]),'stime_ticks':int(v[12]),'sample_m':time.monotonic()}
    def reap_known_direct_or_adopted(self,pid):
        if pid not in self.known:raise MissingWitness('unknown_pid_no_wait_or_signal')
        row=self.known[pid];birth=row['birth']
        if row['raw'] is not None:raise MissingWitness('raw_wait_cannot_replay')
        if ProcessBirth.read(os.getpid())!=self.parent:
            raise MissingWitness('outside_parent_birth_changed')
        # A still-present child must currently be direct/adopted by this actual
        # subreaper. Its birth must equal the retained pre-death subject.
        current=ProcessBirth.read(pid)
        if current.ticks!=birth.ticks or current.ppid!=self.parent.pid:
            raise MissingWitness('original_child_not_in_actual_parent_custody')
        poll=select.poll();poll.register(row['fd'],select.POLLIN)
        while time.monotonic()<self.end:
            poll.poll(max(0,min(50,int((self.end-time.monotonic())*1000))))
            before=time.monotonic()
            try:wait_pid,raw=os.waitpid(pid,os.WNOHANG)
            except ChildProcessError as failure:
                self.unknown.append((birth,'raw_wait_unavailable',type(failure).__name__))
                raise MissingWitness('original_raw_wait_unavailable_not_zero') from failure
            after=time.monotonic()
            if wait_pid==0:continue
            if wait_pid!=pid:raise MissingWitness('foreign_raw_wait_subject')
            observed=RawOriginalParentWait(self.parent,birth,row['role'],wait_pid,raw,
                before,after,row['sample'])
            # Persist raw outcome in custody BEFORE any fallible close/check.
            row['raw']=observed;self.waits.append(observed)
            closing=row['fd'];row['fd']=None
            try:os.close(closing);row['close_confirmed']=True
            except BaseException as failure:
                self.close_failures.append((birth,type(failure).__name__));raise
            if pid==self.frontend.pid:self.frontend.returncode=os.waitstatus_to_exitcode(raw)
            return observed
        self.unknown.append((birth,'original_cleanup_deadline_expired'))
        raise MissingWitness('STOP_UNCONFIRMED_original_parent_cleanup_expired')
    def normal_frontend_alive(self):
        birth=self.known[self.frontend.pid]['birth']
        if ProcessBirth.read(birth.pid)!=birth or self.frontend.returncode is not None:
            raise MissingWitness('normal_original_frontend_not_same_live_birth')
        return self.sample(birth)
    def closure_frontend_lives(self):
        # Normal operation joins owned helpers while the original frontend
        # remains alive. Its pidfd close is never relabelled as a frontend wait.
        sample=self.normal_frontend_alive()
        unresolved=[r['birth']for p,r in self.known.items()
            if p!=self.frontend.pid and (r['raw'] is None or not r['close_confirmed'])]
        if unresolved or self.unknown or self.close_failures:
            raise MissingWitness('STOP_UNCONFIRMED_normal_helper_custody_remaining')
        row=self.known[self.frontend.pid];closing=row['fd'];row['fd']=None
        try:os.close(closing);row['close_confirmed']=True
        except BaseException as failure:
            self.close_failures.append((row['birth'],type(failure).__name__));raise
        return sample,tuple(self.waits)

    def closure(self):
        unresolved=[r['birth']for r in self.known.values()
            if r['raw'] is None or not r['close_confirmed']]
        if unresolved or self.close_failures or self.unknown:
            raise MissingWitness('STOP_UNCONFIRMED_actual_owned_custody_remaining')
        return tuple(self.waits)
