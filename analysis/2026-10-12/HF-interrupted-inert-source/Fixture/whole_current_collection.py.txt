"""Whole prospective pytest collection consumer. No collection executed here.
This plugin only verifies inventory and binds assertions to actual completed
original captures. It does not generate case journeys or grant effects."""
from pathlib import Path
from live_original_capture import LiveOriginalCapture,MissingWitness

class WholeOriginalCollection:
    def __init__(self,capture,expected_keys,original106_keys,new20_keys):
        if type(capture) is not LiveOriginalCapture:
            raise MissingWitness('actual_original_capture_required')
        self.capture=capture;self.expected=frozenset(expected_keys)
        self.original=frozenset(original106_keys);self.new=frozenset(new20_keys)
        if len(self.expected)!=189 or len(self.original)!=106 or len(self.new)!=20:
            raise MissingWitness('full189_and_original106_required_no_narrowing')
        if self.expected&self.original or self.new&(self.expected|self.original):
            raise MissingWitness('overlapping_case_identity')
    def pytest_configure(self,config):
        if hasattr(config,'_friday_actual_native_capture'):
            raise MissingWitness('capture_cannot_replace_existing')
        config._friday_actual_native_capture=self.capture
    def pytest_collection_finish(self,session):
        actual=[]
        for item in session.items:
            # Real pytest item/CallSpec after actual collection, never caller
            # JSON counts or a manufactured test name used as observation.
            params=item.callspec.params if hasattr(item,'callspec') else {}
            case=params.get('case',params.get('case_id'))
            if type(case) is str:
                key=Path(str(item.path)).name+'::'+case
            else:
                # Original non-parametrized items retain their real collected
                # pytest identity. No invented synthetic106 keys/counts.
                key=Path(str(item.path)).name+'::'+item.name
            if getattr(item,'function',None) is None:
                raise MissingWitness('actual_original_test_body_unbound')
            actual.append(key)
        wanted=self.expected|self.original|self.new
        if len(actual)!=315 or len(set(actual))!=315 or set(actual)!=wanted:
            raise MissingWitness('whole_current_collection_missing_extra_duplicate')
    def pytest_sessionfinish(self,session,exitstatus):
        self.capture.complete_inventory()
        if exitstatus!=0:raise MissingWitness('whole_original_assertions_failed')
