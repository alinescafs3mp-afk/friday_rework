"""Bounded fake FD controls Source only; NOT_RUN, no admission or kernel calls.
Caller supplies exact independently reviewed extracted modules. Every primitive
used by capture is replaced with the finite backend before invocation.
"""
from types import SimpleNamespace
import ast

class FakeFDs:
    O_RDONLY=0;O_DIRECTORY=1;O_NOFOLLOW=2;O_NONBLOCK=4
    O_WRONLY=8;O_CREAT=16;O_EXCL=32
    def __init__(self,custody,fail=None,close_effect=False):
        self.custody=custody;self.fail=fail;self.close_effect=close_effect
        self.primary=OSError(5,'bounded fake primary')
        self.live={10:('root',100,100),11:('evidence',200,200),12:('group',300,300),13:('slot',400,400)}
        self.next_fd=20;self.events=[];self.offset={};self.saved={}
    def label(self,fd):
        rows=[r for r in self.custody['fds'] if r['fd']==fd and not r['closed']]
        assert len(rows)==1, 'every fstat/close fd is already in exterior registry'
        return rows[0]['label']
    def info(self,kind,dev,ino):
        directory=kind in ('root','sub','evidence','group')
        return SimpleNamespace(st_mode=(0o040700 if directory else 0o100400),
            st_dev=dev,st_ino=ino,st_size=3 if kind=='raw' else 0,
            st_nlink=1,st_mtime_ns=1,st_ctime_ns=1,st_uid=1000,st_gid=1000)
    def getuid(self):return 1000
    def getgid(self):return 1000
    def fstat(self,fd):
        label=self.label(fd);self.events.append(('fstat',label,fd))
        if self.fail=='fstat' and label.startswith('trace_directory:'):raise self.primary
        kind,dev,ino=self.live[fd]
        if self.fail=='identity' and label.startswith('trace_directory:'):ino+=1
        return self.info(kind,dev,ino)
    def dup(self,fd):
        self.events.append(('dup',fd))
        if self.fail=='dup':raise self.primary
        value=self.next_fd;self.next_fd+=1;self.live[value]=self.live[fd];return value
    def listdir(self,fd):
        kind=self.live[fd][0]
        return ['sub'] if kind=='root' else ['event.json'] if kind=='sub' else []
    def stat(self,name,dir_fd,follow_symlinks):
        assert follow_symlinks is False
        return self.info('sub',100,101) if name=='sub' else self.info('raw',100,102)
    def open(self,name,flags,mode=None,dir_fd=None):
        self.events.append(('open',name))
        if self.fail=='open' and name=='sub':raise self.primary
        value=self.next_fd;self.next_fd+=1
        self.live[value]=('sub',100,101) if name=='sub' else ('raw',100,102) if name=='event.json' else ('saved',200,value)
        if name not in ('sub','event.json'):self.saved[name]=value
        return value
    def read(self,fd,n):
        assert 0<n<=65536;self.offset[fd]=self.offset.get(fd,0)+1
        return b'abc' if self.offset[fd]==1 else b''
    def write(self,fd,view):
        assert len(view)<=65536;self.events.append(('write',self.label(fd)))
        if self.fail=='write':raise self.primary
        return len(view)
    def fsync(self,fd):self.label(fd);self.events.append(('fsync',fd))
    def close(self,fd):
        label=self.label(fd);self.events.append(('close',label,fd))
        if self.fail=='close' and label=='trace_root_duplicate':
            if self.close_effect:del self.live[fd]
            raise self.primary
        del self.live[fd]

def capture_trial(owner,fdmodule,fail=None,close_effect=False):
    custody={'fds':[],'capacity_release':False,'private_directory_capabilities':{
        'trace':{'fd':11,'path':'/FAKE_ONLY/trace','identity':[200,200],'policy':{'uid':1000,'gid':1000,'mode':0o700}},
        'trace_parent':{'fd':14,'path':'/FAKE_ONLY','identity':[500,500],'policy':{'uid':1000,'gid':1000,'mode':0o700}}}}
    for fd,label in ((10,'output'),(11,'exterior_evidence'),(12,'group'),(13,'helper_slot_duplicate')):
        fdmodule.bind_existing(custody,fd,label)
    fake=FakeFDs(custody,fail,close_effect)
    old_os,old_fd_os,old_time=owner.os,fdmodule.os,owner.time
    owner.os=fake;fdmodule.os=fake;owner.time=SimpleNamespace(monotonic=lambda:0,time=lambda:0)
    old_directory,old_parent,old_output=fdmodule.directory_capability,fdmodule.validate_private_parent,owner.kernel120_revalidate_output_before_effect
    # This pre-existing unit backend isolates FD custody, never claims parent/kernel acceptance.
    def fake_directory(cap,private=True,process_pid=None):
        assert cap['fd']==11 and fake.fstat(11).st_uid==1000
        return cap
    def fake_parent(path,cap,process_pid=None):
        assert path=='/FAKE_ONLY/trace' and cap['path']=='/FAKE_ONLY'
        return cap
    def fake_output(spec):
        assert spec['qualification_case']=='FAKE_ONLY'
        assert fake.fstat(10).st_dev==100
        return None
    fdmodule.directory_capability=fake_directory;fdmodule.validate_private_parent=fake_parent
    owner.kernel120_revalidate_output_before_effect=fake_output
    result=None;caught=None
    try:
        result=owner.existing_root_capture_private_trace({'domain':'kernel120_qualification','qualification_case':'FAKE_ONLY'},
            10,11,[200,200],{'actual_wait':'FAKE_ONLY'},100,100,custody)
    except BaseException as error:caught=error
    finally:
        owner.os=old_os;fdmodule.os=old_fd_os;owner.time=old_time
        fdmodule.directory_capability=old_directory;fdmodule.validate_private_parent=old_parent
        owner.kernel120_revalidate_output_before_effect=old_output
    return result,caught,custody,fake

def test_root_fd_normal_capture(owner,fdmodule):
    result,error,custody,fake=capture_trial(owner,fdmodule)
    assert error is None and result['raw_files']==1 and result['raw_total_bytes']==3
    assert result['capacity_release'] is False and result['normal_umount_not_yet_done'] is True
    assert set(fake.live)=={10,11,12,13}, 'all newly acquired trace FDs closed, exterior mount/group/slot retained'
    acquired=[r for r in custody['fds'] if r['fd']>=20]
    assert acquired and all(r['closed'] and r['close_attempted'] and not r['close_ambiguous'] for r in acquired)
    close_labels=[e[1] for e in fake.events if e[0]=='close']
    assert any(x.startswith('trace_target:') for x in close_labels) and close_labels[-1]=='trace_index'

def test_root_fd_fstat_failure_registered(owner,fdmodule):
    result,error,custody,fake=capture_trial(owner,fdmodule,'fstat')
    assert error is fake.primary and result is None and custody['STOP_UNCONFIRMED'] is True
    row=next(r for r in custody['fds'] if r['label'].startswith('trace_directory:'))
    assert row['fd'] in fake.live and not row['closed'] and not row['close_attempted']
    assert all(fd in fake.live for fd in (10,11,12,13))
    assert not any(e[0]=='close' for e in fake.events)

def test_root_fd_identity_failure_registered(owner,fdmodule):
    result,error,custody,fake=capture_trial(owner,fdmodule,'identity')
    assert error is not None and result is None and custody['STOP_UNCONFIRMED'] is True
    row=next(r for r in custody['fds'] if r['label'].startswith('trace_directory:'))
    assert row['fd'] in fake.live and not row['close_attempted']
    assert not any(e[0]=='close' for e in fake.events)

def test_root_fd_open_failure_no_lost_handle(owner,fdmodule):
    result,error,custody,fake=capture_trial(owner,fdmodule,'open')
    assert error is fake.primary and result is None and custody['STOP_UNCONFIRMED'] is True
    assert {r['fd'] for r in custody['fds'] if not r['closed']}==set(fake.live)
    assert custody['capacity_release'] is False and not any(e[0]=='close' for e in fake.events)

def test_root_fd_dup_failure_retains_exterior(owner,fdmodule):
    result,error,custody,fake=capture_trial(owner,fdmodule,'dup')
    assert error is fake.primary and result is None and custody['STOP_UNCONFIRMED'] is True
    assert set(fake.live)=={10,11,12,13} and len(custody['fds'])==4
    assert not any(e[0]=='close' for e in fake.events)

def test_root_fd_close_ambiguity_no_retry(owner,fdmodule):
    for effect in (False,True):
        result,error,custody,fake=capture_trial(owner,fdmodule,'close',effect)
        assert error is fake.primary and result is None and custody['STOP_UNCONFIRMED'] is True
        row=next(r for r in custody['fds'] if r['label']=='trace_root_duplicate')
        assert row['close_attempted'] and row['close_ambiguous'] and not row['closed']
        assert sum(e[0]=='close' and e[2]==row['fd'] for e in fake.events)==1
        assert all(fd in fake.live for fd in (10,11,12,13))
        assert custody['capacity_release'] is False

def test_root_fd_partial_target_failure_retained(owner,fdmodule):
    result,error,custody,fake=capture_trial(owner,fdmodule,'write')
    assert error is fake.primary and result is None and custody['STOP_UNCONFIRMED'] is True
    assert len(custody['partial_evidence_names'])==1
    name=custody['partial_evidence_names'][0];fd=fake.saved[name]
    row=next(r for r in custody['fds'] if r['fd']==fd and not r['closed'])
    assert fd in fake.live and not row['close_attempted']
    assert all(x in fake.live for x in (10,11,12,13))

def test_root_fd_full_finalizer_order_static(exact_owner_source):
    """Check actual final helper Source order, not a fabricated runtime pass."""
    tree=ast.parse(exact_owner_source)
    fn=next(n for n in tree.body if isinstance(n,ast.FunctionDef) and n.name=='existing_root_finalize')
    code=ast.get_source_segment(exact_owner_source,fn)
    stock_wait=code.index("os.waitpid(stock_birth['pid'],0)")
    actor_wait=code.index("os.waitpid(actor_birth['pid'],0)")
    capture=code.index('raw_trace=existing_root_capture_private_trace')
    output_close=code.index('root_fd_custody.closed(retained_fd_custody,fd)')
    umount=code.index('libc.umount2')
    slot_close=code.index('root_fd_custody.closed(retained_fd_custody,slot_fd)')
    assert stock_wait<actor_wait<capture<output_close<umount<slot_close
    assert 'retained_fd_custody' in code and 'except BaseException as error:' in code
    assert 'root_fd_custody.stopped(retained_fd_custody,error)' in code
    assert 'LOCK_UN' in code and 'flock(' not in code
    assert 'known_handles_closed_once' in code and 'capacity_release' in code

def run_root_fd_memory_controls(owner,fdmodule,exact_owner_source):
    test_root_fd_normal_capture(owner,fdmodule)
    test_root_fd_fstat_failure_registered(owner,fdmodule)
    test_root_fd_identity_failure_registered(owner,fdmodule)
    test_root_fd_open_failure_no_lost_handle(owner,fdmodule)
    test_root_fd_dup_failure_retains_exterior(owner,fdmodule)
    test_root_fd_close_ambiguity_no_retry(owner,fdmodule)
    test_root_fd_partial_target_failure_retained(owner,fdmodule)
    test_root_fd_full_finalizer_order_static(exact_owner_source)
    return {'controls':8,'close_ambiguity_variants':2,'actual_kernel_calls':0,'Source_only':True,'capacity_release':False}
