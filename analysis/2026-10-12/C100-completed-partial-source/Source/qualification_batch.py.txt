class BatchRefusal(ValueError):
    def __init__(self,code,detail):
        self.code=code
        super().__init__(code+': '+detail)


def need(ok,code,detail):
    if not ok:raise BatchRefusal(code,detail)


def finite(v):return type(v) in (int,float) and math.isfinite(v)








def drive_all32_actual_assertions(owner,tests,Root_expected,trace_refs,clock_pin):
    """Existing Root actual trace interface, not an executor/provider substitute.
    Verify all real pinned inputs first; call every unchanged intended assertion
    on its own complete actual Root trace. No partial total or unrelated-error
    pass. A shared positive cohort still has ten distinct cause observations.
    Source author cannot use this as independent final acceptance.
    """
    import json
    ids=[x[0] for x in CASES]
    mapping=original_case_clock_bindings(Root_expected,clock_pin)
    need(type(Root_expected)is dict and Root_expected.get('independent_final_Source_safety_acceptance') is not None
         and Root_expected.get('actual_specific_kernel120_admission') is not None,
         'BATCH_NOT_ADMITTED','outside Root actual final body/action admission required')
    need(Root_expected.get('actual_complete_batch_finished_under_original_clock_ref') is not None,
         'BATCH_PHYSICAL_FIT_UNBOUND','actual full32 oneclock completion prerequisite missing')
    need(type(trace_refs)is dict and set(trace_refs)==set(ids),'BATCH_SCOPE','exact all32 Root raw trace pins')
    traces={};functions={}
    for case_id,kind,name,cause in CASES:
        trace=json.loads(owner.same_read_pin(trace_refs[case_id],65536))
        need(trace.get('case_id')==case_id and trace.get('intended_cause')==cause
             and trace.get('original_clock_ref')==mapping[case_id]['original_clock_ref']
             and trace.get('Root_expected_original_clock_ref')==mapping[case_id]['Root_expected_original_clock_ref']
             and trace.get('simulated') is False and trace.get('unknown_as_zero') is False
             and trace.get('old_clock_or_Source_credit_transferred') is False,
             'BATCH_TRACE_CAUSE','actual exact case/clock/cause, no Source history transfer')
        if kind=='source_uncertainty_after_actual_operation':
            need(trace.get('accepted_original_clock_elapsed_at_entry')
                 ==mapping[case_id]['accepted_original_clock_elapsed_at_entry'],
                 'BATCH_TRACE_CAUSE','same actual original producer entry elapsed, no substituted batch anchor')
        function=getattr(tests,name,None)
        need(callable(function),'BATCH_ASSERTION_MISSING',name)
        traces[case_id]=trace;functions[case_id]=function
    result=[]
    for case_id,kind,name,cause in CASES:
        try:functions[case_id](traces[case_id])
        except Exception as error:
            result.append({'id':case_id,'kind':kind,'state':'FAIL','error_type':type(error).__name__})
        else:result.append({'id':case_id,'kind':kind,'state':'PASS'})
    need(len(result)==32,'BATCH_SCOPE','full mandatory assertions, no early green partial return')
    return {'results':result,'passed':sum(r['state']=='PASS' for r in result),
            'failed':sum(r['state']=='FAIL' for r in result),'total':32,'clock_reset':False,
            'independent_acceptance':'ROOT_SEPARATE_REVIEW_REQUIRED','capacity_release':False,
            'native_genuine_or_installed_runtime_acceptance':False}

def original32_serial_cut_conflict(owner,catalog):

    need(type(catalog)is Original32Catalog,'ORIGINAL_CATALOG','current catalog')

    funcs={n.name:n for n in tree.body if isinstance(n,ast.FunctionDef)}
    hold=funcs['retain_until_stock_guard']
    need(any(isinstance(n,ast.While) and isinstance(n.test,ast.Constant) and n.test.value is True for n in ast.walk(hold))
         and any(isinstance(n,ast.Call) and isinstance(n.func,ast.Attribute) and n.func.attr=='pause' for n in ast.walk(hold)),
         'CUT_CONTRACT_DRIFT','unchanged host voluntary guardian retention')
    need('kernel120_effect_boundary' in funcs,'CUT_CONTRACT_DRIFT','bounded existing Root AFTER-real notification')
    cuts=[r[0] for r in CASES if r[1]=='source_uncertainty_after_actual_operation']
    need(len(cuts)==4 and all(catalog.planned[i]['planned_age_window']==[19,21] for i in cuts),
         'CUT_CONTRACT_DRIFT','four original commonclock windows')
    return {'code':'ORIGINAL_EARLY_OWNED_STOP_REQUIRES_ACTUAL_FIT','clock_pin':catalog.clock_pin,
       'cut_count':4,'window_seconds':2,'serial_owner_count':1,'physical_fit':'UNPROVEN',
       'voluntary_host_retention_preserved':True,'latest_actor_stop_age':100,'latest_stock_stop_age':110,
       'lower_bound_from_voluntary_hold_claimed':False,'worst244_305_used_as_minimum':False}

         and Root_expected.get('actual_specific_action_admission') is not None,
         'BATCH_NOT_ADMITTED','external exact Source/action admission')
    result=assert_complete_plan(plan,clock_pin,absolute,now_m,now_u,Root_expected)
    need(Root_expected.get('retained_holder_verified') is not None,
         'EXISTING_CALLER_CUSTODY_UNBOUND','actual independent original capability holder required')
    need(set(Root_expected.get('produced_phase_input_refs',{}))=={p['phase_id'] for p in plan['phases']},
         'ACTUAL_INPUT_BINDINGS','current named clock-derived producer before all phase effects')
    original32_serial_cut_conflict(Root_expected['actual_accepted_same_read_owner_object'],
                                  Root_expected['original_catalog_object'])
    return result


def capture_original_case_clock(owner,case_id,spec,original_case_catalog,original_global_clock):
    import json,time
    c=original_case_catalog
    need(type(c)is Original32Catalog and case_id in c.planned and case_id not in c.actual,
         'ORIGINAL_CATALOG','one actual entry per cause, no replay or guessed historical row')
    need(spec['clock']==c.clock_pin and original_global_clock==c.clock
         and json.loads(owner.same_read_pin(spec['clock'],65536))==c.clock,
         'ORIGINAL_CLOCK','same immutable current original clock argument')
    clock=c.clock;m=time.monotonic();u=time.time();age=m-clock['started_at_monotonic']
    need(0<=age<120 and abs(age-(u-clock['started_at_unix']))<=2,'ORIGINAL_CLOCK','actual captured dualclock')
    kind=next(r[1] for r in CASES if r[0]==case_id)
    if kind=='source_uncertainty_after_actual_operation':
        need(19<=age<=21 and 19<=u-clock['started_at_unix']<=21,'BATCH_OWNER_CUT_WINDOW','original four19..21 preserved')
    need(c.active_phase is None,'SERIAL_OWNER','previous phase must have actual fresh closure before next entry')
    row={'original_clock_ref':c.clock_pin,'Root_expected_original_clock_ref':c.clock_pin,
      'original_clock_producer_ref':c.producer_pin,'catalog_frozen_before_batch_ref':c.frozen_ref,
      'original_seconds':120,'cleanup_seconds':30,'clock_start_m':clock['started_at_monotonic'],
      'clock_start_u':clock['started_at_unix'],'entry_m':m,'entry_u':u,'accepted_original_clock_elapsed_at_entry':age,
      'original_final_m':clock['started_at_monotonic']+120,'original_final_u':clock['final_deadline_unix'],
      'original_useful_m':clock['started_at_monotonic']+90,'original_useful_u':clock['useful_deadline_unix'],
      'clock_reset':False,'new_actual_entry_producer':'capture_original_case_clock','capacity_release':False}
    c.actual[case_id]=row;c.active_phase=case_id
    if owner.root_fd_custody._physical_capture is not None:
        owner.root_fd_custody._physical_capture.observe_state('case_entry',dict(row,case_id=case_id))
    return row


def original_case_clock_bindings(Root_expected,clock_pin):
    catalog=Root_expected.get('original_catalog_object')
    need(type(catalog)is Original32Catalog and catalog.clock_pin==clock_pin
         and set(catalog.actual)=={r[0] for r in CASES} and catalog.active_phase is None,
         'CLOCK_CASE_MAPPING_UNBOUND','actual all32 entries are a final observation prerequisite, never a pre-GO plan')
    for case_id,row in catalog.actual.items():
        need(row['original_clock_ref']==clock_pin and row['Root_expected_original_clock_ref']==clock_pin
             and row['clock_start_m']==catalog.clock['started_at_monotonic']
             and row['clock_start_u']==catalog.clock['started_at_unix']
             and row['original_final_m']==catalog.clock['started_at_monotonic']+120
             and row['original_final_u']==catalog.clock['final_deadline_unix']
             and row['clock_reset'] is False,'CLOCK_CASE_MAPPING_UNBOUND','exact actual common origin')
        if catalog.planned[case_id]['kind']=='source_uncertainty_after_actual_operation':
            need(19<=row['entry_m']-row['clock_start_m']<=21
                 and 19<=row['entry_u']-row['clock_start_u']<=21
                 and 19<=row['accepted_original_clock_elapsed_at_entry']<=21,
                 'BATCH_OWNER_CUT_WINDOW','all four actual original windows')
    return catalog.actual


def validate_actual_serial_intervals(catalog):
    """Actual producer rows only; serial intervals cannot overlap or renew."""
    need(type(catalog)is Original32Catalog,'ORIGINAL_CATALOG','same actual catalog')
    previous_m=catalog.clock['started_at_monotonic'];previous_u=catalog.clock['started_at_unix']
    for row in catalog.phase_intervals:
        need(all(finite(row.get(k)) for k in ('entry_m','entry_u','terminal_m','terminal_u'))
             and previous_m<=row['entry_m']<=row['terminal_m']<catalog.clock['started_at_monotonic']+120
             and previous_u<=row['entry_u']<=row['terminal_u']<catalog.clock['final_deadline_unix'],
             'SERIAL_OWNER','actual no overlapping owner; common original final')
        previous_m=row['terminal_m'];previous_u=row['terminal_u']
    return {'actual_interval_count':len(catalog.phase_intervals),'serial_owner_count':1}

def execute_original32_serial(owner,adapter,base,Root_expected,clock,catalog):
    """Fixed serial caller. Safe contained FAIL continues; uncertain custody stops.
    No measured-fit claim, new clock, per-case120, retries or synthetic passes.
    """
    import time,json
    plan=make_original32_plan(catalog);outcomes=[]
    controls=Root_expected['controls_module']
    for phase in plan['phases']:
        phase_id=phase['phase_id'];case_id=phase['case_ids'][0]
        need(catalog.active_phase is None,'SERIAL_OWNER','previous actual cohort closure required')
        owner.root_fd_custody.require_custody(Root_expected['retained_fd_custody'])
        if 'wait_original_age' in phase:owner.kernel120_wait_original_age(clock,phase['wait_original_age'])
        ref=Root_expected['produced_phase_input_refs'][phase_id]
        frozen=json.loads(owner.same_read_pin(ref,65536))
        need(frozen['schema']=='kernel120-derived-phase-input.v3'
             and frozen['original_clock_ref']==catalog.clock_pin
             and frozen['candidate_spec']==Root_expected['actual_phase_specs'][phase_id],
             'ACTUAL_INPUT_BINDINGS','current exact named producer/consumer schema')
        spec=frozen['candidate_spec'];clean=frozen['clean_owned_fixture_spec']
        context=Root_expected['phase_contexts'][phase_id]
        context['retained_fd_custody']=Root_expected['retained_fd_custody']
        need(context['original_clock_pin']==catalog.clock_pin,'ORIGINAL_CLOCK','same commonclock context')
        custody=context['retained_fd_custody']
        owner.root_fd_custody.begin_phase(custody,phase_id)
        for fd,label in ((clean['existing_owner']['group_fd'],'group'),
                         (clean['existing_owner']['slot_fd'],'original_exterior_slot'),
                         (context['control_dir_fd'],'control_directory'),(context['trace_dir_fd'],'trace_directory')):
            owner.root_fd_custody.bind_existing(custody,fd,label)
        for fd in context['output_owned_fds']:owner.root_fd_custody.bind_existing(custody,fd,'Root_output')
        if phase['first_birth_required']:
            need(time.monotonic()+45<clock['started_at_monotonic']+90
                 and time.time()+45<clock['useful_deadline_unix'],
                 'BATCH_REAL_BUDGET_EXHAUSTED','original45 before actual birth only')
        if case_id in controls.NEGATIVE_CONTRACTS:
            if not phase['first_birth_required']:
                capture_original_case_clock(owner,case_id,clean,catalog,clock)
            result=controls.dispatch_negative(owner,adapter,spec,clean,context,clock,catalog,case_id)
        else:result=adapter.kernel120_named_root_case_adapter(owner,spec,context,clock,catalog,case_id)
        need(result['actual_cohort_closure'] is True and result['capacity_release'] is False
             and result['original_slot_retained'] is True,'SERIAL_OWNER','actual normal cleanup; original slot retained')
        owner.root_fd_custody.complete_phase(custody,phase_id)
        terminal_m=time.monotonic();terminal_u=time.time()
        need(terminal_m<clock['started_at_monotonic']+120 and terminal_u<clock['final_deadline_unix'],
             'BATCH_REAL_BUDGET_EXHAUSTED','actual original final, never reset')
        original=catalog.actual[case_id]
        for item in phase['case_ids'][1:]:catalog.actual[item]=dict(original)
        catalog.phase_intervals.append({'case_ids':phase['case_ids'],'entry_m':original['entry_m'],
              'entry_u':original['entry_u'],'terminal_m':terminal_m,'terminal_u':terminal_u})
        catalog.active_phase=None;outcomes.append(result)
    need(catalog.active_phase is None and len(catalog.actual)==32,'SERIAL_OWNER','whole actual original scope')
    owner.root_fd_custody.require_custody(Root_expected['retained_fd_custody'])
    validate_actual_serial_intervals(catalog)
    slot=base['existing_owner']['slot_fd']
    owner.root_fd_custody.finalize_batch_roles(Root_expected['retained_fd_custody'],slot)
    # CLI duplicate closes last HERE; actual external caller holds the last original
    # until its independently admitted exact GNU wait/fresh kernel terminal hook.
    return {'outcomes':outcomes,'case_clock_bindings':catalog.actual,'actual_original_scope':32,
       'physical_fit':'ROOT_MUST_INDEPENDENTLY_VERIFY','all32_assertion_acceptance':'PENDING_ROOT_RAW_TRACE_REVIEW',
       'existing_caller_last_slot_close_required':True,'capacity_release':False,
       'clock_reset':False,'native_genuine_or_installed_runtime_acceptance':False}

def produce_whole_role_plan(owner,Root_expected,clean_specs,contexts,phase_order,auxiliary):
    """Complete physical FD identities with explicit close ownership/lifetimes."""
    plans=[]
    def add(fd,role,lifetime,phase,purpose,path=None):
        plans.append({'fd':fd,'role':role,'identity':owner.root_fd_custody.handle_identity(fd),
                      'lifetime':lifetime,'phase':phase,'purpose':purpose,'path':path})
    for phase in phase_order:
        spec=clean_specs[phase];context=contexts[phase]
        add(spec['existing_owner']['group_fd'],'group','phase',phase,'normal_current_group_remove',spec['native']['group'])
        add(context['trace_dir_fd'],'trace_directory','phase',phase,'current_raw_trace_until_cleanup',
            context['private_directory_capabilities']['trace']['path'])
        for fd in context['output_owned_fds']:
            add(fd,'output','phase',phase,'current_output_normal_unmount',str(owner.Path(spec['record']).parent))
        add(spec['existing_owner']['slot_fd'],'slot','batch',None,'CLI_duplicate_external_caller_retains_original')
        add(context['control_dir_fd'],'control_directory','batch',None,'all_derived_inputs_and_original_clock',
            context['private_directory_capabilities']['control']['path'])
        for key in ('output_parent','control_parent','trace_parent'):
            cap=context['private_directory_capabilities'][key]
            add(cap['fd'],'private_parent','batch',None,'original_private_parent_retention',cap['path'])
        cap=context['group_parent_capability']
        add(cap['fd'],'group_parent','batch',None,'original_group_parent_retention',cap['path'])
        for cap in context.get('auxiliary_parent_capabilities',[]):
            role='group_parent' if owner.root_fd_custody.retained_parent_is_cgroup(cap) else 'private_parent'
            add(cap['fd'],role,'batch',None,'original_group_parent_retention' if role=='group_parent' else 'original_private_parent_retention',cap['path'])
    for row in auxiliary['owned']+auxiliary['borrowed']:
        for fd in row['fds']:
            add(fd,row['role'],'phase',row['phase'],'current_auxiliary_once_only_cleanup/'+row['consumer'],row.get('path'))
    guardian=Root_expected['retained_fd_custody']['actual_original_guardian_pidfd']
    add(guardian,'original_guardian_pidfd','batch',None,'original_GNU120_guard_until_all_phase_closure')
    return owner.root_fd_custody.preflight_role_plan(Root_expected['retained_fd_custody'],plans,phase_order)
