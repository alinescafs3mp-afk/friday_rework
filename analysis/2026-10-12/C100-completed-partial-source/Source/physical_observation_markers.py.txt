def _tracer_pid():
    rows=[r.split()[1] for r in Path('/proc/self/status').read_text().splitlines()
          if r.startswith('TracerPid:')]
    if len(rows)!=1 or not rows[0].isdecimal():raise ValueError('ACTUAL_TRACER_PID_UNKNOWN')
    return int(rows[0])

def configure(binding):
    """Called by admitted outside code after exact pins, BEFORE target effects.

    Binding is a declaration; the live TracerPid is read independently. The
    observer separately verifies the matching session and current source bytes.
    There is no observer fallback and no registration sourced from a t fixture.
    """
    global _active
    if _active is not None:raise ValueError('OBSERVATION_SESSION_REBIND_FORBIDDEN')
    if type(binding)is not dict or set(binding)!={'observer_pid','session_word','subject_birth'}:
        raise ValueError('OBSERVATION_EXTERNAL_BINDING_UNBOUND')
    if binding['observer_pid']<=0 or _tracer_pid()!=binding['observer_pid']:
        raise ValueError('GENUINE_INDEPENDENT_OBSERVER_NOT_ATTACHED')
    fields=Path('/proc/self/stat').read_text().rsplit(')',1)[1].split()
    if fields[19]!=str(binding['subject_birth']):raise ValueError('OBSERVATION_SUBJECT_BIRTH')
    if type(binding['session_word'])is not int or not 0<binding['session_word']<2**63:
        raise ValueError('OBSERVATION_EXACT_SESSION')
    library=ctypes.CDLL(None,use_errno=True)
    library.syscall.restype=ctypes.c_long
    _active=(binding['observer_pid'],binding['session_word'],library)

def mark(name,a=0,b=0,c=0):
    if _active is None:raise ValueError('GENUINE_RAW_OBSERVER_UNBOUND_BEFORE_EFFECT')
    observer,word,library=_active
    if _tracer_pid()!=observer:raise ValueError('GENUINE_RAW_OBSERVER_LOST_STOP')
    values=(MAGIC,word,EVENTS[name],a,b,c)
    if any(type(x)is not int or not 0<=x<2**64 for x in values):
        raise ValueError('RAW_MARKER_ARGUMENT_SHAPE')
    got=library.syscall(ctypes.c_long(SYS_GETTID),
        *(ctypes.c_ulonglong(x) for x in values))
    if got!=os.getpid():
        raise ValueError('RAW_MARKER_ACTUAL_SYSCALL_OR_UNEXPECTED_THREAD')
    return got

def refusal(error):
    # The value comes only from the caught REAL exception, never expected code.
    actual_text=str(error)
    code=getattr(error,'code',None) or (actual_text if actual_text in REFUSALS else actual_text.split(':',1)[0])
    if isinstance(error,OSError) and error.errno==9:code='KERNEL_CLOSE_ERRNO_9'
    number=REFUSALS.index(code)+1 if code in REFUSALS else 0
    mark('refusal',number)
    return code

def observe_state(kind,state):
    """Publish real consumer arguments/state for independent ptrace reading.
    These bytes are DECLARATIONS; observer reducers must cross-check FD roles,
    identities, close outcomes and waits against actual kernel observations.
    No t, expected assertion value or authorization may be published here.
    """
    import json
    kinds={'custody_before':1,'custody_after':2,'phase_begin':3,'phase_closed':4,
           'batch_closed':5,'scan_dirfd':6,'scan_closed':7,'child_hygiene':8,
           'child_hygiene_closed':9,'auxiliary_before':10,'auxiliary_after':11,
           'owned_wait_complete':12,'case_entry':13,'operation_result':14,
           'caller_closed':15,'STOP_custody':16}
    raw=json.dumps(state,sort_keys=True,separators=(',',':'),allow_nan=False).encode()
    if not 0<len(raw)<=1048576:raise ValueError('ACTUAL_SEMANTIC_STATE_BOUND')
    buffer=ctypes.create_string_buffer(raw)
    mark('semantic_read',ctypes.addressof(buffer),len(raw),kinds[kind])
    # Buffer stays live through the independently intercepted entry/exit.
    return None
