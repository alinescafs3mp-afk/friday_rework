import copy

def _refuses(fn,code,*args):
    try:fn(*args)
    except ValueError as error:assert error.code==code
    else:raise AssertionError('expected concrete refusal')

def test_c45_missing_actual_root_provider(adapter):
    _refuses(adapter.verify_existing_root_facility,'MISSING_EXISTING_ROOT_FACILITY',{}, {}, {},0,0,'boot')

def test_c45_wrong_original_source_budget(adapter,valid_clock):
    for seconds in (1800,3600,True,120.0):
        clock=copy.deepcopy(valid_clock);clock['original_seconds']=seconds
        _refuses(adapter.original_root_clock,'ORIGINAL_CLOCK',clock,clock['boot_id'],clock['started_at_monotonic']+20,clock['started_at_unix']+20)

def test_c45_dualclock_drift(adapter,valid_clock):

    _refuses(adapter.original_root_clock,'ORIGINAL_CLOCK',c,c['boot_id'],c['started_at_monotonic']+20,c['started_at_unix']+23)

def test_c45_renewed_final(adapter,valid_clock):

    _refuses(adapter.original_root_clock,'ORIGINAL_CLOCK',c,c['boot_id'],c['started_at_monotonic']+20,c['started_at_unix']+20)

def test_c45_nonfinite_bool_clock(adapter,valid_clock):
    for value in (True,float('inf'),float('nan')):
        c=copy.deepcopy(valid_clock);c['started_at_unix']=value
        _refuses(adapter.original_root_clock,'ORIGINAL_CLOCK',c,c['boot_id'],20,20)

def test_c45_complete_original_clock_positive(adapter,valid_clock):
    c=valid_clock;r=adapter.original_root_clock(c,c['boot_id'],c['started_at_monotonic']+20,c['started_at_unix']+20)
    assert r['actor_stop_m']==c['started_at_monotonic']+100 and r['stock_stop_m']==c['started_at_monotonic']+110
    assert r['final_u']==c['final_deadline_unix']

def memory_conflict_plan(batch):
    ids=[x[0] for x in batch.CASES];positive=[x[0] for x in batch.CASES if x[1]=='positive']
    negative=[x[0] for x in batch.CASES if x[1]=='negative']
    cuts=[x[0] for x in batch.CASES if x[1]=='source_uncertainty_after_actual_operation']
    pin='SYNTHETIC_MEMORY_ONLY_NOT_AUTHORITY'
    absolute={'start_m':1000,'start_u':2000,'useful_m':1090,'useful_u':2090,'final_m':1120,'final_u':2120}
    phases=[]
    # Intentionally overlapping cut intervals are a NEGATIVE fixture, never an
    # accepted execution schedule or proof that the real full32 packet fits.
    for members,entry,terminal,birth in [(negative,0,1,False),(positive,1,62,True)]+[( [cut],20,81,True) for cut in cuts]:
        phase={'case_ids':members,'original_clock_pin':pin,'reset_clock':False,'entry_m':1000+entry,
               'entry_u':2000+entry,'terminal_m':1000+terminal,'terminal_u':2000+terminal,
               'worst_time_s':61 if birth else 1,'IO_bytes_upper':75497472 if birth else 65536,
               'inodes_upper':512 if birth else 32,'first_birth_required':birth}
        if birth:phase.update({'whole_memory':2147483648,'swap':0,'Linux_tasks':128,'cpu_max':'200000 100000',
            'AS_per_outside_role':268435456,'physical_child_seconds':46,'physical_outside_remaining_s':51})
        if members==positive:phase['probe_mode']='ALL_SAFE'
        phases.append(phase)
    return {'original_clock_pin':pin,'case_ids':ids,'actual_complete_time_IO_resource_fit_ref':'SYNTHETIC_MEMORY_ONLY',
        'Root_independently_verified_complete_physical_fit':True,'phases':phases,'provider_max_simultaneous_cohorts':1,
        'whole_lifetime_IO_upper':536870912,'whole_lifetime_inode_upper':4096},pin,absolute

def run_c45_memory_controls(adapter,batch):
    """Finite pure consistency controls only; never Root/physical admission.
    Independent Root may call ONLY after final-body safety review. This entry
    never executes a launcher, candidate Source import, fixture/kernel effect.
    """
    c=memory_clock();spec,expected,c2=memory_wait_inputs();p,pin,absolute=memory_conflict_plan(batch)
    test_c45_missing_actual_root_provider(adapter)
    test_c45_wrong_original_source_budget(adapter,c)
    test_c45_dualclock_drift(adapter,c)
    test_c45_renewed_final(adapter,c)
    test_c45_nonfinite_bool_clock(adapter,c)
    test_c45_complete_original_clock_positive(adapter,c)

    test_c45_missing_or_repeated_original_case(batch,p,pin,absolute,1000,2000)
