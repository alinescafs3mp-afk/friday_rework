class BatchRefusal(ValueError):
    def __init__(self,code,detail):
        self.code=code
        super().__init__(code+': '+detail)


def need(ok,code,detail):
    if not ok:raise BatchRefusal(code,detail)


def finite(v):return type(v) in (int,float) and math.isfinite(v)


def assert_complete_plan(plan,original_clock_pin,absolute,now_m,now_u):
    need(type(plan)is dict and plan.get('original_clock_pin')==original_clock_pin,
         'BATCH_ORIGINAL_CLOCK','one original clock, no per-case+120 or Source1800')
    ids=[x[0] for x in CASES]
    need(plan.get('case_ids')==ids and len(ids)==len(set(ids))==32,
         'BATCH_SCOPE','exact original order/10positive/18negative/4owner cuts')
    need(plan.get('actual_complete_time_IO_resource_fit_ref') is not None
         and plan.get('Root_independently_verified_complete_physical_fit') is True,
         'BATCH_PHYSICAL_FIT_UNBOUND','no measured/admitted whole32 originalclock fit')
    phases=plan.get('phases');need(type(phases)is list and 1<=len(phases)<=32,
                               'BATCH_PHASES','finite existing facility ordering')
    seen=[];positive=set(x[0] for x in CASES if x[1]=='positive')
    cuts=set(x[0] for x in CASES if x[1]=='source_uncertainty_after_actual_operation')
    total_io=0;total_inodes=0;intervals=[];cohorts=0
    for phase in phases:
        need(type(phase)is dict and phase.get('original_clock_pin')==original_clock_pin
             and phase.get('reset_clock') is False,'BATCH_ORIGINAL_CLOCK','same immutable phase clock')
        members=phase.get('case_ids');need(type(members)is list and members and len(members)==len(set(members))
              and set(members)<=set(ids),'BATCH_SCOPE','exact unique intended causes')
        seen.extend(members)
        fields=('entry_m','entry_u','terminal_m','terminal_u','worst_time_s')
        need(all(finite(phase.get(k)) for k in fields),'BATCH_TIME','finite independently bound interval')
        need(phase['entry_m']>=now_m and phase['entry_u']>=now_u and phase['worst_time_s']>=0
             and phase['terminal_m']>=phase['entry_m']+phase['worst_time_s']
             and phase['terminal_u']>=phase['entry_u']+phase['worst_time_s']
             and phase['terminal_m']<absolute['final_m'] and phase['terminal_u']<absolute['final_u']
             and abs((phase['entry_m']-absolute['start_m'])-(phase['entry_u']-absolute['start_u']))<=2
             and abs((phase['terminal_m']-absolute['start_m'])-(phase['terminal_u']-absolute['start_u']))<=2,
             'BATCH_TIME','actual remaining full phase/cleanup fits BOTH immutable final clocks')
        for k in ('IO_bytes_upper','inodes_upper'):
            need(type(phase.get(k))is int and phase[k]>=0,'BATCH_IO','actual finite integer reservation')
        total_io+=phase['IO_bytes_upper'];total_inodes+=phase['inodes_upper']
        if phase.get('first_birth_required') is True:
            cohorts+=1
            need(phase['entry_m']+45<absolute['useful_m'] and phase['entry_u']+45<absolute['useful_u'],
                 'BATCH_USEFUL','unchanged45 useful BEFORE each actual firstbirth')
            need(phase.get('whole_memory')==2147483648 and phase.get('swap')==0
                 and phase.get('Linux_tasks')==128 and phase.get('cpu_max')=='200000 100000'
                 and phase.get('AS_per_outside_role')==268435456,
                 'BATCH_RESOURCES','original whole2GiB/swap0/tasks128/CPU2 and AS256 outside roles')
            child=phase.get('physical_child_seconds');outside=phase.get('physical_outside_remaining_s')
            need(type(child)is int and 45<child<=85 and finite(outside)
                 and child+5<=outside<=90 and phase['worst_time_s']>=outside+10,
                 'BATCH_PHYSICAL_WAIT_BOUND','actual ALREADY-ELAPSED GNU guardian remainder plus original Root cleanup; no zero-time cut')
            intervals.append((phase['entry_m'],phase['terminal_m'],phase['entry_u'],phase['terminal_u']))
        if set(members)&positive:
            need(set(members)==positive and phase.get('probe_mode') in ('ALL_SAFE','ALL_SAFE_THEN_UNKNOWN'),
                 'BATCH_POSITIVE_COHORT','all10 unchanged positives share the complete actual probe workload')
        cut=set(members)&cuts
        if cut:
            need(len(cut)==len(members)==1 and phase.get('first_birth_required') is True,
                 'BATCH_OWNER_CUT','four distinct actual-owner cut contexts, never labels on one execution')
            need(19<=phase['entry_m']-absolute['start_m']<=21
                 and 19<=phase['entry_u']-absolute['start_u']<=21,
                 'BATCH_OWNER_CUT_WINDOW','each original assertion retains actual19..21sec entry')
    need(len(seen)==len(set(seen))==32 and set(seen)==set(ids),'BATCH_SCOPE','no missing/repeated cause')
    need(cohorts==5,'BATCH_PHYSICAL_COHORTS','one allsafe plus four actual distinct owner cuts')
    need(type(plan.get('provider_max_simultaneous_cohorts'))is int
         and plan['provider_max_simultaneous_cohorts']==1,
         'MISSING_EXISTING_SERIAL_ROOT_FACILITY','existing childless finite owner is serial; no invented parallel topology')
    intervals.sort()
    for previous,current in zip(intervals,intervals[1:]):
        need(previous[1]<=current[0] and previous[3]<=current[2],
             'BATCH_SERIAL_WINDOW_CONFLICT','all4 original19..21 cuts cannot overlap exclusive existing owner')
    need(type(plan.get('whole_lifetime_IO_upper'))is int and total_io<=plan['whole_lifetime_IO_upper']
         and type(plan.get('whole_lifetime_inode_upper'))is int and total_inodes<=plan['whole_lifetime_inode_upper'],
         'BATCH_LIFETIME_IO','aggregate all phases, never occupancy or percase budget reset')
    # Binding a measured full32 fit is NOT supported merely by this arithmetic.
    # Root must supply existing admitted provider/trace collection before GO.
    return {'total':32,'positive':10,'negative':18,'owner_cuts':4,'physical_cohorts':5,
            'total_IO_upper':total_io,'total_inode_upper':total_inodes,'clock_reset':False,
            'capacity_release':False,'Source_arithmetic_is_not_admission':True}


def before_any_batch_GO(Root_expected,plan,clock_pin,absolute,now_m,now_u):
    need(type(Root_expected)is dict and Root_expected.get('existing_supported_whole32_batch_source') is not None
         and Root_expected.get('actual_before_firstbirth_wait_provider') is not None,
         'MISSING_EXISTING_ROOT_BATCH_PROVIDER','INPUTS providers are UNBOUND; no launch/new facility/G8 retry')
    need(Root_expected.get('independent_final_Source_safety_acceptance') is not None
         and Root_expected.get('actual_specific_kernel120_admission') is not None,
         'BATCH_NOT_ADMITTED','candidate Source/JSON does not grant kernel120')
    return assert_complete_plan(plan,clock_pin,absolute,now_m,now_u)


def drive_all32_actual_assertions(owner,tests,Root_expected,trace_refs,clock_pin):
    """Existing Root actual trace interface, not an executor/provider substitute.
    Verify all real pinned inputs first; call every unchanged intended assertion
    on its own complete actual Root trace. No partial total or unrelated-error
    pass. A shared positive cohort still has ten distinct cause observations.
    Source author cannot use this as independent final acceptance.
    """
    import json
    ids=[x[0] for x in CASES]
    need(type(Root_expected)is dict and Root_expected.get('independent_final_Source_safety_acceptance') is not None
         and Root_expected.get('actual_specific_kernel120_admission') is not None,
         'BATCH_NOT_ADMITTED','outside Root actual final body/action admission required')
    need(Root_expected.get('actual_complete_batch_finished_under_original_clock_ref') is not None,
         'BATCH_PHYSICAL_FIT_UNBOUND','actual full32 oneclock completion prerequisite missing')
    need(type(trace_refs)is dict and set(trace_refs)==set(ids),'BATCH_SCOPE','exact all32 Root raw trace pins')
    traces={};functions={}
    for case_id,kind,name,cause in CASES:
        trace=json.loads(owner.same_read_pin(trace_refs[case_id],65536))
        need(trace.get('case_id')==case_id and trace.get('intended_cause')==cause
             and trace.get('original_clock_ref')==clock_pin
             and trace.get('Root_expected_original_clock_ref')==clock_pin
             and trace.get('simulated') is False and trace.get('unknown_as_zero') is False
             and trace.get('old_clock_or_Source_credit_transferred') is False,
             'BATCH_TRACE_CAUSE','actual exact case/clock/cause, no Source history transfer')
        function=getattr(tests,name,None)
        need(callable(function),'BATCH_ASSERTION_MISSING',name)
        traces[case_id]=trace;functions[case_id]=function
    result=[]
    for case_id,kind,name,cause in CASES:
        try:functions[case_id](traces[case_id])
        except Exception as error:
            result.append({'id':case_id,'kind':kind,'state':'FAIL','error_type':type(error).__name__})
        else:result.append({'id':case_id,'kind':kind,'state':'PASS'})
    need(len(result)==32,'BATCH_SCOPE','full mandatory assertions, no early green partial return')
    return {'results':result,'passed':sum(r['state']=='PASS' for r in result),
            'failed':sum(r['state']=='FAIL' for r in result),'total':32,'clock_reset':False,
            'independent_acceptance':'ROOT_SEPARATE_REVIEW_REQUIRED','capacity_release':False,
            'native_genuine_or_installed_runtime_acceptance':False}
