def same_read_pin(ref,maximum=2097152):
    require(type(ref) is dict and set(ref)=={'path','bytes','sha256'},'exact pin')
    p=Path(ref['path']);require(p.is_absolute() and '..' not in p.parts
        and str(p)==os.path.normpath(str(p)),'absolute normalized pin')
    require(type(ref['bytes']) is int and 0<ref['bytes']<=maximum,'bounded pin bytes')
    for parent in p.parents:require(stat.S_ISDIR(parent.lstat().st_mode),'pin ancestor no symlink')
    fd=os.open(p,os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK)
    try:
        before=os.fstat(fd);require(stat.S_ISREG(before.st_mode) and before.st_nlink==1
            and before.st_size==ref['bytes'],'exact regular single-link pin')
        raw=bytearray()
        while len(raw)<=ref['bytes']:
            b=os.read(fd,min(65536,ref['bytes']+1-len(raw)))
            if not b:break
            raw.extend(b)
        after=os.fstat(fd)
        require((before.st_dev,before.st_ino,before.st_size,before.st_mtime_ns,before.st_ctime_ns)
                ==(after.st_dev,after.st_ino,after.st_size,after.st_mtime_ns,after.st_ctime_ns)
                and len(raw)==ref['bytes'] and hashlib.sha256(raw).hexdigest()==ref['sha256'],
                'same-read digest and kernel identity')
        return bytes(raw)
    finally:os.close(fd)


def exact_birth(pid):

    return {'pid':pid,'birth':v[19],'state':v[0],'ppid':int(v[1]),'pgid':int(v[2]),'sid':int(v[3])}

def existing_owner_kernel_observation(spec,group_fd):
    """ONE bounded Root checkpoint, not a polling worker or authority receipt.
    Exact private kernel group is the membership source; count Linux TIDs,
    including threads, descendants and Z. Never infer unknown as zero.
    """
    group=Path(spec['native']['group']);q=os.fstat(group_fd);g=group.stat()
    require([q.st_dev,q.st_ino]==[g.st_dev,g.st_ino]==spec['native']['first_birth_group_identity'],
            'actual retained group inode at observation')
    def small(path):
        fd=os.open(path,os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK)
        try:
            b=os.read(fd,65537);require(len(b)<=65536,'bounded actual kernel observation');return b.decode()
        finally:os.close(fd)
    first={k:small(group/k) for k in ('pids.current','cgroup.events','cgroup.procs','cgroup.threads',
                                    'memory.max','memory.swap.max','pids.max','cpu.max','memory.events','pids.events','cpu.stat')}
    ids=[int(x) for x in first['cgroup.threads'].split()];require(len(ids)<=128,'actual bounded Linux TID inventory')
    require(not ({r['pid'] for r in spec['existing_owner']['protected_writers']} & set(ids)), 'protected writer excluded')
    rows=[]
    for tid in ids:
        try:
            identity=exact_birth(tid);c=small('/proc/'+str(tid)+'/cgroup').strip()
            require(c=='0::'+spec['native']['group_relative'],'only current private owned cohort')
            status=small('/proc/'+str(tid)+'/status')
            row={'tid':tid,'identity':identity,'cgroup':c,'status_raw':status}
            if identity['state']=='Z':row['zombie_no_live_probe_or_reap_credit']=True
            else:row['limits_raw']=small('/proc/'+str(tid)+'/limits')
            rows.append(row)
        except FileNotFoundError:rows.append({'tid':tid,'state':'UNKNOWN_EXITED_DURING_SAME_CHECKPOINT_NOT_ZERO'})
    last={k:small(group/k) for k in ('pids.current','cgroup.events','cgroup.threads')}
    empty=first['pids.current'].strip()==last['pids.current'].strip()=='0'
    require(not empty or (not ids and not last['cgroup.threads'].strip()
            and 'populated 0' in first['cgroup.events'] and 'populated 0' in last['cgroup.events']),
            'actual fresh kernel0 agreement, no existence/wait surrogate')
    return {'group_identity':[q.st_dev,q.st_ino],'first_raw':first,'last_raw':last,'owned_task_rows':rows,
        'actual_kernel_zero':empty,'capacity_release':False,'at_unix':time.time(),'at_monotonic':time.monotonic(),
        'unknown_as_zero':False,'Root_must_independently_pin_and_accept_this_actual_checkpoint':True}

def existing_root_capture_private_trace(spec, output_fd, evidence_fd, evidence_identity, root_wait, final_m, final_u):
    """Finite extension of existing Root custody, AFTER direct/adopted wait.
    Supplied destination is an ALREADY-owned private evidence directory outside
    the output mount, independently admitted by Root. No mkdir/new provider.
    Copy every remaining finite raw file, same-read/pin BEFORE normal umount.
    Overlarge/unknown files retain mount/group/slot, never discard evidence.
    """
    require(spec['domain']=='kernel120_qualification','harmless Root trace only')
    require(type(evidence_fd) is int and evidence_fd>2 and evidence_fd!=output_fd,
            'existing distinct Root evidence capability; no invented FD')
    q=os.fstat(evidence_fd);o=os.fstat(output_fd)
    require(stat.S_ISDIR(q.st_mode) and [q.st_dev,q.st_ino]==evidence_identity
            and stat.S_IMODE(q.st_mode)==0o700 and q.st_dev!=o.st_dev,
            'Root existing private evidence directory OUTSIDE output filesystem')
    require(not os.listdir(evidence_fd),'one new admitted evidence slice, no overwrite/replay')
    # 2MiB total raw trace is the original exterior evidence reservation.
    # Live quota debris beyond it is a concrete refusal; cannot truncate it.
    def before_IO():
        require(time.monotonic()+5<final_m and time.time()+5<final_u,
                'Root original final reserve BEFORE raw trace IO; outside original guardian still mandatory')
    before_IO()
    pending=[(os.dup(output_fd),'')];rows=[];total=0;entries=0;created=[]
    try:
        while pending:
            before_IO()
            directory,prefix=pending.pop()
            try:
                names=sorted(os.listdir(directory));entries+=len(names)
                require(entries<=512,'finite existing output inode inventory')
                for name in names:
                    before_IO()
                    require(type(name)is str and name not in ('.','..') and '/' not in name
                            and '\x00' not in name,'exact output component')
                    rel=prefix+name;s=os.stat(name,dir_fd=directory,follow_symlinks=False)
                    if stat.S_ISDIR(s.st_mode):
                        require(prefix.count('/')<8,'finite raw directory depth')
                        child=os.open(name,os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW,dir_fd=directory)
                        f=os.fstat(child)
                        require([f.st_dev,f.st_ino]==[s.st_dev,s.st_ino] and f.st_dev==o.st_dev,
                                'same private mount directory')
                        pending.append((child,rel+'/'));continue
                    require(stat.S_ISREG(s.st_mode) and s.st_nlink==1 and s.st_dev==o.st_dev
                            and 0<=s.st_size<=2097152-total,'all raw files finite/owned/singlelink')
                    source=os.open(name,os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK,dir_fd=directory)
                    try:
                        before=os.fstat(source);require([before.st_dev,before.st_ino]==[s.st_dev,s.st_ino],
                                                       'same actual raw file birth')
                        raw=bytearray()
                        while len(raw)<=s.st_size:
                            before_IO()
                            part=os.read(source,min(65536,s.st_size+1-len(raw)))
                            if not part:break
                            raw.extend(part)
                        after=os.fstat(source)
                        require((before.st_dev,before.st_ino,before.st_size,before.st_mtime_ns,before.st_ctime_ns)
                                ==(after.st_dev,after.st_ino,after.st_size,after.st_mtime_ns,after.st_ctime_ns)
                                and len(raw)==s.st_size,'exact stable raw read after actual waits')
                    finally:os.close(source)
                    digest=hashlib.sha256(raw).hexdigest();target='raw-'+str(len(rows)).zfill(4)+'-'+digest
                    fd=os.open(target,os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o400,dir_fd=evidence_fd)
                    created.append(target)
                    try:
                        view=memoryview(raw)
                        while view:
                            before_IO()
                            n=os.write(fd,view);require(n>0,'actual bounded raw evidence progress');view=view[n:]
                        before_IO();os.fsync(fd);saved=os.fstat(fd)
                    finally:os.close(fd)
                    total+=len(raw);rows.append({'original_relative':rel,'bytes':len(raw),'sha256':digest,
                        'saved_name':target,'saved_identity':[saved.st_dev,saved.st_ino],
                        'original_identity':[before.st_dev,before.st_ino]})
            finally:os.close(directory)
        document={'schema':'existing-root-private-raw-trace.v1','case':spec['qualification_case'],
                  'root_actual_wait':root_wait,'files':rows,'total_bytes':total,'capacity_release':False,
                  'candidate_raw_trace_is_not_independent_acceptance':True}
        raw=json.dumps(document,sort_keys=True,separators=(',',':'),allow_nan=False).encode()
        require(len(raw)<=65536,'bounded Root raw trace index')
        before_IO()
        fd=os.open('root-raw-index.json',os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o400,dir_fd=evidence_fd)
        try:
            view=memoryview(raw)
            while view:
                before_IO()
                n=os.write(fd,view);require(n>0,'Root raw index progress');view=view[n:]
            before_IO();os.fsync(fd)
        finally:os.close(fd)
        before_IO();os.fsync(evidence_fd)
        return {'index_name':'root-raw-index.json','bytes':len(raw),'sha256':hashlib.sha256(raw).hexdigest(),
                'raw_files':len(rows),'raw_total_bytes':total,'actual_wait_before_copy':True,
                'normal_umount_not_yet_done':True,'capacity_release':False}
    except BaseException:
        # Existing Root retains custody of partial evidence as well as its
        # original mount/group/flock. No unlink/replay/lazy unmount on failure.
        for fd,_ in pending:
            try:os.close(fd)
            except OSError:pass
        raise
