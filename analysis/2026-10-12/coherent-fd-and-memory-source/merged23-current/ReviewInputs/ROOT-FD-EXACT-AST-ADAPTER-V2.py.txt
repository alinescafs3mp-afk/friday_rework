"""Concrete future memory-only RootFD AST adapter, SOURCE_ONLY/NOT_RUN.
No imports, file reads, process/thread/FD/native effects, sys.modules changes,
Root provider, real finalizer calls, or authority creation. Root separately
reviews final package and authenticates a new bounded-memory permission.
Only fixed pinned fake controls may call extracted bodies after that admission.
"""
SOURCE_NAMES = {'root_fd_custody.py': ('require_custody', 'own', 'bind_existing', 'opened', 'duplicated', 'closed', 'stopped'), 'stock_install_owner.py': ('require', 'existing_root_capture_private_trace'), 'test_c45_root_fd_custody.py': ('FakeFDs', 'capture_trial', 'test_root_fd_normal_capture', 'test_root_fd_fstat_failure_registered', 'test_root_fd_identity_failure_registered', 'test_root_fd_open_failure_no_lost_handle', 'test_root_fd_dup_failure_retains_exterior', 'test_root_fd_close_ambiguity_no_retry', 'test_root_fd_partial_target_failure_retained', 'test_root_fd_full_finalizer_order_static', 'run_root_fd_memory_controls')}
SOURCE_PINS = {'root_fd_custody.py': {'path': 'pin-paths/p000139', 'bytes': 2326, 'sha256': 'bb0d8527f7986f16b09277a27b329cc2b224de4964bad0cc1190de293e977555'}, 'stock_install_owner.py': {'path': 'pin-paths/p000141', 'bytes': 34485, 'sha256': '34c1e9f3e0a6d62034fc978702db16a05fcac0e7bacb7052f255b4e4b4f431ba'}, 'test_c45_root_fd_custody.py': {'path': 'pin-paths/p000143', 'bytes': 9385, 'sha256': '6c5baa39d439a47b39db4459b0863925132be84818efed829e66536395ed2e0b'}}
LIMITS = {'AS': 268435456, 'CPU': 30, 'FSIZE': 67108864, 'CORE': 0, 'affinity': [0, 1], 'GNU_KILL_SECONDS': 30}

def root_admitted(root_context):
    # These fields are consistency checks; Root authentication is external.
    assert type(root_context) is dict
    assert root_context['decision'] == 'ADMIT_EXACT_BOUNDED_MEMORY_AST'
    assert root_context['scope'] == 'MEMORY_ONLY_NO_NATIVE_KERNEL_LIVE_CREDIT'
    assert root_context['external_root_authenticated'] is True
    assert root_context['final_corrected_source_and_harness_independently_accepted'] is True
    assert root_context['exact_adapter_independently_reviewed'] is True
    assert root_context['actual_caps_before_GNU'] == LIMITS
    assert root_context['original_clocks_usage_stops_preserved'] is True
    assert root_context['source_pins'] == SOURCE_PINS

def memory_builtins():
    return {'__build_class__': __build_class__, 'BaseException': BaseException,
            'Exception': Exception, 'AssertionError': AssertionError,
            'ValueError': ValueError, 'OSError': OSError, 'type': type,
            'dict': dict, 'list': list, 'tuple': tuple, 'set': set,
            'str': str, 'int': int, 'bool': bool, 'bytes': bytes,
            'bytearray': bytearray, 'memoryview': memoryview, 'len': len,
            'min': min, 'max': max, 'sorted': sorted, 'next': next,
            'enumerate': enumerate, 'all': all, 'any': any,
            'isinstance': isinstance, 'sum': sum}

def extract_fd_memory_modules(source_blobs, root_context, ast_module,
                              types_module, json_module, hashlib_module, stat_module):
    root_admitted(root_context)
    assert type(source_blobs) is dict and set(source_blobs) == set(SOURCE_NAMES)
    modules = {}
    for filename in SOURCE_NAMES:
        raw = source_blobs[filename]
        expected = SOURCE_PINS[filename]
        assert type(raw) is bytes and 0 < len(raw) == expected['bytes'] <= 2097152
        assert hashlib_module.sha256(raw).hexdigest() == expected['sha256']
        tree = ast_module.parse(raw, filename=filename)
        selected = []
        found = set()
        for node in tree.body:
            if not isinstance(node, (ast_module.FunctionDef, ast_module.ClassDef)):
                continue
            if node.name not in SOURCE_NAMES[filename]:
                continue
            assert node.name not in found and not node.decorator_list
            for child in ast_module.walk(node):
                assert not isinstance(child, (ast_module.Import, ast_module.ImportFrom))
                if isinstance(child, (ast_module.FunctionDef, ast_module.ClassDef)):
                    assert not child.decorator_list
                if isinstance(child, ast_module.Name):
                    assert child.id not in {'__import__','open','eval','exec','compile',
                        'Path','resource','sys','threading','subprocess','ctypes',
                        'signal','select','fcntl','globals','locals'}
                if isinstance(child, ast_module.Attribute):
                    assert child.attr not in {'__globals__','__code__','__subclasses__','__mro__'}
                if isinstance(child, ast_module.FunctionDef):
                    assert child.returns is None
                    for arg in child.args.posonlyargs + child.args.args + child.args.kwonlyargs:
                        assert arg.annotation is None
                    for default in child.args.defaults + [v for v in child.args.kw_defaults if v is not None]:
                        ast_module.literal_eval(default)
            found.add(node.name)
            selected.append(node)
        assert found == set(SOURCE_NAMES[filename])
        module = types_module.ModuleType('_memory_only_' + filename.replace('.', '_'))
        module.__dict__.update(__builtins__=memory_builtins(),
            os=types_module.SimpleNamespace(),
            time=types_module.SimpleNamespace(),
            json=json_module, hashlib=hashlib_module, stat=stat_module,
            ast=ast_module, SimpleNamespace=types_module.SimpleNamespace)
        if filename == 'stock_install_owner.py':
            module.root_fd_custody = modules['root_fd_custody.py']
        # No Source top import or real OS object exists in these namespaces.
        exact_nodes = ast_module.Module(body=selected, type_ignores=[])
        code = compile(ast_module.fix_missing_locations(exact_nodes), filename + ':memory-only', 'exec')
        exec(code, module.__dict__, module.__dict__)
        modules[filename] = module
    return modules

def run_fd_memory_controls(source_blobs, root_context, ast_module,
                           types_module, json_module, hashlib_module, stat_module):
    modules = extract_fd_memory_modules(source_blobs, root_context, ast_module,
                                       types_module, json_module, hashlib_module, stat_module)
    controls = modules['test_c45_root_fd_custody.py']
    # capture_trial installs exact FakeFDs and finite fake clocks before each
    # capture call, then restores the inert objects in the same module dicts.
    result = controls.run_root_fd_memory_controls(modules['stock_install_owner.py'],
        modules['root_fd_custody.py'], source_blobs['stock_install_owner.py'].decode())
    assert result['controls'] == 8 and result['close_ambiguity_variants'] == 2
    assert result['actual_kernel_calls'] == 0 and result['capacity_release'] is False
    return {'observations': result, 'credit': 'FAKE_MEMORY_ONLY_NO_RUNTIME_ACCEPTANCE',
            'Source310_rerun': False, 'real_finalizer_executed': False}
