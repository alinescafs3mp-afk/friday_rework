def expect_code(rows,label,fn,code,budget):
    budget.step()
    try:fn()
    except (ValueError,AssertionError) as error:
        if getattr(error,'code',None)!=code:raise AssertionError(label+': wrong refusal '+type(error).__name__)
    else:raise AssertionError(label+': negative did not refuse')
    rows.append({'id':label,'state':'MEMORY_EXPECTED_REFUSAL','code':code,'actual_acceptance':False})

def extractor_negative_controls(sources,budget):
    rows=[];name='existing_root_adapter.py';rule=SPEC['rules'][name]['original_root_clock']
    for text,code in [("def original_root_clock():\n import os\n",'EXTRACT_FORBIDDEN_AST'),
        ("def original_root_clock():\n while True: pass\n",'EXTRACT_FORBIDDEN_AST'),
        ("def original_root_clock():\n return __import__('os')\n",'EXTRACT_CALL'),
        ("def original_root_clock():\n os.open('x',0)\n",'EXTRACT_CALL'),
        ("def original_root_clock():\n return clock.__class__\n",'EXTRACT_DUNDER'),
        ("def original_root_clock():\n return getattr(clock,'get')\n",'EXTRACT_CALL'),
        ("@evil\ndef original_root_clock():\n return 0\n",'EXTRACT_DECORATOR'),
        ("def original_root_clock(x=evil()):\n return x\n",'EXTRACT_DEFAULT')]:
        node=ast.parse(text).body[0]
        expect_code(rows,'extractor_forbidden_'+str(len(rows)),lambda node=node:check_pure_node(node,rule,budget),code,budget)
    changed=dict(sources);changed[name]=sources[name]+b'\n# drift\n'
    expect_code(rows,'extractor_source_pin_drift',lambda:extract_modules(changed,budget),'EXTRACT_SOURCE_PIN',budget)
    changed=dict(sources);del changed['qualification_batch.py']
    expect_code(rows,'extractor_missing_source',lambda:extract_modules(changed,budget),'EXTRACT_SCOPE',budget)
    changed=dict(sources);changed['extra.py']=b'pass\n'
    expect_code(rows,'extractor_extra_source',lambda:extract_modules(changed,budget),'EXTRACT_SCOPE',budget)
    return rows

        (1<<129,'MEMORY_INTEGER','integer129bits')]:

    cycle=[];cycle.append(cycle)

    return rows
