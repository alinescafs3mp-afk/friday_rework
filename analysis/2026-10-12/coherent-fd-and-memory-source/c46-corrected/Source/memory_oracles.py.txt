        need(t.get(key)==e.get(key) and type(t.get(key)) is list and len(t[key])==2,'MEMORY_OWNED_IDENTITY')

def assert_memory_all32_shape(envelope,records,cases,budget):
    e=copy_graph(envelope,budget);ids=[x[0] for x in cases]
    need(e.get('case_ids')==ids and len(ids)==len(set(ids))==32,'MEMORY_32_SCOPE')
    need(type(e.get('trace_refs')) is dict and list(e['trace_refs'])==ids,'MEMORY_32_ORDER')
    need(e.get('original_clock_pin')==TOKEN and e.get('clock_reset') is False and e.get('capacity_release') is False,'MEMORY_32_CLOCK')
    owner=MemorySameReadOwner(records,budget);counts={'positive':0,'negative':0,'source_uncertainty_after_actual_operation':0}
    for case_id,kind,name,cause in cases:
        raw=owner.same_read_pin(e['trace_refs'][case_id],65536)
        trace=copy_graph(json.loads(raw),budget)
        need(type(trace) is dict,'MEMORY_32_CAUSE')
        need(trace.get('memory_only') is True and trace.get('simulated') is True
             and trace.get('actual_admission') is False and trace.get('producer_consumer_shape_only') is True,'MEMORY_NOT_AUTHORITY')
        need(trace.get('case_id')==case_id and trace.get('kind')==kind
             and trace.get('assertion_name')==name and trace.get('intended_cause')==cause,'MEMORY_32_CAUSE')
        need(trace.get('original_clock_ref')==TOKEN and trace.get('Root_expected_original_clock_ref')==TOKEN
             and trace.get('unknown_as_zero') is False and trace.get('old_clock_or_Source_credit_transferred') is False,'MEMORY_32_CLOCK')
