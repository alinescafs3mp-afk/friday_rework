"""Finite named MEMORY fakes. No OS, real FD, clock, stock, wait or admission.
Only this module's stdlib math/hashlib/json primitives are used. Custom input
objects, cycles, arbitrary callbacks and authority-looking imports are refused.
"""
import hashlib
import json
import math

TOKEN='SYNTHETIC_MEMORY_ONLY_NOT_AUTHORITY'

class MemoryRefusal(ValueError):
    def __init__(self,code):
        self.code=code
        super().__init__(code)

class MemoryBudget:
    __slots__=('operations','bytes','cumulative_bytes','limit','byte_limit')
    def __init__(self,limit=100000,byte_limit=1048576):
        if type(limit) is not int or not 1<=limit<=100000 or type(byte_limit) is not int or not 1<=byte_limit<=1048576:
            raise MemoryRefusal('MEMORY_LIMIT')
        self.operations=0;self.bytes=0;self.cumulative_bytes=0;self.limit=limit;self.byte_limit=byte_limit
    def step(self,n=1):
        if type(n) is not int or n<0:raise MemoryRefusal('MEMORY_LIMIT')
        self.operations+=n
        if self.operations>self.limit:raise MemoryRefusal('MEMORY_OPERATIONS')
    def byte(self,n,graph=None):
        if type(n) is not int or n<0:raise MemoryRefusal('MEMORY_BYTES')
        self.step();self.cumulative_bytes+=n
        if graph is None:admitted=n
        else:graph[1]+=n;admitted=graph[1]
        if admitted>self.byte_limit:raise MemoryRefusal('MEMORY_BYTES')
        self.bytes=max(self.bytes,admitted) # max single bounded graph, not RSS
        # Cumulative rechecks are diagnostics, not lifetime kernel IO or a
        # physical memory peak. Same immutable strings are reused by cloning.


def copy_graph(value,budget,depth=0,active=None,nodes=None):
    """One bounded plain dict/list/tuple graph clone, no object hooks."""
    if type(budget) is not MemoryBudget:raise MemoryRefusal('MEMORY_BUDGET_TYPE')
    budget.step()
    if nodes is None:nodes=[0,0]
    nodes[0]+=1
    if nodes[0]>8192:raise MemoryRefusal('MEMORY_GRAPH_NODES')
    if depth>24:raise MemoryRefusal('MEMORY_DEPTH')
    kind=type(value)
    if value is None or kind is bool:return value
    if kind is int:
        if value.bit_length()>128:raise MemoryRefusal('MEMORY_INTEGER')
        return value
    if kind is float:return value # deliberately preserves inf/nan for negatives
    if kind in (str,bytes):
        n=len(value)
        if n>65536:raise MemoryRefusal('MEMORY_WIRE')
        budget.byte(n*4 if kind is str else n,nodes)
        return value
    if kind not in (dict,list,tuple):raise MemoryRefusal('MEMORY_OBJECT')
    if len(value)>128:raise MemoryRefusal('MEMORY_CONTAINER')
    if active is None:active=set()
    ident=id(value)
    if ident in active:raise MemoryRefusal('MEMORY_CYCLE')
    active.add(ident)
    try:
        if kind is dict:
            out={}
            for k,v in value.items():
                if type(k) is not str or len(k)>256:raise MemoryRefusal('MEMORY_KEY')
                budget.byte(len(k)*4,nodes)
                out[k]=copy_graph(v,budget,depth+1,active,nodes)
            return out
        out=[copy_graph(v,budget,depth+1,active,nodes) for v in value]
        return tuple(out) if kind is tuple else out
    finally:active.remove(ident)

class MemoryCopy:
    __slots__=('budget',)
    def __init__(self,budget):self.budget=budget
    def deepcopy(self,value):return copy_graph(value,self.budget)

class MemoryMath:
    __slots__=()
    @staticmethod
    def isfinite(value):
        if type(value) not in (int,float):raise MemoryRefusal('MEMORY_NUMBER')
        return math.isfinite(value)

class MemoryNamespace:
    __slots__=('_values',)
    def __init__(self,values):self._values=values
    def __getattr__(self,name):
        if type(name) is not str or name.startswith('__') or name not in self._values:
            raise AttributeError(name)
        return self._values[name]

class MemorySameReadOwner:
    """Dict of pinned BYTES, never OS files or actual owner objects."""
    __slots__=('records','budget','reads')
    def __init__(self,records,budget):
        self.records=copy_graph(records,budget);self.budget=budget;self.reads=0
    def same_read_pin(self,pin,cap):
        self.budget.step();self.reads+=1
        if self.reads>32 or type(cap) is not int or not 1<=cap<=65536:raise MemoryRefusal('MEMORY_WIRE')
        if type(pin) is not dict or set(pin)!= {'kind','key','bytes','sha256'} or pin['kind']!='MEMORY_BYTES_ONLY':
            raise MemoryRefusal('MEMORY_PIN')
        raw=self.records.get(pin['key'])
        if type(raw) is not bytes or len(raw)>cap or type(pin['bytes']) is not int or len(raw)!=pin['bytes']:
            raise MemoryRefusal('MEMORY_PIN')
        if hashlib.sha256(raw).hexdigest()!=pin['sha256']:raise MemoryRefusal('MEMORY_PIN')
        return raw


def memory_wire(value,budget):
    graph=copy_graph(value,budget)
    try:raw=json.dumps(graph,sort_keys=True,separators=(',',':'),allow_nan=False).encode()
    except (ValueError,TypeError):raise MemoryRefusal('MEMORY_JSON')
    if len(raw)>65536:raise MemoryRefusal('MEMORY_WIRE')
    budget.byte(len(raw));return raw


def memory_pin(key,raw):
    if type(key) is not str or not key.startswith('MEMORY-') or type(raw) is not bytes or len(raw)>65536:
        raise MemoryRefusal('MEMORY_PIN')
    return {'kind':'MEMORY_BYTES_ONLY','key':key,'bytes':len(raw),'sha256':hashlib.sha256(raw).hexdigest()}
