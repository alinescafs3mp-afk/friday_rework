"""Source-only bounded entry-key controls. No function is called at import.
Future admitted Root supplies exact reviewed invoke/keys/expected exception.
All custom-object creation occurs outside the entry and is accounted separately.
"""
def entry_keyguard_controls(invoke, keys, scope_error):
    assert type(keys) is tuple and 1 <= len(keys) <= 16
    assert all(type(key) is str for key in keys)
    count = {'hash': 0, 'eq': 0, 'iter': 0, 'len': 0}
    target = keys[0]
    class Collision:
        def __hash__(self):
            count['hash'] += 1
            return hash(target)
        def __eq__(self, other):
            count['eq'] += 1
            raise AssertionError('CUSTOM_EQ_ENTERED')
    class StringKey(str):
        def __hash__(self):
            count['hash'] += 1
            return hash(target)
        def __eq__(self, other):
            count['eq'] += 1
            raise AssertionError('STRING_SUBCLASS_EQ_ENTERED')
    class Dictionary(dict):
        def __len__(self):
            count['len'] += 1
            raise AssertionError('DICT_SUBCLASS_LEN_ENTERED')
        def __iter__(self):
            count['iter'] += 1
            raise AssertionError('DICT_SUBCLASS_ITER_ENTERED')
    rows = []
    def refused(name, mapping):
        before = tuple(count.values())
        try:
            invoke(mapping)
        except (ValueError, AssertionError) as error:
            assert type(error.args) is tuple and error.args == (scope_error,)
        else:
            raise AssertionError('SCOPE_WAS_NOT_REFUSED')
        assert tuple(count.values()) == before
        rows.append({'id': name, 'state': 'PASS_MEMORY_ONLY', 'callbacks_in_entry': 0})
    for name, bad in (('CUSTOM_COLLIDING_KEY', Collision()),
                      ('STR_SUBCLASS_COLLIDING_KEY', StringKey(target)),
                      ('EXACT_INT_KEY', 7), ('EXACT_BOOL_KEY', True)):
        mapping = {bad: b''}
        for key in keys[1:]:
            mapping[key] = b''
        assert len(mapping) == len(keys)
        refused(name, mapping)
    refused('OVERSIZED_BUILTIN_DICT', {str(i): b'' for i in range(128)})
    refused('EMPTY_BUILTIN_DICT', {})
    refused('DICT_SUBCLASS', Dictionary())
    return rows
