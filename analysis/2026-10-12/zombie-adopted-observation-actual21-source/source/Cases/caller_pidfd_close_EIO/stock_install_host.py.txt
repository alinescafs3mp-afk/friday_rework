def require(ok, reason):
    if not ok:
        raise ValueError(reason)


def identity(pid):
    v = Path('/proc/' + str(pid) + '/stat').read_text().rsplit(')', 1)[1].split()
    return {'pid': pid, 'birth': v[19], 'ppid': int(v[1]),
            'pgid': int(v[2]), 'sid': int(v[3])}


def absent(row):
    try:
        return identity(row['pid'])['birth'] != row['birth']
    except FileNotFoundError:
        return True


def read_pin(ref, maximum=1048576):
    require(type(ref) is dict and set(ref) == {'path', 'bytes', 'sha256'}, 'pin shape')
    p = Path(ref['path'])
    require(p.is_absolute() and '..' not in p.parts and str(p) == os.path.normpath(str(p)), 'pin path')
    require(type(ref['bytes']) is int and 0 <= ref['bytes'] <= maximum, 'pin size')
    for parent in reversed(p.parents):
        require(stat.S_ISDIR(parent.lstat().st_mode), 'pin ancestor')
    fd = os.open(p, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK)
    try:
        a = os.fstat(fd)
        require(stat.S_ISREG(a.st_mode) and a.st_nlink == 1 and a.st_uid == os.getuid()
                and not a.st_mode & 0o077 and a.st_size <= maximum, 'private regular pin')
        chunks, size = [], 0
        while True:
            b = os.read(fd, min(8192, maximum + 1 - size))
            if not b:
                break
            chunks.append(b)
            size += len(b)
            require(size <= maximum, 'read bound')
        raw = b''.join(chunks)
        key = lambda s: (s.st_dev, s.st_ino, s.st_size, s.st_mode, s.st_mtime_ns, s.st_ctime_ns)
        require(key(a) == key(os.fstat(fd)) == key(p.lstat()), 'pin drift')
        require(len(raw) == ref['bytes'] and hashlib.sha256(raw).hexdigest() == ref['sha256'], 'pin mismatch')
        return raw
    finally:
        os.close(fd)


def put(path, value):
    raw = (json.dumps(value, sort_keys=True, allow_nan=False) + '\n').encode()
    require(len(raw) <= LIMIT, 'evidence bound')
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o400)
    with os.fdopen(fd, 'wb') as f:
        f.write(raw)
        f.flush()
        os.fsync(f.fileno())


def original(clock):
    fields = ('started_at_unix', 'started_at_monotonic', 'original_seconds',
              'cleanup_seconds', 'useful_deadline_unix', 'final_deadline_unix')
    require(all(type(clock.get(k)) in (int, float) and math.isfinite(clock[k]) for k in fields), 'clock numeric')
    elapsed = time.monotonic() - clock['started_at_monotonic']
    require(3 <= clock['cleanup_seconds'] < clock['original_seconds'] and elapsed >= 0
            and abs(time.time() - clock['started_at_unix'] - elapsed) <= 2
            and clock['boot_id'] == Path('/proc/sys/kernel/random/boot_id').read_text().strip()
            and clock['useful_deadline_unix'] == clock['started_at_unix'] + clock['original_seconds'] - clock['cleanup_seconds']
            and clock['final_deadline_unix'] == clock['started_at_unix'] + clock['original_seconds'], 'original clock consistency')
    return (clock['started_at_monotonic'] + clock['original_seconds'] - clock['cleanup_seconds'],
            clock['started_at_monotonic'] + clock['original_seconds'])



def proc_read(path, maximum=4096):
    # Read proc metadata, never reopen/duplicate a retained custody handle.
    with open(path, 'rb', buffering=0) as f:
        raw = f.read(maximum + 1)
    require(len(raw) <= maximum, 'bounded proc observation')
    return raw.decode('utf-8', 'replace')


def fd_observation(fd, cached=None, pid=None):
    prefix = '/proc/' + ('self' if pid is None else str(pid))
    row = {'fd': fd, 'cached_identity': cached if cached is not None else 'UNKNOWN', 'current_identity': 'UNKNOWN',
           'target': 'UNKNOWN', 'fdinfo': 'UNKNOWN', 'errors': []}
    for key, get in (
            ('current_identity', lambda: os.fstat(fd) if pid is None else os.stat(prefix + '/fd/' + str(fd))),
            ('target', lambda: os.readlink(prefix + '/fd/' + str(fd))),
            ('fdinfo', lambda: proc_read(prefix + '/fdinfo/' + str(fd)))):
        try:
            value = get()
            if key == 'current_identity':
                value = [value.st_dev, value.st_ino, stat.S_IFMT(value.st_mode)]
                if cached is not None:
                    require(value == list(cached), 'retained FD identity changed during observation')
            row[key] = value
        except BaseException as exc:
            row['errors'].append({'field': key, 'class': type(exc).__name__, 'detail': str(exc)[:120]})
    return row


def process_snapshot(pid):
    # Identity and state from ONE bounded stat read; never combine two births.
    values = proc_read('/proc/' + str(pid) + '/stat').rsplit(')', 1)[1].split()
    require(len(values) >= 20, 'complete worker stat snapshot')
    return {'identity': {'pid': pid, 'birth': values[19], 'ppid': int(values[1]),
                         'pgid': int(values[2]), 'sid': int(values[3])}, 'state': values[0]}


def confirm_worker_sample(actual, after):
    require(after['identity'] == actual, 'worker identity drift during observation')


def worker_observation(c):
    row = {'identity': 'UNKNOWN', 'state': 'UNKNOWN', 'wchan': 'UNKNOWN',
           'syscall': 'UNKNOWN', 'lock_fds': [], 'physical_flock': 'UNKNOWN', 'errors': []}
    try:
        birth_path = Path(c['spec']['evidence']) / 'WORKER-BIRTH.json'
        s = birth_path.lstat()
        require(stat.S_ISREG(s.st_mode) and s.st_nlink == 1 and s.st_uid == os.getuid()
                and not s.st_mode & 0o077 and s.st_size <= LIMIT, 'private worker birth observation')
        raw = proc_read(str(birth_path), LIMIT)
        after = birth_path.lstat()
        require((s.st_dev, s.st_ino, s.st_size, s.st_mtime_ns, s.st_ctime_ns) ==
                (after.st_dev, after.st_ino, after.st_size, after.st_mtime_ns, after.st_ctime_ns), 'worker birth observation drift')
        saved = json.loads(raw)
        require(type(saved.get('pid')) is int and type(saved.get('birth')) is str, 'observed worker birth shape')
        row['recorded_birth'] = {'pid': saved['pid'], 'birth': saved['birth']}
    except FileNotFoundError:
        row['state'] = 'BIRTH_NOT_AVAILABLE_AT_THIS_OBSERVATION'
        return row
    except BaseException as exc:
        row['errors'].append({'field': 'recorded_birth', 'class': type(exc).__name__})
        return row
    try:
        sample = process_snapshot(saved['pid'])
        actual = sample['identity']
        if actual['birth'] != saved['birth']:
            row['state'] = 'RECORDED_BIRTH_ABSENT_PID_REUSED'
            return row
        row['identity'] = actual
        require(actual['ppid'] in (os.getpid(), c['proc'].pid) and actual['pgid'] == c['host']['pgid'],
                'observed worker belongs to actual caller or finite subreaper')
        row['state'] = sample['state']
        c['observed_worker'] = {'pid': actual['pid'], 'birth': actual['birth'], 'kind': 'Source_worker'}
    except FileNotFoundError:
        row['state'] = 'RECORDED_BIRTH_FRESH_ABSENT'
        return row
    except BaseException as exc:
        row['errors'].append({'field': 'owned_birth', 'class': type(exc).__name__})
        return row
    row['capacity_release'] = False
    row['direct_wait_credit_from_state'] = False
    row['sample_unix'], row['sample_monotonic'] = time.time(), time.monotonic()
    last = c.get('last_live_worker_sample')
    row['last_live_sample'] = last if last is not None and last['identity']['pid'] == actual['pid'] and last['identity']['birth'] == actual['birth'] else None
    row['last_live_sample_is_historical_not_current_authority'] = True
    if row['state'] == 'Z':
        # A confirmed owned zombie has no executing syscall/open FD sample.
        # The retained host pidfd is still sampled separately, including errors.
        row['current_probe_status'] = 'NOT_APPLICABLE_ZOMBIE_NONRUNNING'
        row['wchan'] = row['syscall'] = 'NOT_APPLICABLE_ZOMBIE_NONRUNNING'
        row['physical_flock'] = 'NOT_APPLICABLE_ZOMBIE_NONRUNNING'
        row['closure'] = 'UNREAPED; ACTUAL_DIRECT_WAIT_AND_FRESH_ABSENCE_REQUIRED'
        try:
            after = process_snapshot(actual['pid'])
            confirm_worker_sample(actual, after)
            require(after['state'] == 'Z', 'zombie state changed during observation')
            row['same_birth_after_sample'] = True
        except BaseException as exc:
            row['same_birth_after_sample'] = 'UNKNOWN'
            row['errors'].append({'field': 'zombie_sample_end_identity', 'class': type(exc).__name__})
        return row  # No remote syscall/wchan/fd iteration on this exact branch.
    row['current_probe_status'] = 'LIVE_SAMPLE_REQUIRED; ALL_REAL_ERRORS_REFUSE'
    for key in ('wchan', 'syscall'):
        try:
            row[key] = proc_read('/proc/' + str(actual['pid']) + '/' + key).strip()
        except BaseException as exc:
            row['errors'].append({'field': key, 'class': type(exc).__name__})
    lock = c['descriptor'].get('lock')
    if c['descriptor'].get('case') != 'worker_flock' or lock is None:
        row['physical_flock'] = 'NOT_REQUESTED'
        return row
    try:
        target = Path(lock).lstat()
        require(stat.S_ISREG(target.st_mode) and target.st_nlink == 1 and target.st_uid == os.getuid(), 'private fixture lock inode')
        lock_key = [target.st_dev, target.st_ino, stat.S_IFMT(target.st_mode)]
        for leaf in sorted(Path('/proc/' + str(actual['pid']) + '/fd').iterdir(), key=lambda p: int(p.name))[:32]:
            fd = int(leaf.name)
            try:
                s = leaf.stat()
                if [s.st_dev, s.st_ino, stat.S_IFMT(s.st_mode)] == lock_key:
                    fd_row = fd_observation(fd, pid=actual['pid'])
                    row['lock_fds'].append(fd_row)
                    row['errors'].extend({'field': 'physical_lock_fd_' + str(fd), **error} for error in fd_row['errors'])
            except FileNotFoundError:
                continue
        # Preserve actual kernel values. Root must judge them with the holder's

        row['physical_flock'] = 'KERNEL_SAMPLE_CAPTURED_ROOT_MUST_INTERPRET_BLOCK' if row['wchan'] not in ('UNKNOWN', '0', '') and row['syscall'] not in ('UNKNOWN', 'running', '') and row['lock_fds'] else 'UNKNOWN'
    except BaseException as exc:
        row['errors'].append({'field': 'physical_lock_fd', 'class': type(exc).__name__})
    try:
        after = process_snapshot(actual['pid'])
        confirm_worker_sample(actual, after)
        require(after['state'] != 'Z', 'live-to-zombie transition during active sample')
        row['same_birth_after_sample'] = True
    except BaseException as exc:
        row['same_birth_after_sample'] = 'UNKNOWN'
        row['errors'].append({'field': 'sample_end_birth', 'class': type(exc).__name__})

            'lock_fds': [{'fd': f['fd'], 'current_identity': f['current_identity']} for f in row['lock_fds']],

    return row


def actor_observation(c):
    # Kernel-discovered outside actor is observed, never signalled or adopted as
    # host ownership. Root separately owns its actual direct wait/fresh closure.
    rows = []
    if c['descriptor'].get('case') != 'worker_flock':
        return rows
    target = Path(c['descriptor']['lock']).lstat()
    require(stat.S_ISREG(target.st_mode) and target.st_nlink == 1 and target.st_uid == os.getuid(), 'fixture lock inode')
    key = [target.st_dev, target.st_ino, stat.S_IFMT(target.st_mode)]
    for line in proc_read('/proc/locks', LIMIT).splitlines():
        parts = line.split()
        if len(parts) < 8 or parts[1:4] != ['FLOCK', 'ADVISORY', 'WRITE']:
            continue  # blocked waiter (->) is not the holder.
        dev = parts[5].split(':')
        if len(dev) != 3 or [int(dev[0], 16), int(dev[1], 16), int(dev[2])] != [os.major(target.st_dev), os.minor(target.st_dev), target.st_ino]:
            continue
        require(len(rows) < 4, 'bounded actual stock lock holders')
        pid = int(parts[4])
        actual = identity(pid)
        fds = []
        for leaf in sorted(Path('/proc/' + str(pid) + '/fd').iterdir(), key=lambda p: int(p.name))[:32]:
            try:
                s = leaf.stat()
                if [s.st_dev, s.st_ino, stat.S_IFMT(s.st_mode)] == key:
                    fds.append(fd_observation(int(leaf.name), pid=pid))
            except FileNotFoundError:
                continue
        command = proc_read('/proc/' + str(pid) + '/cmdline').split('\0')[:-1]
        expected = ['/usr/bin/sleep', '15']
        row = {'identity': actual, 'kernel_holder_row': line, 'lock_identity': key, 'lock_fds': fds,
               'command': command, 'affinity': sorted(os.sched_getaffinity(pid)),
               'limits': proc_read('/proc/' + str(pid) + '/limits'), 'readiness': 'UNKNOWN',
               'owner': 'EXTERNAL_ROOT_STOCK_ACTOR; NOT_HOST_CHILD_OWNERSHIP'}
        row['same_birth_after_sample'] = identity(pid)['birth'] == actual['birth']
        if command == expected and fds and row['same_birth_after_sample'] and any('FLOCK' in f['fdinfo'] and 'lock:' in f['fdinfo'] for f in fds):
            row['readiness'] = 'ACTUAL_KERNEL_FLOCK_HOLDER_AND_OPEN_LOCK_FD_OBSERVED'
        rows.append(row)
    return rows


def _observe_impl(c, phase, include_worker=False):
    # One bounded observation in this existing finite host; no observer process,
    # thread, custody transfer, new grant, FD replacement, close or wait probe.
    c['phase'] = phase
    if c['observation_count'] >= 24:
        c['observation_errors'].append({'phase': phase, 'class': 'ObservationBound'})
        return False
    c['observation_count'] += 1
    errors = []
    frame = {'kind': 'FINITE_HOST_OBSERVATION', 'sequence': c['observation_count'], 'phase': phase,
             'observed_unix': time.time(), 'observed_monotonic': time.monotonic(),
             'host': 'UNKNOWN', 'outer_guard': c['outer'], 'caller': 'UNKNOWN',
             'caller_direct_wait': c['code'] is not None, 'caller_returncode': c['code'],
              'adopted_direct_wait': list(c['adopted']), 'adopted_pending': list(c['adopted_pending']),
              'custody_faults': c['faults'][:8],
             'retained_fds': {}, 'ambiguous_closes': dict(c['ambiguous']),
             'capacity_release': False, 'observations_are_not_authority_or_independent_acceptance': True}
    try:
        require(identity(os.getpid()) == c['host'], 'same actual finite host birth')
        frame['host'] = c['host']
        require(identity(c['outer']['pid']) == c['outer'], 'same actual existing stock guard birth')
        frame['actual_stock_guard_same_birth'] = True
        if c['proc'] is not None:
            frame['caller_popen_pid'] = c['proc'].pid
            try:
                frame['caller'] = identity(c['proc'].pid)
                if c['caller'] is not None:
                    require(frame['caller']['birth'] == c['caller']['birth'], 'caller birth drift during observation')
            except FileNotFoundError:
                frame['caller'] = 'FRESH_PID_ABSENT; direct wait shown separately'
        require(len(c['fds']) <= 24, 'bounded retained host FD set')
        for name, fd in c['fds'].items():
            if fd is not None:
                row = fd_observation(fd, c['ids'].get(name))
                frame['retained_fds'][name] = row
                errors.extend({'fd_label': name, **e} for e in row['errors'])
        if include_worker:
            frame['worker'] = worker_observation(c)
            # Permission/read failure is uncertainty; never claim physical block.
            errors.extend(frame['worker']['errors'])
            frame['stock_actor'] = actor_observation(c)
            for actor in frame['stock_actor']:
                for fd_row in actor['lock_fds']:
                    errors.extend({'field': 'stock_actor_lock_fd', **error} for error in fd_row['errors'])
    except BaseException as exc:
        errors.append({'class': type(exc).__name__, 'detail': str(exc)[:180]})
    frame['errors'] = errors[:16]
    try:
        raw = (json.dumps(frame, sort_keys=True, allow_nan=False) + '\n').encode()
        require(len(raw) <= 16384 and c['observation_bytes'] + len(raw) <= LIMIT, 'whole bounded host observation output')
        c['observation_bytes'] += len(raw)  # Charge before a fallible/partial write.
        offset = 0
        while offset < len(raw):
            n = os.write(1, raw[offset:])  # Existing stock stdout, never reopen it.
            require(n > 0, 'zero observation write')
            offset += n
    except BaseException as exc:
        errors.append({'class': type(exc).__name__, 'detail': str(exc)[:180]})
    if errors:
        c['observation_errors'].extend({'phase': phase, **e} for e in errors[:16])
    return not errors


def observe(c, phase, include_worker=False):
    # All construction/time/proc/write/error bookkeeping is covered here.
    # No observation exception escapes the actual finite host custody.
    try:
        return _observe_impl(c, phase, include_worker)
    except BaseException as exc:
        try:
            c['observation_errors'].append({'phase': phase, 'class': type(exc).__name__})
        except BaseException:
            pass
        return False


def retained_pause(c, phase):
    observe(c, phase, include_worker=True)
    retain_until_stock_guard()  # Actual same host keeps every uncertain handle.


def wait_caller_observed(c, seconds):
    # Sample once while the original caller deadline is live. Popen's direct
    # wait, existing stock guard and final wait remain the physical owners.
    started = time.monotonic()
    first = min(seconds, 0.5)
    try:
        return c['proc'].wait(timeout=first)
    except subprocess.TimeoutExpired:
        if not observe(c, 'caller_wait_live_before_any_stop', include_worker=True):
            retained_pause(c, 'caller_wait_observation_unconfirmed')
        remaining = max(0, min(seconds - (time.monotonic() - started),
                               c['caller_deadline_monotonic'] - time.monotonic(),
                               c['caller_deadline_unix'] - time.time()))
        return c['proc'].wait(timeout=remaining)


def retain_fd(c, label, fd):
    # Assign immediately after open, BEFORE fallible identity capture.
    require(label not in c['fds'], 'occupied retained host FD label')
    c['fds'][label] = fd
    if label.startswith('adopted_pidfd_'):
        observe(c, 'adopted_pidfd_registered_before_fstat', include_worker=True)
    s = os.fstat(fd)
    c['ids'][label] = (s.st_dev, s.st_ino, stat.S_IFMT(s.st_mode))
    return fd


def close_once(c, label):
    fd = c['fds'].get(label)
    if fd is None or label in c['ambiguous']:
        return
    if label in ('stdout', 'stderr', 'caller_pidfd') or label.startswith('adopted_pidfd_'):
        if not observe(c, 'before_close_' + label):
            c['ambiguous'][label] = 'OBSERVATION_UNCONFIRMED'
            return
    try:
        s = os.fstat(fd)
        require((s.st_dev, s.st_ino, stat.S_IFMT(s.st_mode)) == c['ids'].get(label), 'retained host FD drift')
        os.close(fd)
        c['fds'][label] = None
    except BaseException as exc:
        c['ambiguous'][label] = type(exc).__name__
        observe(c, 'ambiguous_close_' + label)
        try:
            os.fstat(fd)
        except OSError as probe:
            if probe.errno == 9:  # EBADF confirms this FD number is absent; never retry close.
                c['fds'][label] = None
                c['ambiguous'].pop(label, None)
        except BaseException:
            pass


def put_retained(path, value, c, label):
    raw = (json.dumps(value, sort_keys=True, allow_nan=False) + '\n').encode()
    require(len(raw) <= LIMIT, 'evidence bound')
    fd = retain_fd(c, label, os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o400))
    try:
        offset = 0
        while offset < len(raw):
            count = os.write(fd, raw[offset:])
            require(count > 0, 'zero evidence write')
            offset += count
        os.fsync(fd)
    finally:
        close_once(c, label)
    require(c['fds'].get(label) is None and label not in c['ambiguous'], 'retained evidence close unconfirmed')


def retain_until_stock_guard():
    # Do not voluntarily drop a live finite owner. Independently admitted stock
    # timeout remains armed. This is no new retry/daemon/authority facility.
    while True:
        try:
            signal.pause()
        except BaseException:
            pass  # Existing armed stock SIGKILL bounds the SAME finite host.


def main():
    require(len(sys.argv) == 4, 'exact input: path bytes SHA256')
    ref = {'path': sys.argv[1], 'bytes': int(sys.argv[2]), 'sha256': sys.argv[3]}
    spec = json.loads(read_pin(ref, LIMIT))
    keys = {'host', 'caller', 'helper', 'entry', 'descriptor', 'clock', 'evidence',
            'fixture_root', 'record', 'wall_cap', 'caller_wall_seconds',
            'host_cleanup_seconds', 'outer_timeout_seconds'}
    require(set(spec) == keys and spec['host']['path'] == SELF, 'exact one-shot input fields')
    for key in ('host', 'caller', 'helper', 'entry', 'descriptor'):
        read_pin(spec[key])
    require(spec['caller']['path'] == str(Path(SELF).with_name('stock_install_caller.py')), 'fixed caller')
    descriptor = json.loads(read_pin(spec['descriptor'], LIMIT))
    clock = json.loads(read_pin(spec['clock'], LIMIT))
    useful, final = original(clock)
    for key in ('wall_cap', 'caller_wall_seconds', 'host_cleanup_seconds', 'outer_timeout_seconds'):
        require(type(spec[key]) in (int, float) and math.isfinite(spec[key]) and spec[key] > 0, 'finite launch caps')
    require(spec['wall_cap'] <= 30 and spec['caller_wall_seconds'] <= spec['wall_cap']
            and spec['host_cleanup_seconds'] <= 5
            and spec['caller_wall_seconds'] + spec['host_cleanup_seconds'] + 2 <= spec['outer_timeout_seconds'] <= 60,
            'shortening-only physical caps')
    require(time.monotonic() + spec['outer_timeout_seconds'] < useful
            and time.time() + spec['outer_timeout_seconds'] < clock['useful_deadline_unix'], 'outer guard fits original useful clock')
    for kind, cap in ((resource.RLIMIT_AS, AS), (resource.RLIMIT_CPU, 60), (resource.RLIMIT_FSIZE, MAX)):
        require(resource.getrlimit(kind) == (cap, cap), 'hard host resources before caller')
    require(resource.getrlimit(resource.RLIMIT_CORE) == (0, 0)
            and 0 < len(os.sched_getaffinity(0)) <= 2
            and signal.getsignal(signal.SIGCHLD) == signal.SIG_DFL
            and len(list(Path('/proc/self/task').iterdir())) == 1, 'host custody boundary')
    children = Path('/proc/self/task/' + str(os.getpid()) + '/children')
    require(not children.read_text().strip(), 'finite host starts without children')
    host = identity(os.getpid())
    outer = identity(os.getppid())
    expected_outer = [b'/usr/bin/timeout', b'-s', b'KILL',
                      (str(spec['outer_timeout_seconds']) + 's').encode(),
                      PYTHON.encode(), b'-I', b'-B', SELF.encode(),
                      ref['path'].encode(), str(ref['bytes']).encode(), ref['sha256'].encode()]
    require(Path('/proc/' + str(outer['pid']) + '/cmdline').read_bytes().split(b'\0')[:-1] == expected_outer,
            'actual stock outer invocation differs')
    require(host['pgid'] == outer['pgid'], 'whole cohort outer guard')
    libc = ctypes.CDLL(None, use_errno=True)
    require(libc.prctl(36, 1, 0, 0, 0) == 0, 'finite host subreaper')
    require(libc.prctl(1, signal.SIGKILL, 0, 0, 0) == 0 and os.getppid() == outer['pid'], 'host parent death')
    record = Path(spec['record'])
    e = Path(spec['evidence'])
    require(record.parent == e.parent and record != e and record.is_absolute()
            and '..' not in record.parts and str(record) == os.path.normpath(str(record)), 'exact private records')
    for parent in reversed(record.parents):
        require(stat.S_ISDIR(parent.lstat().st_mode), 'record ancestor')
    require(record.parent.stat().st_uid == os.getuid() and not record.parent.stat().st_mode & 0o077, 'private record parent')
    require(not e.exists(), 'no attempt replay')
    record.mkdir(mode=0o700)
    deadline_mono = min(useful, time.monotonic() + spec['caller_wall_seconds'])
    deadline_unix = min(clock['useful_deadline_unix'], time.time() + spec['caller_wall_seconds'])
    launch = {'input': ref, 'host': host, 'outer_guard': outer,
              'deadline_monotonic': deadline_mono, 'deadline_unix': deadline_unix}
    put(record / 'LAUNCH.json', launch)
    put(record / 'HOST-BIRTH.json', {'host': host, 'outer_guard': outer,
        'limits': Path('/proc/self/limits').read_text(), 'affinity': sorted(os.sched_getaffinity(0)),
        'Source_caller_worker_hard_AS_sum': 2 * AS, 'whole_timeout_host_caller_worker_hard_AS_sum': 4 * AS,
        'original_clock': clock, 'unknown_initial_metadata_RSS': 'UNKNOWN',
        'authority': 'Root actual independently admitted exact external invocation; this file is consistency only'})
    # Local state of the already admitted finite host, established BEFORE Popen.
    # Not a new issuer/protocol: exact external Root invocation is still required.
    c = {'fds': {}, 'ids': {}, 'ambiguous': {}, 'proc': None, 'caller': None, 'fork_attempted': False,
         'code': None, 'known': [], 'adopted': [], 'faults': [], 'timed_out': False,
         'host': host, 'outer': outer, 'spec': spec, 'descriptor': descriptor, 'phase': 'OWNED_BEFORE_POPEN',
         'caller_deadline_monotonic': deadline_mono, 'caller_deadline_unix': deadline_unix,
         'observed_worker': None, 'last_live_worker_sample': None, 'adopted_pending': [],
          'observation_count': 0, 'observation_bytes': 0, 'observation_errors': []}
    def fault(phase, exc):
        if len(c['faults']) < 32:
            c['faults'].append({'phase': phase, 'class': type(exc).__name__, 'detail': str(exc)[:300]})
    def wait_seconds(cleanup=False):
        if cleanup:
            return max(0, min(spec['host_cleanup_seconds'], final - time.monotonic(), clock['final_deadline_unix'] - time.time()))
        return max(0, min(deadline_mono - time.monotonic(), deadline_unix - time.time()))
    try:
        outfd = retain_fd(c, 'stdout', os.open(record / 'stdout.log', os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o400))
        errfd = retain_fd(c, 'stderr', os.open(record / 'stderr.log', os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o400))
        c['fork_attempted'] = True
        c['proc'] = subprocess.Popen([PYTHON, '-I', '-B', spec['caller']['path'], str(record / 'LAUNCH.json')],
                                    stdout=outfd, stderr=errfd, close_fds=True, start_new_session=False)
        proc = c['proc']  # Retain direct-child object/PID before any birth/fstat/write.
        pfd = retain_fd(c, 'caller_pidfd', os.pidfd_open(proc.pid))
        caller = identity(proc.pid)
        c['caller'] = caller
        c['known'].append(caller)
        require(caller['ppid'] == os.getpid() and caller['pgid'] == host['pgid'], 'actual owned caller')
        put_retained(record / 'CALLER-BIRTH.json', caller, c, 'caller_birth_leaf')
        if not observe(c, 'caller_owned_after_birth_before_direct_wait'):
            retained_pause(c, 'initial_caller_observation_unconfirmed')
        try:
            c['code'] = wait_caller_observed(c, wait_seconds())
        except subprocess.TimeoutExpired:
            c['timed_out'] = True
            if not observe(c, 'caller_timeout_before_pidfd_stop', include_worker=True):
                retained_pause(c, 'caller_timeout_observation_unconfirmed')
            signal.pidfd_send_signal(pfd, signal.SIGKILL)
            c['code'] = proc.wait(timeout=wait_seconds(True))
    except BaseException as exc:
        fault('caller_scope', exc)
        if c['proc'] is not None and c['code'] is None and c['fds'].get('caller_pidfd') is not None:
            try:
                fd = c['fds']['caller_pidfd']
                s = os.fstat(fd)
                require((s.st_dev, s.st_ino, stat.S_IFMT(s.st_mode)) == c['ids'].get('caller_pidfd'), 'retained caller pidfd')
                if not observe(c, 'caller_recovery_before_pidfd_stop', include_worker=True):
                    retained_pause(c, 'caller_recovery_observation_unconfirmed')
                try:
                    signal.pidfd_send_signal(fd, signal.SIGKILL)
                except ProcessLookupError:
                    pass
                c['code'] = c['proc'].wait(timeout=wait_seconds(True))
            except BaseException as stop:
                fault('caller_stop_direct_wait', stop)
    # No finally closes actual custody while a direct wait/fresh absence is unknown.
    whole_closed = False
    closure = []
    try:
        cleanup_until = min(time.monotonic() + spec['host_cleanup_seconds'], final)
        while c['code'] is not None:
            values = children.read_text().split()
            if not values:
                break
            require(time.monotonic() < cleanup_until and time.time() < clock['final_deadline_unix'], 'STOP_UNCONFIRMED adopted child')
            for value in values:
                row = identity(int(value))
                require(row['ppid'] == os.getpid() and row['pgid'] == host['pgid'], 'owned adopted identity')
                if row not in c['known']:
                    c['known'].append(row)
                if not any(p['pid'] == row['pid'] and p['birth'] == row['birth'] for p in c['adopted_pending']):
                    c['adopted_pending'].append({'pid': row['pid'], 'birth': row['birth']})
                label = 'adopted_pidfd_' + str(row['pid']) + '_' + row['birth']
                if label not in c['fds']:
                    retain_fd(c, label, os.pidfd_open(row['pid']))
                fd = c['fds'][label]
                if not observe(c, 'owned_adopted_before_pidfd_stop_and_wait', include_worker=True):
                    retained_pause(c, 'adopted_observation_unconfirmed')
                s = os.fstat(fd)
                require((s.st_dev, s.st_ino, stat.S_IFMT(s.st_mode)) == c['ids'].get(label), 'retained adopted pidfd')
                try:
                    signal.pidfd_send_signal(fd, signal.SIGKILL)
                except ProcessLookupError:
                    pass
                if not observe(c, 'before_adopted_direct_wait', include_worker=True):
                    retained_pause(c, 'adopted_pre_wait_observation_unconfirmed')
                try:
                    pid, status = os.waitpid(row['pid'], os.WNOHANG)
                    if pid:
                        require(pid == row['pid'], 'exact adopted direct wait')
                        require(absent(row), 'adopted direct wait without fresh absence')
                        c['adopted'].append({'pid': pid, 'birth': row['birth'], 'direct_wait': True, 'returncode': os.waitstatus_to_exitcode(status)})
                except BaseException as exc:
                    fault('adopted_direct_wait', exc)
                    retained_pause(c, 'adopted_direct_wait_unconfirmed')
                    # Keep all pidfds until whole cohort closure is established.
            select.select([], [], [], 0.01)
        worker_file = e / 'WORKER-BIRTH.json'
        if worker_file.exists():
            s = worker_file.lstat()
            require(stat.S_ISREG(s.st_mode) and s.st_nlink == 1 and s.st_uid == os.getuid() and not s.st_mode & 0o077 and s.st_size <= LIMIT, 'worker birth evidence inode')
            worker = json.loads(worker_file.read_bytes())
            require(type(worker['pid']) is int and type(worker['birth']) is str, 'worker birth evidence shape')
            if not any(row['pid'] == worker['pid'] and row['birth'] == worker['birth'] for row in c['known']):
                c['known'].append({'pid': worker['pid'], 'birth': worker['birth'], 'kind': 'Source_worker'})
        closure = [dict(row, fresh_birth_absent=absent(row)) for row in c['known']]
        adopted_waits_confirmed = all(any(a['pid'] == p['pid'] and a['birth'] == p['birth'] and a['direct_wait'] is True
                                           for a in c['adopted']) for p in c['adopted_pending'])
        whole_closed = c['code'] is not None and c['caller'] is not None and adopted_waits_confirmed and not children.read_text().strip() and all(r['fresh_birth_absent'] for r in closure)
        # Missing Popen handle after an attempted spawn is never no-execution credit.
        if c['fork_attempted'] and c['proc'] is None:
            whole_closed = False
    except BaseException as exc:
        fault('whole_cohort_closure', exc)
        whole_closed = False
    if c['observed_worker'] is not None and not any(r['pid'] == c['observed_worker']['pid'] and r['birth'] == c['observed_worker']['birth'] for r in c['known']):
        c['known'].append(c['observed_worker'])
        closure.append(dict(c['observed_worker'], fresh_birth_absent='UNKNOWN; additional actual observed birth'))
        whole_closed = False  # Additional known birth requires fresh actual closure, never omit it.
    result = {'caller_direct_wait': c['code'] is not None, 'caller_returncode': c['code'],
              'caller_timed_out': c['timed_out'], 'known_fresh_closure': closure,
              'adopted_direct_wait': c['adopted'], 'whole_owned_directchild_cohort_closed': whole_closed,
              'fault': c['faults'] or None, 'capacity_release': False,
              'retained_resources_before_release': {'fds': {n:fd for n,fd in c['fds'].items() if fd is not None},
                                                    'identities': c['ids'], 'ambiguous_closes': c['ambiguous']},
              'attempt_outcome': 'UNKNOWN_NO_REPLAY' if c['faults'] or c['code'] != 0 or not whole_closed or c['observation_errors'] else 'CALLER_RETURN_OBSERVED_NOT_INSTALL_ACCEPTANCE',
              'scope_release': 'Root decides only after actual host direct return and fresh closure',
              'installed_admission': 'NOT_CLAIMED', 'original_clock': clock,
              'observation_errors': c['observation_errors'][:32],
              'observation_output': 'bounded JSON lines in existing stock stdout; not independent acceptance'}
    if not observe(c, 'before_outside_closure_evidence'):
        retained_pause(c, 'outside_observation_unconfirmed')
    try:
        put_retained(record / 'OUTSIDE-CLOSURE.json', result, c, 'outside_closure_leaf')
    except BaseException as exc:
        fault('outside_closure_evidence', exc)
        retained_pause(c, 'outside_closure_evidence_unconfirmed')
    if not whole_closed:
        retained_pause(c, 'whole_cohort_STOP_UNCONFIRMED')  # No handle released.
    if c['observation_errors']:
        retained_pause(c, 'observations_unconfirmed_before_any_release')
    # Whole actual child cohort closed first. Each close is attempted at most once.
    for label in list(c['fds']):
        if 'pidfd' not in label:
            close_once(c, label)
    if c['ambiguous']:
        retained_pause(c, 'non_pidfd_final_close_ambiguous')  # Keep pidfds and ambiguous FD.
    for label in list(c['fds']):
        if 'pidfd' in label:
            close_once(c, label)
    if c['ambiguous'] or any(fd is not None for fd in c['fds'].values()):
        retained_pause(c, 'pidfd_final_close_ambiguous')
    try:
        require(not children.read_text().strip() and all(absent(row) for row in c['known']), 'fresh whole closure before host exit')
        if not observe(c, 'fresh_children_closed_before_host_return'):
            retained_pause(c, 'final_host_observation_unconfirmed')
        print(json.dumps(result, sort_keys=True))
    except BaseException as exc:
        fault('final_host_evidence', exc)
        retained_pause(c, 'final_host_evidence_unconfirmed')
    return 0 if c['code'] == 0 and not c['faults'] else 2


_real_close = os.close
def _fault_close(fd):
    try: target=os.readlink('/proc/self/fd/'+str(fd))
    except OSError: target=''
    if target == 'anon_inode:[pidfd]':
        raise OSError(5,'exact pidfd close EIO before release')
    return _real_close(fd)
os.close=_fault_close
