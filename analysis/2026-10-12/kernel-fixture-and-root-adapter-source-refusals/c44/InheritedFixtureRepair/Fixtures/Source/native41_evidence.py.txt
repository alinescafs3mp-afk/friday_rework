"""Exact same-read consumer for Root's existing collector, never authority.
Pure custody validation, no runner/controller/receipt/new owner or retry.
Only after independent Source admission may Root supply already-retained FDs.
"""
import hashlib
import json
import os
from pathlib import Path
import stat
import zlib

def same_read(fd,pin,maximum=65536):
    before=os.fstat(fd);p=Path(pin['path']);a=p.lstat()
    key=lambda s:(s.st_dev,s.st_ino,s.st_mode,s.st_size,s.st_mtime_ns,s.st_ctime_ns)
    assert stat.S_ISREG(before.st_mode) and before.st_nlink==1
    assert 0<before.st_size==pin['bytes']<=maximum and key(before)==key(a)
    raw=os.pread(fd,before.st_size+1,0);after=os.fstat(fd)
    assert len(raw)==pin['bytes'] and key(before)==key(after)==key(p.lstat())
    assert hashlib.sha256(raw).hexdigest()==pin['sha256']
    return raw

def caller_evidence(index_fd,index_pin,segment_fds):
    index=json.loads(same_read(index_fd,index_pin))
    assert index['schema']=='friday.causal.lossless-index.v1'
    refs=index['segments'];assert 1<=len(refs)<=18
    assert len(segment_fds)==len(refs) and len({r['path'] for r in refs})==len(refs)
    chunks=[];offset=0
    for ordinal,(fd,pin) in enumerate(zip(segment_fds,refs)):
        leaf=json.loads(same_read(fd,pin))
        assert set(leaf)=={'schema','case','index','offset','compressed_hex'}
        assert leaf['schema']=='friday.causal.lossless-segment.v1'
        assert leaf['case']==index['case'] and leaf['index']==ordinal and leaf['offset']==offset
        chunk=bytes.fromhex(leaf['compressed_hex']);assert 0<len(chunk)<=24576
        if ordinal<len(refs)-1:assert len(chunk)==24576
        chunks.append(chunk);offset+=len(chunk)
    compressed=b''.join(chunks)
    assert 0<len(compressed)==index['compressed_bytes']<=430080
    assert hashlib.sha256(compressed).hexdigest()==index['compressed_sha256']
    decoder=zlib.decompressobj()
    raw=decoder.decompress(compressed,16777217)
    assert len(raw)<=16777216 and decoder.eof and not decoder.unconsumed_tail and not decoder.unused_data
    assert len(raw)==index['raw_bytes'] and hashlib.sha256(raw).hexdigest()==index['raw_sha256']
    assert raw.endswith(b'\n')
    events=[];owned={};births={};birth_order=[]
    for ordinal,line in enumerate(raw.splitlines()):
        event=json.loads(line);assert event['case']==index['case'] and event['index']==ordinal
        if 'owned_delta' in event:
            for name in event.pop('owned_delta_deleted'):assert name in owned;del owned[name]
            owned.update(event.pop('owned_delta'))
            for identity in event.pop('known_birth_deleted'):assert identity in births;del births[identity]
            births.update(event.pop('known_birth_delta'))
            if 'known_birth_order' in event:birth_order=event.pop('known_birth_order')
            assert len(birth_order)==len(set(birth_order))==len(births)<=128 and set(birth_order)==set(births)
            owned['known']=[births[key] for key in birth_order]
            event['owned']=json.loads(json.dumps(owned))
        events.append(event)
    assert len(events)==index['event_count'] and len(events)>0
    assert events[0]['operation']=='fixture_attach'
    assert index['capacity_release'] is False
    assert index['root_outside_wait']=='REQUIRED_NOT_OBSERVED_INSIDE'
    assert index['observation_failed'] is False and index['failure_count']==0
    return {'index':index,'events':events,'inside_closure_is_not_outside_acceptance':True}

def failure_evidence(fd,pin):
    failure=json.loads(same_read(fd,pin))
    assert failure['schema']=='friday.causal.reserved-failure.v1'
    assert failure['normal_evidence_complete'] is False and failure['capacity_release'] is False
    assert failure['actual_primary_exception']['class']
    assert failure['kernel'] is not None and failure['owned'] is not None
    assert len(failure['owned']['known'])<=128
    return failure # Actual cause/custody only, never successful closure/credit.
