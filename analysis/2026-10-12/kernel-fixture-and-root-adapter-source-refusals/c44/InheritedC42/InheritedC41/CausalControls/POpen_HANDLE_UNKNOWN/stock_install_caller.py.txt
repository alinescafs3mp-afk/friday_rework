def require(ok, reason):

def identity(pid):
    v = Path('/proc/' + str(pid) + '/stat').read_text().rsplit(')', 1)[1].split()
    return {'pid': pid, 'birth': v[19], 'ppid': int(v[1]), 'pgid': int(v[2]), 'sid': int(v[3])}


def read_pin(ref, maximum=1048576):
    require(type(ref) is dict and set(ref) == {'path', 'bytes', 'sha256'}, 'pin shape')
    p = Path(ref['path'])
    require(p.is_absolute() and '..' not in p.parts and str(p) == os.path.normpath(str(p)), 'pin path')
    require(type(ref['bytes']) is int and 0 <= ref['bytes'] <= maximum, 'pin size')
    for parent in reversed(p.parents):
        require(stat.S_ISDIR(parent.lstat().st_mode), 'pin ancestor')
    fd = os.open(p, os.O_RDONLY | os.O_NONBLOCK | os.O_NOFOLLOW)
    try:
        a = os.fstat(fd)
        require(stat.S_ISREG(a.st_mode) and a.st_nlink == 1 and a.st_uid == os.getuid()
                and not a.st_mode & 0o077 and a.st_size <= maximum, 'private regular pin')
        chunks, size = [], 0
        while True:
            b = os.read(fd, min(8192, maximum + 1 - size))
            if not b:
                break
            chunks.append(b)
            size += len(b)
            require(size <= maximum, 'pin read bound')
        raw = b''.join(chunks)
        key = lambda s: (s.st_dev, s.st_ino, s.st_size, s.st_mode, s.st_mtime_ns, s.st_ctime_ns)
        require(key(a) == key(os.fstat(fd)) == key(p.lstat()), 'pin drift')
        require(len(raw) == ref['bytes'] and hashlib.sha256(raw).hexdigest() == ref['sha256'], 'pin mismatch')
        return raw
    finally:
        os.close(fd)


def put(path, value):
    raw = (json.dumps(value, sort_keys=True, allow_nan=False) + '\n').encode()
    require(len(raw) <= LIMIT, 'evidence bound')
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o400)
    with os.fdopen(fd, 'wb') as f:
        f.write(raw)
        f.flush()
        os.fsync(f.fileno())


def caller_limits():
    for kind, cap in ((resource.RLIMIT_AS, AS), (resource.RLIMIT_CPU, 60), (resource.RLIMIT_FSIZE, MAX)):
        require(resource.getrlimit(kind) == (cap, cap), 'hard caller resources before Source')
    require(resource.getrlimit(resource.RLIMIT_CORE) == (0, 0)
            and 0 < len(os.sched_getaffinity(0)) <= 2, 'caller core/affinity')
    return {'pid': os.getpid(), 'birth': identity(os.getpid())['birth'],
            'AS': [AS, AS], 'CPU': [60, 60], 'FSIZE': [MAX, MAX], 'CORE': [0, 0],
            'affinity': sorted(os.sched_getaffinity(0))}

    def _observe(self, phase):
        # Only this actual caller and the SAME already held helper custody.
        # Existing host stdout.log is the bounded output; no new channel/issuer.
        try:
            c = self.custody
            frame = {'kind': 'FINITE_CALLER_OBSERVATION', 'phase': phase,
                     'observed_unix': time.time(), 'observed_monotonic': time.monotonic(),
                     'caller': identity(os.getpid()), 'actual_host': identity(os.getppid()),
                     'capacity_release': False, 'custody': 'NOT_HELD_YET', 'errors': []}
            require(frame['caller']['birth'] == self.caller['birth'] and frame['actual_host'] == self.launch['host'],
                    'same actual caller and host births in observation')
            if c is not None:
                frame['custody'] = {'pid': c.pid, 'birth': c.ticks, 'waited': c.waited,
                                    'state': c.state, 'retained_fds': {}, 'ambiguous': dict(c.fd_errors),
                                    'direct_wait_observed': c.waited, 'fresh_birth_absent': c.child_absent,
                                    'process_state_is_never_reap_or_release_authority': True}
                require(len(c.fds) <= 24, 'bounded caller observation handles')
                for name, fd in c.fds.items():
                    if fd is None:
                        continue
                    row = {'fd': fd, 'cached_identity': c.fd_ids.get(name), 'current_identity': 'UNKNOWN',
                           'target': 'UNKNOWN', 'fdinfo': 'UNKNOWN'}
                    for key in ('current_identity', 'target', 'fdinfo'):
                        try:
                            if key == 'current_identity':
                                s = os.fstat(fd)
                                row[key] = [s.st_dev, s.st_ino, stat.S_IFMT(s.st_mode)]
                                if c.fd_ids.get(name) is not None:
                                    require(row[key] == list(c.fd_ids[name]), 'retained caller FD identity drift')
                            elif key == 'target':
                                row[key] = os.readlink('/proc/self/fd/' + str(fd))
                            else:
                                with open('/proc/self/fdinfo/' + str(fd), 'rb', buffering=0) as f:
                                    raw = f.read(4097)
                                require(len(raw) <= 4096, 'bounded caller fdinfo')
                                row[key] = raw.decode('utf-8', 'replace')
                        except BaseException as exc:
                            frame['errors'].append({'label': name, 'field': key, 'class': type(exc).__name__})
                    frame['custody']['retained_fds'][name] = row
            raw = (json.dumps(frame, sort_keys=True, allow_nan=False) + '\n').encode()
            require(len(raw) <= 16384 and self.observation_bytes + len(raw) <= LIMIT, 'whole caller observation output')
            self.observation_bytes += len(raw)
            offset = 0
            while offset < len(raw):
                n = os.write(1, raw[offset:])
                require(n > 0, 'zero caller observation write')
                offset += n
            if frame['errors']:
                self.observation_errors.extend(frame['errors'][:16])
                return False
            return True
        except BaseException as exc:
            self.observation_errors.append({'phase': phase, 'class': type(exc).__name__})
            return False

    def _observe_or_retain(self, phase):

            confirmed = self._observe(phase)
        except BaseException:
            confirmed = False
        if not confirmed:
            # No raw observation exception drops the SAME caller/custody. Actual
            # already admitted host deadline/parent-death/stock guard remain live.
            while True:
                try:
                    signal.pause()
                except BaseException:
                    pass  # Existing host/stock stop bounds the SAME caller.

        read_pin(self.spec['host'])

                and (c.pid is None or (c.waited and c.child_absent)), 'STOP_UNCONFIRMED retained resources')

                'caller_closure': 'outside stock Popen direct wait still required'}

    def native_scan(self,reap=True):

        # is visible here. Every task was already reserved by pids/memory/CPU.

        require(len(pids)<=127,'finite native processes')
        for pid in pids:
            try:
                raw=Path('/proc/'+str(pid)+'/stat').read_text(); v=raw.rsplit(')',1)[1].split()
            except FileNotFoundError:
                # Vanishing before registration is unknown, never zero/reap.
                raise RuntimeError('STOP_UNCONFIRMED unregistered disappearing native birth')
            row={'pid':pid,'birth':v[19],'state':v[0],'ppid':int(v[1]),'namespace':self.namespace}
            if row['state']!='Z':
                require(os.stat('/proc/'+str(pid)+'/ns/pid').st_ino==self.namespace,'descendant namespace escape')
                require(Path('/proc/'+str(pid)+'/cgroup').read_text().strip()=='0::'+self.spec['native']['group_relative'],'descendant scope escape')
            else:row['active_probes']='NOT_APPLICABLE_ZOMBIE_UNREAPED_NOT_CLOSURE'

        return pids

def native_sandbox(temp_root):

    import platform

    require(libc.syscall(446,rules_fd,0)==0,'actual inherited Landlock restriction')
    os.close(root_fd);os.close(rules_fd)
    class Filter(ctypes.Structure):_fields_=[('code',ctypes.c_ushort),('jt',ctypes.c_ubyte),('jf',ctypes.c_ubyte),('k',ctypes.c_uint)]
    class Prog(ctypes.Structure):_fields_=[('len',ctypes.c_ushort),('filter',ctypes.POINTER(Filter))]
    code=[(0x20,0,0,4),(0x15,1,0,0xc000003e),(0x06,0,0,0x80000000),(0x20,0,0,0)]

    arr=(Filter*len(code))(*(Filter(*x) for x in code));prog=Prog(len(code),arr)
    require(libc.prctl(22,2,ctypes.byref(prog),0,0)==0,'actual inherited namespace/network escape filter')

def read_small(path,cap=65536):
    fd=os.open(path,os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK)
    try:
        raw=os.read(fd,cap+1)
        if len(raw)>cap:raise RuntimeError('finite fixture read exhausted')
        return raw.decode()
    finally:os.close(fd)

def _private_mutation(owner,fixture):
    # Root creates a DISPOSABLE original-byte-exact private copy before pinning.
    # Never edit C39, installed modules/resources, account files or shared data.
    path=Path(fixture['disposable_capture_path'])
    assert path.is_relative_to(owner.temp_root) and path.resolve(strict=True).is_relative_to(owner.temp_root)
    before=path.lstat();assert stat.S_ISREG(before.st_mode) and before.st_nlink==1
    assert before.st_mode&0o022==0 and before.st_mode&0o200
    original=owner.capture_raw[str(path)]
    assert len(original)==before.st_size and 0<len(original)<=16777216
    raw=bytes([original[0]^1])+original[1:]
    label='c40_disposable_mutation'
    fd=owner.native_track(label,os.open(path,os.O_WRONLY|os.O_NOFOLLOW))
    pos=0
    while pos<len(raw):n=os.write(fd,raw[pos:]);assert n>0;pos+=n
    os.fsync(fd);owner.native_close_fd(label)
    after=path.lstat()
    assert before.st_mtime_ns!=after.st_mtime_ns or before.st_ctime_ns!=after.st_ctime_ns
    assert owner.capture_raw[str(path)]==original
    return path,original,raw

def dispatch(owner,fixture,observe):
    case_id=fixture['id'];owner.native_check()
    if case_id=='CAPTURE_UNKNOWN_IMPORT_NO_DISK_FALLBACK':
        observe.capture_unknown(owner);return
    if case_id=='CAPTURE_PACKAGE_RESOURCE_NO_REOPEN':
        path=Path(fixture['disposable_capture_path'])
        original=owner.capture_raw[str(path)]
        # Actual Root/host changes only its disposable private Source copy
        # AFTER capture, before GO. Caller Landlock writes remain temp-only.
        fd=owner.native_track('c40_resource_after_capture_read',os.open(path,os.O_RDONLY|os.O_NOFOLLOW))
        changed=os.pread(fd,len(original)+1,0);owner.native_close_fd('c40_resource_after_capture_read')
        assert len(changed)==len(original) and changed!=original
        observe.captured_resource(owner,str(path));return
    if case_id=='CAPTURE_CYCLIC_CANONICAL_NAMESPACES':
        # Root binds ACTUAL genuine modules and their canonical observed
        # relationships. No fixture mini-library replaces genuine dependencies.
        modules=fixture['genuine_cyclic_fromlist_dataclass_modules'];assert len(modules)>=3
        results=[]
        for name in modules:
            module=owner.native_module(name)
            assert module is sys.modules[name] and module.__spec__.name==name
            assert module.__spec__.loader is owner or name in owner.capture_extensions
            results.append({'name':name,'module_id':id(module),'spec_name':module.__spec__.name})
        owner._c40_add({'operation':'canonical_genuine_imports','phase':'actual','modules':results})
        return
    modes=fixture['real_workload_modes']
    for mode in modes:
        result=observe.real_workload(owner,fixture['python_path'],fixture['payload_path'],mode)
        # EOF positivity requires actual raw count and exact stdout/stderr
        # producer pins; CompletedProcess text decoding alone is insufficient.
        if mode=='positive_eof':

            assert result.returncode==0
    if modes:return
    # Remaining controls operate at actual existing host/stock or exact Source
    # fault sites. A caller-only invocation must not fabricate their outcomes.
    raise RuntimeError('case requires its separately Root-bound existing host/stock causal site: '+case_id)

def native41_bootstrap_probe(group_path,group_relative,context):
    # Real fork plus real thread BEFORE caller GO or DURING host captured import.
    # Same private namespace/group, no new worker/controller/protocol service.
    assert context in ('first_caller_before_GO','host_captured_import')
    group=Path(group_path)
    expected={'memory.max':'2147483648','memory.swap.max':'0','pids.max':'128','cpu.max':'200000 100000'}
    def sample():
        assert Path('/proc/self/cgroup').read_text().strip()=='0::'+group_relative
        actual={key:(group/key).read_text().strip() for key in expected};assert actual==expected
        denied=None
        try:
            fd=os.open(group/'memory.max',os.O_WRONLY|os.O_NOFOLLOW)
        except OSError as error:
            denied=error.errno
        else:
            os.close(fd)
            raise RuntimeError('C41 controller write open unexpectedly permitted; NO write/retry')
        assert denied in (13,30) # Actual Landlock/readonly kernel refusal, no fake predicate.
        fields=Path('/proc/self/stat').read_text().rsplit(')',1)[1].split()
        assert resource.getrlimit(resource.RLIMIT_AS)==(1073741824,1073741824)
        return {'pid':os.getpid(),'tid':threading.get_native_id(),'birth':fields[19],
                'group':group_relative,'controllers':actual,'context':context,'actual_controller_write_denial_errno':denied}
    thread_rows=[];errors=[]
    def target():
        try:thread_rows.append(sample())
        except BaseException as error:errors.append(type(error).__name__)
    thread=threading.Thread(target=target);thread.start();thread.join(timeout=5)
    assert not thread.is_alive() and len(thread_rows)==1 and not errors
    read_fd,write_fd=os.pipe();pid=os.fork()
    if pid==0:
        try:
            os.close(read_fd);raw=json.dumps(sample(),sort_keys=True).encode();assert len(raw)<=4096
            assert os.write(write_fd,raw)==len(raw);os.close(write_fd);os._exit(0)
        except BaseException:os._exit(125)
    # Same finite probe retains exact child handle before pipe/sample/wait.
    retained={'pid':pid,'pidfd':None,'waited':False,'closed':False}
    retained['pidfd']=os.pidfd_open(pid)
    before=os.fstat(retained['pidfd'])
    os.close(write_fd);raw=os.read(read_fd,4097);assert 0<len(raw)<=4096
    os.close(read_fd);waited,status=os.waitpid(pid,0)
    assert waited==pid and status==0 and not Path('/proc/'+str(pid)).exists()
    retained['waited']=True
    after=os.fstat(retained['pidfd']);assert (before.st_dev,before.st_ino)==(after.st_dev,after.st_ino)
    os.close(retained['pidfd']);retained['closed']=True
    child=json.loads(raw);assert child['pid']==pid and child['controllers']==expected
    return {'thread':thread_rows[0],'child':child,'direct_wait':True,'fresh_child_absent':True,
            'real_fork_and_thread':True,'actual_inherited_before_GO':True,'not_controller_migration':True}
