def test_cached_declared_six_expected_origins(actual):
    _real(actual);apps=actual['six_declared_application_names'];assert len(set(apps))==6
    rows=actual['cached_application_cases'];assert len(rows)==18
    assert {(r['name'],r['variant']) for r in rows}=={(n,v) for n in apps for v in ('canonical','alias','stale_reference')}
    for r in rows:
        assert r['actual_expected_canonical_name_origin'] is True
        assert r['actual_code_differs_from_captured_bytes'] is True
        assert r['actual_refusal_reason'] in ('cached declared application refused before captured execution','stale bootstrap application module reference')
        assert r['actual_captured_application_exec_count']==0 and r['actual_genuine_receipt_writes']==0
        assert r['intended_cause_observed'] is True and r['unrelated_error_as_pass'] is False
