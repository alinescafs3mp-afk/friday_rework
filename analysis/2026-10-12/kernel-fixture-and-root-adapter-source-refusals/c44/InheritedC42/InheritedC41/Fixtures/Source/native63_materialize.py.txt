def final_bytes(final_sources,name,binding):
    assert name in final_sources and name in binding['final_source_pins']
    raw=final_sources[name];ref=binding['final_source_pins'][name]
    assert type(raw)is bytes and type(ref['bytes'])is int and len(raw)==ref['bytes']<=2097152
    assert hashlib.sha256(raw).hexdigest()==ref['sha256']
    return raw


def materialize(case_id,final_sources,descriptor,spec,binding,recipes,lane):
    assert case_id in CASE_IDS and binding['case_id']==case_id
    assert binding['independent_source_acceptance'] is not None
    assert binding['final_source_pins'] is not None and binding['fixture_source_pins'] is not None
    # Consistency is NOT authority; the external existing Root/stock guard must
    # independently approve this distinct instrumented control before execution.
    sources={name:final_bytes(final_sources,name,binding) for name in final_sources}
    inputs={'descriptor':copy.deepcopy(descriptor),'spec':copy.deepcopy(spec)}
    rule=recipes['recipes'][case_id]
    assert lane in ('Source_branch_only','actual_causal')
    fault=rule['source_fault_provenance_only'] if lane=='Source_branch_only' else None
    if fault is not None:
        name=fault['file'];text=sources[name].decode()
        # Exactly one explicit old anchor; final independent reviewer accepts
        # this new final-source-derived CONTROL, never the old50 source hashes.
        assert text.count(fault['anchor'])==1,'changed final Source anchor needs new Root binding/review'
        text=text.replace(fault['anchor'],fault['replacement'],1)
        sources[name]=text.encode()
    f=rule['pinned_input_provenance_only'] if lane=='Source_branch_only' else None
    if f is not None:
        parts=INPUT_COMPONENTS[case_id]
        root=inputs['spec'] if parts[0]=='native' else inputs['descriptor']
        cur=root
        for component in parts[:-1]:cur=cur[component]
        assert parts[-1] in cur
        if f['op']=='remove':del cur[parts[-1]]
        else:
            assert f['op']=='replace';cur[parts[-1]]=f['value']
    return {'sources':sources,'inputs':inputs,'case_id':case_id,
            'changed_control_requires_distinct_review_and_pin':True,
            'runtime_executed':False,'runtime_authority':None}

def instrument_host(raw,binding):
    assert binding['instrumented_control_review'] is not None
    text=raw.decode()
    init='    capture=None;group_created=False;mounted=False;receipt_may_changed=False;caller_waited=False\n'
    assert text.count(init)==1
    helper='''    c40_events=[]
    def c40_note(operation):
        require(len(c40_events)<128,'finite actual host causal observations')
        c40_events.append({'operation':operation,'at_monotonic':time.monotonic(),'at_unix':time.time(),
            'attempted':attempted,'caller_waited':caller_waited,'mounted':mounted,'group_created':group_created,
            'receipt_may_changed':receipt_may_changed,'ambiguous':dict(c['ambiguous']),
            'live_labels':[k for k,v in c['fds'].items() if v is not None],
            'actual_fd_ids':{k:list(v) for k,v in c['ids'].items() if k.startswith('lock_')},
            'locks_acquired':list(locks),'capacity_release':False})
'''
    text=text.replace(init,init+helper,1)
    sites=[
     ("        targets_current()\n        # Same existing caller class",'before_capture'),
     ("        snapshot=capture.native_capture(descriptor)\n",'capture_returned'),
     ("        attempted=True\n",'before_Popen_attempt'),
     ("        caller_waited=True\n        require(code==0",'caller_waited_absent'),
     ("            receipt_may_changed=True # BEFORE rename/fsync; failure is UNKNOWN_NO_REPLAY.\n",'before_receipt_writer'),
     ("            updater.atomic_private_json(Path(staged['path']),staged['payload']) # SAME original genuine writer.\n",'genuine_writer_returned'),
     ("        normal_cleanup()\n        # Namespace lifetime",'normal_cleanup_returned')]
    for anchor,operation in sites:
        assert text.count(anchor)==1,'changed host causal site requires new Root binding'
        # Insert after single-statement sites; before multi-line branch anchors.
        if operation in ('before_capture','caller_waited_absent','normal_cleanup_returned'):
            first,rest=anchor.split('\n',1)
            indent=first[:len(first)-len(first.lstrip())]
            text=text.replace(anchor,first+'\n'+indent+"c40_note("+repr(operation)+")\n"+rest,1)
        else:
            indent=anchor[:len(anchor)-len(anchor.lstrip())]
            text=text.replace(anchor,anchor+indent+"c40_note("+repr(operation)+")\n",1)
    end="        close_locks()\n        return 0\n"
    assert text.count(end)==1
    text=text.replace(end,"        c40_note('before_final_lock_release')\n        final_record('C40-HOST-CAUSE.json',{'events':c40_events,'capacity_release':False})\n"+end,1)
    failure="            final_record('NATIVE-STOP.json',{'class':type(exc).__name__"
    assert text.count(failure)==1
    text=text.replace(failure,"            c40_note('actual_exception')\n            final_record('C40-HOST-CAUSE.json',{'events':c40_events,'class':type(exc).__name__,'detail':str(exc)[:300],'capacity_release':False})\n"+failure,1)
    setup='        snapshot=capture.native_capture(descriptor)\n'
    assert text.count(setup)==1
    before="""        fixture=n['fixture']
        if fixture['id']=='CAPTURE_SAME_READ_DIGEST_DRIFT' and fixture['same_read_real_mutation']:
            capture._c40_same_read_mutation={'path':fixture['disposable_capture_path'],'Root_bound_disposable_root':fixture['Root_bound_disposable_root']}
"""
    after="""        if fixture['id']=='CAPTURE_PACKAGE_RESOURCE_NO_REOPEN':
            path=Path(fixture['disposable_capture_path']);private=Path(fixture['Root_bound_disposable_root'])
            require(path.is_relative_to(private) and path.resolve(strict=True).is_relative_to(private),'Root-bound disposable capture mutation only')
            original=snapshot[str(path)];require(0<len(original)<=16777216,'finite captured disposable resource')
            fd=own('c40_disposable_capture_writer',os.open(path,os.O_WRONLY|os.O_NOFOLLOW))
            require(os.pwrite(fd,bytes([original[0]^1]),0)==1,'actual postcapture private byte mutation')
            os.fsync(fd);close_native('c40_disposable_capture_writer');c40_note('actual_resource_mutation_after_capture')
"""
    before="        capture.native41_fixture=n['fixture']\n"+before
    text=text.replace(setup,before+setup+after,1)
    return text.encode()
