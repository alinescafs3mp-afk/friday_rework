def native41_bootstrap_probe(group_path,group_relative,context):
    # Real fork plus real thread BEFORE caller GO or DURING host captured import.
    # Same private namespace/group, no new worker/controller/protocol service.
    assert context in ('first_caller_before_GO','host_captured_import')
    group=Path(group_path)
    expected={'memory.max':'2147483648','memory.swap.max':'0','pids.max':'128','cpu.max':'200000 100000'}
    def sample():
        assert Path('/proc/self/cgroup').read_text().strip()=='0::'+group_relative
        actual={key:(group/key).read_text().strip() for key in expected};assert actual==expected
        denied=None
        try:
            fd=os.open(group/'memory.max',os.O_WRONLY|os.O_NOFOLLOW)
        except OSError as error:
            denied=error.errno
        else:
            os.close(fd)
            raise RuntimeError('C41 controller write open unexpectedly permitted; NO write/retry')
        assert denied in (13,30) # Actual Landlock/readonly kernel refusal, no fake predicate.
        fields=Path('/proc/self/stat').read_text().rsplit(')',1)[1].split()
        assert resource.getrlimit(resource.RLIMIT_AS)==(1073741824,1073741824)
        return {'pid':os.getpid(),'tid':threading.get_native_id(),'birth':fields[19],
                'group':group_relative,'controllers':actual,'context':context,'actual_controller_write_denial_errno':denied}
    thread_rows=[];errors=[]
    def target():
        try:thread_rows.append(sample())
        except BaseException as error:errors.append(type(error).__name__)
    thread=threading.Thread(target=target);thread.start();thread.join(timeout=5)
    assert not thread.is_alive() and len(thread_rows)==1 and not errors
    read_fd,write_fd=os.pipe();pid=os.fork()
    if pid==0:
        try:
            os.close(read_fd);raw=json.dumps(sample(),sort_keys=True).encode();assert len(raw)<=4096
            assert os.write(write_fd,raw)==len(raw);os.close(write_fd);os._exit(0)
        except BaseException:os._exit(125)
    # Same finite probe retains exact child handle before pipe/sample/wait.
    retained={'pid':pid,'pidfd':None,'waited':False,'closed':False}
    retained['pidfd']=os.pidfd_open(pid)
    before=os.fstat(retained['pidfd'])
    os.close(write_fd);raw=os.read(read_fd,4097);assert 0<len(raw)<=4096
    os.close(read_fd);waited,status=os.waitpid(pid,0)
    assert waited==pid and status==0 and not Path('/proc/'+str(pid)).exists()
    retained['waited']=True
    after=os.fstat(retained['pidfd']);assert (before.st_dev,before.st_ino)==(after.st_dev,after.st_ino)
    os.close(retained['pidfd']);retained['closed']=True
    child=json.loads(raw);assert child['pid']==pid and child['controllers']==expected
    return {'thread':thread_rows[0],'child':child,'direct_wait':True,'fresh_child_absent':True,
            'real_fork_and_thread':True,'actual_inherited_before_GO':True,'not_controller_migration':True}
