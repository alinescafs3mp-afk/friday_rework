def read_small(path,limit=65536):
    fd=os.open(path,os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK)
    try:
        b=os.read(fd,limit+1);require(len(b)<=limit,'finite read');return b
    finally:os.close(fd)


def pin(ref):
    require(type(ref) is dict and set(ref)=={'path','bytes','sha256'},'exact probe input pin')
    p=Path(ref['path']);require(p.is_absolute() and p.resolve(strict=True)==p,'probe exact pin path')
    fd=os.open(p,os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK)
    try:
        a=os.fstat(fd);require(stat.S_ISREG(a.st_mode) and a.st_nlink==1 and a.st_size==ref['bytes']
            and 0<ref['bytes']<=2097152,'probe finite regular pin')
        b=bytearray()
        while len(b)<=ref['bytes']:
            x=os.read(fd,min(65536,ref['bytes']+1-len(b)))
            if not x:break
            b.extend(x)
        z=os.fstat(fd)
        require((a.st_dev,a.st_ino,a.st_size,a.st_mtime_ns,a.st_ctime_ns)
                ==(z.st_dev,z.st_ino,z.st_size,z.st_mtime_ns,z.st_ctime_ns)
                and len(b)==ref['bytes'] and hashlib.sha256(b).hexdigest()==ref['sha256'],
                'probe same-read captured bytes')
        return bytes(b)
    finally:os.close(fd)


def birth(pid):
    raw=read_small('/proc/'+str(pid)+'/stat',8192).decode();s=raw.rsplit(')',1)[1].split()
    return {'pid':pid,'birth':s[19],'state':s[0],'ppid':int(s[1]),'pgid':int(s[2]),'sid':int(s[3])}


def absent(row):
    try:return birth(row['pid'])['birth']!=row['birth']
    except FileNotFoundError:return True


def snapshot(spec,label,pid=None,tid=None):
    pid=os.getpid() if pid is None else pid;row=birth(pid);group=Path(spec['native']['group']);q=group.stat()
    require(read_small('/proc/'+str(pid)+'/cgroup',8192).decode().strip()=='0::'+spec['native']['group_relative'],
            'actual inherited reservation from birth')
    controllers={k:read_small(group/k,4096).decode().strip() for k in CONTROLLERS}
    require(controllers==CONTROLLERS and [q.st_dev,q.st_ino]==spec['native']['first_birth_group_identity'],
            'same actual inherited controller inode/values')
    tasks=int(read_small(group/'pids.current',4096));require(0<tasks<=128,'actual Linux tasks including threads')
    result={'label':label,'identity':row,'tid':tid,'controllers':controllers,'group_identity':[q.st_dev,q.st_ino],
        'kernel_pids_current':tasks,'pidns_inode':os.stat('/proc/'+str(pid)+'/ns/pid').st_ino,
        'userns_inode':os.stat('/proc/'+str(pid)+'/ns/user').st_ino,
        'mountns_inode':os.stat('/proc/'+str(pid)+'/ns/mnt').st_ino,
        'at_unix':time.time(),'at_monotonic':time.monotonic()}
    # Z has no live syscall/wchan/remote-FD evidence. Existence != actual wait.
    if row['state']=='Z':result['zombie_no_live_probe_or_reap_credit']=True
    return result


def deny_open(path):
    try:fd=os.open(path,os.O_WRONLY|os.O_NOFOLLOW|os.O_NONBLOCK)
    except OSError as e:
        require(e.errno in (errno.EACCES,errno.EROFS),'intended actual kernel write denial')
        return {'path':str(path),'actual_errno':e.errno,'write_count':0}
    else:
        os.close(fd);raise RuntimeError('write-open unexpectedly allowed; NO write/retry')


def real_thread_and_child(spec,context):
    rows=[];errors=[];gate=threading.Event()
    def target():
        try:
            rows.append(snapshot(spec,context+'_thread',tid=threading.get_native_id()))
            rows.append({'denial':deny_open(Path(spec['native']['group'])/'memory.max')})
        except BaseException as e:errors.append(type(e).__name__)
        finally:gate.set()
    thread=threading.Thread(target=target);thread.start()
    require(gate.wait(5),'bounded actual thread sample');thread.join(timeout=5)
    require(not thread.is_alive() and not errors and len(rows)==2,'actual thread joined')
    r,w=os.pipe();pid=os.fork()
    if pid==0:
        try:
            os.close(r);data={'snapshot':snapshot(spec,context+'_child'),
                'denial':deny_open(Path(spec['native']['group'])/'memory.max')}
            raw=json.dumps(data).encode();require(len(raw)<=8192,'bounded real child output')
            require(os.write(w,raw)==len(raw),'actual child pipe write');os.close(w);os._exit(0)
        except BaseException:os._exit(125)
    retained={'pid':pid,'pidfd':None,'pipe_r':r,'pipe_w':w,'waited':False}
    # The SAME finite probe retains all handles before fallible evidence work.
    retained['pidfd']=os.pidfd_open(pid);fdid=os.fstat(retained['pidfd']);os.close(w);retained['pipe_w']=None
    require(select.select([r,retained['pidfd']],[],[],5)[0],'actual child ready/terminal')
    raw=os.read(r,8193);require(0<len(raw)<=8192,'actual bounded child bytes');child=json.loads(raw)
    waited,status=os.waitpid(pid,0);require(waited==pid and status==0,'actual direct child wait')
    require(absent(child['snapshot']['identity']),'fresh exact child birth absent AFTER wait')
    q=os.fstat(retained['pidfd']);require((q.st_dev,q.st_ino)==(fdid.st_dev,fdid.st_ino),'same retained child pidfd')
    os.close(r);os.close(retained['pidfd']);retained['waited']=True
    return {'thread':rows,'child':child,'direct_wait_status':status,'fresh_exact_child_absent':True,
            'thread_joined':True,'real_fork_and_thread':True}


def zombie_and_adopted(spec):

    waited,status=os.waitpid(pid,0);require(waited==pid and status==0 and absent(row),'actual zombie direct wait/fresh absence')
    os.close(fd)

            require(os.write(w,str(leaf).encode())>0,'actual adopted identity');os.close(w);os.close(go_r);os._exit(0)
        except BaseException:os._exit(125)

    os.close(leaffd);os.close(midfd)
    return {'zombie':z,'zombie_actual_wait':True,'adopted_live':live,'adopted_actual_wait':True,

def task_count_limit(spec):
    group=Path(spec['native']['group']);before=read_small(group/'pids.events',4096).decode()
    release=threading.Event();threads=[];error=None
    try:
        for _ in range(129):
            t=threading.Thread(target=lambda:release.wait(5))
            try:t.start()
            except RuntimeError as e:error=type(e).__name__;break
            threads.append(t)
        peak=int(read_small(group/'pids.current',4096));after=read_small(group/'pids.events',4096).decode()
        counters=lambda raw:dict((k,int(v)) for k,v in (x.split() for x in raw.splitlines()))
        require(error is not None and peak==128 and counters(after)['max']>counters(before)['max'],
                'actual kernel pids128 task/thread refusal, not Python exception alone')
    finally:
        release.set()
        for t in threads:t.join(timeout=5)
    require(all(not t.is_alive() for t in threads),'all actual finite quota threads joined')
    return {'created_real_threads':len(threads),'actual_peak_tasks':peak,'actual_pids_events_before':before,
            'actual_pids_events_after':after,'intended_kernel_limit_observed':True}


def output_occupancy(spec):
    root=Path(spec['record']);files=[];written=0;bytes_errno=None;inode_errno=None;chunk=b'Q'*1048576
    try:
        for i in range(19):
            p=root/('bytes-'+str(i));fd=os.open(p,os.O_WRONLY|os.O_CREAT|os.O_EXCL,0o400);files.append(p)
            try:
                for _ in range(4):
                    require(written+len(chunk)<=MAX_WRITE,'finite aggregate lifetime write reservation')
                    try:n=os.write(fd,chunk)
                    except OSError as e:
                        require(e.errno==errno.ENOSPC,'actual bounded tmpfs byte exhaustion');bytes_errno=e.errno;break
                    written+=n;require(n==len(chunk),'finite complete chunk or explicit unknown')
            finally:os.close(fd)
            if bytes_errno is not None:break
        v=os.statvfs(root);byte_sample={'blocks':v.f_blocks,'available':v.f_bavail,'size':v.f_frsize}
        require(bytes_errno==errno.ENOSPC and v.f_bavail==0,'intended actual tmpfs byte boundary')
    finally:
        for p in files:p.unlink()
    files=[]
    try:
        for i in range(513):
            p=root/('inode-'+str(i))
            try:fd=os.open(p,os.O_WRONLY|os.O_CREAT|os.O_EXCL,0o400)
            except OSError as e:
                require(e.errno==errno.ENOSPC,'actual bounded tmpfs inode exhaustion');inode_errno=e.errno;break
            files.append(p);os.close(fd)
        v=os.statvfs(root);inode_sample={'files':v.f_files,'free':v.f_ffree}
        require(inode_errno==errno.ENOSPC and v.f_ffree==0,'intended actual tmpfs inode boundary')
    finally:
        for p in files:p.unlink()
    return {'actual_byte_errno':bytes_errno,'actual_inode_errno':inode_errno,'bytes_written':written,
            'lifetime_probe_write_upper':MAX_WRITE,'actual_byte_sample':byte_sample,'actual_inode_sample':inode_sample,
            'occupancy_not_cumulative_IO_credit':True}
