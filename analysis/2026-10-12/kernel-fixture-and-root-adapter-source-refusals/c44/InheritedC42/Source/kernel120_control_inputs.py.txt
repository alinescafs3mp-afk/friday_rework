"""Finite existing-Root input preparation, never admission or runtime runner.
Actual bad-controller groups/alternate handles must ALREADY be independently
owned, bounded and admitted by Root. This module never creates/writes a cgroup,
starts a process, modifies a protected path or resets an original clock.
"""
import copy


def bind_preflight_control(base,case_id,Root_actual_control_bindings):
    spec=copy.deepcopy(base);actual=Root_actual_control_bindings[case_id]
    assert actual['Root_only_actual_private_control_provenance'] is not None
    assert actual['original_clock_ref']==base['clock']
    if case_id in ('K120_UNBOUNDED_MEMORY','K120_NONZERO_SWAP','K120_UNBOUNDED_TASKS','K120_UNBOUNDED_CPU',
                   'K120_SHARED_WRITER_SCOPE'):
        for key in ('group','group_relative','first_birth_group_identity'):spec['native'][key]=actual[key]
        spec['existing_owner']['group_fd']=actual['group_fd']
    elif case_id=='K120_FOREIGN_GROUP_FD':spec['existing_owner']['group_fd']=actual['foreign_group_fd']
    elif case_id in ('K120_FOREIGN_SLOT_FD','K120_UNHELD_SLOT'):spec['existing_owner']['slot_fd']=actual['alternate_slot_fd']
    elif case_id=='K120_RELATIVE1800_GUARD':spec['native']['stock_guard_seconds']=1800
    elif case_id=='K120_OUTPUT_NOT_BOUNDED':spec['record']=actual['private_unbounded_output_record']
    else:
        # Clock/acceptance/role/unsupported-facility controls are performed by
        # EXISTING Root before entry or by actual kernel refusal. Never fake
        # getrlimit/clock/syscall outcomes in candidate code.
        raise ValueError('Root-owned before-effect or actual-kernel control; no synthetic candidate outcome')
    return spec


def validate_existing_Root_before_entry(spec,Root_expected):
    # Root_expected comes from the already-authorized existing owner context,
    # NEVER an untrusted spec/JSON grant. All refs remain unbound in C42.
    assert Root_expected['actual_source_safety_acceptance'] is not None
    assert Root_expected['actual_specific_action_admission'] is not None
    assert spec['clock']==Root_expected['original_clock_pin']
    assert spec['owner_host']==Root_expected['accepted_current_host_pin']
    assert spec['existing_owner']['group_fd']==Root_expected['actual_private_group_fd']
    assert spec['existing_owner']['slot_fd']==Root_expected['actual_existing_slot_fd']
    assert Root_expected['whole_original_batch_plan_fits'] is True
    assert Root_expected['kernel120_NOT_GRANTED'] is False
    return spec # consistency only, does not create Root authority or a process.
