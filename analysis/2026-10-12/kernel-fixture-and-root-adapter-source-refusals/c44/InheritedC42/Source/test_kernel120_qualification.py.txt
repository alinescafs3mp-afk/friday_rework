def _actual(trace,case_id):
    assert trace['case_id']==case_id and trace['actual_Root_independent_trace_ref'] is not None
    assert trace['Root_control_fixture_independent_Source_acceptance_ref'] is not None
    assert trace['Root_control_authorization_ref'] is not None
    assert trace['original_clock_ref']==trace['Root_expected_original_clock_ref']
    assert trace['clock_reset'] is False and trace['old_clock_or_Source_credit_transferred'] is False
    assert trace['simulated'] is False and trace['unknown_as_zero'] is False
    assert trace['protected_writer_controller_changes']==0
    assert trace['native_genuine_receipt_or_install_calls']==0
    assert trace['capacity_release_before_Root_wait_kernel_mount_FD_closure'] is False


def _positive(trace,case_id,cause):
    _actual(trace,case_id)
    assert trace['candidate_independent_exact_Source_acceptance_ref'] is not None
    assert trace['candidate_actual_specific_action_admission_ref'] is not None
    assert trace['intended_cause']==cause and trace['actual_operation_ref'] is not None
    assert trace['raw_kernel_observations_ref'] is not None and trace['unrelated_error_as_pass'] is False
    assert trace['actual_Root_stock_wait_ref'] is not None and trace['fresh_exact_stock_actor_absence_ref'] is not None
    assert trace['actual_Root_after_wait_kernel_pids_current']==0
    assert trace['actual_Root_after_wait_kernel_populated']==0
    assert trace['normal_empty_group_removal_ref'] is not None
    assert trace['normal_umount_and_fresh_mount_absence_ref'] is not None
    assert trace['known_Root_FD_close_once_ref'] is not None


def _negative(trace,case_id,cause):
    _actual(trace,case_id)
    assert trace['intended_cause']==cause and trace['actual_refusal_trace_ref'] is not None
    assert trace['unrelated_error_as_pass'] is False and trace['fallback_or_retry_count']==0
    if case_id=='K120_PIDFD_OR_NAMESPACE_UNSUPPORTED':
        assert trace['actual_clone3_syscall_result']<0
        assert trace['actual_kernel_errno'] in (1,22,38,95)
        assert trace['actual_Root_fresh_private_kernel0_ref'] is not None
    else:
        assert trace['actual_first_birth_call_count']==0
        assert trace['actual_Root_no_child_and_private_group_unchanged_ref'] is not None
