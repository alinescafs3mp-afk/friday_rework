"""Bounded test instrumentation on the EXISTING one-shot native owner.
No new owner, worker, grant, watchdog, socket, service or protocol. Root alone
may load this captured fixture after independent admission. Exceptions and all
original return values propagate; kernel snapshots never invent missing zeros.
"""
import hashlib
import json
import os
from pathlib import Path
import stat
import time
import types
BIRTH_FIELDS={'pid','birth','state','ppid','waited','absent','returncode'}
OWNED_FIELDS={'known','direct_pids','unknown_spawn','ambiguous','output_bytes','live_labels','pending_receipt'}
DELTA_FIELDS={'owned_delta','owned_delta_deleted','known_birth_delta','known_birth_deleted','known_birth_order'}

def strict_json(raw):
    def pairs(rows):
        value={}
        for key,item in rows:
            assert key not in value, 'duplicate JSON object key'
            value[key]=item
        return value
    def constant(value):raise AssertionError('nonfinite JSON constant')
    return json.loads(raw,object_pairs_hook=pairs,parse_constant=constant)

def identity_key(row):
    assert type(row)is dict and set(row)==BIRTH_FIELDS
    assert type(row['pid'])is int and row['pid']>0
    assert type(row['birth'])is str and row['birth'].isascii() and row['birth'].isdecimal()
    assert 0<len(row['birth'])<=20 and str(int(row['birth']))==row['birth'] and int(row['birth'])>0
    assert type(row['state'])is str and row['state'] in ('R','S','D','Z','T','t','X','x','K','W','P','I')
    assert type(row['ppid'])is int and row['ppid']>=0
    assert type(row['waited'])is bool and type(row['absent'])is bool
    assert row['returncode'] is None or type(row['returncode'])is int
    assert not row['absent'] or row['waited'], 'absence is not unobserved wait credit'
    return json.dumps([row['pid'],row['birth']],separators=(',',':'),allow_nan=False)

def owned_state(value):
    assert type(value)is dict and set(value)==OWNED_FIELDS
    rows=value['known'];assert type(rows)is list and len(rows)<=128
    keys=[identity_key(row) for row in rows];assert len(keys)==len(set(keys))
    direct=value['direct_pids'];assert type(direct)is list and len(direct)<=8
    assert all(type(pid)is int and pid>0 for pid in direct) and len(direct)==len(set(direct))
    for pid in direct:assert len([row for row in rows if row['pid']==pid and not row['waited']])==1
    assert type(value['unknown_spawn'])is bool and type(value['pending_receipt'])is bool
    assert type(value['ambiguous'])is dict and len(value['ambiguous'])<=128
    assert type(value['output_bytes'])is int and 0<=value['output_bytes']<=33554432+8192
    labels=value['live_labels'];assert type(labels)is list and len(labels)<=128
    assert all(type(label)is str and 0<len(label)<=256 for label in labels) and len(labels)==len(set(labels))
    return keys


CONTROLLERS={'memory.max':'2147483648','memory.swap.max':'0','pids.max':'128','cpu.max':'200000 100000'}
METHODS=('native_authority','native_capture_read','native_capture','native_capture_install',
         'native_module','native_spawn','native_run','native_charge','native_wait',
         'native_scan','native_children_closed','native_stage_receipt','native_close_selector',
         'native_close_stream','native_close_fd','native_capture_restore','native_finish')


def read_small(path,cap=65536):
    fd=os.open(path,os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK)
    try:
        raw=os.read(fd,cap+1)
        if len(raw)>cap:raise RuntimeError('finite fixture read exhausted')
        return raw.decode()
    finally:os.close(fd)


def kernel_snapshot(owner):
    group=owner.group
    result={'monotonic':time.monotonic(),'unix':time.time(),'controllers':{},'counters':{},
            'group_identity':None,'own_cgroup':None,'tasks':None,'namespace':None,'error':None}
    try:
        s=group.stat();result['group_identity']=[s.st_dev,s.st_ino]
        result['controllers']={n:read_small(group/n,4096).strip() for n in CONTROLLERS}
        for n in ['memory.events','pids.events','cpu.stat','cgroup.events']:
            result['counters'][n]={k:int(v) for k,v in (row.split() for row in read_small(group/n).splitlines())}
        result['tasks']=int(read_small(group/'pids.current',4096))
        result['own_cgroup']=read_small('/proc/self/cgroup',4096).strip()
        result['namespace']=os.stat('/proc/self/ns/pid').st_ino
    except BaseException as exc:
        result['error']={'class':type(exc).__name__,'detail':str(exc)[:200]}
    return result


def owned_snapshot(owner):
    # Copy bounded primitive observations, never a CompletedProcess/JSON claim.
    known=[{k:v.get(k) for k in ['pid','birth','state','ppid','waited','absent','returncode']}
           for v in owner.known.values()]
    if len(known)>128:raise RuntimeError('native fixture birth inventory exhausted')
    return {'known':known,'direct_pids':list(owner.direct),'unknown_spawn':owner.unknown_spawn,
            'ambiguous':dict(owner.ambiguous),'output_bytes':owner.output_bytes,
            'live_labels':[k for k,v in owner.fds.items() if v is not None],
            'pending_receipt':owner.pending_receipt is not None}


# The SAME native_put consumer is kept at 65536 bytes. Evidence is a
# reversible delta stream, compressed in bounded memory, then split into exact
# independently pinned leaves. No event-count truncation or lost custody rows.
TRACE_COMPRESSED_MAX=430080
TRACE_RAW_MAX=16777216
SEGMENT_COMPRESSED_MAX=24576
SEGMENT_WIRE_MAX=65536
FAILURE_WIRE_MAX=65536
import zlib


def wire(value):
    return (json.dumps(value,sort_keys=True,allow_nan=False)+'\n').encode()


def observation_failure(owner,error,primary=None):
    # Reserved evidence is outside the normal stream; no failed normal stream
    # can consume it. Exact primary exception survives instrumentation errors.
    owner._c40_observation_failed=True
    owner._c40_failure_count+=1
    row={'class':type(error).__name__,'detail':str(error)[:300],
         'primary':primary,'event_count':owner._c40_count}
    if owner._c40_first_failure is None:owner._c40_first_failure=row
    owner._c40_last_failure=row


def reserved_failure(owner,primary):
    if owner._c40_reserved_attempted:return
    owner._c40_reserved_attempted=True
    writing=owner._c40_writing;owner._c40_writing=True
    try:
        failure={'schema':'friday.causal.reserved-failure.v1','case':owner._c40_case,
                 'normal_evidence_complete':False,'capacity_release':False,
                 'actual_primary_exception':primary,'event_count':owner._c40_count,
                 'first_failure':owner._c40_first_failure,'last_failure':owner._c40_last_failure,
                 'failure_count':owner._c40_failure_count,
                 'kernel':kernel_snapshot(owner),'owned':owned_snapshot(owner)}
        if len(wire(failure))>FAILURE_WIRE_MAX:raise RuntimeError('reserved actual failure cap')
        owner.native_put(owner.temp_root/'C41-OBSERVATION-FAILURE.json',failure)
    except BaseException as error:
        owner._c40_reserved_write_error={'class':type(error).__name__,'detail':str(error)[:300]}
    finally:owner._c40_writing=writing


def attach(owner,case_id):
    if getattr(owner,'_c40_observing',False):raise RuntimeError('one fixture attachment per distinct case')
    assert type(case_id)is str and 0<len(case_id)<=64
    owner.native_check();owner._c40_observing=True;owner._c40_case=case_id
    owner._c40_count=0;owner._c40_raw_bytes=0;owner._c40_compressed=bytearray()
    owner._c40_encoder=zlib.compressobj(6);owner._c40_raw_hash=hashlib.sha256()
    owner._c40_previous_owned={};owner._c40_observation_failed=False
    owner._c40_first_failure=None;owner._c40_last_failure=None;owner._c40_failure_count=0
    owner._c40_persist_attempted=False;owner._c40_writing=False;owner._c40_reserved_attempted=False
    owner._c40_original={name:getattr(owner,name) for name in METHODS}
    def add(value):
        try:
            if owner._c40_observation_failed:raise RuntimeError('causal observation already failed; no success credit')
            value={'case':case_id,'index':owner._c40_count,**value}
            if 'owned' in value:
                current=value.pop('owned')
                owned_state(current)
                previous=owner._c40_previous_owned
                value['owned_delta']={k:v for k,v in current.items() if k!='known' and (k not in previous or previous[k]!=v)}
                value['owned_delta_deleted']=[k for k in previous if k not in current]
                before={identity_key(r):r for r in previous.get('known',[])}
                after={identity_key(r):r for r in current['known']}
                if len(after)!=len(current['known']):raise RuntimeError('duplicate actual birth identity')
                value['known_birth_delta']={k:v for k,v in after.items() if k not in before or before[k]!=v}
                value['known_birth_deleted']=[k for k in before if k not in after]
                if list(before)!=list(after):value['known_birth_order']=list(after)
                owner._c40_previous_owned=current
            raw=json.dumps(value,sort_keys=True,separators=(',',':'),allow_nan=False).encode()+b'\n'
            if len(raw)>65536 or owner._c40_raw_bytes+len(raw)>TRACE_RAW_MAX:
                raise RuntimeError('bounded lossless causal stream raw cap exhausted')
            compressed=owner._c40_encoder.compress(raw)
            # Reserve the zlib terminal overhead before admitting a new record.
            if len(owner._c40_compressed)+len(compressed)+1024>TRACE_COMPRESSED_MAX:
                raise RuntimeError('bounded compressed causal stream exhausted')
            owner._c40_compressed.extend(compressed);owner._c40_raw_hash.update(raw)
            owner._c40_raw_bytes+=len(raw);owner._c40_count+=1
        except BaseException as error:
            observation_failure(owner,error)
            raise
    owner._c40_add=add
    cleanup={'native_wait','native_children_closed','native_close_selector',
             'native_close_stream','native_close_fd','native_capture_restore','native_finish'}
    for name in METHODS:
        original=owner._c40_original[name]
        def wrapper(self,*args,_original=original,_name=name,**kw):
            if self._c40_writing:return _original(*args,**kw)
            quiet=(_name=='native_charge' and self.output_bytes+len(args[0])<=33554432) or _name=='native_scan'
            if not quiet:
                try:add({'operation':_name,'phase':'enter','owned':owned_snapshot(self)})
                except BaseException as error:
                    observation_failure(self,error)
                    if _name not in cleanup:raise
            try:result=_original(*args,**kw)
            except BaseException as exc:
                primary={'class':type(exc).__name__,'detail':str(exc)[:300]}
                try:add({'operation':_name,'phase':'exception',**primary,
                         'owned':owned_snapshot(self),'kernel':kernel_snapshot(self)})
                except BaseException as error:observation_failure(self,error,primary)
                reserved_failure(self,primary)
                raise # The actual original exception, never a persist exception.
            if not quiet:
                value={'operation':_name,'phase':'return','owned':owned_snapshot(self)}
                if _name in ('native_wait','native_spawn'):value['kernel']=kernel_snapshot(self)
                try:add(value)
                except BaseException as error:
                    observation_failure(self,error)
                    if _name not in cleanup:raise
            return result
        setattr(owner,name,types.MethodType(wrapper,owner))
    add({'operation':'fixture_attach','phase':'actual','kernel':kernel_snapshot(owner),'owned':owned_snapshot(owner)})


def persist(owner):
    if owner._c40_persist_attempted:raise RuntimeError('one evidence publication attempt; no replay')
    owner._c40_persist_attempted=True
    owner._c40_writing=True
    primary=None;refs=[]
    try:
        tail=owner._c40_encoder.flush()
        if len(owner._c40_compressed)+len(tail)>TRACE_COMPRESSED_MAX:
            raise RuntimeError('reserved compressed terminal cap exhausted')
        owner._c40_compressed.extend(tail)
        compressed=bytes(owner._c40_compressed)
        for start in range(0,len(compressed),SEGMENT_COMPRESSED_MAX):
            index=len(refs);chunk=compressed[start:start+SEGMENT_COMPRESSED_MAX]
            leaf=owner.temp_root/('C41-CAUSE-SEGMENT-%03d.json'%index)
            value={'schema':'friday.causal.lossless-segment.v1','case':owner._c40_case,
                   'index':index,'offset':start,'compressed_hex':chunk.hex()}
            raw=wire(value)
            if len(raw)>SEGMENT_WIRE_MAX or len(raw)>2*len(chunk)+512:raise RuntimeError('same native_put segment boundary')
            # Retains native_track-before-fstat and native_put actual fsync/close.
            owner.native_put(leaf,value)
            refs.append({'path':str(leaf),'bytes':len(raw),'sha256':hashlib.sha256(raw).hexdigest()})
        final_owned=owned_snapshot(owner)
        owned_state(final_owned)
        if final_owned!=owner._c40_previous_owned:raise RuntimeError('terminal custody changed after last admitted event')
        value={'schema':'friday.causal.lossless-index.v1','case':owner._c40_case,
               'segments':refs,'compressed_bytes':len(compressed),
               'compressed_sha256':hashlib.sha256(compressed).hexdigest(),
               'raw_bytes':owner._c40_raw_bytes,'raw_sha256':owner._c40_raw_hash.hexdigest(),
               'event_count':owner._c40_count,'owned_delta_initial':{},
               'kernel':kernel_snapshot(owner),'owned':final_owned,
               'observation_failed':owner._c40_observation_failed,
               'failure_count':owner._c40_failure_count,
               'first_failure':owner._c40_first_failure,'last_failure':owner._c40_last_failure,
               'capacity_release':False,'root_outside_wait':'REQUIRED_NOT_OBSERVED_INSIDE'}
        if len(wire(value))>SEGMENT_WIRE_MAX:raise RuntimeError('same native_put index boundary')
        owner.native_put(owner.temp_root/'C40-CALLER-CAUSE.json',value)
        return value
    except BaseException as error:
        primary=error
        observation_failure(owner,error)
        reserved_failure(owner,{'class':type(primary).__name__,'detail':str(primary)[:300]})
        raise primary
    finally:owner._c40_writing=False


def real_workload(owner,python_path,payload_path,mode):
    # Both executable and captured immutable script MUST be externally bound;
    # no inferred permission from their hashes, this function or the mode name.
    owner.native_check()
    before=kernel_snapshot(owner)
    command=[python_path,'-I','-B',payload_path,mode,str(owner.group),str(owner.temp_root)]
    result=None;failure=None
    try:result=owner.native_run(command,timeout=30)
    except BaseException as exc:failure=exc
    # The original native_run exception object always wins over instrumentation
    # exhaustion/latching, including errors while formatting or sampling it.
    primary=None
    if failure is not None:
        primary={'class':type(failure).__name__,'detail':'UNAVAILABLE_WHEN_FORMATTING_FAILS'}
    try:
        after=kernel_snapshot(owner)
        if failure is not None:primary['detail']=str(failure)[:300]
        owner._c40_add({'operation':'real_workload','phase':'actual','mode':mode,'before':before,'after':after,
                       'returncode':None if result is None else result.returncode,'failure':primary})
    except BaseException as observation_error:
        owner._c40_observation_failed=True
        try:observation_failure(owner,observation_error,primary)
        except BaseException:
            fallback={'class':type(observation_error).__name__,'detail':'UNAVAILABLE_SECONDARY_LOGGER_ERROR',
                      'primary':primary,'event_count':owner._c40_count}
            if owner._c40_first_failure is None:owner._c40_first_failure=fallback
            owner._c40_last_failure=fallback # truthful separate failure even when formatting broke
        try:
            reserved_failure(owner,primary if primary is not None else
                {'class':type(observation_error).__name__,'detail':'ADAPTER_OBSERVATION_FAILED'})
        except BaseException:pass # Existing reserved_attempted latch: once, no replay.
        if failure is None:raise
    if failure is not None:raise failure
    return result


def capture_unknown(owner):
    owner.native_check();name='c40_undeclared_application'
    assert name not in owner.capture_catalog['modules']
    try:owner.find_spec(name)
    except ModuleNotFoundError as exc:
        assert exc.name==name
        owner._c40_add({'operation':'find_spec','phase':'actual_refusal','name':name,
                        'finder_ids':[id(x) for x in __import__('sys').meta_path],
                        'owner_id':id(owner),'class':type(exc).__name__})
        return
    raise AssertionError('unknown import used disk fallback')


def captured_resource(owner,path):
    original=owner.capture_raw[path]
    assert owner.get_data(path)==original
    row=next((r for r in owner.capture_catalog['modules'].values()
              if r['kind']=='package' and path.startswith(str(Path(r['logical_path']).parent)+'/')),None)
    assert row is not None
    name=next(n for n,r in owner.capture_catalog['modules'].items() if r is row)
    reader=owner.get_resource_reader(name)
    relative=str(Path(path).relative_to(Path(row['logical_path']).parent))
    assert reader.files().joinpath(*Path(relative).parts).read_bytes()==original
    try:reader.resource_path(relative)
    except FileNotFoundError:pass
    else:raise AssertionError('captured resource reopened disk path')
    owner._c40_add({'operation':'captured_resource','phase':'actual','path':path,
                    'sha256':hashlib.sha256(original).hexdigest(),'bytes':len(original)})
