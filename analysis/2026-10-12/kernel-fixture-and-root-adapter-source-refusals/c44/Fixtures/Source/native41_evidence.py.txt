"""Exact same-read consumer for Root's existing collector, never authority.
Pure custody validation, no runner/controller/receipt/new owner or retry.
Only after independent Source admission may Root supply already-retained FDs.
"""
import hashlib
import json
import os
from pathlib import Path
import stat
import zlib

BIRTH_FIELDS={'pid','birth','state','ppid','waited','absent','returncode'}
OWNED_FIELDS={'known','direct_pids','unknown_spawn','ambiguous','output_bytes','live_labels','pending_receipt'}
DELTA_FIELDS={'owned_delta','owned_delta_deleted','known_birth_delta','known_birth_deleted','known_birth_order'}

def strict_json(raw):
    def pairs(rows):
        value={}
        for key,item in rows:
            assert key not in value, 'duplicate JSON object key'
            value[key]=item
        return value
    def constant(value):raise AssertionError('nonfinite JSON constant')
    return json.loads(raw,object_pairs_hook=pairs,parse_constant=constant)

def identity_key(row):
    assert type(row)is dict and set(row)==BIRTH_FIELDS
    assert type(row['pid'])is int and row['pid']>0
    assert type(row['birth'])is str and row['birth'].isascii() and row['birth'].isdecimal()
    assert 0<len(row['birth'])<=20 and str(int(row['birth']))==row['birth'] and int(row['birth'])>0
    assert type(row['state'])is str and row['state'] in ('R','S','D','Z','T','t','X','x','K','W','P','I')
    assert type(row['ppid'])is int and row['ppid']>=0
    assert type(row['waited'])is bool and type(row['absent'])is bool
    assert row['returncode'] is None or type(row['returncode'])is int
    assert not row['absent'] or row['waited'], 'absence is not unobserved wait credit'
    return json.dumps([row['pid'],row['birth']],separators=(',',':'),allow_nan=False)

def owned_state(value):
    assert type(value)is dict and set(value)==OWNED_FIELDS
    rows=value['known'];assert type(rows)is list and len(rows)<=128
    keys=[identity_key(row) for row in rows];assert len(keys)==len(set(keys))
    direct=value['direct_pids'];assert type(direct)is list and len(direct)<=8
    assert all(type(pid)is int and pid>0 for pid in direct) and len(direct)==len(set(direct))
    for pid in direct:assert len([row for row in rows if row['pid']==pid and not row['waited']])==1
    assert type(value['unknown_spawn'])is bool and type(value['pending_receipt'])is bool
    assert type(value['ambiguous'])is dict and len(value['ambiguous'])<=128
    assert type(value['output_bytes'])is int and 0<=value['output_bytes']<=33554432+8192
    labels=value['live_labels'];assert type(labels)is list and len(labels)<=128
    assert all(type(label)is str and 0<len(label)<=256 for label in labels) and len(labels)==len(set(labels))
    return keys


def same_read(fd,pin,maximum=65536):
    assert type(pin)is dict and set(pin)=={'path','bytes','sha256'}
    assert type(pin['bytes'])is int and 0<pin['bytes']<=maximum
    before=os.fstat(fd);p=Path(pin['path']);a=p.lstat()
    key=lambda s:(s.st_dev,s.st_ino,s.st_mode,s.st_size,s.st_mtime_ns,s.st_ctime_ns)
    assert stat.S_ISREG(before.st_mode) and before.st_nlink==1
    assert 0<before.st_size==pin['bytes']<=maximum and key(before)==key(a)
    raw=os.pread(fd,before.st_size+1,0);after=os.fstat(fd)
    assert len(raw)==pin['bytes'] and key(before)==key(after)==key(p.lstat())
    assert hashlib.sha256(raw).hexdigest()==pin['sha256']
    return raw

def caller_evidence(index_fd,index_pin,segment_fds):
    index=strict_json(same_read(index_fd,index_pin))
    assert type(index)is dict and set(index)=={'schema','case','segments','compressed_bytes','compressed_sha256','raw_bytes','raw_sha256','event_count','owned_delta_initial','kernel','owned','observation_failed','failure_count','first_failure','last_failure','capacity_release','root_outside_wait'}
    assert index['schema']=='friday.causal.lossless-index.v1'
    assert type(index['case'])is str and 0<len(index['case'])<=64
    assert all(type(index[k])is int and index[k]>0 for k in ('compressed_bytes','raw_bytes','event_count'))
    assert type(index['failure_count'])is int and index['failure_count']==0
    assert index['first_failure'] is None and index['last_failure'] is None
    assert type(index['kernel'])is dict
    assert index['owned_delta_initial']=={}, 'producer begins from exact empty delta state'
    owned_state(index['owned'])
    refs=index['segments'];assert type(refs)is list and 1<=len(refs)<=18
    assert len(segment_fds)==len(refs) and len({r['path'] for r in refs})==len(refs)
    chunks=[];offset=0
    for ordinal,(fd,pin) in enumerate(zip(segment_fds,refs)):
        leaf=strict_json(same_read(fd,pin))
        assert set(leaf)=={'schema','case','index','offset','compressed_hex'}
        assert leaf['schema']=='friday.causal.lossless-segment.v1'
        assert leaf['case']==index['case'] and type(leaf['index'])is int and leaf['index']==ordinal
        assert type(leaf['offset'])is int and leaf['offset']==offset
        chunk=bytes.fromhex(leaf['compressed_hex']);assert 0<len(chunk)<=24576
        if ordinal<len(refs)-1:assert len(chunk)==24576
        chunks.append(chunk);offset+=len(chunk)
    compressed=b''.join(chunks)
    assert 0<len(compressed)==index['compressed_bytes']<=430080
    assert hashlib.sha256(compressed).hexdigest()==index['compressed_sha256']
    decoder=zlib.decompressobj()
    raw=decoder.decompress(compressed,16777217)
    assert len(raw)<=16777216 and decoder.eof and not decoder.unconsumed_tail and not decoder.unused_data
    assert len(raw)==index['raw_bytes'] and hashlib.sha256(raw).hexdigest()==index['raw_sha256']
    assert raw.endswith(b'\n')
    events=[];owned={};births={};birth_order=[]
    for ordinal,line in enumerate(raw.splitlines()):
        assert 0<len(line)<=65536, 'same finite per-event producer boundary'
        event=strict_json(line);assert type(event)is dict
        assert event['case']==index['case'] and type(event['index'])is int and event['index']==ordinal
        assert 'owned' not in event, 'wire never mixes reconstructed state with delta state'
        if 'owned_delta' in event:
            assert DELTA_FIELDS-{'known_birth_order'}<=set(event)
            delta=event.pop('owned_delta');deleted=event.pop('owned_delta_deleted')
            assert type(delta)is dict and set(delta)<=OWNED_FIELDS-{'known'}
            assert type(deleted)is list and all(type(key)is str for key in deleted)
            assert len(deleted)==len(set(deleted)) and set(deleted)<=set(owned)-{'known'}
            assert not set(deleted)&set(delta)
            for name in deleted:del owned[name]
            owned.update(delta)
            removed=event.pop('known_birth_deleted');updates=event.pop('known_birth_delta')
            assert type(removed)is list and all(type(key)is str for key in removed)
            assert len(removed)==len(set(removed)) and set(removed)<=set(births)
            assert type(updates)is dict and len(updates)<=128 and not set(removed)&set(updates)
            for key,row in updates.items():assert type(key)is str and key==identity_key(row), 'canonical key/pid/birth identity'
            for identity in removed:del births[identity]
            births.update(updates)
            if 'known_birth_order' in event:birth_order=event.pop('known_birth_order')
            assert type(birth_order)is list and all(type(key)is str for key in birth_order)
            assert len(birth_order)==len(set(birth_order))==len(births)<=128 and set(birth_order)==set(births)
            owned['known']=[births[key] for key in birth_order]
            assert owned_state(owned)==birth_order
            event['owned']=strict_json(json.dumps(owned,allow_nan=False))
        else:assert not DELTA_FIELDS&set(event), 'orphan delta component'
        events.append(event)
    assert events and 'owned' in events[0], 'complete initial custody snapshot'
    assert owned==index['owned'], 'terminal reconstructed custody equals independently pinned index'
    assert len(events)==index['event_count'] and len(events)>0
    assert events[0]['operation']=='fixture_attach'
    assert index['capacity_release'] is False
    assert index['root_outside_wait']=='REQUIRED_NOT_OBSERVED_INSIDE'
    assert index['observation_failed'] is False and index['failure_count']==0
    return {'index':index,'events':events,'inside_closure_is_not_outside_acceptance':True}

def failure_evidence(fd,pin):
    failure=json.loads(same_read(fd,pin))
    assert failure['schema']=='friday.causal.reserved-failure.v1'
    assert failure['normal_evidence_complete'] is False and failure['capacity_release'] is False
    assert failure['actual_primary_exception']['class']
    assert failure['kernel'] is not None and failure['owned'] is not None
    assert len(failure['owned']['known'])<=128
    return failure # Actual cause/custody only, never successful closure/credit.
