"""Read actual constructor-owned DB/association/delivery history; no writes or remint."""
from __future__ import annotations
import hashlib
import json
import os
from pathlib import Path
import sqlite3
import stat
import time
from driver.qualification import Unbound
from driver.native import object_key
from .custody import canonical_path

def pure_snapshot(value, *, limit=65536):
    def checked(item, depth=0):
        if depth > 16:
            raise Unbound("history_depth_limit")
        if item is None or type(item) in (str, int, bool):
            return item
        if type(item) is list:
            if len(item) > 4096:
                raise Unbound("history_list_limit")
            return [checked(x, depth+1) for x in item]
        if type(item) is dict:
            if len(item) > 4096 or any(type(k) is not str for k in item):
                raise Unbound("history_mapping_limit")
            return {key: checked(x, depth+1) for key, x in item.items()}
        if type(item) is float:
            import math
            if not math.isfinite(item):
                raise Unbound("history_number_not_finite")
            return item
        raise Unbound("history_not_original_pure_data")
    copied = checked(value)
    body = json.dumps(copied, sort_keys=True, ensure_ascii=True, allow_nan=False).encode()
    if len(body) > limit:
        raise Unbound("history_page_capacity")
    return {"actual": copied, "bytes": len(body), "sha256": hashlib.sha256(body).hexdigest()}

def read_native_history(caller):
    from .observation import OriginalCaller
    if type(caller) is not OriginalCaller:
        raise Unbound("history_caller_not_exact_original_Source")
    caller.runtime.require_effects(("history_read","session_db","plugin_state"),useful=False)
    if time.monotonic() >= caller.deadline:
        return {"unknown":["history_not_started_after_original_caller_deadline"]}
    binding = caller.binding
    if binding is None:
        return {"unknown": ["genuine_original_binding_missing"]}
    registry = binding.constructor_registry
    database, host = binding.database, binding.host
    if (not registry.observed("database", database) or not registry.observed("host", host)
            or host.ctx is not binding.ctx or binding.cli.agent is not binding.agent
            or binding.agent._session_db is not database):
        return {"unknown": ["constructor_owned_DB_or_host_replaced"]}
    path = Path(canonical_path(database.db_path))
    home_custody = caller.runtime.home_custody
    home_custody.verify()
    if path != binding.cap.home / "state.db":
        return {"unknown": ["original_DB_path_changed"]}
    before = os.stat("state.db",dir_fd=home_custody.fd,follow_symlinks=False)
    if (not stat.S_ISREG(before.st_mode) or before.st_uid != os.getuid() or before.st_nlink != 1
            or before.st_mode & 0o077):
        return {"unknown": ["original_DB_privacy_or_file_contract_changed"]}
    session_id = binding.agent.session_id
    if type(session_id) is not str or session_id != binding.cli.session_id:
        return {"unknown": ["original_native_session_changed"]}
    entered = time.monotonic()
    # Kernel directory FD anchor + fixed safe leaf; WAL/SHM names stay in the
    # SAME retained directory. Source forbids pathname/ancestor repinning.
    anchored_uri = "file:/proc/self/fd/"+str(home_custody.fd)+"/state.db?mode=ro"
    home_custody.verify()
    connection = sqlite3.connect(anchored_uri,uri=True,timeout=.05)
    try:
        connection.execute("PRAGMA query_only=ON")
        session = connection.execute("SELECT source,user_id,profile_name,origin_json FROM sessions WHERE id=?",
                                     (session_id,)).fetchall()
        rows = connection.execute("SELECT id,role,content FROM messages WHERE session_id=? ORDER BY id LIMIT 257",
                                  (session_id,)).fetchall()
        if len(session) != 1 or len(rows) > 256:
            raise Unbound("actual_native_session_missing_or_history_capacity")
        # Actual source/profile/origin rows are evidence, never reconstructed
        # admission/session capabilities and never current-delivery authority.
        messages = []
        for ident, role, content in rows:
            if content is not None and type(content) is not str:
                raise Unbound("native_message_content_not_original_text")
            encoded = b"" if content is None else content.encode("utf-8")
            messages.append({"id": ident, "role": role, "actual_content_bytes": len(encoded),
                             "actual_content_sha256": hashlib.sha256(encoded).hexdigest(),
                             "content_null": content is None})
        db_snapshot = pure_snapshot({"session": [list(x) for x in session], "messages": messages})
    finally:
        connection.close()
    home_custody.verify()
    after = os.stat("state.db",dir_fd=home_custody.fd,follow_symlinks=False)
    if (before.st_dev, before.st_ino) != (after.st_dev, after.st_ino):
        raise Unbound("original_DB_file_replaced_during_read")
    # Read the original native durable PluginState file as audit data after
    # closed leases. This creates no tool/admission capability, calls no result
    # action, never fabricates quiescence, and cannot resurrect a historical send.
    state_directory = Path(canonical_path(host.store._cli_directory))
    state_custody = binding.original_state_directory_custody
    if state_custody.path != state_directory:
        raise Unbound("native_state_directory_path_changed_no_repin")
    state_custody.verify()
    directory_info = os.fstat(state_custody.fd)
    if (not stat.S_ISDIR(directory_info.st_mode) or directory_info.st_mode & 0o077
            or directory_info.st_uid != os.getuid()
            or (directory_info.st_dev, directory_info.st_ino) != binding.original_state_directory_identity):
        raise Unbound("actual_original_state_directory_changed")
    state_path = state_directory / "state.json"
    state_fd = state_custody.open_leaf("state.json",os.O_RDONLY)
    try:
        state_before = os.fstat(state_fd)
        if (not stat.S_ISREG(state_before.st_mode) or state_before.st_uid != os.getuid()
                or state_before.st_nlink != 1 or state_before.st_mode & 0o077 or state_before.st_size > 1048576):
            raise Unbound("actual_original_plugin_state_file_contract")
        chunks, size = [], 0
        while True:
            chunk = os.read(state_fd, min(65536, 1048577-size))
            if not chunk:
                break
            chunks.append(chunk)
            size += len(chunk)
            if size > 1048576:
                raise Unbound("actual_original_plugin_state_file_capacity")
        state_raw = b"".join(chunks)
        try:
            state_data = json.loads(state_raw.decode("utf-8-sig"))
        except (ValueError,UnicodeError,RecursionError,MemoryError) as exc:
            return {"DB":db_snapshot,"actual_native_state_bytes":state_raw,"actual_native_state_sha256":hashlib.sha256(state_raw).hexdigest(),
                    "unknown":["actual_durable_state_parse:"+type(exc).__name__],"human_read":"UNCONFIRMED"}
        if type(state_data) is not dict:
            return {"DB":db_snapshot,"actual_native_state_bytes":state_raw,"unknown":["actual_durable_state_mapping_type"],"human_read":"UNCONFIRMED"}
        state_after = os.fstat(state_fd)
        state_custody.verify()
        actual_path = os.stat("state.json",dir_fd=state_custody.fd,follow_symlinks=False)
        if ((state_before.st_dev, state_before.st_ino, state_before.st_size, state_before.st_mtime_ns)
                != (state_after.st_dev, state_after.st_ino, state_after.st_size, state_after.st_mtime_ns)
                or (actual_path.st_dev, actual_path.st_ino) != (state_after.st_dev, state_after.st_ino)):
            raise Unbound("actual_native_durable_state_changed_during_audit")
        original = state_data.get("associations.v1")
        current = None if type(original) is not dict else original.get("jobs")
    finally:
        state_custody.close_leaf(state_fd)
    if type(current) is not dict:
        raise Unbound("original_durable_association_jobs_wrong_type")
    associations = []
    # Only original constructor-owned locally retained jobs, not historical
    # rows imported to fabricate a current owner's live capability.
    for reference, owner_refs in tuple(host._local_owners.items()):
        if any(owner_refs.get(k) is not v for k, v in
               (("cli", binding.cli), ("agent", binding.agent), ("database", database),
                ("context", binding.ctx), ("manager", binding.manager), ("state", binding.state))):
            continue
        row = current.get(reference)
        if type(row) is not dict or row.get("existing_task_id") != reference:
            raise Unbound("original_owned_association_missing_from_durable_snapshot")
        observed = {key: row.get(key) for key in ("existing_task_id", "worker_kind", "owner", "deadline_unix",
                    "elapsed_seconds", "stop_intent", "submission_observation", "goal_verification", "delivery", "result")}
        actual_host = row.get("host")
        observed["actual_host_quiescence"] = None if type(actual_host) is not dict else actual_host.get("quiescence")
        associations.append(pure_snapshot(observed))
    if len(associations) > 64:
        raise Unbound("original_owned_association_capacity")
    returned = time.monotonic()
    return {"entered": entered, "returned": returned, "database": object_key(database),
            "database_identity": (before.st_dev, before.st_ino), "session_id": session_id,
            "DB": db_snapshot, "associations": associations,
            "actual_native_state_file": {"path": str(state_path), "identity": (state_before.st_dev, state_before.st_ino),
                                       "bytes": len(state_raw), "sha256": hashlib.sha256(state_raw).hexdigest()},
            "current_delivery_authority": "NOT_CREATED_FROM_HISTORY",
            "human_read": "UNCONFIRMED", "unknown": [] if returned <= caller.deadline else ["history_read_past_original30"]}

def terminal_history_prefix(before, after):
    # Meaningful durable append-only history oracle. No status remint or send.
    if type(before) is not list or type(after) is not list:
        raise Unbound("delivery_history_not_original_arrays")
    return {"before_count": len(before), "after_count": len(after),
            "unchanged_actual_prefix": after[:len(before)] == before,
            "new_actual_attempts": after[len(before):],
            "current_operation_authority": "NONE_FROM_HISTORICAL_ROWS"}
