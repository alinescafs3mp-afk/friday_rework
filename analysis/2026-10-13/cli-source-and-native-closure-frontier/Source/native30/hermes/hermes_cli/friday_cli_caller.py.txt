"""One original CLI caller clock, born before admission or activity thread.

Source-only external contract: the current native guardian must claim the
exact original executor thread and complete/reap all participants by end.
No guardian implementation or runtime permission is supplied by this file.
Without that retained facility the new result path refuses before effects.
"""
import asyncio
from contextvars import ContextVar
from contextlib import contextmanager
import math
import threading
import time

_CURRENT = ContextVar("friday_whole_original_caller", default=None)
_REGISTRY = {}

class CallerUnconfirmed(RuntimeError):
    pass

def current(required=False):
    span = _CURRENT.get()
    if required and span is None:
        raise CallerUnconfirmed("original_caller_missing")
    return span

class WholeCaller:
    def __init__(self, agent, call_id, task_id, entry):
        from hermes_cli.friday_cli_work import _TURN
        now = time.monotonic()
        if type(entry) not in (int, float) or not math.isfinite(entry) or not 0 < entry <= now:
            raise CallerUnconfirmed("original_entry_clock_invalid")
        if type(call_id) is not str or not call_id or len(call_id.encode()) > 512:
            raise CallerUnconfirmed("original_call_id_invalid")
        self.entry = entry
        self.end = self.entry + 30.0
        self.useful = self.end - 3.0
        self.agent = agent
        self.call_id = call_id
        self.task_id = task_id
        turn = _TURN.get()
        if type(turn) is not dict or turn.get("active") is not True:
            raise CallerUnconfirmed("original_turn_not_retained")
        if agent is not turn["agent"] or task_id != turn["session_id"]:
            raise CallerUnconfirmed("foreign_original_caller")
        self.owner = turn["output"]
        if self.owner.anchor is None or self.owner.frontend is None:
            raise CallerUnconfirmed("original_descriptor_admission_not_available")
        self.thread = threading.current_thread()
        self.thread_id = threading.get_native_id()
        self.stop = threading.Event()
        self.activities = []
        self.phases = []
        self.deferred = []
        self.errors = []
        self.terminal = None
        self.completed = False
        self.tail_ready = False
        self.stop_dispatch_started = False
        self.stop_unconfirmed = False
        self.guard = None
        self.receipt_selections = {}
        from hermes_cli.friday_cli_native_custody import OriginalParticipantCustody
        self.custody = OriginalParticipantCustody(self)

    def attach_guard(self):
        # This is a proposal for integration into the CURRENT existing native
        # guard, not a new broker, generic callback or serialized grant.
        provider = getattr(self.owner.cli, "_friday_original_caller_guard", None)
        if provider is None:
            raise CallerUnconfirmed("whole_original_caller_guard_NOT_GRANTED")
        self.guard = provider.claim_original_caller(
            agent=self.agent, call_id=self.call_id, task_id=self.task_id,
            thread=self.thread, native_tid=self.thread_id, entry=self.entry,
            deadline=self.end, frontend=self.owner, session=self.owner.cap)
        if self.guard is None:
            raise CallerUnconfirmed("whole_original_caller_guard_refused")
        # Provider/source/topology and this lease's opaque authority still need
        # whole independent binding; a returned bool/dict is not a lease.
        if type(self.guard) in (bool, dict, str, int, tuple, list):
            raise CallerUnconfirmed("serialized_caller_guard_not_authority")

    def check(self, phase, *, cleanup=False):
        end = self.end if cleanup else self.useful
        if self.stop.is_set() and not cleanup:
            raise CallerUnconfirmed("original_caller_stop_requested")
        if time.monotonic() >= end:
            raise TimeoutError("original_whole_caller_deadline:" + phase)
        self.owner.check_lifetime()
        self.owner.anchor.validate()
        if self.guard is None:
            raise CallerUnconfirmed("original_caller_guard_missing")
        self.guard.check_original_caller(self, phase=phase, deadline=end)
        if len(self.phases) >= 4096:
            raise CallerUnconfirmed("original_phase_evidence_limit")
        self.phases.append((phase, time.monotonic()))

    def narrow(self, remaining):
        if type(remaining) not in (float, int) or not math.isfinite(remaining) or remaining <= 0:
            raise CallerUnconfirmed("association_budget_exhausted")
        self.end = min(self.end, time.monotonic() + remaining)
        self.useful = min(self.useful, self.end - 3.0)
        self.guard.narrow_original_caller(self, deadline=self.end)
        self.check("association_narrow")

    def loop_deadline(self, *, cleanup=False):
        loop = asyncio.get_running_loop()
        self.check("loop_deadline", cleanup=cleanup)
        return loop.time() + max(0, (self.end if cleanup else self.useful) - time.monotonic())

    def sync(self, phase, fn, *args, cleanup=False, **kwargs):
        self.check(phase + ":before", cleanup=cleanup)
        # Executes on the registered ORIGINAL native caller/owned persistent
        # loop thread. The external guard must bound blocking syscalls; these
        # checks alone never establish kernel preemption or <=30.
        value = fn(*args, **kwargs)
        self.check(phase + ":after", cleanup=cleanup)
        return value

    def activity(self, thread, stop):
        self.check("activity_register")
        record = {"thread": thread, "stop": stop, "started": False, "joined": False}
        self.activities.append(record)  # before thread.start
        self.custody.reserve_activity(record)
        self.guard.own_activity_thread(self, thread=thread, stop=stop)
        return record

    def join_activity(self, record):
        record["stop"].set()
        record["thread"].join(timeout=min(2.0, max(0, self.end - time.monotonic())))
        if record["thread"].is_alive():
            self.errors.append("activity_join_UNCONFIRMED")
            self.stop.set()
            raise CallerUnconfirmed("activity_thread_not_reaped")
        record["joined"] = True
        self.check("activity_join", cleanup=True)

    def defer_receipt(self, commit, *args):
        # Body presentation is not yet delivery. Durable UNKNOWN stays current
        # until real original executor/DB/UI/proxy tails have completed.
        self.deferred.append((commit, args))

    def finish(self):
        self.check("native_tail_before_barrier", cleanup=True)
        state = self.owner.frontend
        if self.owner.prompt is not None:
            future = state.barrier()
            value = future.result(timeout=max(0, self.end - time.monotonic()))
            if value is not True:
                raise CallerUnconfirmed("native_proxy_barrier_unknown")
            self.owner.prompt.render_barrier(self.end)
        if self.errors or any(not r["joined"] for r in self.activities) or state.pending or state.errors:
            raise CallerUnconfirmed("native_frontend_or_activity_tail_unconfirmed")
        self.check("metadata_DB_UI_and_frontend_closed", cleanup=True)
        self.custody.wait_natural_settlement(self.end)
        self.check("native_tail_ready", cleanup=True)
        self.tail_ready = True
        self.terminal = {"state": "TAIL_READY_NOT_RETURNED",
                         "at": time.monotonic(), "end": self.end}
        # Arm the CURRENT retained native observer on the exact original frame.
        # There is no self-issued caller-return certificate and no early ledger
        # DELIVERED. Observer work/durable commit is inside the SAME whole fence.
        from hermes_cli.friday_cli_return import arm_original_return
        arm_original_return(
            self, frame_site="_original_sequential_caller_slice",
            phases=tuple(self.phases), deadline=self.end,
            observed_return=self.accept_observed_return)

    def accept_observed_return(self, proof):
        if not self.tail_ready or self.completed or self.stop.is_set():
            raise CallerUnconfirmed("late_duplicate_or_stopped_original_return")
        token = _CURRENT.set(self)
        try:
            from hermes_cli.friday_cli_return import validate_original_return
            validate_original_return(
                self, proof=proof, original_entry=self.entry, deadline=self.end,
                original_thread=self.thread, original_native_tid=self.thread_id)
            self.check("actual_original_return_before_receipts", cleanup=True)
            self.guard.commit_original_receipts(
                self, proof=proof, operations=tuple(self.deferred), deadline=self.end,
                native_tail=tuple(self.phases), frontend=self.owner.frontend)
            self.check("actual_observer_and_immutable_receipts_closed", cleanup=True)
            self.guard.finish_original_caller(self, proof=proof, outcome="COMPLETED",
                                             phases=tuple(self.phases), deadline=self.end)
            self.completed = True
            self.terminal = {"state": "CLOSED", "at": time.monotonic(), "end": self.end}
            if self.owner.frontend.caller is not self:
                raise CallerUnconfirmed("observed_return_foreign_frontend")
            self.owner.frontend.caller = None
            _REGISTRY.pop((id(self.agent), self.call_id))
        except BaseException as exc:
            self.abandon(exc)
            raise
        finally:
            _CURRENT.reset(token)

    def abandon(self, error):
        self.stop.set()
        try:
            self.custody.request_stop()
        except BaseException as cleanup:
            self.stop_unconfirmed = True
            self.errors.append("participant_stop:" + type(cleanup).__name__)
        self.errors.append(type(error).__name__)
        self.terminal = {"state": "UNKNOWN", "at": time.monotonic(),
                         "error": type(error).__name__, "end": self.end}
        if self.guard is not None and not self.stop_dispatch_started:
            self.stop_dispatch_started = True
            try:
                self.guard.stop_original_caller(self, deadline=self.end,
                                              activity=tuple(self.activities),
                                              frontend=self.owner.frontend)
            except BaseException as cleanup:
                self.stop_unconfirmed = True
                self.errors.append("original_guard_cleanup:" + type(cleanup).__name__)
                # Preserve the primary cancellation/error and every cleanup
                # failure. No repeated stop request or hidden substitute.

        # Keep exact span/participants/deferred attempts in registry. Neither
        # uncertain return nor a timeout manufactures FAILED or resend.

@contextmanager
def enter_original(agent, call_id, task_id, *, entry):
    key = (id(agent), call_id)
    if key in _REGISTRY or _CURRENT.get() is not None:
        raise CallerUnconfirmed("duplicate_original_caller")
    span = WholeCaller(agent, call_id, task_id, entry)
    if getattr(span.owner.cli, "_friday_original_caller_guard", None) is None:
        raise CallerUnconfirmed("whole_original_caller_guard_NOT_GRANTED")
    _REGISTRY[key] = span
    token = _CURRENT.set(span)
    state = span.owner.frontend
    if state.caller is not None:
        _CURRENT.reset(token)
        raise CallerUnconfirmed("original_frontend_other_caller")
    state.caller = span
    try:
        span.attach_guard()
        span.check("native_entry")
        yield span
        if not span.tail_ready:
            raise CallerUnconfirmed("original_caller_tail_not_published")
    except BaseException as exc:
        span.abandon(exc)
        raise
    finally:
        _CURRENT.reset(token)
        # The span remains bound through native publication. It is not closed
        # at ResultTool.handle or the persistent _run_async coroutine return.

@contextmanager
def publish_original(agent, call_id):
    key = (id(agent), call_id)
    span = _REGISTRY.get(key)
    if span is None:
        yield None
        return
    token = _CURRENT.set(span)
    try:
        span.check("native_publication", cleanup=True)
        yield span
        span.finish()
    except BaseException as exc:
        span.abandon(exc)
        raise
    finally:
        _CURRENT.reset(token)
        # Pending exact native observer/commit remains owned and UNKNOWN.
        # accept_observed_return alone can close its current registry slot.
