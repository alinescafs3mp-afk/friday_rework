"""Concrete semantic fencing of a deferred ordinary CLI delivery attempt.

This retains immutable selection bytes from the actual locked UNKNOWN row.
It does not certify named-state atomicity, directory durability or whole-call
stop/reap. The existing original guardian still owns the real durable commit.
"""
import hashlib
import json


class ReceiptSelectionUnconfirmed(RuntimeError):
    pass


def _canonical(value):
    return json.dumps(value, sort_keys=True, ensure_ascii=False,
                      allow_nan=False, separators=(",", ":")).encode("utf-8")


def _selection(row, reference, number):
    result = row.get("result", {})
    artifacts = result.get("artifacts", [])
    matching = [item for item in artifacts if item.get("reference") == reference]
    attempts = result.get("deliveries", {}).get(reference, [])
    if (len(matching) != 1 or not attempts
            or type(number) is not int or attempts[-1].get("number") != number):
        raise ReceiptSelectionUnconfirmed("original_delivery_selection_not_current")
    attempt = attempts[-1]
    if attempt.get("state") != "UNKNOWN" or attempt.get("receipt") is not None:
        raise ReceiptSelectionUnconfirmed("original_unknown_attempt_already_changed")
    # Captures the actual owner authority generation, invocation and route.
    # No absent global store generation is invented.
    value = {
        "existing_task_id": row["existing_task_id"], "owner": row["owner"],
        "admission_hash": row["admission_hash"],
        "worker_kind": row["worker_kind"],
        "workspace_reference": row["workspace_reference"],
        "supervisor": row["supervisor"], "native": row["native"],
        "deadline_unix": row["deadline_unix"],
        "binding": row["host"]["binding"],
        "source_paths": result["source_paths"], "artifacts": artifacts,
        "reference": reference, "attempts": attempts,
    }
    body = _canonical(value)
    if len(body) > 65536:
        raise ReceiptSelectionUnconfirmed("original_receipt_selection_limit")
    return body


class OriginalReceiptSelection:
    def __init__(self, store, span, row, reference, number):
        if row["owner"].get("platform") != "cli":
            raise ReceiptSelectionUnconfirmed("not_original_cli_delivery")
        self.store = store
        self.state_object = store.state
        self.directory = store._cli_directory
        self.namespace = store._cli_namespace
        self.span = span
        self.task_id = row["existing_task_id"]
        self.reference = reference
        self.number = number
        self.body = _selection(row, reference, number)
        self.sha256 = hashlib.sha256(self.body).hexdigest()
        self.phase = "UNKNOWN_SAVE_NOT_YET_RETURNED"

    def unknown_save_returned(self):
        if self.phase != "UNKNOWN_SAVE_NOT_YET_RETURNED":
            raise ReceiptSelectionUnconfirmed("duplicate_unknown_save_return")
        self.phase = "UNKNOWN_SAVE_RETURNED"

    def begin_commit(self, store, span, row, reference, number):
        if (store is not self.store or span is not self.span
                or store.state is not self.state_object
                or store._cli_directory != self.directory
                or store._cli_namespace != self.namespace
                or self.phase != "UNKNOWN_SAVE_RETURNED"
                or row["existing_task_id"] != self.task_id
                or reference != self.reference or number != self.number
                or _selection(row, reference, number) != self.body):
            raise ReceiptSelectionUnconfirmed("late_foreign_or_changed_original_receipt")
        self.phase = "COMMIT_ENTERED"

    def save_returned(self):
        if self.phase != "COMMIT_ENTERED":
            raise ReceiptSelectionUnconfirmed("receipt_save_return_without_commit")
        self.phase = "SAVE_RETURNED_NOT_WHOLE_ACCEPTED"
