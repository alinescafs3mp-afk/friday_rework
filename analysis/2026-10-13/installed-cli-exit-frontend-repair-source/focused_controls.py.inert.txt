def owner(stream=None):
    stream=io.StringIO() if stream is None else stream
    cli=types.SimpleNamespace(console=types.SimpleNamespace(file=stream,_lock=threading.RLock()),_friday_cli_scope=object())
    result=ClassicConsoleOutput(cli);cli._friday_cli_output=result
    assert result.anchor is not None and result.frontend is frontend.lookup(stream)
    return result

def refused(fn, exc, message):
    try:fn()
    except exc as error: assert str(error)==message, str(error)
    else:raise AssertionError('required refusal missing: '+message)

def native_release_method():
    source=(root/'cli.py').read_text();tree=ast.parse(source)
    cls=next(x for x in tree.body if isinstance(x,ast.ClassDef) and x.name=='HermesCLI')
    method=next(x for x in cls.body if isinstance(x,ast.FunctionDef) and x.name=='_release_active_session')
    namespace={'logger':logging.getLogger('isolated-release')}
    exec(compile(ast.Module(body=[method],type_ignores=[]),str(root/'cli.py'),'exec'),namespace)
    return namespace['_release_active_session'], hashlib.sha256(ast.get_source_segment(source,method).encode()).hexdigest()

results=[]
def check(name,fn):
    fn();results.append({'control':name,'state':'PASS'})

def own_repeat():
    o=owner();o.close();o.close();frontend.unregister(o)
    assert o.anchor.closed and frontend.lookup(o.stream) is None

def foreign_current():
    o=owner();clone=types.SimpleNamespace(stream=o.stream,anchor=o.anchor,frontend=o.frontend,_admitted_frontend=o.frontend,_admitted_anchor=o.anchor)
    refused(lambda:frontend.unregister(clone),frontend.FrontendUnconfirmed,'foreign_registry_close')
    assert frontend.lookup(o.stream) is o.frontend;o.close()
check('foreign_owner_current_registry_rejected',foreign_current)

def foreign_reregistered():

def first_pending():
    o=owner();pending=object();o.frontend.pending[1]=pending
    refused(o.close,OutputCleanupUnconfirmed,'native_frontend_lifecycle_close_unconfirmed')
    assert o.frontend.pending[1] is pending and frontend.lookup(o.stream) is o.frontend and not o.anchor.closed and o.frontend._registry_close is None
    o.frontend.pending.clear();o.close()
check('live_pending_first_close_retains_custody',first_pending)

def repeat_pending():
    o=owner();o.close();pending=object();o.frontend.pending[1]=pending
    refused(o.close,OutputCleanupUnconfirmed,'native_frontend_lifecycle_close_unconfirmed')
    refused(lambda:frontend.unregister(o),frontend.FrontendUnconfirmed,'frontend_lifecycle_close_not_confirmed')
    assert o.frontend.pending[1] is pending;o.frontend.pending.clear();o.close()
check('live_pending_repeated_close_not_suppressed',repeat_pending)

def proxy_case(repeated,closed):
    o=owner()
    if repeated:o.close()
    stop=threading.Event();entered=threading.Event()
    def worker():entered.set();stop.wait(2)
    thread=threading.Thread(target=worker);thread.start();assert entered.wait(.25)
    proxy=types.SimpleNamespace(closed=closed,_flush_thread=thread);o.frontend.proxy=proxy
    try:
        refused(o.close,frontend.FrontendUnconfirmed,'frontend_lifecycle_close_not_confirmed')
        assert o.frontend.proxy is proxy and thread.is_alive()
        if not repeated:assert frontend.lookup(o.stream) is o.frontend and o.frontend._registry_close is None
    finally:stop.set();thread.join(.5);assert not thread.is_alive();proxy.closed=True;o.close()
check('open_live_proxy_first_close_rejected',lambda:proxy_case(False,False))
check('live_proxy_even_marked_closed_repeat_rejected',lambda:proxy_case(True,True))
