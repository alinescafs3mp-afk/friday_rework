def ordinary_cli_runtime(home, marker):
    """An owned invocation is separate from immutable installation clocks."""
    from hermes_cli.friday_dashboard_owner import read_owned
    import time, math, hashlib, re
    rows = Path('/proc/self/cgroup').read_text().splitlines()
    require(len(rows) == 1 and rows[0].startswith('0::/'), 'cli_runtime_unified_cgroup_required')
    group = rows[0][3:]
    unit = group.rsplit('/', 1)[-1]
    require(re.fullmatch(r'friday-rework-cli-runtime-[A-Za-z0-9-]+\.service', unit)
            and '..' not in Path(group).parts, 'cli_runtime_owned_native_unit_required')
    raw = read_owned(Path(home)/'preparation/cli-runtime'/(unit+'.intent.json'), private=True)
    intent = json.loads(raw)
    require(set(intent) == {'schema','state','request','clock','physical','history'}
            and intent['schema'] == 'friday.installed-cli-invocation.v1'
            and intent['state'] == 'CONSUMED_NOT_COMPLETE', 'cli_runtime_consumed_intent_required')
    request = intent['request']; clock = intent['clock']; physical = intent['physical']
    require(physical['cgroup'] == group and physical['unit'] == unit
            and re.fullmatch('[0-9a-f]{32}', physical['invocation_id']), 'cli_runtime_native_owner_changed')
    caps = {'memory.max':'2147483648','memory.swap.max':'0','pids.max':'128','cpu.max':'200000 100000'}
    root = Path('/sys/fs/cgroup')/group.lstrip('/')
    require(physical['caps'] == caps and all((root/k).read_text().strip() == v for k,v in caps.items()),
            'cli_runtime_physical_caps_changed')
    require(set(clock) == {'schema','seconds','cleanup_seconds','started_mono','deadline_mono',
            'useful_deadline_mono','boot_id','gnu_pid','gnu_starttime_ticks'}
            and clock['schema'] == 'friday.installed-cli-clock.v1'
            and clock['seconds'] == 180 and clock['cleanup_seconds'] == 30
            and type(clock['started_mono']) in (int,float) and math.isfinite(clock['started_mono'])
            and clock['deadline_mono'] == clock['started_mono']+180
            and clock['useful_deadline_mono'] == clock['started_mono']+150
            and clock['boot_id'] == Path('/proc/sys/kernel/random/boot_id').read_text().strip()
            and type(clock['gnu_pid']) is int and clock['gnu_pid'] > 1
            and type(clock['gnu_starttime_ticks']) is int and clock['gnu_starttime_ticks'] > 0
            and clock['started_mono'] <= time.monotonic() < clock['useful_deadline_mono'],
            'cli_runtime_clock_expired_or_invalid')
    oldraw = read_owned(Path(request['historical_marker']['path']), private=True)
    old = json.loads(oldraw)
    require(hashlib.sha256(oldraw).hexdigest() == request['historical_marker']['sha256']
            and marker['postinstall_original_marker'] == request['historical_marker']
            and marker['postinstall_source_input'] == request['repaired_input']
            and {k:v for k,v in marker.items() if k not in {'source_receipt_sha256',
                'postinstall_original_marker','postinstall_source_input','postinstall_native_launcher'}} ==
                {k:v for k,v in old.items() if k != 'source_receipt_sha256'}
            and old['input_sha256'] == request['installation_input']['sha256'],
            'cli_runtime_installation_history_changed')
    require(set(intent['history']) == {'FRIDAY-INSTALL.cli-segment.intent.json',
            'FRIDAY-INSTALL.cli-host.intent.json','FRIDAY-INSTALL.json.original-cli-segment'}
            and all(hashlib.sha256(read_owned(Path(home)/name, private=True)).hexdigest() == pin
                    for name,pin in intent['history'].items()), 'cli_runtime_history_changed')
    import sys
    from hermes_constants import get_hermes_home
    require(type(request['argv']) is list and request['argv'][:2] == ['-p','default']
            and get_hermes_home() == Path(request['home']) == Path(home)
            and list(sys.argv[1:]) == request['argv'][2:], 'cli_runtime_invocation_changed')
    import resource
    require(os.sched_getaffinity(0) == {0,1}
            and resource.getrlimit(resource.RLIMIT_CORE) == (0,0)
            and resource.getrlimit(resource.RLIMIT_FSIZE) == (268435456,268435456),
            'cli_runtime_native_limits_changed')
    # Repeated gate calls from this birth are allowed; another native birth
    # cannot replay the same consumed unit intent during the useful window.
    birth = Path('/proc/self/stat').read_text().rsplit(')',1)[1].split()[19]
    owner = {'boot_id':clock['boot_id'],'pid':os.getpid(),'starttime_ticks':int(birth),
             'pid_namespace':os.readlink('/proc/self/ns/pid')}
    owner_path = Path(home)/'preparation/cli-runtime'/(unit+'.native-owner.json')
    try:
        fd = os.open(owner_path, os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW, 0o600)
    except FileExistsError:
        require(json.loads(read_owned(owner_path, private=True)) == owner, 'cli_runtime_native_replay_refused')
    else:
        with os.fdopen(fd,'w') as stream:
            json.dump(owner,stream,sort_keys=True);stream.flush();os.fsync(stream.fileno())
    return clock
