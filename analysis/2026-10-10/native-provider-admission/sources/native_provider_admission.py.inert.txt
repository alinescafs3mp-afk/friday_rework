def require(ok, reason):
    if not ok:
        raise RuntimeError('NATIVE_PROVIDER_ADMISSION:' + reason)


def canonical_sha(value):
    return hashlib.sha256(json.dumps(value, sort_keys=True, separators=(',', ':')).encode()).hexdigest()


def elf_image(data, candidate=True):
    """Pure exact ELF64/x86_64 file parser; no loader or candidate execution."""
    require(data[:6] == b'\x7fELF\x02\x01', 'ELF64_LE_REQUIRED')
    def unpack(fmt, at):
        require(type(at) is int and 0 <= at and at + struct.calcsize(fmt) <= len(data), 'ELF_BOUNDS')
        return struct.unpack_from(fmt, data, at)
    h = unpack('<16sHHIQQQIHHHHHH', 0)
    require(h[2] == 62 and h[1] in (2, 3) and h[9] == 56 and h[11] == 64, 'ELF_ARCH_LAYOUT')
    require(0 < h[10] <= 64 and 0 < h[12] <= 65535 and h[13] < h[12], 'ELF_TABLE_COUNT')
    ph = [unpack('<IIQQQQQQ', h[5] + i * 56) for i in range(h[10])]
    sh = [unpack('<IIQQQQIIQQ', h[6] + i * 64) for i in range(h[12])]
    def body(s):
        require(s[4] + s[5] <= len(data), 'ELF_SECTION_BOUNDS')
        return data[s[4]:s[4] + s[5]]
    names = body(sh[h[13]])
    def string(buf, at):
        require(0 <= at < len(buf), 'ELF_STRING_BOUNDS')
        return buf[at:buf.index(b'\0', at)].decode()
    sections = {string(names, s[0]): s for s in sh}
    require(len(sections) == len(sh), 'ELF_DUPLICATE_SECTION')
    def sec(name):
        return body(sections[name])
    symsec = sections['.dynsym']; require(symsec[9] == 24 and symsec[6] < len(sh), 'ELF_DYNSYM_LAYOUT')
    strings = body(sh[symsec[6]])
    symbols = []; symbol_bytes = sec('.dynsym')
    for at in range(0, len(symbol_bytes), 24):
        n, info, other, ndx, value, size = struct.unpack_from('<IBBHQQ', symbol_bytes, at)
        symbols.append(dict(name=string(strings, n), bind=info >> 4, type=info & 15,
                            visibility=other & 3, ndx=ndx, value=value, size=size))
    vs = list(struct.unpack('<' + 'H' * (len(sec('.gnu.version')) // 2), sec('.gnu.version'))) if '.gnu.version' in sections else [1] * len(symbols)
    require(len(vs) == len(symbols), 'ELF_VERSYM_COUNT')
    needs, definitions = {}, {}
    if '.gnu.version_r' in sections:
        buf = sec('.gnu.version_r'); at = 0; seen = set()
        while True:
            require(at not in seen and at + 16 <= len(buf), 'ELF_VERNEED_BOUNDS'); seen.add(at)
            version, count, file, aux, nxt = struct.unpack_from('<HHIII', buf, at)
            require(version == 1 and 0 < count < 1024, 'ELF_VERNEED_VERSION'); a = at + aux
            for _ in range(count):
                _, _, index, name, anxt = struct.unpack_from('<IHHII', buf, a)
                needs[index & 0x7fff] = [string(strings, file), string(strings, name)]; a += anxt
            if not nxt: break
            at += nxt
    if '.gnu.version_d' in sections:
        buf = sec('.gnu.version_d'); at = 0; seen = set()
        while True:
            require(at not in seen and at + 20 <= len(buf), 'ELF_VERDEF_BOUNDS'); seen.add(at)
            version, _, index, count, _, aux, nxt = struct.unpack_from('<HHHHIII', buf, at)
            require(version == 1 and count > 0, 'ELF_VERDEF_VERSION')
            name, _ = struct.unpack_from('<II', buf, at + aux)
            definitions[index & 0x7fff] = string(strings, name)
            if not nxt: break
            at += nxt
    dynamic = [struct.unpack_from('<qQ', sec('.dynamic'), at) for at in range(0, len(sec('.dynamic')), 16)]
    imports, relatives = [], []
    require(not candidate or not any(s[2] & 2 and s[1] in (9, 19) for s in sh), 'UNREVIEWED_REL_OR_RELR')
    for name, s in sections.items():
        if not candidate or s[1] != 4 or not s[2] & 2: continue  # Allocated SHT_RELA only.
        require(s[9] == 24 and s[6] == list(sections).index('.dynsym'), 'ELF_RELA_LAYOUT')
        for at in range(0, len(body(s)), 24):
            offset, info, addend = struct.unpack_from('<QQq', body(s), at)
            index, kind = info >> 32, info & 0xffffffff
            if kind == 8 and index == 0:
                relatives.append(dict(offset=offset, addend=addend)); continue
            require(kind in (6, 7) and 0 < index < len(symbols) and addend == 0, 'UNREVIEWED_RELOCATION_KIND')
            sy = symbols[index]; require(sy['ndx'] == 0 and sy['bind'] in (1, 2), 'UNREVIEWED_IMPORT')
            version = needs.get(vs[index] & 0x7fff)
            require((vs[index] & 0x7fff) <= 1 or version is not None, 'MISSING_IMPORT_VERSION')
            imports.append(dict(offset=offset, kind=kind, symbol=sy['name'], bind=sy['bind'], version=version))
    loads = [list(p) for p in ph if p[0] == 1]
    for row in imports + relatives:
        require(any(p[1] & 2 and p[3] <= row['offset'] and row['offset'] + 8 <= p[3] + p[6] for p in loads), 'RELOCATION_NOT_OWN_DATA')
    exports = {s['name']: s['value'] for s in symbols if s['ndx'] and s['bind'] == 1 and s['type'] == 2}
    return dict(type=h[1], loads=loads, symbols=symbols, versym=vs, verdef=definitions,
                imports=imports, relatives=relatives, exports=exports,
                needed=[string(strings, v) for tag, v in dynamic if tag == 1],
                dynamic_address=sections['.dynamic'][3],
                load_hooks={str(tag): v for tag, v in dynamic if tag in (12, 13, 25, 26, 27, 28, 32, 33)},
                forbidden_dynamic=any(tag in (15, 29, 22, 0x6ffffefb, 0x6ffffefc) for tag, _ in dynamic))


def static_elf_policy(image):
    return {k: image[k] for k in ('type', 'loads', 'imports', 'relatives', 'exports',
                                'needed', 'dynamic_address', 'load_hooks', 'forbidden_dynamic')}


def definition(image, name, version):
    matches = []
    for i, sy in enumerate(image['symbols']):
        if sy['name'] != name or not sy['ndx'] or sy['bind'] not in (1, 2) or sy['visibility'] != 0: continue
        v = image['versym'][i]
        if (version is None and not v & 0x8000) or (version is not None and image['verdef'].get(v & 0x7fff) == version):
            matches.append(sy)
    require(len(matches) == 1 and matches[0]['type'] in (2, 10), 'MISSING_AMBIGUOUS_PROVIDER_DEFINITION:' + name)
    return matches[0]


def validate_binding_record(expected, observed):
    """Pure comparison, also used by finite negative data controls."""
    require(set(observed) == {'offset', 'symbol', 'version', 'target', 'expected_target',
                              'provider', 'provider_sha256', 'provider_link_map',
                              'expected_link_map', 'target_executable', 'definition_type', 'method'}, 'OBSERVATION_SCHEMA')
    for key in ('offset', 'symbol', 'version', 'provider', 'provider_sha256', 'definition_type'):
        require(observed[key] == expected[key], 'RELOCATION_POLICY_MISMATCH:' + key)
    require(type(observed['target']) is int and type(observed['expected_target']) is int, 'TARGET_TYPE')
    if expected['provider'] is None:
        require(observed['target'] == observed['expected_target'] == 0 and observed['method'] == 'ACTUAL_GOT_NULL_WEAK', 'WEAK_NOT_NULL')
        require(observed['provider_link_map'] is None and observed['expected_link_map'] is None and observed['target_executable'] is False, 'NULL_WEAK_PROVIDER')
    else:
        require(observed['target'] > 0 and observed['target'] == observed['expected_target'], 'ACTUAL_GOT_TARGET_MISMATCH')
        require(type(observed['provider_link_map']) is int and observed['provider_link_map'] > 0 and observed['provider_link_map'] == observed['expected_link_map'], 'ACTUAL_PROVIDER_LINK_MAP_MISMATCH')
        require(observed['target_executable'] is True, 'TARGET_NOT_PROVIDER_EXECUTABLE')
        method = 'ACTUAL_GOT_IFUNC_DLVSYM_SAME_PROVIDER' if expected['definition_type'] == 10 else 'ACTUAL_GOT_EXACT_VERSIONED_DEFINITION'
        require(observed['method'] == method, 'LOOKUP_IS_NOT_RELOCATION_PROOF')
    return True


def identity(st):
    return [st.st_dev, st.st_ino, st.st_size, st.st_mtime_ns, st.st_ctime_ns]


def current_caller():
    fields = Path('/proc/self/stat').read_text().rsplit(')', 1)[1].split()
    return dict(pid=os.getpid(), start_ticks=int(fields[19]),
                boot_id=Path('/proc/sys/kernel/random/boot_id').read_text().strip())


def observe_caller_lock(fd, clock):
    require(type(fd) is int and fd >= 3, 'MISSING_WHOLE_CALLER_QUALITY_FD')
    st = os.fstat(fd)
    actual = [st.st_dev, st.st_ino, st.st_uid, st.st_gid, st.st_nlink, st.st_size, stat.S_IMODE(st.st_mode)]
    require(actual == QUALITY_ID, 'QUALITY_FD_IDENTITY')
    live = QUALITY.stat(); require((live.st_dev, live.st_ino) == (st.st_dev, st.st_ino), 'QUALITY_PATH_IDENTITY')
    token = f'{os.major(st.st_dev):02x}:{os.minor(st.st_dev):02x}:{st.st_ino}'
    lines = Path('/proc/self/fdinfo/' + str(fd)).read_text().splitlines()
    require(any(line.startswith('lock:') and 'FLOCK' in line.split() and 'WRITE' in line.split()
                and str(os.getpid()) in line.split() and token in line.split() for line in lines), 'WHOLE_CALLER_FLOCK_NOT_HELD')
    caller = current_caller(); require(caller['boot_id'] == clock['boot_id'], 'CLOCK_BOOT')
    return {**caller, 'fd': fd, 'identity': actual}


def validate_authority(policy, clock, purpose):
    """Static authority is explicitly distinct from later live observation."""
    require(policy['schema'] == 'friday.native-provider-load-policy.v1', 'POLICY_SCHEMA')
    require(policy['staged_load_authorized'] is True and policy['runtime_binding_observed'] is False,
            'STATIC_AUTHORITY_IS_NOT_OBSERVED_BINDING')
    require(policy['retry_authorized'] is False and policy['product_mode_admission'] is False, 'NO_RETRY_OR_PRODUCT_GRANT')
    require(type(policy['generation']) is int and policy['generation'] > 0
            and policy['assignment'] == clock['assignment'] and policy['generation'] == clock['generation']
            and policy['original_clock'] == clock, 'POLICY_CLOCK_ASSIGNMENT')
    require(policy['purpose'] == purpose, 'POLICY_SELECTED_PURPOSE')
    return True


def validate_caller_tuple(expected, observed):
    require(all(observed[k] == expected[k] for k in ('pid', 'start_ticks', 'boot_id', 'fd', 'identity')), 'WRONG_OR_STALE_CALLER')
    return True


def verify_static_policy(library, ref, clock, purpose):
    require(type(ref) is dict and set(ref) == {'path', 'sha256'}, 'MISSING_STAGED_POLICY_REF')
    raw = Path(ref['path']).read_bytes(); require(hashlib.sha256(raw).hexdigest() == ref['sha256'], 'POLICY_HASH')
    p = json.loads(raw)
    validate_authority(p, clock, purpose)
    s = p['static']; require(Path(s['dso']['path']).resolve() == Path(library).resolve(), 'POLICY_DSO_PATH')
    review_ref = p['independent_admission_review']; raw = Path(review_ref['path']).read_bytes()
    require(hashlib.sha256(raw).hexdigest() == review_ref['sha256'], 'POLICY_REVIEW_HASH')
    review = json.loads(raw)
    require(review['state'] == 'STAGED_NATIVE_LOAD_POLICY_SOURCE_REVIEWED' and review['accepted_by_independent_root'] is True,
            'MISSING_INDEPENDENT_STAGED_LOAD_REVIEW')
    require(review['static_policy_sha256'] == canonical_sha(s) and review['source_manifest_sha256'] == s['source_manifest']['sha256'], 'REVIEW_EXACT_POLICY_TUPLE')
    mref = s['source_manifest']
    require(Path(mref['path']).resolve() == Path(library).resolve().parent/'source-manifest.json', 'POLICY_SOURCE_MANIFEST_PATH')
    raw = Path(mref['path']).read_bytes()
    require(hashlib.sha256(raw).hexdigest() == mref['sha256'], 'POLICY_SOURCE_MANIFEST')
    rows = json.loads(raw)['files']; require(len({r['path'] for r in rows}) == len(rows), 'MANIFEST_DUPLICATES')
    require(str(Path(__file__).resolve()) in {str(Path(r['path']).resolve()) for r in rows}, 'LOADER_NOT_IN_REVIEWED_MANIFEST')
    for row in rows:
        raw = Path(row['path']).read_bytes()
        require(len(raw) == row['bytes'] and hashlib.sha256(raw).hexdigest() == row['sha256'], 'MANIFEST_PAYLOAD_DRIFT:' + row['path'])
    release = Path('/proc/sys/kernel/osrelease').read_text().strip()
    require(release == s['kernel_release'], 'POLICY_KERNEL_RELEASE')
    require(hashlib.sha256(Path('/boot/config-' + release).read_bytes()).hexdigest() == s['kernel_config_sha256'], 'POLICY_KERNEL_CONFIG')
    hooks = s['load_time_review']
    require(hooks['accepted_by_independent_root'] is True and hooks['dso_sha256'] == s['dso']['sha256']
            and hooks['init_fini_reviewed'] is True and hooks['dependencies_already_loaded_only'] is True
            and hooks['null_weak_before_load'] == list(NULL_WEAK) and hooks['IFUNC_resolvers_reviewed'] == ['memcmp'], 'LOAD_TIME_EFFECTS_NOT_REVIEWED')
    return p

class _Function:
    def __init__(self, owner, name, function):
        object.__setattr__(self, '_owner', owner); object.__setattr__(self, '_name', name); object.__setattr__(self, '_function', function)
    def __getattr__(self, name):
        require(name in ('argtypes', 'restype'), 'RAW_FUNCTION_ACCESS'); return getattr(self._function, name)
    def __setattr__(self, name, value):
        require(name in ('argtypes', 'restype'), 'RAW_FUNCTION_MUTATION'); setattr(self._function, name, value)
    def __call__(self, *args):
        self._owner.require_current(cleanup=self._name in CLEANUP_APIS)
        self._owner._api_call_attempts += 1
        return self._function(*args)


class AdmittedNative:
    """One retained PyDLL and actual relocation observation, never a cache label."""
    def __init__(self, library, *, policy_ref, clock, caller_lock, evidence_path, purpose):
        self._valid = False; self._files = {}; self._mem = None; self._functions = {}
        self._api_call_attempts = 0; self._load_attempted = False
        self._library = str(Path(library).resolve()); self._clock = dict(clock)
        self._caller = dict(caller_lock); self._evidence = Path(evidence_path)
        self._bound = []; self._loaded = None; self._stage = 'STATIC_POLICY'
        try:
            self._policy = verify_static_policy(library, policy_ref, clock, purpose)
            self._context()
            require(sys.implementation.name == 'cpython' and sys.version_info[:3] == (3, 14, 7)
                    and sys._is_gil_enabled() and sys.flags.isolated and sys.flags.no_site
                    and sys.dont_write_bytecode and not sys.flags.optimize, 'QUALIFIED_INTERPRETER_FLAGS')
            require(not any(k.startswith('LD_') or k.startswith('GLIBC_TUNABLES') for k in os.environ), 'LOADER_ENVIRONMENT')
            s = self._policy['static']
            for row in s['namespace_files'] + [s['dso']]: self._pin(row)
            require(Path('/proc/self/exe').resolve() in [Path(r['path']) for r in s['namespace_files']], 'CURRENT_INTERPRETER_NOT_PINNED')
            self._image = self._files[self._library]['elf']
            require(static_elf_policy(self._image) == s['elf'] and not self._image['forbidden_dynamic']
                    and self._image['needed'] == ['libc.so.6'], 'EXACT_REVIEWED_ELF_LAYOUT')
            self._maps = self._read_maps()
            require(not any(self._is_file(m, self._files[self._library]) for m in self._maps), 'SECOND_OR_PREEXISTING_CANDIDATE_LOAD_REFUSED')
            self._mem = os.open('/proc/self/mem', os.O_RDONLY | os.O_CLOEXEC)
            self._bootstrap()  # Pinned glibc inspection functions, NOT candidate APIs.
            self._namespace(False)
            for name in NULL_WEAK:
                self._dl.dlerror()
                require(not self._dl.dlsym(None, name.encode()), 'LOAD_TIME_WEAK_HOOK_RESOLVED:' + name)
            self._stage = 'STAGED_LOAD_UNOBSERVED'
            self._check_files()
            self._context()
            self._load_attempted = True
            self._loaded = ctypes.PyDLL('/proc/self/fd/' + str(self._files[self._library]['fd']), mode=os.RTLD_NOW | os.RTLD_LOCAL)
            self._stage = 'LOADED_BINDING_UNVERIFIED'
            self._check_files()
            self._maps = self._read_maps(); self._namespace(True)
            self._base, self._link = self._info(self._loaded._handle)
            require(self._base == self._base_for(self._library), 'LOADED_OBJECT_BIAS')
            prefix = struct.unpack('<QQQQQ', self._read(self._link, 40))
            require(prefix[2] == self._base + self._image['dynamic_address'], 'LOADED_OBJECT_DYNAMIC_IDENTITY')
            self._observe_relocations()
            self._validate_exports()
            self._namespace_bound = list(self._namespace_rows)
            self._context(); self._stage = 'ACTUAL_CURRENT_GOT_PROVIDERS_BOUND_BEFORE_CANDIDATE_APIS'
            require(self._api_call_attempts == 0, 'CANDIDATE_API_BEFORE_ADMISSION')
            self._record(True); self._valid = True
        except BaseException as error:
            self._valid = False
            try:
                self._record(False, repr(error))
            except BaseException as record_error:
                error.add_note('ADMISSION_DIAGNOSTIC_WRITE_FAILED:' + repr(record_error))
            finally:
                self.release_inspection_fds()
            raise

    def _pin(self, row):
        path = str(Path(row['path']).resolve()); require(path == row['path'] and path not in self._files, 'CANONICAL_UNIQUE_PROVIDER_PATH')
        fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_CLOEXEC)
        try:
            before = os.fstat(fd); require(stat.S_ISREG(before.st_mode), 'PROVIDER_NOT_REGULAR')
            with os.fdopen(os.dup(fd), 'rb') as stream: raw = stream.read()
            require(identity(before) == identity(os.fstat(fd)) and len(raw) == row['bytes'] and hashlib.sha256(raw).hexdigest() == row['sha256'], 'OPENED_PROVIDER_BYTES_CHANGED')
            self._files[path] = dict(fd=fd, identity=identity(before), sha256=row['sha256'], elf=elf_image(raw, candidate=path == self._library), path=path)
        except BaseException:
            os.close(fd); raise

    def _context(self, cleanup=False):
        c = self._clock; start, end = c['started_monotonic'], c['deadline_monotonic']
        require(type(c['generation']) is int and c['generation'] > 0 and c['assignment']
                and c['budget_seconds'] == 180 and c['cpu_seconds'] == 45
                and all(type(v) in (int, float) and math.isfinite(v) for v in (start, end))
                and end - start == 180 and start <= time.monotonic(), 'ORIGINAL_CLOCK')
        require(cleanup or time.monotonic() < end, 'ORIGINAL_DEADLINE_NO_FRESH_ADMISSION')
        now = observe_caller_lock(self._caller['fd'], c)
        self._last_caller_observation = now
        validate_caller_tuple(self._caller, now)
        require(len(list(Path('/proc/self/task').iterdir())) == 1, 'CONCURRENT_LOADER_THREAD_REFUSED')

    def _read_maps(self):
        rows = []
        for line in Path('/proc/self/maps').read_text().splitlines():
            fields = line.split(None, 5); a, b = fields[0].split('-'); dev = fields[3].split(':')
            rows.append(dict(start=int(a, 16), end=int(b, 16), permissions=fields[1], offset=int(fields[2], 16),
                             dev=os.makedev(int(dev[0], 16), int(dev[1], 16)), inode=int(fields[4]), path=fields[5] if len(fields) == 6 else ''))
        return rows

    def _is_file(self, mapping, pinned):
        return (mapping['dev'], mapping['inode']) == tuple(pinned['identity'][:2])

    def _base_for(self, path):
        p = self._files[path]; first = min(x[3] for x in p['elf']['loads']) & ~4095
        bases = {m['start'] - first for m in self._maps if m['offset'] == 0 and self._is_file(m, p)}
        require(len(bases) == 1, 'MISSING_OR_DUPLICATE_MAPPED_PROVIDER:' + path)
        return bases.pop()

    def _read(self, address, size):
        require(type(address) is int and address > 0 and 0 < size <= 4096
                and any('r' in m['permissions'] and m['start'] <= address and address + size <= m['end'] for m in self._maps), 'UNREADABLE_LOADED_ADDRESS')
        raw = os.pread(self._mem, size, address); require(len(raw) == size, 'SHORT_LOADED_READ'); return raw

    def _cstring(self, address):
        raw = bytearray()
        for i in range(4096):
            byte = self._read(address + i, 1)
            if byte == b'\0': return raw.decode()
            raw.extend(byte)
        raise RuntimeError('NATIVE_PROVIDER_ADMISSION:UNTERMINATED_LINK_MAP_NAME')

    def _bootstrap(self):
        self._dl = ctypes.PyDLL(None)
        libc_path = self._policy['static']['inspection_libc']
        libc = self._files[libc_path]; base = self._base_for(libc_path)
        signatures = {'dlinfo': ((ctypes.c_void_p, ctypes.c_int, ctypes.c_void_p), ctypes.c_int),
                      'dladdr1': ((ctypes.c_void_p, ctypes.POINTER(_DlInfo), ctypes.POINTER(ctypes.c_void_p), ctypes.c_int), ctypes.c_int),
                      'dlvsym': ((ctypes.c_void_p, ctypes.c_char_p, ctypes.c_char_p), ctypes.c_void_p),
                      'dlsym': ((ctypes.c_void_p, ctypes.c_char_p), ctypes.c_void_p),
                      'dlerror': ((), ctypes.c_char_p)}
        for name, (args, result) in signatures.items():
            f = getattr(self._dl, name); sy = definition(libc['elf'], name, None)
            require(sy['type'] == 2 and ctypes.cast(f, ctypes.c_void_p).value == base + sy['value'], 'UNTRUSTED_INSPECTION_FUNCTION:' + name)
            f.argtypes = args; f.restype = result

    def _info(self, handle):
        pointer = ctypes.c_void_p(); lmid = ctypes.c_long()
        require(self._dl.dlinfo(handle, 2, ctypes.byref(pointer)) == 0 and pointer.value, 'RTLD_DI_LINKMAP')
        require(self._dl.dlinfo(handle, 1, ctypes.byref(lmid)) == 0 and lmid.value == 0, 'RTLD_DI_LMID_BASE_REQUIRED')
        return struct.unpack('<QQQQQ', self._read(pointer.value, 40))[0], pointer.value

    def _namespace(self, loaded):
        _, head = self._info(self._dl._handle)
        require(struct.unpack('<QQQQQ', self._read(head, 40))[4] == 0, 'MAIN_LINK_MAP_HEAD')
        allowed = {r['path'] for r in self._policy['static']['namespace_files']}
        if loaded: allowed.add(self._library)
        observed = []; seen = set(); previous = 0; node = head
        while node:
            require(node not in seen and len(seen) < 64, 'LINK_MAP_CYCLE_OR_BOUND'); seen.add(node)
            base, name, _, nxt, prev = struct.unpack('<QQQQQ', self._read(node, 40))
            require(prev == previous, 'LINK_MAP_CHAIN_CHANGED')
            text = self._cstring(name) if name else ''
            if text == 'linux-vdso.so.1':
                require(any(m['path'] == '[vdso]' and m['start'] == base and 'x' in m['permissions'] for m in self._maps), 'VDSO_MAPPING')
                path = text
            else:
                path = str(Path(text or '/proc/self/exe').resolve())
                require(path in allowed and base == self._base_for(path), 'UNKNOWN_NAMESPACE_OBJECT:' + path)
            observed.append(dict(path=path, base=base, link_map=node)); previous, node = node, nxt
        require({r['path'] for r in observed if r['path'] != 'linux-vdso.so.1'} == allowed, 'EXPECTED_NAMESPACE_NOT_EXACT')
        self._namespace_rows = observed
        for m in self._maps:
            if 'x' not in m['permissions'] or m['path'] in ('[vdso]', '[vsyscall]'): continue
            require(any(path in allowed and self._is_file(m, self._files[path]) for path in self._files), 'UNKNOWN_EXECUTABLE_MAPPING')

    def _observe_relocations(self):
        policy_rows = self._policy['static']['relocation_providers']
        require(len(policy_rows) == len(self._image['imports']) and len({r['offset'] for r in policy_rows}) == len(policy_rows), 'INCOMPLETE_RELOCATION_POLICY')
        self._provider_handles = {}
        for imp, expected in zip(self._image['imports'], policy_rows):
            require(all(expected[k] == imp[k] for k in ('offset', 'symbol', 'version')), 'RELOCATION_POLICY_ORDER')
            slot = self._base + imp['offset']
            require(any(self._is_file(m, self._files[self._library]) and m['start'] <= slot and slot + 8 <= m['end'] for m in self._maps), 'GOT_NOT_CANDIDATE_MAPPING')
            target = struct.unpack('<Q', self._read(slot, 8))[0]
            row = {k: expected[k] for k in ('offset', 'symbol', 'version', 'provider', 'provider_sha256', 'definition_type')}
            if expected['provider'] is None:
                require(imp['bind'] == 2 and imp['symbol'] in NULL_WEAK, 'UNREVIEWED_NULL_WEAK')
                row.update(target=target, expected_target=0, provider_link_map=None, expected_link_map=None,
                           target_executable=False, method='ACTUAL_GOT_NULL_WEAK')
            else:
                path = expected['provider']; pinned = self._files[path]
                require(pinned['sha256'] == expected['provider_sha256'], 'PROVIDER_POLICY_HASH')
                version = imp['version'][1] if imp['version'] else None
                sy = definition(pinned['elf'], imp['symbol'], version)
                require(sy['type'] == expected['definition_type'], 'PROVIDER_SYMBOL_KIND')
                if path not in self._provider_handles:
                    # The main executable already has its public process handle.
                    # NOLOAD cannot initialize a new libc/dependency.
                    self._provider_handles[path] = (self._dl if path == str(Path('/proc/self/exe').resolve())
                        else ctypes.PyDLL(path, mode=os.RTLD_NOLOAD | os.RTLD_NOW | os.RTLD_LOCAL))
                handle = self._provider_handles[path]
                provider_base, provider_link = self._info(handle._handle)
                require(provider_base == self._base_for(path), 'PROVIDER_HANDLE_NOT_EXPECTED_OBJECT')
                wanted = provider_base + sy['value']
                method = 'ACTUAL_GOT_EXACT_VERSIONED_DEFINITION'
                if sy['type'] == 10:
                    require(imp['symbol'] == 'memcmp' and version is not None, 'UNREVIEWED_IFUNC')
                    self._dl.dlerror(); wanted = self._dl.dlvsym(handle._handle, imp['symbol'].encode(), version.encode())
                    require(wanted and self._dl.dlerror() is None, 'IFUNC_VERSION_LOOKUP_FAILED')
                    method = 'ACTUAL_GOT_IFUNC_DLVSYM_SAME_PROVIDER'
                info = _DlInfo(); link = ctypes.c_void_p()
                require(self._dl.dladdr1(target, ctypes.byref(info), ctypes.byref(link), 2) != 0, 'ACTUAL_GOT_PROVIDER_UNKNOWN')
                executable = any(self._is_file(m, pinned) and 'x' in m['permissions'] and m['start'] <= target < m['end'] for m in self._maps)
                row.update(target=target, expected_target=int(wanted), provider_link_map=link.value,
                           expected_link_map=provider_link, target_executable=executable, method=method)
            validate_binding_record(expected, row); self._bound.append(row)
        for r in self._image['relatives']:
            require(struct.unpack('<Q', self._read(self._base + r['offset'], 8))[0] == self._base + r['addend'], 'OWN_RELATIVE_RELOCATION_MISMATCH')
        self._maps = self._read_maps(); self._namespace(True)

    def _record(self, admitted, error=None):
        value = dict(schema='friday.actual-caller-native-provider-binding.v1',
                     state=self._stage, admission_passed=admitted, error=error,
                     load_attempted=self._load_attempted, load_returned_handle=self._loaded is not None,
                     candidate_api_call_attempts_before_record=self._api_call_attempts,
                     caller=current_caller(), caller_quality_requirement=self._caller,
                     caller_quality_observation=getattr(self, '_last_caller_observation', None),
                     original_clock=self._clock, library=self._library,
                     binary_sha256=self._files.get(self._library, {}).get('sha256'),
                     pinned_file_identities={k: {'identity': v['identity'], 'sha256': v['sha256']} for k, v in self._files.items()},
                     policy_static_sha256=canonical_sha(self._policy['static']) if hasattr(self, '_policy') else None,
                     observed_monotonic=time.monotonic(), relocations=self._bound,
                     namespace=getattr(self, '_namespace_rows', []), runtime_acceptance=False, retry_authorized=False,
                     loaded_handle_retained_until_same_caller_exit=self._loaded is not None)
        with self._evidence.open('x') as stream:
            json.dump(value, stream, indent=2); stream.write('\n'); stream.flush(); os.fsync(stream.fileno())

    def require_current(self, library=None, deadline=None, cleanup=False):
        require(self._valid, 'UNADMITTED_OR_REVOKED_HANDLE')
        try:
            require(library is None or str(Path(library).resolve()) == self._library, 'CACHED_WRONG_DSO')
            require(deadline is None or deadline == self._clock['deadline_monotonic'], 'CACHED_WRONG_CLOCK')
            self._context(cleanup)
            self._check_files()
            self._maps = self._read_maps(); self._namespace(True)
            require(self._namespace_rows == self._namespace_bound, 'BOUND_NAMESPACE_CHANGED')
            require(self._info(self._loaded._handle) == (self._base, self._link), 'LOADED_HANDLE_CHANGED')
            for row in self._bound:
                require(struct.unpack('<Q', self._read(self._base + row['offset'], 8))[0] == row['target'], 'BOUND_GOT_CHANGED')
            self._validate_exports()
        except BaseException:
            self._valid = False  # No API, even cleanup, after binding authority is lost.
            raise
        return self

    def _check_files(self):
        for path, pinned in self._files.items():
            require(identity(os.fstat(pinned['fd'])) == pinned['identity']
                    and identity(os.stat(path)) == pinned['identity'], 'PINNED_FILE_OR_PATH_DRIFT')

    def _validate_exports(self):
        for name, value in self._image['exports'].items():
            function = getattr(self._loaded, name)
            require(ctypes.cast(function, ctypes.c_void_p).value == self._base + value, 'CANDIDATE_EXPORT_INTERPOSED')

    def __getattr__(self, name):
        if name.startswith('_'): raise AttributeError(name)
        require(self._valid and name in self._image['exports'], 'UNREVIEWED_CANDIDATE_API')
        if name not in self._functions: self._functions[name] = _Function(self, name, getattr(self._loaded, name))
        return self._functions[name]

    def release_inspection_fds(self):
        self._valid = False
        if self._mem is not None: os.close(self._mem); self._mem = None
        for pinned in self._files.values():
            if pinned['fd'] is not None: os.close(pinned['fd']); pinned['fd'] = None
        # No explicit dlclose: reviewed fini and loaded state belong to the
        # caller through exit. No unload/reload/reset of an attempted tuple.


def load_native(library, *, policy_ref, clock, caller_lock, evidence_path, purpose):
    return AdmittedNative(library, policy_ref=policy_ref, clock=clock,
                          caller_lock=caller_lock, evidence_path=evidence_path, purpose=purpose)
