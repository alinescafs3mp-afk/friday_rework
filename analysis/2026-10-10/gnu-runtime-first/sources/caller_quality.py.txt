# Inert analytical source export; no execution or deployment grant.
"""GNU-only read-side proof of the existing caller's retained quality flock.

No lock acquisition, FD transfer, clock creation or release in this module.
The protected launch record is attached to the exact root permit and original
clock; live proc identity and two kernel lock views must independently agree.
"""
import hashlib
import json
import os
from pathlib import Path
import stat
import time

LOCK = Path('/PRIVATE_RUNTIME/friday-quality-ui-exclusive.lock')
LOCK_ID = [PUBLIC_LOCK_DEV, PUBLIC_LOCK_INODE, PUBLIC_LOCK_UID, PUBLIC_LOCK_GID, 1, 0, 0o664]
SCHEMA = 'friday.gnu-caller-quality-custody.v1'
STATE = 'CALLER_QUALITY_CUSTODY_HELD_UNTIL_CALLER_EXIT'
RELEASE = 'NOT_OBSERVED_REQUIRES_EXTERNAL_POST_CALLER_EXIT_REACQUISITION'


def digest_bytes(data):
    return hashlib.sha256(data).hexdigest()


def clock_digest(clock):
    return digest_bytes(json.dumps(clock, sort_keys=True, separators=(',', ':'),
                                  allow_nan=False).encode())


def file_identity(st):
    assert stat.S_ISREG(st.st_mode), 'QUALITY_NOT_REGULAR'
    return [st.st_dev, st.st_ino, st.st_uid, st.st_gid, st.st_nlink,
            st.st_size, stat.S_IMODE(st.st_mode)]


def caller_identity(pid):
    assert type(pid) is int and pid > 1
    raw = Path('/proc/'+str(pid)+'/stat').read_text()
    assert int(raw.split(' ', 1)[0]) == pid
    tail = raw[raw.rfind(')')+2:].split()
    assert tail[0] not in ('Z', 'X', 'x'), 'CALLER_NOT_LIVE'
    return {'pid': pid, 'start_ticks': int(tail[19]),
            'boot_id': Path('/proc/sys/kernel/random/boot_id').read_text().strip()}


def protected_record(path):
    path = Path(path)
    assert path.is_absolute() and path.resolve() == path
    parent = path.parent.stat()
    assert parent.st_uid == os.getuid() and stat.S_IMODE(parent.st_mode) == 0o700
    fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_CLOEXEC)
    try:
        before = os.fstat(fd)
        assert stat.S_ISREG(before.st_mode) and before.st_nlink == 1
        assert before.st_uid == os.getuid() and stat.S_IMODE(before.st_mode) == 0o600
        assert 0 < before.st_size <= 65536
        data = os.read(fd, 65537)
        assert len(data) == before.st_size
        after = os.fstat(fd)
        assert (before.st_dev, before.st_ino, before.st_size, before.st_mtime_ns,
                before.st_ctime_ns) == (after.st_dev, after.st_ino, after.st_size,
                                      after.st_mtime_ns, after.st_ctime_ns)
        named = path.lstat()
        assert (named.st_dev, named.st_ino) == (before.st_dev, before.st_ino)
        assert stat.S_ISREG(named.st_mode) and named.st_nlink == 1
        return json.loads(data), digest_bytes(data)
    finally:
        os.close(fd)


def verify_contract(here, admission):
    here = Path(here).resolve()
    contract = admission['caller_quality_custody']
    assert contract['schema'] == SCHEMA
    assert contract['lock_path'] == str(LOCK) and contract['lock_identity'] == LOCK_ID
    assert contract['binding_path'] == str(here.parent.parent/'launch-binding.json')
    assert contract['holder'] == 'EXISTING_CALLER_SAME_PID_EXEC_UNTIL_EXIT'
    assert contract['native_child_fd_transfer'] is False
    assert contract['release_evidence'] == RELEASE
    assert admission['root_launcher']['path'] == str(here/'root_gnu_launch.py')
    assert digest_bytes((here/'root_gnu_launch.py').read_bytes()) == admission['root_launcher']['sha256']
    return contract


def _lock_rows(text, pid, identity, fdinfo=False):
    # Require exactly one unambiguous whole-file FLOCK WRITE, owned by the
    # original caller, on this device/inode. Blocked/foreign/other lock types
    # on this file cannot be promoted to evidence of our retained flock.
    rows = []
    key = (os.major(identity[0]), os.minor(identity[0]), identity[1])
    for line in text.splitlines():
        if fdinfo:
            if not line.startswith('lock:'):
                continue
            line = line.partition(':')[2].strip()
        words = line.split()
        matches = []
        for token in words:
            parts = token.split(':')
            if len(parts) == 3:
                try:
                    if (int(parts[0], 16), int(parts[1], 16), int(parts[2])) == key:
                        matches.append(token)
                except ValueError:
                    pass
        if not matches:
            if fdinfo:
                raise RuntimeError('FOREIGN_OR_MALFORMED_FD_LOCK')
            continue
        assert len(matches) == 1 and len(words) == 8, 'AMBIGUOUS_QUALITY_LOCK'
        assert words[1:4] == ['FLOCK', 'ADVISORY', 'WRITE'], 'NOT_CALLER_FLOCK_WRITE'
        assert words[4] == str(pid) and words[5] == matches[0], 'FOREIGN_QUALITY_HOLDER'
        assert words[6:] == ['0', 'EOF'], 'NOT_WHOLE_FILE_FLOCK'
        rows.append(words[1:])
    assert len(rows) == 1, 'MISSING_OR_AMBIGUOUS_CALLER_FLOCK'
    return rows[0]


def validate_quality_custody(here, admission, owner_binding=None):
    """Caller validates itself; owner validates its exact still-live parent."""
    here = Path(here).resolve()
    contract = verify_contract(here, admission)
    if owner_binding is None:
        pid = os.getpid()
        expected = caller_identity(pid)
    else:
        pid = owner_binding['caller_pid']
        assert pid == os.getppid(), 'NOT_ORIGINAL_CALLER_PARENT'
        expected = {'pid': pid, 'start_ticks': owner_binding['caller_identity']['start_ticks'],
                    'boot_id': admission['original_clock']['boot_id']}
        assert owner_binding['caller_identity']['pid'] == pid
    assert caller_identity(pid) == expected
    record, record_sha = protected_record(contract['binding_path'])
    assert caller_identity(pid) == expected
    assert record['schema'] == SCHEMA and record['state'] == STATE
    assert record['caller_identity'] == expected, 'STALE_OR_FOREIGN_LAUNCH_RECORD'
    assert record['assignment'] == admission['assignment']
    assert record['generation'] == admission['generation'] == 1
    assert record['qualification_directory'] == str(here)
    assert record['source_manifest_sha256'] == admission['source_manifest_sha256']
    assert digest_bytes((here/'source-manifest.json').read_bytes()) == admission['source_manifest_sha256']
    assert record['root_launcher'] == admission['root_launcher']
    assert record['permit_path'] == admission['permit_path']
    permit_data = Path(admission['permit_path']).read_bytes()
    assert json.loads(permit_data) == admission
    assert record['permit_sha256'] == digest_bytes(permit_data)
    assert record['original_clock'] == admission['original_clock']
    assert record['original_clock_sha256'] == clock_digest(admission['original_clock'])
    assert expected['boot_id'] == admission['original_clock']['boot_id']
    assert (admission['original_clock']['started_monotonic'] <= record['monotonic_before_exec']
            <= time.monotonic() < admission['original_clock']['deadline_monotonic'])
    assert record['lock_path'] == str(LOCK) and record['lock_identity'] == LOCK_ID
    assert record['new_supervisor'] is False and record['new_budget'] is False
    fd = record['quality_lock_fd']
    assert type(fd) is int and fd >= 3
    argv = [admission['interpreter'], '-I', '-B', '-S',
            str(here/'invoke_qualification.py'), '--gnu-primitive', admission['permit_path']]
    assert record['argv'] == argv
    proc = Path('/proc')/str(pid)
    assert caller_identity(pid) == expected
    actual_argv = [v.decode() for v in (proc/'cmdline').read_bytes().split(b'\0') if v]
    assert actual_argv == argv, 'CALLER_SAME_PID_EXEC_ARGV_MISMATCH'
    assert digest_bytes((proc/'exe').read_bytes()) == admission['interpreter_sha256']
    assert caller_identity(pid) == expected
    fdpath = proc/'fd'/str(fd)
    assert os.readlink(fdpath) == str(LOCK)
    assert file_identity(LOCK.lstat()) == file_identity(fdpath.stat()) == LOCK_ID
    info = (proc/'fdinfo'/str(fd)).read_text()
    fields = [line.split(':', 1)[1].strip() for line in info.splitlines() if line.startswith('ino:')]
    assert fields == [str(LOCK_ID[1])], 'FDINFO_INODE_MISMATCH'
    flags = [line.split(':', 1)[1].strip() for line in info.splitlines() if line.startswith('flags:')]
    assert len(flags) == 1 and not (int(flags[0], 8) & os.O_CLOEXEC)
    assert _lock_rows(info, pid, LOCK_ID, fdinfo=True) == _lock_rows(Path('/proc/locks').read_text(), pid, LOCK_ID)
    assert file_identity(fdpath.stat()) == file_identity(LOCK.lstat()) == LOCK_ID
    assert caller_identity(pid) == expected, 'CALLER_CHANGED_DURING_QUALITY_READ'
    if owner_binding is not None:
        assert os.getppid() == pid, 'CALLER_PARENT_CHANGED'
        assert owner_binding['caller_quality_binding_sha256'] == record_sha
    return {'state': STATE, 'caller_quality_custody_held': True,
            'caller_identity': expected, 'quality_lock_fd': fd, 'lock_identity': LOCK_ID,
            'launch_binding_sha256': record_sha, 'permit_sha256': record['permit_sha256'],
            'original_clock_sha256': record['original_clock_sha256'],
            'source_manifest_sha256': admission['source_manifest_sha256'],
            'release_evidence': RELEASE}


def consume_quality_receipt(receipt, current):
    assert receipt['caller_quality_custody_held'] is True
    assert receipt['caller_quality_custody'] == current
    assert 'lock_returned' not in receipt and 'quality_lock_returned' not in receipt
    assert receipt['quality_release_evidence'] == RELEASE
