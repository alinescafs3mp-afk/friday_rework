# Inert analytical source export; no execution or deployment grant.
"""Existing-owner event pump: source-only, never called by Sol.

This is joined to the real once-only owner, not a separate observer process.
The parent/GNU/mode credits, original clock and persistent direct custody stay.
Actual caller boundary, built shim and Linux/ptrace/ABI delivery NOT_QUALIFIED.
"""
import ctypes
import math
import os
import resource
import signal
import struct
import time
from pathlib import Path
from finite_custody import Custody, StopUnconfirmed
from gnu_primitive_inputs import mode_args,original_room

SEIZE = 0x4206
GETEVENTMSG = 0x4201
GETSIGINFO = 0x4202
CONT = 7
DETACH = 17
LISTEN = 0x4208
FORK, VFORK, CLONE, EXEC, EXIT, EVENT_STOP = 1, 2, 3, 4, 6, 128
OPTIONS = 2 | 4 | 8 | 16 | 64 | (1 << 20)  # full inheritance + EXITKILL
WALL = 0x40000000
RECORD = struct.Struct('<8s9Q32s')
MAGIC = b'FGNUv1\x00\x00'

def ptrace(request, pid, data=0):
    lib = ctypes.CDLL(None, use_errno=True)
    lib.ptrace.argtypes = [ctypes.c_uint, ctypes.c_int, ctypes.c_void_p,
                          ctypes.c_void_p]
    lib.ptrace.restype = ctypes.c_long
    value = data if isinstance(data, int) else ctypes.cast(data, ctypes.c_void_p)
    ctypes.set_errno(0)
    result = lib.ptrace(request, pid, None, value)
    error = ctypes.get_errno()
    if result == -1 and error:
        raise OSError(error, 'ptrace:' + str(request))
    return result

def children_usage():
    r = resource.getrusage(resource.RUSAGE_CHILDREN)
    return r.ru_utime, r.ru_stime

def status_fields(pid):
    values = {}
    for line in Path('/proc/' + str(pid) + '/status').read_text().splitlines():
        k, _, v = line.partition(':')
        if k in ('Tgid', 'TracerPid', 'PPid'):
            values[k] = int(v.strip())
    if set(values) != {'Tgid', 'TracerPid', 'PPid'}:
        raise StopUnconfirmed('INCOMPLETE_TRACE_IDENTITY')
    return values

class TraceCustody(Custody):
    def __init__(self, proc, owner_pid, proc_stat, proc_children, group_members,
                 digest, deadline, admission, step, gate_write, ready_read):
        super().__init__(proc, owner_pid, proc_stat, proc_children, group_members)
        self.digest = digest
        self.deadline = deadline
        self.admission = admission
        self.step = step
        self.gate_write, self.ready_read = gate_write, ready_read
        os.set_blocking(ready_read, False)
        self.buffer = bytearray()
        self.pipe_eof = False
        self.launcher_ready = self.gnu_hello = self.arm_record = None
        self.seized = self.gnu_exec = self.gnu_detached = False
        self.gnu_fork_stop = self.mode_released = False
        self.mode_pid = None
        self.traces = {}
        self.quarantined = {}
        self.trace_released = set()
        self.trace_events = []
        self.trace_waits = []
        self.bootstrap_events = []
        here = Path(__file__).resolve().parent
        mode = step['mode']
        from owner_step_inputs import MODES
        args = mode_args(here, mode, admission, MODES)
        self.expected_gnu_argv = ['gnutimeout', '--signal=KILL', '--verbose', '60s',
                                  admission['interpreter'], '-I', '-B', '-S', *args]

    def _room(self):
        # Same original180. No new phase clock or shortened healthy drain.
        try:
            original_room(self.deadline, time.monotonic())
        except RuntimeError as error:
            raise StopUnconfirmed(str(error)) from error

    def _read_pipe(self):
        while not self.pipe_eof:
            try:
                block = os.read(self.ready_read, RECORD.size * 4)
            except BlockingIOError:
                break
            except InterruptedError:
                continue
            if not block:
                self.pipe_eof = True
                break
            self.buffer.extend(block)
            if len(self.buffer) > RECORD.size * 3:
                raise StopUnconfirmed('EXTRA_READINESS_BYTES')
            while len(self.buffer) >= RECORD.size:
                fields = RECORD.unpack(self.buffer[:RECORD.size])
                del self.buffer[:RECORD.size]
                magic, pid, ppid, kind, api, timer, clock, signo, duration, arm, nonce = fields
                if magic != MAGIC or nonce.decode('ascii') != self.step['bootstrap_nonce']:
                    raise StopUnconfirmed('READINESS_NONCE_OR_ABI')
                if pid != self.proc.pid or ppid != self.owner_pid:
                    raise StopUnconfirmed('READINESS_WRONG_GNU_IDENTITY')
                record = dict(pid=pid, ppid=ppid, kind=kind, api=api,
                              timer_id=timer, clock_id=clock, signo=signo,
                              duration_ns=duration, armed_monotonic_ns=arm)
                self.bootstrap_events.append(record)
                if kind in (1, 2) and any((api, timer, clock, signo, duration, arm)):
                    raise StopUnconfirmed('HELLO_MUST_NOT_CLAIM_TIMER')
                if kind == 1 and self.launcher_ready is None and not self.seized:
                    self.launcher_ready = record
                elif kind == 2 and self.launcher_ready and self.gnu_hello is None and self.gnu_exec:
                    self.gnu_hello = record
                elif kind == 3 and self.gnu_hello and self.arm_record is None and self.gnu_detached:
                    if api not in (1, 2) or clock != 0 or signo != int(signal.SIGALRM):
                        raise StopUnconfirmed('READINESS_WRONG_TIMER_OR_SIGNAL')
                    if duration != 60_000_000_000 or not (0 < arm <= time.monotonic_ns()):
                        raise StopUnconfirmed('READINESS_WRONG_DURATION_OR_TIME')
                    if api == 2 and timer != 0:
                        raise StopUnconfirmed('ALARM_WRONG_TIMER_IDENTITY')
                    self.arm_record = record
                else:
                    raise StopUnconfirmed('DUPLICATE_OR_UNORDERED_READINESS')
        if self.pipe_eof and (self.buffer or self.arm_record is None):
            raise StopUnconfirmed('EARLY_OR_PARTIAL_READINESS_EOF')

    def _identity(self, pid, natural_parent=None):
        birth = self.proc_stat(pid)
        status = status_fields(pid)
        if natural_parent is not None and status['PPid'] != natural_parent:
            raise StopUnconfirmed('WRONG_NATURAL_PARENT')
        if birth['pgid'] != self.proc.pid or birth['sid'] != self.proc.pid:
            raise StopUnconfirmed('TRACE_OUTSIDE_ORIGINAL_GROUP')
        return {'birth': birth, 'status': status, 'stopped': False,
                'terminal': False, 'released': False}

    def bootstrap(self):
        while time.monotonic() < self.deadline:
            self._read_pipe()
            if self.launcher_ready and not self.seized:
                self._room()
                _, failures = self.discover()
                if failures or self.birth is None:
                    raise StopUnconfirmed('NO_EXACT_LAUNCHER_CUSTODY:' + repr(failures))
                if self.digest('/proc/' + str(self.proc.pid) + '/exe') != self.admission['interpreter_sha256']:
                    raise StopUnconfirmed('BARRIER_NOT_AFTER_LAUNCHER_EXEC')
                identity = self._identity(self.proc.pid, self.owner_pid)
                if identity['status']['TracerPid'] != 0:
                    raise StopUnconfirmed('EXISTING_TRACER')
                # All options atomically supplied with SEIZE, before barrier release.
                ptrace(SEIZE, self.proc.pid, OPTIONS)
                self.traces[self.proc.pid] = identity
                self.seized = True
                if status_fields(self.proc.pid)['TracerPid'] != self.owner_pid:
                    raise StopUnconfirmed('SEIZE_NOT_OBSERVED')
                if os.write(self.gate_write, b'G') != 1:
                    raise StopUnconfirmed('BARRIER_SHORT_RELEASE')
                os.close(self.gate_write)
                self.gate_write = None
                self.bootstrap_events.append({'barrier_released': True, 'full_options': OPTIONS})
            if self.seized:
                self.reap_ready()
                self._read_pipe()
                self._maybe_detach()
                if self.arm_record and not self.mode_released:
                    if self.gnu_reap is not None or not self.owned.get(self.proc.pid, {}).get('authority'):
                        raise StopUnconfirmed('GNU_ALREADY_TERMINAL_BEFORE_FIRST_MODE_CONT')
                    self._room()
                    # arm_ns is sampled AFTER successful original call: an upper
                    # bound on its actual start, not a renewed timer. Rechecking
                    # now+120 is stronger than remaining native interval+60.
                    if self.arm_record['armed_monotonic_ns'] + 120_000_000_000 >= int(self.deadline * 1e9):
                        raise StopUnconfirmed('ARM_OUTSIDE_ORIGINAL_60_PLUS_60_ROOM')
                    item = self.traces[self.mode_pid]
                    if not item['stopped'] or item['terminal'] or item['released']:
                        raise StopUnconfirmed('MODE_NOT_HELD_BEFORE_USERSPACE')
                    current = self.proc_stat(self.proc.pid)
                    if current['start_ticks'] != self.birth['start_ticks'] or current['ppid'] != self.owner_pid:
                        raise StopUnconfirmed('LOST_GNU_BEFORE_MODE_RELEASE')
                    raw = Path('/proc/' + str(self.proc.pid) + '/stat').read_text()
                    if raw[raw.rfind(')')+2:].split()[0] in ('Z', 'X'):
                        raise StopUnconfirmed('GNU_ZOMBIE_BEFORE_MODE_RELEASE')
                    self._room()
                    ptrace(CONT, self.mode_pid, 0)
                    item['stopped'] = False
                    self.mode_released = True
                    self.bootstrap_events.append({'first_mode_CONT': self.mode_pid,
                        'original_remaining_seconds': self.deadline-time.monotonic(),
                        'GNU_untraced': True, 'successful_original_timer_arm': self.arm_record})
                    return
            if self.gnu_reap is not None:
                raise StopUnconfirmed('GNU_TERMINAL_DURING_BOOTSTRAP')
            time.sleep(min(0.002, max(0, self.deadline-time.monotonic())))
        raise StopUnconfirmed('ORIGINAL_DEADLINE_BOOTSTRAP_NOT_RUN')

    def _register(self, parent, child, event):
        if child in self.traces or child in self.trace_released:
            raise StopUnconfirmed('REPEATED_KERNEL_CHILD_IDENTITY')
        item = self._identity(child)
        if item['status']['TracerPid'] != self.owner_pid:
            raise StopUnconfirmed('CHILD_NOT_INHERITED_TRACER')
        item['origin_event'] = {'parent': parent, 'event': event, 'child': child}
        self.traces[child] = item
        if parent == self.proc.pid and event == FORK and self.mode_pid is None:
            if item['status']['PPid'] != self.proc.pid or item['status']['Tgid'] != child:
                raise StopUnconfirmed('GNU_FORK_NOT_EXACT_MODE')
            self.mode_pid = child
            self.gnu_fork_stop = True
        elif event == CLONE and parent in self.traces and item['status']['Tgid'] == self.mode_pid:
            # Real mandatory foreign thread, same process credit. No extra root.
            if self.step['mode'] != 'foreign':
                self._anomaly('UNEXPECTED_THREAD_OUTSIDE_FOREIGN_CONTROL')
        else:
            self._anomaly('UNEXPECTED_FORK_OR_VFORK_NO_NEW_CREDIT')
        if child in self.quarantined:
            stop = self.quarantined.pop(child)
            self._stop(child, stop)

    def _maybe_detach(self):
        if self.gnu_detached or not (self.gnu_fork_stop and self.gnu_hello and self.mode_pid):
            return
        gnu, mode = self.traces[self.proc.pid], self.traces[self.mode_pid]
        if not gnu['stopped'] or not mode['stopped']:
            return
        self._room()
        mappings = Path('/proc/' + str(self.proc.pid) + '/maps').read_text().splitlines()
        paths = {str(Path(row.split(maxsplit=5)[5]).resolve()) for row in mappings
                 if len(row.split(maxsplit=5)) == 6 and row.split(maxsplit=5)[5].startswith('/')
                 and not row.endswith(' (deleted)')}
        for path in ('/lib64/ld-linux-x86-64.so.2', self.admission['gnu_shim']['libc'],
                     self.admission['gnu_shim']['path']):
            if str(Path(path).resolve()) not in paths:
                raise StopUnconfirmed('MISSING_PINNED_GNU_LOADED_PROVIDER')
        # GNU still inside fork, mode held at inherited initial kernel stop.
        # If owner stops BEFORE this call, failed bootstrap retains original180
        # EXITKILL; no timer60 claim is made for that NOT_RUN mode.
        self._room()
        ptrace(DETACH, self.proc.pid, 0)
        gnu['released'] = True
        gnu['stopped'] = False
        self.trace_released.add(self.proc.pid)
        self.gnu_detached = True
        self.bootstrap_events.append({'GNU_DETACH_BEFORE_ARM': self.proc.pid,
                                      'mode_still_held': self.mode_pid})

    def _stop(self, pid, status):
        event, sig = status >> 16, os.WSTOPSIG(status)
        if pid not in self.traces:
            if event != EVENT_STOP:
                raise StopUnconfirmed('UNKNOWN_NOT_AUTO_ATTACHED_STOP')
            if pid in self.quarantined or pid in self.trace_released:
                raise StopUnconfirmed('REPEATED_UNKNOWN_STOP')
            self.quarantined[pid] = status
            return  # No signal/CONT/authority until exact GETEVENTMSG joins it.
        item = self.traces[pid]
        if item['terminal'] or item['released']:
            raise StopUnconfirmed('STOP_AFTER_AUTHORITY_RELEASE')
        current = self.proc_stat(pid)
        if current['start_ticks'] != item['birth']['start_ticks']:
            raise StopUnconfirmed('TRACE_BIRTH_CHANGED')
        item['stopped'] = True
        self.trace_events.append({'pid': pid, 'event': event, 'signal': sig,
                                  'birth': current['start_ticks']})
        if event in (FORK, VFORK, CLONE):
            child = ctypes.c_ulong()
            ptrace(GETEVENTMSG, pid, ctypes.byref(child))
            self._register(pid, child.value, event)
            if pid != self.proc.pid:
                ptrace(CONT, pid, 0)
                item['stopped'] = False
        elif event == EXEC:
            if pid == self.proc.pid:
                if self.gnu_exec or self.digest('/proc/' + str(pid) + '/exe') != self.admission['gnu_sha256']:
                    raise StopUnconfirmed('WRONG_OR_REPEATED_GNU_EXEC')
                argv = [v.decode() for v in Path('/proc/' + str(pid) + '/cmdline').read_bytes().split(b'\0') if v]
                if argv != self.expected_gnu_argv:
                    raise StopUnconfirmed('GNU_SIGNAL_DURATION_OR_ARGV_CHANGED')
                self.gnu_exec = True
            elif pid == self.mode_pid:
                if self.digest('/proc/' + str(pid) + '/exe') != self.admission['interpreter_sha256']:
                    raise StopUnconfirmed('WRONG_MODE_EXEC')
                if self.admission['gnu_shim']['path'] in Path('/proc/' + str(pid) + '/maps').read_text():
                    raise StopUnconfirmed('SHIM_RETAINED_ACROSS_MODE_EXEC')
            else:
                raise StopUnconfirmed('UNEXPECTED_EXEC')
            ptrace(CONT, pid, 0)
            item['stopped'] = False
        elif event == EXIT:
            value = ctypes.c_ulong()
            ptrace(GETEVENTMSG, pid, ctypes.byref(value))
            item['exit_event_status'] = value.value
            ptrace(CONT, pid, 0)
            item['stopped'] = False
        elif event == EVENT_STOP:
            if not item.get('initial_stop_seen'):
                item['initial_stop_seen'] = True
                if pid == self.mode_pid and not self.mode_released:
                    return
                ptrace(CONT, pid, 0)
            elif sig in (signal.SIGSTOP, signal.SIGTSTP, signal.SIGTTIN, signal.SIGTTOU):
                ptrace(LISTEN, pid, 0)  # Real job-control group stop, no forged signal.
            else:
                raise StopUnconfirmed('UNEXPECTED_EVENT_STOP')
            item['stopped'] = False
        elif event == 0:
            info = (ctypes.c_ubyte * 128)()
            try:
                ptrace(GETSIGINFO, pid, ctypes.byref(info))
            except OSError as error:
                if error.errno != 22 or sig not in (signal.SIGSTOP, signal.SIGTSTP, signal.SIGTTIN, signal.SIGTTOU):
                    raise
                ptrace(LISTEN, pid, 0)
            else:
                # Genuine signal-delivery stop, including a real SIGTRAP.
                if struct.unpack_from('<i', bytes(info))[0] != sig:
                    raise StopUnconfirmed('SIGINFO_SIGNAL_MISMATCH')
                ptrace(CONT, pid, sig)
            item['stopped'] = False
        else:
            raise StopUnconfirmed('UNREVIEWED_PTRACE_EVENT')

    def _parent_reap(self, pid, status, usage, delta, identity):
        if pid in self.released:
            raise StopUnconfirmed('PARENT_REAP_TWICE')
        raw = (usage.ru_utime, usage.ru_stime)
        # The real wait has already consumed this child. Cache terminal identity
        # and relinquish numeric signal authority BEFORE any validation failure.
        row = {'pid': pid, 'identity': identity, 'wait_status': status,
               'exit_code': os.waitstatus_to_exitcode(status), 'user': raw[0],
               'system': raw[1], 'rusage': list(usage), 'kernel_usage': True,
               'natural_parent_children_credit': list(delta),
               'includes_only_descendants_actually_waited_by_this_child': True}
        self.rows.append(row)
        self.released.add(pid)
        self.unidentified.discard(pid)
        if pid in self.owned:
            self.owned[pid]['reap'] = row
            self.owned[pid]['authority'] = False
        if pid == self.proc.pid:
            self.gnu_reap = row
            self.proc.returncode = row['exit_code']
        if any(not math.isfinite(v) or v < 0 for v in (*raw, *delta)):
            self._anomaly('MALFORMED_KERNEL_USAGE')
            raise StopUnconfirmed('MALFORMED_KERNEL_USAGE')
        if any(abs(a-b) > 2e-6 for a,b in zip(raw, delta)):
            self._anomaly('PARENT_REAP_CHILDREN_CREDIT_MISMATCH')
            raise StopUnconfirmed('PARENT_REAP_CHILDREN_CREDIT_MISMATCH')

    def reap_ready(self):
        failures = []
        while True:
            before = children_usage()
            # One exclusive thread/reaper. No Popen.poll/wait or other wait pump.
            try:
                pid, status, usage = os.wait4(-1, os.WNOHANG | WALL)
            except InterruptedError:
                continue
            except ChildProcessError:
                break
            after = children_usage()
            if pid == 0:
                break
            delta = tuple(b-a for a,b in zip(before, after))
            if os.WIFSTOPPED(status):
                if any(abs(v) > 2e-6 for v in delta):
                    raise StopUnconfirmed('STOP_UNEXPECTED_CHILD_CPU_CHARGE')
                self._stop(pid, status)
                continue
            if not (os.WIFEXITED(status) or os.WIFSIGNALED(status)):
                raise StopUnconfirmed('UNKNOWN_WAIT_STATUS')
            item = self.traces.get(pid)
            raw = (usage.ru_utime, usage.ru_stime)
            self.trace_waits.append({'pid': pid, 'status': status,
                'raw_rusage_not_automatically_additive': list(usage),
                'owner_children_delta': list(delta)})
            if pid == self.proc.pid:
                identity = self.birth
                if identity is None:
                    self._anomaly('GNU_BIRTH_UNKNOWN')
                if item is not None:
                    item['terminal'] = item['released'] = True
                    self.trace_released.add(pid)
                self._parent_reap(pid, status, usage, delta, identity)
            elif item is not None:
                if item['terminal']:
                    # An EXIT_TRACE notification and later adopted EXIT_DEAD are
                    # distinct kernel waits, not a second trace/process credit.
                    if not any(delta):
                        raise StopUnconfirmed('REPEATED_NONADDITIVE_TERMINAL')
                item['terminal'] = True
                item['released'] = True
                self.trace_released.add(pid)
                if any(not math.isfinite(v) or v < 0 for v in (*raw, *delta)):
                    self._anomaly('MALFORMED_TRACE_KERNEL_USAGE')
                    raise StopUnconfirmed('MALFORMED_TRACE_KERNEL_USAGE_RETAINED')
                if item['status']['Tgid'] != pid:
                    if any(abs(v) > 2e-6 for v in delta):
                        raise StopUnconfirmed('NONLEADER_DOUBLE_CHARGE')
                elif any(delta):
                    # Actual exclusive RUSAGE_CHILDREN increment proves this is
                    # the real-parent reap, not EXIT_TRACE. No PGID inference.
                    self._parent_reap(pid, status, usage, delta, item['birth'])
                elif not any(raw) and (self.gnu_reap is not None or pid in self.owned):
                    self._parent_reap(pid, status, usage, delta,
                                      self.owned.get(pid, {}).get('birth', item['birth']))
                elif not any(raw):
                    self._anomaly('ZERO_USAGE_PARENT_ROLE_UNCONFIRMED')
                # Otherwise EXIT_TRACE: retain raw row, deliberately no charge.
            elif pid in self.owned or pid in self.unidentified:
                self._parent_reap(pid, status, usage, delta,
                                  self.owned.get(pid, {}).get('birth'))
                if pid not in self.owned:
                    self._anomaly('ADOPTED_REAP_BIRTH_UNKNOWN')
            else:
                raise StopUnconfirmed('UNKNOWN_TERMINAL_IDENTITY')
        return failures

    def signal_owned(self):
        failures = super().signal_owned()  # Exact unreaped direct children only.
        for pid, item in self.traces.items():
            if pid == self.proc.pid or item['terminal'] or item['released']:
                continue
            if pid in self.owned and self.owned[pid]['authority']:
                continue  # Already covered by the persistent direct-child path.
            try:
                now = self.proc_stat(pid)
                if now['start_ticks'] != item['birth']['start_ticks'] or status_fields(pid)['TracerPid'] != self.owner_pid:
                    raise StopUnconfirmed('LOST_EXACT_TRACE_SIGNAL_CUSTODY')
                # Our unreaped traced child cannot recycle before this exclusive
                # wait pump consumes its terminal notification. Never killpg.
                os.kill(pid, signal.SIGKILL)
                self.trace_events.append({'pid': pid, 'exact_birth': now['start_ticks'],
                                          'signal': 'SIGKILL_OWNED_UNREAPED_TRACE'})
            except ProcessLookupError:
                self.trace_events.append({'pid': pid, 'signal': 'ESRCH_NOT_CLOSURE'})
            except (OSError, StopUnconfirmed) as error:
                failures.append({'trace_signal': pid, 'error': repr(error)})
        return failures

    def finish(self, deadline, terminate=False):
        if deadline != self.deadline:
            raise StopUnconfirmed('CLOCK_RENEWAL_REFUSED')
        if terminate and self.gate_write is not None:
            os.close(self.gate_write)  # Blocked launcher sees EOF and aborts.
            self.gate_write = None
        super().finish(deadline, terminate)
        if self.quarantined or any(not x['terminal'] and not x['released'] for p,x in self.traces.items() if p != self.proc.pid):
            self.closed = False
            raise StopUnconfirmed('TRACE_KERNEL_HANDLES_NOT_TERMINAL')
        if self.ready_read is not None:
            os.close(self.ready_read)
            self.ready_read = None

    def evidence(self):
        result = super().evidence()
        result.update(bootstrap_events=self.bootstrap_events, trace_events=self.trace_events,
                      trace_waits=self.trace_waits, held_unjoined_stops=self.quarantined,
                      mode_released=self.mode_released, GNU_detached=self.gnu_detached,
                      owned_pipe_fds={'barrier_write':self.gate_write,'readiness_read':self.ready_read},
                      trace_released_no_reacquire=sorted(self.trace_released),
                      source_prepared_not_kernel_qualified=True)
        return result
