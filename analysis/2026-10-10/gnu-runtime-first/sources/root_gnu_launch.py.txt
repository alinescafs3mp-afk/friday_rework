# Inert analytical source export; no execution or deployment grant.
"""Future single GNU launch: existing caller holds quality until its own exit.

Invoke with the pinned interpreter -I -B -S and one exact root-issued permit.
No clock/authority is created here. The false draft refuses before lock/record.
Same-PID exec only; native child close_range(CLOEXEC) is unchanged.
"""
import fcntl
import json
import os
from pathlib import Path
import sys
import time


def main():
    assert len(sys.argv) == 2
    here = Path(__file__).resolve().parent
    sys.path.insert(0, str(here))
    from gnu_primitive_inputs import verify_permit, check_clock
    from caller_quality import (LOCK, LOCK_ID, SCHEMA, STATE, verify_contract,
                                file_identity, caller_identity, clock_digest, digest_bytes)
    admission = verify_permit(here, sys.argv[1])
    contract = verify_contract(here, admission)
    assert admission['root_launcher']['path'] == str(Path(__file__).resolve())
    os.umask(0o077)
    for name in ('run', 'clock.json', 'admission.json', 'owner-binding.json'):
        assert not (here/name).exists(), 'ORIGINAL_ONCE_ONLY_INPUT_ALREADY_CONSUMED'
    binding = Path(contract['binding_path'])
    parent = binding.parent.stat()
    assert parent.st_uid == os.getuid() and parent.st_mode & 0o7777 == 0o700
    assert not binding.exists(), 'LAUNCH_ALREADY_CONSUMED_NO_RETRY'
    identity = caller_identity(os.getpid())
    assert identity['boot_id'] == admission['original_clock']['boot_id']
    fd = os.open(LOCK, os.O_RDONLY | os.O_NOFOLLOW | os.O_CLOEXEC)
    assert fd >= 3
    assert file_identity(os.fstat(fd)) == file_identity(LOCK.lstat()) == LOCK_ID
    fcntl.flock(fd, fcntl.LOCK_EX | fcntl.LOCK_NB)
    os.set_inheritable(fd, True)  # Only across this caller's own exec.
    check_clock(admission['original_clock'], admission)
    assert caller_identity(os.getpid()) == identity
    argv = [admission['interpreter'], '-I', '-B', '-S',
            str(here/'invoke_qualification.py'), '--gnu-primitive', admission['permit_path']]
    permit_data = Path(admission['permit_path']).read_bytes()
    assert json.loads(permit_data) == admission
    record = {'schema': SCHEMA, 'state': STATE, 'caller_identity': identity,
              'assignment': admission['assignment'], 'generation': admission['generation'],
              'qualification_directory': str(here), 'root_launcher': admission['root_launcher'],
              'source_manifest_sha256': admission['source_manifest_sha256'],
              'permit_path': admission['permit_path'], 'permit_sha256': digest_bytes(permit_data),
              'original_clock': admission['original_clock'],
              'original_clock_sha256': clock_digest(admission['original_clock']),
              'monotonic_before_exec': time.monotonic(), 'argv': argv,
              'lock_path': str(LOCK), 'lock_identity': LOCK_ID, 'quality_lock_fd': fd,
              'new_supervisor': False, 'new_budget': False}
    out = os.open(binding, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600)
    with os.fdopen(out, 'w') as stream:
        json.dump(record, stream, indent=2, allow_nan=False)
        stream.write('\n')
        stream.flush()
        os.fsync(stream.fileno())
    dirfd = os.open(binding.parent, os.O_RDONLY | os.O_DIRECTORY | os.O_CLOEXEC)
    try:
        os.fsync(dirfd)
    finally:
        os.close(dirfd)
    assert caller_identity(os.getpid()) == identity
    check_clock(admission['original_clock'], admission)
    # No finally-close/unlock: caller owns this exact flock through its own exit.
    os.execve(admission['interpreter'], argv, {'PATH': '/usr/bin:/bin', 'LANG': 'C', 'LC_ALL': 'C'})


if __name__ == '__main__':
    main()
