"""Fixed, small actual GNU mode; no product import, fork or substitute supervisor."""
import resource
if resource.getrlimit(resource.RLIMIT_AS) != (2 << 30, 2 << 30):
    raise SystemExit('NO_INHERITED_GNU_MODE_AS')
import json
import os
import signal
import sys
import time
from pathlib import Path
HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE))
from owner_step_inputs import check, digest, proc_stat


def main():
    expected = [str(HERE/'gnu_primitive_target.py'), str(HERE/'run'/'gnu-primitive-target.json')]
    # Both actual validation failures occur before check's timer/PDEATHSIG calls.
    negatives = []
    for name, mode, argv in (('wrong-mode-refusal', 'positive', expected),
                            ('wrong-target-argv-refusal', 'gnu-primitive', [*expected, 'extra'])):
        try:
            check(mode, argv)
        except AssertionError as error:
            negatives.append(dict(control=name, refused=True, error=repr(error),
                                  no_timer_or_process_created=True))
        else:
            raise RuntimeError('INVALID_TARGET_INPUT_ACCEPTED')
    joined = check('gnu-primitive', expected)  # Actual envelope, tracer, original timers.
    admission = json.loads((HERE/'admission.json').read_bytes())
    step = joined['step']
    assert os.getppid() == step['gnu_identity']['pid']
    assert not Path('/proc/self/task/'+str(os.getpid())+'/children').read_text().strip()
    assert len(list(Path('/proc/self/task').iterdir())) == 1
    assert not any(key == 'LD_PRELOAD' or key.startswith('FGNU_') for key in os.environ)
    maps = Path('/proc/self/maps').read_text()
    assert admission['gnu_shim']['path'] not in maps
    calls = []
    def trap(signum, frame):
        calls.append(signum)
    signal.signal(signal.SIGTRAP, trap)
    os.kill(os.getpid(), signal.SIGTRAP)  # Genuine traced delivery; owner must forward it.
    assert calls == [signal.SIGTRAP]
    assert time.monotonic() < joined['clock']['deadline_monotonic']
    value = dict(state='GNU_PRIMITIVE_TARGET_OBSERVED_UNACCEPTED', pid=os.getpid(),
                 identity=proc_stat(os.getpid()), GNU_identity=step['gnu_identity'],
                 clock_sha256=digest(HERE/'clock.json'), admission_sha256=digest(HERE/'admission.json'),
                 owner_step_sha256=digest(HERE/'run'/'owner-step-gnu-primitive.json'),
                 source_manifest_sha256=admission['source_manifest_sha256'],
                 strict_environment_verified=True, shim_absent_after_exec=True,
                 actual_environment_keys=sorted(os.environ),
                 actual_inherited_limits=list(resource.getrlimit(resource.RLIMIT_CPU)),
                 actual_affinity=sorted(os.sched_getaffinity(0)), timers=joined['timers'],
                 genuine_SIGTRAP_handler_calls=len(calls), negative_controls=negatives,
                 target_self_cpu_at_record=time.clock_gettime(time.CLOCK_PROCESS_CPUTIME_ID),
                 target_write_exit_cpu='REQUIRE_ACTUAL_GNU_OR_ADOPTED_WAIT4', runtime_acceptance=False)
    with Path(sys.argv[1]).open('x') as stream:
        json.dump(value, stream, indent=2); stream.write('\n')
        stream.flush(); os.fsync(stream.fileno())
    return 0


if __name__ == '__main__':
    raise SystemExit(main())
