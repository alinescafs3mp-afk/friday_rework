def digest(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def verify_refs(value):
    if isinstance(value,dict):
        if 'path' in value and 'sha256' in value:
            p=Path(value['path']);assert digest(p)==value['sha256'],p
            if 'bytes' in value:assert p.stat().st_size==value['bytes'],p
            return 1
        return sum(verify_refs(v) for v in value.values())
    if isinstance(value,list):return sum(verify_refs(v) for v in value)
    return 0
