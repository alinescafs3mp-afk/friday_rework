def read(p, expected=None):
    p = Path(p)
    assert p.is_relative_to(R)
    raw = p.read_bytes()
    digest = hashlib.sha256(raw).hexdigest()
    assert expected is None or digest == expected, (str(p), digest, expected)
    s = p.lstat()
    assert stat.S_ISREG(s.st_mode) and not stat.S_ISLNK(s.st_mode)
    pins[str(p)] = dict(path=str(p), sha256=digest, bytes=len(raw),
                       mode=oct(stat.S_IMODE(s.st_mode)), uid=s.st_uid,
                       gid=s.st_gid, nlink=s.st_nlink)
    return raw

def load(ref):
    return json.loads(read(ref['path'], ref['sha256']))

def check(name, value):
    assert value, name
    cases.append(dict(case=name, status='PASS'))

def reject(name, action):
    try:
        action()
    except (AssertionError, RuntimeError):
        cases.append(dict(case=name, status='PASS_EXPECTED_REFUSAL'))
    else:
        raise AssertionError('expected refusal: ' + name)

def functions(source, wanted, env):
    tree = ast.parse(source)
    nodes = [n for n in tree.body if isinstance(n, (ast.FunctionDef, ast.ClassDef))
             and n.name in wanted]
    assert {n.name for n in nodes} == set(wanted)
    module = ast.Module(body=nodes, type_ignores=[])
    exec(compile(ast.fix_missing_locations(module), '<selected-PURE-input-predicates>', 'exec'), env)
    return env
