"""Existing native routing authorizes the recipient; the saved pin only constrains it."""
from copy import deepcopy


def process_notice_visible(runner, event, source=None):
    from tools.process_registry_authority import event_visible
    from hermes_cli.friday_user_scope import ScopeDenied
    try:
        source = source or runner._build_process_event_source(event)
        if source is None:
            # Scoped raw API self-POST has no final authority envelope yet.
            return event.get("user_authority") is None and event_visible(event)
        if not runner._is_user_authorized_for_source(source):
            return False
        with runner._profile_scope_for_source(source):
            return event_visible(event)
    except (ScopeDenied, ValueError, OSError, TypeError):
        return False


def process_notice_metadata(event):
    if event.get("type") == "async_delegation":
        return {}
    return {"process_authority_event": {"type": event.get("type", "completion"),
            "user_authority": deepcopy(event.get("user_authority"))}}


def admit_process_notice(runner, event, source):
    metadata = getattr(event, "metadata", None) or {}
    if metadata.get("notification_origin") != "process_registry_synthetic":
        return True
    saved = metadata.get("process_authority_event")
    # Old synthetic process events have no pin. The actual recipient context
    # still rejects them in a protected profile; never infer a fresh pin.
    if saved is None:
        saved = {"type": "completion", "user_authority": None}
    if not isinstance(saved, dict) or saved.get("type") == "async_delegation":
        return False
    return process_notice_visible(runner, saved, source)
