def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def check(name, function):
    try:
        result = function()
        assert result is not False, 'predicate returned False'
        rows.append({'name': name, 'status': 'PASS'})
    except Exception as exc:
        rows.append({'name': name, 'status': 'FAIL', 'error': repr(exc)})

def importer(name, globals=None, locals=None, fromlist=(), level=0):
    assert level == 0 and name in modules, ('forbidden subject import', name)
    return modules[name]

def execute(nodes, env, filename):
    env['__builtins__'] = {**vars(builtins), '__import__': importer}
    env.setdefault('__name__', 'inert_selected_source')
    exec(compile(ast.fix_missing_locations(ast.Module(body=nodes, type_ignores=[])), str(filename), 'exec', flags=0x1000000), env)
    return env

def selected(path, names, env=None, cls=None, decorated=False):
    pins[str(path)] = sha(path)
    tree = ast.parse(path.read_text())
    source = tree.body if cls is None else next(n.body for n in tree.body if isinstance(n, ast.ClassDef) and n.name == cls)
    nodes = [copy.deepcopy(n) for n in source if isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)) and n.name in names]
    assert len(nodes) == len(names), names
    if not decorated:
        for n in nodes:
            n.decorator_list = []
    return execute(nodes, {} if env is None else env, path)

    lines=(ROOT/'source.patch').read_text().splitlines(keepends=True)

        out=[];cursor=0

            match=re.fullmatch(r'@@ -(\d+)(?:,(\d+))? \+(\d+)(?:,(\d+))? @@\n',lines[index]);assert match,lines[index]
            old_start,old_count,new_start,new_count=match.groups();start=max(0,int(old_start)-1)
            out.extend(original[cursor:start]);cursor=start;index+=1;minus=plus=0
            while index<len(lines) and not lines[index].startswith(('@@ ','--- ')):

            assert minus==int(old_count or 1) and plus==int(new_count or 1)
        out.extend(original[cursor:]);data=''.join(out).encode();assert data==(NATIVE/name).read_bytes()
        reconstructed[name]=hashlib.sha256(data).hexdigest()
