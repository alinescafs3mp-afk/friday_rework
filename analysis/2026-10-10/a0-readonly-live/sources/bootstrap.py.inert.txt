def need(value,reason):
    if not value:raise RuntimeError(reason)

def capture(reference):
    path=Path(reference['path']);need(path.is_absolute(),'absolute exact input')
    fd=os.open(path,os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK|os.O_CLOEXEC)
    error=None
    try:
        before=os.fstat(fd)
        need(stat.S_ISREG(before.st_mode) and before.st_uid==before.st_gid==PRIVATE_OWNER_UID and
             before.st_nlink==1 and stat.S_IMODE(before.st_mode)==0o600 and
             0<before.st_size<=1048576,'private input metadata')
        body=os.read(fd,1048577);after=os.fstat(fd)
        fields=('st_dev','st_ino','st_mode','st_uid','st_gid','st_nlink','st_size','st_mtime_ns','st_ctime_ns')
        need(all(getattr(before,k)==getattr(after,k) for k in fields) and
             len(body)==before.st_size and hashlib.sha256(body).hexdigest()==reference['sha256'],
             'input changed or hash mismatch')
        return body
    except BaseException as problem:error=problem;raise
    finally:
        try:os.close(fd)
        except BaseException as problem:
            raise RuntimeError('STOP_UNCONFIRMED_INPUT_CLOSE; original='+repr(error)) from problem

def pairs(items):
    result={}
    for key,value in items:need(key not in result,'duplicate input key');result[key]=value
    return result

def save(path,value):
    body=(json.dumps(value,indent=2)+'\n').encode()
    need(len(body)<=65536,'finite owner evidence')
    fd=os.open(path,os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW|os.O_CLOEXEC,0o600)
    try:stream=os.fdopen(fd,'wb')
    except BaseException as original:
        # Raw descriptor is ours until the stream constructor returns.
        try:os.close(fd)
        except BaseException as problem:
            raise RuntimeError('STOP_UNCONFIRMED_EVIDENCE_CLOSE; original='+repr(original)) from problem
        raise
    with stream:stream.write(body);stream.flush();os.fsync(stream.fileno())
    return {'path':str(path),'sha256':hashlib.sha256(body).hexdigest()}

def ticks(pid):
    return int(Path('/proc',str(pid),'stat').read_text().rsplit(')',1)[1].split()[19])
