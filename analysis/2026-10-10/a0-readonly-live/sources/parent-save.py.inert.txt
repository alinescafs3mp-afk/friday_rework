def bootstrap_need(ok, why):
    if not ok:

def digest_pin(pin):
    bootstrap_need(type(pin) is str and len(pin) == 64 and
                   all(c in '0123456789abcdef' for c in pin), 'SHA256 pin')
    return pin

def exact(path, pin, limit=1048576):
    digest_pin(pin)
    fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK | os.O_CLOEXEC)
    try:
        before = os.fstat(fd)
        bootstrap_need(stat.S_ISREG(before.st_mode) and
                       stat.S_IMODE(before.st_mode) == 0o600 and
                       before.st_uid == before.st_gid == PRIVATE_OWNER_UID and
                       before.st_nlink == 1 and 0 < before.st_size <= limit,
                       'regular private owner file metadata/size')
        # One bounded read, no stream loop or reopen. A short read refuses.
        raw = os.read(fd, limit + 1)
        after = os.fstat(fd)
        fields = ('st_dev', 'st_ino', 'st_mode', 'st_uid', 'st_gid',
                  'st_nlink', 'st_size', 'st_mtime_ns', 'st_ctime_ns')
        bootstrap_need(all(getattr(before, k) == getattr(after, k) for k in fields),
                       'same-FD metadata changed during read')
        bootstrap_need(type(raw) is bytes and len(raw) == before.st_size and
                       hashlib.sha256(raw).hexdigest() == pin, 'exact source bytes')
        return raw
    finally:
        os.close(fd)

def unique_pairs(items):
    value = {}
    for key, item in items:
        bootstrap_need(key not in value, 'duplicate JSON field')
        value[key] = item
    return value
