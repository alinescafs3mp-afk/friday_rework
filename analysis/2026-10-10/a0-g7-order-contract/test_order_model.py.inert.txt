import ast

from pathlib import Path

import unittest

from order_model import (NS, STOP_NS, complete_hook_proof, execstop_wait,
                         gated_post_chain, stop_job_orders)

HERE = Path(__file__).resolve().parent
INTAKE = json.loads((HERE / 'INTAKE.json').read_bytes())
PINS = {Path(p['path']): p['sha256'] for p in INTAKE['pins']}
CANDIDATE = Path('/PRIVATE_CONSUMED_G7_CANDIDATE')
DIAG = Path('/PRIVATE_HISTORICAL_DIAGNOSIS')


def pinned(path):
    raw = path.read_bytes()
    if hashlib.sha256(raw).hexdigest() != PINS[path]:
        raise ValueError('input drift: ' + str(path))
    return raw


def consumed_replay():
    """Actual original predicate on original bytes, historical cessation only.

    same_process is replaced with the already recorded cessation fact. This
    deliberately makes NO new /proc query and cannot establish current death.
    """
    data = json.loads(pinned(CANDIDATE / 'native-evidence/native-result.json'))
    recipe = json.loads(pinned(CANDIDATE / 'recipe.json'))
    tree = ast.parse(pinned(CANDIDATE / 'driver.py'))
    fn = next(n for n in tree.body if isinstance(n, ast.FunctionDef)
              and n.name == 'expected_properties')
    cls = next(n for n in tree.body if isinstance(n, ast.ClassDef)
               and n.name == 'Driver')
    methods = [n for n in cls.body if isinstance(n, ast.FunctionDef)
               and n.name in ('select', 'terminal', 'timeout_binding_case')]
    replay = ast.ClassDef(name='Replay', bases=[], keywords=[], body=methods,
                          decorator_list=[])
    module = ast.fix_missing_locations(ast.Module(body=[fn, replay], type_ignores=[]))

    def require(ok, why):
        if not ok:
            raise ValueError(why)

    env = dict(ROOT=CANDIDATE,
               NAMES={r: recipe['prefix']+'-'+r+'.service'
                      for r in data['starts_attempted']},
               re=re, signal=signal, require=require,
               show_argv=lambda: recipe['read_all_argv'],
               MANAGER_SHOW=recipe['read_original_manager_argv'],
               same_process=lambda item:
                   not data['cases']['native-timeout-binding']['setsid_children_ceased'])
    exec(compile(module, 'pinned-driver:selected-pure-AST', 'exec'), env)
    obj = env['Replay']()
    obj.events = copy.deepcopy(data['events'])
    obj.original = data['original_invocations']
    obj.calls = data['manager_calls']
    obj.recipe = recipe
    return obj, data


class OrderTests(unittest.TestCase):
    def test_exact_input_bytes_unchanged(self):
        for p in PINS:
            pinned(p)

    def test_exact_consumed_predicate_stays_fail(self):
        replay, data = consumed_replay()
        result = replay.timeout_binding_case()
        self.assertEqual(result, data['cases']['native-timeout-binding'])
        self.assertEqual(result['status'], 'FAIL')
        self.assertFalse(result['reverse_stop_hook_order'])

    def test_actual_trace_interpretation(self):
        trace = json.loads(pinned(DIAG / 'TRACE.json'))
        outer, dep = (trace['terminal_events'][r]
                      for r in ('outer-timeout', 'dependent-timeout'))
        self.assertEqual(dep['monotonic_ns']-outer['monotonic_ns'], 3_406_665)
        self.assertTrue(all(x['native_capture_valid'] for x in (outer, dep)))
        self.assertEqual(outer['properties']['Job'], '')
        self.assertEqual(dep['properties']['Job'], 'PRIVATE_DEPENDENT_JOB_ID')
        times = trace['case']['native_times_us']
        self.assertEqual(times['dependent_stopped']-times['outer_stopped'], 248)
        self.assertGreaterEqual(times['dependent_exited']-times['dependent_stopped'], 2_000_000)
        self.assertLess(times['dependent_exited'], times['dependent_entered']+12_000_000)

    def test_entry_record_is_not_cleanup_completion(self):
        source = pinned(CANDIDATE / 'observer.py').decode()
        tree = ast.parse(source)
        main = next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == 'main')
        emit = next(n.lineno for n in ast.walk(main) if isinstance(n, ast.Call)
                    and isinstance(n.func, ast.Name) and n.func.id == 'emit')
        sleep = next(n.lineno for n in ast.walk(main) if isinstance(n, ast.Call)
                     and isinstance(n.func, ast.Attribute) and n.func.attr == 'sleep')
        self.assertLess(emit, sleep)
        trace = json.loads(pinned(DIAG / 'TRACE.json'))
        for e in trace['terminal_events'].values():
            self.assertEqual(e['properties']['SubState'], 'stop-post')
            self.assertEqual(e['properties']['ControlPID'], str(e['pid']))

    def test_marker_reorder_cannot_fix_physical_trace(self):
        replay, data = consumed_replay()
        dep = replay.terminal('dependent-timeout')
        original_dep = dep['monotonic_ns']
        outer = replay.terminal('outer-timeout')['monotonic_ns']
        dep['monotonic_ns'] = outer - 1
        self.assertEqual(replay.timeout_binding_case()['status'], 'PASS')
        self.assertGreater(original_dep, outer)
        self.assertFalse(data['cases']['native-timeout-binding']['reverse_stop_hook_order'])

    def test_stop_jobs_have_reverse_partial_order(self):
        orders = stop_job_orders(True)
        self.assertTrue(orders)
        self.assertTrue(all(p.index('dep-post') < p.index('outer-post') for p in orders))

    def test_direct_timeout_permits_both_hook_orders(self):
        orders = stop_job_orders(False)
        self.assertTrue(any(p.index('outer-post') < p.index('dep-post') for p in orders))
        self.assertTrue(any(p.index('dep-post') < p.index('outer-post') for p in orders))

    def test_single_execstop_cannot_ack_before_same_two_second_deadline(self):
        #256 finite traces for each of ENTRY and COMPLETION acknowledgment.
        for entry_only in (True, False):
            for gap, launch, work, outer in itertools.product(
                    (1, 248_000, 10_000_000, 100_000_000),
                    (0, 3_406_665, 50_000_000, 100_000_000),
                    (0, 750_000_000, NS, 1_500_000_000),
                    (0, 1_000_000, 50_000_000, 100_000_000)):
                t = execstop_wait(gap, launch, work, outer,
                                  acknowledge_completion=not entry_only)
                self.assertFalse(t.timely_ack)
                self.assertEqual(t.outer_kill, STOP_NS)

    def test_expiry_provides_concrete_reverse_order_counterexample(self):
        t = execstop_wait(248_000, 50_000_000, 750_000_000, 1_000_000)
        self.assertFalse(t.reverse_entry)
        self.assertFalse(t.reverse_completion_to_entry)
        self.assertEqual(t.outer_hook_entry, 2_001_000_000)
        self.assertEqual(t.dependent_hook_entry, 2_050_248_000)

    def test_accidental_reverse_does_not_prove_wait_worked(self):
        t = execstop_wait(1, 0, 750_000_000, NS)
        self.assertTrue(t.reverse_entry)
        self.assertFalse(t.timely_ack)

    def test_second_post_command_is_not_first_hook_entry(self):
        t = gated_post_chain(2*NS, 2*NS+3_406_665, 2*NS+750_000_000, 4*NS)
        self.assertTrue(t['cleanup_before_second_command'])
        self.assertFalse(t['first_hook_reverse'])
        self.assertFalse(t['physical_hook_entry_accepted'])

    def test_post_gate_failures_do_not_admit_second_command(self):
        for valid, completion in [(False, 3*NS), (True, 4*NS), (True, 5*NS)]:
            t = gated_post_chain(2*NS, 2*NS+1, completion, 4*NS, proof_valid=valid)
            self.assertIsNone(t['second_command_entry'])
            self.assertFalse(t['physical_hook_entry_accepted'])

    def test_completion_receipt_requires_real_bound_facts(self):
        binding = {'attempt':'a'*64, 'boot':'boot-original', 'manager':'manager-original',
                   'dependent_invocation':'b'*32, 'role':'dependent-timeout',
                   'original_deadline_ns':120*NS}
        proof = {k:v for k,v in binding.items() if k!='original_deadline_ns'}
        proof.update(entry_ns=10*NS, completion_ns=11*NS,
                     original_children_ceased=True, original_cgroup_empty=True,
                     hook_process_exited=True, capture_valid=True)
        self.assertTrue(complete_hook_proof(proof, binding))
        for key in ('original_children_ceased','original_cgroup_empty','hook_process_exited','capture_valid'):
            for missing in (False, None, 'UNKNOWN'):
                bad = dict(proof, **{key:missing})
                self.assertFalse(complete_hook_proof(bad, binding))
        for key in ('attempt','boot','manager','dependent_invocation','role'):
            self.assertFalse(complete_hook_proof(dict(proof, **{key:'foreign'}), binding))
        for bad in ({'entry_ns':True}, {'completion_ns':120*NS},
                    {'completion_ns':9*NS}, {'completion_ns':None}, {'unexpected':1}):
            self.assertFalse(complete_hook_proof(dict(proof, **bad), binding))
        self.assertFalse(complete_hook_proof({'entry_ns':1}, binding))

    def test_all_original_scope_and_properties_remain(self):
        _, data = consumed_replay()
        self.assertEqual(len(data['cases']), 6)
        self.assertEqual(len(data['starts_attempted']), 8)
        self.assertEqual(sum(data['starts_attempted'].values()), 9)
        self.assertEqual(sum(x['status']=='PASS' for x in data['cases'].values()), 5)
        recipe = json.loads(pinned(CANDIDATE / 'recipe.json'))
        self.assertEqual([recipe['submission_cutoff_seconds'],recipe['driver_finish_deadline_seconds'],
                          recipe['maximum_whole_grant_seconds']], [35,110,120])
        for role, argv in recipe['native_start_argv'].items():
            self.assertIn('--property=TimeoutStopSec=2s', argv)
            self.assertIn('--property=TimeoutStopFailureMode=kill', argv)
            self.assertIn('--property=KillMode=control-group', argv)
            self.assertIn('--property=SendSIGKILL=yes', argv)
            self.assertIn('--property=FinalKillSignal=SIGKILL', argv)
        self.assertEqual(recipe['dropin_setup']['setup_seconds_from_original_acceptance'], 5)
        self.assertEqual(recipe['bus_transport_source']['preparation_seconds'], 4)
        self.assertEqual(recipe['bus_transport_source']['admission_seconds'], 5)

    def test_primary_source_has_the_required_timeout_bypass(self):
        primary = HERE/'primary'
        pins = json.loads((primary/'PINS.json').read_bytes())
        for p in pins:
            self.assertEqual(hashlib.sha256(Path(p['path']).read_bytes()).hexdigest(),p['sha256'])
        s = (primary/'src-core-service.c').read_text()
        wait = s[s.index('static void service_enter_stop(Service *s, ServiceResult f) {'):]
        wait = wait[:wait.index('\nstatic ', 1)]
        self.assertLess(wait.index('service_spawn(s'), wait.index('service_set_state(s, SERVICE_STOP)'))
        timer = s[s.index('static int service_dispatch_timer(sd_event_source *source, usec_t usec, void *userdata) {'):]
        running = timer[timer.index('case SERVICE_RUNNING:'):timer.index('case SERVICE_REFRESH_EXTENSIONS:')]
        self.assertIn('service_enter_stop(s, SERVICE_FAILURE_TIMEOUT)', running)
        stop = timer[timer.index('case SERVICE_STOP:'):timer.index('case SERVICE_STOP_WATCHDOG:')]
        self.assertIn('case SERVICE_TIMEOUT_KILL:', stop)
        self.assertIn('service_enter_signal(s, SERVICE_STOP_SIGKILL', stop)
        self.assertNotIn('dependent', stop)


if __name__ == '__main__':
    unittest.main(verbosity=2)
