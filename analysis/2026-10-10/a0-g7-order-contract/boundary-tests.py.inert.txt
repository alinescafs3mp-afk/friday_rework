def require(test, message):

def check(name, test, **details):

def functions(path, names):
    tree = ast.parse(path.read_text())
    found = [n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name in names]
    assert {n.name for n in found} == set(names)
    return found

def build(path):
    tree = ast.parse(path.read_text())
    driver = next(n for n in tree.body if isinstance(n, ast.ClassDef) and n.name == 'Driver')
    names = {'select', 'terminal', 'timeout_binding_case', 'ceased', 'snapshot',
             'scan_events', 'remember', 'scenario_validity', 'late_dependency_case', 'evaluate', 'report'}
    methods = [n for n in driver.body if isinstance(n, ast.FunctionDef) and n.name in names]
    assert len(methods) == len(names)
    selected = functions(ROOT / 'common.py', {'same_process', 'absent', 'own_cgroup', 'within'})
    selected += functions(ROOT / 'observer.py', {'late_ready_case'})
    selected += functions(path, {'expected_properties'})
    selected.append(ast.ClassDef(name='Replay', bases=[], keywords=[], body=methods, decorator_list=[]))
    module = ast.fix_missing_locations(ast.Module(body=selected, type_ignores=[]))
    env = dict(ROOT=ROOT, NAMES=NAMES, ROLES=ROLES, re=re, signal=signal,
               require=require, Refused=Refused, read_json=lambda p: json.loads(p.read_text()),
               show_argv=lambda: recipe['read_all_argv'], MANAGER_SHOW=recipe['read_original_manager_argv'],
               MANDATORY_CHECKPOINTS=CHECKPOINTS,
               CHECKPOINT_DEADLINES=dict(zip(CHECKPOINTS, (11, 14, 19, 22, 29, 30))))
    exec(compile(module, str(path) + ':selected-AST-only', 'exec'), env)
    return env

    x.cgroups = {r: set() for r in ROLES}
    for r, records in x.identities.items():
        for record in records:
            for line in record['cgroup'].splitlines():
                if line.startswith('0::'):
                    x.cgroups[r].add(line[3:])
    x.last = copy.deepcopy(data['samples'][-1]['properties'])
    x.last_cgroups = copy.deepcopy(data['samples'][-1]['cgroups'])
    x.samples = [copy.deepcopy(data['samples'][-1])]
    for k, v in data['cessation_evidence'].items():

    x.submission_uncertain = data['submission_uncertain']

    x.issues = []

        and e['monotonic_ns'] > data['cases']['late-dependency']['attempt_interval_ns'][1])
    x.late_started_ns, x.late_observation_finished_ns = data['cases']['late-dependency']['attempt_interval_ns']

def trial(name, mutate, expected='FAIL', method='timeout_binding_case', message=None):
    for reverse in (False, True):
        x = fresh(reverse=reverse)
        mutate(x)
        try:
            result = getattr(x, method)()
            observed = result['status'] if isinstance(result, dict) else result

            observed = 'SAFE_REFUSAL'
            if message:
                assert message in str(exc), (name, str(exc))
        assert observed == expected, (name, reverse, observed, expected)
    CHECKS.append({'name': name, 'status': 'PASS_OFFLINE', 'both_R_orders': True, 'observed': expected})
