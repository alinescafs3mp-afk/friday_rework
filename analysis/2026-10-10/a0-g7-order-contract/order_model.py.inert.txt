"""Inert, integer-clock counterexamples; never a native execution adapter.

The model encodes only the pinned v259 transitions needed to disprove a
guarantee. Scheduling/propagation delays are explicit inputs, not measured
systemd bounds. It does not infer PID/cgroup cessation or authorize effects.
"""
from dataclasses import dataclass
from itertools import permutations

NS = 1_000_000_000
STOP_NS = 2 * NS


@dataclass(frozen=True)
class WaitTrace:
    dependent_term: int
    dependent_kill: int
    dependent_hook_entry: int
    dependent_hook_completion: int
    outer_wait_deadline: int
    timely_ack: bool
    outer_kill: int
    outer_hook_entry: int
    reverse_entry: bool
    reverse_completion_to_entry: bool


def execstop_wait(propagation_ns, dependent_launch_ns, dependent_work_ns,
                  outer_launch_ns, *, acknowledge_completion=True):
    """Outer ExecStop waits; original TimeoutStop2/FailureMode=kill retained.

    Clock zero is the outer ExecStop native timer arm, BEFORE spawn/state
    change/BindsTo propagation. A strictly positive propagation input includes
    that work. Ignore-TERM dependent needs its full original two seconds.
    The successful-spawn path alone suffices for a counterexample.
    """
    values = (propagation_ns, dependent_launch_ns,
              dependent_work_ns, outer_launch_ns)
    if any(type(v) is not int or v < 0 for v in values):
        raise ValueError('nonnegative integer time required')
    if propagation_ns == 0:
        raise ValueError('state change and propagation must follow timer arm')
    dep_kill = propagation_ns + STOP_NS
    dep_entry = dep_kill + dependent_launch_ns
    dep_done = dep_entry + dependent_work_ns
    ack = dep_done if acknowledge_completion else dep_entry
    timely = ack < STOP_NS
    # Successful ExecStop exit resumes normal TERM->KILL after2s. On expiry,
    # pinned kill mode jumps directly to SIGKILL; no extra TERM grace exists.
    outer_kill = ack + STOP_NS if timely else STOP_NS
    outer_entry = outer_kill + outer_launch_ns
    return WaitTrace(propagation_ns, dep_kill, dep_entry, dep_done,
                     STOP_NS, timely, outer_kill, outer_entry,
                     dep_entry <= outer_entry, dep_done <= outer_entry)


def stop_job_orders(outer_job):
    """Finite legal interleavings of TERM/KILL/post-entry for two units.

    After reverses two queued stop jobs. Without the outer job the pinned
    runnable predicate has no other->job to compare. BindsTo propagation
    remains represented by outer TERM before dependent TERM.
    """
    events = ('outer-term', 'outer-kill', 'outer-post',
              'dep-term', 'dep-kill', 'dep-post')
    edges = [('outer-term', 'outer-kill'), ('outer-kill', 'outer-post'),
             ('dep-term', 'dep-kill'), ('dep-kill', 'dep-post')]
    if outer_job:
        edges.append(('dep-post', 'outer-term'))
    else:
        edges.append(('outer-term', 'dep-term'))
    return [p for p in permutations(events)
            if all(p.index(a) < p.index(b) for a, b in edges)]


def gated_post_chain(outer_first_entry, dependent_entry, dependent_done,
                     gate_deadline, *, proof_valid=True):
    """Evaluate a REJECTED-within-current-contract two-command alternative.

    The first outer ExecStopPost synchronization command ALREADY entered.
    A second observer command may be actually spawned after valid completion.
    This can order that second command, never the first hook entry. No marker
    rewrite or successful substitution of one invariant for another occurs.
    """
    if not (0 <= outer_first_entry < gate_deadline
            and 0 <= dependent_entry <= dependent_done):
        raise ValueError('invalid trace')
    observer = (max(outer_first_entry, dependent_done)
                if proof_valid and dependent_done < gate_deadline else None)
    return {'first_hook_reverse': dependent_entry <= outer_first_entry,
            'second_command_entry': observer,
            'cleanup_before_second_command':
                observer is not None and dependent_done <= observer,
            'physical_hook_entry_accepted':
                observer is not None and dependent_entry <= outer_first_entry}


def complete_hook_proof(proof, binding):
    """Pure hypothetical completion-consumer specification, not a producer.

    These facts are deliberately stronger than the current observer's emit.
    No such completion receipt is supplied by the consumed G7 package. A
    future producer needs independent ownership/native evidence qualification.
    """
    if not isinstance(proof, dict) or not isinstance(binding, dict):
        return False
    required = {'attempt', 'boot', 'manager', 'dependent_invocation',
                'role', 'entry_ns', 'completion_ns',
                'original_children_ceased', 'original_cgroup_empty',
                'hook_process_exited', 'capture_valid'}
    if set(proof) != required:
        return False
    if any(proof[k] != binding.get(k) for k in
           ('attempt', 'boot', 'manager', 'dependent_invocation', 'role')):
        return False
    if proof['role'] != 'dependent-timeout':
        return False
    if not all(proof[k] is True for k in
               ('original_children_ceased', 'original_cgroup_empty',
                'hook_process_exited', 'capture_valid')):
        return False
    entry, done = proof['entry_ns'], proof['completion_ns']
    return (type(entry) is int and type(done) is int
            and 0 < entry <= done < binding['original_deadline_ns'])
