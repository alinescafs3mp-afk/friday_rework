"""Pure prospective V2 reviewer decoder. Caller supplies independently pinned bytes.
No file access, execution, cleanup, FD9 operations or admission inference.
A verified inventory is evidence data, never evidence of overall success.
"""
import hashlib
import json

FINAL_CONTRACT = 'SINGLE_FINAL_BUS_INVENTORY_BEFORE_FD9_RELEASE_V2'
REFUSAL_CONTRACT = 'ONCE_FINAL_REFUSAL_AFTER_WITNESS_FD9_OUTCOME_COMPACT_V2'
FINAL_LIMIT = 65536
PHASE_LIMIT = 32 * 1024 * 1024


def need(ok, reason):
    if not ok:
        raise ValueError(reason)


def same_json(left, right):
    # Identical strict comparison to the independently reviewed producer helper.
    return json.dumps(left,sort_keys=True,separators=(',',':'),allow_nan=False) == \
           json.dumps(right,sort_keys=True,separators=(',',':'),allow_nan=False)


def pairs(items):
    result = {}
    for key, value in items:
        need(key not in result, 'duplicate JSON key')
        result[key] = value
    return result


def pin(value):
    need(type(value) is str and len(value) == 64 and
         all(c in '0123456789abcdef' for c in value), 'SHA256 type/value')


def reference(ref):
    need(type(ref) is dict and set(ref) == {'path', 'sha256'}, 'exact reference shape')
    path = ref['path']
    need(type(path) is str and path.startswith('/') and
         all(p not in ('', '.', '..') for p in path.split('/')[1:]), 'canonical absolute path')
    pin(ref['sha256'])
    return path


def parse(raw, digest, limit):
    pin(digest)
    need(type(raw) is bytes and 0 < len(raw) <= limit, 'bounded nonempty bytes')
    need(hashlib.sha256(raw).hexdigest() == digest, 'exact bytes hash')
    value = json.loads(raw, object_pairs_hook=pairs,
                       parse_constant=lambda _: need(False, 'nonfinite JSON'))
    # Reject numeric overflow (e.g. 1e999) as well as NaN tokens.
    same_json(value, value)
    need(type(value) is dict, 'JSON object required')
    return value


def pinned(raw, ref, expected, filename):
    path = reference(ref)
    reference(expected)
    need(same_json(ref, expected), 'reference differs from caller pin')
    need(path.rsplit('/', 1)[-1] == filename, 'wrong evidence filename')
    return parse(raw, ref['sha256'], FINAL_LIMIT), path.rsplit('/', 1)[0]


def flags(value):
    need(value.get('retry_authorized') is False and
         value.get('native_job_cessation_not_inferred_from_bus_close') is True,
         'evidence cannot authorize retry or cessation')


def decode_final(raw, ref, expected_ref, *, parent, session_binding,
                 phase_raw=None, phase_ref=None):
    """Resolve exact full records; context/pins must come from independent custody.
    Inline records carry no session fields: their caller pin is the session binding.
    Phase bytes additionally bind exact typed parent/session and fixed JSON pointers.
    """
    value, root = pinned(raw, ref, expected_ref, 'parent-final-bus-inventory.json')
    need(set(value) == {'contract', 'state', 'original_reason', 'cleanup_error',
         'snapshot_error', 'canonical_fd9', 'bus_inventory', 'transport_inventory',
         'bus_present', 'native_job_cessation_not_inferred_from_bus_close',
         'retry_authorized', 'inventory_storage', 'inventory_ref'}, 'final V2 shape')
    need(value['contract'] == FINAL_CONTRACT, 'final V2 required')
    flags(value)
    need(value['state'] in ('STOP_UNCONFIRMED', 'INVENTORY_ONLY_NOT_CESSATION_AUTHORITY'), 'final state')
    need(type(value['bus_present']) is bool, 'bus_present bool')
    need(type(value['canonical_fd9']) is str, 'FD9 diagnostic type')
    for key in ('original_reason', 'cleanup_error', 'snapshot_error'):
        need(value[key] is None or type(value[key]) is str, key + ' type')
    if any(value[k] is not None for k in ('original_reason', 'cleanup_error', 'snapshot_error')):
        need(value['state'] == 'STOP_UNCONFIRMED', 'error state must remain stopped')
    transports, bus = value['transport_inventory'], value['bus_inventory']
    if value['inventory_storage'] == 'INLINE':
        need(value['inventory_ref'] is None, 'mixed inline/reference form')
    else:
        need(value['inventory_storage'] == 'COMPLETE_PHASE_REFERENCE', 'storage version')
        need(transports is None and bus is None, 'mixed compact/inline form')
        embedded = value['inventory_ref']
        need(type(embedded) is dict and set(embedded) == {'path', 'sha256', 'bytes',
             'transport_json_pointer', 'bus_json_pointer', 'transport_count'}, 'phase reference shape')
        path = reference({k: embedded[k] for k in ('path', 'sha256')})
        reference(phase_ref)
        need(same_json({k: embedded[k] for k in ('path', 'sha256')}, phase_ref), 'phase caller pin')
        need(type(embedded['bytes']) is int and 0 < embedded['bytes'] <= PHASE_LIMIT, 'phase size type/bound')
        need(type(embedded['transport_count']) is int and embedded['transport_count'] >= 0, 'transport count type')
        need(embedded['transport_json_pointer'] == '/owned_transports' and
             embedded['bus_json_pointer'] == '/owned_direct_bus', 'fixed JSON pointers')
        snapshot = parse(phase_raw, embedded['sha256'], PHASE_LIMIT)
        need(len(phase_raw) == embedded['bytes'], 'exact phase size')
        phase = snapshot.get('phase')
        need(phase in ('before_cleanup', 'after_cleanup_reload') and
             path == root + '/parent-cessation-' + phase + '.json', 'exact complete phase path')
        need(type(parent) is dict and type(session_binding) is dict and
             same_json(snapshot.get('parent'), parent) and
             same_json(snapshot.get('session_binding'), session_binding), 'typed parent/session binding')
        need(snapshot.get('driver_JSON_used_for_cessation') is False, 'driver cannot be cessation authority')
        transports, bus = snapshot.get('owned_transports'), snapshot.get('owned_direct_bus')
        need(type(transports) is list and len(transports) == embedded['transport_count'], 'full transport count')
    need(transports is None or type(transports) is list, 'transport inventory type')
    need(bus is None or type(bus) is dict, 'bus inventory type')
    if transports is not None:
        need(all(type(item) is dict for item in transports), 'transport record type')
    if transports is None:
        need(bus is None and value['snapshot_error'] is not None and value['state'] == 'STOP_UNCONFIRMED', 'missing snapshot is unavailable')
    elif value['bus_present']:
        need(type(bus) is dict, 'present bus needs complete data')
    else:
        need(bus is None, 'absent bus cannot carry invented data')
    return {'state': value['state'], 'data_status': 'UNAVAILABLE' if transports is None else 'COMPLETE',
            'transport_inventory': transports, 'bus_inventory': bus, 'envelope': value,
            'native_cessation_inferred': False, 'execution_authorized': False}


def summary_for(transports, bus):
    # Reproduce the existing producer summary only. The returned inventory itself
    # preserves None exactly; this historical digest convention never fills it in.
    digest_bus = bus or {}
    return {'transport_count': None if transports is None else len(transports),
        'transport_sha256': hashlib.sha256(json.dumps(transports,sort_keys=True,separators=(',',':')).encode()).hexdigest(),
        'bus_sha256': hashlib.sha256(json.dumps(digest_bus,sort_keys=True,separators=(',',':')).encode()).hexdigest(),
        'bus_status': {'closed': digest_bus.get('closed') is True,
            'socket_closed': digest_bus.get('socket_closed') is True,
            'failed_present': digest_bus.get('failed') is not None,
            'cleanup_errors_present': bool(digest_bus.get('cleanup_errors')),
            'pending_call_present': digest_bus.get('pending_call') is not None,
            'job_terminal': digest_bus.get('job_state') == 'TERMINAL',
            'no_reference_acquired': digest_bus.get('no_reference_acquired') is True}}


def decode_refusal(raw, ref, expected_ref, *, refusal_contract, parent, session_binding, original_anchor,
                   final_raw=None, expected_final_ref=None, phase_raw=None, phase_ref=None):
    """Refusal has no in-band version: caller MUST pin the reviewed V2 contract.
    Unavailable data remains unavailable; a late STOP overrides snapshot appearance.
    """
    need(refusal_contract == REFUSAL_CONTRACT, 'external compact refusal V2 contract required')
    value, root = pinned(raw, ref, expected_ref, 'parent-refusal.json')
    need(set(value) == {'state', 'reason', 'finalization_errors', 'final_bus_evidence_failure',
        'original_attempt', 'original_attempt_ref', 'canonical_fd9', 'fd9_release_confirmed',
        'dropins', 'inventory_summary', 'final_bus_inventory_ref', 'observation_refs',
        'retry_authorized', 'native_job_cessation_not_inferred_from_bus_close'}, 'compact refusal V2 shape')
    need(value['state'] == 'STOP_UNCONFIRMED', 'refusal stays stopped')
    flags(value)
    need(type(value['fd9_release_confirmed']) is bool and type(value['canonical_fd9']) is str, 'FD9 diagnostic types')
    original = value['original_attempt']
    need(type(original) is dict and same_json(original.get('original_parent'), parent) and
        same_json(original.get('original_anchor'), original_anchor) and
        same_json(original.get('original_attempt_token'), session_binding['attempt_token']) and
        same_json(original.get('original_boot_id'), session_binding['boot_id']) and
        same_json(original.get('manifest_sha256'), session_binding['manifest_sha256']), 'refusal session binding')
    # The producer's full anchor also contains accepted_realtime_ns. Bind the
    # entire caller-pinned anchor, not a lossy two-clock reconstruction.
    if original_anchor is not None:
        need(type(original_anchor) is dict and same_json(
            {k: original_anchor[k] for k in ('accepted_monotonic_ns','accepted_boottime_ns')},
            {k: session_binding[k] for k in ('accepted_monotonic_ns','accepted_boottime_ns')}), 'refusal original clocks')
    summary = value['inventory_summary']
    need(type(summary) is dict and set(summary) == {'transport_count','transport_sha256',
        'bus_sha256','bus_status','complete_data_ref','data_ref_status','full_inventory_not_inlined'}, 'summary V2 shape')
    need(summary['full_inventory_not_inlined'] is True, 'no mixed inline refusal')
    count = summary['transport_count']
    need(count is None or (type(count) is int and count >= 0), 'summary count type')
    pin(summary['transport_sha256']); pin(summary['bus_sha256'])
    status = summary['bus_status']
    need(type(status) is dict and set(status) == {'closed','socket_closed','failed_present',
        'cleanup_errors_present','pending_call_present','job_terminal','no_reference_acquired'} and
        all(type(v) is bool for v in status.values()), 'typed compact bus status')
    need(same_json(summary['complete_data_ref'], value['final_bus_inventory_ref']), 'refusal reference disagreement')
    complete_ref = summary['complete_data_ref']
    decoded = None
    if complete_ref is None:
        need(summary['data_ref_status'] == 'UNAVAILABLE_RETAIN_PARENT', 'missing ref must stay unavailable')
        need(final_raw is None, 'unreferenced final bytes')
    else:
        need(summary['data_ref_status'] == 'AVAILABLE', 'mixed reference availability')
        need(reference(complete_ref) == root + '/parent-final-bus-inventory.json', 'refusal final path')
        if expected_final_ref is not None:
            reference(expected_final_ref)
            need(same_json(complete_ref, expected_final_ref), 'refusal final caller pin')
        if final_raw is not None:
            decoded = decode_final(final_raw, complete_ref, expected_final_ref, parent=parent,
                session_binding=session_binding, phase_raw=phase_raw, phase_ref=phase_ref)
            expected_summary = summary_for(decoded['transport_inventory'], decoded['bus_inventory'])
            need(same_json({k: summary[k] for k in expected_summary}, expected_summary), 'full summary digest/count/status mismatch')
    return {'state': 'STOP_UNCONFIRMED',
        'data_status': 'UNAVAILABLE' if decoded is None else decoded['data_status'],
        'transport_inventory': None if decoded is None else decoded['transport_inventory'],
        'bus_inventory': None if decoded is None else decoded['bus_inventory'],
        'reported_summary': summary, 'refusal': value,
        'native_cessation_inferred': False, 'execution_authorized': False}
