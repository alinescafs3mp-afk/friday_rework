def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def ok(name, condition):
    assert condition, name
    checks.append(name)

def pin(path, expected=None):
    path = Path(path)
    digest = sha(path)
    if expected is not None:
        assert digest == expected, str(path)
    pins[str(path)] = digest
    return {'path': str(path), 'sha256': digest}

def read(path):
    pin(path)
    return json.loads(Path(path).read_bytes())

def verify_refs(v, owner):
    if isinstance(v, list):
        for n in v: verify_refs(n, owner)
    elif isinstance(v, dict):
        for k, n in v.items():
            if k.startswith(str(C)+'/') and isinstance(n, str) and len(n) == 64:
                assert sha(k) == n, (owner, k); refs.append([owner,k])
            verify_refs(n, owner)
        if isinstance(v.get('path'), str) and v['path'].startswith(str(C)+'/') and 'sha256' in v:
            assert sha(v['path']) == v['sha256'], (owner, v['path'])
            if 'bytes' in v: assert Path(v['path']).stat().st_size == v['bytes']
            refs.append([owner,v['path']])

def rebind(value):
    if isinstance(value, str): return value.replace(str(B), str(C))
    if isinstance(value, list): return [rebind(x) for x in value]
    if not isinstance(value, dict): return value
    d = {k.replace(str(B),str(C)): rebind(v) for k,v in value.items()}
    for k,v in d.items():
        if k.startswith(str(C)+'/') and isinstance(v,str) and len(v)==64: d[k]=sha(k)
    if isinstance(d.get('path'),str) and d['path'].startswith(str(C)+'/'):
        if 'sha256' in d: d['sha256']=sha(d['path'])
        if 'bytes' in d: d['bytes']=Path(d['path']).stat().st_size
    return d
