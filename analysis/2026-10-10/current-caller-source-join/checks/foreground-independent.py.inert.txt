def pin(path, expected=None):
    path = Path(path); raw = path.read_bytes(); digest = hashlib.sha256(raw).hexdigest()
    assert expected is None or digest == expected, str(path)
    pins[str(path)] = {'sha256': digest, 'bytes': len(raw)}
    return raw
