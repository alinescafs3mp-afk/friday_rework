def require(value, why):
    if not value:
        raise ValueError(why)


def read_pinned(path, expected, *, private=False):
    path = Path(path)
    require(path.is_absolute() and path.resolve() == path, "noncanonical_pin")
    fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK)
    try:
        st = os.fstat(fd)
        require(stat.S_ISREG(st.st_mode) and st.st_uid == 1000 and st.st_nlink == 1
                and st.st_size <= 16 << 20, "unsafe_pin")
        if private:
            require(stat.S_IMODE(st.st_mode) == 0o600, "private_pin_required")
        with os.fdopen(fd, "rb", closefd=False) as stream:
            raw = stream.read((16 << 20) + 1)
        require(hashlib.sha256(raw).hexdigest() == expected, "pin_changed")
        return raw
    finally:
        os.close(fd)


def ref(path, digest=None):
    return {"path": str(path), "sha256": digest or hashlib.sha256(Path(path).read_bytes()).hexdigest()}

    require(Path("/proc/sys/kernel/random/boot_id").read_text().strip() == pins["boot_id"], "boot_changed")

def acquire_lock():
    fd = os.open(LOCK, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK)
    try:
        st = os.fstat(fd)
        require(stat.S_ISREG(st.st_mode) and
                (st.st_dev, st.st_ino, st.st_uid, st.st_gid, stat.S_IMODE(st.st_mode),
                 st.st_nlink, st.st_size) == LOCK_ID, "quality_lock_identity_changed")
        fcntl.flock(fd, fcntl.LOCK_EX | fcntl.LOCK_NB)
        return fd
    except BaseException:
        os.close(fd)
        raise


def retain_unknown(fd):
    # No receipt, serialization, print/flush, input, retry, replacement owner or
    # release condition precedes this retention. Original parent and FD stay live.
    while True:
        try:
            signal.pause()
        except BaseException:
            pass


def invoke_with_custody(fd, invoke, report):
    settled = False
    code = 3
    try:
        try:
            invoke()
            code, settled = 0, True
        except SystemExit as exc:
            if type(exc.code) is int and exc.code in (0, 2):
                code, settled = exc.code, True
        except BaseException:
            pass
        report(code, settled)  # Diagnostics never decide whether custody survives.
    finally:
        if not settled:
            retain_unknown(fd)
        os.close(fd)
    return code


def write_once(name, value):
    with (E / name).open("x") as stream:
        json.dump(value, stream, indent=2)
        stream.write("\n")
        stream.flush()
        os.fsync(stream.fileno())


def interrupted(signum, frame):
    raise KeyboardInterrupt


def normal_main(args=None):
    args = sys.argv[1:] if args is None else args
    require(len(args) == 2, "NOT_ADMITTED_NOT_RUN: exact review and admission SHA256 required")
    os.umask(0o077)
    sys.dont_write_bytecode = True
    preflight(*args)
    # Future actual native execution retains the existing per-process policy;
    # it makes no claim of an aggregate 20GiB cgroup limit.
    fd = acquire_lock()
    try:
        preflight(*args)
        require(os.isatty(0) and os.tcgetpgrp(0) == os.getpgrp(), "visible_foreground_original_parent_required")
        os.sched_setaffinity(0, sorted(os.sched_getaffinity(0))[:8])
        resource.setrlimit(resource.RLIMIT_AS, (20 << 30, 20 << 30))
        resource.setrlimit(resource.RLIMIT_CORE, (0, 0))
        for sig in (signal.SIGTERM, signal.SIGINT, signal.SIGHUP):
            signal.signal(sig, interrupted)
        write_once("consumed.json", {"state": "ONE_ENTRY_CONSUMED_RUNTIME_UNACCEPTED",
                   "pid": os.getpid(), "lock_fd": fd,
                   "starttime_ticks": int(Path("/proc/self/stat").read_text().rsplit(")", 1)[1].split()[19]),
                   "input": ref(I), "original_install_seconds": 3600,
                   "clock": "Existing public friday_install.main starts before parser/input IO; no launcher clock or reset",
                   "aggregate_memory_limit_claimed": False})
    except BaseException:
        os.close(fd)  # No native invocation has been crossed.
        raise

    def invoke():
        # File/flush/close failures are uncertain too; no diagnostic can bypass
        # the unconditional same-parent retention in invoke_with_custody.
        with (E / "stdout.private.log").open("x") as out, (E / "stderr.private.log").open("x") as err:
            with contextlib.redirect_stdout(out), contextlib.redirect_stderr(err):
                sys.argv = [str(Q / "scripts/friday_install.py"), "install", "--input", str(I)]
                sys.path.insert(0, str(Q))
                runpy.run_path(str(Q / "scripts/friday_install.py"), run_name="__main__")

    def report(code, settled):
        record = {"state": "ENTRY_RETURNED_RUNTIME_UNACCEPTED" if settled else "STOP_UNCONFIRMED",
                  "returncode": code, "same_parent_pid": os.getpid(), "lock_fd": fd,
                  "runtime_accepted": False, "automatic_retry": False}
        write_once("returned.json", record)
        print(json.dumps(record), flush=True)

    return invoke_with_custody(fd, invoke, report)

def selected_object(raw):
    def unique(pairs):
        value = {}
        for key, item in pairs:
            require(key not in value, 'duplicate_selected_json_key')
            value[key] = item
        return value
    value = json.loads(raw, object_pairs_hook=unique,
                       parse_constant=lambda _: (_ for _ in ()).throw(ValueError('nonfinite_selected_json')))
    require(type(value) is dict, 'selected_object_required')
    return value

def selected_preflight(config_path, config_sha, review_path, review_sha):
    require(not sys.flags.optimize and 'PYTHONOPTIMIZE' not in os.environ, 'optimized_entry_refused')
    config = selected_object(read_pinned(config_path, config_sha, private=True))
    review_ref = {'path': str(Path(review_path)), 'sha256': review_sha}
    review = selected_object(read_pinned(review_path, review_sha, private=True))
    admission_ref = config['admission']
    admission = selected_object(read_pinned(admission_ref['path'], admission_ref['sha256'], private=True))
    producer_ref = {'path': str(SELECTED_PRODUCER), 'sha256': SELECTED_PRODUCER_SHA}
    validate_selected_binding(config, admission, review, ref(Path(__file__).resolve()), review_ref, producer_ref)
    producer = selected_object(read_pinned(SELECTED_PRODUCER, SELECTED_PRODUCER_SHA))
    require(producer['schema'] == 'friday.sol.source-successor.v1'
            and len(producer['files']) == len(SELECTED_FILES)
            and {row['relative']: row['after']['sha256'] for row in producer['files']} == SELECTED_FILES,
            'actual_producer_manifest_required')
    for row in producer['files']:
        name = row['relative']
        require(row['after']['path'] == str(SELECTED_SOURCE / name), 'producer_path_changed')
        read_pinned(row['after']['path'], SELECTED_FILES[name])
    permit_ref = config['permit']
    permit = selected_object(read_pinned(permit_ref['path'], permit_ref['sha256'], private=True))
    require(permit.get('schema') == 'friday.native-fault-observation-permit.v1'
            and permit.get('source_reviewed_observation_authorized') is True
            and permit.get('retry_authorized') is False
            and permit.get('product_mode_admission') is False
            and permit.get('original_clock') == config['original_clock']
            and permit.get('qualification_directory') == str(SELECTED_SOURCE / 'qualification')
            and permit['caller_control']['topology'] == SELECTED_TOPOLOGY,
            'same_selected_permit_source_clock_required')
    # Constructor independently verifies interpreter, patched binary, original
    # live interval, clean output and exact resources before prepare/Popen.
    return {'config': config, 'permit': permit, 'config_ref': {'path': config_path, 'sha256': config_sha}}


def selected_modules():
    # A fresh exact namespace prevents an ambient scripts package/__init__.py
    # from substituting the two pinned modules. No candidate is imported here.
    import importlib
    import types
    require(not any(name == 'scripts' or name.startswith('scripts.') for name in sys.modules),
            'fresh_exact_scripts_namespace_required')
    package = types.ModuleType('scripts')
    package.__path__ = [str(SELECTED_SOURCE / 'scripts')]
    sys.modules['scripts'] = package
    containment = importlib.import_module('scripts.install_containment')
    dsh = importlib.import_module('scripts.dsh_prepare')
    require(Path(containment.__file__).resolve() == SELECTED_SOURCE / 'scripts/install_containment.py'
            and Path(dsh.__file__).resolve() == SELECTED_SOURCE / 'scripts/dsh_prepare.py',
            'exact_loaded_existing_modules_required')
    return containment, dsh


def selected_owner():
    return dict(containment=None, control=None, failure=None, original_failure=None,
                process=None, selector=None, fd_records=None, channel_records=None,
                cleanup_errors=None, result=None, diagnostic_failure=None,
                invocation_entered=False, settled=False, retaining=False,
                state='PRE_BIRTH', returncode=3)


def capture_selected_custody(owner):
    # Strong aliases to the ORIGINAL objects, never an integer-FD substitute or
    # serialized snapshot. The control also retains every original ownership
    # record when a close may already have released/reused the numeric fd.
    control = owner['control']
    if control is not None:
        owner['process'] = control.owned_process
        owner['selector'] = control.owned_selector
        owner['fd_records'] = control.fd_records
        owner['channel_records'] = control.channel_records
        owner['cleanup_errors'] = control.cleanup_errors
    failure = owner['failure']
    if failure is not None:
        owner['original_failure'] = getattr(failure, 'original_failure', None)
        # failure retains its traceback, its own caller_control, owned_process
        # and owned_selector even if different from the expected control.


def selected_prebirth(control):
    # No returned-process test alone: after prepare, an unreturned Popen/exec
    # failure is specifically UNQUALIFIED (G2), not verified pre-birth refusal.
    return (not control.consumed and control.owned_process is None
            and control.owned_selector is None and control.monitor_fd is None
            and control.caller_fd is None and not control.fd_records
            and control.channel is None and control.child_channel is None)


def selected_closed_terminal(control, observation):
    return (control.closed is True and control.terminal_confirmed is True
            and not control.cleanup_errors
            and all(state == 'CLOSED' for state in control.fd_records.values())
            and all(state == 'CLOSED' for state in control.channel_records.values())
            and control.owned_process is not None
            and control.owned_process.returncode is not None
            and observation.get('reaped') is True
            and observation.get('namespace_init_exit_verified') is True
            and observation.get('caller_pidfd_exit_observed') is True)


def invoke_selected_with_custody(owner, settings, containment_module, dsh, report):
    """Concrete same-parent join; existing Containment.run calls existing dsh.run.

    Only that producer releases custody. This entry never closes, reacquires or
    retries an FD/PID, and never mints a grant or extends the original clock.
    """
    try:
        try:
            config, permit = settings['config'], settings['permit']
            clock = config['original_clock']
            budget = containment_module.Budget(180, started=clock['started_monotonic'],
                                               deadline=clock['deadline_monotonic'])
            owner['containment'] = containment_module.Containment(
                permit['caller_control']['binary'], budget,
                {'PATH': '/usr/bin:/bin', 'LANG': 'C', 'LC_ALL': 'C'}, caller_control=config)
            owner['control'] = owner['containment'].caller_control
            command = [permit['interpreter'], '-I', '-B', '-S',
                       str(Path(permit['qualification_directory']) / 'native_fault_launch.py'),
                       config['permit']['path']]
            owner['invocation_entered'] = True
            owner['result'] = owner['containment'].run(
                command, Path(permit['qualification_directory']).parent, timeout=180, read_only=True)
        except BaseException as exc:
            owner['failure'] = exc  # Keep exact typed/untyped object and traceback.
        capture_selected_custody(owner)
        failure, control = owner['failure'], owner['control']
        if isinstance(failure, dsh.StopUnconfirmed):
            owner['state'] = 'STOP_UNCONFIRMED'
        elif failure is not None and not owner['invocation_entered']:
            # Actual pinned constructor has acquired no persistent native
            # process/selector/quality custody; temporary input IO is not G2 proof.
            owner.update(state='PRE_BIRTH_REFUSED', settled=True, returncode=2)
        elif failure is not None and control is not None and selected_prebirth(control):
            owner.update(state='PRE_BIRTH_REFUSED', settled=True, returncode=2)
        else:
            observation = (failure.observation if isinstance(failure, dsh.CommandFailed)
                           else owner['result'][1] if failure is None and owner['result'] is not None else {})
            if (failure is None or isinstance(failure, dsh.CommandFailed)) and selected_closed_terminal(control, observation):
                owner.update(state='ENTRY_RETURNED_RUNTIME_UNACCEPTED', settled=True,
                             returncode=0 if failure is None else 2)
            else:
                owner['state'] = 'STOP_UNCONFIRMED'
        try:
            report(owner)
        except BaseException as exc:
            owner['diagnostic_failure'] = exc
            if owner['settled']:
                owner['returncode'] = 2
        return owner['returncode']
    finally:
        if not owner['settled']:
            owner['retaining'] = True
            retain_unknown(owner)  # Exact full custody stays in original parent.


def selected_report(owner):
    control = owner['control']
    quality = control.quality if control is not None else None
    record = {'state': owner['state'], 'returncode': owner['returncode'],
              'same_parent_pid': os.getpid(), 'runtime_accepted': False, 'automatic_retry': False,
              'quality_descriptor_record': control.fd_records.get(quality) if control is not None else None,
              'quality_lock_held': 'NOT_REOBSERVED_NOT_CLAIMED',
              'exact_ownership_records_retained': not owner['settled']}
    # Failure of this optional diagnostic cannot decide custody or replace the
    # original typed failure. G2 synchronous output latency is still unqualified.
    print(json.dumps(record), flush=True)


def selected_interrupted(signum, frame):
    if SELECTED_OWNER is not None and SELECTED_OWNER['retaining']:
        return  # The existing same-parent hold remains active until reconciliation.
    raise KeyboardInterrupt


def selected_main(config_path, config_sha, review_path, review_sha):
    global SELECTED_OWNER
    require(SELECTED_OWNER is None, 'no_second_selected_invocation')
    os.umask(0o077)
    sys.dont_write_bytecode = True
    settings = selected_preflight(config_path, config_sha, review_path, review_sha)
    require(os.isatty(0) and os.tcgetpgrp(0) == os.getpgrp(), 'visible_foreground_original_parent_required')
    require(sorted(os.sched_getaffinity(0)) == [0, 1]
            and resource.getrlimit(resource.RLIMIT_AS) == (2 << 30, 2 << 30)
            and all(v == resource.RLIM_INFINITY or v >= 45 for v in resource.getrlimit(resource.RLIMIT_CPU)),
            'existing_selected_ancestor_resource_envelope_required')
    containment_module, dsh = selected_modules()
    # Recheck exact supplied admission and bytes after import, before native
    # ownership. The selected route NEVER calls legacy acquire_lock().
    require(selected_preflight(config_path, config_sha, review_path, review_sha) == settings,
            'selected_inputs_changed_during_module_load')
    for sig in (signal.SIGTERM, signal.SIGINT, signal.SIGHUP):
        signal.signal(sig, selected_interrupted)
    owner = selected_owner()
    SELECTED_OWNER = owner
    try:
        return invoke_selected_with_custody(owner, settings, containment_module, dsh, selected_report)
    finally:
        # A second outer exit guard also covers interruption/finalizer failure
        # escaping the invocation guard. No diagnostics precede this hold.
        if not owner['settled']:
            owner['retaining'] = True
            retain_unknown(owner)


def main(args=None):
    args = sys.argv[1:] if args is None else args
    if args and args[0] == '--selected-fault':
        require(len(args) == 5, 'NOT_ADMITTED_NOT_RUN: config/ref and independent entry review/ref required')
        return selected_main(*args[1:])
    return normal_main(args)
