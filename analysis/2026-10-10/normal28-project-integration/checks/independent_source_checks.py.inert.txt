def clock():

def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def read(path):
    clock()
    path = Path(path)
    before = path.lstat()
    assert stat.S_ISREG(before.st_mode) and path.resolve() == path and before.st_size < 32 * 1024**2
    raw = path.read_bytes()
    after = path.lstat()
    fields = ('st_dev', 'st_ino', 'st_mode', 'st_size', 'st_mtime_ns', 'st_ctime_ns')
    assert all(getattr(before, f) == getattr(after, f) for f in fields), 'changed_read'
    return raw


def pin(path):
    raw = read(path)
    return {'path': str(path), 'sha256': sha(raw), 'bytes': len(raw)}


def unique(pairs):
    out = {}
    for key, value in pairs:
        assert key not in out, 'duplicate_json_key'
        out[key] = value
    return out


def load(path):
    return json.loads(read(path), object_pairs_hook=unique)


def put(name, value):
    with (W / name).open('x') as f:
        f.write(json.dumps(value, indent=2) + '\n')

def check(name, call):

def refuse(call, reason):
    try:
        call()
    except (ValueError, RuntimeError, PermissionError) as exc:
        assert str(exc) == reason, ('wrong_refusal', str(exc), reason)
    else:
        raise AssertionError('missing_refusal_' + reason)

def selected(path, name, env):
    raw = read(path)
    node = next(n for n in ast.parse(raw).body if isinstance(n, (ast.FunctionDef, ast.ClassDef)) and n.name == name)
    assert not node.decorator_list, ('unexpected_decorator', name)
    env['__builtins__'] = {**vars(builtins), '__import__': importer}
    nodes = [ast.ImportFrom(module='__future__', names=[ast.alias(name='annotations')], level=0), copy.deepcopy(node)]
    exec(compile(ast.fix_missing_locations(ast.Module(body=nodes, type_ignores=[])), str(path), 'exec'), env)
    selected_pins.append({'path': str(path), 'file_sha256': sha(raw), 'name': name,
        'line': node.lineno, 'end_line': node.end_lineno,
        'body_sha256': sha(ast.get_source_segment(raw.decode(), node).encode()), 'body_modified': False})
    return env[name]


def git(args, data=None):
    clock()
    proc = subprocess.run(['git', '--no-optional-locks', '-c', 'core.fsmonitor=false',
        '-c', 'core.hooksPath=/dev/null', *args], cwd=MAIN, input=data,
        stdout=subprocess.PIPE, stderr=subprocess.PIPE,
        env=dict(os.environ, GIT_OPTIONAL_LOCKS='0', GIT_TERMINAL_PROMPT='0'),
        timeout=min(30, (deadline-datetime.now(MSK)).total_seconds()))
    assert proc.returncode == 0, ('read_only_git_failed', proc.stderr[:200])
    return proc.stdout
