"""Finite actual overlay syntax controls; never import or execute product source."""
import datetime
import hashlib
import json
import os
from pathlib import Path
import resource
import time

D = Path(__file__).resolve().parent
started = time.monotonic()
os.sched_setaffinity(0, sorted(os.sched_getaffinity(0))[:2])
resource.setrlimit(resource.RLIMIT_AS, (1 << 30, 1 << 30))
resource.setrlimit(resource.RLIMIT_CORE, (0, 0))
receipt = json.loads((D / 'hermes.source.json').read_text())
paths = set()
manifests = []
for layer in receipt['layers']:
    data = (D / 'candidate' / layer['manifest']).read_bytes()
    assert hashlib.sha256(data).hexdigest() == layer['manifest_sha256']
    manifest = json.loads(data)
    if 'files' in manifest:
        paths.update(row['path'] for row in manifest['files'])
    else:
        paths.update(manifest['expected_patched_files'])
    manifests.append({'path': layer['manifest'], 'sha256': layer['manifest_sha256']})
rows = []
for relative in sorted(paths):
    if time.monotonic() - started > 120:
        raise RuntimeError('source_syntax120_budget')
    path = D / 'hermes' / relative
    data = path.read_bytes()
    assert hashlib.sha256(data).hexdigest() == receipt['files'][relative]['sha256']
    if path.suffix == '.py':
        compile(data, str(path), 'exec', dont_inherit=True)
        rows.append({'path': relative, 'sha256': receipt['files'][relative]['sha256'],
                     'status': 'COMPILED_WITHOUT_IMPORT_OR_EXECUTION'})
result = {
    'schema': 'friday.normal28.overlay-syntax-source-controls.v1',
    'at_msk': datetime.datetime.now(datetime.timezone(datetime.timedelta(hours=3))).isoformat(),
    'status': 'PASS_SOURCE_ONLY', 'manifest_count': len(manifests),
    'unique_overlay_paths': len(paths), 'actual_python_sources_compiled': len(rows),
    'rows': rows, 'manifests': manifests,
    'elapsed_seconds': time.monotonic() - started, 'original_seconds': 120,
    'affinity': sorted(os.sched_getaffinity(0)), 'address_space_limit_bytes': 1 << 30,
    'bytecode_written': False, 'subject_imports': 0, 'runtime_grant': False,
    'installed_acceptance': False,
    'preparation_failure': 'First inline collector assumed uniform files key; refused before compilation/output. Historical manifests use expected_patched_files too. Subject unchanged.'}
output = D / 'OVERLAY-SYNTAX-CONTROLS.json'
output.write_text(json.dumps(result, indent=2) + '\n')
output.chmod(0o600)
print(json.dumps({key: value for key, value in result.items() if key not in ('rows', 'manifests')}))
