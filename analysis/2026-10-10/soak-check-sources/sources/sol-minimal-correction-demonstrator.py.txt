os.umask(0o077)
os.sched_setaffinity(0, sorted(os.sched_getaffinity(0))[:2])
resource.setrlimit(resource.RLIMIT_AS,(1<<30,1<<30))
resource.setrlimit(resource.RLIMIT_CPU,(30,30))
resource.setrlimit(resource.RLIMIT_CORE,(0,0))
E=pathlib.Path(__file__).resolve().parent
W=pathlib.Path('/PUBLIC_LABEL/ASTRA_EVIDENCE_ROOT')
P=pathlib.Path('/PUBLIC_LABEL/PRODUCT_CHECKOUT')
A=pathlib.Path('/PUBLIC_LABEL/ARCHIVE_SOURCE_ROOT')
V=W/'daily-soak-source-astra/public/validation/validate_daily_data.py'

original=V.read_bytes()
assert hashlib.sha256(original).hexdigest()=='4851b1ef5c2b285293cfa97acd7a09457563424095dcb87cb4ce53b67bcd394d'
d=json.loads((C/'tasks.json').read_bytes())
guard="""    # Selected source policy only; retain the older pending-input semantics.
    need(soak['definition_sources'] == %(definition)s, 'missing selected policy definition references')
    need(soak['later_evidence'] == %(evidence)s, 'missing selected mandatory future evidence')
    need(type(soak['planned_mix']) is list and all(
        type(row) is dict and set(row) == {'task_id', 'occurrences', 'minimum_coverage'}
        and row['minimum_coverage'] == %(coverage)r
        for row in soak['planned_mix']), 'unknown mix field or changed mandatory coverage')
    need(type(soak['required_interruptions']) is list and all(
        type(row) is dict and type(row.get('minimum_count')) is int
        for row in soak['required_interruptions']), 'untyped interruption count')
""" % {'definition':repr(d['soak']['definition_sources']),'evidence':repr(d['soak']['later_evidence']),'coverage':d['soak']['planned_mix'][0]['minimum_coverage']}
anchor="    positive_integer(soak['duration_seconds'], 86400, 'invalid finite soak duration')\n"
assert original.decode().count(anchor)==1
new=original.decode().replace(anchor,guard+anchor)
path=E/'proposed_guard.DEMONSTRATION.py';path.write_text(new)
(E/'MINIMAL-CORRECTION.diff').write_text(''.join(difflib.unified_diff(original.decode().splitlines(keepends=True),new.splitlines(keepends=True),fromfile='reviewed validator 4851b1ef',tofile='PRIVATE guard demonstration NOT integrated or accepted')))
spec=importlib.util.spec_from_file_location('private_correction_demo',path)
m=importlib.util.module_from_spec(spec);spec.loader.exec_module(m)
pins=m.load((C/'source-pins.json').read_bytes());roots={'product':P,'archive':A};start=time.monotonic();checks=[]

for g in json.loads((E/'INDEPENDENT-RESULT.json').read_bytes())['gaps']:

    try:m.validate(x,pins,C,roots)
    except (ValueError,TypeError,KeyError) as ex:checks.append({'name':g['name'],'status':'PASS_EXPECTED_REFUSAL','reason':str(ex)})

old=m.load((W/'daily-long-integration/public/fixtures/daily-use/tasks.json').read_bytes())

out={'state':'PRIVATE_MINIMAL_SELECTED_POLICY_GUARD_DEMONSTRATION_ONLY','checks':checks,'count':len(checks),'elapsed_seconds':time.monotonic()-start,'original_candidate_sha256':hashlib.sha256(original).hexdigest(),'demonstration_sha256':hashlib.sha256(new.encode()).hexdigest(),'original_candidate_status':'BLOCKED_PENDING_AUTHOR_FIX_AND_EXACT_INDEPENDENT_REVIEW','author_candidate_modified':False,'main_integrated':False,'runtime_authority':False,'handles':0,'children':0}
