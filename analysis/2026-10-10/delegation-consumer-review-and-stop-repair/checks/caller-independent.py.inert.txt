def definitions(source):
    result = {}
    def visit(nodes, prefix=''):
        for n in nodes:
            if isinstance(n, (ast.FunctionDef, ast.ClassDef)):
                result[prefix+n.name] = ast.dump(n, include_attributes=False)
                if isinstance(n, ast.ClassDef): visit(n.body, prefix+n.name+'.')
    visit(ast.parse(source).body); return result

results=[]
def check(name,fn):
    try:
        facts=fn();results.append({'name':name,'pass':True,'facts':facts})
    except BaseException as exc:
        results.append({'name':name,'pass':False,'error':repr(exc)})

def run_case(version, faults, expected, nonzero=False):
    p=Probe(version,faults,nonzero);r=p.run();c=p.control
    assert r.kind==expected,(r.kind,repr(r.exc),p.events)
    if expected=='StopUnconfirmed':
        assert r.exc.caller_control is c
        if version=='after':
            assert r.exc.owned_process is (None if faults in ('popen_error','selector_create_error') else p.process)
            assert r.exc.owned_selector is (None if faults=='selector_create_error' else p.selector)
            assert c.cleanup_errors or not c.terminal_confirmed
    return p,r

def gate(version):
    p,r=run_case(version,'gate_cancel_after_close','StopUnconfirmed' if version=='after' else 'OSError')
    ev=p.events
    assert ['write',15,'1'] in ev
    assert not any(e[0]=='inert_existing_handshake' for e in ev)
    if version=='after':
        assert sum(e==['close_fd',15] for e in ev)==1
        assert ['pidfd_signal',13] in ev and any(e[0]=='wait4' and e[1]==222 for e in ev)
        assert p.control.fd_records[15]=='CLOSE_UNKNOWN'
        assert not any(e[0]=='REUSED_FD_CLOSE' for e in ev)
    else:
        assert not any(e[0] in ('pidfd_signal','wait4') for e in ev)
        assert any(e[0]=='REUSED_FD_CLOSE' for e in ev)
    return {'outward':r.kind,'events':ev,'records':p.control.fd_records}

check('repaired_F1_same_after_close_interruption',lambda:gate('after'))

        assert ['stdout_close'] in ev and ['stderr_close'] in ev

        assert any(e[0]=='wait4' and e[1]==222 for e in ev)
    if fault in ('gate_close_error',('gate_close_error','signal_error')): assert ['pidfd_signal',13] in ev
    if fault in ('selector_close_error','stdout_close_error','stderr_close_error',('selector_close_error','stdout_close_error','stderr_close_error'),'other_close_error','channel_close_error'):
        assert ['close_fd',19] not in ev and not c.closed
    if fault=='quality_close_error':
        assert c.fd_records[19]=='CLOSE_UNKNOWN' and 19 not in p.live and not c.closed
    if expected in ('return','CommandFailed'):

def direct_stop(fault=None,stale=False,expired=False):
    p=Probe(fault=fault);c=p.control;c.monitor_fd=14;c._own(14);c.caller_fd=13;c._own(13)

def finalizer_regression(version, finalizer):
    p=Probe(version,finalizer)
    # Actual terminal() refuses to confirm a non-ready inert caller PIDFD.
    # This yields the same prior typed-unknown route, before finalizers run.
    p.ns['__builtins__']['__import__']=lambda name,*a,**kw: N(select=lambda *args:([],[],[])) if name=='select' else p.importer(name,*a,**kw)
    r=p.run()
    assert r.kind==('StopUnconfirmed' if version=='after' else 'OSError'),(r.kind,p.events)
    if version=='before':
        chain=r.exc.__context__
        assert isinstance(chain,p.ns['StopUnconfirmed']) and chain.caller_control is p.control
        if finalizer=='selector_close_error': assert ['stdout_close'] not in p.events
        else:assert ['stderr_close'] not in p.events
    else:
        assert ['stdout_close'] in p.events and ['stderr_close'] in p.events
        assert r.exc.original_failure is not None
        assert r.exc.caller_control is p.control and r.exc.owned_process is p.process and r.exc.owned_selector is p.selector
    return {'outward':r.kind,'events':p.events}
for version in ('before','after'):
    for finalizer in ('selector_close_error','stdout_close_error'):
        check(version+'_prior_finalizer_'+finalizer,lambda v=version,f=finalizer:finalizer_regression(v,f))

            super().__setitem__(k,v)

        assert c._close_channel('channel')

def deadline_during_error():
    p=Probe(fault='selector_select_error')
    original=p.selector.select
    def select_then_expire(timeout):
        p.now=180.
        return original(timeout)
    p.selector.select=select_then_expire

    r=p.run()

    return {'events':p.events,'cleanup':p.control.cleanup_errors}
check('error_at_original_D_gets_no_new_wait_or_signal_budget',deadline_during_error)
