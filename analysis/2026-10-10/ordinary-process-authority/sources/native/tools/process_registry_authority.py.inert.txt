"""Ordinary-user access to the existing process registry.

The retained capability is the original principal/profile/admission generation,
not a task ID or a new grant. Trusted teardown keeps the native registry API:
revoking a user's read/write access must never prevent its owned-process cleanup.
This boundary alone does not authorize an ordinary terminal backend.
"""
import json


def capture_authority():
    from hermes_cli.friday_user_scope import delegation_authority
    return delegation_authority()


def visible(pin):
    from hermes_cli.friday_user_scope import delegation_visible
    return delegation_visible(pin)


def require_visible(session):
    from hermes_cli.friday_user_scope import ScopeDenied
    if session is None or not visible(session.user_authority):
        raise ScopeDenied()


def resolve_visible(registry, session_id):
    """Resolve only within the caller's authority, before any PID refresh.

    Canonical IDs returned here are used by the native handlers. An invisible
    exact match is unavailable, while invisible prefixes neither disclose nor
    make an otherwise unique visible prefix ambiguous.
    """
    from tools.process_registry_results import load_completed_results
    capture_authority()  # Validate even an empty registry / lost context.
    if not isinstance(session_id, str) or not session_id:
        return None
    query = session_id.strip()
    if not query.startswith("proc_"):
        query = "proc_" + query
    if len(query) - len("proc_") < registry._MIN_PREFIX_CHARS:
        return None
    sessions = load_completed_results(query)
    with registry._lock:
        sessions.update({sid: s for store in (registry._finished, registry._running)
                         for sid, s in store.items() if sid.startswith(query)})
    exact = sessions.get(query)
    if exact is not None:
        chosen = exact if visible(exact.user_authority) else None
    else:
        matches = [s for s in sessions.values() if visible(s.user_authority)]
        chosen = matches[0] if len(matches) == 1 else None
    if chosen is None:
        return None
    require_visible(chosen)
    refreshed = registry._refresh_detached_session(chosen)
    if refreshed is not None:
        require_visible(refreshed)
    return refreshed


def check_handoff(child, parent, session):
    from hermes_cli.friday_user_scope import check_agent
    require_visible(session)
    # Both agents retain the original scope; ambient equality alone cannot
    # authorize a handoff to a foreign, revoked, or cancelled parent/child.
    check_agent(child)
    check_agent(parent)


def handle_process(args, **kw):
    from hermes_cli.friday_user_scope import ScopeDenied, guard_tool
    from tools import process_registry as native

    try:
        authority = capture_authority()
        # Preserve the ordinary tool allowlist, including terminal's current
        # refusal until its complete backend and supervision are qualified.
        guard_tool("process_manage", args)
        action = args.get("action", "")
        sid = str(args.get("session_id", "")) if args.get("session_id") is not None else ""
        if action == "list":
            result = native._list_processes(kw.get("task_id"))
            capture_authority()
        elif action == "handoff" or action in native._SESSION_ACTIONS:
            if not sid:
                return native.tool_error(f"session_id is required for {action}")
            session = resolve_visible(native.process_registry, sid)
            if session is None:
                raise ScopeDenied()
            require_visible(session)
            if action == "handoff":
                result = native._handoff_process(session.id, args, kw.get("task_id"))
            else:
                handler, redact = native._SESSION_ACTIONS[action]
                result = handler(session.id, args)
                if redact:
                    result = native._redact_process_result(result)
            # A blocking wait can span revocation. Never return its output
            # under a subsequent grant, even if the native operation finished.
            require_visible(session)
        else:
            return native.tool_error(f"Unknown process action: {action}. Use: list, poll, log, wait, kill, write, submit, close, handoff")
        guard_tool("process_manage", args)
        if capture_authority() != authority:
            raise ScopeDenied()
        return json.dumps(result, ensure_ascii=False)
    except ScopeDenied:
        return native.tool_error("product_user_scope_refused")
