"""Reproduce the independently performed in-memory patch and inventory checks."""
import hashlib,json,os,pathlib,re,resource
resource.setrlimit(resource.RLIMIT_AS,(1073741824,1073741824))
os.sched_setaffinity(0,set(sorted(os.sched_getaffinity(0))[:2]))
r=pathlib.Path('<PRIVATE_CANDIDATE_ROOT>')
receipt=json.loads((r/'hermes.source.json').read_text());base=r.parent/'normal-skills-native-join/hermes'
patch=(r/'project/patches/hermes/process-authority.patch').read_text().splitlines(keepends=True)
results=[];i=0
while i<len(patch):
    assert patch[i].startswith('diff --git a/')
    name=patch[i].split(' b/',1)[1].strip();i+=1
    while not patch[i].startswith('--- '):i+=1
    old=patch[i][4:].strip();i+=2
    before=[] if old=='/dev/null' else (base/name).read_text().splitlines(keepends=True)
    out=[];cursor=0
    while i<len(patch) and not patch[i].startswith('diff --git '):
        match=re.match(r'@@ -(\d+)(?:,(\d+))? \+(\d+)(?:,(\d+))? @@',patch[i]);assert match
        oldstart,oldcount,newstart,newcount=match.groups();oldstart=int(oldstart);oldcount=int(oldcount or 1);newcount=int(newcount or 1)
        offset=max(oldstart-1,0);assert offset>=cursor;out.extend(before[cursor:offset]);cursor=offset;i+=1;seenold=seennew=0
        while i<len(patch) and not patch[i].startswith(('@@ ','diff --git ')):
            line=patch[i];i+=1
            if line.startswith('\\ No newline'):raise AssertionError('unsupported no-newline marker')
            assert line[:1] in (' ','+','-')
            if line[0] in ' -':assert before[cursor]==line[1:];cursor+=1;seenold+=1
            if line[0] in ' +':out.append(line[1:]);seennew+=1
        assert (seenold,seennew)==(oldcount,newcount)
    out.extend(before[cursor:]);data=''.join(out).encode()
    results.append({'path':name,'patch_matches_native':data==(r/'native'/name).read_bytes(),'sha256':hashlib.sha256(data).hexdigest()})
assert all(x['patch_matches_native'] for x in results)
actual={str(p.relative_to(r/'hermes')) for p in (r/'hermes').rglob('*') if p.is_file() or p.is_symlink()}
extras=sorted(actual-set(receipt['files']));missing=sorted(set(receipt['files'])-actual)
report={'pure_in_memory_patch_reconstruction':results,'tree_extras':extras,'tree_missing':missing,'physical_file_set_equal_receipt':not extras and not missing}
(r/'review/composition-extra-observations.json').write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps(report))
