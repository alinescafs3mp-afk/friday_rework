+"""Bind native persisted prompt reuse to the existing ordinary admission.
+
+Provenance lives in the session's existing model_config, beside native lifecycle
+metadata; the native prompt writer stores both in one transaction. It is not
+an authorization grant, and unscoped donor sessions retain their behavior.

+import json

+def _store(cap, db):

+        raise ScopeDenied()
+    check_database(cap)

+def _stamp(cap, session_id, prompt):
+    return {"schema": "friday.prompt-scope.v1", "principal_id": cap.key,
+            "generation": cap.admission_generation, "home": str(cap.home),
+            "directory_identity": list(cap.directory_identity), "session_id": session_id,
+            "prompt_sha256": hashlib.sha256(prompt.encode("utf-8")).hexdigest()}

+def matches_provenance(stored, prompt, admission):
+    """Validate the complete stored stamp against this admission and its prompt.
+
+    The admission may describe a newly built prompt. Recompute only its digest
+    for the stored bytes; every other field, including the key set, must match.
+    Restore and atomic tool-pin invalidation use this same predicate.
+    """
+    if not isinstance(prompt, str) or not prompt:
+        return False
+    expected = {**admission, "prompt_sha256": hashlib.sha256(prompt.encode("utf-8")).hexdigest()}
+    return isinstance(stored, dict) and stored == expected
+

+    cap = current()
+    if cap is not None:
+        check_agent(agent)
+        if agent._session_db is not None:
+            _store(cap, agent._session_db)

+    cap = current()
+    if cap is None:
+        return None
+    _store(cap, db)
+    result = _stamp(cap, session_id, prompt) if isinstance(prompt, str) else None
+    cap.check()

+            if identities[rel] != (_identity(after), after.st_size, after.st_mtime_ns, after.st_ctime_ns, after.st_mode):
+                raise ScopeDenied()
+    _private(os.fstat(fd), True)
+    identity = _identity(os.fstat(fd))
+    visit(fd, '', 0)
+    return files, dirs, (identity, identities)

+def _snapshot(root):
+    entered = False
+    try:

+            check()
+            return result

+def _verify_materialized(parent, name, identity, manifest, files, dirs):
+    """Bind bytes, shape, inode identities and modes to the validated tx.
+
+    This is a consistency check within the admitted writer fence, not an
+    exclusion primitive against arbitrary same-UID host code.
+    """
+    fd = os.open(name, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=parent)
+    try:
+        actual_files, actual_dirs, actual = _snapshot_fd(fd)
+        if (actual[0] != identity or actual_files != files or actual_dirs != dirs
+                or set(actual[1]) != set(manifest)
+                or any((info[0], info[-1]) != manifest[rel] for rel, info in actual[1].items())
+                or _identity(os.stat(name, dir_fd=parent, follow_symlinks=False)) != identity):
+            raise ScopeDenied()
+        return actual
+    finally:
+        os.close(fd)

+def walk_files(root):

+def write_bytes(path, data):
+    if not isinstance(data, bytes) or len(data) > _limit(path):
+        raise ScopeDenied()
+    tx, rel = _view(path)
+    if tx is not None:

+            raise ScopeDenied()

+        return

+            _private(old)

+            old = None

+            check()
+        finally:
+            try:
+                os.unlink(tmp, dir_fd=fd)
+            except FileNotFoundError:

+def remove(path, *, tree=False, missing_ok=False):
+    tx, rel = _view(path)
+    if tx is not None:
+        if not rel:
+            raise ScopeDenied()
+        if tree:
+            if rel not in tx.dirs:
+                raise FileNotFoundError(str(path))
+            tx.files = {k: v for k, v in tx.files.items() if k != rel and not k.startswith(rel + '/')}
+            tx.dirs = {k for k in tx.dirs if k != rel and not k.startswith(rel + '/')}
+        elif rel in tx.files:
+            del tx.files[rel]
+        elif not missing_ok:

+        return
+    if tree:
+        raise ScopeDenied()  # tree deletes require atomic native transaction
+    with _directory(Path(path).parent) as (fd, check):
+        try:
+            _private(os.stat(Path(path).name, dir_fd=fd, follow_symlinks=False))
+            check()
+            os.unlink(Path(path).name, dir_fd=fd)
+            check()
+        except FileNotFoundError:
+            if not missing_ok:
+                raise

+def _erase_at(parent, name, budget=None, depth=0, *, expected=None, manifest=None, prefix=''):
+    """Reclaim captured private storage while every admitted writer is fenced.
+
+    Callers must verify the complete materialized tree first. Path-based unlink
+    is safe only inside this cooperative product ownership boundary; an
+    arbitrary same-UID host can still substitute a name at the syscall. This
+    helper does not claim security against a compromised host. Unknown entries
+    observed before reclamation are retained for explicit reconciliation.
+    """
+    from hermes_cli.skill_lock import require_skill_lock
+    # Retained transaction supplies ownership even after its authority expires.
+    tx = _TX.get()
+    if tx is None or expected is None or manifest is None:
+        raise ScopeDenied()
+    require_skill_lock(tx.cap.home, authority=False)
+    budget = [MAX_ENTRIES] if budget is None else budget
+    if depth > MAX_DEPTH:
+        raise ScopeDenied()
+    fd = os.open(name, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=parent)
+    try:
+        identity = _identity(os.fstat(fd))
+        if expected is not None and identity != expected:
+            raise ScopeDenied()
+        children = _names(fd)
+        if manifest is not None:
+            wanted = {rel[len(prefix):].split('/')[0] for rel in manifest if rel.startswith(prefix)}
+            if set(children) != wanted:
+                raise ScopeDenied()
+        for child in children:
+            if budget[0] <= 0:
+                raise ScopeDenied()
+            budget[0] -= 1
+            rel = prefix + child
+            st = os.stat(child, dir_fd=fd, follow_symlinks=False)
+            child_identity = _identity(st)
+            if manifest is not None and manifest.get(rel) != child_identity:
+                raise ScopeDenied()
+            if stat.S_ISDIR(st.st_mode):
+                _erase_at(fd, child, budget, depth + 1, expected=child_identity,
+                          manifest=manifest, prefix=rel + '/')
+            else:
+                os.unlink(child, dir_fd=fd)
+        if _identity(os.stat(name, dir_fd=parent, follow_symlinks=False)) != identity:
+            raise ScopeDenied()
+    finally:
+        os.close(fd)
+    os.rmdir(name, dir_fd=parent)

+class _Transaction:
+    def __init__(self):
+        self.cap = current(required=True)
+        require_skill_lock(self.cap.home)
+        # A failed/crashed transaction owns at most one bounded retained tree.
+        # Never adopt unknown remnants or accumulate another stage behind it.
+        with _directory(self.cap.home) as (fd, check):

+                raise RuntimeError('skill_tree_reconciliation_required')

+        self.before = dict(self.files)
+        self.before_dirs = set(self.dirs)
+        self.abort = False
+        self.published = False
+
+    def check(self):
+        if current(required=True) is not self.cap:
+            raise ScopeDenied()
+        self.cap.check()

+    def bounded(self):
+        if len(self.files) + len(self.dirs) > MAX_ENTRIES or sum(map(len, self.files.values())) > MAX_TREE:
+            raise ScopeDenied()
+
+    def publish(self):
+        self.check(); self.bounded()
+        require_skill_lock(self.cap.home)
+        if self.abort or (self.files == self.before and self.dirs == self.before_dirs):
+            return
+        # Fail before effects if atomic directory exchange is unavailable.
+        libc = ctypes.CDLL(None, use_errno=True)
+        rename = getattr(libc, 'renameat2', None)
+        if rename is None:
+            raise ScopeDenied()

+        rename.restype = ctypes.c_int
+        with _directory(self.cap.home) as (homefd, check):
+            stage = '.skills-txn-' + uuid.uuid4().hex
+            os.mkdir(stage, 0o700, dir_fd=homefd)
+            stage_identity = None
+            stage_manifest = {}
+            publication_conflict = False
+            try:
+                stagefd = os.open(stage, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=homefd)
+                stage_identity = _identity(os.fstat(stagefd))
+                fds = {'': stagefd}
+                try:
+                    if _names(stagefd):
+                        raise ScopeDenied()
+                    for rel in sorted(self.dirs - {''}, key=lambda s: (s.count('/'), s)):
+                        parent, _, leaf = rel.rpartition('/')
+                        os.mkdir(leaf, 0o700, dir_fd=fds[parent])
+                        fds[rel] = os.open(leaf, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=fds[parent])
+                        stage_manifest[rel] = (_identity(os.fstat(fds[rel])), os.fstat(fds[rel]).st_mode)
+                    for rel, data in sorted(self.files.items()):
+                        parent, _, leaf = rel.rpartition('/')
+                        mode = self.original[1].get(rel, (0o600,))[-1] & 0o700 if self.original else 0o600
+                        item = os.open(leaf, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, mode, dir_fd=fds[parent])
+                        stage_manifest[rel] = (_identity(os.fstat(item)), os.fstat(item).st_mode)
+                        with os.fdopen(item, 'wb') as stream:
+                            stream.write(data); stream.flush(); os.fsync(stream.fileno())
+                    for fd in fds.values():
+                        os.fsync(fd)
+                finally:
+                    for fd in reversed(list(fds.values())):
+                        os.close(fd)
+                files, dirs, identity = _snapshot(self.root)
+                if identity != self.original or files != self.before or dirs != self.before_dirs:
+                    raise ScopeDenied()
+                check(); self.check()
+                if _identity(os.stat(stage, dir_fd=homefd, follow_symlinks=False)) != stage_identity:
+                    publication_conflict = True
+                    raise RuntimeError('skill_tree_staging_identity_conflict')
+                try:
+                    _verify_materialized(homefd, stage, stage_identity, stage_manifest, self.files, self.dirs)
+                except (OSError, ScopeDenied):
+                    publication_conflict = True
+                    raise RuntimeError('skill_tree_staging_content_conflict') from None
+                check(); self.check()
+                if rename(homefd, stage.encode(), homefd, b'skills', 2 if self.original else 1):
+                    raise ScopeDenied()
+                self.published = True
+                if (_identity(os.stat('skills', dir_fd=homefd, follow_symlinks=False)) != stage_identity
+                        or (self.original and _identity(os.stat(stage, dir_fd=homefd, follow_symlinks=False)) != self.original[0])):
+                    publication_conflict = True
+                    raise RuntimeError('skill_tree_publication_identity_conflict')
+                try:
+                    _verify_materialized(homefd, 'skills', stage_identity, stage_manifest, self.files, self.dirs)
+                except (OSError, ScopeDenied):
+                    publication_conflict = True
+                    raise RuntimeError('skill_tree_published_content_conflict') from None
+                os.fsync(homefd)
+                check(); self.check()
+            finally:
+                # Cleanup owns only a verified materialized tree under this
+                # same fence. Any unknown artifact is retained, and the next
+                # transaction refuses before allocating further storage.
+                try:
+                    if (not publication_conflict and stage_identity is not None
+                            and (not self.published or self.original is not None)):
+                        expected = self.original[0] if self.published and self.original else stage_identity
+                        manifest = ({rel: (info[0], info[-1]) for rel, info in self.original[1].items()}
+                                    if self.published and self.original else stage_manifest)
+                        expected_files = self.before if self.published and self.original else self.files
+                        expected_dirs = self.before_dirs if self.published and self.original else self.dirs
+                        # On failed partial construction this check retains the
+                        # bounded remnant rather than adopting partial contents.
+                        _verify_materialized(homefd, stage, expected, manifest, expected_files, expected_dirs)
+                        _erase_at(homefd, stage, expected=expected,
+                                  manifest={rel: info[0] for rel, info in manifest.items()})
+
+                except (OSError, ScopeDenied):
+                    if self.published:
+                        raise RuntimeError("skill_tree_published_cleanup_incomplete") from None
+                    raise RuntimeError("skill_tree_not_published_cleanup_incomplete") from None

+class DocumentPath:

+    def __str__(self):
+        return str(self.path)

+        return DocumentPath(self.path / part)

+    @property

+    @property

+    @property

+        return self.path.parts

+    def relative_to(self, other):

+    def exists(self):

+
+    def is_dir(self):

+            return False

+            st = os.stat(self.name, dir_fd=fd, follow_symlinks=False)

+@contextmanager
+def managed_home_lock(home):
+    """Yield True under the managed fence, False for an unmarked donor home.
+
+    Home/marker/lock descriptors and their ancestor links are checked before
+    and after flock's actual wait. A replaced lock is refused, never reopened
+    to silently follow a different inode. No path inside skills/ is a lock.
+    """
+    home = Path(home)
+    cap = current()
+    if cap is not None and home != cap.home:
+        raise ScopeDenied()
+    held = _held()
+    entry = held.get(home)
+    if entry is not None:
+        if entry[0] != os.getpid():
+            raise ScopeDenied()  # fork must not inherit a false reentrant claim
+        entry[1]()
+        if cap is not None:
+            cap.check()
+        yield True
+        entry[1]()
+        if cap is not None:
+            cap.check()
+        return
+    try:
+        os.lstat(home / MARKER)
+    except FileNotFoundError:
+        if cap is not None:
+            raise ScopeDenied()
+        yield False
+        return
+    if (not home.is_absolute() or '..' in home.parts or held):
+        raise ScopeDenied()  # one managed home at a time; no cross-home lock cycle
+    try:
+        import fcntl
+    except ImportError:
+        raise ScopeDenied() from None
+    fds, links = [], []
+    item = None
+    locked = False
+    try:
+        fds.append(os.open('/', os.O_RDONLY | os.O_DIRECTORY))
+        for name in home.parts[1:]:
+            parent = fds[-1]
+            fd = os.open(name, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=parent)
+            fds.append(fd)
+            links.append((parent, name, _identity(os.fstat(fd))))
+        homefd = fds[-1]
+        _private(os.fstat(homefd), True)
+        if cap is not None and _identity(os.fstat(homefd)) != cap.directory_identity:
+            raise ScopeDenied()
+        marker = os.open(MARKER, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK, dir_fd=homefd)
+        fds.append(marker)
+        _private(os.fstat(marker))
+        marker_identity = _identity(os.fstat(marker))
+        item = os.open('.skills.lock', os.O_RDWR | os.O_CREAT | os.O_NOFOLLOW | os.O_NONBLOCK,
+                       0o600, dir_fd=homefd)
+        _private(os.fstat(item))
+        lock_identity = _identity(os.fstat(item))
+
+        def check(*, authority=True):
+            if authority and cap is not None:
+                cap.check()
+            for parent, name, identity in links:
+                st = os.stat(name, dir_fd=parent, follow_symlinks=False)
+                if not stat.S_ISDIR(st.st_mode) or _identity(st) != identity:
+                    raise ScopeDenied()
+            _private(os.fstat(homefd), True)
+            for name, fd, identity in ((MARKER, marker, marker_identity), ('.skills.lock', item, lock_identity)):
+                st = os.stat(name, dir_fd=homefd, follow_symlinks=False)
+                _private(st); _private(os.fstat(fd))
+                if _identity(st) != identity:
+                    raise ScopeDenied()
+
+        check()
+        fcntl.flock(item, fcntl.LOCK_EX)
+        locked = True
+        check()
+        held[home] = (os.getpid(), check)
+        yield True
+        check()
+    finally:
+        held.pop(home, None)
+        if item is not None:
+            try:
+                if locked:
+                    fcntl.flock(item, fcntl.LOCK_UN)
+            finally:
+                os.close(item)
+        for fd in reversed(fds):
+            os.close(fd)

+def skill_write_window(home, *, admin_only=False, with_arguments=False):
+    """Fence an existing native operation, including its validation and metadata.
+
+    `home` is a call-time resolver for the actual destination, not a cached
+    profile. The body/feature gates and unmarked donor semantics are retained.

+    def decorate(fn):

+                return fn(*args, **kwargs)
+        return fenced
+    return decorate

+def native_mkdir(path, mode=0o777, parents=False, exist_ok=False):
+    path = Path(path)
+    home = managed_output(path)
+    if home is None:
+        return path.mkdir(mode=mode, parents=parents, exist_ok=exist_ok)
+    # Path.mkdir(parents=True, mode=0700) leaves intermediate parents at0777.
+    # Create each explicit output directory privately; never chmod old objects.
+    if path == home:
+        _private(path.lstat(), True)
+        if not exist_ok:
+            raise FileExistsError(str(path))
