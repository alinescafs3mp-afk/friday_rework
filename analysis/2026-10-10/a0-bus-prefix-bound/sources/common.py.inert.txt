class Refused(RuntimeError):

def require(test, message):
    if not test:
        raise Refused(message)


def digest(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def json_pairs(pairs):
    obj = {}
    for key, value in pairs:
        require(key not in obj, 'duplicate JSON key: ' + key)
        obj[key] = value
    return obj


def private_bytes(path, mode=0o600):
    fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW)
    try:
        s = os.fstat(fd)
        require(stat.S_ISREG(s.st_mode) and s.st_uid == os.getuid() and
                s.st_nlink == 1 and stat.S_IMODE(s.st_mode) == mode, 'private file metadata')
        require(s.st_size <= 1024 * 1024, 'private file too large')
        raw = os.read(fd, 1024 * 1024 + 1)
        require(len(raw) == s.st_size, 'file changed during read')
        return raw
    finally:
        os.close(fd)


def read_json(path):
    return json.loads(private_bytes(path), object_pairs_hook=json_pairs)


def write_once(path, obj):
    raw = obj if isinstance(obj, bytes) else (json.dumps(obj, sort_keys=True) + '\n').encode()
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600)
    with os.fdopen(fd, 'wb') as stream:
        stream.write(raw)
        stream.flush()
        os.fsync(stream.fileno())


def clocks():
    return {'monotonic_ns': time.monotonic_ns(),
            'boottime_ns': time.clock_gettime_ns(time.CLOCK_BOOTTIME),
            'realtime_ns': time.time_ns()}


def within(r, seconds, now=None):
    now = clocks() if now is None else now
    return all(0 <= now[k] - r['accepted_' + k] < seconds * 10**9
               for k in ('monotonic_ns', 'boottime_ns'))


def remaining(r, seconds):
    now = clocks()
    return max(0, min(seconds - (now[k] - r['accepted_' + k]) / 1e9
                      for k in ('monotonic_ns', 'boottime_ns')))

def identity(pid):
    raw = Path('/proc/%d/stat' % pid).read_text()
    fields = raw[raw.rfind(')') + 2:].split()
    status = Path('/proc/%d/status' % pid).read_text()
    ignored = next(line.split()[1] for line in status.splitlines() if line.startswith('SigIgn:'))
    return {'pid': pid, 'starttime_ticks': int(fields[19]), 'parent_pid': int(fields[1]),
            'process_group': int(fields[2]), 'session_id': int(fields[3]), 'ignored_signals': int(ignored, 16),
            'state': fields[0], 'cgroup': Path('/proc/%d/cgroup' % pid).read_text()}

def emit(role, event, extra=None):
    require(role in ROLES, 'unknown role')
    item = {'role': role, 'event': event, **clocks(),
            'invocation_id': os.environ.get('INVOCATION_ID'), **identity(os.getpid())}
    r=read_json(EVIDENCE/'receipt.json')
    item['session_binding']={k:r[k] for k in ('assignment','generation','attempt_token','manifest_sha256','boot_id','accepted_monotonic_ns','accepted_boottime_ns')}
    if extra:
        item.update(extra)
    # One emitter per PID, immutable boot/monotonic identity, private directory.
    # Publish only a fully flushed record; observers never parse a partial JSON.
    stem = '%s.%s.%d.%d' % (role, event, os.getpid(), item['monotonic_ns'])
    pending, final = EVIDENCE / ('.' + stem + '.pending'), EVIDENCE / (stem + '.json')
    require(not final.exists(), 'duplicate event identity')
    write_once(pending, item)
    os.rename(pending, final)
    return item

HISTORY_ENV = {**ENV, 'LANG':'C', 'SYSTEMD_LOG_LEVEL':'debug',
               'SYSTEMD_LOG_TARGET':'console', 'SYSTEMD_LOG_COLOR':'0',
               'SYSTEMD_LOG_LOCATION':'0', 'SYSTEMD_LOG_TIME':'0', 'SYSTEMD_LOG_TID':'0'}
HISTORY_PROFILE = 'admission-history-c-debug-console-v1'
CAPTURE_LIMIT = 65536


def history_diagnostics(result, original):
    """Strict v259.5 benign grammar for the one exact admission call.

    No filesystem lookup, journal inventory claim or integrity certification.
    Unknown diagnostics and unsupported but possibly benign variants refuse.
    Source/limits and the silent zero-bucket boundary are pinned in the package.
    """
    require(result['argv'] == history_argv(original), 'diagnostic exact history argv')
    require(result.get('diagnostic_profile') == HISTORY_PROFILE and
            result.get('environment') == HISTORY_ENV, 'diagnostic exact environment')
    require(result.get('capture_complete') is True and
            result.get('stream_eof') == {'stdout':True,'stderr':True} and
            result.get('decode_error') is False and not result.get('capture_error') and
            result['returncode'] == 0 and result['client_reaped'] is True and
            result['timed_out'] is False and result['output_truncated'] is False,
            'incomplete history capture')
    raw = {}
    for name in ('stdout','stderr'):
        encoded = result[name+'_raw_hex']
        require(type(encoded) is str and len(encoded) <= CAPTURE_LIMIT*2 and
                re.fullmatch(r'(?:[0-9a-f]{2})*',encoded) is not None, 'raw capture encoding')
        raw[name] = bytes.fromhex(encoded)
        require(raw[name].decode('utf-8','strict') == result[name], 'raw/text capture mismatch')
    require(sum(map(len,raw.values())) <= CAPTURE_LIMIT, 'history capture byte bound')
    require(raw['stdout'] == b'', 'prior exact-name row or malformed stdout')
    text = raw['stderr'].decode('ascii','strict')
    require(text and text.endswith('\n') and
            all(c == '\n' or ' ' <= c <= '~' for c in text), 'diagnostic encoding/framing')
    lines = text[:-1].split('\n')
    require(len(lines) <= 512 and all(0 < len(line) <= 2048 for line in lines),
            'diagnostic line bounds')
    roots = ('/run/log/journal','/var/log/journal')
    considered, directories, removed, files = [], {}, set(), set()
    pending = None
    settings = False
    have_settings = False
    phase = 'open'
    boot = original['boot_id'].replace('-','')
    found = 'Found boot '+boot+' for '+boot+'+0'
    # Match tree has one OR term for USER_UNIT, and one ANDed boot term.
    # Reordering equal OR operands is harmless; missing/extra/duplicate is not.
    expected_units = {'USER_UNIT='+NAMES[role] for role in ROLES}
    def match_filter(line):
        prefix, suffix = 'Journal filter: ((', ') AND _BOOT_ID='+boot+')'
        require(line.startswith(prefix) and line.endswith(suffix), 'history filter grammar')
        terms = line[len(prefix):-len(suffix)].split(' OR ')
        require(len(terms) == 8 and set(terms) == expected_units, 'exact eight history OR terms')
    def path_kind(path):
        if path in roots:
            return 'root'
        require(re.fullmatch(r'/(?:run|var)/log/journal/[0-9a-f]{32}',path) is not None,
                'ambiguous/unsupported diagnostic directory')
        require(str(Path(path).parent) in directories, 'directory parent not opened')
        return 'directory'
    for line in lines:
        if phase == 'closed':
            raise Refused('diagnostic after teardown')
        if phase == 'filter':
            match_filter(line); phase = 'stats'; continue
        if phase == 'stats':
            m = re.fullmatch(r'mmap cache statistics: (0|[1-9][0-9]{0,19}) category cache hit, '
                r'(0|[1-9][0-9]{0,19}) window list hit, (0|[1-9][0-9]{0,19}) miss, '
                r'(0|[1-9][0-9]{0,19}) files, (0|[1-9][0-9]{0,19}) windows, '
                r'(0|[1-9][0-9]{0,19}) unused',line)
            require(m is not None, 'unknown/error or missing mmap summary')
            counts = tuple(map(int,m.groups()))
            require(all(n < 2**64 for n in counts) and counts[3] == len(files) and
                    counts[5] <= counts[4], 'inconsistent mmap summary')
            phase = 'teardown'; continue
        if phase == 'teardown':
            m = re.fullmatch(r'(Root directory|Directory) ([^ ]+) removed\.',line)
            require(m is not None, 'unknown/error or incomplete directory teardown')
            kind, path = m.groups()
            require(directories.get(path) == kind and path not in removed,
                    'unopened/repeated/mismatched directory teardown')
            removed.add(path)
            if removed == set(directories): phase = 'closed'
            continue
        if line == found:
            require(considered == list(roots) and files and have_settings and not settings and
                    (pending is None or pending[0] == 'Root directory'),
                    'incomplete open/boot diagnostic context')
            pending = None; phase = 'filter'; continue
        m = re.fullmatch(r"Considering (root directory|directory) '([^']+)'\.",line)
        if m:
            require(not settings and (pending is None or pending[0] == 'Root directory'),
                    'incomplete file/directory diagnostic context')
            kind, path = m.groups()
            actual = path_kind(path)
            require((kind == 'root directory') == (actual == 'root'), 'directory class mismatch')
            if actual == 'root':
                require(len(considered) < 2 and path == roots[len(considered)], 'root discovery order')
                considered.append(path)
            else:
                require(path not in directories and considered and
                        str(Path(path).parent) == considered[-1], 'subdirectory discovery context')
            pending = ('Root directory' if actual == 'root' else 'Directory',path)
            continue
        m = re.fullmatch(r'(Root directory|Directory) ([^ ]+) added\.',line)
        if m:
            require(not settings and pending == m.groups() and m[2] not in directories,
                    'unconsidered/duplicate directory')
            directories[m[2]] = m[1]; pending = None; continue
        if re.fullmatch(r'Journal effective settings seal=(?:yes|no) keyed_hash=(?:yes|no) '
                        r'compress=(?:NONE|XZ|LZ4|ZSTD) compress_threshold_bytes=8B',line):
            require(pending is None and directories and not settings,
                    'out-of-context effective settings')
            settings = have_settings = True; continue
        m = re.fullmatch(r'File ([^ ]+) added\.',line)
        if m:
            path = m[1]; parent = str(Path(path).parent)
            require(path == parent + '/' + Path(path).name, 'noncanonical diagnostic file path')
            require(pending is None and have_settings and parent in directories and
                    path not in files and considered and path.startswith(considered[-1]+'/'),
                    'unopened/duplicate/ambiguous diagnostic file')
            require(re.fullmatch(r'user-PRIVATE_OWNER_UID(?:@[0-9a-f]{32}-[0-9a-f]{16}-[0-9a-f]{16})?\.journal~?',
                                Path(path).name) is not None, 'unsupported selected user journal path')
            files.add(path); settings = False; continue
        raise Refused('unknown/error/ignored/malformed history diagnostic')
    require(phase == 'closed' and removed == set(directories), 'incomplete history diagnostic transcript')
    return {'profile':HISTORY_PROFILE,'known_benign_diagnostics':True,
            'physical_corpus_certified':False,'global_history_certified':False,
            'files_observed':len(files),'directories_observed':len(directories)}


def journal_read_allowed(argv, original, now):
    # Two time values are the only variable tokens. Canonical integer seconds
    # range over at most 25 values in the original22s window, never new anchors.
    require(within(original,22,now), 'original journal window')
    require(len(argv) == 19 and type(original.get('accepted_realtime_ns')) is int,
            'finite journal query shape')
    require(re.fullmatch(r'--since=@[1-9][0-9]{0,11}',argv[8]) is not None and
            re.fullmatch(r'--until=@[1-9][0-9]{0,11}',argv[9]) is not None,
            'canonical journal time values')
    since, until = int(argv[8][9:]), int(argv[9][9:])
    floor = original['accepted_realtime_ns']//10**9-1
    ceiling = min(original['accepted_realtime_ns']//10**9+23, now['realtime_ns']//10**9+1)
    require(floor <= since <= until <= ceiling, 'journal time outside original finite range')
    manager = original['user_manager']
    require(type(manager['pid']) is int and manager['pid'] > 1, 'exact manager PID')
    expected = ['/usr/bin/journalctl', '--user', '--quiet', '--no-pager', '--output=json',
                '--output-fields='+','.join(JOURNAL_FIELDS), '--lines=+65',
                '--boot='+original['boot_id'].replace('-', ''), '--since=@%d'%since, '--until=@%d'%until,
                'USER_UNIT='+NAMES['dependent-late'], '_PID='+str(manager['pid']),
                '_UID=PRIVATE_OWNER_UID', '_EXE=/usr/lib/systemd/systemd', '_COMM=systemd',
                'JOB_TYPE=start', *['MESSAGE_ID='+value for value in JOURNAL_JOB_MESSAGE_IDS]]
    require(argv == expected, 'exact journal boot/unit/manager/fields/options')
    return True


def parse_show(text, roles=ROLES):
    blocks = {}
    for block in text.strip().split('\n\n'):
        if not block.strip():
            continue
        pairs = [line.split('=', 1) for line in block.splitlines() if '=' in line]
        item = json_pairs(pairs)
        require(item.get('Id') in [NAMES[r] for r in roles], 'unexpected manager name')
        require(item['Id'] not in blocks, 'duplicate manager block')
        blocks[item['Id']] = item
    require(set(blocks) == {NAMES[r] for r in roles}, 'incomplete manager response')
    return {r: blocks[NAMES[r]] for r in roles}

def cgroup_sample(role, value):
    require(value.startswith('/') and '..' not in Path(value).parts and value.endswith('/' + NAMES[role]), 'cgroup scope')
    path = Path('/sys/fs/cgroup') / value.lstrip('/')
    result = {'path': value, **clocks(), 'exists': path.exists(), 'pids': []}
    if not result['exists']:
        return result
    result['events'] = (path / 'cgroup.events').read_text()
    files = [path / 'cgroup.procs'] + list(path.rglob('*/cgroup.procs'))
    for f in files:
        try:
            pids = [int(x) for x in f.read_text().split()]
        except FileNotFoundError:
            continue
        for pid in pids:
            try:
                result['pids'].append(identity(pid))
            except FileNotFoundError:
                result.setdefault('vanished_during_sample', []).append(pid)
    return result

def _bounded_capture(argv, timeout, *, popen, absolute):
    """64KiB raw-prefix capture under the original dual deadline/reap reserve.

    Always retain raw bytes, including malformed UTF-8, partial reads and
    exceptions. Only the exact admission history record selects debug logging.
    This function never classifies or suppresses stderr.
    """
    require(timeout > 0.12, 'no capture/reap reserve')
    begin = clocks()
    deadline = {a:absolute['deadline_'+a] for a in ('monotonic_ns','boottime_ns')}
    def left():
        now=clocks()
        return min((deadline[a]-now[a])/1e9 for a in deadline)
    profile = absolute.get('diagnostic_profile')
    environment = dict(HISTORY_ENV if profile == HISTORY_PROFILE else ENV)
    limit, used = CAPTURE_LIMIT, 0
    output = {'stdout':bytearray(),'stderr':bytearray()}
    eof = {'stdout':False,'stderr':False}
    timed_out = overflow = False
    p = None
    result = {'argv':list(argv),'started':begin,'environment':environment,
              'diagnostic_profile':profile,'capture_error':None}
    try:
        p = popen(argv,stdin=subprocess.DEVNULL,stdout=subprocess.PIPE,
                  stderr=subprocess.PIPE,env=environment,close_fds=True)
        with selectors.DefaultSelector() as selector:
            for name,stream in (('stdout',p.stdout),('stderr',p.stderr)):
                os.set_blocking(stream.fileno(),False)
                selector.register(stream,selectors.EVENT_READ,name)
            while selector.get_map():
                budget=left()-0.1
                if budget <= 0:
                    timed_out=True; break
                for key,_ in selector.select(budget):
                    chunk=os.read(key.fileobj.fileno(),min(4096,limit-used+1))
                    if not chunk:
                        eof[key.data]=True
                        selector.unregister(key.fileobj); continue
                    retained=chunk[:limit-used]
                    output[key.data].extend(retained); used+=len(retained)
                    if len(retained) != len(chunk):
                        overflow=True; break
                if overflow: break
        if not (timed_out or overflow):
            try:
                p.wait(timeout=max(0,left()-0.1))
            except subprocess.TimeoutExpired:
                timed_out=True
        if p.poll() is None:
            p.kill()
        if p.returncode is None and left()>0:
            try: p.wait(timeout=left())
            except subprocess.TimeoutExpired: timed_out=True
    except BaseException as error:
        result['capture_error'] = type(error).__name__+': '+str(error)[:1024]
        raise
    finally:
        try:
            if p is not None:
                if p.poll() is None: p.kill()
                if p.returncode is None and left()>0: p.wait(timeout=left())
        except BaseException as error:
            result['capture_error'] = type(error).__name__+': '+str(error)[:1024]
        finally:
            result.update({'finished':clocks(),'returncode':None if p is None else p.returncode,
                'timed_out':timed_out,'output_truncated':overflow,
                'client_reaped':p is not None and p.returncode is not None,
                'capture_complete':all(eof.values()) and not timed_out and not overflow and
                                   result['capture_error'] is None,
                'stream_eof':dict(eof),'decode_error':False})
            for name,raw in output.items():
                result[name+'_raw_hex'] = bytes(raw).hex()
                try: result[name] = bytes(raw).decode('utf-8','strict')
                except UnicodeError:
                    result[name] = None
                    result['decode_error'] = True
            # The original registry owns this snapshot even if capture raises.
            absolute['capture'] = result
            if p is not None:
                p.stdout.close(); p.stderr.close()
    return result


def bounded_capture_call(argv, timeout, *, expected='read', popen=None):
    return call(argv,timeout,expected=expected,popen=popen)

class ParentTransport:
    """Finite in-process custody, independent of Driver JSON and booleans.

    Every attempted native call is recorded before Popen; dispatch waits for
    actual Popen/starttime/pidfd registration. A failed/denied/late call latches
    uncertainty forever, even if its child is subsequently reaped. Only exact
    read clients and the two reviewed reloads remain after closing Driver.
    """
    def __init__(self, original, start_argv, clock=clocks, popen=subprocess.Popen):
        import copy
        self.original = copy.deepcopy(original)
        self.start_argv = tuple(tuple(a) for a in start_argv)
        self.clock, self.popen = clock, popen
        self.parent_pid = os.getpid()
        self.parent_identity = identity(self.parent_pid)
        self.records = []
        self.phase = 'admission'
        self.uncertain = False
        self.driver_started = self.driver_returned = False
        self.reload_count = 0
        self.journal_reads = 0
        self.starts = {}
        self.busy = False
        self.current = None
        self.closed_at = None

    def left(self, seconds):
        now = self.clock()
        require(within(self.original, seconds, now), 'original transport clock expired/future')
        return min(seconds-(now[k]-self.original['accepted_'+k])/1e9
                   for k in ('monotonic_ns','boottime_ns'))

    def begin_driver(self):
        self.left(5)
        require(self.phase == 'admission' and not self.uncertain and not self.busy,
                'driver entry refuses open/unknown admission')
        self.phase = 'driver'
        self.driver_started = True

    def close_driver(self):
        # Called in the owner's finally, after the synchronous callable has
        # returned/unwound. There is no detached Driver actor or FD9 duplicate.
        require(self.phase == 'driver' and self.driver_started, 'single driver closure')
        self.phase = 'cleanup'
        self.closed_at = self.clock()
        if self.busy or not within(self.original,110,self.closed_at):
            self.uncertain = True
        self.driver_returned = True

    def refuse_setup(self):
        require(self.phase == 'admission' and not self.driver_started, 'setup refusal phase')
        self.phase = 'setup_refused'
        self.uncertain = True  # fail closed; original parent custody is retained

    def classify(self, argv, expected, *, direct_job=False):
        require(type(argv) is list and all(type(a) is str for a in argv), 'exact argv')
        require(expected in ('read','start','reload'), 'caller expected effect class')
        require(not self.uncertain and not self.busy, 'uncertain/reentrant native transport')
        if expected == 'start':
            require(tuple(argv) in self.start_argv, 'exact approved start argv')
            require(direct_job == ('--unit='+NAMES['dependent-late'] in argv), 'late role requires single direct job, never CLI resubmission')
            require(self.phase == 'driver' and not self.driver_returned, 'future submitter closed')
            self.left(35)
            key = tuple(argv)
            count = self.starts.get(key,0)
            maximum = 2 if '--unit='+NAMES['outer-bound'] in argv else 1
            require(count < maximum, 'native start replay')
            self.starts[key] = count+1
            return 'start', 35
        if expected == 'reload':
            require(argv == RELOAD_ARGV, 'exact approved reload argv')
            require((self.phase,self.reload_count) in (('admission',0),('cleanup_reload',1)),
                    'unexpected/repeated reload')
            self.reload_count += 1
            return 'reload', 5 if self.phase == 'admission' else 120
        require(self.phase in ('admission','driver','cleanup','cleanup_reload','closed'), 'closed native lane')
        fixed = [show_argv(), *[show_argv((role,)) for role in ROLES], MANAGER_SHOW, JOBS_ARGV]
        if self.phase == 'admission':
            fixed += [PREFLIGHT_MANAGER, PREFLIGHT_UNITS, UNIT_PATH_ARGV]
        require(argv in fixed, 'complete approved read argv; archive never admission/result authority')
        return 'read', 5 if self.phase == 'admission' else (108 if self.phase == 'driver' else 120)

    def spawn(self, argv, **kwargs):
        r = self.current
        require(r is not None and r['argv'] == argv and r['process'] is None, 'registered single Popen')
        rd, wr = os.pipe2(os.O_CLOEXEC)
        try:
            p = self.popen(['/usr/bin/python3','-I','-S','-B','-c',EXEC_LATCH,
                            str(rd),str(r['deadline_monotonic_ns']),str(r['deadline_boottime_ns']),*argv],
                           pass_fds=(rd,), **kwargs)
            r['process'] = p  # Custody survives identity/read/dispatch exceptions.
            r['pidfd'] = os.pidfd_open(p.pid)
            ident = identity(p.pid)
            require(ident['parent_pid'] == self.parent_pid and ident['state'] not in ('Z','X'), 'owned native client identity')
            r['identity'] = ident
            r['registered_at'] = self.clock()
            require(all(r['registered_at'][a] < r['deadline_'+a] for a in ('monotonic_ns','boottime_ns')), 'late native registration')
            r['dispatch_attempted'] = True
            require(os.write(wr,b'1') == 1, 'native dispatch latch write')
            return p
        finally:
            os.close(rd)
            os.close(wr)

    def call(self, argv, timeout, *, expected='read'):
        kind, seconds = self.classify(argv, expected)
        is_history = self.phase == 'admission' and expected == 'read' and argv == history_argv(self.original)
        left = min(float(timeout), self.left(seconds), 1.0 if is_history else float(timeout))
        require(left > 0.12 and self.parent_pid == os.getpid() and
                identity(self.parent_pid)['starttime_ticks'] == self.parent_identity['starttime_ticks'],
                'native call needs finite capture/reap reserve and original parent')
        at = self.clock()
        r = {'sequence':len(self.records),'argv':list(argv),'kind':kind,'expected':expected,'phase':self.phase,
             'started':at,'process':None,'pidfd':None,'identity':None,
             'dispatch_attempted':False,'reaped':False,'uncertain':True,
             'diagnostic_profile':HISTORY_PROFILE if is_history else None,
             **{'deadline_'+a:min(at[a]+int(left*1e9),self.original['accepted_'+a]+seconds*10**9)
                for a in ('monotonic_ns','boottime_ns')}}
        self.records.append(r)
        self.current, self.busy = r, True
        try:
            result = _bounded_capture(argv,left,popen=self.spawn,absolute=r)
            # Retain the exact bounded capture before ANY acceptance predicate.
            # A rc0 warning/timeout/truncation/late result remains diagnostic
            # evidence on refusal, never evidence of admission or cessation.
            r['capture'] = result
            p = r['process']
            require(p is not None and r['identity'] is not None and r['dispatch_attempted'], 'native custody absent')
            # Never consume result['client_reaped'] as proof; use owned Popen.
            r['reaped'] = p.returncode is not None
            r['returncode'] = p.returncode
            allowed = (0,)  # every denied transport remains unknown, including read clients
            r['finished'] = self.clock()
            require(r['reaped'] and result['returncode'] == p.returncode and p.returncode in allowed and
                    not result['timed_out'] and not result['output_truncated'] and
                    result.get('capture_complete') is True and result.get('decode_error') is False and
                    not result.get('capture_error') and
                    all(r['finished'][a] < r['deadline_'+a] for a in ('monotonic_ns','boottime_ns')),
                    'denied/unknown/late native transport')
            if is_history:
                r['diagnostic_classification'] = history_diagnostics(result,self.original)
            else:
                require(not result['stderr'], 'any stderr from ordinary native call refuses')
            if hasattr(self,'native_observer'): self.native_observer(result)
            r['uncertain'] = False
            return result
        finally:
            p = r['process']
            if p is not None:
                # A capture exception may occur after Popen but before returning.
                # Finite finalization never fabricates a reap or clears uncertainty.
                try:
                    if p.poll() is None:
                        p.kill()
                    budget = min((r['deadline_'+a]-self.clock()[a])/1e9 for a in ('monotonic_ns','boottime_ns'))
                    if p.returncode is None and budget > 0:
                        p.wait(timeout=budget)
                except BaseException:
                    pass
                r['reaped'] = p.returncode is not None
                r['returncode'] = p.returncode
                if r['reaped'] and r['pidfd'] is not None:
                    os.close(r['pidfd']); r['pidfd'] = None
            if r['uncertain'] or not r['reaped']:
                self.uncertain = True
            final=self.clock()
            if not all(final[a]<r['deadline_'+a] for a in ('monotonic_ns','boottime_ns')):
                r['uncertain']=True; self.uncertain=True
            self.current, self.busy = None, False
            require(all(final[a]<r['deadline_'+a] for a in ('monotonic_ns','boottime_ns')),
                    'native finalization exceeded original call deadline')

    def inventory(self):
        return [{k:v for k,v in r.items() if k not in ('process','pidfd')} for r in self.records]

    def confirm_closed(self):
        self.left(120)
        require(self.driver_returned and self.closed_at is not None and
                within(self.original,110,self.closed_at) and
                self.phase in ('cleanup','cleanup_reload','closed') and
                not self.uncertain and not self.busy and bool(self.records), 'submission inventory unknown/open')
        for r in self.records:
            require(not r['uncertain'] and r['reaped'] and r['process'] is not None and
                    r['process'].returncode == r['returncode'] and r['identity'] is not None and
                    r['dispatch_attempted'], 'actual native client not reaped/registered')
        require(hasattr(self,'job_bus') and self.job_bus.closed and self.job_bus.failed is None and
                self.job_bus.fd is not None and self.job_bus.fd_closed and self.job_bus.no_reference_acquired and
                self.job_bus.pending is None and self.job_bus.correlation is not None and
                self.job_bus.correlation.state == 'TERMINAL', 'direct bus call/event custody not closed')
        return self.inventory()

class JobCorrelation:
    """One returned job, bounded early events, irreversible uncertainty."""
    def __init__(self, owner, unit, original, clock=clocks):
        require(type(owner) is str and re.fullmatch(r':[0-9]+\.[0-9]+', owner), 'unique manager owner')
        self.owner, self.unit, self.original, self.clock = owner, unit, dict(original), clock
        self.state, self.failure = 'UNSENT', None
        self.path = self.serial = self.terminal = None
        self.events = []
        self.sent_at = None

    def fail(self, why):
        self.state, self.failure = 'UNKNOWN', why
        raise Refused(why)

    def gate(self):
        if self.failure is not None: self.fail(self.failure)
        if not within(self.original, 22, self.clock()): self.fail('original late dual deadline')

    def begin(self):
        self.gate()
        if self.state != 'UNSENT': self.fail('no repeat after attempted submission')
        self.state, self.sent_at = 'ATTEMPTED', self.clock()

    def reply(self, owner, serial, path):
        self.gate()
        if self.state != 'ATTEMPTED' or owner != self.owner or type(serial) is not int or serial <= 0:
            self.fail('foreign/duplicate method reply')
        if type(path) is not str or re.fullmatch(r'/org/freedesktop/systemd1/job/[1-9][0-9]*', path) is None:
            self.fail('invalid returned job path')
        if int(path.rsplit('/',1)[1]) > 0xffffffff: self.fail('job ID overflow')
        self.serial, self.path, self.state = serial, path, 'REPLIED'
        for item in self.events:
            if item['path'] != path: self.fail('stale/wrong job for exact unit')
        self._terminal()

    def event(self, owner, path, job_id, unit, result):
        self.gate()
        if self.state not in ('ATTEMPTED','REPLIED','TERMINAL'): self.fail('terminal before current dispatch')
        if owner != self.owner or unit != self.unit: self.fail('foreign sender/unit')
        if type(job_id) is not int or not 0 < job_id <= 0xffffffff or path != BUS_PATH+'/job/'+str(job_id):
            self.fail('job path/ID mismatch')
        if result not in ('done','canceled','timeout','failed','dependency','skipped'):
            self.fail('unknown typed job result')
        if self.events: self.fail('duplicate/conflicting terminal')
        self.events.append({'owner':owner,'path':path,'job_id':job_id,'unit':unit,
                            'result':result,'received_at':self.clock()})
        if self.path is not None and path != self.path: self.fail('wrong returned job')
        self._terminal()

    def _terminal(self):
        if self.path is not None and self.events:
            self.terminal = {**self.events[0], 'request_serial':self.serial,
                'submitted_at':self.sent_at, 'protocol':JOB_PROTOCOL,
                'assignment':self.original['assignment'],'generation':self.original['generation'],
                'attempt_token':self.original['attempt_token'],'boot_id':self.original['boot_id']}
            self.state = 'TERMINAL'

                self.lock_check(),'original bus parent/FD9/clock/uncertainty')

             self.select_api,self.struct_api)=trusted_jeepney_imports()
            self.gate(4)
            self.parser=self.library.Parser();self.gate(4)
            self.prepared=True;self.prefix_state='PREPARED_NO_SOCKET'
        except BaseException as error:
            self.failed=type(error).__name__+': '+str(error)[:512];self.transport.uncertain=True
            self.cleanup_socket();raise

    def io(self, deadline, seconds, operation, *args):
        self.gate(seconds)
        left=min((deadline[a]-self.clock()[a])/1e9 for a in deadline)
        require(left>0,'original socket operation deadline; no send')
        self.sock.settimeout(left)  # CPython timeout bounds connect/sendall/recv.
        require(all(self.clock()[a]<deadline[a] for a in deadline),'deadline after timeout setup; no operation')
        if self.pending is not None and operation==self.sock.sendall:self.pending['send_started']=True
        result=operation(*args)
        self.gate(seconds)
        require(all(self.clock()[a]<deadline[a] for a in deadline),'late socket operation')
        return result

    def phase_deadline(self, seconds, cap=None):
        self.gate(seconds);at=self.clock()
        return {a:min(self.transport.original['accepted_'+a]+seconds*10**9,
                      at[a]+int(cap*1e9) if cap is not None else 2**63-1)
                for a in ('monotonic_ns','boottime_ns')}

    def connect_auth(self):
        self.gate(5);require(self.prepared and self.sock is None,'one owned socket')
        deadline=self.phase_deadline(5)
        # socket() has no connect/authentication prefix; NONBLOCK/CLOEXEC at birth.
        # Assign the returned object BEFORE any connect/auth/peer query or send.
        self.sock=self.socket_api.socket(self.socket_api.AF_UNIX,
            self.socket_api.SOCK_STREAM|self.socket_api.SOCK_NONBLOCK|self.socket_api.SOCK_CLOEXEC)
        self.prefix_state='SOCKET_OWNED_BEFORE_CONNECT'
        self.fd=self.sock.fileno()
        require(type(self.fd) is int and self.fd>=0 and self.fd not in (8,9),'owned socket FD, no custody FD reuse')
        st=os.fstat(self.fd);require(stat.S_ISSOCK(st.st_mode),'actual socket FD')
        self.fd_identity=(st.st_dev,st.st_ino,st.st_mode)
        self.io(deadline,5,self.sock.connect,'/run/user/PRIVATE_OWNER_UID/bus')
        cred=self.sock.getsockopt(self.socket_api.SOL_SOCKET,self.socket_api.SO_PEERCRED,12)
        pid,uid,gid=self.struct_api.unpack('3i',cred)
        require(pid>0 and uid==PRIVATE_OWNER_UID,'local bus kernel peer credentials')
        self.peer={'pid':pid,'uid':uid,'gid':gid,'starttime_ticks':identity(pid)['starttime_ticks']}
        self.gate(5);self.prefix_state='CONNECTED_BEFORE_AUTH'
        auth=self.auth_type(enable_fds=False,inc_null_byte=False)
        self.io(deadline,5,self.sock.sendall,b'\0')
        exchanges=0;received=0
        for data in auth:
            exchanges+=1;require(exchanges<=128,'bounded SASL exchanges')
            if data:self.io(deadline,5,self.sock.sendall,data)
            incoming=self.io(deadline,5,self.sock.recv,1024)
            require(incoming,'disconnect during authentication');received+=len(incoming)
            require(received<=8192,'bounded complete SASL response')
            auth.feed(incoming)
        require(auth.authenticated,'EXTERNAL authentication required')
        self.io(deadline,5,self.sock.sendall,self.begin)
        self.prefix_state='AUTHENTICATED_BEFORE_HELLO'

    def receive(self, deadline, seconds):
        # Library parser receives exactly requested bytes; reject oversized frame
        # BEFORE buffering its body. No negotiated or accepted ancillary FDs.
        for _ in range(128):
            self.gate(seconds)
            desired=self.parser.bytes_desired()
            require(type(desired) is int and 0<=desired<=65536 and
                    self.parser.buf.bytes_buffered+desired<=65536,'finite DBus frame')
            m=self.parser.get_next_message()
            if m is not None:
                self.frames+=1;require(self.frames<=1024,'finite total message inventory')
                H=self.library.HeaderFields
                require(m.header.fields.get(H.unix_fds,0)==0,'no ancillary descriptor protocol')
                return m
            require(desired>0,'parser made no progress')
            raw=self.io(deadline,seconds,self.sock.recv,min(desired,4096))
            if not raw:self.fail('bus disconnected')
            self.parser.add_data(raw)
        self.fail('bounded frame dispatch exceeded')

    def signal(self, message):
        try:
            L=self.library;H=L.HeaderFields;f=message.header.fields
            if message.header.message_type!=L.MessageType.signal:self.fail('unsolicited non-signal/reply')
            if f.get(H.interface)=='org.freedesktop.DBus' and f.get(H.member)=='NameAcquired':
                require(f.get(H.sender)=='org.freedesktop.DBus' and f.get(H.path)=='/org/freedesktop/DBus' and
                        f.get(H.signature)=='s' and len(message.body)==1 and type(message.body[0]) is str and
                        re.fullmatch(r':[0-9]+\.[0-9]+',message.body[0]) and self.name_acquired is None and
                        (self.unique_name is None or message.body[0]==self.unique_name),'exact one Hello NameAcquired')
                self.name_acquired=message.body[0];return
            if f.get(H.interface)=='org.freedesktop.DBus' and f.get(H.member)=='NameLost':
                self.fail('bus connection unique name lost')
            if f.get(H.interface)=='org.freedesktop.DBus' and f.get(H.member)=='NameOwnerChanged':
                require(f.get(H.sender)=='org.freedesktop.DBus' and f.get(H.path)=='/org/freedesktop/DBus' and
                        f.get(H.signature)=='sss' and len(message.body)==3,'authenticated owner-change signal')
                if message.body[0]==BUS_NAME:self.fail('manager unique owner changed')
                return
            if f.get(H.interface)=='org.freedesktop.DBus.Local' and f.get(H.member)=='Disconnected':
                self.fail('bus disconnect signal')
            require(f.get(H.interface)==BUS_IFACE and f.get(H.member)=='JobRemoved' and
                    f.get(H.path)==BUS_PATH and f.get(H.sender)==self.owner and
                    f.get(H.signature)=='uoss' and len(message.body)==4,'authenticated exact JobRemoved')
            self.events+=1
            if self.events>128:self.fail('finite job event overflow')
            job_id,path,unit,result=message.body
            if unit!=NAMES['dependent-late']:return
            require(self.correlation is not None,'same-unit terminal before current dispatch')
            self.correlation.event(self.owner,path,job_id,unit,result)
        except BaseException as error:
            self.failed=type(error).__name__+': '+str(error)[:512];self.transport.uncertain=True
            self.cleanup_socket();raise

    def pump(self, seconds):
        try:
            self.gate(seconds)
            # No model/mail polling. This is finite event dispatch inside original
            # existing Driver observation, using native socket readiness once/frame.
            for _ in range(128):
                self.gate(seconds)
                if self.parser.buf.bytes_buffered==0 and not self.select_api.select([self.sock],[],[],0)[0]:return
                deadline=self.phase_deadline(seconds,.5)
                self.signal(self.receive(deadline,seconds))
            self.fail('bus dispatch overflow')
        except BaseException as error:
            self.failed=type(error).__name__+': '+str(error)[:512];self.transport.uncertain=True
            self.cleanup_socket();raise

    def rpc(self, dest, path, iface, method, signature='', args=(), seconds=5, effect=False):
        self.gate(seconds);require(not self.in_call and self.pending is None,'single finite bus call')
        require(len(self.calls)<64 and seconds in (5,22,110),'original phase/call bounds')
        if dest=='org.freedesktop.DBus':
            require(path=='/org/freedesktop/DBus' and iface=='org.freedesktop.DBus' and not effect,'exact bus daemon route')
            require((method=='Hello' and signature=='' and args==() and self.unique_name is None and self.owner is None) or
                    (method=='GetNameOwner' and signature=='s' and args==(BUS_NAME,)) or
                    (method in ('GetConnectionUnixProcessID','GetConnectionUnixUser') and not self.subscribed and
                     signature=='s' and args==(self.owner,)) or
                    (method=='AddMatch' and not self.subscribed and signature=='s' and args in ((self.match,),
                     ("type='signal',sender='org.freedesktop.DBus',interface='org.freedesktop.DBus',member='NameOwnerChanged',arg0='"+BUS_NAME+"'",))),
                    'no generic/repeated daemon calls')
        else:
            require(dest==self.owner and path==BUS_PATH,'pinned unique manager only')
            require((method=='Get' and iface=='org.freedesktop.DBus.Properties' and not self.subscribed and not effect and
                     signature=='ss' and args==(BUS_IFACE,'Version')) or
                    (method=='Subscribe' and iface==BUS_IFACE and not self.subscribed and not effect and signature=='' and args==()) or
                    (method=='StartTransientUnit' and iface==BUS_IFACE and effect and seconds==22 and
                     self.correlation is not None and self.correlation.state=='ATTEMPTED' and
                     not any(x['method']=='StartTransientUnit' for x in self.calls) and
                     signature=='ssa(sv)a(sa(sv))' and len(args)==4 and args[:2]==(NAMES['dependent-late'],'fail')),
                    'no generic manager/ref/unref/repeated start')
        deadline=self.phase_deadline(seconds,.5)
        record={'method':method,'destination':dest,'path':path,'started':self.clock(),'effect':effect,
                'attempted':False,'send_started':False,'finished':None,'serial':None,'reply_sender':None,'unknown':True}
        self.calls.append(record);self.in_call=True
        try:
            self.audit({'bus_call_before_send':dict(record)})
            L=self.library;H=L.HeaderFields
            m=L.new_method_call(L.DBusAddress(path,bus_name=dest,interface=iface),method,signature,tuple(args))
            self.serial+=1;require(self.serial<=0xffffffff,'serial bound')
            raw=m.serialise(serial=self.serial);require(len(raw)<=65536,'finite outgoing message')
            record['serial']=self.serial
            self.gate(seconds)
            require(all(self.clock()[a]<deadline[a] for a in deadline),'audit/encoding deadline; no send')
            # A plain protected local record replaces libdbus PendingCall. There
            # is no callback or detached library task to cancel; retain on loss.
            self.pending=record;record['attempted']=True
            self.io(deadline,seconds,self.sock.sendall,raw)
            for _ in range(128):
                reply=self.receive(deadline,seconds);f=reply.header.fields
                if reply.header.message_type==L.MessageType.signal:self.signal(reply);continue
                require(reply.header.message_type==L.MessageType.method_return and f.get(H.sender)==dest and
                        f.get(H.reply_serial)==self.serial,'foreign/error/duplicate/lost method reply')
                expected={'Hello':'s','GetNameOwner':'s','GetConnectionUnixProcessID':'u','GetConnectionUnixUser':'u',
                          'AddMatch':'','Subscribe':'','StartTransientUnit':'o','Get':'v'}[method]
                require(f.get(H.signature,'')==expected,'exact method return signature')
                record.update(finished=self.clock(),reply_sender=dest,reply_signature=expected,unknown=False)
                values=reply.body
                self.audit({'bus_call_completed':dict(record)})
                self.gate(seconds)
                require(all(self.clock()[a]<deadline[a] for a in deadline),'late reply finalization')
                self.pending=None;return values,self.serial
            self.fail('finite method reply/event overflow')
        except BaseException as error:
            record['unknown']=True;self.failed=type(error).__name__+': '+str(error)[:512];self.transport.uncertain=True
            self.cleanup_socket();raise
        finally:self.in_call=False

    def open(self):
        try:
            self.connect_auth()
            values,_=self.rpc('org.freedesktop.DBus','/org/freedesktop/DBus','org.freedesktop.DBus','Hello')
            require(len(values)==1 and type(values[0]) is str and re.fullmatch(r':[0-9]+\.[0-9]+',values[0]),'one unique Hello')
            self.unique_name=values[0]
            require(self.name_acquired is None or self.name_acquired==self.unique_name,'Hello/NameAcquired join')
            values,_=self.rpc('org.freedesktop.DBus','/org/freedesktop/DBus','org.freedesktop.DBus','GetNameOwner','s',(BUS_NAME,))
            require(len(values)==1 and type(values[0]) is str and re.fullmatch(r':[0-9]+\.[0-9]+',values[0]),'unique manager owner')
            self.owner=values[0]
            for method,wanted in [('GetConnectionUnixProcessID',self.manager['pid']),('GetConnectionUnixUser',PRIVATE_OWNER_UID)]:
                values,_=self.rpc('org.freedesktop.DBus','/org/freedesktop/DBus','org.freedesktop.DBus',method,'s',(self.owner,))
                require(len(values)==1 and type(values[0]) is int and values[0]==wanted,'authenticated manager PID/UID')
            values,_=self.rpc(self.owner,BUS_PATH,'org.freedesktop.DBus.Properties','Get','ss',(BUS_IFACE,'Version'))
            require(tuple(values)==(('s','259.5-0ubuntu3.4'),),'installed native manager Version variant')
            self.match="type='signal',sender='"+self.owner+"',path='"+BUS_PATH+"',interface='"+BUS_IFACE+"',member='JobRemoved'"
            for rule in (self.match,"type='signal',sender='org.freedesktop.DBus',interface='org.freedesktop.DBus',member='NameOwnerChanged',arg0='"+BUS_NAME+"'"):
                self.rpc('org.freedesktop.DBus','/org/freedesktop/DBus','org.freedesktop.DBus','AddMatch','s',(rule,))
            self.rpc(self.owner,BUS_PATH,BUS_IFACE,'Subscribe');self.subscribed=True
            self.verify_owner(5);self.prefix_state='SUBSCRIBED_READY';return self
        except BaseException as error:
            self.failed=type(error).__name__+': '+str(error)[:512];self.transport.uncertain=True
            self.cleanup_socket();raise

    def verify_owner(self, seconds):
        self.gate(seconds)
        values,_=self.rpc('org.freedesktop.DBus','/org/freedesktop/DBus','org.freedesktop.DBus','GetNameOwner','s',(BUS_NAME,),seconds=seconds)
        require(tuple(values)==(self.owner,),'same unique manager owner')

    def submit(self, argv):
        self.gate(22);require(self.subscribed and self.correlation is None,'one pre-subscribed current direct job')
        props=direct_properties(argv,NAMES,ROOT);require('AddRef' not in props,'original private-route acquires no reference')
        self.verify_owner(22)
        kind,_=self.transport.classify(argv,'start',direct_job=True)
        self.correlation=JobCorrelation(self.owner,NAMES['dependent-late'],self.transport.original,self.clock)
        self.correlation.begin()
        try:
            self.audit({'direct_job_attempted':{'argv':argv,'at':self.clock(),'kind':kind,'protocol':JOB_PROTOCOL}})
            typed=[(key,(sig,bytes(value) if sig=='ay' else value)) for key,(sig,value) in props.items()]
            values,serial=self.rpc(self.owner,BUS_PATH,BUS_IFACE,'StartTransientUnit','ssa(sv)a(sa(sv))',
                (NAMES['dependent-late'],'fail',typed,[]),seconds=22,effect=True)
            require(len(values)==1 and type(values[0]) is str,'typed returned job path')
            self.correlation.reply(self.owner,serial,values[0])
            self.verify_owner(22)
            return {'protocol':JOB_PROTOCOL,'job_path':self.correlation.path,'request_serial':serial,'argv':argv}
        except BaseException as error:
            self.failed=type(error).__name__+': '+str(error)[:512];self.transport.uncertain=True
            self.cleanup_socket();raise

    def terminal(self):
        self.pump(22);self.correlation.gate();return self.correlation.terminal

    def cleanup_socket(self):
        # This is cleanup only, independent of clocks, gate or owner verification.
        # No retry of possibly reused descriptor. Unknown job/dispatch stays unknown.
        if self.close_attempted:return
        self.close_attempted=True
        try:
            if self.sock is not None:
                if self.fd_identity is not None:
                    st=os.fstat(self.fd)
                    require((st.st_dev,st.st_ino,st.st_mode)==self.fd_identity,'socket identity changed; retain handle')
                self.sock.close()
                require(self.sock.fileno()==-1,'socket object closure unconfirmed')
                if self.fd is not None:
                    try:os.fstat(self.fd)
                    except OSError as error:
                        import errno
                        require(error.errno==errno.EBADF,'socket descriptor closure unknown');self.fd_closed=True
                    else:raise Refused('old FD remains open/reused; never close by number')
            self.closed=True
        except BaseException as error:
            self.cleanup_errors.append(type(error).__name__+': '+str(error)[:512])
            self.transport.uncertain=True
            if self.failed is None:self.failed='socket cleanup unconfirmed'

    def close(self):
        problem=None
        try:
            self.transport.left(110)
            if not self.closed and self.failed is None and self.pending is None:self.verify_owner(110)
        except BaseException as error:
            problem=error;self.failed=type(error).__name__+': '+str(error)[:512];self.transport.uncertain=True
        finally:self.cleanup_socket()
        if problem is not None:raise problem
        if not (self.closed and self.fd is not None and self.fd_closed and not self.cleanup_errors and
                self.pending is None and self.failed is None and not self.transport.uncertain and
                self.correlation is not None and self.correlation.state=='TERMINAL' and self.no_reference_acquired):
            self.fail('socket closure is not known native outcome/cessation')
        require(self.closed and self.fd is not None and self.fd_closed and not self.cleanup_errors and
                self.pending is None and self.failed is None and not self.transport.uncertain and
                self.correlation is not None and self.correlation.state=='TERMINAL' and self.no_reference_acquired,
                'socket closure is not known native outcome/cessation')
        self.audit({'direct_bus_closed':self.inventory()})

    def inventory(self):
        return {'parent_pid':self.transport.parent_pid,'owner':self.owner,'manager':self.manager,
                'kernel_bus_peer':self.peer,'socket_fd':self.fd,'socket_identity':self.fd_identity,
                'socket_closed':self.fd_closed,'closed':self.closed,'failed':self.failed,
                'close_attempted':self.close_attempted,'cleanup_errors':self.cleanup_errors,
                'pending_call':self.pending,'calls':self.calls,'events':self.events,'frames':self.frames,
                'prefix_state':self.prefix_state,'library_sha256':JEEPNEY_WHEEL_SHA,
                'protocol':JOB_PROTOCOL,'no_reference_acquired':self.no_reference_acquired,
                'job_state':None if self.correlation is None else self.correlation.state,
                'terminal':None if self.correlation is None else self.correlation.terminal,
                'native_job_cessation_not_inferred_from_bus_close':True}
