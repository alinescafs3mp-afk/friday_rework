"""One-shot launch gate. No Hermes imports, retries, shell loop or recovery.

Only the checked supervisor releases the fixed payload. The gate also checks
the actual mounted directory identities before bash sees command bytes.
"""
import hashlib
import json
import os
from pathlib import Path
import time

CONTROL = Path('/.hermes-terminal-control')
READY = Path('/.hermes-terminal-ready/ready.json')


def run():
    raw = (CONTROL / 'payload.json').read_bytes()
    if len(raw) > 2_000_000:
        raise RuntimeError('terminal_gate_payload_bound')
    payload = json.loads(raw)
    pin = hashlib.sha256(raw).hexdigest()
    for row in payload['mounts']:
        p = Path(row['path'])
        st = p.stat()
        if p.resolve() != p or [st.st_dev, st.st_ino] != row['identity']:
            raise RuntimeError('terminal_gate_mount_changed')
    if os.environ.get('DBUS_SESSION_BUS_ADDRESS') or os.environ.get('SSH_AUTH_SOCK'):
        raise RuntimeError('terminal_gate_ambient_credential')
    ready = {'nonce': payload['nonce'], 'payload_sha256': pin, 'mounts': payload['mounts']}
    fd = os.open(READY, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600)
    with os.fdopen(fd, 'w') as f:
        json.dump(ready, f); f.flush(); os.fsync(f.fileno())
    while time.monotonic() < payload['startup_deadline']:
        if (CONTROL / 'STOP').exists():
            raise RuntimeError('terminal_gate_stopped')
        permit = CONTROL / 'permit.json'
        if permit.exists():
            if json.loads(permit.read_bytes()) != {'nonce': payload['nonce'], 'payload_sha256': pin}:
                raise RuntimeError('terminal_gate_invalid_permit')
            if time.monotonic() >= payload['command_deadline'] or (CONTROL / 'STOP').exists():
                raise RuntimeError('terminal_gate_deadline_or_stop')
            # stdin/PTY remains the native service transport; READY was never
            # mixed with user output and command input was never a shell gate.
            os.chdir(payload['cwd'])
            os.execve('/bin/bash', ['/bin/bash', '--noprofile', '--norc', '-c', payload['command']],
                      payload['environment'])
        time.sleep(.01)
    raise TimeoutError('terminal_gate_startup_deadline')


if __name__ == '__main__':
    run()
