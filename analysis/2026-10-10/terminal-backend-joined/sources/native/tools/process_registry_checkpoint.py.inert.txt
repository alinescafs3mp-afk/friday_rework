"""Running-process checkpoint persistence and PID-safe recovery."""

import json
import logging
import time
from typing import Any, Dict, List, Optional

from agent.redact import redact_sensitive_text

logger = logging.getLogger("tools.process_registry")


class ProcessCheckpointMixin:
    def _detached_host_fate(self, pid: Optional[int], expected_start: Optional[int]) -> str:
        """Subclass supplies the live PID decision. See ProcessRegistry."""
        raise NotImplementedError

    # ----- Checkpoint (crash recovery) -----

    def _write_checkpoint(self, extra_entries: Optional[List[Dict[str, Any]]] = None, *, strict=False):
        """Write running process metadata to the checkpoint file atomically."""
        if (getattr(self, '_terminal_recovery_blocked', False)
                or getattr(self, '_custody_recovery_blocked', False)):
            raise RuntimeError('STOP_UNCONFIRMED')
        from tools.process_registry import _checkpoint_path, _CHECKPOINT_FIELDS

        custody_strict = strict
        try:
            with self._lock:
                custody_strict = strict or any(s.terminal_custody or s.web_custody for s in self._running.values())
                entries = []
                for s in self._running.values():
                    if s.exited and not ((s.terminal_custody or s.web_custody) and s._scope_stop_pending):
                        continue
                    # Backfill the start time so recovery can detect PID recycling
                    # even for sessions spawned before this field existed.
                    if s.host_start_time is None and s.pid_scope == "host" and s.pid:
                        s.host_start_time = self._safe_host_start_time(s.pid)
                    entry = {"session_id": s.id, **{f: getattr(s, f) for f in _CHECKPOINT_FIELDS}}
                    # Redact inline credentials before persisting (~/.hermes/processes.json).
                    # Recovery uses command only for display (adoption re-validates the
                    # PID, never re-runs it), so masking is lossless.
                    # See #77484.
                    entry["command"] = redact_sensitive_text(s.command, code_file=True)
                    entry["owner_task_id"] = s.owner_task_id or s.task_id
                    entries.append(entry)
                if extra_entries:
                    tracked_ids = {item.get("session_id") for item in entries}
                    entries.extend(item for item in extra_entries if item.get("session_id") not in tracked_ids)
                from utils import atomic_json_write
                if custody_strict:
                    from tools.terminal_containment import durable_json
                    durable_json(_checkpoint_path(), entries)
                else:
                    atomic_json_write(_checkpoint_path(), entries)
        except Exception as e:
            if custody_strict:
                raise
            logger.debug("Failed to write checkpoint file: %s", e, exc_info=True)

    def recover_from_checkpoint(self) -> int:
        """On gateway startup, probe PIDs from the checkpoint file; returns how many
        were recovered as detached sessions."""
        from tools.process_registry import (
            ProcessSession, _CHECKPOINT_FIELDS, _checkpoint_path,
            _CHECKPOINT_DEFAULTS, _WATCHER_ROUTE_KEYS, _stop_systemd_unit,
        )

        checkpoint_path = _checkpoint_path()
        if (getattr(self, '_terminal_recovery_blocked', False)
                or getattr(self, '_custody_recovery_blocked', False)):
            raise RuntimeError('STOP_UNCONFIRMED')
        try:
            checkpoint_path.lstat()
        except FileNotFoundError:
            return 0
        except OSError:
            self._terminal_recovery_blocked = True
            self._custody_recovery_blocked = True
            raise RuntimeError('STOP_UNCONFIRMED') from None
        try:
            from tools.terminal_containment import private_bytes, validate_recovered_entry
            raw = private_bytes(checkpoint_path)
            if len(raw) > 8_000_000:
                raise ValueError('checkpoint_bound')
            entries = json.loads(raw.decode('utf-8-sig'))
            if not isinstance(entries, list):
                raise ValueError('checkpoint_list_required')
            seen = set()
            # Validate the entire mixed inventory before adoption or writes.
            # Unknown custody stays blocked across both native launch paths.
            for entry in entries:
                if not isinstance(entry, dict):
                    raise ValueError('checkpoint_record_required')
                sid = entry.get('session_id')
                if not isinstance(sid, str) or not sid or sid in seen:
                    raise ValueError('checkpoint_unique_session_required')
                seen.add(sid)
                if type(entry.get('terminal_custody', False)) is not bool or type(entry.get('web_custody', False)) is not bool:
                    raise ValueError('invalid_custody_flag')
                if entry.get('terminal_custody') and entry.get('web_custody'):
                    raise ValueError('ambiguous_checkpoint_custody')
                unit = entry.get('systemd_unit', '')
                if not isinstance(unit, str):
                    raise ValueError('invalid_checkpoint_unit')
                # A saved strict service or its original custody metadata cannot
                # downgrade to legacy PID/EOF recovery by clearing its flag.
                # Ordinary scopes keep their empty strict-field defaults.
                if not entry.get('terminal_custody'):
                    metadata = entry.get('terminal_metadata', {})
                    if (unit.startswith('hermes-worker-terminal-')
                            or not isinstance(metadata, dict) or metadata):
                        raise ValueError('terminal_checkpoint_custody_downgrade')
                    for key in ('terminal_stop_requested', 'terminal_settled'):
                        if entry.get(key, False) is not False:
                            raise ValueError('terminal_checkpoint_custody_downgrade')
                if not entry.get('web_custody'):
                    if unit.startswith('hermes-worker-web-'):
                        raise ValueError('web_checkpoint_custody_downgrade')
                    for key in ('web_invocation', 'web_cgroup', 'web_boot_id',
                                'web_tool_call_id', 'web_turn_id'):
                        if entry.get(key, '') != '' or not isinstance(entry.get(key, ''), str):
                            raise ValueError('web_checkpoint_custody_downgrade')
                    identity = entry.get('web_cgroup_identity', [])
                    deadline = entry.get('web_deadline', 0.0)
                    if (not isinstance(identity, list) or identity
                            or type(deadline) not in (int, float) or deadline != 0):
                        raise ValueError('web_checkpoint_custody_downgrade')
                    for key in ('web_stop_requested', 'web_settled'):
                        if entry.get(key, False) is not False:
                            raise ValueError('web_checkpoint_custody_downgrade')
                # *_unconfirmed=True is a benign default even on legacy rows;
                # it alone is not evidence of strict service custody.
                if entry.get('terminal_custody'):
                    validate_recovered_entry(entry)
                if entry.get('web_custody'):
                    from tools.process_registry_web import validate_web_checkpoint
                    validate_web_checkpoint(entry)
        except Exception:
            self._terminal_recovery_blocked = True
            self._custody_recovery_blocked = True
            raise RuntimeError('STOP_UNCONFIRMED') from None
        recovered = 0
        unresolved_scope_entries: List[Dict[str, Any]] = []
        for entry in entries:
            pid, pid_scope = entry.get("pid"), entry.get("pid_scope", "host")
            if entry.get('terminal_custody'):
                from tools.terminal_containment import validate_recovered_entry
                try:
                    validate_recovered_entry(entry)
                except Exception:
                    self._terminal_recovery_blocked = True
                    raise RuntimeError('STOP_UNCONFIRMED') from None
                # Retain pre-Popen intent even without a wrapper PID. Recovered
                # pipes/PTY/grants are deliberately never fabricated.
                session = ProcessSession(id=entry['session_id'],
                    **{f: entry.get(f, _CHECKPOINT_DEFAULTS[f]) for f in _CHECKPOINT_FIELDS}, detached=True)
                session.terminal_unconfirmed = True
                session._scope_stop_pending = True
                with self._lock:
                    if session.id not in self._running:
                        self._running[session.id] = session
                        recovered += 1
                continue
            if entry.get("web_custody"):
                # Adopt intent even with no PID, or with a dead/reused wrapper.
                # Never infer cgroup cessation from a failed gateway's pipes.
                session = ProcessSession(id=entry["session_id"],
                    **{f:entry.get(f,_CHECKPOINT_DEFAULTS[f]) for f in _CHECKPOINT_FIELDS}, detached=True)
                session.web_unconfirmed = True
                session._scope_stop_pending = True
                with self._lock:
                    if session.id not in self._running:
                        self._running[session.id] = session
                        recovered += 1
                continue
            if not pid:
                continue
            # The registry is process-global, so every profile's checkpoint carries every live
            # process; a multiplexer recovering several homes must adopt each session once.
            with self._lock:
                already_tracked = entry.get("session_id") in self._running
            if already_tracked:
                continue
            if pid_scope != "host":  # in-sandbox PIDs mean nothing once the env handle is gone
                logger.info(
                    "Skipping recovery for non-host process: %s (pid=%s, scope=%s)",
                    entry.get("command", "unknown")[:60], pid, pid_scope)
                continue
            # Alive and still ours: re-attach. A start-time probe that cannot
            # be read is not proof the PID was reused — dropping it would leave
            # a live child unsupervised, and marking it exited would invent a
            # completion. A positive mismatch means the number was recycled:
            # do not adopt it and do not signal it.
            fate = self._detached_host_fate(pid, entry.get("host_start_time"))
            if fate == "reused":
                logger.info(
                    "Not recovering session %s: pid %d is alive but its "
                    "start time no longer matches — PID was recycled onto "
                    "an unrelated process; refusing to adopt or signal it.",
                    entry.get("session_id", "?"), pid)
                continue
            if fate != "running":
                systemd_unit = entry.get("systemd_unit", "")
                if systemd_unit and not _stop_systemd_unit(systemd_unit):
                    logger.warning(
                        "Could not reap persisted scope %s for dead wrapper pid %s; "
                        "retaining checkpoint entry for the next startup",
                        systemd_unit, pid)
                    unresolved_scope_entries.append(entry)
                continue
            fields = {f: entry.get(f, _CHECKPOINT_DEFAULTS[f]) for f in _CHECKPOINT_FIELDS}
            fields.update(
                command=entry.get("command", "unknown"),
                owner_task_id=entry.get("owner_task_id", "") or entry.get("task_id", ""),
                started_at=entry.get("started_at", time.time()))
            # detached: can't read output, but can report status + kill
            session = ProcessSession(id=entry["session_id"], detached=True, **fields)
            with self._lock:
                self._running[session.id] = session
            recovered += 1
            logger.info("Recovered detached process: %s (pid=%d)", session.command[:60], pid)
            # Re-enqueue watcher so gateway can resume notifications
            if session.watcher_interval > 0:
                from tools.process_registry_authority import copy_authority
                self.pending_watchers.append({
                    "session_id": session.id,
                    "user_authority": copy_authority(session),
                    "check_interval": session.watcher_interval,
                    "session_key": session.session_key,
                    **{key: getattr(session, f"watcher_{key}") for key in _WATCHER_ROUTE_KEYS},
                    "notify_on_complete": session.notify_on_complete,
                    "parent_session_id": session.parent_session_id,
                })
        self._write_checkpoint(extra_entries=unresolved_scope_entries)
        return recovered
