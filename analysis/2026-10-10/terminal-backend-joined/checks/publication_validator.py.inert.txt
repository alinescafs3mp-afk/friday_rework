"""Finite read-only integrity/privacy/link check of an inert source publication.

Standard library only. Never imports or executes the inspected source/checkers,
applies patches, initializes Hermes, or contacts processes/services/network.
Output is JSON on stdout; the caller owns any output file.
"""
import argparse
import hashlib
import json
from pathlib import Path, PurePosixPath
import re
import stat
from urllib.parse import unquote, urlsplit

PREFIX = 'analysis/2026-10-10/terminal-backend-joined/'
MAX_FILES = 128
MAX_FILE_BYTES = 12 * 1024 * 1024
MAX_TOTAL_BYTES = 24 * 1024 * 1024


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def require(condition, message):
    if not condition:
        raise ValueError(message)


def local(root, value, base=None):
    require(isinstance(value, str) and bool(value), 'invalid local reference')
    parsed = urlsplit(value)
    require(not parsed.scheme and not parsed.netloc, 'external reference forbidden')
    name = unquote(parsed.path)
    if name.startswith(PREFIX):
        name = name[len(PREFIX):]
        parent = root
    else:
        parent = base or root
    require(bool(name) and not name.startswith('/') and '\\' not in name,
            'absolute/empty/non-POSIX reference forbidden')
    parts = PurePosixPath(name).parts
    require('..' not in parts and '.' not in parts, 'traversal reference forbidden')
    target = parent.joinpath(*parts)
    require(target.is_relative_to(root), 'reference outside bundle')
    require(target.is_file() and not target.is_symlink(), 'missing/nonregular reference')
    return target


def validate(root):
    root = Path(root).absolute()
    require(root.is_dir() and not root.is_symlink(), 'invalid bundle root')
    files = {}
    total = 0
    for p in sorted(root.rglob('*')):
        require(not p.is_symlink(), 'symlink in publication')
        if p.is_dir():
            continue
        require(stat.S_ISREG(p.stat().st_mode), 'nonregular publication entry')
        rel = p.relative_to(root).as_posix()
        require(not p.stat().st_mode & 0o111, 'executable publication entry')
        require(p.stat().st_size <= MAX_FILE_BYTES, 'oversized publication file')
        total += p.stat().st_size
        require(total <= MAX_TOTAL_BYTES, 'oversized publication bundle')
        files[rel] = p.read_bytes()
        require(len(files) <= MAX_FILES, 'too many publication files')
    require('MANIFEST.json' in files, 'missing publication manifest')
    manifest = json.loads(files['MANIFEST.json'])
    require(isinstance(manifest, dict), 'invalid publication manifest')
    require(set(manifest) == set(files) - {'MANIFEST.json'}, 'manifest inventory mismatch')
    for rel, pin in manifest.items():
        local(root, rel)
        require(isinstance(pin, dict) and set(pin) == {'sha256', 'bytes'}, 'invalid file pin')
        require(pin == {'sha256': sha(files[rel]), 'bytes': len(files[rel])}, 'file hash/size mismatch')

    privacy_patterns = [
        ('private Linux home', r'/' r'home/(?!user(?:[/\s]|$)|example(?:[/\s]|$))[^\s\"\'<>]+'),
        ('private macOS home', r'/' r'Users/(?!me(?:[/\s]|$)|user(?:[/\s]|$)|example(?:[/\s]|$))[^\s\"\'<>]+'),
        ('developer state path', r'(?:\.co' r'dex/|\.config/co' r'dex-multi/|\.jericho/' r'runtime/|Friday_rework_' r'handoff/|SESSION_' r'HANDOFF\.md)'),
        ('native binding metadata', r'(?:CODEX_' r'THREAD_ID|backend_' r'endpoint|app-server-' r'control\.sock)'),
        ('private key block', r'-----BEGIN (?:RSA |EC |OPENSSH )?PRIVATE KEY-----'),
        ('provider credential', r'\b(?:sk-(?:proj-)?[A-Za-z0-9_-]{24,}|gh[pousr]_[A-Za-z0-9]{30,}|AKIA[A-Z0-9]{16})\b'),
        ('Telegram credential', r'\b[0-9]{8,12}:[A-Za-z0-9_-]{32,}\b'),
        ('JWT credential', r'\beyJ[A-Za-z0-9_-]{16,}\.[A-Za-z0-9_-]{16,}\.[A-Za-z0-9_-]{16,}\b'),
    ]
    json_docs = {}
    markdown_links = 0
    for rel, raw in files.items():
        text = raw.decode('utf-8', errors='strict')
        for label, pattern in privacy_patterns:
            require(not re.search(pattern, text), 'privacy scan: ' + label + ' in ' + rel)
        if rel.endswith('.json') or rel.endswith('.json.inert.txt'):
            json_docs[rel] = json.loads(text)
        if rel.endswith('.md'):
            for target in re.findall(r'(?<!!)\[[^\]\n]+\]\(([^)\n]+)\)', text):
                target = target.strip().split(' ', 1)[0].strip('<>')
                local(root, target, root / rel.rsplit('/', 1)[0] if '/' in rel else root)
                markdown_links += 1

    index = json_docs['source-index.json']
    require(index['status'] == 'INERT_SOURCE_COPIES', 'source status drift')
    require(index['native_source_count'] == 18, 'native source count drift')
    source_rows = index['files']
    require(len(source_rows) == 20, 'source/patch inventory drift')
    logicals = set()
    indexed_paths = set()
    for row in source_rows:
        require(row['logical_source'] not in logicals, 'duplicate logical source')
        logicals.add(row['logical_source'])
        p = local(root, row['path'])
        require(p.name.endswith('.inert.txt'), 'non-inert source path')
        rel = p.relative_to(root).as_posix()
        require(rel not in indexed_paths, 'duplicate published source path')
        indexed_paths.add(rel)
        require(sha(files[rel]) == row['sha256'] and len(files[rel]) == row['bytes'], 'source pin mismatch')
    actual_source_paths = {rel for rel in files if rel.startswith('sources/')}
    require(actual_source_paths == indexed_paths, 'unindexed source copy')
    joined = json_docs['sources/project/patches/hermes/terminal-backend-joined-manifest.json.inert.txt']
    patch_raw = files['sources/project/patches/hermes/terminal-backend-joined.patch.inert.txt']
    require(sha(patch_raw) == joined['patch_sha256'], 'patch binding mismatch')
    receipt = json_docs['source-receipt.json']
    pins = json_docs['source-r5-pins.json']
    native_rows = {row['logical_source']: row for row in source_rows if row['kind'] == 'native'}
    require(len(native_rows) == len(joined['files']) == len(pins) == 18, 'exact18 mismatch')
    require(set(native_rows) == set(pins) == {x['path'] for x in joined['files']}, 'exact18 inventory mismatch')
    for row in joined['files']:
        name = row['path']
        published = native_rows[name]
        require(row['sha256'] == published['sha256'] == pins[name]['sha256'] == receipt['files'][name]['sha256'], 'joined source hash mismatch')
        require(row['bytes'] == published['bytes'] == pins[name]['bytes'] == receipt['files'][name]['bytes'], 'joined source size mismatch')
        require(row['mode'] == receipt['files'][name]['mode'], 'source mode metadata mismatch')
    patch_paths = re.findall(r'^diff --git a/(\S+) b/(\S+)$', patch_raw.decode(), re.M)
    require(len(patch_paths) == 18 and all(a == b for a, b in patch_paths), 'patch file inventory mismatch')
    require({a for a, b in patch_paths} == set(native_rows), 'patch source inventory mismatch')
    require(len(receipt['layers']) == 28, 'layer inventory drift')
    require(receipt['layers'][-1]['patch_sha256'] == joined['patch_sha256'], 'last patch layer drift')
    require(receipt['layers'][-1]['manifest_sha256'] == sha(files['sources/project/patches/hermes/terminal-backend-joined-manifest.json.inert.txt']), 'last manifest layer drift')
    comp = json_docs['composition.json']
    require(comp['files_verified'] == len(receipt['files']) == 17647, 'full source receipt count drift')
    require(comp['source_receipt_sha256'] == sha(files['source-receipt.json']), 'composition receipt mismatch')
    require(comp['source_r5_pins_sha256'] == sha(files['source-r5-pins.json']), 'composition pins mismatch')
    require(comp['all_bytes_modes_match'] is True and comp['runtime_grant'] is False, 'composition claim drift')
    require(receipt['dependencies_installed'] is False and receipt['services_started'] is False, 'runtime receipt drift')

    provenance = json_docs['checker-provenance.json']
    for checker in provenance['files']:
        p = local(root, checker['path'])
        raw = files[p.relative_to(root).as_posix()]
        require(sha(raw) == checker['sha256'], 'checker hash mismatch')
        require(checker['published_copy_executed'] is False, 'inert checker claim drift')
        if checker['transformation'] == 'NONE_EXACT_BYTES':
            require(checker['sha256'] == checker['executed_private_sha256'], 'original executed checker provenance mismatch')
    terminal = json_docs['terminal-author-controls.json']
    web = json_docs['web-author-controls.json']
    require(terminal['passed'] == 74 and web['passed'] == 80 and terminal['failed'] == web['failed'] == 0, 'author controls drift')
    for controls in [terminal, web]:
        require(len(controls['controls']) == controls['passed'] and all(x['status'] == 'PASS' for x in controls['controls']), 'author control detail mismatch')
    for name, row in web['source_bodies'].items():
        require(row['sha256'] == receipt['files'][name]['sha256'], 'author source body pin mismatch')
    require(terminal['runtime_acceptance'] == web['runtime'] == 'NOT_RUN', 'author runtime claim drift')
    outcome = json_docs['outcome.json']
    require(outcome['independent_review'] == 'LIMITED_PASS_SOURCE_ONLY_RUNTIME_NOT_RUN', 'review ownership drift')
    review = json_docs['independent-source-review.json']
    require(review['original_private_result_sha256'] == '8afcc2b73913ecd1a25ebb2f7b370a1078f9934fca384c96f0857f4180520ddd'
            and review['status'] == outcome['independent_review']
            and review['review_complete_for_assignment'] is True
            and review['runtime_acceptance'] is False, 'independent verdict drift')
    for item in list(review['final_snapshot'].values()) + [review['independent_controls']['proof'],
                     review['independent_controls']['reused_unchanged_backend']['proof'],
                     outcome['independent_source_review']]:
        if isinstance(item, dict) and 'path' in item:
            path = local(root, item['path'])
            raw = files[path.relative_to(root).as_posix()]
            require(sha(raw) == item['sha256'] and len(raw) == item['bytes'], 'independent proof binding mismatch')
    independent = json_docs['independent-r5-controls.json']
    require(independent['passed'] == 107 and independent['failed'] == 0
            and len(independent['controls']) == 107
            and all(x['status'] == 'PASS' for x in independent['controls']), 'independent control detail mismatch')
    require(independent['native_effects'] == 0 and independent['handles'] == 0, 'independent runtime claim drift')
    for row in independent['selected']:
        name = row['logical_source'].removeprefix('hermes/')
        require(row['file_sha256'] == receipt['files'][name]['sha256'], 'independent source body mismatch')
    require(review['independent_controls']['new_final_r5_passed'] == 107
            and review['independent_controls']['reused_unchanged_backend']['count'] == 29
            and review['independent_controls']['author_controls_counted_as_independent'] is False,
            'independent accounting drift')
    require(all(f['status'] == 'FIX_CONFIRMED_ON_EXACT_FINAL_R5' for f in review['findings'])
            and {f['id'] for f in review['findings']} == {'F1','F2','F3'}, 'independent findings unresolved')
    require(outcome['runtime_admission'] is False and outcome['release_accepted'] is False, 'admission/release claim drift')
    require(outcome['SAFE_terminal_and_process_manage'] == 'DENIED_UNCHANGED', 'SAFE drift')
    require(outcome['full_native_imports'] == outcome['runtime_tests'] == 'NOT_RUN', 'unexecuted gate drift')
    json_links = 0
    def check_json_refs(value):
        nonlocal json_links
        if isinstance(value, dict):
            for child in value.values():
                check_json_refs(child)
        elif isinstance(value, list):
            for child in value:
                check_json_refs(child)
        elif isinstance(value, str) and value.startswith(PREFIX):
            local(root, value)
            json_links += 1
    for doc in json_docs.values():
        check_json_refs(doc)
    return {'status': 'PASS_INERT_PUBLICATION_INTEGRITY_PRIVACY_LINKS',
            'published_files': len(files), 'manifest_entries': len(manifest),
            'native_sources': 18, 'exact_source_and_patch_copies': 20,
            'receipt_files': 17647, 'layers': 28, 'author_controls': 154,
            'markdown_links': markdown_links, 'json_local_links': json_links,
            'json_files_parsed': len(json_docs), 'bytes': total,
            'manifest_sha256': sha(files['MANIFEST.json']),
            'privacy_scope': 'Finite private-home/developer-state/known credential pattern scan; source never executed',
            'runtime': 'NOT_RUN', 'independent_review': 'LIMITED_PASS_SOURCE_ONLY_RUNTIME_NOT_RUN',
            'runtime_grant': False, 'release_accepted': False}


if __name__ == '__main__':
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('bundle')
    args = parser.parse_args()
    try:
        result = validate(args.bundle)
    except (ValueError, KeyError, TypeError, OSError, UnicodeError) as error:
        print(json.dumps({'status': 'FAIL', 'error': str(error)}))
        raise SystemExit(1)
    print(json.dumps(result, indent=2))
