def check(name, f):
    try:
        f()
        REPORT.append({'name': name, 'status': 'PASS'})
    except Exception as exc:
        REPORT.append({'name': name, 'status': 'FAIL', 'error': type(exc).__name__ + ': ' + str(exc)})


def equal(value, expected):
    assert value == expected, (value, expected)


def denied(f, needle=''):
    try:
        f()
    except (PermissionError, RuntimeError, ValueError, OSError) as exc:
        assert needle in str(exc), (needle, str(exc))
    else:
        raise AssertionError('unexpected admission')


def fixture_module(name, **values):
    m = types.ModuleType(name)
    m.__dict__.update(values)
    sys.modules[name] = m
    return m


def body_module(rel, name, extra=None, selected=None):
    tree = ast.parse((B / rel).read_text())
    nodes = []
    for n in tree.body:
        if isinstance(n, (ast.FunctionDef, ast.ClassDef)):
            if selected is None or n.name in selected:
                nodes.append(n)
        elif selected is None and isinstance(n, (ast.Assign, ast.AnnAssign)):
            nodes.append(n)
        elif isinstance(n, (ast.Import, ast.ImportFrom)):
            names = [a.name.split('.')[0] for a in n.names] if isinstance(n, ast.Import) else [(n.module or '').split('.')[0]]
            if all(v in {'dataclasses', 'hashlib', 'json', 'math', 'os', 'pathlib', 'stat', 'time',
                         're', 'subprocess', 'select', 'uuid', 'copy', 'threading', 'decimal'} for v in names):
                nodes.append(n)
    future = ast.ImportFrom(module='__future__', names=[ast.alias(name='annotations')], level=0)
    m = fixture_module(name)
    m.__dict__.update(extra or {})
    m.__dict__['__file__'] = str(B / rel)
    code = ast.fix_missing_locations(ast.Module(body=[future, *nodes], type_ignores=[]))
    exec(compile(code, str(B / rel), 'exec'), m.__dict__)
    return m

class InertRegistry(mix.TerminalProcessMixin):
    def __init__(self):
        tree=ast.parse((B/'tools/process_registry.py').read_text())
        cls=next(n for n in tree.body if isinstance(n,ast.ClassDef) and n.name=='ProcessRegistry')
        init=next(n for n in cls.body if isinstance(n,ast.FunctionDef) and n.name=='__init__')
        expr=next(n.value for n in ast.walk(init) if isinstance(n,ast.Assign)
                  and any(isinstance(t,ast.Attribute) and t.attr=='_lock' for t in n.targets))
        self._running = {}; self._lock = eval(compile(ast.Expression(expr),'native registry lock expression','eval'),{'threading':threading}); self.events = []
    def _new_session(self, command, task_id, owner_task_id, session_key, cwd, **extra):
        return types.SimpleNamespace(id='proc_' + os.urandom(6).hex(), command=command, task_id=task_id,
            owner_task_id=owner_task_id, session_key=session_key, cwd=cwd, process=None, pid=None,
            host_start_time=None, terminal_stop_requested=False, terminal_unconfirmed=True, terminal_settled=False,
            _terminal_binding=None, _terminal_foreground=False, _reader_thread=None, _pty=None, detached=False,
            _terminal_settle_lock=threading.RLock(), _terminal_stdin_lock=threading.Lock(),
            parent_session_id='', exit_code=None, _scope_stop_pending=False, **extra)
    def _checked_scoped_checkpoint(self, session):
        self.events.append('checkpoint:'+session.terminal_metadata['phase'])
    def _safe_host_start_time(self, pid): return 101
    def _finish_exited(self, session, rc, **kw):
        self.events.append('finish'); session.exit_code=rc; self._running.pop(session.id, None)
    def _track_started(self, *a): self.events.append('reader')
    def _reader_loop(self, *a): raise AssertionError('INERT reader must not run')
    def _detached_host_fate(self, *a): return 'gone'
