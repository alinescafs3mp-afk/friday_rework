"""Finite pure-publication controls; mutate only owned inert fixture copies.

No subject imports, patch application, processes/services/models/network.
The original publication remains unchanged. Re-signing editorial file pins in
negative fixtures tests semantic validation beyond the first manifest check.
"""
from pathlib import Path
import hashlib
import json
import os
import resource
import runpy
import shutil
import time

W = Path(__file__).parent
B = W / 'public-candidate/analysis/2026-10-10/terminal-backend-joined'
F = W / 'publication-fixtures-final'
F.mkdir(mode=0o700)
os.umask(0o077)
resource.setrlimit(resource.RLIMIT_AS, (1024**3, 1024**3))
os.sched_setaffinity(0, sorted(os.sched_getaffinity(0))[:2])
validate = runpy.run_path(str(W / 'validate_publication_final.py'))['validate']
REPORT = []
started = time.monotonic()


def js(p):
    return json.loads(p.read_text())


def put(p, value):
    p.write_text(json.dumps(value, indent=2) + '\n')


def resign(root):
    pins = {}
    for p in sorted(root.rglob('*')):
        if p.is_file() and p.name != 'MANIFEST.json':
            raw = p.read_bytes()
            pins[p.relative_to(root).as_posix()] = {'sha256': hashlib.sha256(raw).hexdigest(), 'bytes': len(raw)}
    put(root / 'MANIFEST.json', pins)


def rewrite(root, name, mutate):
    p = root / name
    value = js(p)
    mutate(value)
    put(p, value)
    resign(root)


def control(name, mutate, expected_error=None, rehash=True):
    root = F / str(len(REPORT) + 1)
    shutil.copytree(B, root, symlinks=True)
    mutate(root)
    if rehash:
        resign(root)
    try:
        observation = validate(root)
    except (ValueError, KeyError, TypeError, OSError) as error:
        if expected_error is None or expected_error not in str(error):
            REPORT.append({'name': name, 'status': 'FAIL', 'error': str(error)})
        else:
            REPORT.append({'name': name, 'status': 'PASS', 'observed_refusal': str(error)})
    else:
        REPORT.append({'name': name, 'status': 'PASS' if expected_error is None else 'FAIL',
                       'observed': observation['status']})


control('exact publication is admitted as source integrity only', lambda root: None)
control('legitimate editorial change with exact refreshed pin',
        lambda root: (root / 'README.md').write_text((root / 'README.md').read_text() + '\nEditorial fixture only.\n'))
control('one source byte cannot borrow old manifest pin',
        lambda root: (root / 'sources/native/tools/terminal_binding.py.inert.txt').write_bytes(b'changed'),
        'file hash/size mismatch', rehash=False)
control('missing source refused before any effect',
        lambda root: (root / 'sources/native/tools/terminal_binding.py.inert.txt').unlink(),
        'manifest inventory mismatch', rehash=False)
control('extra unpinned file refused', lambda root: (root / 'unexpected.txt').write_text('INERT'),
        'manifest inventory mismatch', rehash=False)
control('re-signed extra source cannot hide outside source index',
        lambda root: (root / 'sources/native/unlisted.py.inert.txt').write_text('INERT'),
        'unindexed source copy')
control('re-signed missing markdown link refused',
        lambda root: (root / 'README.md').write_text((root / 'README.md').read_text() + '\n[missing](missing.json)\n'),
        'missing/nonregular reference')
control('re-signed encoded traversal in source index refused',
        lambda root: rewrite(root, 'source-index.json', lambda value: value['files'][0].update(path='%2e%2e/escape.txt')),
        'traversal reference forbidden')
control('re-signed external local JSON reference refused',
        lambda root: rewrite(root, 'source-index.json', lambda value: value['files'][0].update(path='https://example.com/fixture.txt')),
        'external reference forbidden')
control('re-signed private home marker refused',
        lambda root: (root / 'README.md').write_text((root / 'README.md').read_text() + '\n/' + 'home/privacy-control/private.txt\n'),
        'privacy scan: private Linux home')
control('re-signed developer state marker refused',
        lambda root: (root / 'README.md').write_text((root / 'README.md').read_text() + '\n.' + 'codex/private.txt\n'),
        'privacy scan: developer state path')
control('re-signed provider credential marker refused',
        lambda root: (root / 'README.md').write_text((root / 'README.md').read_text() + '\nsk-' + 'A' * 40 + '\n'),
        'privacy scan: provider credential')
control('re-signed Telegram credential marker refused',
        lambda root: (root / 'README.md').write_text((root / 'README.md').read_text() + '\n' + '123456789:' + 'A' * 35 + '\n'),
        'privacy scan: Telegram credential')
control('re-signed duplicate logical source refused',
        lambda root: rewrite(root, 'source-index.json', lambda value: value['files'][1].update(logical_source=value['files'][0]['logical_source'])),
        'duplicate logical source')
control('re-signed source index hash substitution cannot borrow unchanged bytes',
        lambda root: rewrite(root, 'source-index.json', lambda value: value['files'][0].update(sha256='0' * 64)),
        'source pin mismatch')
def replace_indexed_bytes(root, kind):
    index = js(root / 'source-index.json')
    row = next(row for row in index['files'] if row['kind'] == kind)
    rel = row['path'].removeprefix('analysis/2026-10-10/terminal-backend-joined/')
    p = root / rel
    p.write_bytes(p.read_bytes() + b'\nINERT_MUTATION_ONLY\n')
    raw = p.read_bytes()
    row.update(sha256=hashlib.sha256(raw).hexdigest(), bytes=len(raw))
    put(root / 'source-index.json', index)
control('re-signed source and index cannot evade original joined source pins',
        lambda root: replace_indexed_bytes(root, 'native'), 'joined source hash mismatch')
control('re-signed patch and index cannot evade original patch pin',
        lambda root: replace_indexed_bytes(root, 'project'), 'patch binding mismatch')
control('re-signed published checker cannot invent executed-original provenance',
        lambda root: rewrite(root, 'checker-provenance.json', lambda value: value['files'][0].update(executed_private_sha256='0' * 64)),
        'original executed checker provenance mismatch')
control('re-signed author source-body substitution refused',
        lambda root: rewrite(root, 'web-author-controls.json', lambda value: value['source_bodies']['tools/process_registry.py'].update(sha256='0' * 64)),
        'author source body pin mismatch')
control('re-signed author green counter cannot omit observation',
        lambda root: rewrite(root, 'terminal-author-controls.json', lambda value: value['controls'].pop()),
        'author control detail mismatch')
control('re-signed runtime admission claim refused',
        lambda root: rewrite(root, 'outcome.json', lambda value: value.update(runtime_admission=True)),
        'admission/release claim drift')
control('re-signed SAFE change claim refused',
        lambda root: rewrite(root, 'outcome.json', lambda value: value.update(SAFE_terminal_and_process_manage='ENABLED')),
        'SAFE drift')
control('re-signed invented independent review refused',
        lambda root: rewrite(root, 'outcome.json', lambda value: value.update(independent_review='ACCEPTED')),
        'review ownership drift')
control('re-signed independent result substitution refused',
        lambda root: rewrite(root, 'independent-source-review.json', lambda value: value.update(original_private_result_sha256='0' * 64)),
        'independent verdict drift')
control('re-signed independent observation omission refused',
        lambda root: rewrite(root, 'independent-r5-controls.json', lambda value: value['controls'].pop()),
        'independent proof binding mismatch')
control('re-signed changed independent source proof refused',
        lambda root: rewrite(root, 'independent-source-review.json', lambda value: value['final_snapshot']['source_receipt'].update(sha256='0' * 64)),
        'independent proof binding mismatch')
control('inert source cannot carry executable mode',
        lambda root: (root / 'sources/native/tools/terminal_binding.py.inert.txt').chmod(0o700),
        'executable publication entry')
def symlink(root):
    p = root / 'README.md'
    p.unlink()
    p.symlink_to(B / 'README.md')
control('publication symlink cannot borrow outside bytes', symlink, 'symlink in publication', rehash=False)

failed = sum(row['status'] == 'FAIL' for row in REPORT)
result = {'schema': 'friday.inert-publication-controls.v1', 'passed': len(REPORT) - failed,
    'failed': failed, 'controls': REPORT, 'scope': 'Pure read-only publication validator; isolated owned inert-copy mutations only',
    'elapsed_seconds': round(time.monotonic() - started, 3),
    'cpu_affinity': sorted(os.sched_getaffinity(0)), 'memory_limit_bytes': resource.getrlimit(resource.RLIMIT_AS)[0],
    'native_imports': 0, 'subject_processes': 0, 'subject_services': 0, 'subject_threads': 0,
    'model_calls': 0, 'network_calls': 0, 'children': 0, 'runtime_acceptance': 'NOT_RUN'}
put(W / 'PUBLICATION-CONTROLS.json', result)
print(json.dumps({'passed': result['passed'], 'failed': failed, 'elapsed_seconds': result['elapsed_seconds']}))
if failed:
    print(json.dumps([row for row in REPORT if row['status'] == 'FAIL']))
    raise SystemExit(1)
