"""Fixed single native-fault target; no original modes/G2 controls are replayed."""
import hashlib
import json
import os
from pathlib import Path
import resource
import signal
import sys
import time


def main():
    here = Path(__file__).resolve().parent
    sys.path.insert(0, str(here))
    from finite_kernel import arm
    from existing_caller_join import exclusive
    run = Path.cwd()  # Exact native C chdir to the once-only runtime directory.
    assert run == here.parent.parent/'native-fault-run'
    permit = json.loads((run/'permit.json').read_bytes())
    assert permit['runtime_directory'] == str(run) and permit['qualification_directory'] == str(here)
    clock = json.loads((run/'clock.json').read_bytes())
    assert len(sys.argv) == 2 and sys.argv[1].isdecimal()
    assert clock['budget_seconds'] == 180 and clock['cpu_seconds'] == 45
    assert clock['deadline_monotonic']-clock['started_monotonic'] == 180
    assert clock['boot_id'] == Path('/proc/sys/kernel/random/boot_id').read_text().strip()
    assert time.monotonic() < clock['deadline_monotonic']
    assert resource.getrlimit(resource.RLIMIT_CPU) == (6, 45)
    assert resource.getrlimit(resource.RLIMIT_AS) == (2 << 30, 2 << 30)
    assert sorted(os.sched_getaffinity(0)) == [0, 1]
    assert signal.getsignal(signal.SIGXCPU) == signal.SIG_DFL
    blocked = signal.pthread_sigmask(signal.SIG_BLOCK, set())
    assert signal.SIGXCPU not in blocked
    assert sys.flags.isolated and sys.flags.no_site and sys.dont_write_bytecode
    assert not sys.flags.optimize and sys.version_info[:3] == (3, 14, 7)
    wall = arm(time.CLOCK_MONOTONIC, clock['deadline_monotonic'])
    cpu = arm(time.CLOCK_PROCESS_CPUTIME_ID, 6)
    exclusive(run/'target-entry.json', {'pid': os.getpid(), 'ppid': os.getppid(),
        'cpu_limit': list(resource.getrlimit(resource.RLIMIT_CPU)),
        'as_limit': list(resource.getrlimit(resource.RLIMIT_AS)),
        'affinity': sorted(os.sched_getaffinity(0)),
        'clock_sha256': hashlib.sha256((run/'clock.json').read_bytes()).hexdigest(),
        'wall': wall, 'cpu': cpu, 'before_caller_gate': True})
    assert os.read(int(sys.argv[1]), 1) == b'B'
    os.close(int(sys.argv[1]))
    return 0


if __name__ == '__main__':
    raise SystemExit(main())
