def digest(path):
    with Path(path).open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()


def clock_valid(clock):
    assert clock['budget_seconds'] == 180 and clock['cpu_seconds'] == 45
    start, end = clock['started_monotonic'], clock['deadline_monotonic']
    assert all(type(x) in (int, float) and math.isfinite(x) for x in (start, end))
    assert end - start == 180
    assert start <= time.monotonic() < end
    assert clock['boot_id'] == Path('/proc/sys/kernel/random/boot_id').read_text().strip()
    assert clock['assignment'] and type(clock['generation']) is int and clock['generation'] > 0
    assert 'owner_pid' not in clock

def validate_caller_lock(p):
    ref = Path(p['launch_binding'])
    launch = json.loads(ref.read_bytes())
    assert launch['pid'] == os.getpid()
    assert launch['boot_id'] == p['original_clock']['boot_id']
    assert launch['permit_sha256'] == digest(p['permit_path'])
    assert launch['original_clock'] == p['original_clock']
    current = Path('/proc/self/stat').read_text()
    start = int(current.rsplit(')', 1)[1].split()[19])
    assert start == launch['start_ticks']
    fd = launch['quality_fd']; assert type(fd) is int and fd >= 3
    s = os.fstat(fd)
    actual = [s.st_dev, s.st_ino, s.st_uid, s.st_gid, s.st_nlink, s.st_size, stat.S_IMODE(s.st_mode)]
    assert actual == launch['quality_identity'] == QUALITY_ID
    info = Path('/proc/self/fdinfo/'+str(fd)).read_text().splitlines()
    locks = [line.split() for line in info if line.startswith('lock:')]
    major_minor_inode = f'{os.major(s.st_dev):02x}:{os.minor(s.st_dev):02x}:{s.st_ino}'
    assert any('FLOCK' in row and 'WRITE' in row and str(os.getpid()) in row
               and major_minor_inode in row for row in locks)
    assert os.stat(QUALITY).st_ino == s.st_ino and os.stat(QUALITY).st_dev == s.st_dev
    return {'pid': os.getpid(), 'start_ticks': start, 'fd': fd, 'identity': actual,
            'fdinfo_lock_lines': [row for row in locks], 'caller_custody_held': True,
            'release_requires_external_caller_exit': True}


def usage(which):
    value = resource.getrusage(which)
    return value.ru_utime + value.ru_stime


def native_snapshot(lib):
    state = {name: int(lib.frw_q_get(i)) for i, name in enumerate(STATE_FIELDS)}
    state['wall_ns'] = [int(lib.frw_q_get(30+i)) for i in range(5)]
    state['cpu_ns'] = [int(lib.frw_q_get(40+i)) for i in range(5)]
    return state


def recover_and_finish(wall, custody, proc, deadline, helpers, failures, skip_native_signal=False):
    Custody, proc_stat, direct_children, group_members = helpers
    # R2: recover exact C-published PID independently of a failed pidfd signal.
    try:
        if proc is None and wall is not None and wall.pid:
            proc = SimpleNamespace(pid=wall.pid, returncode=None)
        if proc is not None and custody is None:
            custody = Custody(proc, os.getpid(), proc_stat, direct_children, group_members)
    except BaseException as error:
        failures.append({'phase': 'retained_custody_recovery', 'error': repr(error)})
    if wall is not None and not skip_native_signal:
        try:
            wall.kill_root()
        except BaseException as error:
            failures.append({'phase': 'native_signal', 'error': repr(error), 'errno': getattr(error, 'errno', None)})
    if custody is not None:
        try:
            custody.finish(deadline, terminate=True)
        except BaseException as error:
            failures.append({'phase': 'original_custody_finish', 'error': repr(error)})
            # One bounded harvest, not a new clock or retry of the observation.
            try:
                _, errors = custody.discover()
                errors.extend(custody.signal_owned()); errors.extend(custody.reap_ready())
                failures.extend(errors)
            except BaseException as error:
                failures.append({'phase': 'final_harvest', 'error': repr(error)})
    return proc, custody


def cause_observed(code, state, error, before, after, sentinel_before, sentinel_after):
    actual_errno = getattr(error, 'errno', None)
    if code == 1:
        return state['actual_clone_syscalls'] == 1 and state['actual_clone_errno'] == actual_errno == errno.EINVAL
    if code in (2, 3, 4, 5):
        expected = errno.ENOSYS if code == 2 else errno.EPERM if code == 3 else errno.ENOMEM
        return (actual_errno == expected and state['injected_operations'] == 1
                and state['actual_clone_syscalls'] == 0 and not state['root_pid']
                and state['birth_fd'] == -1 and (code != 5 or sentinel_before == sentinel_after))
    if code in (6, 7):
        return (bool(state['cpu_hit']) and state['cpu_ns'][4] >= 6000000000
                and (state['actual_clone_syscalls'] == 0 and state['root_pid'] == 0 if code == 6
                     else state['actual_clone_syscalls'] == 1 and state['root_pid'] > 0))
    if code in (10, 11):
        return isinstance(error, KeyboardInterrupt) and state['pending_signal_errno'] == 0
    if code == 12:
        return actual_errno == state['actual_timer_create_errno'] == errno.EINVAL
    if code == 13:
        return actual_errno == state['actual_timer_settime_errno'] == errno.EINVAL
    if code in (*range(14, 20), 28, 29):
        return actual_errno == errno.EIO and state['injected_operations'] >= 1
    if code == 20:
        return actual_errno == errno.EINVAL and state['root_pid'] > 0 and state['birth_fd'] >= 0
    if code in (21, 22, 23, 24, 25, 27):
        expected = {21: errno.ENOENT, 22: errno.EBADF, 23: errno.EEXIST,
                    24: errno.EINVAL, 25: errno.EINVAL, 27: errno.EBUSY}[code]
        return actual_errno == expected
    if code == 26:
        return isinstance(error, RuntimeError) and str(error) == 'NONDEFAULT_PYTHON_SIGNAL_TABLE'
    if code == 30:
        return isinstance(error, RuntimeError) and str(error).startswith('OWNER_EXEC_ERROR_PROTOCOL:')
    return False

    run.mkdir(mode=0o700)  # Consumed directory; never reopen/remove for retry.
    from existing_caller_join import exclusive, direct_children, group_members, wait_exec_pipe
    from finite_custody import Custody

    assert not direct_children() and len(list(Path('/proc/self/task').iterdir())) == 1
    assert signal.getsignal(signal.SIGCHLD) == signal.SIG_DFL
    assert signal.getsignal(signal.SIGINT) == signal.default_int_handler
    assert subreaper() == 1
    exclusive(run/'clock.json', clock)
    exclusive(run/'permit.json', p)
    exclusive(run/'claim.json', {'clock_sha256': digest(run/'clock.json'), 'plan': p['plan'],
                                 'consumed': True, 'retry_authorized': False,
                                 'one_native_birth_maximum': 1})
    before = {'cpu': list(resource.getrlimit(resource.RLIMIT_CPU)),
              'as': list(resource.getrlimit(resource.RLIMIT_AS)),
              'affinity': sorted(os.sched_getaffinity(0)),
              'children_cpu': usage(resource.RUSAGE_CHILDREN),
              'caller_cpu_lifetime': usage(resource.RUSAGE_SELF)}

    # NativeBirth constructor cannot publish a Python wall object. No new DSO.
    lib = ctypes.PyDLL(p['qualification_binary'], mode=os.RTLD_NOW | os.RTLD_LOCAL)
    lib.frw_q_configure.argtypes = (ctypes.c_int, ctypes.c_int)
    lib.frw_q_configure.restype = ctypes.c_int
    lib.frw_q_get.argtypes = (ctypes.c_int,); lib.frw_q_get.restype = ctypes.c_longlong
    lib.frw_close.argtypes = (ctypes.c_int,); lib.frw_close.restype = ctypes.c_int
    fds = set(); sentinel_before = sentinel_after = None
    if code == 5:

    gate_r = gate_w = err_r = err_w = None
    try:
        if code == 24:
            # Actual pipe2 flags error; NOT a real resource-exhausted pipe.
            os.pipe2(-1)
            raise RuntimeError('INVALID_PIPE_FLAGS_UNEXPECTEDLY_SUCCEEDED')
        gate_r, gate_w = os.pipe2(os.O_CLOEXEC); fds.update((gate_r, gate_w))
        err_r, err_w = os.pipe2(os.O_CLOEXEC); fds.update((err_r, err_w))
        if code == 26:
            action = signal.signal(signal.SIGXCPU, signal.SIG_IGN)
        if code == 27:
            mask = signal.pthread_sigmask(signal.SIG_BLOCK, {signal.SIGALRM})

            native_born = native_before_close['root_pid'] != 0
            # No-child cases can be positively closed WITHOUT inventing a wait4.

                    primary = error; record_error = repr(error)
                else:
                    failures.append({'phase': 'receipt_conflict_not_refused'})
            observed = cause_observed(code, native_before_close, primary, before, after,
                                      sentinel_before, sentinel_after)
            result = {'schema': 'friday.native-fault-observation.v1',
                      'state': 'CAUSE_OBSERVED_PENDING_CLOSURE_UNACCEPTED' if observed else 'FAILED_OR_UNKNOWN_NO_RETRY',
                      'plan': p['plan'], 'observation_class': category, 'cause_observed': observed,

                      'cpu_endpoint_overshoot_from6_seconds': native_before_close['cpu_ns'][4]/1e9-6 if code in (6, 7) and native_before_close['cpu_hit'] else None,
                      'delivery_guard1_qualified': False,
                      'resources_restored': all(before[k] == after[k] for k in ('cpu', 'as', 'affinity')),
                      'before_parent_hooks_paired': native_before_close['before_fork_returns'] == native_before_close['after_fork_returns'],
                      'rows': rows, 'custody': custody.evidence() if custody is not None else None,
                      'kernel_accounting_complete': accounted, 'no_child_observed': no_child,
                      'known_reaped_cpu_lower_bound': lower_bound,
                      'accounted_tree_cpu': lower_bound if accounted else 0 if no_child and delta == 0 else None,
                      'caller_children_delta': delta, 'caller_cpu_lifetime_at_record': usage(resource.RUSAGE_SELF),
                      'caller_write_exit_tail_cpu': 'UNKNOWN_NOT_ZERO_REQUIRES_EXTERNAL_WITNESS',
                      'caller_quality_custody': validate_caller_lock(p),
                      'sentinel_preserved': sentinel_before == sentinel_after if code == 5 else None,
                      'errors': failures, 'original_wall_met_at_record': time.monotonic() < deadline,
                      'native_close': 'PENDING_SEPARATE_RECORD', 'runtime_acceptance': False,
                      'F6_complete': False, 'product_mode_admission': False,
                      'not_selected_plans': {name: 'NOT_RUN_NO_RENEWED_ALLOWANCE' for name in PLANS if name != p['plan']}}
            try:
                if code != 23:
                    exclusive(run/'observation.json', result)
                else:
                    exclusive(run/'receipt-failure-evidence.json', result)
            except BaseException as error:
                record_error = repr(error)
        finally:
            # Neither assembly nor receipt failure may skip native close. A
            # retained fast-exit birth with unknown identity stays unconfirmed.
            try:
                if wall is not None:
                    wall.close_after_observation(no_child or terminal)
                else:
                    error = lib.frw_close(False)
                    if error: raise OSError(error, 'UNPUBLISHED_ARM_OBJECT_CLOSE')
            except BaseException as error:
                close_error = repr(error)
            native_after_close = native_snapshot(lib)
            for fd in list(fds):
                try:
                    os.close(fd); fds.remove(fd)
                except OSError as error:
                    failures.append({'phase': 'pipe_or_sentinel_close', 'error': repr(error)})
            close_at = time.monotonic()
            close_record = {'schema': 'friday.native-fault-close.v1', 'error': close_error,
                            'native_after_close': native_after_close,
                            'native_state_released': not any(native_after_close[k] for k in ('armed', 'timer_live', 'alarm_installed', 'cpu_installed', 'borrowing'))
                                                     and native_after_close['birth_fd'] == -1,
                            'actual_restoration_getter': native_after_close['actual_restoration_getter'],
                            'original_wall_met': close_at < deadline,
                            'record_write_error': record_error, 'no_child': no_child,
                            'terminal_custody': terminal, 'clock_sha256': digest(run/'clock.json'),
                            'caller_quality_custody': validate_caller_lock(p),
                            'external_caller_exit_cpu_release_required': True}
            exclusive(run/'native-close.json', close_record)
    # A measured expected cause is NOT complete kernel/production acceptance.
    return 0 if (observed and close_error is None and record_error is None and close_at < deadline
                 and close_record['native_state_released'] and native_after_close['actual_restoration_getter'] == 1
                 and (accounted or no_child and delta == 0) and lower_bound <= 45
                 and all(before[k] == after[k] for k in ('cpu', 'as', 'affinity'))
                 and native_before_close['before_fork_returns'] == native_before_close['after_fork_returns']) else 1
