# Inert analytical source export; no execution or deployment grant.
"""Source-only same-PID launcher. Original clock comes from ROOT, never reset.

Root must materialize an exact permit/source/qualification ELF before use.
Keeps the existing canonical FLOCK through exec/record/exit; no supervisor.
"""
import fcntl
import hashlib
import json
import os
from pathlib import Path
import stat
import sys
import time


def main():
    here = Path(__file__).resolve().parent
    assert len(sys.argv) == 2
    permit = Path(sys.argv[1]).resolve()
    p = json.loads(permit.read_bytes())
    assert p['schema'] == 'friday.native-fault-observation-permit.v1'
    assert p['source_reviewed_observation_authorized'] is True
    assert p['product_mode_admission'] is False and p['retry_authorized'] is False
    assert p['scope'] == 'ONE_SELECTED_FAULT_PLAN_ORIGINAL_CLOCK_ONE_BIRTH_NO_RETRY'
    assert p['plan'] not in ('original180-before-publication-source', 'original180-after-publication-source')
    assert p['qualification_directory'] == str(here) and p['permit_path'] == str(permit)
    clock = p['original_clock']
    assert clock['budget_seconds'] == 180 and clock['cpu_seconds'] == 45
    assert clock['deadline_monotonic']-clock['started_monotonic'] == 180
    assert clock['started_monotonic'] <= time.monotonic() < clock['deadline_monotonic']
    assert clock['boot_id'] == Path('/proc/sys/kernel/random/boot_id').read_text().strip()
    runtime = here.parent.parent/'native-fault-run'
    assert p['runtime_directory'] == str(runtime) and not runtime.exists()
    assert hashlib.sha256((here/'source-manifest.json').read_bytes()).hexdigest() == p['source_manifest_sha256']
    assert hashlib.sha256(Path(p['interpreter']).read_bytes()).hexdigest() == p['interpreter_sha256']
    binding = Path(p['launch_binding'])
    assert binding == here.parent.parent/'native-fault-launch-binding.json' and not binding.exists()
    fd = os.open('/PRIVATE_RUNTIME/friday-quality-ui-exclusive.lock', os.O_RDONLY | os.O_NOFOLLOW)
    s = os.fstat(fd)
    identity = [s.st_dev, s.st_ino, s.st_uid, s.st_gid, s.st_nlink, s.st_size, stat.S_IMODE(s.st_mode)]
    assert identity == [PUBLIC_LOCK_DEV, PUBLIC_LOCK_INODE, PUBLIC_LOCK_UID, PUBLIC_LOCK_GID, 1, 0, 0o664]
    fcntl.flock(fd, fcntl.LOCK_EX | fcntl.LOCK_NB)
    os.set_inheritable(fd, True)
    argv = [p['interpreter'], '-I', '-B', '-S', str(here/'invoke_qualification.py'), '--native-faults', str(permit)]
    record = {'pid': os.getpid(), 'start_ticks': int(Path('/proc/self/stat').read_text().rsplit(')', 1)[1].split()[19]),
              'boot_id': clock['boot_id'], 'permit_sha256': hashlib.sha256(permit.read_bytes()).hexdigest(),
              'quality_identity': identity, 'quality_fd': fd, 'original_clock': clock,
              'argv': argv, 'same_pid_exec': True, 'new_supervisor': False, 'new_budget': False}
    out = os.open(binding, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600)
    with os.fdopen(out, 'w') as stream:
        json.dump(record, stream, indent=2); stream.write('\n'); stream.flush(); os.fsync(stream.fileno())
    directory = os.open(binding.parent, os.O_RDONLY | os.O_DIRECTORY)
    os.fsync(directory); os.close(directory)
    os.execve(p['interpreter'], argv, {'PATH': '/usr/bin:/bin', 'LANG': 'C', 'LC_ALL': 'C'})


if __name__ == '__main__':
    main()
