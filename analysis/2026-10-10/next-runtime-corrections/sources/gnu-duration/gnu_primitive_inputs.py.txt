import hashlib
import json
import math
import sys
import time
from pathlib import Path

SCOPE = 'ONE_GNU_PRIMITIVE_BUNDLE_ORIGINAL_CLOCK_NO_RETRY'
MODE = 'gnu-primitive'
# Exact pinned GNU 9.7 conversion of nominal CLI 60s; no timer tolerance.
GNU_POSIX_DURATION_NS = 60_000_000_001
GNU_ALARM_DURATION_NS = 60_000_000_000  # Existing 60s-only fallback policy.
# Exact rebuilt ELF binding. Independent final review is required before effects.
GNU_SHIM_BUILT_SHA256 = 'f08c839804f89012cbf403ab87a2892bf29bc4a0b9bef10e1c7fa6b256dd2f2c'
LIMITS = dict(wall_seconds=180, tree_cpu_seconds=45, root_cpu_seconds=6,
              as_bytes=2 << 30, affinity=[0, 1], gnu_seconds=60, closure_seconds=60)
CONTROLS = ('real-gnu-arm-mode-exec-result', 'wrong-mode-refusal',
            'wrong-target-argv-refusal', 'exhausted-cpu-allocation-refusal',
            'nonfinite-cpu-allocation-refusal', 'late-original-wall-refusal',
            'genuine-sigtrap-forwarding', 'clock-renewal-refusal',
            'terminal-wait4-reentry')


def digest(path):
    with Path(path).open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()


def is_primitive(admission):
    return admission.get('qualification_scope') == SCOPE


def original_room(deadline, now):
    if not (math.isfinite(deadline) and math.isfinite(now)):
        raise RuntimeError('NO_ORIGINAL_GNU60_FULL_CLOSURE60_ROOM')
    # Reserve the real POSIX interval plus full closure60 inside original180.
    if math.ceil(now * 1e9) + GNU_POSIX_DURATION_NS + 60_000_000_000 >= math.floor(deadline * 1e9):
        raise RuntimeError('NO_ORIGINAL_GNU60_FULL_CLOSURE60_ROOM')


def check_clock(clock, admission):
    assert is_primitive(admission) and admission['product_mode_admission'] is False
    assert clock['assignment'] == admission['assignment']
    assert type(clock['generation']) is int and clock['generation'] == admission['generation'] == 1
    assert clock['budget_seconds'] == 180 and clock['cpu_seconds'] == 45
    assert clock['deadline_monotonic'] - clock['started_monotonic'] == 180
    assert clock['started_monotonic'] <= time.monotonic() < clock['deadline_monotonic']
    assert clock['boot_id'] == Path('/proc/sys/kernel/random/boot_id').read_text().strip()
    assert 'owner_pid' not in clock and 'owner_start_ticks' not in clock
    original_room(clock['deadline_monotonic'], time.monotonic())


def mode_args(here, mode, admission, ordinary_modes):
    here = Path(here)
    if is_primitive(admission):
        assert admission['product_mode_admission'] is False
        assert mode == MODE and admission['primitive_controls'] == list(CONTROLS)
        return [str(here/'gnu_primitive_target.py'), str(here/'run'/'gnu-primitive-target.json')]
    assert mode in ordinary_modes
    return [str(here/'run.py')] if mode == 'source-cases' else [
        str(here/'callback_probe.py'), mode, str(here/'run'/('callback-'+mode+'.json'))]


def verify_bound_files(here, admission):
    # Actual existing files[] manifest schema, not a parallel invented pin map.
    here = Path(here).resolve()
    assert digest(here/'source-manifest.json') == admission['source_manifest_sha256']
    rows = json.loads((here/'source-manifest.json').read_bytes())['files']
    pins = {row['path']: row['sha256'] for row in rows}
    assert len(pins) == len(rows)
    for row in rows:
        path = Path(row['path'])
        assert path.is_absolute() and path.resolve() == path
        assert path.is_relative_to(here.parent)
        assert digest(path) == row['sha256'] and path.stat().st_size == row['bytes']
    for name in ('invoke_qualification.py', 'existing_caller_join.py', 'run_bounded_r2.py',
                 'launch.py', 'finite_trace.py', 'owner_step_inputs.py', 'gnu_primitive_inputs.py',
                 'gnu_primitive_target.py', 'finite_owner_entry.py', 'caller_wall.py',
                 'native_birth.py', 'finite_birth.c', 'finite_birth.so', 'gnu_arm_shim.so',
                 'gnu_arm_shim.c', 'gnu_arm_shim.map', 'finite_custody.py', 'finite_kernel.py',
                 'finite_credit.py', 'caller_quality.py', 'root_gnu_launch.py'):
        assert str(here/name) in pins
    assert admission['native_birth_binary_sha256'] == pins[str(here/'finite_birth.so')]
    assert admission['gnu_shim']['path'] == str(here/'gnu_arm_shim.so')
    assert admission['gnu_shim']['sha256'] == pins[str(here/'gnu_arm_shim.so')]
    assert admission['gnu_shim']['source_sha256'] == pins[str(here/'gnu_arm_shim.c')]
    assert admission['gnu_shim']['version_script_sha256'] == pins[str(here/'gnu_arm_shim.map')]
    assert admission['gnu_shim']['libc'] == '/lib/x86_64-linux-gnu/libc.so.6'
    assert admission['gnu_shim']['build_status'] == 'INDEPENDENTLY_VERIFIED_EXACT_BUILT_ELF'
    assert admission['root_launcher']['sha256'] == pins[str(here/'root_gnu_launch.py')]
    return pins

    assert admission['schema'] == 'friday.gnu-primitive-first-observation-permit.v1'
    assert is_primitive(admission) and admission['source_reviewed_observation_authorized'] is True
    assert admission['execution_authorized'] is True and admission['production_browser_grant'] is False
    assert admission['product_mode_admission'] is False
    assert admission['qualification_directory'] == str(here)
    assert admission['budget_authority'] == 'EXPLICIT_ROOT_ORIGINAL_CLOCK_NOT_ALL11_RETRY'
    assert admission['limits'] == LIMITS and admission['primitive_controls'] == list(CONTROLS)
    assert admission['assignment'] and type(admission['generation']) is int and admission['generation'] == 1
    assert sys.flags.isolated and sys.flags.no_site and sys.dont_write_bytecode
    assert not sys.flags.optimize
    assert digest('/proc/self/exe') == digest(admission['interpreter']) == admission['interpreter_sha256']
    assert admission['native_birth_binary_sha256'] == '20cbb92a027914ae9356452be59fb11f04bbb7697f4b551dee697e2fec85a3fe'
    assert GNU_SHIM_BUILT_SHA256 is not None, 'GNU_SHIM_REBUILD_AND_INDEPENDENT_REVIEW_REQUIRED'
    assert admission['gnu_shim']['sha256'] == GNU_SHIM_BUILT_SHA256
    assert digest('/usr/bin/gnutimeout') == admission['gnu_sha256'] == '1ba715580334044dbd32873bddd7f8a19687cda1cf5e9f44be7266af6c048766'
    dependencies = admission['resolved_runtime_dependencies']
    assert admission['runtime_dependency_binding_status'] == 'EXACT_FINAL_PROVIDERS_BOUND_BY_ROOT'
    assert dependencies['/lib/x86_64-linux-gnu/libc.so.6'] == '85e64f97e348786a8fb4d9f3d52fec289e2fb86bba20f0731dfe61990525e0f7'

    for path, sha in dependencies.items():

    assert review['root_launcher'] == admission['root_launcher']
    assert review['caller_quality_custody'] == admission['caller_quality_custody']
    assert review['limits'] == LIMITS

    # No observed=True predicate: observation is the future output, not a prerequisite.
    check_clock(admission['original_clock'], admission)
    return admission


def readonly_negative_controls(deadline, previous):
    from finite_credit import allocate_cpu
    rows = []
    for name, value in (('exhausted-cpu-allocation-refusal', 35.0),
                        ('nonfinite-cpu-allocation-refusal', float('nan'))):
        try:
            allocate_cpu(value)
        except (ValueError, RuntimeError) as error:
            rows.append(dict(control=name, refused=True, error=str(error), no_process_attempt=True))
        else:
            raise RuntimeError('INVALID_CPU_ALLOCATION_ACCEPTED')
    try:
        original_room(deadline, deadline-120)
    except RuntimeError as error:
        rows.append(dict(control='late-original-wall-refusal', refused=True,
                         error=str(error), original_clock_unchanged=True))
    else:
        raise RuntimeError('SHORTENED_CLOSURE_ROOM_ACCEPTED')
    cap, fork_hard, credit = allocate_cpu(previous)
    assert previous+6+credit <= 45
    return rows


def consume_target(here, custody, step, clock, admission):
    path = Path(here)/'run'/'gnu-primitive-target.json'
    value = json.loads(path.read_bytes())
    assert value['state'] == 'GNU_PRIMITIVE_TARGET_OBSERVED_UNACCEPTED'
    assert value['pid'] == custody.mode_pid and value['owner_step_sha256'] == digest(Path(here)/'run'/'owner-step-gnu-primitive.json')
    assert value['clock_sha256'] == digest(Path(here)/'clock.json')
    assert value['admission_sha256'] == digest(Path(here)/'admission.json')
    assert value['source_manifest_sha256'] == admission['source_manifest_sha256']
    assert value['identity']['start_ticks'] == custody.traces[custody.mode_pid]['birth']['start_ticks']
    assert value['GNU_identity'] == step['gnu_identity']
    assert value['genuine_SIGTRAP_handler_calls'] == 1
    assert any(row['pid'] == custody.mode_pid and row['event'] == 0 and row['signal'] == 5
               for row in custody.trace_events)
    assert value['strict_environment_verified'] is True and value['shim_absent_after_exec'] is True
    assert [row['control'] for row in value['negative_controls']] == ['wrong-mode-refusal', 'wrong-target-argv-refusal']
    assert all(row['refused'] for row in value['negative_controls'])
    assert custody.arm_record and custody.mode_released and custody.gnu_detached
    assert custody.gnu_reap['exit_code'] == 0 and custody.closed
    return value


def main(here, permit_path):
    here = Path(here).resolve()
    admission = verify_permit(here, permit_path)
    from caller_quality import validate_quality_custody, consume_quality_receipt
    validate_quality_custody(here, admission)  # Refuse unattached launch before run/clock/native.
    # The root owns this clock. This entry never starts/renews one, even on failure.
    run = here/'run'
    run.mkdir(mode=0o700)
    (run/'home').mkdir(mode=0o700)
    (run/'tmp').mkdir(mode=0o700)
    from existing_caller_join import invoke_existing_owner
    result = invoke_existing_owner(here, admission['original_clock'], admission, primitive=True)
    consume_quality_receipt(result, validate_quality_custody(here, admission))
    return 0 if result['package_observation_complete'] else 1
