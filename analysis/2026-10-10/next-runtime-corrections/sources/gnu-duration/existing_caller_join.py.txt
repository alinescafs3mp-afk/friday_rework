# Inert analytical source export; no execution or deployment grant.
"""Same existing terminal caller -> finite owner through native clone3/exec.
SOURCE ONLY; independent root review and F6 qualification are still required.
Root fork-to-exit CPU + its actually waited/adopted tree is measured once.
The same external caller reports its own observed CPU separately, never zero.
"""
import json
import os
from pathlib import Path
import resource
import select
import signal
import struct
import sys
import time
from types import SimpleNamespace
from finite_kernel import subreaper
from finite_custody import Custody, StopUnconfirmed
from owner_step_inputs import digest, proc_stat, MODES
from caller_wall import CallerWall


def ensure_f4_bootstrap():
    # The live stop is retained. Root may replace this only after independent
    # source review, final binary binding and F6 on the actual merged artifact.
    raise RuntimeError('F4_NATIVE_SOURCE_ONLY_REQUIRES_INDEPENDENT_REVIEW_AND_F6')


def exclusive(path, value):
    with path.open('x') as stream:
        json.dump(value, stream, indent=2)
        stream.write('\n')
        stream.flush()
        os.fsync(stream.fileno())


def direct_children():
    return [int(v) for v in Path('/proc/self/task/'+str(os.getpid())+'/children').read_text().split()]


def group_members(pgid):
    rows, errors = [], []
    for item in Path('/proc').iterdir():
        if not item.name.isdecimal():
            continue
        try:
            row = proc_stat(int(item.name))
            if row['pgid'] == pgid:
                rows.append(row)
        except (FileNotFoundError, ProcessLookupError):
            pass
        except (OSError, ValueError, IndexError) as error:
            errors.append({'pid': item.name, 'error': repr(error)})
    return rows, errors


def self_cpu():
    usage = resource.getrusage(resource.RUSAGE_SELF)
    return usage.ru_utime+usage.ru_stime


def wait_exec_pipe(fd, deadline):
    # The child error pipe is CLOEXEC. Successful exec closes it; Popen's
    # blocking preexec error-pipe wait is replaced by this exact absolute wait.
    while time.monotonic() < deadline:
        ready, _, _ = select.select([fd], [], [], max(0, deadline-time.monotonic()))
        if ready:
            error = os.read(fd, 4096)
            if error:
                if len(error) == struct.calcsize('=ii'):
                    stage, code = struct.unpack('=ii', error)
                    raise OSError(code, 'OWNER_NATIVE_EXEC_STAGE:' + str(stage))
                raise RuntimeError('OWNER_EXEC_ERROR_PROTOCOL:' + repr(error))
            return
    raise StopUnconfirmed('OWNER_EXEC_PREFIX_ORIGINAL_DEADLINE_EXHAUSTED')


def invoke_existing_owner(here, original_clock, admission, primitive=False):
    if primitive:
        from gnu_primitive_inputs import check_clock, is_primitive, verify_bound_files
        assert is_primitive(admission) and admission['source_reviewed_observation_authorized'] is True
        assert admission['product_mode_admission'] is False
        check_clock(original_clock, admission)
        verify_bound_files(here, admission)
    else:
        ensure_f4_bootstrap()
    here = Path(here).resolve()
    clock = dict(original_clock)
    assert clock['budget_seconds'] == 180 and clock['cpu_seconds'] == 45
    assert clock['deadline_monotonic']-clock['started_monotonic'] == 180
    assert clock['started_monotonic'] <= time.monotonic() < clock['deadline_monotonic']
    assert clock['boot_id'] == Path('/proc/sys/kernel/random/boot_id').read_text().strip()
    assert 'owner_pid' not in clock and 'owner_start_ticks' not in clock
    if not primitive:
        assert clock['assignment'] == 'FRW041-CALLBACK-COST-CHANGED-SOURCE-QUALIFICATION'
    assert clock['generation'] == admission['generation'] == 1
    assert admission['assignment'] == clock['assignment']
    assert admission['execution_authorized'] is True
    assert admission['production_browser_grant'] is False
    assert admission['finite_sequence_sha256'] == digest(here/'finite-sequence.json')
    assert admission['source_manifest_sha256'] == digest(here/'source-manifest.json')
    assert admission['entry_limits_source_sha256'] == digest(here/'finite_owner_entry.py')
    assert admission['caller_join_source_sha256'] == digest(here/'existing_caller_join.py')
    assert admission['caller_entry_source_sha256'] == digest(here/'invoke_qualification.py')
    assert admission['caller_wall_source_sha256'] == digest(here/'caller_wall.py')
    assert admission['native_birth_source_sha256'] == digest(here/'finite_birth.c')
    assert admission['native_birth_binding_sha256'] == digest(here/'native_birth.py')
    native_library = here/'finite_birth.so'
    assert admission['native_birth_binary_sha256'] == digest(native_library)
    # Config is a source precondition, never runtime acceptance by itself.
    assert admission['kernel_release'] == Path('/proc/sys/kernel/osrelease').read_text().strip()
    config = Path('/boot/config-' + admission['kernel_release'])
    assert admission['kernel_config_sha256'] == digest(config)
    assert 'CONFIG_POSIX_CPU_TIMERS_TASK_WORK=y' in config.read_text().splitlines()
    assert admission['interpreter_sha256'] == digest(admission['interpreter']) == digest('/proc/self/exe')
    assert sys.flags.isolated and sys.flags.no_site and sys.dont_write_bytecode
    assert not direct_children() and len(list(Path('/proc/self/task').iterdir())) == 1
    # This is the EXISTING required independent owner-birth receipt. This
    # prepared overlay does not create it, set accepted=True, or authorize itself.
    if primitive:
        # Explicit first-observation source permission, never an observed receipt.
        from gnu_primitive_inputs import verify_permit
        assert verify_permit(here, admission['permit_path']) == admission
        from caller_quality import validate_quality_custody
        quality_before_birth = validate_quality_custody(here, admission)
    else:
        ref = admission['independent_boundary_receipts']['owner-birth-bound']
        assert digest(ref['path']) == ref['sha256']
        birth_review = json.loads(Path(ref['path']).read_bytes())
        assert birth_review['source_manifest_sha256'] == admission['source_manifest_sha256']
        assert birth_review['observed'] is True and birth_review['accepted_by_independent_root'] is True
        assert birth_review['terminal_observer_boundary'] == 'OWNER_FORK_THROUGH_FINAL_WAIT4_SEPARATE_OBSERVER_CPU'
        assert birth_review['inherited_effective_limits'] == 'CPU_SOFT6_DEFAULT_UNBLOCKED_AS_SOFT2GiB_AFFINITY0,1'
        assert birth_review['native_birth_binary_sha256'] == admission['native_birth_binary_sha256']
        assert birth_review['native_birth_source_sha256'] == admission['native_birth_source_sha256']
        assert birth_review['kernel_config_sha256'] == admission['kernel_config_sha256']
        assert birth_review['birth_api'] == 'clone3_CLEAR_SIGHAND_PIDFD_NATIVE_EXEC_ONLY'
        assert birth_review['original_deadline_signal_before_python_return_qualified'] is True
    signal.signal(signal.SIGCHLD, signal.SIG_DFL)
    assert subreaper() == 1
    previous = resource.getrusage(resource.RUSAGE_CHILDREN)
    observer_start = self_cpu()  # Whole observer lifetime-so-far, not a reset.
    exclusive(here/'clock.json', clock)
    exclusive(here/'admission.json', admission)
    gate_read = gate_write = error_read = error_write = None
    proc = custody = wall = None
    errors = []
    caller_identity = proc_stat(os.getpid())
    try:
        gate_read, gate_write = os.pipe2(os.O_CLOEXEC)
        error_read, error_write = os.pipe2(os.O_CLOEXEC)
        argv = [admission['interpreter'], '-I', '-B', '-S', str(here/'run_bounded_r2.py'),
                '--caller-gate', str(gate_read)]
        wall = CallerWall(native_library, clock['deadline_monotonic'])
        pid = wall.spawn(argv, here, gate_read, error_write)
        # Native state already owns root_pid + birth pidfd; no pidfd_open.
        proc = SimpleNamespace(pid=pid, returncode=None)
        custody = Custody(proc, os.getpid(), proc_stat, direct_children, group_members)
        os.close(gate_read); gate_read = None
        os.close(error_write); error_write = None
        wait_exec_pipe(error_read, clock['deadline_monotonic'])
        birth = proc_stat(proc.pid)
        assert birth['ppid'] == os.getpid() and birth['pgid'] == birth['sid'] == proc.pid
        assert digest('/proc/'+str(proc.pid)+'/exe') == admission['interpreter_sha256']
        actual_argv = [v.decode() for v in Path('/proc/'+str(proc.pid)+'/cmdline').read_bytes().split(b'\0') if v]
        assert actual_argv == argv
        exclusive(here/'owner-binding.json', {
            'owner_pid': proc.pid, 'owner_start_ticks': birth['start_ticks'],
            'owner_identity': birth, 'caller_pid': os.getpid(),
            'caller_identity': caller_identity, 'argv': argv, 'actual_argv': actual_argv,
            'interpreter_sha256': admission['interpreter_sha256'],
            'clock_sha256': digest(here/'clock.json'),
            'entry_sha256': digest(here/'finite_owner_entry.py'),
            'caller_join_sha256': digest(here/'existing_caller_join.py'),
            'root_pidfd_retained': wall.root_fd >= 0,
            'root_birth_api': 'clone3_CLEAR_SIGHAND_PIDFD_NATIVE_EXEC_ONLY',
            'native_birth_binary_sha256': admission['native_birth_binary_sha256'],
            'native_birth_source_sha256': admission['native_birth_source_sha256'],
            'native_birth_binding_sha256': admission['native_birth_binding_sha256'],
            'caller_entry_source_sha256': admission['caller_entry_source_sha256'],
            'caller_wall_source_sha256': admission['caller_wall_source_sha256'],
            'kernel_config_sha256': admission['kernel_config_sha256'],
            'birth_effective_cpu': 6, 'birth_effective_as': 2 << 30,
            'birth_affinity': [0, 1],
            'birth_hard_cpu': 'INHERITED_OBSERVER_CEILING_NOT_ASSERTED_45',
            'measured_cpu_boundary': 'OWNER_FORK_THROUGH_EXIT_AND_OWNED_TREE',
            **({'caller_quality_binding_sha256': quality_before_birth['launch_binding_sha256']}
               if primitive else {})})
        if wall.expired or time.monotonic() >= clock['deadline_monotonic']:
            raise StopUnconfirmed('NO_ORIGINAL_WALL_AT_GATE_RELEASE')
        if primitive:
            assert validate_quality_custody(here, admission) == quality_before_birth
        wall.require_admission_open()  # CPU endpoint/delivery/rollback latches also close this gate.
        assert os.write(gate_write, b'B') == 1
        os.close(gate_write); gate_write = None
        custody.finish(clock['deadline_monotonic'])
    except BaseException as error:
        errors.append({'caller_error': repr(error)})
        if wall is not None:
            # A Python pending exception / failed native restoration may hide
            # the call's return, but cannot erase its persistent native handle.
            # Recover exact born state independently of signal delivery. A
            # failed kill cannot skip available wait4 custody or invent zero.
            try:
                native_pid = wall.pid
                if proc is None and native_pid:
                    proc = SimpleNamespace(pid=native_pid, returncode=None)
                if proc is not None and custody is None:
                    custody = Custody(proc, os.getpid(), proc_stat, direct_children, group_members)
            except BaseException as native_cleanup:
                errors.append({'native_custody_error': repr(native_cleanup)})
            try:
                wall.kill_root()
            except BaseException as native_delivery:
                errors.append({'native_delivery_error': repr(native_delivery)})
        if gate_write is not None:
            try:
                os.close(gate_write)
            except OSError as pipe_close:
                errors.append({'gate_close_error': repr(pipe_close)})
            gate_write = None
        if custody is not None:
            try:
                custody.finish(clock['deadline_monotonic'], terminate=True)
            except BaseException as cleanup:
                errors.append({'caller_cleanup_error': repr(cleanup)})
                # One finite final harvest, not a renewed wall window. Missing
                # terminal usage remains UNKNOWN. Never hide rows already reaped.
                try:
                    _, failures = custody.discover()
                    failures += custody.signal_owned()
                    failures += custody.reap_ready()
                    if failures:
                        errors.append({'final_harvest_errors': failures})
                except BaseException as harvest:
                    errors.append({'final_harvest_error': repr(harvest)})
    finally:
        for fd in (gate_read, gate_write, error_read, error_write):
            if fd is not None:
                try:
                    os.close(fd)
                except OSError as pipe_close:
                    errors.append({'pipe_close_error': repr(pipe_close)})
    try:
        return finish_observation(here, clock, previous, observer_start, proc,
                                  custody, wall, errors, primitive)
    finally:
        if wall is not None:
            # Every post-reap error (including receipt writes) reaches closure.
            # Incomplete custody retains the exact handle and fails explicitly.
            wall.close_after_observation(not wall.pid or bool(custody and custody.closed))


def finish_observation(here, clock, previous, observer_start, proc, custody, wall, errors, primitive=False):
    final = resource.getrusage(resource.RUSAGE_CHILDREN)
    delta = final.ru_utime+final.ru_stime-previous.ru_utime-previous.ru_stime
    rows = custody.rows if custody is not None else []
    row_sum = sum(row['user']+row['system'] for row in rows)
    root_reap = custody.gnu_reap if custody is not None else None
    accounting_complete = bool(custody and custody.closed and not custody.anomalies
        and root_reap and not errors and abs(delta-row_sum) <= 2e-6)
    owner_record = None
    quality_now = None
    if primitive:
        from caller_quality import validate_quality_custody, consume_quality_receipt, RELEASE
        try:
            admission = json.loads((here/'admission.json').read_bytes())
            quality_now = validate_quality_custody(here, admission)
        except BaseException as error:
            errors.append({'caller_quality_error': repr(error)})
    if accounting_complete and proc.returncode == 0:
        try:
            owner_record = json.loads((here/'run'/'finite-owner-result.json').read_bytes())
            assert owner_record['clock_sha256'] == digest(here/'clock.json')
            assert owner_record['owner']['pid'] == proc.pid
            assert owner_record['owner']['start_ticks'] == root_reap['identity']['start_ticks']
            if primitive:
                from gnu_primitive_inputs import MODE, CONTROLS, GNU_POSIX_DURATION_NS, GNU_ALARM_DURATION_NS
                assert quality_now is not None
                consume_quality_receipt(owner_record, quality_now)
                assert owner_record['state'] == 'GNU_PRIMITIVE_COMPLETE_PENDING_ACTUAL_OWNER_REAP'
                assert owner_record['primitive_controls'] == list(CONTROLS)
                observations = owner_record['primitive_observations']
                assert [v['control'] for v in observations] == list(CONTROLS)
                assert len({v['control'] for v in observations}) == len(observations)
                by_control = {v['control']: v for v in observations}
                for name in ('wrong-mode-refusal', 'wrong-target-argv-refusal',
                             'exhausted-cpu-allocation-refusal', 'nonfinite-cpu-allocation-refusal',
                             'late-original-wall-refusal', 'clock-renewal-refusal'):
                    assert by_control[name]['refused'] is True
                assert by_control['terminal-wait4-reentry']['rows_unchanged'] is True
                assert by_control['terminal-wait4-reentry']['children_usage_unchanged'] is True
                assert by_control['genuine-sigtrap-forwarding']['target_handler_calls'] == 1
                assert by_control['genuine-sigtrap-forwarding']['trace_delivery_rows']
                assert by_control['real-gnu-arm-mode-exec-result']['GNU_exit'] == 0
                arm = by_control['real-gnu-arm-mode-exec-result']['arm_record']
                assert arm['api'] in (1, 2)
                assert arm['duration_ns'] == (GNU_POSIX_DURATION_NS if arm['api'] == 1 else GNU_ALARM_DURATION_NS)
                assert [r['mode'] for r in owner_record['mode_receipts']] == [MODE]
            else:
                assert owner_record['state'] == 'SEQUENCE_COMPLETE_PENDING_ACTUAL_OWNER_REAP'
                assert [r['mode'] for r in owner_record['mode_receipts']] == list(MODES)
            for receipt in owner_record['mode_receipts']:
                assert receipt['accounting_complete'] and receipt['process_unit_closed'] and receipt['content_accepted']
                if primitive:
                    consume_quality_receipt(receipt, quality_now)
                else:
                    assert receipt['lock_returned']
        except BaseException as error:
            errors.append({'owner_record_error': repr(error)})
            owner_record = None
    # root wait4 already contains its actually waited descendants. Extra adopted
    # direct rows are disjoint. Never add owner_record child totals a second time.
    measured_cpu = row_sum if accounting_complete else None
    original_wall_met = time.monotonic() < clock['deadline_monotonic']
    if wall is not None and wall.signal_error is not None:
        errors.append({'original_wall_signal_error': wall.signal_error})
    complete = bool(accounting_complete and measured_cpu <= 45 and owner_record
                    and original_wall_met and not errors)
    result = {'state': 'FINITE_PACKAGE_OBSERVED' if complete else 'FAILED_OR_UNKNOWN_NO_RETRY',
        'clock_sha256': digest(here/'clock.json'), 'root_wait4': root_reap,
        'root_tree_and_adopted_cpu': measured_cpu,
        'known_reaped_cpu_lower_bound': row_sum, 'caller_children_delta': delta,
        'rows': rows, 'errors': errors,
        'custody': custody.evidence() if custody is not None else None,
        'kernel_accounting_complete': accounting_complete,
        'package_observation_complete': complete, 'original_wall_met': original_wall_met,
        'root_exit': proc.returncode if proc is not None else None,
        'observer_cpu_lifetime_at_pre_fork_preparation_snapshot': observer_start,
        'observer_cpu_lifetime_at_final_record_snapshot': self_cpu(),
        'observer_cpu_scope': 'SEPARATE_TERMINAL_OBSERVER_NOT_ZERO_NOT_INCLUDED_IN45',
        'observer_final_write_exit_cpu': 'NOT_OBSERVED_BY_SELF_NOT_CLAIMED_COMPLETE',
        'native_observer_teardown': 'PENDING_AFTER_THIS_WRITE_REQUIRE_CALLER_EXIT_AND_CLOSURE',
        'runtime_acceptance': False, 'production_browser_grant': False,
        'qualification_scope': 'GNU_PRIMITIVE_ONLY' if primitive else 'ORIGINAL_ALL11',
        'owner_record': owner_record if primitive else None}
    if primitive:
        result.update(caller_quality_custody_held=quality_now is not None,
                      caller_quality_custody=quality_now, quality_release_evidence=RELEASE)
    # The measured root's own final write/reap/teardown CPU is in root_wait4.
    # This observer receipt is outside that process; do not claim its later CPU.
    exclusive(here/'run'/'caller-final-wait4.json', result)
    return result
