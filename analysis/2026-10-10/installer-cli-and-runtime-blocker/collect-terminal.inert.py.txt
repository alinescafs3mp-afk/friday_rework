def summarize_records(p, data, external, observed_monotonic, *, record_hashes):
    """Pure raw-record projection. No process, clock, native or lock effects."""
    c = p['original_clock']
    launch = data.get('launch', {})
    binding = data.get('provider-binding', {})
    close = data.get('native-close', {})
    observation = data.get('observation', {})
    caller = binding.get('caller', {})
    same_caller = (launch.get('pid') == caller.get('pid')
                   and launch.get('start_ticks') == caller.get('start_ticks')
                   and launch.get('boot_id') == caller.get('boot_id') == c['boot_id']
                   and type(caller.get('pid')) is int and caller['pid'] > 0
                   and type(caller.get('start_ticks')) is int and caller['start_ticks'] > 0)
    bound = (binding.get('admission_passed') is True
             and binding.get('state') == 'ACTUAL_CURRENT_GOT_PROVIDERS_BOUND_BEFORE_CANDIDATE_APIS'
             and type(binding.get('candidate_api_call_attempts_before_record')) is int
             and binding['candidate_api_call_attempts_before_record'] == 0
             and binding.get('load_returned_handle') is True
             and binding.get('binary_sha256') == p['qualification_binary_sha256']
             and binding.get('original_clock') == c and same_caller)
    requirement = binding.get('caller_quality_requirement', {})
    quality_observation = binding.get('caller_quality_observation', {})
    closed_quality = close.get('caller_quality_custody', {})
    quality_consistent = (type(launch.get('quality_fd')) is int and launch['quality_fd'] >= 3
                          and launch.get('quality_identity') == [2050,9044963,1000,1000,1,0,436]
                          and all(row.get('pid') == launch.get('pid')
                                  and row.get('start_ticks') == launch.get('start_ticks')
                                  and row.get('boot_id') == c['boot_id']
                                  and row.get('fd') == launch['quality_fd']
                                  and row.get('identity') == launch['quality_identity']
                                  for row in (requirement,quality_observation,closed_quality))
                          and closed_quality.get('caller_custody_held') is True)
    recorded_hashes_consistent = (record_hashes.get('root-permit') is not None
                                  and record_hashes.get('root-permit') == record_hashes.get('permit') == launch.get('permit_sha256')
                                  and record_hashes.get('clock') is not None
                                  and record_hashes.get('clock') == close.get('clock_sha256') == observation.get('clock_sha256')
                                  and record_hashes.get('provider-binding') is not None
                                  and record_hashes.get('provider-binding') == observation.get('actual_provider_binding_sha256')
                                  and observation.get('original_clock') == c
                                  and observation.get('source_manifest_sha256') == p['source_manifest_sha256']
                                  and observation.get('qualification_binary_sha256') == p['qualification_binary_sha256'])
    clock_equal = (data.get('clock') == c and data.get('permit') == p
                   and launch.get('original_clock') == c and close.get('original_wall_met') is True)
    terminal = (external.get('state') == 'EXACT_OWNED_CALLER_REAPED_ORIGINAL_CONTROL_CLOSED'
                and external.get('caller_pid') == launch.get('pid')
                and external.get('caller_start_ticks') == launch.get('start_ticks')
                and external.get('boot_id') == c['boot_id'] and external.get('original_clock') == c
                and external.get('quality_custody_through_exit') is True
                and external.get('control_retained_through_exit') is True
                and external.get('completed_before_original_deadline') is True)
    witness = observation.get('cpu_signal_witness')
    witness26 = (isinstance(witness,dict) and set(witness)==set(CPU_SIGNAL_FIELDS)
                 and all(type(v) is int for v in witness.values())
                 and witness == observation.get('native_before_close',{}).get('cpu_signal_witness'))
    confirmed = (bound and clock_equal and terminal and quality_consistent and recorded_hashes_consistent and witness26
                 and close.get('state') == 'NATIVE_CLOSE_OBSERVED_EXTERNAL_EXIT_PENDING'
                 and close.get('native_state_released') is True
                 and close.get('actual_restoration_getter') == 1
                 and close.get('native_snapshot_error') is None and close.get('error') is None
                 and close.get('record_write_error') is None
                 and close.get('local_close_errors') == [] and close.get('local_fds_remaining') == []
                 and close.get('evidence_errors') == []
                 and type(observed_monotonic) in (int,float)
                 and math.isfinite(observed_monotonic)
                 and c['started_monotonic'] <= observed_monotonic < c['deadline_monotonic'])
    return {'state':'RAW_SOURCE_CONTRACT_TERMINAL_JOIN_NOT_ACCEPTANCE' if confirmed else 'STOP_UNCONFIRMED',
            'actual_same_caller_provider_record_consistent':bound,
            'original_clock_consistent':clock_equal,'external_owned_terminal_consistent':terminal,
            'FLOCK_record_join_consistent':quality_consistent,'raw_hash_joins_consistent':recorded_hashes_consistent,
            'cpu_witness26_present':witness26,'raw_cpu_signal_witness':witness,
            'runtime_acceptance':False,'product_acceptance':False,'retry_authorized':False,
            'exclusive_enqueue_provenance':'NOT_PROVEN_STANDARD_SIGNAL_COALESCENCE',
            'caller_write_exit_tail_cpu':'UNKNOWN_NOT_ZERO_REQUIRES_EXTERNAL_WITNESS',
            'load_to_exit_mechanism_proven_by_this_collector':False}

def checked(path, expected):
    raw = Path(path).read_bytes()
    if hashlib.sha256(raw).hexdigest() != expected: raise ValueError('EXACT_PIN_REQUIRED')
    return json.loads(raw)

def main():
    # This pure projection does not justify lifting the unconditional stop.
    if not __debug__ or any(value is None for value in ROOT_BINDINGS.values()):
        raise SystemExit('INERT_UNBOUND_ROOT_REVIEW_REQUIRED_NO_EFFECT')
    assert len(sys.argv) == 3
    tool_exit, last_call_wait = int(sys.argv[1]),float(sys.argv[2])
    assert math.isfinite(last_call_wait) and last_call_wait >= 0
    p = checked(ROOT_BINDINGS['root_runtime_permit_path'],ROOT_BINDINGS['root_runtime_permit_sha256'])
    assert p['assignment'] == ROOT_BINDINGS['runtime_assignment'] and p['generation'] == ROOT_BINDINGS['runtime_generation']
    checked(ROOT_BINDINGS['root_launch_collection_acceptance_path'], ROOT_BINDINGS['root_launch_collection_acceptance_sha256'])
    launcher_raw = Path(ROOT_BINDINGS['materialized_launcher_path']).read_bytes()
    assert hashlib.sha256(launcher_raw).hexdigest() == ROOT_BINDINGS['materialized_launcher_sha256']
    external = checked(ROOT_BINDINGS['external_owned_terminal_path'],ROOT_BINDINGS['external_owned_terminal_sha256'])
    run = Path(p['runtime_directory'])
    paths = {'launch':Path(p['launch_binding']),'clock':run/'clock.json','claim':run/'claim.json',
             'permit':run/'permit.json','provider-binding':run/'provider-binding.json',
             'observation':run/'observation.json','native-close':run/'native-close.json',
             'receipt-failure-evidence':run/'receipt-failure-evidence.json'}
    data, hashes, missing = {}, {}, []
    for key,path in paths.items():
        try: raw = path.read_bytes()
        except FileNotFoundError: missing.append(key)
        else: data[key] = json.loads(raw); hashes[key] = hashlib.sha256(raw).hexdigest()
    record_hashes = dict(hashes, **{'root-permit':ROOT_BINDINGS['root_runtime_permit_sha256']})
    result = summarize_records(p,data,external,time.monotonic(),record_hashes=record_hashes)
    result.update(raw_records=data, hashes=hashes, missing=missing, external_terminal=external,
                  tool_exit_code=tool_exit,tool_last_call_wait_seconds=last_call_wait,
                  tool_wait_is_not_total_runtime=True,new_signal_or_process_control=False,
                  quality_lock_probe='NOT_PERFORMED_COLLECTOR_IS_NOT_CONTROL_OWNER')
    # Source-only currently: never run this writer to test it.
    out = ROOT/'external-terminal-observation.json'
    fd = os.open(out,os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o600)
    with os.fdopen(fd,'w') as stream:
        json.dump(result,stream,indent=2,allow_nan=False);stream.write('\n');stream.flush();os.fsync(stream.fileno())

if __name__ == '__main__':
    main()
