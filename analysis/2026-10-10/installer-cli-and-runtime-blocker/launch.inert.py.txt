ROOT_BINDINGS = {
    'root_runtime_permit_path': None,
    'root_runtime_permit_sha256': None,
    'root_launch_collection_acceptance_path': None,
    'root_launch_collection_acceptance_sha256': None,
    'root_load_to_exit_control_path': None,
    'root_load_to_exit_control_sha256': None,
    'runtime_assignment': None,
    'runtime_generation': None,
}
# Not a bindable flag, source receipt, capability template or execution grant.
raise SystemExit('NF_LOAD_EXIT1_UNPROVEN_EXTERNAL_SAME_CALLER_PIDFD_ORIGINAL_DEADLINE_LOAD_THROUGH_EXIT')

from pathlib import Path
import hashlib
import json
import math
import os
import time

def ref(p):
    p = Path(p)
    b = p.read_bytes()
    return {'path': str(p), 'sha256': hashlib.sha256(b).hexdigest()}

def checked(p, sha):
    assert ref(p)['sha256'] == sha, str(p)
    return json.loads(Path(p).read_bytes())

ROOT = Path('/private/ASTRA_RUNTIME/native-provider-final-composition')
Q = ROOT/'candidate/qualification'
FINAL_MANIFEST_SHA256 = 'efa28e7524091e2bbbb89cfbf28f874869251f01cd0b3ca3e7af09ebd387553c'

def main():
    # UNREACHABLE source mechanics, never permission to remove NF_LOAD_EXIT1.
    if not __debug__ or any(value is None for value in ROOT_BINDINGS.values()):
        raise SystemExit('INERT_UNBOUND_ROOT_REVIEW_REQUIRED_NO_EFFECT')
    permit_path = Path(ROOT_BINDINGS['root_runtime_permit_path'])
    p = checked(permit_path, ROOT_BINDINGS['root_runtime_permit_sha256'])
    decision = checked(ROOT_BINDINGS['root_launch_collection_acceptance_path'],
                       ROOT_BINDINGS['root_launch_collection_acceptance_sha256'])
    control = checked(ROOT_BINDINGS['root_load_to_exit_control_path'],
                      ROOT_BINDINGS['root_load_to_exit_control_sha256'])
    assert decision['accepted_by_independent_root'] is True
    assert p['source_reviewed_observation_authorized'] is True
    assert p['product_mode_admission'] is False and p['retry_authorized'] is False
    assert p['plan'] == 'real-cpu6-before-publication' and p['one_native_birth_maximum'] == 1
    assert p['assignment'] == ROOT_BINDINGS['runtime_assignment']
    assert p['generation'] == ROOT_BINDINGS['runtime_generation']
    assert p['limits'] == {'wall_seconds':180,'tree_cpu_seconds':45,'root_cpu_seconds':6,
                            'as_bytes':2147483648,'affinity':[0,1]}
    c = p['original_clock']
    assert c['assignment'] == p['assignment'] and c['generation'] == p['generation']
    assert c['budget_seconds'] == 180 and c['cpu_seconds'] == 45
    assert all(type(c[k]) in (int,float) and math.isfinite(c[k]) for k in ('started_monotonic','deadline_monotonic'))
    assert c['deadline_monotonic']-c['started_monotonic'] == 180
    assert c['started_monotonic'] <= time.monotonic() < c['deadline_monotonic']
    assert c['boot_id'] == Path('/proc/sys/kernel/random/boot_id').read_text().strip()
    # A JSON assertion is not the absent control mechanism. The unconditional
    # stop above is retained even if somebody populates all these fields.
    assert control['original_clock'] == c and control['assignment'] == p['assignment']
    assert control['generation'] == p['generation'] and control['caller_pid'] == os.getpid()
    assert control['caller_start_ticks'] == int(Path('/proc/self/stat').read_text().rsplit(')',1)[1].split()[19])
    assert control['covers_load_through_exit'] is True
    assert p['qualification_directory'] == str(Q) and p['permit_path'] == str(permit_path)
    assert permit_path == ROOT/'ROOT-ISSUED-NATIVE-FAULT-PERMIT.json'
    assert p['source_manifest_sha256'] == FINAL_MANIFEST_SHA256
    m = checked(Q/'source-manifest.json', FINAL_MANIFEST_SHA256)
    assert len(m['files']) == 48
    for row in m['files']:
        assert ref(row['path'])['sha256'] == row['sha256']
        assert Path(row['path']).stat().st_size == row['bytes']
    assert p['runtime_provider_binding_status'] == 'STATIC_REVIEWED_STAGED_LOAD_ACTUAL_BINDING_REQUIRED'
    assert 'resolved_runtime_dependencies' not in p and p['expected_runtime_dependencies']
    policy = checked(p['provider_admission_policy']['path'], p['provider_admission_policy']['sha256'])
    assert policy['original_clock'] == c and policy['assignment'] == p['assignment'] and policy['generation'] == p['generation']
    assert policy['purpose'] == p['plan'] and policy['staged_load_authorized'] is True
    assert policy['runtime_binding_observed'] is False and policy['retry_authorized'] is False
    assert policy['product_mode_admission'] is False
    assert policy['static']['source_manifest']['sha256'] == FINAL_MANIFEST_SHA256
    assert policy['static']['dso']['path'] == p['qualification_binary']
    assert policy['static']['dso']['sha256'] == p['qualification_binary_sha256']
    for path, sha in p['expected_runtime_dependencies'].items():
        assert ref(path)['sha256'] == sha
    for name in ('native-fault-run','native-fault-launch-binding.json','external-terminal-observation.json'):
        output = ROOT/name
        assert not output.exists() and not output.is_symlink()
    # Exact existing same-PID launcher acquires and keeps its original FLOCK
    # through exec/native records/exit. No Popen/fork/writer or new authority here.

              {'PATH':'/usr/bin:/bin','LANG':'C','LC_ALL':'C'})
