class Refused(RuntimeError):
    pass


def need(value, why):
    if not value:
        raise Refused(why)


def identity(s):
    return (s.st_dev, s.st_ino)


def metadata(s):
    return tuple(getattr(s, k) for k in ('st_dev', 'st_ino', 'st_mode', 'st_uid',
                 'st_gid', 'st_nlink', 'st_size', 'st_mtime_ns', 'st_ctime_ns'))


def gate(context, reserve=10):
    # One original clock, never restarted for reads/publication/cleanup.
    need(time.monotonic() < context['deadline_monotonic'] - reserve,
         'original_root_operation_budget_exhausted')


class CleanupUnconfirmed(Refused):
    def __init__(self, causal, errors):
        self.causal=causal
        self.cleanup_errors=errors
        super().__init__('STOP_UNCONFIRMED_CLEANUP; original='+repr(causal)+'; errors='+repr(errors))

def close_owned(owned, causal=None):
    # Pop BEFORE a fallible close. Every remaining owned FD is independently
    # attempted once, even if an earlier close succeeded but raised afterward.
    errors=[]
    for name in tuple(owned):
        fd=owned.pop(name)
        try:os.close(fd)
        except BaseException as error:
            errors.append({'handle':name,'fd':fd,'state':'CLOSE_ATTEMPTED_UNCONFIRMED',
                           'error':type(error).__name__+': '+str(error)[:256]})
    if errors:raise CleanupUnconfirmed(causal,errors) from causal

def safe_parent(path, allowed_uids, context):
    path=Path(path)
    need(path.is_absolute() and '..' not in path.parts,'absolute_clean_path')
    owned={}
    try:
        owned['parent']=os.open('/',os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW|os.O_CLOEXEC)
        root=os.fstat(owned['parent']);chain=[identity(root)]
        need(root.st_uid==root.st_gid==0 and stat.S_ISDIR(root.st_mode) and
             not root.st_mode&0o022,'unsafe_root_ancestor')
        for part in path.parts[1:-1]:
            gate(context)
            owned['next']=os.open(part,os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW|
                                  os.O_NONBLOCK|os.O_CLOEXEC,dir_fd=owned['parent'])
            s=os.fstat(owned['next'])
            need(stat.S_ISDIR(s.st_mode) and s.st_uid in allowed_uids and
                 not s.st_mode&0o022,'unsafe_ancestor_owner_mode_type')
            if allowed_uids==(0,):need(s.st_gid==0,'root_ancestor_gid')
            chain.append(identity(s))
            previous=owned.pop('parent')  # transfer BEFORE close, never recover old number
            owned['parent']=owned.pop('next')
            close_owned({'previous':previous})
        fd=owned.pop('parent')
        return fd,path.name,chain
    finally:
        close_owned(owned,sys.exc_info()[1])


def read_fd(fd, expected_sha, size, uid, mode, context, nlink=1):
    gate(context)
    before = os.fstat(fd)
    need(stat.S_ISREG(before.st_mode) and before.st_uid == uid and
         (uid != 0 or before.st_gid == 0) and before.st_nlink == nlink and
         stat.S_IMODE(before.st_mode) == mode and before.st_size == size,
         'file_owner_mode_size_link_type')
    raw = os.read(fd, size + 1)
    after = os.fstat(fd)
    need(metadata(before) == metadata(after) and len(raw) == size and
         hashlib.sha256(raw).hexdigest() == expected_sha,
         'exact_same_fd_bytes_changed_or_wrong_hash')
    gate(context)
    return raw, after


def capture_source(context):
    owned={}
    try:
        parent,leaf,chain=safe_parent(SOURCE,(0,PRIVATE_OWNER_UID),context);owned['parent']=parent
        owned['source']=os.open(leaf,os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK|os.O_CLOEXEC,dir_fd=parent)
        raw,s=read_fd(owned['source'],WHEEL_SHA,WHEEL_SIZE,PRIVATE_OWNER_UID,0o400,context)
        need(metadata(s)==metadata(os.stat(leaf,dir_fd=parent,follow_symlinks=False)), 'source_name_replaced')
        check,_,current=safe_parent(SOURCE,(0,PRIVATE_OWNER_UID),context);owned['check']=check
        need(chain==current,'source_ancestor_replaced')
        return raw,{'path':str(SOURCE),'sha256':WHEEL_SHA,'bytes':len(raw),
                    'identity':list(identity(s)),'captured_before_destination_effects':True}
    finally:
        close_owned(owned,sys.exc_info()[1])


def verify_target(parent, leaf, context, owned_fd=None):
    fd = os.open(leaf, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK |
                 os.O_CLOEXEC, dir_fd=parent)
    try:
        _, s = read_fd(fd, WHEEL_SHA, WHEEL_SIZE, 0, 0o444, context)
        need(metadata(s) == metadata(os.stat(leaf, dir_fd=parent, follow_symlinks=False)),
             'target_name_changed')
        if owned_fd is not None:
            need(identity(s) == identity(os.fstat(owned_fd)), 'published_inode_changed')
        return {'path': str(DEST), 'sha256': WHEEL_SHA, 'bytes': WHEEL_SIZE,
                'identity': list(identity(s)), 'uid': 0, 'gid': 0, 'mode': '0444',
                'nlink': 1, 'world_readable': True}
    finally:
        close_owned({'target':fd},sys.exc_info()[1])


def publish(context, review):
    need(os.getuid() == os.geteuid() == os.getegid() == 0, 'existing_root_rights_required')
    need(review.get('source_accepted') is True and review.get('contract') == CONTRACT and
         review.get('wheel_sha256') == WHEEL_SHA and review.get('wheel_bytes') == WHEEL_SIZE and
         review.get('destination') == str(DEST) and review.get('runtime_activation') is False and
         review.get('installer_sha256') == context['installer_sha256'],
         'independent_exact_publication_review_required')
    record = {'contract': CONTRACT, 'state': 'PREPARED_NO_DESTINATION_EFFECT', 'source': None,
              'review_sha256': context['review_sha256'], 'original_clock': dict(context),
              'directory_created': False, 'publication_entered': False, 'published': False,
              'target': None, 'anonymous_inode': None, 'cleanup_errors': [],
              'candidate_runtime_started': False, 'retry_authorized': False}
    base = directory = temporary = check = None
    chain = None
    error = None
    try:
        raw,source=capture_source(context)
        record['source']=source
        base, leaf, chain = safe_parent(BASE / 'friday', (0,), context)
        gate(context)
        # mkdir is the only directory effect; do not adopt/chmod/chown shared
        # ancestors. Existing directory is checked without changing its modes.
        try:
            directory = os.open(leaf, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW |
                                os.O_CLOEXEC, dir_fd=base)
        except FileNotFoundError:
            record['directory_creation_entered'] = True
            os.mkdir(leaf, 0o755, dir_fd=base)
            record['directory_created'] = True
            directory = os.open(leaf, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW |
                                os.O_CLOEXEC, dir_fd=base)
            record['directory_identity'] = list(identity(os.fstat(directory)))
            os.fsync(base)
        ds = os.fstat(directory)
        need(stat.S_ISDIR(ds.st_mode) and ds.st_uid == ds.st_gid == 0 and
             not ds.st_mode & 0o022 and ds.st_mode & 0o005 == 0o005 and
             identity(ds) == identity(os.stat(leaf, dir_fd=base, follow_symlinks=False)),
             'destination_directory_owner_mode_identity')
        record['directory_identity'] = list(identity(ds))
        try:
            record['target'] = verify_target(directory, WHEEL_NAME, context)
        except FileNotFoundError:
            # Linux ready filesystem primitive: anonymous O_TMPFILE inode,
            # fixed captured bytes, link via held /proc/self/fd FD. No named
            # partial file or path-based replacement; no fallback if unsupported.
            gate(context)
            record['anonymous_creation_entered'] = True
            temporary = os.open('.', os.O_RDWR | os.O_TMPFILE | os.O_CLOEXEC,
                                0o600, dir_fd=directory)
            record['anonymous_inode'] = list(identity(os.fstat(temporary)))
            view = memoryview(raw)
            while view:
                gate(context)
                n = os.write(temporary, view)
                need(0 < n <= len(view), 'short_or_zero_write')
                view = view[n:]
            os.fchmod(temporary, 0o444)
            os.fsync(temporary)
            os.lseek(temporary, 0, os.SEEK_SET)
            read_fd(temporary, WHEEL_SHA, WHEEL_SIZE, 0, 0o444, context, nlink=0)
            gate(context)
            record['publication_entered'] = True  # unknown interruption is never success
            # follow_symlinks=True uses Linux linkat AT_SYMLINK_FOLLOW for the
            # held procfd. The destination name must be ABSENT; EEXIST refuses.
            os.link('/proc/self/fd/' + str(temporary), WHEEL_NAME,
                    dst_dir_fd=directory, follow_symlinks=True)
            record['published'] = True
            os.fsync(directory)
            record['target'] = verify_target(directory, WHEEL_NAME, context, temporary)
        check, _, current = safe_parent(BASE / 'friday', (0,), context)
        try:
            need(current == chain and identity(os.fstat(check)) == identity(os.fstat(base)) and
                 identity(ds) == identity(os.stat('friday', dir_fd=check, follow_symlinks=False)),
                 'destination_ancestor_replaced')
        finally:
            closing=check;check=None
            close_owned({'check':closing},sys.exc_info()[1])
        gate(context)
        record['state'] = 'EXACT_PROTECTED_WHEEL_PUBLISHED' if record['published'] else 'EXACT_EXISTING_WHEEL_VERIFIED'
    except BaseException as problem:
        error = problem
        record['error'] = type(problem).__name__ + ': ' + str(problem)[:512]
        if isinstance(problem,CleanupUnconfirmed):
            record['cleanup_errors'].extend(problem.cleanup_errors)
            record['original_error']=repr(problem.causal)[:1024]
        record['state'] = 'STOP_UNCONFIRMED_PUBLICATION' if record['publication_entered'] else 'REFUSED_NOT_PUBLISHED'
        if record.get('directory_creation_entered') and not record['directory_created'] and not isinstance(problem, FileExistsError):
            record['state'] = 'STOP_UNCONFIRMED_DIRECTORY_CREATION'
        if record.get('anonymous_creation_entered') and temporary is None and not isinstance(problem, OSError):
            record['state'] = 'STOP_UNCONFIRMED_ANONYMOUS_FD_CREATION'
        # Do not unlink a concurrent or partially confirmed destination, remove
        # a shared directory, retry publication or claim rollback. A newly
        # created directory is retained and its identity recorded for the owner.
    finally:
        remaining={name:fd for name,fd in (('anonymous',temporary),('destination_directory',directory),('base',base),('check',check)) if fd is not None}
        temporary=directory=base=check=None  # ownership transferred before closes
        try:close_owned(remaining,error)
        except CleanupUnconfirmed as problem:record['cleanup_errors'].extend(problem.cleanup_errors)
        if record['cleanup_errors']:
            record['state'] = 'STOP_UNCONFIRMED_CLEANUP'
        record['elapsed_seconds'] = time.monotonic() - context['started_monotonic']
        record['success'] = error is None and not record['cleanup_errors'] and record['elapsed_seconds'] < 60
        if record['elapsed_seconds'] >= 60:
            record['state'] = 'STOP_UNCONFIRMED_LATE'
    return record


def main(context, review):
    os.umask(0o022)  # only future newly-created directory/anonymous inode; no shared chmod
    os.sched_setaffinity(0, {0, 1})
    resource.setrlimit(resource.RLIMIT_AS, (1073741824, 1073741824))
    need(context['deadline_monotonic'] == context['started_monotonic'] + 60,
         'original_exact60_root_entry_clock')
    def expired(*_):
        raise Refused('original_root_entry60_exhausted')
    for sig in (signal.SIGALRM, signal.SIGINT, signal.SIGTERM, signal.SIGHUP):
        signal.signal(sig, expired)
    signal.setitimer(signal.ITIMER_REAL, context['deadline_monotonic'] - time.monotonic())
    try:
        record = publish(context, review)
        raw = (json.dumps(record, sort_keys=True, separators=(',', ':')) + '\n').encode()
        need(len(raw) <= 8192, 'owner_receipt_bound')
        os.set_blocking(1, False)
        need(os.write(1, raw) == len(raw), 'STOP_UNCONFIRMED_owner_receipt_short_or_unavailable')
        if not record['success']:
            raise Refused('publication_refused_or_unconfirmed; inspect exact owner receipt; no retry')
        gate(context, 0)
        return record
    finally:
        signal.setitimer(signal.ITIMER_REAL, 0)
