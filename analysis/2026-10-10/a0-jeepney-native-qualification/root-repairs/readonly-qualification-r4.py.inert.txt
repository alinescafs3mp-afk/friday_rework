def readonly_bus_type(common):
    # Only a method-admission guard; actual ready-library parser/auth/socket/
    # rpc/ownership implementations and all clocks remain the frozen producer.
    class ReadonlyBus(common.ParentJobBus):
        def rpc(self,dest,path,iface,method,signature='',args=(),seconds=5,effect=False):
            daemon=(dest,path,iface)==('org.freedesktop.DBus','/org/freedesktop/DBus','org.freedesktop.DBus')
            manager=bool(self.owner) and dest==self.owner and path==common.BUS_PATH
            rule="type='signal',sender='org.freedesktop.DBus',interface='org.freedesktop.DBus',member='NameOwnerChanged',arg0='"+common.BUS_NAME+"'"
            allowed=daemon and ((method=='Hello' and signature=='' and args==()) or
                (method=='GetNameOwner' and signature=='s' and args==(common.BUS_NAME,)) or
                (method in ('GetConnectionUnixProcessID','GetConnectionUnixUser') and signature=='s' and args==(self.owner,)) or
                (method=='AddMatch' and signature=='s' and args in ((getattr(self,'match',None),),(rule,))))
            allowed=allowed or (manager and ((iface=='org.freedesktop.DBus.Properties' and method=='Get' and signature=='ss' and args==(common.BUS_IFACE,'Version')) or
                (iface==common.BUS_IFACE and method=='Subscribe' and signature=='' and args==())))
            need(allowed and seconds==5 and effect is False,'READONLY_FORBIDDEN_METHOD_BEFORE_SEND')
            return super().rpc(dest,path,iface,method,signature,args,seconds,effect)
        def submit(self,*args,**kwargs):
            raise RuntimeError('READONLY_FORBIDDEN_JOB_BEFORE_SEND')
    return ReadonlyBus

    signal.signal(signal.SIGALRM,expired)

            bus.prepare()

            bus.open()
