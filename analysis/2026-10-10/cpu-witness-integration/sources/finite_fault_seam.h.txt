/* Qualification-only native seam included AFTER finite_birth native globals.
 * Production C/binary is unchanged. No macro interposition/constructor/thread.
 * The distinct compiled unit needs independent source/ELF/provider review.
 * Injections test handling, not production kernel returns. No reset API.
 */
#ifndef FRW_FAULT_QUALIFICATION
#error "Explicit qualification-only compilation required"
#endif
static int q_plan, q_configured, q_sentinel = -1;
static volatile sig_atomic_t q_before_fork, q_after_fork, q_kernel_clone_calls;
static volatile sig_atomic_t q_injected_calls, q_injected_stage;
static volatile sig_atomic_t q_actual_clone_errno, q_create_errno, q_settime_errno;
static volatile sig_atomic_t q_pending_errno;
static volatile sig_atomic_t q_pending_attempts, q_pending_site, q_pending_sent;
/* Persistent endpoint snapshot: constructor rollback can close a successful
 * arm before Python records the selected pending exception. Never reset. */
static volatile sig_atomic_t q_pending_state[7];
static long long q_wall_ns[5], q_cpu_ns[5];
/* Qualification-only first handler-entry witness, preserved through close.
 * SIGXCPU also has non-RLIMIT_CPU producers. Never promote cpu_hit alone.
 * Every handler-written cell is sig_atomic_t; timestamps use sec/nsec cells.
 * No timer, signal injection, threshold tolerance, or reset is introduced. */
static volatile sig_atomic_t q_cpu_soft_installed;
static volatile sig_atomic_t q_cpu_info[26];

static void q_cpu_clock(clockid_t id, int slot)
{
    struct timespec ts;
    if (clock_gettime(id, &ts) < 0) {
        if (!q_cpu_info[6]) q_cpu_info[6] = errno ? errno : EIO;
        return;
    }
    if (ts.tv_sec < 0 || ts.tv_sec > INT_MAX ||
        ts.tv_nsec < 0 || ts.tv_nsec >= 1000000000L) {
        if (!q_cpu_info[6]) q_cpu_info[6] = ERANGE;
        return;
    }
    q_cpu_info[slot] = (sig_atomic_t)ts.tv_sec;
    q_cpu_info[slot+1] = (sig_atomic_t)ts.tv_nsec;
}

static void q_cpu_witness(const siginfo_t *info)
{
    struct rlimit limit;
    if (q_cpu_info[0]) {
        q_cpu_info[0] = 2; /* Multiple deliveries invalidate one-cause credit. */
        return;
    }
    q_cpu_info[0] = 1;
    q_cpu_info[1] = info ? info->si_signo : -1;
    q_cpu_info[2] = info ? info->si_code : -1;
    q_cpu_info[3] = info ? info->si_pid : -1;
    q_cpu_info[4] = info ? (sig_atomic_t)info->si_uid : -1;
    q_cpu_info[5] = info ? info->si_errno : -1;
    q_cpu_info[13] = q_cpu_soft_installed;
    q_cpu_info[16] = borrowing;
    q_cpu_info[17] = armed;
    q_cpu_info[18] = attempted;
    q_cpu_info[19] = q_before_fork;
    q_cpu_info[20] = q_after_fork;
    q_cpu_info[21] = q_kernel_clone_calls;
    q_cpu_info[22] = root_pid;
    q_cpu_info[23] = birth_fd;
    q_cpu_info[24] = cpu_installed;
    q_cpu_info[25] = can_abort;
    if (syscall(SYS_prlimit64, 0, RLIMIT_CPU, NULL, &limit) < 0) {
        q_cpu_info[6] = errno ? errno : EIO;
    } else {
        q_cpu_info[14] = limit.rlim_cur == RLIM_INFINITY ? -1 :
            limit.rlim_cur > INT_MAX ? -2 : (sig_atomic_t)limit.rlim_cur;
        q_cpu_info[15] = limit.rlim_max == RLIM_INFINITY ? -1 :
            limit.rlim_max > INT_MAX ? -2 : (sig_atomic_t)limit.rlim_max;
    }
    /* Linux make_process_cpuclock(0, CPUCLOCK_PROF) = -8: raw utime+stime,
     * the RLIMIT_CPU comparison domain. CLOCK_PROCESS_CPUTIME_ID is SCHED.
     * These are handler-entry reads, not the kernel enqueue timestamp. */
    q_cpu_clock((clockid_t)-8, 7);
    q_cpu_clock(CLOCK_PROCESS_CPUTIME_ID, 9);
    q_cpu_clock(CLOCK_MONOTONIC, 11);
}

static void q_mark(int point)
{
    struct timespec now;
    int saved = errno;
    if (clock_gettime(CLOCK_MONOTONIC, &now) == 0)
        q_wall_ns[point] = (long long)now.tv_sec * 1000000000LL + now.tv_nsec;
    if (clock_gettime(CLOCK_PROCESS_CPUTIME_ID, &now) == 0)
        q_cpu_ns[point] = (long long)now.tv_sec * 1000000000LL + now.tv_nsec;
    errno = saved;
}

int frw_q_configure(int plan, int sentinel)
{
    if (q_configured || armed || attempted || timer_live || alarm_installed || cpu_installed)
        return EALREADY;
    if (plan < 1 || plan > 30) return EINVAL;
    if (plan == 5 && (sentinel < 0 || fcntl(sentinel, F_GETFD) < 0)) return EBADF;
    q_plan = plan; q_sentinel = sentinel; q_configured = 1;
    q_mark(0);
    return 0;
}

/* Returns injected EIO without calling that selected operation. This is NEVER
 * recorded as an observed errno from the real production syscall. */
static int q_fail(int stage)
{
    int selected = (q_plan == 14 && stage == 105) ||
                   (q_plan == 15 && stage == 104) ||
                   (q_plan == 16 && stage == 103) ||
                   (q_plan == 17 && stage == 201) ||
                   (q_plan == 18 && stage == 202) ||
                   (q_plan == 19 && stage == 203) ||
                   (q_plan == 28 && stage == 101) ||
                   (q_plan == 29 && stage == 102);
    if (!selected) return 0;
    ++q_injected_calls; q_injected_stage = stage; errno = EIO;
    return 1;
}

static int q_create(struct sigevent *event, timer_t *id)
{
    int rc = timer_create(q_plan == 12 ? -1 : CLOCK_MONOTONIC, event, id);
    q_create_errno = rc < 0 ? errno : 0;
    return rc;
}

static int q_settime(timer_t id, const struct itimerspec *spec)
{
    struct itimerspec local = *spec;
    if (q_plan == 13) local.it_value.tv_nsec = 1000000000L;
    int rc = timer_settime(id, TIMER_ABSTIME, &local, NULL);
    q_settime_errno = rc < 0 ? errno : 0;
    return rc;
}

static void q_pending(int site)
{
    int saved = errno;
    ++q_pending_attempts;
    q_pending_site = site;
    q_pending_state[0] = armed;
    q_pending_state[1] = attempted;
    q_pending_state[2] = root_pid;
    q_pending_state[3] = birth_fd;
    q_pending_state[4] = q_before_fork;
    q_pending_state[5] = q_after_fork;
    q_pending_state[6] = q_kernel_clone_calls;
    /* Actual OS SIGINT under the selected CPython default KeyboardInterrupt
     * disposition. No PyErr_SetString/fabricated Python raise at the endpoint. */
    if (syscall(SYS_tgkill, getpid(), syscall(SYS_gettid), SIGINT) < 0)
        q_pending_errno = errno;
    else
        ++q_pending_sent;
    errno = saved;
}

static long q_clone(struct clone_args *args, size_t size)
{
    q_mark(2);
    /* Real borrowed soft6 crossing: neither timer nor CPU limit is shortened.
     * Actual bound_signal delivers/longjmps; these loops NEVER call it directly.
     * Clock failure returns an observed errno; there is no fallback timer. */
    if (q_plan == 8) {
        while (!expired) (void)syscall(SYS_pause); /* Original ALRM, no busy CPU6. */
        errno = ECANCELED; return -1;
    }
    if (q_plan == 6) {
        while (!cpu_hit && !expired) {
            struct timespec now;
            if (clock_gettime(CLOCK_MONOTONIC, &now) < 0) return -1;
        }
        errno = ECANCELED; return -1;
    }
    if (q_plan >= 2 && q_plan <= 5) {
        ++q_injected_calls; q_injected_stage = 301;
        /* An EXISTING real sentinel descriptor, not a fake owned pidfd.
         * A failed syscall's stale output must be cleared without closing it.
         * This is a USERSPACE STATE SUBSTITUTE, not kernel put_user rollback. */
        if (q_plan == 5) birth_fd = q_sentinel;
        errno = q_plan == 2 ? ENOSYS : q_plan == 3 ? EPERM : ENOMEM;
        return -1;
    }
    if (q_plan == 1) args->flags |= CLONE_THREAD; /* invalid WITHOUT CLONE_VM */
    ++q_kernel_clone_calls;
    long pid = syscall(SYS_clone3, args, size);
    if (pid == 0) return 0; /* No clock/Python/libffi operation in child branch. */
    int saved = errno;
    q_actual_clone_errno = pid < 0 ? saved : 0;
    q_mark(3);
    if (pid > 0 && q_plan == 9)
        while (!expired) (void)syscall(SYS_pause);
    if (pid > 0 && q_plan == 7) {
        while (!cpu_hit && !expired) {
            struct timespec now;
            if (clock_gettime(CLOCK_MONOTONIC, &now) < 0) break;
        }
    }
    errno = saved;
    return pid;
}

/* Read actual call/state counters; no successful observation is inferred.
 * Index ABI is fixed in native_fault_controls.py. No field reset/second birth. */
static int q_restored(void)
{
    struct sigaction a, c, child;
    sigset_t mask;
    if (sigaction(SIGALRM, NULL, &a) < 0 || sigaction(SIGXCPU, NULL, &c) < 0 ||
        sigaction(SIGCHLD, NULL, &child) < 0 || sigprocmask(SIG_SETMASK, NULL, &mask) < 0)
        return -errno;
    if (a.sa_handler != SIG_DFL || c.sa_handler != SIG_DFL || child.sa_handler != SIG_DFL ||
        (child.sa_flags & SA_NOCLDWAIT) || sigismember(&mask, SIGALRM) ||
        sigismember(&mask, SIGXCPU) || sigismember(&mask, SIGCHLD)) return 0;
    if (q_wall_ns[2]) {
        struct rlimit cpu, as;
        cpu_set_t affinity;
        CPU_ZERO(&affinity);
        if (getrlimit(RLIMIT_CPU, &cpu) < 0 || getrlimit(RLIMIT_AS, &as) < 0 ||
            sched_getaffinity(0, sizeof affinity, &affinity) < 0) return -errno;
        if (cpu.rlim_cur != old_cpu.rlim_cur || cpu.rlim_max != old_cpu.rlim_max ||
            as.rlim_cur != old_as.rlim_cur || as.rlim_max != old_as.rlim_max ||
            !CPU_EQUAL(&affinity, &old_affinity)) return 0;
    }
    return 1;
}

long long frw_q_get(int index)
{
    switch (index) {
    case 0: return q_plan;
    case 1: return q_before_fork;
    case 2: return q_after_fork;
    case 3: return q_kernel_clone_calls;
    case 4: return q_injected_calls;
    case 5: return q_injected_stage;
    case 6: return q_actual_clone_errno;
    case 7: return q_create_errno;
    case 8: return q_settime_errno;
    case 9: return q_pending_errno;
    case 10: return armed;
    case 11: return attempted;
    case 12: return timer_live;
    case 13: return alarm_installed;
    case 14: return cpu_installed;
    case 15: return borrowing;
    case 16: return root_pid;
    case 17: return birth_fd;
    case 18: return cpu_hit;
    case 19: return expired;
    case 20: return signal_errno;
    case 21: return rollback_errno;
    case 22: return q_restored();
    case 23: return q_pending_attempts;
    case 24: return q_pending_site;
    case 25: return q_pending_sent;
    default:
        if (index >= 30 && index < 35) return q_wall_ns[index - 30];
        if (index >= 40 && index < 45) return q_cpu_ns[index - 40];
        if (index >= 50 && index < 57) return q_pending_state[index - 50];
        if (index >= 60 && index < 86) return q_cpu_info[index - 60];
        return -1;
    }
}
