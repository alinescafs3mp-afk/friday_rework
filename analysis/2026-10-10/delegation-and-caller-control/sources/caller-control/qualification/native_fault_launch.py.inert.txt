# INERT ANALYSIS COPY; NOT AN INSTALLATION INPUT OR EXECUTION GRANT.
"""Source-only same-PID launcher. Original clock comes from ROOT, never reset.

Root must materialize an exact permit/source/qualification ELF before use.
Keeps the existing canonical FLOCK through exec/record/exit; no supervisor.
"""
import fcntl
import hashlib
import json
import os
from pathlib import Path
import stat
import sys
import time


def main():
    if len(sys.argv) == 6 and sys.argv[2] == '--caller-control':
        return contained_main(Path(sys.argv[1]).resolve(), *(int(v) for v in sys.argv[3:]))
    here = Path(__file__).resolve().parent
    assert len(sys.argv) == 2
    permit = Path(sys.argv[1]).resolve()
    p = json.loads(permit.read_bytes())
    assert p['schema'] == 'friday.native-fault-observation-permit.v1'
    assert p['source_reviewed_observation_authorized'] is True
    assert p['product_mode_admission'] is False and p['retry_authorized'] is False
    assert p['scope'] == 'ONE_SELECTED_FAULT_PLAN_ORIGINAL_CLOCK_ONE_BIRTH_NO_RETRY'
    assert p['plan'] not in ('original180-before-publication-source', 'original180-after-publication-source')
    assert p['qualification_directory'] == str(here) and p['permit_path'] == str(permit)
    clock = p['original_clock']
    assert clock['budget_seconds'] == 180 and clock['cpu_seconds'] == 45
    assert clock['deadline_monotonic']-clock['started_monotonic'] == 180
    assert clock['started_monotonic'] <= time.monotonic() < clock['deadline_monotonic']
    assert clock['boot_id'] == Path('/proc/sys/kernel/random/boot_id').read_text().strip()
    runtime = here.parent.parent/'native-fault-run'
    assert p['runtime_directory'] == str(runtime) and not runtime.exists()
    assert hashlib.sha256((here/'source-manifest.json').read_bytes()).hexdigest() == p['source_manifest_sha256']
    assert hashlib.sha256(Path(p['interpreter']).read_bytes()).hexdigest() == p['interpreter_sha256']
    binding = Path(p['launch_binding'])
    assert binding == here.parent.parent/'native-fault-launch-binding.json' and not binding.exists()
    fd = os.open('PRIVATE_QUALITY_LOCK_PATH', os.O_RDONLY | os.O_NOFOLLOW)
    s = os.fstat(fd)
    identity = [s.st_dev, s.st_ino, s.st_uid, s.st_gid, s.st_nlink, s.st_size, stat.S_IMODE(s.st_mode)]
    assert identity == [PRIVATE_QUALITY_DEVICE, PRIVATE_QUALITY_INODE, PRIVATE_QUALITY_UID, PRIVATE_QUALITY_GID, PRIVATE_QUALITY_NLINK, PRIVATE_QUALITY_SIZE, PRIVATE_QUALITY_MODE]
    fcntl.flock(fd, fcntl.LOCK_EX | fcntl.LOCK_NB)
    os.set_inheritable(fd, True)
    argv = [p['interpreter'], '-I', '-B', '-S', str(here/'invoke_qualification.py'), '--native-faults', str(permit)]
    record = {'pid': os.getpid(), 'start_ticks': int(Path('/proc/self/stat').read_text().rsplit(')', 1)[1].split()[19]),
              'boot_id': clock['boot_id'], 'permit_sha256': hashlib.sha256(permit.read_bytes()).hexdigest(),
              'quality_identity': identity, 'quality_fd': fd, 'original_clock': clock,
              'argv': argv, 'same_pid_exec': True, 'new_supervisor': False, 'new_budget': False}
    out = os.open(binding, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600)
    with os.fdopen(out, 'w') as stream:
        json.dump(record, stream, indent=2); stream.write('\n'); stream.flush(); os.fsync(stream.fileno())
    directory = os.open(binding.parent, os.O_RDONLY | os.O_DIRECTORY)
    os.fsync(directory); os.close(directory)
    os.execve(p['interpreter'], argv, {'PATH': '/usr/bin:/bin', 'LANG': 'C', 'LC_ALL': 'C'})


CALLER_TOPOLOGY = 'FOREGROUND_CONTROLLER_BWRAP_MONITOR_PID1_CALLER_ONE_NATIVE_CHILD_V1'


def control_require(ok, reason):
    if not ok:
        raise RuntimeError('CALLER_CONTROL:' + reason)


def control_object(raw):
    def unique(pairs):
        result = {}
        for key, value in pairs:
            control_require(key not in result, 'DUPLICATE_JSON_KEY')
            result[key] = value
        return result
    result = json.loads(raw, object_pairs_hook=unique,
                        parse_constant=lambda _: (_ for _ in ()).throw(ValueError('NONFINITE_JSON')))
    control_require(type(result) is dict, 'OBJECT_REQUIRED')
    return result


def validate_control_grant(grant, ready, clock, controller, now):
    import math
    control_require(type(grant) is dict and set(grant) == {
        'op', 'topology', 'controller', 'monitor', 'caller', 'original_clock', 'permit_sha256', 'quality', 'binary'},
        'GRANT_SCHEMA')
    control_require(ready['op'] == 'READY' and type(ready['pid']) is int and ready['pid'] == 1, 'READY_SCHEMA')
    control_require(grant['op'] == 'GRANT' and grant['topology'] == CALLER_TOPOLOGY
                    and grant['original_clock'] == clock and grant['controller'] == controller
                    and grant['permit_sha256'] == ready['permit_sha256'], 'STALE_WRONG_GRANT')
    start, end = clock.get('started_monotonic'), clock.get('deadline_monotonic')
    control_require(all(type(v) in (int, float) and math.isfinite(v) for v in (start, end, now))
                    and clock['budget_seconds'] == 180 and clock['cpu_seconds'] == 45
                    and end - start == 180 and start <= now < end, 'LATE_OR_RESET_GRANT')
    caller, monitor = grant['caller'], grant['monitor']
    control_require(all(type(v['pid']) is int and v['pid'] > 1
                        and type(v['ppid']) is int and v['ppid'] >= 0
                        and type(v['start_ticks']) is int and v['start_ticks'] > 0
                        for v in (caller, monitor, controller)), 'IDENTITY_FIELD_TYPES')
    control_require(type(caller['pid']) is int and caller['pid'] > 1
                    and caller['nspid'] == [caller['pid'], 1]
                    and caller['ppid'] == monitor['pid'] and monitor['ppid'] == controller['pid']
                    and all(caller[k] == ready[k] for k in
                            ('start_ticks', 'pid_namespace', 'mount_namespace', 'boot_id'))
                    and caller['boot_id'] == clock['boot_id'] == controller['boot_id'], 'WRONG_HOST_CALLER')
    control_require(grant['quality']['owner'] == controller and grant['quality']['identity'] ==
                    [PRIVATE_QUALITY_DEVICE, PRIVATE_QUALITY_INODE, PRIVATE_QUALITY_UID, PRIVATE_QUALITY_GID, PRIVATE_QUALITY_NLINK, PRIVATE_QUALITY_SIZE, PRIVATE_QUALITY_MODE], 'WRONG_LOCK_OWNER')
    return True


def proc_relative_text(directory_fd, name, maximum=16384):
    fd = os.open(name, os.O_RDONLY | os.O_CLOEXEC, dir_fd=directory_fd)
    try:
        raw = os.read(fd, maximum + 1)
        control_require(len(raw) <= maximum, 'PROC_RECORD_OVERSIZE')
        return raw.decode()
    finally:
        os.close(fd)


def observed_controller(procfd):
    raw = proc_relative_text(procfd, 'stat')
    fields = raw.rsplit(')', 1)[1].split()
    nspid = [int(v) for line in proc_relative_text(procfd, 'status').splitlines()
             if line.startswith('NSpid:') for v in line.split()[1:]]
    return dict(pid=int(raw.split('(', 1)[0]), start_ticks=int(fields[19]), ppid=int(fields[1]), nspid=nspid,
                pid_namespace=os.readlink('ns/pid', dir_fd=procfd),
                mount_namespace=os.readlink('ns/mnt', dir_fd=procfd),
                boot_id=Path('/proc/sys/kernel/random/boot_id').read_text().strip())


def control_pidfd_target(text, expected_pid):
    if type(expected_pid) is not int or expected_pid <= 0:
        return False
    return [line.split() for line in text.splitlines() if line.startswith('Pid:')] == [['Pid:', str(expected_pid)]]


def control_live(fd):
    import select
    control_require(not select.select([fd], [], [], 0)[0], 'CONTROLLER_DEAD_OR_PIDFD_CLOSED')
    target = os.readlink('/proc/self/fd/' + str(fd))
    control_require(target == 'anon_inode:[pidfd]', 'NOT_A_PIDFD')


def control_wait(channel, parentfd, deadline, write=False):
    import select
    control_live(parentfd)
    remaining = deadline - time.monotonic()
    control_require(remaining > 0, 'ORIGINAL_DEADLINE_EXPIRED')
    readers, writers, _ = select.select([parentfd] if write else [parentfd, channel],
                                         [channel] if write else [], [], remaining)
    control_require(parentfd not in readers and (channel in writers if write else channel in readers),
                    'CONTROLLER_DEAD_OR_HANDSHAKE_TIMEOUT')
    control_live(parentfd)


def contained_main(permit_path, channel_fd, controller_pidfd, controller_procfd):
    # Only stdlib bootstrap precedes the positive descriptor grant. In particular
    # native_provider_admission/finite_kernel/candidate imports happen after exec.
    import array
    import resource
    import select
    import socket
    here = Path(__file__).resolve().parent
    control_require(sys.flags.isolated and sys.flags.no_site and sys.dont_write_bytecode
                    and not sys.flags.optimize and os.getpid() == 1 and os.getppid() == 0,
                    'NEW_PID1_CALLER_REQUIRED')
    control_require(all(type(fd) is int and fd >= 3 for fd in (channel_fd, controller_pidfd, controller_procfd))
                    and len({channel_fd, controller_pidfd, controller_procfd}) == 3, 'INHERITED_DESCRIPTORS')
    p_raw = permit_path.read_bytes(); p = control_object(p_raw)
    control_require(p['schema'] == 'friday.native-fault-observation-permit.v1'
                    and p['source_reviewed_observation_authorized'] is True
                    and p['product_mode_admission'] is False and p['retry_authorized'] is False
                    and p['scope'] == 'ONE_SELECTED_FAULT_PLAN_ORIGINAL_CLOCK_ONE_BIRTH_NO_RETRY'
                    and p['qualification_directory'] == str(here) and p['permit_path'] == str(permit_path), 'EXACT_PERMIT')
    control_require(p['plan'] not in ('original180-before-publication-source', 'original180-after-publication-source'),
                    'PRESERVED_EXPIRY_STOP')
    c = p['original_clock']; deadline = c['deadline_monotonic']
    controller = observed_controller(controller_procfd)
    control_require(controller['nspid'] == [controller['pid']], 'HOST_CONTROLLER_REQUIRED')
    control_live(controller_pidfd)
    # Existing pinned bwrap installs child PDEATHSIG *before* exec. Its monitor
    # installs its PDEATHSIG *before* child-pid publication. This post-exec check
    # against the inherited exact controller PIDFD closes death-before-arm races;
    # no candidate module is reached if the parent already died. Subsequent death
    # is handled by that already-armed kernel chain, not a Python heartbeat.
    fields = Path('/proc/self/stat').read_text().rsplit(')', 1)[1].split()
    ready = dict(op='READY', pid=1, start_ticks=int(fields[19]),
                 pid_namespace=os.readlink('/proc/self/ns/pid'), mount_namespace=os.readlink('/proc/self/ns/mnt'),
                 boot_id=Path('/proc/sys/kernel/random/boot_id').read_text().strip(),
                 permit_sha256=hashlib.sha256(p_raw).hexdigest())
    # Root6 is installed by the existing native operation and may become soft7;
    # hard45 must survive. These per-process limits are NOT aggregate tree45.
    control_require(all(v == resource.RLIM_INFINITY or v >= 45 for v in resource.getrlimit(resource.RLIMIT_CPU)),
                    'HARD45_NATIVE_CONTRACT')
    resource.setrlimit(resource.RLIMIT_CPU, (45, 45))
    resource.setrlimit(resource.RLIMIT_AS, (2 << 30, 2 << 30))
    os.sched_setaffinity(0, {0, 1})
    channel = socket.socket(fileno=channel_fd); channel.setblocking(False)
    control_wait(channel, controller_pidfd, deadline, write=True)
    raw = json.dumps(ready, sort_keys=True, separators=(',', ':')).encode()
    control_require(channel.send(raw) == len(raw), 'SHORT_READY')
    control_wait(channel, controller_pidfd, deadline)
    raw, ancillary, flags, _ = channel.recvmsg(16385, socket.CMSG_SPACE(4 * array.array('i').itemsize), socket.MSG_CMSG_CLOEXEC)
    received = []
    try:
        for level, kind, body in ancillary:
            control_require(level == socket.SOL_SOCKET and kind == socket.SCM_RIGHTS, 'UNEXPECTED_CONTROL_MESSAGE')
            integers = array.array('i'); integers.frombytes(body); received.extend(integers)
        control_require(raw and len(raw) <= 16384 and not flags & ~socket.MSG_CMSG_CLOEXEC
                        and len(ancillary) == 1 and len(received) == 4 and len(set(received)) == 4,
                        'GRANT_EOF_TRUNCATION_OR_MISSING_RIGHTS')
        grant = control_object(raw)
        validate_control_grant(grant, ready, c, controller, time.monotonic())
        control_require(p['caller_control']['topology'] == CALLER_TOPOLOGY
                        and p['caller_control']['binary'] == grant['binary']
                        and grant['binary']['path'] == '/usr/local/libexec/friday/bwrap-frw-success-only-v1'
                        and grant['binary']['source_sha256'] == 'ee31ef74f0f01efe0aaba4324821d3ccd3b0874fc8099dd07026fd57012b5d0b'
                        and grant['binary']['gate'] == 'SUCCESS_ONLY_BYTE_1_EOF_ERROR_REFUSE', 'PATCHED_LIFETIME_SOURCE')
        quality_fd, self_pidfd, monitor_pidfd, caller_procfd = received
        control_live(controller_pidfd); control_live(self_pidfd); control_live(monitor_pidfd)
        self_info = Path(f'/proc/self/fdinfo/{self_pidfd}').read_text().splitlines()
        control_require([line.split()[1:] for line in self_info if line.startswith('Pid:')] == [['1']],
                        'GRANTED_PIDFD_IS_NOT_THIS_CALLER')
        st = os.fstat(quality_fd)
        actual = [st.st_dev, st.st_ino, st.st_uid, st.st_gid, st.st_nlink, st.st_size, stat.S_IMODE(st.st_mode)]
        control_require(actual == grant['quality']['identity'], 'GRANTED_QUALITY_IDENTITY')
        # Sender acquired this OFD once; SCM_RIGHTS duplicates THAT open file
        # description into this process. No reopen, flock conversion or unlock.
        # Host procfds render the external owner's PID. Inner /proc may hide
        # an out-of-namespace lock owner, so it is NOT an ownership oracle.
        owner_info = proc_relative_text(controller_procfd, 'fdinfo/' + str(grant['quality']['fd']))
        token = f'{os.major(st.st_dev):02x}:{os.minor(st.st_dev):02x}:{st.st_ino}'
        rows = [line.split() for line in owner_info.splitlines() if line.startswith('lock:')]
        control_require(any(len(r) == 9 and r[2:5] == ['FLOCK', 'ADVISORY', 'WRITE']
                            and r[5] == str(controller['pid']) and r[6] == token for r in rows), 'OWNER_FLOCK_NOT_VISIBLE')
        host_self = proc_relative_text(caller_procfd, 'stat')
        control_require(int(host_self.split('(', 1)[0]) == grant['caller']['pid']
                        and int(host_self.rsplit(')', 1)[1].split()[19]) == ready['start_ticks'],
                        'HOST_PROC_FD_IS_NOT_THIS_CALLER')
        # Host proc fdinfo renders the actual targets even when controller and
        # monitor are invisible in inner /proc. Different fd numbers/alive bits
        # alone would also accept three descriptors for the same wrong process.
        for handle, target in ((controller_pidfd, controller['pid']),
                               (self_pidfd, grant['caller']['pid']),
                               (monitor_pidfd, grant['monitor']['pid'])):
            control_live(handle)
            info = proc_relative_text(caller_procfd, 'fdinfo/' + str(handle))
            control_require(control_pidfd_target(info, target), 'WRONG_LIVE_HOST_PIDFD_TARGET')
            control_live(handle)
        inherited = [line.split() for line in proc_relative_text(caller_procfd, 'fdinfo/' + str(quality_fd)).splitlines()
                     if line.startswith('lock:')]
        control_require(any(len(r) == 9 and r[2:5] == ['FLOCK', 'ADVISORY', 'WRITE']
                            and r[5] == str(controller['pid']) and r[6] == token for r in inherited),
                        'CALLER_SHARED_OFD_LOCK_NOT_VISIBLE_IN_HOST_PROC')
        output = Path(p['caller_control']['output_directory'])
        control_require(output.is_absolute() and output.resolve() == output
                        and output != here.parent and output not in here.parent.parents and here.parent not in output.parents
                        and p['runtime_directory'] == str(output/'native-fault-run')
                        and p['launch_binding'] == str(output/'launch-binding.json'), 'EXACT_WRITABLE_OUTPUT')
        control_require(hashlib.sha256((here/'source-manifest.json').read_bytes()).hexdigest() == p['source_manifest_sha256']
                        and hashlib.sha256(Path(p['interpreter']).read_bytes()).hexdigest() == p['interpreter_sha256']
                        and hashlib.sha256(Path('/proc/self/exe').read_bytes()).hexdigest() == p['interpreter_sha256'],
                        'SOURCE_INTERPRETER_CHANGED')
        argv = [p['interpreter'], '-I', '-B', '-S', str(here/'invoke_qualification.py'), '--native-faults', str(permit_path)]
        external = dict(topology=CALLER_TOPOLOGY, controller=controller, caller=grant['caller'], monitor=grant['monitor'],
                        controller_pidfd=controller_pidfd, controller_procfd=controller_procfd,
                        caller_pidfd=self_pidfd, monitor_pidfd=monitor_pidfd, caller_procfd=caller_procfd,
                        controller_quality_fd=grant['quality']['fd'], scm_rights_shared_open_description=True)
        record = dict(pid=1, host_pid=grant['caller']['pid'], start_ticks=ready['start_ticks'], boot_id=c['boot_id'],
                      permit_sha256=ready['permit_sha256'], original_clock=c, quality_fd=quality_fd,
                      quality_identity=actual, external_control=external, argv=argv, same_pid_exec=True,
                      new_supervisor=False, new_budget=False, topology_newly_qualified=False)
        fd = os.open(p['launch_binding'], os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600)
        with os.fdopen(fd, 'w') as stream:
            json.dump(record, stream, indent=2); stream.write('\n'); stream.flush(); os.fsync(stream.fileno())
        for fd in (quality_fd, self_pidfd, monitor_pidfd, controller_pidfd, controller_procfd, caller_procfd): os.set_inheritable(fd, True)
        control_wait(channel, controller_pidfd, deadline, write=True)
        raw = json.dumps({'op': 'ACCEPTED', 'start_ticks': ready['start_ticks'], 'quality_fd': quality_fd}, separators=(',', ':')).encode()
        control_require(channel.send(raw) == len(raw), 'SHORT_ACCEPT')
        control_live(controller_pidfd)
        control_require(time.monotonic() < deadline, 'LATE_BEFORE_CANDIDATE_EXEC')
        channel.close()
        os.execve(p['interpreter'], argv, {'PATH': '/usr/bin:/bin', 'LANG': 'C', 'LC_ALL': 'C'})
    finally:
        # Successful exec retains the exact shared lock/lifetime descriptors.
        # A failed handoff reaches no candidate import and does not unlock the
        # parent's OFD; process exit closes only this process's descriptor copies.
        for fd in received:
            try: os.close(fd)
            except OSError: pass
        channel.close()


if __name__ == '__main__':
    main()
