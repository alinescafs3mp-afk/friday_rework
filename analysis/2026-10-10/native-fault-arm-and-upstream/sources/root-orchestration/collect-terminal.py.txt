# Inert analytical source export; no execution or deployment grant.
"""One post-tool-terminal read, never a launcher, retry, signaler or runtime grant.
Invoke only after the original exact launch command returned a terminal exit.
Persists raw closure observations; independent acceptance is a later decision.
"""
import fcntl
import hashlib
import json
import os
from pathlib import Path
import stat
import sys
import time
from datetime import datetime
ROOT = Path(__file__).resolve().parent
SOURCE = ROOT
PERMIT = SOURCE/'ROOT-ISSUED-NATIVE-FAULT-PERMIT.json'
Q = SOURCE/'candidate'/'qualification'


def main():
    assert len(sys.argv) == 3
    tool_exit, tool_wall = int(sys.argv[1]), float(sys.argv[2])
    permit = json.loads(PERMIT.read_bytes())
    clock = permit['original_clock']
    assert permit['qualification_directory'] == str(Q)
    assert clock['deadline_monotonic'] - clock['started_monotonic'] == 180
    assert clock['boot_id'] == Path('/proc/sys/kernel/random/boot_id').read_text().strip()
    data, hashes, missing = {}, {}, []
    run = SOURCE/'native-fault-run'
    assert permit['runtime_directory'] == str(run)
    paths = [SOURCE/'native-fault-launch-binding.json', run/'clock.json',
             run/'observation.json', run/'receipt-failure-evidence.json', run/'native-close.json']
    for path in paths:
        if path.exists():
            raw = path.read_bytes()
            hashes[str(path)] = hashlib.sha256(raw).hexdigest()
            data[str(path)] = json.loads(raw)
        else:
            missing.append(str(path))
    identities = {}
    def visit(value):
        if isinstance(value, dict):
            if type(value.get('pid')) is int and type(value.get('start_ticks')) is int:
                identities[(value['pid'], value['start_ticks'])] = True
            for child in value.values():
                visit(child)
        elif isinstance(value, list):
            for child in value:
                visit(child)
    visit(data)
    observations = []
    for pid, start in sorted(identities):
        path = Path('/proc')/str(pid)/'stat'
        try:
            raw = path.read_text()
        except FileNotFoundError:
            observations.append({'pid':pid, 'expected_start_ticks':start, 'state':'ABSENT'})
        else:
            actual = int(raw[raw.rfind(')')+2:].split()[19])
            observations.append({'pid':pid, 'expected_start_ticks':start, 'actual_start_ticks':actual,
                                 'state':'SAME_PROCESS_PRESENT' if actual == start else 'PID_REUSED_DIFFERENT_PROCESS'})
    lock = {'lock_path': '/PRIVATE_RUNTIME/friday-quality-ui-exclusive.lock',
            'lock_identity': [PUBLIC_LOCK_DEV, PUBLIC_LOCK_INODE, PUBLIC_LOCK_UID, PUBLIC_LOCK_GID, 1, 0, 0o664]}
    fd = None
    reacquired = False
    lock_error = None
    try:
        fd = os.open(lock['lock_path'], os.O_RDONLY | os.O_NOFOLLOW | os.O_CLOEXEC)
        st = os.fstat(fd)
        identity = [st.st_dev, st.st_ino, st.st_uid, st.st_gid, st.st_nlink, st.st_size, stat.S_IMODE(st.st_mode)]
        assert stat.S_ISREG(st.st_mode) and identity == lock['lock_identity']
        assert (Path(lock['lock_path']).lstat().st_dev, Path(lock['lock_path']).lstat().st_ino) == (st.st_dev, st.st_ino)
        assert observations and all(row['state'] != 'SAME_PROCESS_PRESENT' for row in observations)
        fcntl.flock(fd, fcntl.LOCK_EX | fcntl.LOCK_NB)
        reacquired = True
    except BaseException as error:
        lock_error = repr(error)
    finally:
        if fd is not None:
            os.close(fd)
    now = time.monotonic()
    result = {'state':'EXTERNAL_TERMINAL_OBSERVATIONS_NOT_RUNTIME_ACCEPTANCE',
              'atMSK':datetime.now().astimezone().isoformat(), 'observed_monotonic':now,
              'original_deadline_monotonic':clock['deadline_monotonic'],
              'elapsed_from_original_clock':now-clock['started_monotonic'],
              'observed_before_original_deadline':now < clock['deadline_monotonic'],
              'tool_exit_code':tool_exit, 'tool_last_call_wait_seconds':tool_wall,
              'tool_wait_is_not_total_runtime':True,
              'permit_sha256':hashlib.sha256(PERMIT.read_bytes()).hexdigest(),
              'known_recorded_identities':observations, 'files':hashes, 'missing':missing,
              'raw_records':data, 'quality_lock_reacquired_after_known_caller_exit':reacquired,
              'quality_lock_probe_released':fd is None or not os.path.exists('/proc/self/fd/'+str(fd)),
              'quality_lock_error':lock_error, 'new_signal_or_process_control':False,
              'retry_authorized':False, 'product_acceptance':False}
    out = ROOT/'external-terminal-observation.json'
    f = os.open(out, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600)
    with os.fdopen(f,'w') as stream:
        json.dump(result,stream,indent=2,allow_nan=False)
        stream.write('\n');stream.flush();os.fsync(stream.fileno())
    print(json.dumps({k:result[k] for k in ('state','elapsed_from_original_clock','observed_before_original_deadline','tool_exit_code','quality_lock_reacquired_after_known_caller_exit','quality_lock_error')}))


if __name__ == '__main__':
    main()
