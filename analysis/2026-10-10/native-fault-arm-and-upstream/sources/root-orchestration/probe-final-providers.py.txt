# Inert analytical source export; no execution or deployment grant.
"""Read the selected interpreter's symbol providers; never load candidate code."""
import ctypes
import hashlib
import json
import os
from pathlib import Path
import sys
from datetime import datetime

HERE = Path(__file__).resolve().parent
EVIDENCE = Path('/PRIVATE_RUNTIME/reviewer-workspace/FRW041-NATIVE-FAULT-FINAL-REVIEW-evidence/provider-version-checks.json')
assert hashlib.sha256(EVIDENCE.read_bytes()).hexdigest() == 'ee642ea2acdca2663c7987e9fa0259d5c10c763a40aa6e712c84eae418db1882'
assert sys.version_info[:3] == (3, 14, 7) and sys.flags.isolated and sys.flags.no_site and sys.dont_write_bytecode
assert sys._is_gil_enabled() and not any(k.startswith('LD_') for k in os.environ)
assert hashlib.sha256(Path('/proc/self/exe').read_bytes()).hexdigest() == '8dfa9757a52b9c3edf1dedaaa2a7a8c40ea4beb058f20e90bdd47b48f3b1b176'

class DlInfo(ctypes.Structure):
    _fields_ = [('fname', ctypes.c_char_p), ('base', ctypes.c_void_p),
                ('sname', ctypes.c_char_p), ('saddr', ctypes.c_void_p)]

namespace = ctypes.CDLL(None)
dladdr = namespace.dladdr
dladdr.argtypes = [ctypes.c_void_p, ctypes.POINTER(DlInfo)]
dladdr.restype = ctypes.c_int
rows = []
for expected in json.loads(EVIDENCE.read_bytes())['rows']:
    pointer = ctypes.cast(getattr(namespace, expected['symbol']), ctypes.c_void_p)
    info = DlInfo()
    assert dladdr(pointer, ctypes.byref(info)) == 1 and info.fname
    path = Path(os.fsdecode(info.fname)).resolve()
    sha = hashlib.sha256(path.read_bytes()).hexdigest()
    wanted = ('8dfa9757a52b9c3edf1dedaaa2a7a8c40ea4beb058f20e90bdd47b48f3b1b176'
              if expected['current_static_provider'] == 'selected_CPython'
              else '85e64f97e348786a8fb4d9f3d52fec289e2fb86bba20f0731dfe61990525e0f7')
    assert sha == wanted
    rows.append({**expected, 'actual_default_namespace_provider': str(path), 'sha256': sha})
mapped = {}
for line in Path('/proc/self/maps').read_text().splitlines():
    parts = line.split(None, 5)
    if len(parts) == 6 and parts[5].startswith('/'):
        path = Path(parts[5])
        assert '(deleted)' not in parts[5]
        mapped[str(path.resolve())] = hashlib.sha256(path.read_bytes()).hexdigest()
assert not any('finite_birth' in p for p in mapped)
result = {'state': 'SELECTED_INTERPRETER_DEFAULT_NAMESPACE_PROVIDERS_OBSERVED_NO_CANDIDATE_LOAD',
          'atMSK': datetime.now().astimezone().isoformat(), 'version': sys.version,
          'rows': rows, 'mapped_files': mapped, 'candidate_loaded': False,
          'candidate_called': False, 'product_acceptance': False,
          'limitation': 'Current default namespace plus independently checked ELF symbol versions; candidate runtime not observed by this probe.'}
out = HERE/'final-provider-observation.json'
fd = os.open(out, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600)
with os.fdopen(fd, 'w') as f:
    json.dump(result, f, indent=2); f.write('\n'); f.flush(); os.fsync(f.fileno())
print(json.dumps({'symbols': len(rows), 'mapped_files': len(mapped), 'candidate_loaded': False,
                  'result_sha256': hashlib.sha256(out.read_bytes()).hexdigest()}))
