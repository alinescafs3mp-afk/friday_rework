from pathlib import Path
import datetime
import hashlib
import json
import os
import stat

os.umask(0o077)
W = Path('/PRIVATE_RUNTIME/root-workspace')
D = W/'native-fault-final-composition'
Q = D/'candidate/qualification'

def ref(p):
    p = Path(p)
    b = p.read_bytes()
    return {'path': str(p), 'sha256': hashlib.sha256(b).hexdigest()}

def checked(p, sha):
    assert ref(p)['sha256'] == sha, str(p)
    return json.loads(Path(p).read_bytes())

def save(p, value):
    fd = os.open(p, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600)
    with os.fdopen(fd, 'w') as f:
        json.dump(value, f, indent=2, allow_nan=False)
        f.write('\n'); f.flush(); os.fsync(f.fileno())
    fd = os.open(Path(p).parent, os.O_RDONLY | os.O_DIRECTORY)
    os.fsync(fd); os.close(fd)
