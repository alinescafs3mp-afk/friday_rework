import re

p = json.loads(sys.argv[1])
host = p['host_before']
assert time.monotonic() + 3 < p['deadline']
assert Path('/proc/sys/kernel/random/boot_id').read_text().strip() == host['boot_id']
line = Path('/proc/self/cgroup').read_text().strip()

assert {k:(root/k).read_text().strip() for k in host['caps']} == host['caps']
assert 0 < len(os.sched_getaffinity(0)) <= 2
assert os.getpid() == os.getpgrp() == os.getsid(0)

def original():
    assert time.monotonic() + 3 < p['deadline']

def held(fd):
    st = os.fstat(fd)
    assert (st.st_dev,st.st_ino,st.st_uid,st.st_gid,st.st_nlink,st.st_size,stat.S_IMODE(st.st_mode)) == (PRIVATE_LOCK_DEV,PRIVATE_LOCK_INO,PRIVATE_LOCK_UID,PRIVATE_LOCK_GID,1,0,0o664)
    text=Path(f'/proc/self/fdinfo/{fd}').read_text()
    assert any('FLOCK  ADVISORY  WRITE' in line and f'{st.st_ino} ' in line for line in text.splitlines())
    assert any('FLOCK  ADVISORY  WRITE' in line and f' {host["owner_pid"]} ' in line and f'{st.st_ino} ' in line for line in Path('/proc/locks').read_text().splitlines())
    return text


def sealed(name, raw):
    fd = os.memfd_create(name, os.MFD_ALLOW_SEALING | os.MFD_CLOEXEC)
    try:
        view = memoryview(raw)
        while view:
            n = os.write(fd, view)
            if n <= 0: raise ValueError('sealed_observation_short_write')
            view = view[n:]
        # Existing Linux ABI policy: these constants are not exported by the
        # selected CPython. Never modify fcntl globally or accept a writable FD.
        assert sys.platform == 'linux' and os.uname().machine in ('x86_64', 'aarch64')
        fcntl.fcntl(fd, 1033, 15)
        assert fcntl.fcntl(fd, 1034) == 15 and os.fstat(fd).st_nlink == 0
        os.lseek(fd, 0, os.SEEK_SET)
        os.set_inheritable(fd, True)
        return fd
    except BaseException:
        os.close(fd)
        raise


original(); held(p['quality_fd'])
assert int(Path(f'/proc/{host["owner_pid"]}/stat').read_text().rsplit(')',1)[1].split()[19]) == host['owner_birth']
assert int(Path(f'/proc/{host["scope_pid"]}/stat').read_text().rsplit(')',1)[1].split()[19]) == host['scope_birth']
metadata=json.dumps(host,sort_keys=True).encode()
custody=sealed('frw-ts-host-scope-observation',metadata)

mount_ref = p['mounts']; raw = Path(mount_ref['path']).read_bytes()
assert hashlib.sha256(raw).hexdigest() == mount_ref['sha256']
mount = json.loads(raw)
args = ['--json-status-fd', str(p['native_status_fd']), '--unshare-pid',
    '--die-with-parent', '--new-session', '--as-pid-1', '--unshare-net',
    '--unshare-ipc', '--unshare-uts', '--tmpfs', '/']
files = set(mount['support_files']) | set(mount['ro_files'])
links = dict(mount['support_links'], **mount['links'])
ro_roots = mount['ro_roots']
assert ro_roots == [p['ts_root'], '/usr/lib/python3.14']
destinations = set(files) | set(ro_roots) | set(links) | {str(root), p['run'], '/dev', '/proc'}
directories = {'/sys', '/sys/fs', '/sys/fs/cgroup', '/usr', '/usr/lib', '/usr/lib64'}
for item in destinations:
    q = Path(item)
    assert q.is_absolute() and '..' not in q.parts
    directories.update(str(parent) for parent in q.parents if str(parent) != '/')
for directory in sorted(directories, key=lambda value: (value.count('/'), value)):
    # Never create a directory on top of a deliberately selected link.
    if directory not in links: args += ['--dir', directory]
for source in ro_roots: args += ['--ro-bind', source, source]
for source in sorted(files):
    if not any(Path(source).is_relative_to(Path(q)) for q in ro_roots):
        args += ['--ro-bind', source, source]
for link, target in sorted(links.items()): args += ['--symlink', target, link]
args += ['--ro-bind', str(root), str(root), '--bind', p['run'], p['run'],
         '--dev', '/dev', '--proc', '/proc', '--remount-ro', '/', '--']
command = sys.argv[2:]
assert command[-1] == 'FRW_TS_CUSTODY_FD'
command[-1] = str(custody)
args += command
# Existing bwrap --args accepts NUL-delimited argv through a file descriptor.
# This avoids ARG_MAX for the complete unpruned support file/link inventory.
argument_fd = sealed('frw-ts-exact-bwrap-args', b'\0'.join(os.fsencode(a) for a in args) + b'\0')
original(); held(p['quality_fd'])
os.execve('/usr/bin/bwrap', ['/usr/bin/bwrap', '--args', str(argument_fd)], dict(os.environ))
