"""NOT_RUN assertions for root's real bounded native fault controls.

This file launches nothing and supplies no owner/clock/admission. The original
root must pass the actual primary Popen captured at creation, original observed
birth and original deadline after its independently supported fault injection.
"""
def retained_original(error, process, birth, deadline, *, stop_unconfirmed):
    assert error.owned_process is process
    assert error.custody.process is process
    assert error.custody.deadline == deadline
    assert error.custody.starttime == birth
    assert error.custody.cleanup_started is True
    assert error.observation['pid'] == process.pid
    assert error.observation['starttime_ticks'] == birth
    assert error.observation['original_deadline'] == deadline
    assert error.observation['returncode'] == process.returncode
    assert error.observation['reaped'] == (process.returncode is not None)
    assert error.retained_custodies[0] is error.custody
    if stop_unconfirmed:
        assert type(error).__name__ == 'StopUnconfirmed'
    # Deliberately no repair, retry, sleep, kill, renewed deadline or lock release.
    return {'original_handle_retained': True, 'deadline_unchanged': True,
            'reaped': process.returncode is not None, 'runtimeGrant': False}
