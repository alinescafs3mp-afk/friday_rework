"""NOT_RUN. Controls over genuine current evidence, after separate native grant.

No synthetic positive transcript, process launch, new deadline or admission.
The caller supplies the actual current join and original bound/owner. Missing
positive witnesses are NOT_RUN, never a vacuous pass. This file is only parsed
as source during FRW026-TS-REQUEST-IDENTITY-REPAIR.
"""
import copy
import json
from pathlib import Path


def genuine_witnesses(streams):
    rows=[r for stream in streams.values() for r in stream]
    finals=[r for r in rows if r['event']=='VITEST_FINAL_RESULT_SOURCE' and 'canonical' in r]
    transitions=[r for r in rows if r['event']=='VITEST_FINAL_SOURCE_TRANSITION']
    captures={(r['pid'],r['seq']):r for r in rows if r['event']=='VITE_RETURN_CAPTURE'}
    def module(r): return r['pid'],r['environment'],r['id'],r['canonical'],r['transformId']
    pairs=[(a,b) for a in finals for b in finals if module(a)==module(b)
           and a['requestId']!=b['requestId'] and a['seq']<b['seq']
           and a['returnedTransformSha256']==b['returnedTransformSha256']
           and a['finalSourceSha256']==b['finalSourceSha256']]
    if not pairs: raise ValueError('NOT_RUN_NO_GENUINE_SAME_BYTE_DISTINCT_RETURNS')
    overlap=[]
    for a,b in pairs:
        ta=next(t for t in transitions if t['pid']==a['pid'] and t['requestId']==a['requestId'])
        tb=next(t for t in transitions if t['pid']==b['pid'] and t['requestId']==b['requestId'])
        ca=captures[(a['producerPid'],a['returnSeq'])];cb=captures[(b['producerPid'],b['returnSeq'])]
        if (ca['seq']<ta['seq'] and cb['seq']<ta['seq'] and ta['inlineApplied']
                and ta['nodeBeforeSha256']!=ta['nodeAfterSha256']
                and tb['nodeBeforeSha256']==ta['nodeAfterSha256']
                and not tb['inlineApplied'] and b['returnedTransformSha256']!=tb['nodeBeforeSha256']):
            overlap.append((a,b))
    if not overlap: raise ValueError('NOT_RUN_NO_GENUINE_INLINE_MAP_OVERLAP')
    fetch=[r for r in rows if r['event']=='VITEST_WORKER_FETCH' and 'canonical' in r]
    shared=[a for a in fetch if any(a['pid']!=b['pid'] and a['canonical']==b['canonical'] for b in fetch)]
    if not shared: raise ValueError('NOT_RUN_NO_GENUINE_SHARED_MODULE_WORKERS')
    tmp=[r for r in rows if r['event']=='VITEST_TMP_DELIVERY' and r['cacheMetadata']['__frwEvidence']['requestId']!=r['requestId']]
    if not tmp: raise ValueError('NOT_RUN_NO_GENUINE_DISTINCT_REQUEST_TMP_REUSE')
    return {'same_byte_distinct_requests':[(a['pid'],a['requestId'],b['requestId']) for a,b in pairs],
            'actual_inline_overlap':[(a['pid'],a['requestId'],b['requestId']) for a,b in overlap],
            'distinct_request_tmp_reuse':[(r['pid'],r['seq']) for r in tmp],
            'shared_worker_fetches':[(r['pid'],r['seq']) for r in shared]}


def renumber_references(streams):
    """Maintain untouched observed references when deletion shifts process seq.

    Removed endpoints remain invalid and should be rejected by the semantic
    consumer. No upstream request/transform ID or observed byte hash is changed.
    """
    mapping={(r['pid'],r['seq']):n for stream in streams.values() for n,r in enumerate(stream,1)}
    for stream in streams.values():
        for n,r in enumerate(stream,1):
            pid=r['pid']
            for key in ('localRequestSeq','importerRequestSeq','cacheOriginRequestSeq','fetchSeq'):
                if r.get(key) is not None: r[key]=mapping.get((pid,r[key]),-1)
            if r.get('deliverySeq') is not None:
                r['deliverySeq']=mapping.get((r['deliveryPid'],r['deliverySeq']),-1)
            if r.get('returnSeq') is not None:
                r['returnSeq']=mapping.get((r['producerPid'],r['returnSeq']),-1)
            if r.get('cacheMetadata') is not None:
                e=r['cacheMetadata']['__frwEvidence'];e['returnSeq']=mapping.get((e['producerPid'],e['returnSeq']),-1)
            r['seq']=n


def prepared_controls(join, genuine_run, budget, owner, max_bytes, destination):
    baseline=Path(genuine_run)
    join(baseline,budget,owner,max_bytes)
    originals={p.name:p.read_bytes() for p in baseline.iterdir() if p.name in
               ('clock.json','pre-exec-os.json','progress.json') or p.name.startswith('guard-') and p.suffix=='.jsonl'}
    original_streams={n:[json.loads(line) for line in raw.splitlines()] for n,raw in originals.items() if n.startswith('guard-')}
    witnesses=genuine_witnesses(original_streams)
    def flipped(value): return ('0' if value[0]!='0' else '1')+value[1:]
    expectations={
        'returned-hash': {'returned_final_exact_transform_chain_broken'},
        'mutable-node-as-return': {'returned_final_exact_transform_chain_broken'},
        'strip-upstream-id': {'upstream_request_identity_missing_or_malformed'},
        'alter-upstream-id': {'payload_not_joined_to_exact_parent_request','worker_fetch_upstream_delivery_changed'},
        'same-byte-other-parent': {'worker_fetch_upstream_delivery_changed'},
        'temp-storage-id': {'payload_not_joined_to_exact_parent_request','temporary_storage_occurrence_missing'},
        'drop-local-completion': {'local_request_completion_multiplicity_changed'},
        'duplicate-local-completion': {'local_request_completion_multiplicity_changed'},
        'alter-importer-occurrence': {'local_request_completion_identity_changed','worker_importer_evaluation_missing'},
        'drop-shared-worker-import': {'evaluation_without_actual_worker_request','worker_static_import_chain_incomplete','worker_importer_evaluation_missing','cache_without_exact_original_local_request'},
        'typed-inner-omitted': {'vitest_inner_evaluation_missing'},
        'wrapper-changed': {'typed_evaluation_begin_done_multiplicity_changed','wrapper_hash_changed'},
        'plugin-output': {'actual_plugin_output_next_input_chain_broken'},
        'graph-edge': {'returned_graph_edges_changed_without_delta','graph_delta_previous_edges_changed','final_graph_edges_not_equal_to_observed_edges'},
    }
    outcomes={}
    for kind,expected in expectations.items():
        streams=copy.deepcopy(original_streams);rows=[r for stream in streams.values() for r in stream]
        fetch=[r for r in rows if r['event']=='VITEST_WORKER_FETCH' and 'canonical' in r]
        if kind in ('returned-hash','mutable-node-as-return'):
            r=next(r for r in rows if r['event']=='VITEST_FINAL_RESULT_SOURCE' and 'canonical' in r and
                   (kind!='mutable-node-as-return' or any(t['event']=='VITEST_FINAL_SOURCE_TRANSITION' and t['requestId']==r['requestId'] and t['nodeBeforeSha256']!=r['returnedTransformSha256'] for t in rows)))
            r['returnedTransformSha256']=flipped(r['returnedTransformSha256']) if kind=='returned-hash' else next(t['nodeBeforeSha256'] for t in rows if t['event']=='VITEST_FINAL_SOURCE_TRANSITION' and t['requestId']==r['requestId'])
        elif kind=='strip-upstream-id': del fetch[0]['requestId']
        elif kind=='alter-upstream-id': fetch[0]['requestId']+=10000000
        elif kind=='same-byte-other-parent':
            r=next(r for r in fetch if any(f['event']=='VITEST_FINAL_RESULT_SOURCE' and f.get('canonical')==r['canonical'] and f.get('environment')==r['environment'] and f.get('finalSourceSha256')==r['finalSourceSha256'] and f.get('requestId')!=r['requestId'] for f in rows))
            other=next(f for f in rows if f['event']=='VITEST_FINAL_RESULT_SOURCE' and f.get('canonical')==r['canonical'] and f.get('environment')==r['environment'] and f.get('finalSourceSha256')==r['finalSourceSha256'] and f.get('requestId')!=r['requestId'])
            for key in ('producerPid','returnSeq','transformId','requestId','returnedTransformSha256','finalSourceSha256'):r[key]=other[key]
        elif kind=='temp-storage-id':
            r=next(r for r in fetch if r.get('cacheMetadata'))
            r['cacheMetadata']['__frwEvidence']['requestId']+=10000000
        elif kind in ('drop-local-completion','duplicate-local-completion'):
            r=next(r for r in rows if r['event']=='VITE_WORKER_REQUEST_DONE' and 'canonical' in r)
            for stream in streams.values():
                if r in stream:
                    if kind=='drop-local-completion':stream.remove(r)
                    else:
                        duplicate=copy.deepcopy(r);duplicate['seq']=max(x['seq'] for x in stream)+1000000
                        stream.insert(stream.index(r)+1,duplicate)
                    break
            renumber_references(streams)
        elif kind=='alter-importer-occurrence':
            r=next(r for r in rows if r['event']=='VITE_WORKER_REQUEST' and r.get('importerRequestSeq') is not None)
            r['importerRequestSeq']+=10000000
        elif kind=='drop-shared-worker-import':
            f=next(r for r in fetch if any(x['canonical']==r['canonical'] and x['pid']!=r['pid'] for x in fetch))
            req=next(r for r in rows if r['event']=='VITE_WORKER_REQUEST' and r['pid']==f['pid'] and r.get('fetchSeq')==f['seq'] and r['cacheState']=='fresh' and r['importer'] is not None)
            for n,stream in streams.items():streams[n]=[r for r in stream if not (r is req or r['event']=='VITE_WORKER_REQUEST_DONE' and r['pid']==req['pid'] and r['localRequestSeq']==req['seq'])]
            renumber_references(streams)
        elif kind=='typed-inner-omitted':
            first=next(r for r in rows if r['event']=='VITEST_EVAL_BEGIN')
            for n,stream in streams.items():streams[n]=[r for r in stream if not (r['pid']==first['pid'] and r.get('localRequestSeq')==first['localRequestSeq'] and r['event'] in ('VITEST_EVAL_BEGIN','VITEST_EVAL_DONE'))]
            renumber_references(streams)
        elif kind=='wrapper-changed':
            r=next(r for r in rows if r['event']=='VITEST_EVAL_BEGIN');r['wrappedSha256']=flipped(r['wrappedSha256'])
        elif kind=='plugin-output':
            r=next(r for r in rows if r['event']=='VITE_PLUGIN_OUTPUT');r['codeSha256']=flipped(r['codeSha256'])
        elif kind=='graph-edge':next(r for r in rows if r['event']=='VITE_TRANSFORM_GRAPH' and r['imports'])['imports'].pop()
        target=Path(destination)/kind;target.mkdir(mode=0o700,parents=False,exist_ok=False)
        for name,raw in originals.items():
            if name in streams:raw=(''.join(json.dumps(r)+'\n' for r in streams[name])).encode()
            (target/name).write_bytes(raw)
        try:join(target,budget,owner,max_bytes)
        except ValueError as error:
            if str(error) not in expected:raise AssertionError('unrelated refusal '+kind+': '+str(error)) from error
            outcomes[kind]=str(error)
        else:raise AssertionError('semantic mutation accepted: '+kind)
    return {'actual_positive_witnesses':witnesses,'negative_outcomes':outcomes,'synthetic_native_execution':False}
