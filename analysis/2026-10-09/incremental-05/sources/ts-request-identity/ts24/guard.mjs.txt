function pinned(filename) {
  current()
  try {
    const canonical = fs.realpathSync(filename)
    const expected = pins[canonical]
    if (!expected || crypto.createHash('sha256').update(raw.read(canonical)).digest('hex') !== expected) refuse('UNPINNED_MODULE:' + canonical)
    emit({ event: 'NODE_MODULE_OR_PIN_CHECK', path: canonical, sha256: expected })
    return canonical
  } catch (error) { refuse('PIN_READ_ERROR:' + filename) }
}

// Vitest's native cache serializer is flatted, not JSON object encoding.
// Decode its bounded acyclic plain metadata graph; code is never evaluated.
function cacheMetadataDecode(text) {
  const table = JSON.parse(text)
  if (!Array.isArray(table) || !table.length || table.length > 100000) refuse('UNBOUND_FLATTED_CACHE_METADATA')
  const decoded = new Map(), active = new Set()
  function ref(index) {
    if (!Number.isInteger(index) || index < 0 || index >= table.length || active.has(index)) refuse('INVALID_FLATTED_CACHE_REFERENCE')
    if (decoded.has(index)) return decoded.get(index)
    const entry = table[index]
    if (entry === null || typeof entry !== 'object') return entry
    active.add(index)
    const value = Array.isArray(entry) ? [] : Object.create(null)
    for (const [key,item] of Object.entries(entry)) {
      if (typeof item === 'string') {
        if (!/^(0|[1-9][0-9]*)$/.test(item)) refuse('INVALID_FLATTED_CACHE_REFERENCE')
        value[key] = ref(Number(item))
      } else if (item !== null && typeof item === 'object') refuse('NESTED_UNBOUND_FLATTED_METADATA')
      else value[key] = item
    }
    active.delete(index); decoded.set(index,value)
    return value
  }
  return ref(0)
}
function tempSource(code, result, canonical) {
  const marker = '\n//# vitestCache='
  const at = code.lastIndexOf(marker)
  if (at === -1) return { finalSource: code, cacheMetadata: null }
  const suffix = code.slice(at + marker.length)
  if (!/^[A-Za-z0-9+/]+={0,2}$/.test(suffix)) refuse('UNBOUND_CACHE_SUFFIX')
  const cacheMetadata = cacheMetadataDecode(Buffer.from(suffix, 'base64').toString('utf8'))
  if (cacheMetadata.id !== result.id || pinned(cacheMetadata.file) !== canonical) refuse('CACHE_METADATA_IDENTITY_CHANGED')
  identity(cacheMetadata)
  return { finalSource: code.slice(0,at), cacheMetadata }
}

function request(event, meta, importer, requested, dynamic, cacheState, mod, importerRequestSeq = null) {
  const known = fetchedContexts.get(meta)
  const row = { event, id: meta.id, importer, requested, dynamic, cache: cacheState !== 'fresh', cacheState,
    environment: known?.environment ?? null, importerRequestSeq,
    cacheOriginRequestSeq: cacheState === 'fresh' ? null : execution(mod).localRequestSeq }
  if ('externalize' in meta) {
    const externalize = meta.externalize
    row.externalize = externalize
    if (isBuiltin(externalize) || externalize === '/@vite/client') row.builtin = true
    else {
      const resolved = meta.__frwExternalCanonical ?? externalize
      row.externalCanonical = pinned(resolved.startsWith('file:') ? fileURLToPath(resolved) : resolved)
      row.externalSha256 = pins[row.externalCanonical]
    }
  } else {
    const upstream = identity(meta)
    if (!known || !equalIdentity(upstream,known)) refuse('REQUEST_WITHOUT_LOCAL_FETCH_CONTEXT')
    Object.assign(row, known)
    row.canonical = pinned(meta.file)
    row.sourceSha256 = pins[row.canonical]
    row.codeSha256 = hash(meta.code)
  }
  return emit(row)
}
