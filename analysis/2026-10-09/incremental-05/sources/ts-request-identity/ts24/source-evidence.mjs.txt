/** NOT_RUN. Vite's real transform and environment graph seams.
 * Node hooks alone do not observe Vitest's vm.runInThisContext evaluation.
 * These are input/returned-transform observations, not proof of VM evaluation.
 */
import fs from 'node:fs'
import path from 'node:path'
import { fileURLToPath } from 'node:url'
import { guardEvidence as evidence } from './guard.mjs'

export function sourceEvidence() {
  let server
  const checked = new Map()
  function file(id) {
    if (typeof id !== 'string' || id.includes('\0') || id.startsWith('virtual:')) evidence.refuse('UNBOUND_VIRTUAL_TRANSFORM')
    let raw = id.startsWith('file:') ? fileURLToPath(id) : id
    if (raw.startsWith('/@fs/')) raw = raw.slice(4)
    const query = raw.indexOf('?')
    // No raw/url/worker transformations silently substitute a different corpus.
    if (query !== -1) {
      if (!/^\?v=[a-f0-9]+$/.test(raw.slice(query))) evidence.refuse('UNBOUND_TRANSFORM_QUERY:' + id)
      raw = raw.slice(0, query)
    }
    if (!path.isAbsolute(raw)) evidence.refuse('NON_FILE_TRANSFORM:' + id)
    const canonical = evidence.pinned(raw)
    return { canonical, sourceSha256: evidence.rawHash(fs.readFileSync(canonical)) }
  }
  let requestSequence = 0
  const pre = {
    name: 'frw-ts24-pinned-input', enforce: 'pre',
    load(id) { file(id); return null },
    configureServer(value) {
      server = value
      for (const [name, environment] of Object.entries(server.environments)) {
        const graph = environment.moduleGraph
        const original = environment.transformRequest.bind(environment)
        environment.__frwRequestIds = new WeakMap()
        environment.transformRequest = async function (url, ...args) {
          const requestId = ++requestSequence
          const result = await original(url, ...args)
          if (result) {
            // Capture the actual returned string before another await can expose
            // a shared node/result to an overlapping inline-map mutation.
            const observed = { ...result, __frwTransformId: result.__frwTransformId }
            observed.__frwEvidence = Object.freeze({ schema: 2, producerPid: process.pid,
              environment: name, transformId: result.__frwTransformId, requestId,
              returnedTransformSha256: evidence.rawHash(observed.code) })
            const capture = evidence.emit({ event: 'VITE_RETURN_CAPTURE', ...observed.__frwEvidence,
              url, codeSha256: observed.__frwEvidence.returnedTransformSha256 })
            observed.__frwEvidence = Object.freeze({ ...observed.__frwEvidence, returnSeq: capture.seq })
            const node = await graph.getModuleByUrl(url)
            if (!node || !node.id || !node.file) evidence.refuse('UNBOUND_RETURNED_TRANSFORM:' + url)
            // A distinct wrapper object keeps concurrent/cache returns distinct;
            // map/code bytes and native transform behavior are unchanged.
            environment.__frwRequestIds.set(observed, requestId)
            evidence.transform('VITE_RETURNED_TRANSFORM', observed.code, node, name,
              observed.__frwTransformId, { ...observed.__frwEvidence, url })
            evidence.emit({ event: 'VITE_TRANSFORM_GRAPH', environment: name, url, id: node.id,
              transformId: observed.__frwTransformId, requestId, returnSeq: capture.seq, producerPid: process.pid, file: node.file, imports: evidence.graphEdges(node) })
            checked.set(requestId, observed.__frwTransformId)
            return observed
          }
          return result
        }
      }
    },
  }
  function snapshot() {
    if (!server) evidence.refuse('VITE_SERVER_NOT_OBSERVED')
    const graph = []
    for (const [name, environment] of Object.entries(server.environments)) {
      for (const node of environment.moduleGraph.idToModuleMap.values()) {
        if (!node.file) {
          if (typeof node.id !== 'string' || !(evidence.isBuiltin(node.id) || node.id === '/@vite/client') || node.transformResult) evidence.refuse('UNBOUND_GRAPH_FILE:' + node.id)
          graph.push({ environment: name, id: node.id, url: node.url, canonical: null,
            sourceSha256: null, transformedSha256: null,
            imports: [...node.importedModules].map(m => ({ id: m.id, file: m.file, url: m.url })) })
          continue
        }
        const input = file(node.file)
        graph.push({ environment: name, id: node.id, url: node.url, ...input,
          transformedSha256: node.transformResult ? evidence.rawHash(node.transformResult.code) : null,
          imports: [...node.importedModules].map(m => ({ id: m.id, file: m.file, url: m.url })) })
      }
    }
    evidence.emit({ event: 'VITE_FINAL_GRAPH', graph, observedTransformRows: new Set(checked.values()).size, observedRequestRows: checked.size,
      meaning: 'Pinned inputs and returned transform graph; finite coordinator must reconcile actual worker/evaluator streams, and external root must prove terminal scope closure.' })
  }
  return { plugins: [pre], snapshot }
}
