"""SHA/AST checks only. Never import, compile, test or execute candidate bytes."""
from pathlib import Path
import ast
import hashlib
import json
W=Path(__file__).resolve().parent
S=Path('<SOL_WORK>/FRW041-CALLBACK-FINITE-OWNER-evidence')
C=W/'candidate';Q=C/'qualification'
def sha(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def ref(p):return {'path':str(p),'sha256':sha(p),'bytes':p.stat().st_size}
result={'state':'SOURCE_SHA_AST_CHECKS_ONLY_NO_CANDIDATE_EXECUTION','failures':[],
        'internal_sha_checks':0,'external_hashes_currently_read':0,'candidate_imports_tests_or_effects':0}
asts={}
for p in sorted(C.rglob('*.py')):
 asts[p]=ast.parse(p.read_text(),filename=str(p))
for p in sorted((W/'prepared').rglob('*.py')):asts[p]=ast.parse(p.read_text(),filename=str(p))
result['AST_files']=[ref(p) for p in asts]
result['preserved_AST']=[]
old=json.loads((S/'static-verification-final.json').read_text())
for row in old['full_chronology_and_corpus_AST_preservation']:
 rel=Path(row['path']).relative_to(S/'candidate');p=C/rel
 old_source=Path(row['path']).read_text()
 new_source=p.read_text().replace(str(C),str(S/'candidate'))
 equal=ast.dump(ast.parse(old_source),include_attributes=False)==ast.dump(ast.parse(new_source),include_attributes=False)
 assert equal,rel
 result['preserved_AST'].append({'path':str(p),'prior':row['path'],'same_full_AST_after_only_exact_path_rebinding':equal})
external_counts={}
for name in ['input-pins.json','probe-input-pins.json','projection-pins.json']:
 original=json.loads((S/'candidate/qualification'/name).read_text());current=json.loads((Q/name).read_text())
 old_external={k:v for k,v in original.items() if not k.startswith(str(S/'candidate')+'/')}
 new_external={k:v for k,v in current.items() if not k.startswith(str(C)+'/')}
 assert old_external==new_external,name
 external_counts[name]=len(new_external)
result['external_map_records_unchanged_provenance_only']=external_counts
refs=[]
def scan(x):
 if isinstance(x,dict):
  for k,v in x.items():
   if isinstance(k,str) and k.startswith(str(C)+'/') and isinstance(v,str) and len(v)==64:
    assert sha(k)==v,k;refs.append((k,v))
   elif k in ('path','file') and isinstance(v,str) and v.startswith(str(W)+'/') and 'sha256' in x:
    assert sha(v)==x['sha256'],v;refs.append((v,x['sha256']))
    if 'bytes' in x:assert Path(v).stat().st_size==x['bytes'],v
   else:scan(v)
 elif isinstance(x,list):
  for v in x:scan(v)
for p in C.rglob('*.json'):scan(json.loads(p.read_text()))
result['internal_sha_checks']=len(refs)
for name in ['run_bounded_r2.py','finite_custody.py','existing_caller_join.py']:
 tree=asts[Q/name]
 calls=[n for n in ast.walk(tree) if isinstance(n,ast.Call) and isinstance(n.func,ast.Attribute)]
 assert not any(n.func.attr in ('killpg','waitpid','poll','wait') for n in calls),name
 if name=='finite_custody.py':
  assert all(isinstance(n.args[0],ast.Name) and n.args[0].id=='pid' for n in calls if n.func.attr=='wait4')
  result['custody_wait4_calls']=[{'line':n.lineno,'target':ast.unparse(n.args[0])} for n in calls if n.func.attr=='wait4']
owner=asts[Q/'run_bounded_r2.py']
lowers=[n.lineno for n in ast.walk(owner) if isinstance(n,ast.Call) and isinstance(n.func,ast.Attribute) and n.func.attr=='setrlimit' and 'fork_hard' in ast.unparse(n)]
forks=[n.lineno for n in ast.walk(owner) if isinstance(n,ast.Call) and isinstance(n.func,ast.Attribute) and n.func.attr=='Popen']
assert len(lowers)==len(forks)==1 and lowers[0]<forks[0]
result['parent_hard_lowering_before_only_Popen']={'lower_line':lowers[0],'Popen_line':forks[0]}
assert 'time.monotonic();' not in (Q/'existing_caller_join.py').read_text()
sequence=json.loads((Q/'finite-sequence.json').read_text())
assert len(sequence['modes_once_in_order'])==11 and sequence['existing_source_cases']==277 and sequence['existing_source_reports']==831
assert sequence['existing_project_cases']==38 and sequence['existing_project_reports']==114
assert sequence['expected_exits']['deny']==125
assert sequence['budget_seconds']==180 and sequence['cpu_seconds']==45 and sequence['memory_bytes']==2<<30 and sequence['affinity']==[0,1]
assert sequence['gnu']['maximum_step_seconds']==sequence['gnu']['shared_clock_closure_reserve_seconds']==60
controls=json.loads((W/'prepared-controls.json').read_text())
assert len(controls['controls'])==17 and all(r['execution']=='NOT_RUN' for r in controls['controls'])
result['prepared_controls_NOT_RUN']=len(controls['controls'])
result['prepared_model_methods_NOT_RUN']=[n.name for n in ast.walk(asts[W/'prepared/test_custody_source_controls.py']) if isinstance(n,ast.FunctionDef) and n.name.startswith('test_')]
result['source_manifest']=ref(Q/'source-manifest.json')
result['no_runtime_inputs_or_generated_caches']=not any((Q/name).exists() for name in ['clock.json','admission.json','owner-binding.json','run','__pycache__'])
assert result['no_runtime_inputs_or_generated_caches']
(W/'source-checks.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps({'AST':len(asts),'preserved_AST':len(result['preserved_AST']),'internal_SHA':len(refs),'prepared_controls_NOT_RUN':17,'candidate_execution':False,'failures':[]}))
