"""Pure declared credit arithmetic, never proof of kernel delivery margins."""
import math

def allocate_cpu(previous):
    if not math.isfinite(previous) or previous < 0:
        raise ValueError('UNKNOWN_OR_INVALID_PREVIOUS_KERNEL_CPU')
    cap = math.floor((45-6-previous)/2)-1
    while cap > 0 and previous+6+max(6,cap)+cap+2 > 45:
        cap -= 1
    if cap <= 0:
        raise RuntimeError('NO_CUMULATIVE_CPU_CREDIT_WITH_FORK_PREFIX')
    fork_hard = max(6,cap)
    return cap, fork_hard, fork_hard+cap+2
