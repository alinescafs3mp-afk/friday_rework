"""Same-PID existing exec launcher -> pinned GNU -> pinned mode interpreter.
Root installed reserved CPU/AS/affinity BEFORE exec this launcher. No extra
launcher process credit or fresh limit/clock. GNU fork inherits its hard cap.
"""
from pathlib import Path
import os,sys,json,resource,time
HERE=Path(__file__).resolve().parent
sys.path.insert(0,str(HERE))
from owner_step_inputs import MODES,proc_stat,digest
assert len(sys.argv)==2 and sys.argv[1] in MODES
mode=sys.argv[1];RUN=HERE/'run'
clock=json.loads((HERE/'clock.json').read_bytes())
binding=json.loads((HERE/'owner-binding.json').read_bytes())
assert binding['clock_sha256']==digest(HERE/'clock.json')
clock.update(owner_pid=binding['owner_pid'],owner_start_ticks=binding['owner_start_ticks'])
admission=json.loads((HERE/'admission.json').read_bytes())
step=json.loads((RUN/('root-step-'+mode+'.json')).read_bytes())
assert step['mode']==mode and step['ordinal']==MODES.index(mode)
assert step['clock_sha256']==digest(HERE/'clock.json')
assert step['admission_sha256']==digest(HERE/'admission.json')
parent=proc_stat(os.getppid())
assert parent['pid']==clock['owner_pid'] and parent['start_ticks']==clock['owner_start_ticks']
assert sorted(os.sched_getaffinity(0))==[0,1]
cap=step['process_cpu_allocation']
assert resource.getrlimit(resource.RLIMIT_CPU)==(cap,cap)
assert resource.getrlimit(resource.RLIMIT_AS)==(2<<30,2<<30)
assert os.getpgrp()==os.getpid()==os.getsid(0)
from finite_kernel import parent_death
parent_death(parent['pid'])  # Retained by ordinary exec GNU, NOT its fork child.
assert time.monotonic()+120<clock['deadline_monotonic']
assert digest('/usr/bin/gnutimeout')==admission['gnu_sha256']
assert digest(admission['interpreter'])==admission['interpreter_sha256']
argv=[str(HERE/'run.py')] if mode=='source-cases' else [str(HERE/'callback_probe.py'),mode,str(RUN/('callback-'+mode+'.json'))]
gnu=['gnutimeout','--signal=KILL','--verbose','60s',admission['interpreter'],'-I','-B','-S',*argv]
identity=proc_stat(os.getpid())
step.update(gnu_identity={k:identity[k] for k in ('pid','ppid','pgid','sid','start_ticks')},
 expected_pgid=identity['pgid'],gnu_argv=gnu,gnu_seconds=60)
with (RUN/('owner-step-'+mode+'.json')).open('x') as f:json.dump(step,f,indent=2)
os.environ.clear();os.environ.update(PATH='/usr/bin:/bin',LANG='C.UTF-8',LC_ALL='C.UTF-8',
 PYTEST_DISABLE_PLUGIN_AUTOLOAD='1',HOME=str(RUN/'home'),TMPDIR=str(RUN/'tmp'))
os.chdir(RUN)
assert time.monotonic()+120<clock['deadline_monotonic']
os.execv('/usr/bin/gnutimeout',gnu)
