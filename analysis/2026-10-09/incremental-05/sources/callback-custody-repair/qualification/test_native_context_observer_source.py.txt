EVENTS=[('fsync','call',False),('fsync','return',True),('fsync','c_call',True),('fsync','c_return',True),('fsync','c_exception',True),('fsync','exception',False),('open','call',False),('open','return',True),('open','c_return',False),('other_guard','return',False),('copystat','return',True),('copystat','call',False),('copystat','c_return',False),('copy2','return',True),('copy2','call',False),('copy2','c_exception',False),('foreign_shutil','return',False),('wrong_file','return',False),('wrong_globals','c_return',False),('foreign','return',False)]

def observer_env(kind,fail=None):
    def fsync():pass
    def opening():pass
    fence=NS(directory_sync_fsync=fsync,directory_sync_open=opening,STDLIB='/pinned/stdlib',RAW={'fsync':object()})
    module=ModuleType('shutil')
    calls=[];state={};arg=object()
    code=fsync.__code__ if kind=='fsync' else opening.__code__ if kind=='open' else NS(co_filename='/pinned/stdlib/shutil.py',co_name=kind if kind in ('copy2','copystat') else 'foreign')
    globals_=fence.__dict__ if kind in ('fsync','open','other_guard') else module.__dict__ if kind in ('copystat','copy2','foreign_shutil','wrong_file') else {}
    if kind=='fsync':arg=fence.RAW['fsync']
    if kind=='wrong_file':code=NS(co_filename='/foreign/shutil.py',co_name='copy2')
    if kind=='wrong_globals':code=fsync.__code__
    frame=NS(f_globals=globals_,f_code=code,f_lineno=123)
    class Refused(Exception):pass
    def deny(event):calls.append(('deny',event));raise Refused(event)
    fence.deny=deny
    def copy_callback(*args):
        calls.append(('copy',args))
        if fail=='copy':raise ValueError('original copy observation failure')
    def directory_callback(*args):
        calls.append(('directory',args))
        if fail=='directory':raise ValueError('original directory observation failure')
    env=dict(fence=fence,state=state,copy_metadata_profile=copy_callback,directory_profile_event=directory_callback,sys=NS(modules={'shutil':module}))
    return bind(definition('directory_profile'),env),frame,arg,calls,state,Refused

@pytest.mark.parametrize('kind,event,allowed',EVENTS)
def test_directory_prefilter_retains_exact_native_and_copy_events(kind,event,allowed):
    fn,frame,arg,calls,state,_=observer_env(kind)
    fn(frame,event,arg)
    assert calls==([('copy',(frame,event,arg)),('directory',(frame,event,arg))] if allowed else [])
    assert state=={}

@pytest.mark.parametrize('failure',['copy','directory'])
def test_directory_observer_failure_cannot_become_success(failure):
    fn,frame,arg,calls,state,Refused=observer_env('fsync',failure)
    state['metadata_pending']=dict(src=Path('/private/source'),dst=Path('/private/destination'),node='exact-node',source_inode=[1,2],metadata={'retained':True})
    with pytest.raises(Refused,match='directory_observer:observation_failed'):fn(frame,'c_exception',arg)
    assert 'metadata_pending' not in state
    assert state['metadata_failed_pending']==dict(source='/private/source',destination='/private/destination',node='exact-node',source_inode=[1,2],metadata={'retained':True})
    assert state['directory_observer_failure']==dict(type='ValueError',message='original '+failure+' observation failure',event='c_exception',function=frame.f_code.co_name,line=123)
    assert calls[-1]==('deny','directory_observer:observation_failed')

@pytest.mark.parametrize('deleted',[False,True])
def test_snapshot_retains_all_paths_unique_hashes_and_metadata_gate(deleted):
    calls=[];saves={};paths=['/mapped/a','/mapped/a','/mapped/b'+(' (deleted)' if deleted else '')]
    mapping='\n'.join('00-01 r-xp 0 00:00 0 '+p for p in paths)
    def path(raw):assert raw=='/proc/self/maps';return NS(read_text=lambda:mapping)
    def save(name,value):calls.append(('save',name));saves[name]=value
    def sha(raw):calls.append(('hash',raw));return 'hash:'+raw
    state={'all_reports':['preserved']}
    env=dict(fence=NS(verify_distribution_metadata=lambda:calls.append(('metadata',))),save=save,state=state,Path=path,origins=lambda:{'all_modules':'preserved'},sha=sha)
    fn=bind(definition('snapshot'),env)
    if deleted:
        with pytest.raises(AssertionError,match='DELETED_MAPPING'):fn()
    else:fn()
    assert calls[0]==('metadata',)
    assert saves['pytest-progress.json'] is state and saves['loaded-mapping-paths.json']==paths
    assert saves['module-origins.json']=={'all_modules':'preserved'}
    assert calls.index(('save','loaded-mapping-paths.json'))<calls.index(('hash','/mapped/a'))
    assert calls.count(('hash','/mapped/a'))==1
    if deleted:assert 'loaded-mappings.json' not in saves
    else:assert saves['loaded-mappings.json']=={'/mapped/a':'hash:/mapped/a','/mapped/b':'hash:/mapped/b'}

@pytest.mark.parametrize('binding',['missing','nonmodule'])
def test_missing_or_nonmodule_shutil_cannot_reach_handlers(binding):
    # Concrete missing coverage: absence/non-module value has no module globals.
    # Any f_code audit or handler invocation would fail this early no-op oracle.
    calls=[]
    class Frame:
        f_globals={}
        @property
        def f_code(self):raise AssertionError('no-op must reject before f_code audit')
    modules={} if binding=='missing' else {'shutil':object()}
    env=dict(fence=NS(),sys=NS(modules=modules),state={},copy_metadata_profile=lambda *a:calls.append(a),directory_profile_event=lambda *a:calls.append(a))
    bind(definition('directory_profile'),env)(Frame(),'return',None)
    assert calls==[] and env['state']=={}
