class Usage(tuple):
    @property
    def ru_utime(self): return self[0]
    @property
    def ru_stime(self): return self[1]

class KernelModel:
    def __init__(self):
        self.original={'pid':100,'ppid':9,'pgid':100,'sid':100,'start_ticks':500,'cpu_ticks':0}
        self.live={100:dict(self.original)}
        self.waitable={100}
        self.waited=[]
        self.signals=[]
        self.foreign=[]
        self.scans=0
        self.fail_after_first_reap=False
        self.proc=SimpleNamespace(pid=100,returncode=None)
        self.custody=Custody(self.proc,9,self.stat,self.direct,self.members)
    def stat(self,pid): return dict(self.live[pid])
    def direct(self): return list(self.live)
    def members(self,pgid):
        self.scans+=1
        if self.fail_after_first_reap and self.waited:
            self.fail_after_first_reap=False
            return [],[{'error':'one injected scan failure AFTER exact GNU reap'}]
        return [dict(v) for v in self.live.values() if v['pgid']==pgid]+self.foreign,[]
    def wait4(self,pid,flags):
        if pid in self.waited: raise AssertionError('double wait/recharge')
        if pid not in self.waitable: return 0,0,None
        self.waited.append(pid);self.live.pop(pid);self.waitable.remove(pid)
        return pid,0,Usage((1.0,0.25)+(0,)*14)
    def kill(self,pid,sig):
        assert pid in self.live and pid not in self.waited
        self.signals.append(pid);self.waitable.add(pid)
    def patches(self):
        return patch.multiple('finite_custody.os',wait4=self.wait4,kill=self.kill)

class PreparedControls(unittest.TestCase):
    def test_R13_reentrant_after_reap_retains_original_row_and_failed_error(self):
        k=KernelModel();k.fail_after_first_reap=True;errors=[]
        with k.patches(),patch('finite_custody.time.monotonic',return_value=1),patch('finite_custody.time.sleep'):
            try:k.custody.finish(10)
            except StopUnconfirmed as error:errors.append(str(error))
            first=k.custody.gnu_reap
            self.assertIsNotNone(first)
            k.custody.finish(10,terminate=True)
            k.foreign=[dict(k.original,start_ticks=900)]
            scans=k.scans
            k.custody.finish(10,terminate=True)
            self.assertEqual(k.scans,scans)  # Closed state is not rescanned.
            self.assertIs(first,k.custody.gnu_reap)
            self.assertEqual(first['identity'],k.original)
            self.assertEqual(first['rusage'],[1.0,0.25]+[0]*14)
            self.assertEqual(k.waited,[100]);self.assertFalse(k.signals)
            self.assertTrue(errors)  # Original failure must still reject content.
            self.assertTrue(k.custody.closed)

    def test_R14_matching_group_after_release_never_signaled(self):
        k=KernelModel()
        # Foreign numerical match arrives after original GNU's exact reap.
        k.foreign=[dict(k.original,pid=101,start_ticks=901,ppid=44)]
        with k.patches(),patch('finite_custody.time.monotonic',return_value=1):
            with self.assertRaisesRegex(StopUnconfirmed,'UNPROVED_GROUP'):
                k.custody.finish(10,terminate=True)
        self.assertEqual(k.waited,[100]);self.assertFalse(k.signals)
        self.assertNotIn(101,k.custody.owned)
        self.assertIn((101,901),k.custody.discovered)
        self.assertFalse(k.custody.closed)

    def test_R15_fork_prefix_reserved_before_popen(self):
        for previous in [0,1,12,25,29,30]:
            cap,hard,credit=allocate_cpu(previous)
            self.assertEqual(hard,max(6,cap))
            self.assertEqual(credit,hard+cap+2)
            self.assertLessEqual(previous+6+credit,45)
        self.assertEqual(allocate_cpu(30),(1,6,9))
        for previous in [30.00001,35,45]:
            with self.assertRaises(RuntimeError):allocate_cpu(previous)
        tree=ast.parse((Q/'run_bounded_r2.py').read_text())
        lowers=[n.lineno for n in ast.walk(tree) if isinstance(n,ast.Call)
                and isinstance(n.func,ast.Attribute) and n.func.attr=='setrlimit'
                and 'fork_hard' in ast.unparse(n)]
        forks=[n.lineno for n in ast.walk(tree) if isinstance(n,ast.Call)
               and isinstance(n.func,ast.Attribute) and n.func.attr=='Popen']
        self.assertEqual(len(lowers),1);self.assertEqual(len(forks),1)
        self.assertLess(lowers[0],forks[0])
        self.assertNotIn('killpg', (Q/'finite_custody.py').read_text())

    def test_R16_adopted_cardinality_after_discovery_keeps_all_usage(self):
        k=KernelModel()
        for pid in (101,102):
            k.live[pid]=dict(k.original,pid=pid,start_ticks=500+pid,
                             pgid=700 if pid==102 else 100)
        with k.patches(),patch('finite_custody.time.monotonic',return_value=1),patch('finite_custody.time.sleep'):
            k.custody.finish(10)
        self.assertEqual(k.waited,[100,101,102])
        self.assertEqual(k.signals,[101,102])
        self.assertEqual(sum(row['user']+row['system'] for row in k.custody.rows),3.75)
        self.assertIn('UNEXPECTED_DESCENDANT_CARDINALITY',k.custody.anomalies)
        self.assertIn('CHILD_ESCAPED_EXPECTED_GROUP:102',k.custody.anomalies)
        self.assertTrue(k.custody.closed)  # Physical closure, original sequence FAIL.

    def test_R16_fast_exit_missing_stat_keeps_unknown_reap_and_stops(self):

    def test_R16_ESRCH_does_not_skip_final_wait4(self):
        k=KernelModel();k.live[101]=dict(k.original,pid=101,start_ticks=601)
        def vanished(pid,sig):
            self.assertEqual(pid,101)
            k.waitable.add(pid)  # Exit raced with signal, still unreaped child.
            raise ProcessLookupError('ESRCH')
        with patch('finite_custody.os.wait4',k.wait4),patch('finite_custody.os.kill',vanished),patch('finite_custody.time.monotonic',return_value=1),patch('finite_custody.time.sleep'):
            k.custody.finish(10)
        self.assertEqual(k.waited,[100,101])
        self.assertEqual(len(k.custody.rows),2)
        self.assertEqual(k.custody.events[0]['signal_or_stat'],'ESRCH_REAP_STILL_REQUIRED')
        self.assertTrue(k.custody.closed)

    def test_R16_later_proc_error_does_not_erase_already_proved_birth(self):
        k=KernelModel();k.custody.discover()
        def missing(pid):raise FileNotFoundError('late proc read failure')
        k.custody.proc_stat=missing

            with self.assertRaises(StopUnconfirmed):k.custody.finish(10)
            self.assertEqual(k.custody.gnu_reap['identity'],k.original)

            k.custody.proc_stat=k.stat

        self.assertTrue(k.custody.closed);self.assertEqual(k.waited,[100])

    def test_R16_group_scan_exception_still_discovers_and_reaps_direct(self):
