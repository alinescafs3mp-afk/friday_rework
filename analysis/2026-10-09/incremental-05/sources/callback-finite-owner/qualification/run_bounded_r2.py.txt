def exact_lock():
    fd=os.open(LOCK,os.O_RDONLY|os.O_NOFOLLOW)
    try:
        s=os.fstat(fd)
        identity=(s.st_dev,s.st_ino,s.st_uid,s.st_gid,s.st_nlink,s.st_size,stat.S_IMODE(s.st_mode))
        if identity!=LOCK_ID:raise RuntimeError('LOCK_IDENTITY_CHANGED')
        fcntl.flock(fd,fcntl.LOCK_EX|fcntl.LOCK_NB)
        fcntl.flock(fd,fcntl.LOCK_UN)
        return list(identity)
    finally:
        os.close(fd)

def proc_children():
    return [int(v) for v in Path('/proc/self/task/'+str(os.getpid())+'/children').read_text().split()]

def group_members(group):
    members=[];errors=[]
    for p in Path('/proc').iterdir():
        if not p.name.isdecimal():continue
        try:
            row=proc_stat(int(p.name))
            if row['pgid']==group:members.append(row)
        except (FileNotFoundError,ProcessLookupError):pass
        except (OSError,ValueError,IndexError) as error:
            errors.append({'pid':p.name,'error':repr(error)})
    return members,errors

def exclusive(path,value):

def finish_group(proc,birth,deadline,observed,rows):
    """Bounded exact-owned cleanup; no waitpid/Popen.wait can discard rusage."""
    group=proc.pid;gnu_reaped=False;unexpected=False
    while time.monotonic()<deadline:
        members,errors=group_members(group)
        if errors:raise StopUnconfirmed('PROC_SCAN_ERRORS:'+repr(errors))
        for r in members:observed[(r['pid'],r['start_ticks'])]=r
        if len(observed)>2:unexpected=True  # Only pinned GNU single fork, no candidate forks.
        # Existing finite subreaper owns the only child subtree. Unknown new
        # direct children are retained and terminate sequencing, never ignored.
        direct=proc_children()
        for pid in direct:
            try:
                r=proc_stat(pid)
                observed[(pid,r['start_ticks'])]=r
                if r['pgid']!=group:unexpected=True
            except (FileNotFoundError,ProcessLookupError):pass
        while True:
            try:pid,status,usage=os.wait4(-1,os.WNOHANG)
            except ChildProcessError:break
            if pid==0:break
            known=[r for (p,b),r in observed.items() if p==pid]
            if len(known)!=1:
                unexpected=True
            rows.append({'pid':pid,'identity':known[0] if len(known)==1 else None,
                         'wait_status':status,'exit_code':os.waitstatus_to_exitcode(status),
                         'user':usage.ru_utime,'system':usage.ru_stime,
                         'kernel_usage':True,'includes_only_descendants_actually_waited_by_this_child':True})
            if pid==proc.pid:
                proc.returncode=os.waitstatus_to_exitcode(status);gnu_reaped=True
        if gnu_reaped:
            # GNU9.7 self/group KILL can leave adopted mode. The GNU wait4 row
            # omits a mode it did not wait; this owner separately waits it.
            if members:
                # PGID cannot be recycled while retained exact members remain.
                now,scan_errors=group_members(group)
                if scan_errors:raise StopUnconfirmed('PRE_KILL_SCAN_ERRORS')
                if any((r['pid'],r['start_ticks']) not in observed for r in now):
                    raise StopUnconfirmed('UNBOUND_GROUP_MEMBER')
                if now:
                    os.killpg(group,signal.SIGKILL)
            remaining,scan_errors=group_members(group)
            if not remaining and not scan_errors and not proc_children():
                if unexpected:raise StopUnconfirmed('UNEXPECTED_CHILD_OR_UNBOUND_REAP')
                return
        # Finite process wait under original absolute clock, NOT mailbox/model polling.
        time.sleep(min(0.01,max(0,deadline-time.monotonic())))
    raise StopUnconfirmed('ORIGINAL_ABSOLUTE_DEADLINE_EXHAUSTED')

def owned_kill(proc,birth,observed):
    members,errors=group_members(proc.pid)
    for r in members:
        if r['ppid']==os.getpid() or r['pgid']==proc.pid:
            observed[(r['pid'],r['start_ticks'])]=r
    if errors:raise StopUnconfirmed('CLEANUP_SCAN_ERRORS')
    if any((r['pid'],r['start_ticks']) not in observed for r in members):
        raise StopUnconfirmed('CLEANUP_UNBOUND_MEMBER')
    if members:os.killpg(proc.pid,signal.SIGKILL)
