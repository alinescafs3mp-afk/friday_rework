"""Actual finite-owner joins; no cgroup or self-authored reviewed boolean.
No process launch/clock renewal/limit raise. Mode inherits BEFORE exec GNU's
reserved hard CPU allowance, establishes its own non-execing absolute timer
and parent-death binding before source imports/candidate effects.
Pre-mode timer/PDEATHSIG gap in the installed GNU fork/exec seam remains an
explicit SOURCE BLOCKER. This adapter does not pretend those survive fork.
"""
from pathlib import Path
import json, hashlib, os, sys, time, resource, math
HERE=Path(__file__).resolve().parent
MODES=('source-cases','positive','profile-entry','profile-exit','audit-entry','audit-exit',
 'foreign','baseexceptions','deny','overhead-original','overhead-timed')
def digest(p):
 with Path(p).open('rb') as f:return hashlib.file_digest(f,'sha256').hexdigest()
def proc_stat(pid):
 raw=Path('/proc/'+str(pid)+'/stat').read_text();tail=raw[raw.rfind(')')+2:].split()
 return {'pid':pid,'ppid':int(tail[1]),'pgid':int(tail[2]),'sid':int(tail[3]),
  'start_ticks':int(tail[19]),'cpu_ticks':int(tail[11])+int(tail[12])}
def check(mode,argv):
 assert mode in MODES and sys.argv==argv
 clock=json.loads((HERE/'clock.json').read_bytes())
 step=json.loads((HERE/'run'/('owner-step-'+mode+'.json')).read_bytes())
 admission=json.loads((HERE/'admission.json').read_bytes())
 assert clock['assignment']=='FRW041-CALLBACK-COST-CHANGED-SOURCE-QUALIFICATION' and clock['generation']==1
 assert clock['budget_seconds']==180 and clock['cpu_seconds']==45
 assert clock['deadline_monotonic']-clock['started_monotonic']==180
 assert clock['boot_id']==Path('/proc/sys/kernel/random/boot_id').read_text().strip()
 assert admission['assignment']==clock['assignment'] and admission['generation']==1
 assert admission['execution_authorized'] is True and admission['production_browser_grant'] is False
 assert admission['source_manifest_sha256']==digest(HERE/'source-manifest.json')
 assert step['clock_sha256']==digest(HERE/'clock.json') and step['admission_sha256']==digest(HERE/'admission.json')
 assert step['mode']==mode and step['ordinal']==MODES.index(mode)
 owner=proc_stat(clock['owner_pid']);assert owner['start_ticks']==clock['owner_start_ticks']
 parent=proc_stat(os.getppid())
 assert {k:parent[k] for k in ('pid','ppid','pgid','sid','start_ticks')}==step['gnu_identity']
 assert parent['ppid']==owner['pid'] and os.getpgrp()==parent['pgid']==step['expected_pgid']
 assert digest('/proc/'+str(parent['pid'])+'/exe')==admission['gnu_sha256']
 assert os.readlink('/proc/'+str(parent['pid'])+'/exe')=='/usr/bin/gnutimeout'
 expected=['gnutimeout','--signal=KILL','--verbose','60s',admission['interpreter'],'-I','-B','-S',*argv]
 actual=[s.decode() for s in Path('/proc/'+str(parent['pid'])+'/cmdline').read_bytes().split(b'\0') if s]
 assert actual==expected==step['gnu_argv']
 assert digest('/proc/self/exe')==admission['interpreter_sha256']
 assert sys.flags.isolated and sys.flags.no_site and sys.dont_write_bytecode
 assert sorted(os.sched_getaffinity(0))==[0,1]
 cap=step['process_cpu_allocation'];assert type(cap) is int and cap>0
 assert resource.getrlimit(resource.RLIMIT_CPU)==(cap,cap)
 assert resource.getrlimit(resource.RLIMIT_AS)==(2<<30,2<<30)
 assert step['reserved_root_cpu']==6 and step['reserved_live_child_cpu']==2*(cap+1)
 assert step['previous_kernel_child_cpu']+6+2*(cap+1)<=45
 assert step['completed_modes']==list(MODES[:step['ordinal']])
 for i,r in enumerate(step['previous_receipts']):
  assert r['mode']==MODES[i] and r['accounting_complete'] and r['process_unit_closed'] and r['lock_returned']
 assert len(step['previous_receipts'])==step['ordinal']
 assert set(os.environ)<=set(('PATH','HOME','TMPDIR','LANG','LC_ALL','PYTEST_DISABLE_PLUGIN_AUTOLOAD'))
 assert os.environ.get('PATH')=='/usr/bin:/bin' and os.environ.get('PYTEST_DISABLE_PLUGIN_AUTOLOAD')=='1'
 assert Path(os.environ['HOME']).resolve()==(HERE/'run/home').resolve()
 assert Path(os.environ['TMPDIR']).resolve()==(HERE/'run/tmp').resolve()
 assert time.monotonic()+60<clock['deadline_monotonic']
 sys.path.insert(0,str(HERE))
 from finite_kernel import parent_death,arm
 parent_death(parent['pid'])
 timers=[arm(time.CLOCK_MONOTONIC,clock['deadline_monotonic']),
         arm(time.CLOCK_PROCESS_CPUTIME_ID,float(cap))]
 pins=json.loads((HERE/'input-pins.json').read_bytes())
 for path,h in pins.items():assert digest(path)==h,path
 assert digest(HERE/'finite-sequence.json')==admission['finite_sequence_sha256']
 return {'clock':clock,'step':step,'timers':timers,
         'cpu_reserved_total':step['previous_kernel_child_cpu']+6+2*(cap+1),
         'pre_arm_gap_qualified':False}
