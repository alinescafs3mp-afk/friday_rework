"""Narrow ABI for this non-execing finite owner and mode process.
Never arm anything merely by import. All actual calls are NOT_RUN in Sol.
Linux x86_64/glibc layout is pinned by abi-source-pins.json, not guessed runtime
capability. No timer survives exec/fork; mode must establish its own timer.
"""
import ctypes
import os
import signal
import time

class Timespec(ctypes.Structure):
    _fields_ = [('tv_sec', ctypes.c_long), ('tv_nsec', ctypes.c_long)]

class Itimerspec(ctypes.Structure):
    _fields_ = [('it_interval', Timespec), ('it_value', Timespec)]

class Sigevent(ctypes.Structure):
    _fields_ = [('value', ctypes.c_void_p), ('signo', ctypes.c_int),
                ('notify', ctypes.c_int), ('padding', ctypes.c_int * 12)]

TIMERS = []  # Remain armed through final result/lock/pipes closure and process exit.

def libc():
    lib = ctypes.CDLL(None, use_errno=True)
    lib.prctl.argtypes = [ctypes.c_int, ctypes.c_ulong, ctypes.c_ulong,
                         ctypes.c_ulong, ctypes.c_ulong]
    lib.prctl.restype = ctypes.c_int
    lib.timer_create.argtypes = [ctypes.c_int, ctypes.POINTER(Sigevent),
                                 ctypes.POINTER(ctypes.c_void_p)]
    lib.timer_create.restype = ctypes.c_int
    lib.timer_settime.argtypes = [ctypes.c_void_p, ctypes.c_int,
                                  ctypes.POINTER(Itimerspec), ctypes.c_void_p]
    lib.timer_settime.restype = ctypes.c_int
    return lib

def checked(result, operation):
    if result != 0:
        error = ctypes.get_errno()
        raise OSError(error, operation)

def arm(clock, absolute):
    if ctypes.sizeof(Sigevent) != 64 or ctypes.sizeof(Timespec) != 16:
        raise RuntimeError('UNREVIEWED_ABI')
    if time.clock_gettime(clock) >= absolute:
        raise RuntimeError('ALLOCATION_ALREADY_SPENT')
    seconds = int(absolute)
    spec = Itimerspec(Timespec(0, 0), Timespec(seconds, int((absolute-seconds)*1e9)))
    event = Sigevent(); event.signo = signal.SIGKILL; event.notify = 0  # SIGEV_SIGNAL
    timer = ctypes.c_void_p()
    lib = libc()
    checked(lib.timer_create(clock, ctypes.byref(event), ctypes.byref(timer)), 'timer_create')
    # On arming failure abort before Popen/candidate; no capability claim.
    checked(lib.timer_settime(timer, 1, ctypes.byref(spec), None), 'timer_settime(TIMER_ABSTIME)')
    TIMERS.append(timer)
    return {'clock':clock,'absolute':absolute,'signal':int(signal.SIGKILL),
            'armed_return_code':0,'delivery_or_latency_qualified':False}

def parent_death(expected_ppid):
    if os.getppid() != expected_ppid:
        raise RuntimeError('PARENT_ALREADY_LOST')
    checked(libc().prctl(1, signal.SIGKILL, 0, 0, 0), 'PR_SET_PDEATHSIG')
    # Close the parent-death race across prctl. Failure never enters candidate.
    if os.getppid() != expected_ppid:
        raise RuntimeError('PARENT_LOST_DURING_BIND')

def subreaper():
    lib = libc()
    checked(lib.prctl(36, 1, 0, 0, 0), 'PR_SET_CHILD_SUBREAPER')
    # Separate getter with pointer ABI. Do not presume setter success is enough.
    value = ctypes.c_int()
    raw = ctypes.CDLL(None, use_errno=True).prctl
    raw.restype = ctypes.c_int
    checked(raw(37, ctypes.byref(value), 0, 0, 0), 'PR_GET_CHILD_SUBREAPER')
    if value.value != 1:
        raise RuntimeError('SUBREAPER_NOT_EFFECTIVE')
    return 1
