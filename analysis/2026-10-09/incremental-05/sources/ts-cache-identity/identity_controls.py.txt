"""Prepared NOT_RUN controls of genuine current-schema evidence.

Call only after separate native/source qualification. No process, new clock,
admission or synthetic positive. Each negative needs only its own witness.
"""
import copy
import json
from pathlib import Path

EXPECTED = {
    'returned-hash': {'returned_final_exact_transform_chain_broken'},
    'mutable-node-as-return': {'returned_final_exact_transform_chain_broken'},
    'strip-upstream-id': {'upstream_request_identity_missing_or_malformed'},
    'alter-upstream-id': {'payload_not_joined_to_exact_parent_request', 'worker_fetch_upstream_delivery_changed'},
    'same-byte-other-parent': {'worker_fetch_upstream_delivery_changed'},
    'temp-storage-id': {'payload_not_joined_to_exact_parent_request', 'temporary_storage_occurrence_missing'},
    'drop-local-completion': {'local_request_completion_multiplicity_changed'},
    'duplicate-local-completion': {'local_request_completion_multiplicity_changed'},
    'alter-importer-occurrence': {'local_request_completion_identity_changed', 'worker_importer_evaluation_missing'},
    'drop-shared-worker-import': {'evaluation_without_actual_worker_request', 'worker_static_import_chain_incomplete',
        'worker_importer_evaluation_missing', 'cache_without_exact_original_local_request',
        'retained_execution_reuse_identity_changed', 'binding_retained_execution_origin_changed',
        'binding_actual_promise_origin_changed', 'request_native_promise_origin_changed'},
    # Parent delivery remains: missing receiver coverage is semantic refusal.
    'shared-worker-chain': {'ipc_delivery_fetch_occurrence_multiplicity_changed'},
    'typed-inner-omitted': {'vitest_inner_evaluation_missing'},
    'wrapper-changed': {'typed_evaluation_begin_done_multiplicity_changed', 'wrapper_hash_changed'},
    'plugin-output': {'actual_plugin_output_next_input_chain_broken'},
    'graph-edge': {'returned_graph_edges_changed_without_delta', 'graph_delta_previous_edges_changed',
                   'final_graph_edges_not_equal_to_observed_edges'},
    'reuse-origin': {'retained_execution_reuse_identity_changed'},
    'reuse-delivery': {'retained_execution_reuse_identity_changed'},
    'reuse-evaluated-bytes': {'retained_execution_reuse_identity_changed'},
    'drop-reuse': {'cache_reuse_occurrence_multiplicity_changed'},
    'duplicate-reuse': {'cache_reuse_occurrence_multiplicity_changed'},
    'unused-fetch-replacement': {'unused_fetch_relationship_changed'},
    'module-object': {'retained_execution_reuse_identity_changed', 'cache_without_exact_original_local_request'},
}
LOCAL_REFS = ('localRequestSeq', 'importerRequestSeq', 'cacheOriginRequestSeq', 'fetchSeq',
    'moduleObjectSeq', 'bindingSeq', 'priorBindingSeq', 'priorFetchSeq',
    'retainedOriginRequestSeq', 'replacementBindingSeq')


def renumber_references(streams):
    """Translate surviving endpoints, including nested origins. Never redirect
    a deleted endpoint to a surviving row. -1 marks the intentionally removed
    endpoint. Only the control's named semantic refusal is accepted.
    """
    mapping = {(r['pid'], r['seq']): n for stream in streams.values()
               for n, r in enumerate(stream, 1)}
    missing = []
    def translated(pid, seq, key):
        if seq is None: return None
        if (pid, seq) not in mapping:
            missing.append({'pid': pid, 'old_seq': seq, 'field': key})
            return -1
        return mapping[pid, seq]
    def walk(value, inherited_pid=None):
        if isinstance(value, list):
            for child in value: walk(child, inherited_pid)
        elif isinstance(value, dict):
            pid = value.get('pid', inherited_pid)
            for key in LOCAL_REFS:
                if key in value: value[key] = translated(pid, value[key], key)
            if 'deliverySeq' in value:
                value['deliverySeq'] = translated(value['deliveryPid'], value['deliverySeq'], 'deliverySeq')
            if 'returnSeq' in value:
                value['returnSeq'] = translated(value['producerPid'], value['returnSeq'], 'returnSeq')
            if 'seq' in value and 'pid' in value:
                value['seq'] = translated(value['pid'], value['seq'], 'seq')
            for child in value.values():
                if isinstance(child, (dict, list)): walk(child, pid)
    for stream in streams.values():
        for row in stream: walk(row)
    return {'surviving_endpoint_map': [{'pid': p, 'old_seq': s, 'new_seq': n}
        for (p, s), n in sorted(mapping.items())], 'deliberately_removed_endpoint_references': missing}


def worker(r): return r['pid'], r['workerId'], r['poolId']


def positive_witnesses(streams):
    """Inspect only an already consumed genuine positive. No generated events.

These are scenario observations, never substitute acceptance of the native
owner/custody or the original mandatory full gate.
    """
    rows = [r for stream in streams.values() for r in stream]
    fetch = {(r['pid'], r['seq']): r for r in rows if r['event'] in
             ('VITEST_WORKER_FETCH', 'VITEST_ENV_FETCH', 'VITEST_SERVER_FETCH') and 'canonical' in r}
    reuses = [r for r in rows if r['event'] == 'VITE_MODULE_REUSE' and 'canonical' in r]
    bindings = {(r['pid'], r['seq']): r for r in rows if r['event'] == 'VITE_MODULE_BINDING'}
    def alias(r):
        b = bindings[r['pid'], r['bindingSeq']]
        return (r['newDeliveryUnused'] and r['requested'] != r['origin']['requested']
                and r['requestId'] != r['origin']['requestId'] and r['returnSeq'] != r['origin']['returnSeq']
                and r['sameFinalSourceBytes']
                and not b['invalidated'] and b['retainedOriginRequestSeq'] == r['origin']['localRequestSeq'])
    scenarios = {
        'cache-true-retained-meta': [r for r in reuses if not r['newDeliveryUnused'] and any(
            h['event'] == 'VITE_WORKER_CACHE_HINT' and worker(h) == worker(r)
            and h['fetchSeq'] == r['fetchSeq'] and h['moduleObjectSeq'] == r['moduleObjectSeq']
            and h['seq'] < r['localRequestSeq'] for h in rows)],
        'full-alias-evaluated-reuse': [r for r in reuses if alias(r) and r['cacheState'] == 'evaluated-cache'
            and fetch[r['pid'], r['fetchSeq']]['tmp'] is None],
        'tmp-alias-evaluated-reuse': [r for r in reuses if alias(r) and r['cacheState'] == 'evaluated-cache'
            and fetch[r['pid'], r['fetchSeq']]['tmp'] is not None],
        'alias-pending-same-object': [r for r in reuses if alias(r) and r['cacheState'] == 'pending-cache'],
        'native-cycle': [r for r in reuses if r['cacheState'] == 'cycle'],
        'replaced-unrequested-fetch': [r for r in rows if r['event'] == 'VITE_UNUSED_FETCH'],
        'distinct-request-tmp-storage': [r for r in rows if r['event'] == 'VITEST_TMP_DELIVERY'
            and r['cacheMetadata']['__frwEvidence']['requestId'] != r['requestId']],
        'builtin-request': [r for r in rows if r['event'] == 'VITE_WORKER_REQUEST' and r.get('builtin')],
        'pinned-external-request': [r for r in rows if r['event'] == 'VITE_WORKER_REQUEST' and 'externalCanonical' in r],
    }
    finals = [r for r in rows if r['event'] == 'VITEST_FINAL_RESULT_SOURCE' and 'canonical' in r]
    captures = {(r['pid'], r['seq']): r for r in rows if r['event'] == 'VITE_RETURN_CAPTURE'}
    transitions = {(r['pid'], r['requestId']): r for r in rows if r['event'] == 'VITEST_FINAL_SOURCE_TRANSITION'}
    pairs = [(a, b) for a in finals for b in finals if a['pid'] == b['pid']
        and all(a[k] == b[k] for k in ('id', 'canonical', 'environment', 'transformId',
                                      'returnedTransformSha256', 'finalSourceSha256'))
        and a['requestId'] != b['requestId'] and a['seq'] < b['seq']]
    overlap = []
    for a, b in pairs:
        ta = transitions[a['pid'], a['requestId']]; tb = transitions[b['pid'], b['requestId']]
        if (captures[a['producerPid'], a['returnSeq']]['seq'] < ta['seq']
            and captures[b['producerPid'], b['returnSeq']]['seq'] < ta['seq'] and ta['inlineApplied']
            and ta['nodeBeforeSha256'] != ta['nodeAfterSha256']
            and tb['nodeBeforeSha256'] == ta['nodeAfterSha256'] and not tb['inlineApplied']
            and b['returnedTransformSha256'] != tb['nodeBeforeSha256']): overlap.extend((a, b))
    scenarios['same-byte-distinct-parent-returns'] = [r for pair in pairs for r in pair]
    scenarios['actual-inline-map-overlap'] = overlap
    return {name: {'state': 'GENUINE_WITNESS_OBSERVED' if selected else 'NOT_RUN',
                   'reason': None if selected else 'NO_GENUINE_WITNESS',
                   'rows': [{'pid': r['pid'], 'seq': r['seq'], 'event': r['event']} for r in selected]}
            for name, selected in scenarios.items()}


def choose_witness(kind, rows):
    """No global overlap/tmp/shared-worker prerequisite."""
    def first(predicate): return next((r for r in rows if predicate(r)), None)
    fetch = lambda r: r['event'] == 'VITEST_WORKER_FETCH' and 'canonical' in r
    final = lambda r: r['event'] == 'VITEST_FINAL_RESULT_SOURCE' and 'canonical' in r
    if kind == 'returned-hash': return first(final)
    if kind == 'mutable-node-as-return':
        for r in rows:
            if final(r):
                t = first(lambda t: t['event'] == 'VITEST_FINAL_SOURCE_TRANSITION'
                    and t['pid'] == r['pid'] and t['requestId'] == r['requestId']
                    and t.get('nodeBeforeSha256') is not None
                    and t['nodeBeforeSha256'] != r['returnedTransformSha256'])
                if t is not None: return r, t
        return None
    if kind in ('strip-upstream-id', 'alter-upstream-id'): return first(fetch)
    if kind == 'same-byte-other-parent':
        for r in rows:
            if fetch(r):
                other = first(lambda f: final(f) and f['canonical'] == r['canonical']
                    and f['environment'] == r['environment'] and f['requestId'] != r['requestId']
                    and f['returnedTransformSha256'] == r['returnedTransformSha256']
                    and f['finalSourceSha256'] == r['finalSourceSha256'])
                if other is not None: return r, other
        return None
    if kind == 'temp-storage-id': return first(lambda r: fetch(r) and r.get('cacheMetadata') is not None)
    if kind in ('drop-local-completion', 'duplicate-local-completion'):
        return first(lambda r: r['event'] == 'VITE_WORKER_REQUEST_DONE' and 'canonical' in r)
    if kind == 'alter-importer-occurrence':
        return first(lambda r: r['event'] == 'VITE_WORKER_REQUEST' and r.get('importerRequestSeq') is not None)
    if kind in ('drop-shared-worker-import', 'shared-worker-chain'):
        for f in rows:
            if fetch(f) and any(fetch(x) and x['canonical'] == f['canonical'] and worker(x) != worker(f) for x in rows):
                request = first(lambda r: r['event'] == 'VITE_WORKER_REQUEST' and worker(r) == worker(f)
                    and r.get('fetchSeq') == f['seq'] and r['cacheState'] == 'fresh' and r['importer'] is not None)
                if request is not None and all(any(r['event'] == event and worker(r) == worker(f)
                    and r.get('localRequestSeq') == request['seq'] for r in rows)
                    for event in ('VITEST_EVAL_BEGIN', 'VITEST_EVAL_DONE', 'VITE_RUNNER_EVAL_BEGIN', 'VITE_RUNNER_EVAL_DONE')):
                    return f, request
        return None
    if kind in ('typed-inner-omitted', 'wrapper-changed'): return first(lambda r: r['event'] == 'VITEST_EVAL_BEGIN')
    if kind == 'plugin-output': return first(lambda r: r['event'] == 'VITE_PLUGIN_OUTPUT')
    if kind == 'graph-edge': return first(lambda r: r['event'] == 'VITE_TRANSFORM_GRAPH' and r['imports'])
    if kind in ('reuse-origin', 'reuse-delivery', 'reuse-evaluated-bytes', 'drop-reuse', 'duplicate-reuse', 'module-object'):
        return first(lambda r: r['event'] == 'VITE_MODULE_REUSE' and 'canonical' in r)
    if kind == 'unused-fetch-replacement': return first(lambda r: r['event'] == 'VITE_UNUSED_FETCH')
    raise ValueError('unknown prepared control: ' + kind)


def mutate(kind, streams, witness):
    def flipped(value): return ('0' if value[0] != '0' else '1') + value[1:]
    def delete(predicate):
        removed = []
        for name, stream in streams.items():
            removed.extend(r for r in stream if predicate(r))
            streams[name] = [r for r in stream if not predicate(r)]
        return removed
    removed = []
    if kind == 'returned-hash': witness['returnedTransformSha256'] = flipped(witness['returnedTransformSha256'])
    elif kind == 'mutable-node-as-return':
        row, transition = witness; row['returnedTransformSha256'] = transition['nodeBeforeSha256']
    elif kind == 'strip-upstream-id': del witness['requestId']
    elif kind == 'alter-upstream-id': witness['requestId'] += 10000000
    elif kind == 'same-byte-other-parent':
        row, other = witness
        for key in ('producerPid', 'returnSeq', 'transformId', 'requestId', 'returnedTransformSha256', 'finalSourceSha256'):
            row[key] = other[key]
    elif kind == 'temp-storage-id': witness['cacheMetadata']['__frwEvidence']['requestId'] += 10000000
    elif kind in ('drop-local-completion', 'drop-reuse'): removed = delete(lambda r: r is witness)
    elif kind in ('duplicate-local-completion', 'duplicate-reuse'):
        stream = next(s for s in streams.values() if any(r is witness for r in s))
        duplicate = copy.deepcopy(witness); duplicate['seq'] = max(r['seq'] for r in stream) + 1000000
        stream.insert(stream.index(witness) + 1, duplicate)
    elif kind == 'alter-importer-occurrence': witness['importerRequestSeq'] += 10000000
    elif kind == 'drop-shared-worker-import':
        fetch, request = witness
        removed = delete(lambda r: r is request or r['event'] == 'VITE_WORKER_REQUEST_DONE'
            and r['pid'] == request['pid'] and r['localRequestSeq'] == request['seq'])
    elif kind == 'shared-worker-chain':
        fetch, request = witness
        # Full predecessor negative plus new completion/reuse/unused rows.
        # Parent deliveries and all opposite-worker rows survive.
        kinds = ('VITEST_WORKER_FETCH', 'VITE_WORKER_REQUEST', 'VITE_WORKER_REQUEST_DONE',
            'VITEST_EVAL_BEGIN', 'VITEST_EVAL_DONE', 'VITE_RUNNER_EVAL_BEGIN', 'VITE_RUNNER_EVAL_DONE',
            'VITE_MODULE_REUSE', 'VITE_UNUSED_FETCH')
        removed = delete(lambda r: worker(r) == worker(fetch) and r.get('canonical') == fetch['canonical']
                         and r['event'] in kinds)
        if any(worker(r) != worker(fetch) for r in removed): raise AssertionError('opposite worker changed')
    elif kind == 'typed-inner-omitted':
        removed = delete(lambda r: r['pid'] == witness['pid'] and r.get('localRequestSeq') == witness['localRequestSeq']
                         and r['event'] in ('VITEST_EVAL_BEGIN', 'VITEST_EVAL_DONE'))
    elif kind == 'wrapper-changed': witness['wrappedSha256'] = flipped(witness['wrappedSha256'])
    elif kind == 'plugin-output': witness['codeSha256'] = flipped(witness['codeSha256'])
    elif kind == 'graph-edge': witness['imports'].pop()
    elif kind == 'reuse-origin': witness['origin']['requestId'] += 10000000
    elif kind == 'reuse-delivery': witness['deliverySeq'] += 10000000
    elif kind == 'reuse-evaluated-bytes': witness['origin']['codeSha256'] = flipped(witness['origin']['codeSha256'])
    elif kind == 'unused-fetch-replacement': witness['replacementBindingSeq'] += 10000000
    elif kind == 'module-object': witness['origin']['moduleObjectSeq'] += 10000000
    else: raise ValueError('unknown prepared control: ' + kind)
    references = renumber_references(streams) if removed or kind.startswith('duplicate-') else None
    return {'removed_rows': [{'pid': r['pid'], 'old_seq': r['seq'], 'event': r['event']} for r in removed],
            'references': references}


def prepared_controls(join, genuine_run, budget, owner, max_bytes, destination, control_names=None):
    baseline = Path(genuine_run)
    join(baseline, budget, owner, max_bytes)
    originals = {p.name: p.read_bytes() for p in baseline.iterdir() if p.name in
        ('clock.json', 'pre-exec-os.json', 'progress.json') or p.name.startswith('guard-') and p.suffix == '.jsonl'}
    original_streams = {n: [json.loads(line) for line in raw.splitlines()]
                        for n, raw in originals.items() if n.startswith('guard-')}
    outcomes = {}
    for kind in EXPECTED if control_names is None else control_names:
        budget.check()
        streams = copy.deepcopy(original_streams); rows = [r for stream in streams.values() for r in stream]
        witness = choose_witness(kind, rows)
        if witness is None:
            outcomes[kind] = {'state': 'NOT_RUN', 'reason': 'NO_GENUINE_WITNESS_FOR_' + kind}
            continue
        witnesses = witness if isinstance(witness, tuple) else (witness,)
        witness_refs = [{'pid': r['pid'], 'seq': r['seq'], 'event': r['event']} for r in witnesses]
        delta = mutate(kind, streams, witness)
        target = Path(destination) / kind; target.mkdir(mode=0o700, parents=False, exist_ok=False)
        for name, raw in originals.items():
            if name in streams: raw = (''.join(json.dumps(r) + '\n' for r in streams[name])).encode()
            (target / name).write_bytes(raw)
        try: join(target, budget, owner, max_bytes)
        except ValueError as error:
            if str(error) not in EXPECTED[kind]: raise AssertionError('unrelated refusal ' + kind + ': ' + str(error)) from error
            outcomes[kind] = {'state': 'SEMANTIC_REFUSAL_OBSERVED', 'refusal': str(error), 'genuine_witnesses': witness_refs, 'delta': delta}
        else: raise AssertionError('semantic mutation accepted: ' + kind)
    return {'actual_positive_consumed': True, 'positive_scenario_witnesses': positive_witnesses(original_streams),
            'negative_outcomes': outcomes, 'synthetic_native_execution': False,
            'native_effects': [], 'missing_witnesses_are_individual_NOT_RUN': True}
