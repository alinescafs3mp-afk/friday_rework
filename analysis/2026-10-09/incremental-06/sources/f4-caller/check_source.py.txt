# INERT ANALYSIS COPY (.txt); NOT AN INSTALLABLE OR AUTHORIZED ENTRY.
"""Finite read/hash/AST checker; never imports or runs the proposed candidate."""
import ast
import difflib
import hashlib
import json
from pathlib import Path

OUT = Path(__file__).resolve().parent
BASE = OUT.parent/'finite-owner-custody-repair/candidate/qualification'
SRC = OUT/'source/qualification'


def call_name(node):
    if isinstance(node, ast.Call):
        return ast.unparse(node.func)
    return None


def function(tree, name):
    return next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == name)


def first_body_call(node):
    first = node.body[0]
    return call_name(first.value) if isinstance(first, ast.Expr) else None


def ref(path):
    b = path.read_bytes()
    return {'path': str(path), 'sha256': hashlib.sha256(b).hexdigest(), 'bytes': len(b)}


trees = {p.name: ast.parse(p.read_text(), filename=str(p)) for p in SRC.glob('*.py')}
join = trees['existing_caller_join.py']
entry = trees['invoke_qualification.py']
refusal = function(join, 'ensure_f4_bootstrap')
assert len(refusal.body) == 1 and isinstance(refusal.body[0], ast.Raise)
assert first_body_call(function(join, 'invoke_existing_owner')) == 'ensure_f4_bootstrap'
main = function(entry, 'main')
refusal_position = next(i for i,n in enumerate(main.body) if isinstance(n, ast.Expr)
                        and call_name(n.value) == 'ensure_f4_bootstrap')
clock_position = next(i for i,n in enumerate(main.body) if isinstance(n, ast.Assign)
                      and call_name(n.value) == 'time.monotonic')
assert refusal_position < clock_position
invoke = function(join, 'invoke_existing_owner')
calls = [(n.lineno, call_name(n)) for n in ast.walk(invoke) if isinstance(n, ast.Call)]
line = lambda name: min(n for n, name_at in calls if name_at == name)
assert line('install_inherited_envelope') < line('os.fork')
assert line('CallerWall') < line('os.fork') < line('wall.bind_unreaped_root')
assert line('wall.bind_unreaped_root') < line('wait_exec_pipe')
assert line('restore_observer_envelope') > line('os.fork')
assert line('wait_exec_pipe') < line('os.write')
assert line('Custody') < line('wall.bind_unreaped_root')
wall = trees['caller_wall.py']
all_names = {call_name(n) for tree in trees.values() for n in ast.walk(tree) if isinstance(n, ast.Call)}
assert 'os.killpg' not in all_names
assert 'signal.pidfd_send_signal' in all_names
assert 'subprocess.Popen' not in all_names  # Existing measured owner still has original Popen.
assert 'resource.getrusage' in all_names
assert 'os.wait4' not in all_names  # Only existing pinned Custody reaps, never competing reaper.
assert any(isinstance(n, ast.Compare) and 'measured_cpu <= 45' == ast.unparse(n)
           for n in ast.walk(invoke))
files = [ref(p) for p in sorted(SRC.glob('*.py'))]
originals = json.loads((OUT/'input-refs.json').read_text())
changed_inputs = [r['path'] for r in originals
                  if hashlib.sha256(Path(r['path']).read_bytes()).hexdigest() != r['sha256']]
assert not changed_inputs
patches = []
for p in sorted(SRC.glob('*.py')):
    prior = BASE/p.name
    original = prior.read_text().splitlines(keepends=True) if prior.exists() else []
    patches.extend(difflib.unified_diff(original, p.read_text().splitlines(keepends=True),
                                      fromfile=str(prior) if prior.exists() else '/dev/null',
                                      tofile=str(p)))
(OUT/'SOURCE.diff').write_text(''.join(patches))
report = {'state': 'STATIC_ONLY_PROTOTYPE_HARD_REFUSED', 'AST_files': len(trees),
          'candidate_imports': 0, 'candidate_tests': 0, 'candidate_kernel_calls': 0,
          'source_file_refs': files, 'all_pinned_inputs_unchanged': not changed_inputs,
          'entry_refuses_before_clock': True, 'callable_refuses_before_effects': True,
          'unreachable_prototype_order_checks': 'PASS',
          'effective_limit_install_line': line('install_inherited_envelope'),
          'fork_line': line('os.fork'), 'parent_restore_line': line('restore_observer_envelope'),
          'pidfd_bind_line': line('wall.bind_unreaped_root'),
          'known_source_endpoint_gap_still_present': True,
          'runtime_control_outcomes': 'NOT_RUN',
          'no_candidate_import_or_execution_by_checker': True}
(OUT/'source-checks.json').write_text(json.dumps(report, indent=2)+'\n')
print(json.dumps({k:v for k,v in report.items() if k != 'source_file_refs'}))
