"""Child side of the existing caller's explicit fork/exec boundary.
The parent applies effective CPU6/default-unblocked SIGXCPU, AS2GiB and
[0,1] before fork. Hard45/AS2GiB are made irreversible here before exec.
The inherited SOFT CPU bound and retained parent hard ceiling are distinct;
accepting that original-contract interpretation requires independent review.
No limit is installed by importing this module.
"""
import os
import resource
import signal


def finite_owner_entry_limits():
    if resource.getrlimit(resource.RLIMIT_CPU)[0] != 6:
        raise RuntimeError('MISSING_INHERITED_ROOT_CPU_SOFT_BOUND')
    if resource.getrlimit(resource.RLIMIT_AS)[0] != 2 << 30:
        raise RuntimeError('MISSING_INHERITED_ROOT_AS_SOFT_BOUND')
    if sorted(os.sched_getaffinity(0)) != [0, 1]:
        raise RuntimeError('MISSING_INHERITED_ROOT_AFFINITY')
    if signal.getsignal(signal.SIGXCPU) != signal.SIG_DFL:
        raise RuntimeError('ROOT_SOFT_BOUND_HAS_NONDEFAULT_DISPOSITION')
    blocked = signal.pthread_sigmask(signal.SIG_BLOCK, set())
    if signal.SIGXCPU in blocked:
        raise RuntimeError('ROOT_SOFT_BOUND_SIGNAL_BLOCKED')
    # Birth CPU is NOT reset. The inherited effective bound already covered
    # the fork prefix; this installs the exact irreversible owner pair.
    resource.setrlimit(resource.RLIMIT_CPU, (6, 45))
    resource.setrlimit(resource.RLIMIT_AS, (2 << 30, 2 << 30))
    resource.setrlimit(resource.RLIMIT_CORE, (0, 0))
    signal.signal(signal.SIGCHLD, signal.SIG_DFL)
    signal.signal(signal.SIGALRM, signal.SIG_DFL)
    signal.pthread_sigmask(signal.SIG_UNBLOCK, {signal.SIGCHLD, signal.SIGALRM})
