"""One existing terminal caller's original-monotonic deadline, not a worker.
The kernel timer targets this live observer; its handler signals only a pidfd.
No timer/kernel call occurs on import. Delivery/latency is NOT_RUN.
"""
import ctypes
import os
import signal
import time
from finite_kernel import Itimerspec, Sigevent, Timespec, checked, libc


class CallerWall:
    def __init__(self, deadline):
        self.deadline = deadline
        self.root_fd = None
        self.expired = False
        self.signal_error = None
        self.timer = ctypes.c_void_p()
        self.lib = libc()
        self.old_handler = signal.getsignal(signal.SIGALRM)
        if self.old_handler not in (signal.SIG_DFL, signal.SIG_IGN):
            raise RuntimeError('OBSERVER_ALREADY_HAS_ALARM_HANDLER')
        if time.monotonic() >= deadline:
            raise RuntimeError('ORIGINAL_DEADLINE_ALREADY_EXPIRED')
        # One thread; SIGALRM is blocked by the caller during creation/binding.
        signal.signal(signal.SIGALRM, self._expire)
        event = Sigevent()
        event.signo = signal.SIGALRM
        event.notify = 0  # SIGEV_SIGNAL, observer stays alive for terminal wait4.
        seconds = int(deadline)
        spec = Itimerspec(Timespec(0, 0),
                          Timespec(seconds, int((deadline-seconds)*1e9)))
        checked(self.lib.timer_create(time.CLOCK_MONOTONIC, ctypes.byref(event),
                                     ctypes.byref(self.timer)), 'caller timer_create')
        checked(self.lib.timer_settime(self.timer, 1, ctypes.byref(spec), None),
                'caller original TIMER_ABSTIME')

    def _expire(self, signum, frame):
        self.expired = True
        self.kill_root()

    def kill_root(self):
        # A pidfd never retargets a recycled PID, including after wait4.
        if self.root_fd is None:
            return
        try:
            signal.pidfd_send_signal(self.root_fd, signal.SIGKILL)
        except ProcessLookupError:
            pass  # Terminal/reaped is established only by actual wait4.
        except OSError as error:
            self.signal_error = repr(error)

    def bind_unreaped_root(self, pid):
        if self.root_fd is not None:
            raise RuntimeError('ROOT_PIDFD_ALREADY_BOUND')
        # Called immediately after our own fork, before any wait/reaper can
        # release that PID. A zombie is still the same exact child.
        self.root_fd = os.pidfd_open(pid, 0)
        if self.expired or time.monotonic() >= self.deadline:
            self.expired = True
            self.kill_root()

    def close_after_observation(self):
        blocked = signal.pthread_sigmask(signal.SIG_BLOCK, {signal.SIGALRM})
        try:
            self.lib.timer_delete.argtypes = [ctypes.c_void_p]
            self.lib.timer_delete.restype = ctypes.c_int
            checked(self.lib.timer_delete(self.timer), 'caller timer_delete')
            fd, self.root_fd = self.root_fd, None
            if fd is not None:
                os.close(fd)
            # Pending alarm must not be delivered to the prior default action.
            while signal.SIGALRM in signal.sigpending():
                signal.sigwait({signal.SIGALRM})
            signal.signal(signal.SIGALRM, self.old_handler)
        finally:
            signal.pthread_sigmask(signal.SIG_SETMASK, blocked)
