# INERT ANALYSIS COPY (.txt); NOT AN INSTALLABLE OR AUTHORIZED ENTRY.
"""Read-only source/order checks; never import or execute a candidate.

This checks retained inputs and the mandatory compatibility precondition.
It is not a Linux/ptrace/timer test or proof of binary/source equivalence.
"""
import ast
import hashlib
import json
from pathlib import Path

W = Path('<ASTRA_WORK>')
S = Path('<SOL_WORK>')
E = S / 'FRW041-FINITE-GNU-DETACH-evidence'
checks = []


def check(name, condition):
    assert condition, name
    checks.append(name)


def pinned(ref):
    p = Path(ref['path'])
    b = p.read_bytes()
    check('SHA:' + str(p), hashlib.sha256(b).hexdigest() == ref['sha256'])
    if 'bytes' in ref:
        check('LENGTH:' + str(p), len(b) == ref['bytes'])
    return b


inputs = json.loads((W / 'finite-gnu-detach-sol/INPUTS.json').read_bytes())
for ref in inputs.values():
    pinned(ref)
gnu = pinned(inputs['GNU_source']).decode()
review = json.loads(pinned(inputs['ptrace_review']))
fork_source = pinned(review['refs']['linux_fork']).decode()
ptrace_source = pinned(review['refs']['linux_ptrace']).decode()
exit_source = pinned(review['refs']['linux_exit']).decode()
task = (W / 'finite-gnu-detach-sol/TASK.txt').read_text()
sequence = json.loads((W / 'finite-owner-custody-repair/candidate/qualification/finite-sequence.json').read_bytes())
manifest = json.loads(pinned(inputs['base_manifest']))
ast_count = 0
for ref in manifest['files']:
    b = pinned(ref)
    if Path(ref['path']).suffix == '.py':
        ast.parse(b, filename=ref['path'])
        ast_count += 1

check('trusted GNU fork occurs before parent settimeout',
      gnu.index('monitored_pid = fork ();') < gnu.index('settimeout (timeout, true);'))
check('GNU timer success branch is inside settimeout',
      'timer_create (CLOCK_REALTIME, nullptr, &timerid) == 0' in gnu and
      'timer_settime (timerid, 0, &its, nullptr) == 0' in gnu)
check('GNU fallback alarm is also inside settimeout', 'alarm (timeint);' in gnu)
check('kernel FORK event precedes return to GNU userspace',
      fork_source.index('ptrace_event_pid(trace, pid);') <
      fork_source.index('return nr;', fork_source.index('ptrace_event_pid(trace, pid);')))
check('child wake precedes parent FORK event (both notification orders possible)',
      fork_source.index('wake_up_new_task(p);') < fork_source.index('ptrace_event_pid(trace, pid);'))
check('EXITKILL acts on tracer exit, not on mere tracer stop',
      'void exit_ptrace(' in ptrace_source and 'if (unlikely(p->ptrace & PT_EXITKILL))' in ptrace_source and
      'send_sig_info(SIGKILL, SEND_SIG_PRIV, p);' in ptrace_source)
check('terminal trace notification is separate from final parent CPU credit',
      'state == EXIT_DEAD && thread_group_leader(p)' in exit_source and
      'if (state == EXIT_TRACE)' in exit_source and 'ptrace_unlink(p);' in exit_source)
check('mandatory stop-before-detach control retained explicitly', 'owner-stop BEFORE detach' in task)
check('mandatory compatibility refusal, no relabelled admission',
      'stop speculative implementation at that concrete counterexample' in task and
      'relabeling admission' in task)
check('original GNU60 plus closure60 unchanged',
      sequence['gnu']['maximum_step_seconds'] == 60 and
      sequence['gnu']['shared_clock_closure_reserve_seconds'] == 60)
check('prior review B1 exact stated independent-GNU-cutoff gap',
      any(x['id'] == 'B1_TRACED_GNU_TIMEOUT_DEPENDENCY' and
          'does not establish the independent original GNU60 cutoff' in x['finding']
          for x in review['unclosed_or_rejected']))

# Abstract source ordering only. No wait, signal, timer, process or candidate.
# A successful timer wrapper cannot execute while GNU remains at a kernel
# FORK event before its fork returns. DETACH still requires the stopped owner.
ordering = ['GNU_FORK_KERNEL_EVENT', 'OWNER_DETACH', 'GNU_FORK_RETURNS',
            'GNU_SUCCESSFUL_TIMER_ARM', 'SHIM_READY_WRITE', 'OWNER_MODE_RELEASE']
check('proposed readiness is strictly downstream of owner detach',
      ordering.index('OWNER_DETACH') < ordering.index('GNU_SUCCESSFUL_TIMER_ARM'))
reachable_without_owner_after_fork_event = ordering[:1]
check('stop before DETACH cannot yield timer arm/readiness/release',
      not any(x in reachable_without_owner_after_fork_event for x in
              ['GNU_SUCCESSFUL_TIMER_ARM', 'SHIM_READY_WRITE', 'OWNER_MODE_RELEASE']))

out = {'state': 'STATIC_SOURCE_PRECONDITION_CONTRADICTION',
       'checks': checks, 'checks_count': len(checks), 'failures': [],
       'AST_count': ast_count, 'base_files': len(manifest['files']),
       'source_order': ordering,
       'counterexample': 'Owner stops at the existing GNU FORK event before DETACH. '
         'GNU cannot return from fork or arm its original60 timer; shim cannot run. '
         'Owner original180 death/EXITKILL can end retained tracees conditionally '
         'but supplies neither the independent GNU60 cutoff nor full60 closure room.',
       'candidate_imports_or_execution': 0, 'kernel_controls': 'NOT_RUN',
       'independent_review': 'REQUIRED', 'F5_closed': False,
       'not_claimed': ['observed host failure', 'Linux impossibility',
                       'binary/source equivalence', 'original180 failure after validated attachment']}
p = E / 'source-checks.json'
assert not p.exists()
p.write_text(json.dumps(out, ensure_ascii=False, indent=2) + '\n')
p.chmod(0o400)
print(json.dumps({k: out[k] for k in ['state', 'checks_count', 'AST_count', 'base_files', 'F5_closed']}))
