"""Finite SOURCE controls, ALL NOT_RUN, no native containment/cessation proof.

Exact owner methods with explicit inert native observations exercise retention
and admission ordering. They never launch a unit/registry/test interpreter or
grant a clock. Files in pytest tmp_path are model evidence, not live receipts.
Real fault/capability controls require separate independent native admission.
"""
import ast
import copy
import hashlib
import json
import os
import pathlib
import threading
from types import SimpleNamespace
from unittest.mock import Mock

import pytest


ROOT = pathlib.Path(__file__).parent.parent


def definitions(path, names, **namespace):
    tree = ast.parse(path.read_bytes())
    nodes = [n for n in ast.walk(tree) if isinstance(n, ast.FunctionDef) and n.name in names]
    assert {n.name for n in nodes} == set(names)
    for node in nodes:
        node.decorator_list = []
    exec(compile(ast.Module(body=nodes, type_ignores=[]), str(path), 'exec'), namespace)
    return namespace


def owner_model(tmp_path):
    class ModelSupervisorError(RuntimeError):
        pass
    def check(ok, why):
        if not ok:
            raise RuntimeError(why)
    base = definitions(ROOT / 'owner-helper/owner_observer.py',
        {'put', 'private', 'read', 'sha'}, Path=pathlib.Path, os=os,
        stat=__import__('stat'), json=json, hashlib=hashlib, require=check)
    h = SimpleNamespace(**{k: base[k] for k in ('put', 'private', 'read', 'sha')},
                        clock_check=Mock(), cohort=Mock())
    names = {'intent_identity', 'stop_requested', 'persist_stop', 'close',
             'require_supported_containment', 'require_current', 'reserve_binding',
             'retain_registry_disposition', 'original_row_allows_next_phase',
             'submit', 'unit_entry'}
    namespace = definitions(ROOT / 'owner-helper/runtest_outer.py', names,
        check=check, Path=pathlib.Path, SupervisorError=ModelSupervisorError,
        time=SimpleNamespace(monotonic=lambda: 10), hashlib=hashlib)
    Model = type('ExactOwnerSourceModel', (), {k: namespace[k] for k in names})
    owner = Model()
    owner.h = h
    owner.out = tmp_path
    owner.index = 0
    owner.parent = tmp_path / 'prepared-phase'; owner.parent.mkdir(mode=0o700)
    run = owner.parent / 'run'; run.mkdir(mode=0o700)
    base['put'](run / 'clock.json', {'model': 'original-phase'})
    owner.control = owner.parent / 'native-owner'
    owner.pre = {'clock_sha256': 'a'*64}
    owner.phase = {'name': 'prepared-phase', 'runtest_file': 'f000'}
    owner.spec = {'unit': 'original-model.service'}
    owner.original = {'boot_id': 'source-model-boot', 'assignment': 'original-model'}
    owner.deadline = 1800
    owner.row = {'model': True}
    owner.binding = {'original': 'inert-binding'}
    owner.registry_disposition = None
    owner.disposition_reentered = False
    owner._stop_pending = False
    owner.manager = Mock()
    owner._cgroup = Mock(return_value=False)
    owner.supervisor = SimpleNamespace(stop=Mock(), observe=Mock())
    observed = SimpleNamespace(quiescent=True, missing=True, invocation_id='', result='source-model')
    owner.supervisor.stop.return_value = observed
    owner.supervisor.observe.return_value = observed
    return owner


def original_row(owner, **changes):
    row = {'session_id': 'same-session', 'owner_task_id': 'original-model',
           'runtest_custody': True, 'runtest_unit': 'f000', 'runtest_deadline': 1800,
           'runtest_boot_id': 'source-model-boot', 'runtest_outer': owner.binding,
           'pid': 811, 'host_start_time': 33, 'runtest_pgid': 811, 'runtest_sid': 811,
           'runtest_stop_requested': False, 'runtest_launch_pending': False,
           'runtest_persistence_error': '', 'runtest_outer_closed': None,
           'runtest_pidfd_error': ''}
    row.update(changes)
    return row


def checkpoint(owner, row):
    owner.control.mkdir(mode=0o700, exist_ok=True)
    profile = owner.control / 'profile'; profile.mkdir(mode=0o700, exist_ok=True)
    path = profile / 'processes.json'
    path.write_text(json.dumps([row]))
    return owner.h.sha(path)


def recovered_session(owner, **changes):
    # Recovery's new safety fence is deliberately true. It must not overwrite
    # or stand in for the original durable disposition.
    return SimpleNamespace(**(dict(id='same-session', runtest_outer=owner.binding,
        runtest_stop_requested=True, runtest_launch_pending=False,
        runtest_persistence_error='', runtest_pidfd_error='') | changes))


def test_cancel_before_allocation_is_durable_before_native_io(tmp_path):
    owner = owner_model(tmp_path); owner.binding = None
    order = []
    writer = owner.h.put
    def put(*args):
        order.append('intent'); return writer(*args)
    owner.h.put = put
    owner.manager.side_effect = lambda: order.append('manager')
    owner.supervisor.stop.side_effect = lambda row: order.append('stop') or SimpleNamespace()
    with pytest.raises(RuntimeError, match='UNKNOWN_SUBMISSION_RETAINED'):
        owner.close(stop=True)
    assert order == ['intent', 'manager', 'stop']
    assert not owner.control.exists() and owner.stop_requested()
    saved = (owner.parent / 'native-stop.json').read_bytes()
    with pytest.raises(RuntimeError, match='STOP_BEFORE_ALLOCATION'):
        owner.submit()
    with pytest.raises(RuntimeError, match='STOP_BEFORE_UNIT_ENTRY'):
        owner.unit_entry()
    with pytest.raises(RuntimeError, match='OUTER_STOP_RETAINED'):
        owner.reserve_binding()
    assert (owner.parent / 'native-stop.json').read_bytes() == saved


@pytest.mark.parametrize('cut', ['write', 'partial_write', 'readback'])
def test_cancel_persistence_failure_still_attempts_exact_stop(tmp_path, cut):
    owner = owner_model(tmp_path)
    if cut == 'readback':
        owner.h.read = Mock(side_effect=OSError('readback'))
    else:
        def fail(path, value):
            if cut == 'partial_write':
                pathlib.Path(path).write_bytes(b'partial')
                pathlib.Path(path).chmod(0o400)
            raise OSError(cut)
        owner.h.put = fail
    with pytest.raises(RuntimeError, match='INTENT_PERSISTENCE_FAILED'):
        owner.close(stop=True)
    owner.supervisor.stop.assert_called_once_with(owner.row)
    assert not owner.control.exists()
    if cut != 'write':
        assert (owner.parent / 'native-stop.json').exists()
    # No successful cessation/continuation result was returned.
    assert owner.registry_disposition is None
    assert owner._stop_pending
    if cut == 'write':
        with pytest.raises(RuntimeError, match='STOP_BEFORE_ALLOCATION'):
            owner.submit()


def test_cancel_manager_failure_retains_original_intent(tmp_path):
    owner = owner_model(tmp_path)
    owner.manager.side_effect = OSError('native unavailable')
    with pytest.raises(OSError, match='native unavailable'):
        owner.close(stop=True)
    assert owner.stop_requested()
    owner.supervisor.stop.assert_not_called()


@pytest.mark.parametrize('entry', ['submit', 'unit_entry', 'reserve_binding'])
def test_unsupported_user_containment_refuses_before_allocation_or_rpc(tmp_path, entry):
    owner = owner_model(tmp_path)
    with pytest.raises(RuntimeError, match='EFFECTIVE_CGROUP_CONTAINMENT_UNQUALIFIED'):
        getattr(owner, entry)()
    assert not owner.control.exists()
    owner.manager.assert_not_called()
    owner.supervisor.observe.assert_not_called()
    owner.supervisor.stop.assert_not_called()


def test_effective_capability_refusal_does_not_block_owned_stop(tmp_path):
    owner = owner_model(tmp_path)
    proof = owner.close(stop=True)
    assert proof['binding'] == owner.binding
    assert owner.stop_requested()
    owner.supervisor.stop.assert_called_once_with(owner.row)
    # The observation is explicitly inert and grants no real capability.
    assert proof['native_result'] == 'source-model'


@pytest.mark.parametrize('field,value', [
    ('runtest_stop_requested', True), ('runtest_launch_pending', True),
    ('runtest_persistence_error', 'OSError'), ('runtest_stop_requested', None),
    ('runtest_launch_pending', None), ('runtest_persistence_error', None),
    ('runtest_outer_closed', {'prior': 'closure'}),
])
def test_original_disposition_is_not_laundered_by_recovery_or_closure(tmp_path, field, value):
    owner = owner_model(tmp_path)
    row = original_row(owner, **{field: value})
    saved = owner.retain_registry_disposition([row], checkpoint(owner, row))
    assert saved['original_row'][field] == value
    assert not owner.original_row_allows_next_phase(recovered_session(owner))
    before = (owner.parent / 'registry-disposition.json').read_bytes()
    owner.close()
    assert (owner.parent / 'registry-disposition.json').read_bytes() == before


def test_healthy_original_disposition_is_distinct_from_recovery_safety_flag(tmp_path):
    owner = owner_model(tmp_path); row = original_row(owner)
    owner.retain_registry_disposition([row], checkpoint(owner, row))
    assert owner.original_row_allows_next_phase(recovered_session(owner))
    owner.persist_stop()
    assert not owner.original_row_allows_next_phase(recovered_session(owner))


def test_missing_original_fields_do_not_default_to_healthy(tmp_path):
    owner = owner_model(tmp_path); row = original_row(owner)
    row.pop('runtest_outer_closed')
    owner.retain_registry_disposition([row], checkpoint(owner, row))
    assert owner.registry_disposition['fields_complete'] is False
    assert not owner.original_row_allows_next_phase(recovered_session(owner))


def test_second_recovery_keeps_snapshot_and_fences_hidden_transitions(tmp_path):
    owner = owner_model(tmp_path); row = original_row(owner)
    owner.retain_registry_disposition([row], checkpoint(owner, row))
    before = (owner.parent / 'registry-disposition.json').read_bytes()
    later = copy.deepcopy(row); later['runtest_stop_requested'] = True
    owner.retain_registry_disposition([later], checkpoint(owner, later))
    assert owner.disposition_reentered
    assert (owner.parent / 'registry-disposition.json').read_bytes() == before
    assert not owner.original_row_allows_next_phase(recovered_session(owner))


@pytest.mark.parametrize('cut', ['writer', 'readback', 'checkpoint_changed'])
def test_disposition_persistence_failure_prevents_recovery(tmp_path, cut):
    owner = owner_model(tmp_path); row = original_row(owner)
    digest = checkpoint(owner, row)
    if cut == 'writer':
        owner.h.put = Mock(side_effect=OSError('writer'))
    elif cut == 'readback':
        owner.h.read = Mock(side_effect=OSError('readback'))
    else:
        (owner.control / 'profile/processes.json').write_bytes(b'changed')
    with pytest.raises((OSError, RuntimeError)):
        owner.retain_registry_disposition([row], digest)
    assert owner.registry_disposition is None


@pytest.mark.parametrize('mutation', ['clock', 'phase', 'session', 'binding'])
def test_foreign_disposition_refuses(tmp_path, mutation):
    owner = owner_model(tmp_path); row = original_row(owner)
    owner.retain_registry_disposition([row], checkpoint(owner, row))
    path = owner.parent / 'registry-disposition.json'
    record = json.loads(path.read_bytes())
    if mutation == 'clock': record['identity']['original_clock_sha256'] = 'f'*64
    elif mutation == 'phase': record['identity']['phase'] = 'foreign'
    elif mutation == 'session': record['original_row']['session_id'] = 'foreign'
    else: record['original_row']['runtest_outer'] = {'foreign': True}
    path.chmod(0o600); path.write_text(json.dumps(record)); path.chmod(0o400)
    with pytest.raises(RuntimeError, match='DISPOSITION_READBACK_FAILED'):
        owner.retain_registry_disposition([row], checkpoint(owner, row))


def test_exact_caller_persists_before_recovery_and_checks_before_release():
    tree = ast.parse((ROOT / 'owner-helper/owner_observer.py').read_bytes())
    runtime = next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == 'runtime_command')
    calls = [(n.lineno, n.func.attr) for n in ast.walk(runtime)
             if isinstance(n, ast.Call) and isinstance(n.func, ast.Attribute)]
    assert next(line for line, name in calls if name == 'retain_registry_disposition') < \
           next(line for line, name in calls if name == 'recover_from_checkpoint')
    finish = next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == 'finish_runtime_file')
    calls = [(n.lineno, n.func.attr) for n in ast.walk(finish)
             if isinstance(n, ast.Call) and isinstance(n.func, ast.Attribute)]
    assert next(line for line, name in calls if name == 'original_row_allows_next_phase') < \
           next(line for line, name in calls if name == 'finish_runtest')
    decision = next(n for n in ast.walk(finish) if isinstance(n, ast.Assign) and
                    any(isinstance(t, ast.Name) and t.id == 'next_phase' for t in n.targets))
    assert {'original_allows_next', 'drain_allows_next', 'inner_complete'} <= \
           {n.id for n in ast.walk(decision.value) if isinstance(n, ast.Name)}


def finish_model(owner, row, *, drain_error=False, mutate=None, mutate_pidfd=False):
    """Full exact finish caller with declared inert custody, never live proof."""
    digest = checkpoint(owner, row)
    owner.retain_registry_disposition([row], digest)
    session = recovered_session(owner, pid=811, host_start_time=33,
        runtest_pgid=811, runtest_sid=811, runtest_boot_id='source-model-boot',
        runtest_deadline=1800, runtest_unit='f000', owner_task_id='original-model',
        runtest_custody=True, detached=True)
    parent = owner.parent
    put = owner.h.put
    put(parent / 'backend-result.json', {'tool_result': {'exit_code': 0}})
    put(parent / 'backend-start.json', {'source_model': True})
    observation = dict(identity={'model': 'original-phase'},
        backend_result_sha256=owner.h.sha(parent / 'backend-result.json'),
        backend_start_sha256=owner.h.sha(parent / 'backend-start.json'))
    observation.update({k: True for k in ('backend_reaped', 'exact_known_pids_absent',
        'exact_group_scan_quiescent', 'fresh_lock_acquire_unlock_close_returned',
        'original_deadline_cessation_evidenced')})
    put(parent / 'cessation.json', {'observation': observation, 'next_phase_admissible': True,
                                  'status': 'RUNTEST_REPORTED'})
    put(parent / 'runtest-custody.json', {'session_id': session.id,
        'native_outer': owner.binding, 'pid': 811, 'start_ticks': 33,
        'pgid': 811, 'sid': 811, 'boot_id': 'source-model-boot', 'original_deadline': 1800,
        'unit_id': 'f000', 'owner_task_id': 'original-model'})
    put(parent / 'runtest-drain.json', {'session_id': session.id, 'reaped': True, 'exit_code': 0,
        'primary_error': None, 'cleanup_errors': [],
        'registry_persistence_error': 'OSError' if drain_error else '', 'registry_pidfd_error': ''})
    proof = {'binding': owner.binding, 'cessation_before_original_deadline': True,
             'cessation_upper_monotonic': 10, 'source_model_not_actual': True}
    owner.closure_for = Mock(return_value=proof)
    registry = SimpleNamespace(_lock=threading.Lock(), _running={session.id: session})
    def finish(s, **kwargs):
        assert s is session and kwargs['native_owner'] is owner
        s.runtest_stop_requested = True
        return proof
    registry.finish_runtest = Mock(side_effect=finish)
    namespace = dict(require=lambda ok, why: None if ok else (_ for _ in ()).throw(ValueError(why)),
        read=owner.h.read, put=put, private=owner.h.private, sha=owner.h.sha,
        phases_for=lambda original: [owner.phase],
        expected_identity=lambda clock: {'model': clock['model']})
    path = ROOT / 'owner-helper/owner_observer.py'
    tree = ast.parse(path.read_bytes())
    body = next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == 'finish_runtime_file')
    if mutate_pidfd:
        predicate = next(n.value for n in ast.walk(body) if isinstance(n, ast.Assign) and
            any(isinstance(t, ast.Name) and t.id == 'drain_allows_next' for t in n.targets) and
            isinstance(n.value, ast.BoolOp))
        before = len(predicate.values)
        predicate.values = [n for n in predicate.values
                            if ast.unparse(n) != "drain['registry_pidfd_error'] == ''"]
        assert len(predicate.values) == before - 1
    if mutate:
        decision = next(n for n in ast.walk(body) if isinstance(n, ast.Assign) and
            any(isinstance(t, ast.Name) and t.id == 'next_phase' for t in n.targets))
        before = len(decision.value.values)
        decision.value.values = [n for n in decision.value.values
                                 if not (isinstance(n, ast.Name) and n.id == mutate)]
        assert len(decision.value.values) == before - 1
    exec(compile(ast.Module(body=[body], type_ignores=[]), str(path), 'exec'), namespace)
    return lambda: namespace['finish_runtime_file'](owner.out, owner.index, registry, native_owner=owner), registry


@pytest.mark.parametrize('cut', ['healthy', 'cancelled', 'pending', 'persistence',
                                'phase_stop', 'reentered', 'drain_error'])
def test_full_finish_caller_preserves_result_but_never_launders_stopped_intent(tmp_path, cut):
    owner = owner_model(tmp_path)
    changes = {'cancelled': {'runtest_stop_requested': True},
               'pending': {'runtest_launch_pending': True},
               'persistence': {'runtest_persistence_error': 'OSError'}}.get(cut, {})
    finish, registry = finish_model(owner, original_row(owner, **changes), drain_error=cut == 'drain_error')
    if cut == 'phase_stop': owner.persist_stop()
    if cut == 'reentered': owner.disposition_reentered = True
    before = (owner.parent / 'registry-disposition.json').read_bytes()
    finish()
    registry.finish_runtest.assert_called_once()
    release = owner.h.read(owner.parent / 'registry-release.json')
    assert release['next_phase_admissible'] is (cut == 'healthy')
    assert release['exit_code'] == 0 and release['runtime_or_product_accepted'] is False
    assert (owner.parent / 'registry-disposition.json').read_bytes() == before
    assert release['original_registry_disposition_sha256'] == hashlib.sha256(before).hexdigest()


@pytest.mark.parametrize('term', ['original_allows_next', 'drain_allows_next'])
def test_full_finish_models_are_sensitive_to_each_specific_fence(tmp_path, term):
    owner = owner_model(tmp_path)
    row = original_row(owner, runtest_stop_requested=term == 'original_allows_next')
    finish, registry = finish_model(owner, row, drain_error=term == 'drain_allows_next', mutate=term)
    finish()
    # Removing ONLY the selected fence exposes the unsafe admission that the
    # corresponding unchanged healthy/negative pair rejects above.
    assert owner.h.read(owner.parent / 'registry-release.json')['next_phase_admissible'] is True


@pytest.mark.parametrize('cut', ['healthy', 'phase_stop', 'stop_row', 'pending_row',
                                'error_row', 'snapshot_hash', 'reentered'])
def test_next_prepare_consumes_exact_retained_disposition_before_new_allocation(tmp_path, cut):
    owner = owner_model(tmp_path)
    finish, registry = finish_model(owner, original_row(owner))
    finish()
    path = owner.parent / 'registry-disposition.json'
    release_path = owner.parent / 'registry-release.json'
    if cut == 'phase_stop':
        owner.persist_stop()
    elif cut in ('stop_row', 'pending_row', 'error_row', 'snapshot_hash'):
        data = owner.h.read(path)
        key, value = {'stop_row': ('runtest_stop_requested', True),
                      'pending_row': ('runtest_launch_pending', True),
                      'error_row': ('runtest_persistence_error', 'OSError'),
                      'snapshot_hash': ('runtest_persistence_error', 'changed')}[cut]
        data['original_row'][key] = value
        path.chmod(0o600); path.write_text(json.dumps(data)); path.chmod(0o400)
        if cut != 'snapshot_hash':
            # Keep the hash correct so only the relevant intent predicate can
            # distinguish this negative from the healthy positive.
            released = owner.h.read(release_path)
            released['original_registry_disposition_sha256'] = owner.h.sha(path)
            release_path.chmod(0o600); release_path.write_text(json.dumps(released)); release_path.chmod(0o400)
    elif cut == 'reentered':
        released = owner.h.read(release_path); released['recovery_reentered'] = True
        release_path.chmod(0o600); release_path.write_text(json.dumps(released)); release_path.chmod(0o400)
    admission = owner.out / 'model-admission.json'; owner.h.put(admission, {'source_model': True})
    pre = dict(owner.pre, admission=str(admission), admission_sha256=owner.h.sha(admission), reserve_seconds=1)
    original = dict(owner.original, deadline_monotonic=1800)
    phases = [dict(owner.phase, outer=1), dict(name='next-phase', outer=1)]
    class ModelReachedPrelaunch(RuntimeError):
        pass
    def require(ok, why):
        if not ok: raise ValueError(why)
    lock = Mock(side_effect=ModelReachedPrelaunch('validated before allocation'))
    api = definitions(ROOT / 'owner-helper/owner_observer.py', {'prepare'},
        cohort=lambda out: (original, pre), phases_for=lambda clock: phases,
        require=require, read=owner.h.read, sha=owner.h.sha, private=owner.h.private,
        time=SimpleNamespace(monotonic=lambda: 10), lock_probe=lock)
    expected = ModelReachedPrelaunch if cut == 'healthy' else ValueError
    with pytest.raises(expected):
        api['prepare'](owner.out, 1)
    assert lock.call_count == (1 if cut == 'healthy' else 0)
    assert not (owner.out / 'next-phase').exists()
