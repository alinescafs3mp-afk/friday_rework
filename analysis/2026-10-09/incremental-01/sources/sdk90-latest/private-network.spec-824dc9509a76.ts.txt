/** Source-only future controls. NOT RUN; no namespace/kernel/network proof. */
import { describe, it, expect } from 'vitest'
import { prepareHostControl } from '../src/host-control.ts'
import type { HostControlPolicy } from '@deepseek-ai/dsh-sandbox'

const policy: HostControlPolicy = {
  capability: 'host-control-v1', mode: 'workspace-write', workspaceRoot: '/w/task',
  temporaryRoot: '/w/private', readOnly: ['/usr/bin', '/usr/lib'],
  readWrite: ['/w/task', '/w/private'], protectedPaths: ['<USER_HOME>'],
  requiredCapabilities: ['browser-own-cdp', 'web', 'local-chat', 'local-embedding'], statusFd: 8,
  privateNetwork: { driver: 'slirp4netns-1.3.3-frw-private-loopback', blockFd: 9,
    barrierProtocol: 'frw-success-byte-1-v1', resolverFd: 10, resolverSha256: 'a'.repeat(64), nameservers: ['192.168.12.2'] },
}

describe('explicit private network preparation, no admission', () => {
  it('preserves original capabilities and scoped mounts, adds unique barrier and private DNS', () => {
    const prepared = prepareHostControl(['/usr/bin/node', '/w/task/job.js'], policy, '/usr/bin/landlock-run')
    expect(prepared.argv).toContain('--unshare-net')
    expect(prepared.argv.slice(prepared.argv.indexOf('--block-fd'), prepared.argv.indexOf('--block-fd') + 2)).toEqual(['--block-fd', '9'])
    expect(prepared.argv.slice(prepared.argv.indexOf('--ro-bind-data'), prepared.argv.indexOf('--ro-bind-data') + 3)).toEqual(['--ro-bind-data', '10', '/etc/resolv.conf'])
    expect(prepared.argv).not.toContain('/w/private/private-resolv.conf')
    expect(prepared.network).toEqual(policy.privateNetwork)
    expect(prepared.requiredCapabilities).toEqual(policy.requiredCapabilities)
    expect(prepared.runtimeGrant).toBe(false)
  })
  it.each(['127.0.0.53', '::1', '::ffff:127.0.0.1', '10.0.2.3', 'fd00::3', 'localhost'])('refuses DNS host alias %s', address => {
    expect(() => prepareHostControl(['true'], { ...policy, privateNetwork: {
      ...policy.privateNetwork!, nameservers: [address],
    } }, '/usr/bin/landlock-run')).toThrow('EXTERNAL_DNS')
  })
  it.each([0, 2, 8, NaN])('refuses foreign/standard/status barrier FD %s', fd => {
    expect(() => prepareHostControl(['true'], { ...policy, privateNetwork: {
      ...policy.privateNetwork!, blockFd: fd,
    } }, '/usr/bin/landlock-run')).toThrow('PRIVATE_NETWORK')
  })
  it('preserves old preparation without granting network authority', () => {
    const prepared = prepareHostControl(['true'], { ...policy, privateNetwork: undefined }, '/usr/bin/landlock-run')
    expect(prepared.argv).not.toContain('--unshare-net')
    expect(prepared.missing).toContain('REACHABLE_LOCAL_NETWORK_CONTROL_UNMEDIATED')
    expect(prepared.runtimeGrant).toBe(false)
  })
})
