"""Inert source controls, NOT_RUN. Never status/namespace/cgroup/kernel proof."""
import ast
import copy
from pathlib import Path
from types import SimpleNamespace as NS

import pytest

SOURCE = Path(__file__).resolve().parents[1] / 'owner-helper/owner_observer.py'


def functions(*names, **values):
    tree = ast.parse(SOURCE.read_text())
    body = [n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name in names]
    assert len(body) == len(names)
    def require(ok, reason):
        if not ok:
            raise ValueError(reason)
    env = dict(require=require, **values)
    exec(compile(ast.Module(body=body, type_ignores=[]), str(SOURCE), 'exec'), env)
    return env


def receipt():
    before = {'argv': ['/usr/bin/env', '-i', '/usr/bin/gnutimeout', '5', 'guard'],
              'phase_clock_sha256': 'b' * 64, 'original_clock_sha256': 'a' * 64}
    clock = dict(boot_id='boot', deadline_monotonic=100, runtest_file='f000', assignment='original')
    outer = dict(control_group='/original/unit', cgroup_inode=[2, 9], boot_id='boot',
                 original_clock_sha256='a' * 64, phase_clock_sha256='b' * 64, original_deadline=100)
    custody = dict(pid=811, pgid=811, sid=811, start_ticks=37, boot_id='boot',
                   native_outer=outer, unit_id='f000', owner_task_id='original',
                   original_deadline=100, phase_clock_sha256='b' * 64)
    fields = ['S', '800', '811', '811'] + ['0'] * 15 + ['37']
    monitor = dict(pid=811, start_ticks=37, pgid=811, sid=811, boot_id='boot',
                   pid_namespace=[4, 12], cgroup=['0::/original/unit'], cgroup_inode=[2, 9],
                   native_outer=outer)
    start = dict(argv=['/pinned/bwrap', '--', '/pinned/scoped-launcher', '--'] + before['argv'],
                 semantic_argv=before['argv'][:], pid=811, stat='811 (host monitor) ' + ' '.join(fields),
                 boot_id='boot', host_monitor=monitor, owner_pid_namespace=[4, 12],
                 command_binding=dict(prelaunch_sha256='c' * 64,
                     phase_clock_sha256='b' * 64, original_clock_sha256='a' * 64),
                 sandbox_status=dict(state='UNBOUND', sandbox_child=None, timeout_host=None,
                                     guard_host=None, status_pipe=None))
    return start, before, clock, custody


def test_semantic_and_monitor_join_does_not_claim_timeout_identity():
    start, before, clock, custody = receipt()
    api = functions('runtest_start_binding')
    value = api['runtest_start_binding'](start, before, clock, custody, 'c' * 64)
    assert value['semantic_argv'] == before['argv'] and value['exec_argv'] != before['argv']
    assert value['host_monitor']['pid'] == custody['pid'] == 811
    assert value['sandbox_status']['timeout_host'] is None
    assert value['sandbox_status']['sandbox_child'] is None


@pytest.mark.parametrize('cut', ['semantic', 'wrapper', 'source', 'clock', 'birth',
    'monitor_pid', 'sid', 'cgroup', 'cgroup_inode', 'namespace', 'outer_clock',
    'status_promoted', 'namespace_pid_as_host'])
def test_wrong_command_or_host_custody_refuses(cut):
    start, before, clock, custody = copy.deepcopy(receipt())
    if cut == 'semantic': start['semantic_argv'] = ['foreign']
    elif cut == 'wrapper': start['argv'][-1] = 'foreign'
    elif cut == 'source': start['command_binding']['prelaunch_sha256'] = 'x' * 64
    elif cut == 'clock': start['command_binding']['phase_clock_sha256'] = 'x' * 64
    elif cut == 'birth': start['host_monitor']['start_ticks'] = 99
    elif cut == 'monitor_pid': start['host_monitor']['pid'] = 999
    elif cut == 'sid': custody['sid'] = 999
    elif cut == 'cgroup': start['host_monitor']['cgroup'] = ['0::/foreign']
    elif cut == 'cgroup_inode': start['host_monitor']['cgroup_inode'] = [2, 10]
    elif cut == 'namespace': start['host_monitor']['pid_namespace'] = [4, 99]
    elif cut == 'outer_clock': custody['native_outer']['phase_clock_sha256'] = 'x' * 64
    elif cut == 'status_promoted': start['sandbox_status']['state'] = 'QUALIFIED'
    elif cut == 'namespace_pid_as_host':
        start['sandbox_status']['timeout_host'] = {'pid': 1, 'start_ticks': 37}
    with pytest.raises(ValueError):
        functions('runtest_start_binding')['runtest_start_binding'](start, before, clock, custody, 'c' * 64)


def namespace_path(path):
    assert path == '/proc/self/ns/pid'
    return NS(stat=lambda: NS(st_dev=4, st_ino=12))


def test_plain_host_domain_keeps_legacy_observer_path():
    api = functions('require_host_observable_contour', Path=namespace_path)
    api['require_host_observable_contour']({'pid_namespace': [4, 12]})


@pytest.mark.parametrize('contour', [{'pid_namespace': [4, 99]}, {},
    {'pid_namespace': [4, 12], 'state': 'QUALIFIED', 'status_pipe': 'pretend'}])
def test_wrapped_status_never_promoted_and_host_scan_never_reads_inner_pid(contour):
    def forbidden(*args):
        raise AssertionError('must refuse before host proc lookup')
    api = functions('require_host_observable_contour', Path=forbidden)
    with pytest.raises(ValueError, match='SANDBOX_STATUS_CUSTODY_UNQUALIFIED'):
        api['require_host_observable_contour'](contour, wrapped=True)


def test_namespace_local_group_refused_before_host_enumeration():
    api = functions('require_host_observable_contour', 'scan_group', Path=namespace_path)
    with pytest.raises(ValueError, match='PID_NAMESPACE_IS_NOT_HOST'):
        api['scan_group']({'pid_namespace': [4, 99], 'timeout': {'pid': 1}})


def test_all_existing_host_consumers_follow_explicit_wrapped_refusal():
    tree = ast.parse(SOURCE.read_text())
    observe = next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == 'observe')
    joined = next(n for n in ast.walk(observe) if isinstance(n, ast.Call) and
                  isinstance(n.func, ast.Name) and n.func.id == 'runtest_start_binding')
    refused = next(n for n in ast.walk(observe) if isinstance(n, ast.Call) and
                  isinstance(n.func, ast.Name) and n.func.id == 'require_host_observable_contour' and n.keywords)
    consumers = [n for n in ast.walk(observe) if isinstance(n, ast.Call) and
                 isinstance(n.func, ast.Name) and n.func.id in ('scan_group', 'runtest_check', 'collection_check')]
    assert joined.lineno < refused.lineno < min(n.lineno for n in consumers)
    finish = next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == 'finish_runtime_file')
    predicate = next(n.value for n in ast.walk(finish) if isinstance(n, ast.Assign) and
                     any(isinstance(t, ast.Name) and t.id == 'inner_complete' for t in n.targets))
    assert ast.unparse(predicate).startswith('not wrapped_status_unqualified and ')


@pytest.mark.parametrize('field', ['host_monitor', 'sandbox_status'])
def test_wrapped_finish_retains_outer_cleanup_without_next_phase(tmp_path, field):
    # Reuse the retained complete finish fixture and original positive model.
    # Alter only the wrapped-status marker; no host/kernel effect is simulated.
    import test_native_boundary_source as boundary
    import json
    owner = boundary.owner_model(tmp_path)
    finish, registry = boundary.finish_model(owner, boundary.original_row(owner))
    start = owner.parent / 'backend-start.json'
    start.chmod(0o600)
    start.write_text(json.dumps({'source_model': True, field: {'state': 'QUALIFIED'}}))
    start.chmod(0o400)
    done = owner.parent / 'cessation.json'
    record = owner.h.read(done)
    record['observation']['backend_start_sha256'] = owner.h.sha(start)
    done.chmod(0o600); done.write_text(json.dumps(record)); done.chmod(0o400)
    drain = (owner.parent / 'runtest-drain.json').read_bytes()
    finish()
    registry.finish_runtest.assert_called_once()
    release = owner.h.read(owner.parent / 'registry-release.json')
    assert release['next_phase_admissible'] is False
    assert release['runtime_or_product_accepted'] is False
    assert (owner.parent / 'runtest-drain.json').read_bytes() == drain
