/** Source tests only. Kernel and full native qualification are separate. */
import { Context } from '@deepseek-ai/cordis'
import { describe, expect, it, vi } from 'vitest'
import type { HostControlPolicy } from '@deepseek-ai/dsh-sandbox'
import { LocalSandboxProvider } from '../src/index.ts'
import { prepareHostControl, validateHostControlPolicy } from '../src/host-control.ts'

const policy: HostControlPolicy = {
  capability: 'host-control-v1', mode: 'workspace-write',
  workspaceRoot: '/work/fixture', temporaryRoot: '/tmp/donor-private',
  readOnly: ['/usr', '/opt/pinned-runtime', '/work/corpus'],
  readWrite: ['/work/fixture', '/tmp/donor-private'],
  protectedPaths: ['<USER_HOME>/.codex', '<USER_HOME>/.ssh', '/work/credentials'],
  requiredCapabilities: ['native-tools', 'local-inference', 'network', 'browser', 'web'],
  statusFd: 7,
}

describe('host-control source profile', () => {
  it('composes an empty root without a host root, bus, credentials or private network', () => {
    const prepared = prepareHostControl(['/usr/bin/env', '-i', '/usr/bin/true'], policy, '/opt/pinned-runtime/landlock-run')
    expect(prepared.state).toBe('UNQUALIFIED_SOURCE_PREPARATION')
    expect(prepared.runtimeGrant).toBe(false)
    expect(prepared.semanticArgv).toEqual(['/usr/bin/env', '-i', '/usr/bin/true'])
    expect(prepared.argv.slice(0, 2)).toEqual(['/usr/bin/bwrap', '--unshare-user'])
    expect(prepared.argv).not.toContain('--unshare-net')
    expect(prepared.argv).not.toContain('/')
    expect(prepared.argv).toContain('--disable-userns')
    expect(prepared.argv).toContain('--assert-userns-disabled')
    expect(prepared.argv).toContain('--scope-ipc')
    expect(prepared.argv.slice(prepared.argv.indexOf('--json-status-fd'), prepared.argv.indexOf('--json-status-fd') + 2)).toEqual(['--json-status-fd', '7'])
    expect(prepared.requiredCapabilities).toEqual(policy.requiredCapabilities)
    expect(prepared.missing).toContain('REACHABLE_LOCAL_NETWORK_CONTROL_UNMEDIATED')
  })

  it('binds temporary roots after the fresh tmpfs and copies caller arrays', () => {
    const prepared = prepareHostControl(['/usr/bin/true'], policy, '/opt/pinned-runtime/landlock-run')
    expect(prepared.argv.lastIndexOf('/tmp/donor-private')).toBeGreaterThan(prepared.argv.indexOf('--tmpfs'))
    prepared.semanticArgv.push('changed')
    prepared.requiredCapabilities.push('changed')
    expect(policy.requiredCapabilities).not.toContain('changed')
  })

  it.each(['/', '/run', '/sys', '/proc', '/dev', '/home', '/tmp', '/etc', '/var', '<USER_RUNTIME>', '/sys/fs/cgroup'])('rejects broad/control read mount %s', root => {
    expect(() => validateHostControlPolicy({ ...policy, readOnly: [root] })).toThrow()
  })

  it.each(['/work/credentials', '/work', '<USER_HOME>/.ssh/id_ed25519'])('rejects credential ancestors and descendants %s', root => {
    expect(() => validateHostControlPolicy({ ...policy, readOnly: [root] })).toThrow(/CREDENTIAL/)
  })

  it.each(['/outside', '/work/corpus', '/tmp'])('rejects write outside exact roots %s', root => {
    expect(() => validateHostControlPolicy({ ...policy, readWrite: [root] })).toThrow()
  })

  it.each(['relative', '/work/../outside', '/work/file\0tail'])('rejects noncanonical paths %s', root => {
    expect(() => validateHostControlPolicy({ ...policy, workspaceRoot: root })).toThrow()
  })

  it.each([0, 1, 2, -1, 3.5, Infinity])('rejects unowned/invalid status fd %s', statusFd => {
    expect(() => validateHostControlPolicy({ ...policy, statusFd })).toThrow(/DESCRIPTOR/)
  })

  it('refuses read-only write grants and missing credential/capability census', () => {
    expect(() => validateHostControlPolicy({ ...policy, mode: 'read-only' })).toThrow()
    expect(() => validateHostControlPolicy({ ...policy, protectedPaths: [] })).toThrow()
    expect(() => validateHostControlPolicy({ ...policy, requiredCapabilities: [] })).toThrow()
  })
})

describe('host-control admission through actual provider', () => {
  it('refuses despite old file-only probe full and does not call a runner probe', async () => {
    const ctx = new Context()
    const fiber = await ctx.plugin(LocalSandboxProvider, {})
    try {
      const provider = ctx.sandbox as LocalSandboxProvider
      const oldProbe = vi.fn(() => true)
      provider.internals = { probeBwrap: oldProbe }
      await expect(provider.confineHostControl(['/usr/bin/true'], policy)).rejects.toThrow('BUILTIN_LINUX_ROUTE_REQUIRED')
      expect(oldProbe).not.toHaveBeenCalled()
      provider.internals = {}
      await expect(provider.confineHostControl(['/usr/bin/true'], policy)).rejects.toThrow(
        process.platform === 'linux' ? 'REACHABLE_LOCAL_NETWORK_CONTROL_UNMEDIATED' : 'BUILTIN_LINUX_ROUTE_REQUIRED',
      )
    } finally {
      await fiber.dispose()
    }
  })

  it('cannot use custom runner assertion or an aborted call', async () => {
    const ctx = new Context()
    const fiber = await ctx.plugin(LocalSandboxProvider, { runnerCommand: ['/inert/custom'], runnerFailureSignatures: ['inert:'] })
    try {
      const provider = ctx.sandbox as LocalSandboxProvider
      await expect(provider.confineHostControl(['/usr/bin/true'], policy)).rejects.toThrow('BUILTIN_LINUX_ROUTE_REQUIRED')
      const controller = new AbortController()
      controller.abort()
      expect(() => provider.prepareHostControl(['/usr/bin/true'], policy, controller.signal)).toThrow()
    } finally {
      await fiber.dispose()
    }
  })
})
