"""Future source checks; never kernel evidence. NOT_RUN in this assignment."""
import ast
from pathlib import Path
import runpy

import pytest

ROOT = Path(__file__).resolve().parents[1]
API = ROOT / 'owner-helper/host_control.py'
IDENTITY = {'original_clock_sha256': 'a' * 64, 'phase_clock_sha256': 'b' * 64,
            'phase': 'exact-phase', 'unit': 'exact-unit.service',
            'boot_id': 'exact-boot', 'original_deadline': 1234.5}


def test_request_keeps_semantic_argv_original_clock_and_complete_capabilities():
    api = runpy.run_path(str(API))
    argv = ['/usr/bin/env', '-i', '/pinned/python', '-I', '-B', 'guard.py']
    capabilities = ['native-tools', 'local-inference', 'network', 'browser', 'web']
    request = api['preparation_request'](argv, phase_identity=IDENTITY, required_capabilities=capabilities)
    assert request['semanticArgv'] == argv and request['phase'] == IDENTITY
    assert request['requiredCapabilities'] == capabilities and request['runtimeGrant'] is False
    request['semanticArgv'].append('changed')
    assert len(argv) == 6


@pytest.mark.parametrize('value', [[], ['bad\0argv'], 'shell text', [123]])
def test_bad_semantic_argv_refused(value):
    api = runpy.run_path(str(API))
    with pytest.raises(api['HostControlRefusal']):
        api['preparation_request'](value, phase_identity=IDENTITY, required_capabilities=['native-tools'])


def test_missing_original_identity_and_capability_census_refused():
    api = runpy.run_path(str(API))
    with pytest.raises(api['HostControlRefusal']):
        api['preparation_request'](['exact'], phase_identity={}, required_capabilities=['native-tools'])
    with pytest.raises(api['HostControlRefusal']):
        api['preparation_request'](['exact'], phase_identity=IDENTITY, required_capabilities=[])


@pytest.mark.parametrize('mutation', [
    {}, {'runtimeGrant': True}, {'state': 'QUALIFIED'}, {'enforcement': 'full', 'capability': 'file-only'},
    {'semanticArgv': ['different']}, {'capability': 'custom-runner'},
])
def test_preparation_and_promoted_results_always_refuse(mutation):
    api = runpy.run_path(str(API))
    preparation = {'capability': 'host-control-v1', 'state': 'UNQUALIFIED_SOURCE_PREPARATION',
                   'runtimeGrant': False, 'semanticArgv': ['exact']}
    preparation.update(mutation)
    with pytest.raises(api['HostControlRefusal']):
        api['refuse_unqualified_preparation'](preparation, ['exact'])


def test_original_fileowner_refuses_before_any_proposed_wrapper_effect():
    tree = ast.parse((ROOT / 'owner-helper/runtest_outer.py').read_text())
    cls = next(n for n in tree.body if isinstance(n, ast.ClassDef) and n.name == 'FileOwner')
    method = next(n for n in cls.body if isinstance(n, ast.FunctionDef) and n.name == 'confined_donor_argv')
    # The first operative statement is the original admission, before runpy or IO.
    first = method.body[1]
    assert isinstance(first, ast.Expr) and isinstance(first.value, ast.Call)
    assert first.value.func.attr == 'require_supported_containment'
    gate = next(n for n in cls.body if isinstance(n, ast.FunctionDef) and n.name == 'require_supported_containment')
    assert any(isinstance(n, ast.Raise) for n in ast.walk(gate))


def test_existing_spawn_retains_real_registry_and_semantic_wrapper_binding():
    tree = ast.parse((ROOT / 'owner-helper/owner_observer.py').read_text())
    method = next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == 'run_donor_file')
    spawn = next(n for n in method.body if isinstance(n, ast.FunctionDef) and n.name == 'spawn')
    calls = [n for n in ast.walk(spawn) if isinstance(n, ast.Call) and isinstance(n.func, ast.Attribute)]
    assert any(n.func.attr == 'prepare_donor_status' for n in calls)
    assert any(n.func.attr == 'spawn' and n.func.value.id == 'status' for n in calls)
    adapter = ast.parse((ROOT / 'owner-helper/native_status.py').read_text())
    actual = next(n for n in ast.walk(adapter) if isinstance(n, ast.Call) and isinstance(n.func, ast.Attribute) and n.func.attr == 'spawn_runtest')
    assert ast.unparse(actual.args[1]) == 'self.argv'
    assert any(n.arg == 'native_owner' for n in actual.keywords)
    assert not any(n.func.attr in ('wait', 'communicate', 'poll') for n in calls)
