"""Tiny fake-FD preparation called ONLY inside checked timeout-bounded guard.
NOT_RUN. No standalone executable route, inheritance mutation, clock write or exec.
Descriptors remain in this same guard process through pytest; no real keys.
"""
import hashlib, json, os, pathlib, stat, time

DESCRIPTOR_CASES = frozenset('inherited_foreign_descriptor-' + v for v in
                           ('fdopen', 'read', 'fdopen-write', 'write', 'truncate'))
FAKE = b'SYNTHETIC_GUARD_CANARY_ONLY\n'


def prepare(root, clock, clock_sha, identity):
    if (clock.get('selection') != '@canary' or clock.get('canary') not in DESCRIPTOR_CASES or
        clock.get('synthetic_fixture_parent') != str(root.parent)):
        raise SystemExit('EXACT_FAKE_DESCRIPTOR_CANARY_REQUIRED')
    if (pathlib.Path('/proc/sys/kernel/random/boot_id').read_text().strip() != clock['boot_id'] or
        clock['deadline_monotonic'] - time.monotonic() <= 0 or
        hashlib.sha256((root / 'clock.json').read_bytes()).hexdigest() != clock_sha):
        raise SystemExit('BOOT_DEADLINE_OR_CLOCK_DRIFT')
    checked = []
    for key, leaf, flags in (('synthetic_credential_fd', 'auth.json', os.O_RDONLY),
                             ('synthetic_write_fd', 'synthetic-outside.txt', os.O_WRONLY)):
        path = root.parent / leaf
        st = path.lstat()
        if (not stat.S_ISREG(st.st_mode) or stat.S_IMODE(st.st_mode) != 0o600 or
            st.st_uid != os.getuid() or st.st_nlink != 1 or path.read_bytes() != FAKE):
            raise SystemExit('NOT_EXACT_OWNED_SYNTHETIC_FIXTURE')
        checked.append((key, path, flags, st))
    # Validation of BOTH fake files precedes the first descriptor open.
    if clock['deadline_monotonic'] - time.monotonic() <= 0:
        raise SystemExit('ORIGINAL_ASSIGNMENT_EXPIRED')
    fds = {}; identities = {}
    try:
        for key, path, flags, prior in checked:
            fd = os.open(path, flags | os.O_NOFOLLOW | os.O_CLOEXEC)
            fds[key] = fd
            st = os.fstat(fd)
            if (st.st_dev, st.st_ino, st.st_mode, st.st_uid, st.st_nlink) != (
                prior.st_dev, prior.st_ino, prior.st_mode, prior.st_uid, prior.st_nlink):
                raise SystemExit('SYNTHETIC_DESCRIPTOR_IDENTITY_DRIFT')
            identities[key] = {'fd': fd, 'dev': st.st_dev, 'ino': st.st_ino,
                               'mode': st.st_mode, 'source_sha256': hashlib.sha256(FAKE).hexdigest()}
        # Separate descriptor artifact; original phase clock remains byte-exact.
        payload = {'identity': identity, 'descriptors': identities, 'phase_clock_sha256': clock_sha}
        fd = os.open(root / 'synthetic-fd-launch.json', os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600)
        with os.fdopen(fd, 'w') as f:
            f.write(json.dumps(payload, indent=2) + '\n'); f.flush(); os.fsync(f.fileno())
        if hashlib.sha256((root / 'clock.json').read_bytes()).hexdigest() != clock_sha:
            raise SystemExit('PHASE_CLOCK_DRIFT')
        return fds
    except BaseException:
        for fd in fds.values(): os.close(fd)
        raise


if __name__ == '__main__':
    raise SystemExit('STANDALONE_DESCRIPTOR_BOOTSTRAP_FORBIDDEN')
