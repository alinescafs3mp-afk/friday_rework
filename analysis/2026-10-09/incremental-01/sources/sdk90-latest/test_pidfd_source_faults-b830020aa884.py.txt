class Lock:
    """Detect accidentally unguarded descriptor signaling/close without threads."""
    held = False

    def __enter__(self):
        assert not self.held
        self.held = True

    def __exit__(self, *unused):
        self.held = False


def forbidden(*args, **kwargs):

@pytest.mark.parametrize('fault', ['ignored', 'handler', 'threads', 'not_main', 'trace',
                                  'native_handler', 'missing_open', 'missing_signal',
                                  'wrong_python', 'reset', 'readback'])
def test_preflight_failure_is_no_launch_no_retry(fault):
    h = harness()
    if fault == 'ignored': h.handlers[17] = 1
    if fault == 'handler': h.handlers[17] = lambda: None
    if fault == 'threads': h.state.tasks = 2
    if fault == 'not_main': h.threading.current_thread = lambda: object()
    if fault == 'trace': h.sys.gettrace = lambda: object()
    if fault == 'native_handler': h.state.caught |= 1 << 16
    if fault == 'missing_open': h.os.pidfd_open = None
    if fault == 'missing_signal': h.signal.pidfd_send_signal = None
    if fault == 'wrong_python': h.sys.version_info = (3, 13, 0)
    if fault == 'reset': h.signal.signal = lambda *a: (_ for _ in ()).throw(OSError('reset'))
    if fault == 'readback': h.state.ignored = 1 << 16
    with pytest.raises((RuntimeError, OSError)): h.spawn()
    assert h.session.runtest_stop_requested and not h.session.runtest_launch_pending
    assert h.session.process is None and 'Popen' not in h.events
    with pytest.raises(RuntimeError): h.spawn()
    assert 'Popen' not in h.events

@pytest.mark.parametrize('close_first', [False, True])

@pytest.mark.parametrize('cut', ['return', 'after_spawn', 'timeout', 'capture_write'])
def test_actual_donor_finally_keeps_one_handle_one_waiter_and_closes_fd(cut):
    """Exercise the actual unmodified donor primitive with inert OS doubles."""
    import subprocess
    h = harness()
    donor_tree = ast.parse((ROOT / 'donor/scripts/run_tests_parallel.py').read_text())
    primitive = copy.deepcopy(next(n for n in donor_tree.body
        if isinstance(n, ast.FunctionDef) and n.name == '_run_one_file_once'))
    module = ast.Module(body=[ast.ImportFrom(module='__future__', names=[ast.alias(name='annotations')], level=0),
                            primitive], type_ignores=[])
    donor_env = dict(sys=NS(executable='/pinned', platform='linux'),
                     os=NS(environ={}, getpgid=lambda pid: pid),
                     time=NS(monotonic=lambda: 10.),
                     subprocess=NS(PIPE=-1, STDOUT=-2, Popen=forbidden,
                                   TimeoutExpired=subprocess.TimeoutExpired),
                     _kill_tree=forbidden,
                     _parse_pytest_summary=lambda output: {'passed': 1},
                     _describe_interpreter_crash=lambda *a: None,
                     _rmtree_force=lambda root: h.events.append('temp_remove'))
    exec(compile(ast.fix_missing_locations(module), '<actual-donor-primitive>', 'exec'), donor_env)
    captures = []
    def spawn(*a, **k):
        proc = h.spawn()
        if cut == 'return': raise RuntimeError('callback return cut')
        return proc
    def attach(proc):
        h.registry.attach_runtest(h.session, proc)
        if cut == 'after_spawn': raise OSError('diagnostic cut')
    def communicate(**kwargs):
        h.events.append('communicate')
        if cut == 'timeout' and h.events.count('communicate') == 1:
            raise subprocess.TimeoutExpired('exact', 1, output=b'partial')
        h.proc.returncode = -9 if cut == 'timeout' else 0
        return ('retained output', None)
    h.proc.communicate = communicate
    def stop(proc, pgid):
        assert proc is h.session.process and pgid == h.session.runtest_pgid
        if proc.returncode is None: h.registry.stop_runtest(h.session)
    def capture(proc, output, error, cuts):
        captures.append((proc, output))
        h.registry.release_runtest_pidfd(h.session, proc)
        if cut == 'capture_write': raise OSError('receipt write cut')
    def settle(proc, pgid, root):
        assert proc is h.session.process and proc.returncode is not None
    custody = dict(spawn=spawn, after_spawn=attach,
                   retained_handle=lambda: h.session.process,
                   retained_pgid=lambda: h.session.runtest_pgid,
                   stop=stop, capture=capture, settle=settle, remaining=lambda: 1.,
                   terminal=lambda rc: None, temporary_root=lambda: '/retained-temp')
    def run():
        return donor_env['_run_one_file_once'](Path('/fixture/test.py'), [], Path('/fixture'), 1., custody=custody)
    if cut == 'timeout':
        assert run()[1] == 124
    else:
        with pytest.raises((RuntimeError, OSError)): run()
    assert captures == [(h.proc, 'retained output')]
    assert h.events.count('Popen') == 1
    assert h.events.count('communicate') == (2 if cut == 'timeout' else 1)
    assert h.events.count('close') == 1 and h.session._runtest_pidfd is None
    assert h.session.process is h.proc and h.session.runtest_custody
    assert h.session.runtest_outer_closed is None
    assert 'temp_remove' not in h.events
