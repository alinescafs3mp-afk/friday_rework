"""SOURCE ONLY, NOT RUN. Separate admission is required to execute these tests.

Extract only the pure receipt binder and callback method into a synthetic
namespace. Never import/initialize producer, pytest, product or guard modules.
No filesystem writes, collection, fixture execution, network or subprocess.
"""
import ast
from copy import deepcopy
from pathlib import Path
from types import SimpleNamespace
import unittest


SOURCE = Path(__file__).resolve().parent


def isolated(entry):
    tree = ast.parse((SOURCE / entry).read_bytes(), filename=entry)
    binder = next(n for n in tree.body if isinstance(n, ast.FunctionDef)
                  and n.name == 'bind_collector_receipt')
    capture = next(n for n in tree.body if isinstance(n, ast.ClassDef)
                   and n.name == 'Capture')
    callback = next(n for n in capture.body if isinstance(n, ast.FunctionDef)
                    and n.name == 'pytest_collectreport')
    namespace = {'collector_receipts': {}, 'collector_identity_owners': {},
                 'collection_reports': [], 'collection_report_callbacks': [],
                 'collect_only': True}
    exec(compile(ast.Module(body=[binder, callback], type_ignores=[]), entry,
                 'exec'), namespace)
    return namespace


def identity(path='/synthetic/tests', kind='directory', nodeid='tests'):
    return {'path': path, 'object_path': path, 'kind': kind,
            'type': '_pytest.main.Dir', 'nodeid': nodeid,
            'source_relative_path': '../tests'}


def receipt(kind='directory', outcome='passed'):
    return {'identity': identity(kind=kind),
            'parents': [identity('/synthetic/source', 'session', '')],
            'children': [identity('/synthetic/tests/test_a.py', 'module', 'a')],
            'nodeid': 'tests', 'outcome': outcome,
            'detail': None if outcome == 'passed' else 'synthetic detail',
            'callback_count': 0}


def report(row):
    return SimpleNamespace(nodeid=row['nodeid'], outcome=row['outcome'])


class CollectorReceiptSourceTests(unittest.TestCase):
    def each(self):
        for entry in ('controls.py', 'native_controls.py'):
            yield entry, isolated(entry)

    def test_seventeen_same_object_ancestor_makes_one_logical_receipt(self):
        for entry, env in self.each():
            with self.subTest(entry=entry):
                collector = object()
                row = receipt()
                reports = [report(row) for _ in range(17)]
                for made in reports:
                    env['bind_collector_receipt'](collector, made, deepcopy(row))
                self.assertEqual(env['collection_reports'], [row])
                canonical = env['collection_reports'][0]
                self.assertTrue(all(r._friday_collector_receipt is canonical
                                    for r in reports))
                self.assertIs(env['collector_receipts'][id(collector)][0], collector)
                self.assertEqual(env['collection_report_callbacks'], [])
                self.assertEqual(canonical['callback_count'], 0)

    def test_distinct_objects_with_identical_compound_identity_refuse(self):
        for entry, env in self.each():
            with self.subTest(entry=entry):
                row = receipt()
                env['bind_collector_receipt'](object(), report(row), deepcopy(row))
                with self.assertRaisesRegex(ValueError, '^DISTINCT_COLLECTOR_IDENTITY_COLLISION$'):
                    env['bind_collector_receipt'](object(), report(row), deepcopy(row))
                self.assertEqual(len(env['collection_reports']), 1)

    def test_same_display_nodeid_different_path_preserves_compound_identity(self):
        for entry, env in self.each():
            with self.subTest(entry=entry):
                first, second = receipt(), receipt()
                second['identity'] = identity('/synthetic/other', nodeid='tests')
                for row in (first, second):
                    env['bind_collector_receipt'](object(), report(row), row)
                self.assertEqual(env['collection_reports'], [first, second])

    def test_changed_same_object_snapshot_refuses_without_rewriting_original(self):
        mutations = {
            'children': lambda r: r['children'].clear(),
            'parents': lambda r: r['parents'].clear(),
            'type': lambda r: r['identity'].update(type='other.Dir'),
            'relative_path': lambda r: r['identity'].update(source_relative_path='changed'),
            'path': lambda r: r['identity'].update(path='/other'),
            'object_path': lambda r: r['identity'].update(object_path='/other'),
            'nodeid': lambda r: r['identity'].update(nodeid='changed'),
            'detail': lambda r: r.update(detail='unexpected'),
            'callback': lambda r: r.update(callback_count=1),
            'outcome': lambda r: r.update(outcome='failed'),
        }
        for entry in ('controls.py', 'native_controls.py'):
            for name, mutate in mutations.items():
                with self.subTest(entry=entry, mutation=name):
                    env = isolated(entry)
                    collector = object()
                    original = receipt()
                    env['bind_collector_receipt'](collector, report(original), deepcopy(original))
                    changed = deepcopy(original)
                    mutate(changed)
                    with self.assertRaisesRegex(ValueError, '^INCOMPATIBLE_REPEATED_COLLECTOR_RECEIPT$'):
                        env['bind_collector_receipt'](collector, report(changed), changed)
                    self.assertEqual(env['collection_reports'], [original])

    def test_non_directory_and_nonpassed_repeats_refuse(self):
        for entry in ('controls.py', 'native_controls.py'):
            for kind, outcome in (('session', 'passed'), ('module', 'passed'),
                                  ('class', 'passed'), ('directory', 'skipped'),
                                  ('directory', 'failed')):
                with self.subTest(entry=entry, kind=kind, outcome=outcome):
                    env, collector = isolated(entry), object()
                    row = receipt(kind, outcome)
                    env['bind_collector_receipt'](collector, report(row), deepcopy(row))
                    with self.assertRaisesRegex(ValueError, '^INCOMPATIBLE_REPEATED_COLLECTOR_RECEIPT$'):
                        env['bind_collector_receipt'](collector, report(row), deepcopy(row))

    def test_same_report_object_cannot_be_bound_twice(self):
        for entry, env in self.each():
            with self.subTest(entry=entry):
                row, collector = receipt(), object()
                made = report(row)
                env['bind_collector_receipt'](collector, made, deepcopy(row))
                with self.assertRaisesRegex(ValueError, '^DUPLICATE_COLLECTOR_RECEIPT$'):
                    env['bind_collector_receipt'](collector, made, deepcopy(row))

    def test_real_callback_from_any_bound_report_is_preserved(self):
        for entry, env in self.each():
            with self.subTest(entry=entry):
                row, collector = receipt(), object()
                made = [report(row), report(row)]
                for rep in made:
                    env['bind_collector_receipt'](collector, rep, deepcopy(row))
                env['pytest_collectreport'](None, made[-1])
                self.assertEqual(env['collection_reports'][0]['callback_count'], 1)
                self.assertEqual(env['collection_report_callbacks'], [row['identity']])
                # A second actual callback stays visible; unchanged owner checks
                # reject count=2 and duplicate callback identities. Never suppress.
                env['pytest_collectreport'](None, made[0])
                self.assertEqual(env['collection_reports'][0]['callback_count'], 2)
                self.assertEqual(env['collection_report_callbacks'], [row['identity']] * 2)

    def test_repeat_after_callback_refuses(self):
        for entry, env in self.each():
            with self.subTest(entry=entry):
                row, collector = receipt(), object()
                made = report(row)
                env['bind_collector_receipt'](collector, made, deepcopy(row))
                env['pytest_collectreport'](None, made)
                with self.assertRaisesRegex(ValueError, '^INCOMPATIBLE_REPEATED_COLLECTOR_RECEIPT$'):
                    env['bind_collector_receipt'](collector, report(row), deepcopy(row))

    def test_unbound_or_changed_callback_refuses(self):
        for entry in ('controls.py', 'native_controls.py'):
            for change in ('unbound', 'nodeid', 'outcome'):
                with self.subTest(entry=entry, change=change):
                    env, row = isolated(entry), receipt()
                    made = report(row)
                    if change != 'unbound':
                        env['bind_collector_receipt'](object(), made, deepcopy(row))
                        setattr(made, change, 'changed')
                    with self.assertRaisesRegex(ValueError, '^UNBOUND_COLLECTREPORT_CALLBACK$'):
                        env['pytest_collectreport'](None, made)
                    self.assertEqual(env['collection_report_callbacks'], [])

    def test_first_empty_skipped_and_failed_reports_remain_visible(self):
        for entry in ('controls.py', 'native_controls.py'):
            for outcome in ('passed', 'skipped', 'failed'):
                with self.subTest(entry=entry, outcome=outcome):
                    env, row = isolated(entry), receipt('module', outcome)
                    row['children'] = []
                    made = report(row)
                    env['bind_collector_receipt'](object(), made, row)
                    env['pytest_collectreport'](None, made)
                    self.assertEqual(env['collection_reports'], [row])
                    self.assertEqual(row['callback_count'], 1)


if __name__ == '__main__':
    unittest.main()
