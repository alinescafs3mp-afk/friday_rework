"""Precise DATA-ONLY acceptance proposal, NOT_RUN; never launches or observes.
Caller is independent admitting owner, supplies actual existing-backend reap/status,
exact PID/start/group observation, lock probe and before/after hashes. Missing data
means INCONCLUSIVE. This code does not establish truth of supplied observations.
"""


def evaluate(clock, phase, operation_entry, attempts, observation, contour,
             positive_complete=None, wall_entered=None):
    identity = {k: clock.get(k) for k in ('assignment', 'generation', 'nonce', 'entry',
                'selection', 'run_name', 'run_root', 'canary', 'reviewed_source_pins_sha256')}
    identity.update(node=phase['node'], operation=phase['entry_operation'])
    def inconclusive(reason):
        return {'status': 'INCONCLUSIVE', 'reason': reason, 'identity': identity}
    if any(v is None for v in identity.values()):
        return inconclusive('missing exact admission identity')
    if (clock.get('execution_authorized') is not True or clock.get('selection') != '@canary' or
        clock.get('scope') != 'FINITE_SYNTHETIC_CANARY_ONLY'):
        return inconclusive('no exact separately admitted phase')
    if not contour or contour.get('identity') != identity:
        return inconclusive('missing/foreign guard native contour')
    if not operation_entry or operation_entry.get('kind') != 'operation_entry' or operation_entry.get('identity') != identity:
        return inconclusive('selected exact operation was not durably reached')
    guard = contour.get('guard'); timeout = contour.get('timeout')
    if not guard or not timeout or any(k not in d for d in (guard, timeout) for k in ('pid', 'ppid', 'pgid', 'start_ticks', 'exe')):
        return inconclusive('missing exact native process start/group identity')
    if (operation_entry.get('pid') != guard['pid'] or operation_entry.get('pgid') != guard['pgid'] or
        guard['ppid'] != timeout['pid'] or guard['pgid'] != timeout['pgid'] or
        timeout['pgid'] != timeout['pid']):
        return inconclusive('marker is not from exact timeout-owned guard')
    entered = operation_entry.get('monotonic', -1)
    deadline = contour['external_deadline_monotonic']
    if not 0 < entered < deadline < clock['deadline_monotonic']:
        return inconclusive('entry outside admitted original finite phase')
    # Explicit independent facts; no source-only zeros, intent or lock counters.
    required = ('backend_reaped', 'exact_known_pids_absent', 'exact_group_scan_quiescent',
                'outside_synthetic_hashes_unchanged', 'phase_clock_unchanged',
                'all_input_pins_unchanged', 'canonical_lock_identity_verified',
                'fresh_lock_acquire_unlock_close_returned')
    if any(observation.get(k) is not True for k in required):
        return inconclusive('independent post-reap/effect/lock observation missing or uncertain')
    if observation.get('guard_identity') != guard or observation.get('timeout_identity') != timeout:
        return inconclusive('post-reap observation addresses different process start/group')
    if observation.get('identity') != identity:
        return inconclusive('post-reap observation belongs to different phase/nonce')
    if observation.get('exit_code') != phase['expected_exit']:
        return inconclusive('wrong exact terminal code; retain raw failure')
    if observation.get('after_denial_sentinel_present') is not False or observation.get('alarm_swallowed_sentinel_present') is not False:
        return inconclusive('forbidden continuation observed or not checked')
    terms = [a for a in attempts if a.get('kind') == 'termination_intent']
    if phase['expected_exit'] == 0:
        if (not positive_complete or positive_complete.get('kind') != 'positive_assertions_complete' or
            positive_complete.get('identity') != identity or positive_complete.get('pid') != guard['pid'] or positive_complete.get('pgid') != guard['pgid'] or
            not entered <= positive_complete.get('monotonic', -1) < deadline or
            observation.get('selected_test_call_passed') is not True or
            observation.get('later_test_call_passed') is not True or
            observation.get('later_sentinel_bytes') != 'ran' or terms):
            return inconclusive('positive assertions and subsequent test not proven')
    else:
        if observation.get('later_sentinel_present') is not False:
            return inconclusive('later test ran or sentinel absence not observed')
        if phase['expected_exit'] == 137:
            if (not wall_entered or wall_entered.get('kind') != 'wall_entered' or
                wall_entered.get('identity') != identity or wall_entered.get('pid') != guard['pid'] or wall_entered.get('pgid') != guard['pgid'] or
                not entered <= wall_entered.get('monotonic', -1) < deadline or terms):
                return inconclusive('blocked-handler wall was not durably entered or guard terminated first')
            # SIGKILL/group semantics require actual supervisor/status/raw stderr
            # and time evidence, not merely the expected137 or missing receipt.
            if (observation.get('external_timeout_reason') != 'EXTERNAL_TIMEOUT_SIGKILL' or
                observation.get('pinned_timeout_sigkill_diagnostic') is not True or
                observation.get('actual_backend_signal') != 'SIGKILL' or
                observation.get('terminal_monotonic', -1) < deadline):
                return inconclusive('external wall timeout cause not independently observed')
        else:
            if len(terms) != 1 or terms[0].get('identity') != identity:
                return inconclusive('missing or foreign termination intent')
            term = terms[0]
            if (term.get('exit_code') != phase['expected_exit'] or term.get('reason') != phase['expected_reason'] or
                not entered <= term.get('monotonic', -1) < deadline):
                return inconclusive('early/unrelated denial or wrong exact cause')
    return {'status': 'CANARY_PHASE_EVIDENCED', 'identity': identity,
            'scope': 'selected controlled operation only; does not accept guard/product/full gate'}
