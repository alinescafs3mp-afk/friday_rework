def exact_functions(file, names, **values):
    tree = ast.parse(file.read_text())
    definitions = [n for n in ast.walk(tree)
                   if isinstance(n, (ast.FunctionDef, ast.ClassDef)) and n.name in names]
    assert {n.name for n in definitions} == set(names)
    # Definitions only. No candidate top-level imports, guards or pytest.main.
    namespace = dict(values)
    exec(compile(ast.Module(body=definitions, type_ignores=[]), str(file), 'exec'), namespace)
    return namespace

def temp_model(tmp_path):
    parent = tmp_path / 'parent'; parent.mkdir(mode=0o700)
    root = parent / 'r-01234567'; root.mkdir(mode=0o700)
    api = exact_functions(SOURCE / 'admission.py', ['temp_identity', 'verify_donor_temporary_root'],
        pathlib=pathlib, os=os, stat=stat, donor_temp_path=lambda c: root,
        donor_temp_parent=lambda c: parent)
    clock = {'donor_temporary_root': {'path': str(root),
        'identity': api['temp_identity'](root), 'parent_identity': api['temp_identity'](parent)}}
    return api, clock, root, parent


def test_temp_identity_and_actual_absence_model(tmp_path):
    api, clock, root, parent = temp_model(tmp_path)
    assert api['verify_donor_temporary_root'](clock) == root
    with pytest.raises(ValueError, match='CLEANUP_NOT_OBSERVED'):
        api['verify_donor_temporary_root'](clock, removed=True)
    root.rmdir()
    api['verify_donor_temporary_root'](clock, removed=True)
    root.symlink_to(parent / 'absent')
    with pytest.raises(ValueError, match='CLEANUP_NOT_OBSERVED'):
        api['verify_donor_temporary_root'](clock, removed=True)


@pytest.mark.parametrize('drift', ['path', 'inode', 'parent', 'mode', 'symlink', 'missing', 'bad_identity'])
def test_changed_or_foreign_temp_refused(tmp_path, drift):
    api, clock, root, parent = temp_model(tmp_path)
    ref = clock['donor_temporary_root']
    if drift == 'path': ref['path'] += '-foreign'
    elif drift == 'inode': ref['identity'][1] += 1
    elif drift == 'parent': ref['parent_identity'][1] += 1
    elif drift == 'mode': root.chmod(0o755)
    elif drift == 'symlink':
        root.rmdir(); root.symlink_to(parent)
    elif drift == 'missing': root.rmdir()
    else: ref['identity'] = [1, 2, 3]
    with pytest.raises((ValueError, OSError)):
        api['verify_donor_temporary_root'](clock)

@pytest.mark.parametrize('drift', [None, 'writable_parent', 'foreign_parent',
    'symlink_parent', 'resolved_alias', 'no_sticky_base', 'foreign_base'])
def test_temp_parent_refuses_foreign_writable_or_aliased_root(drift):
    base_stat = SimpleNamespace(st_mode=stat.S_IFDIR | 0o1777, st_uid=0)
    parent_stat = SimpleNamespace(st_mode=stat.S_IFDIR | 0o700, st_uid=1000)
    base = SimpleNamespace(lstat=lambda: base_stat)
    parent = SimpleNamespace(parent=base, lstat=lambda: parent_stat)
    parent.resolve = lambda strict: parent
    root = SimpleNamespace(parent=parent)
    if drift == 'writable_parent': parent_stat.st_mode |= 0o020
    elif drift == 'foreign_parent': parent_stat.st_uid = 9999
    elif drift == 'symlink_parent': parent_stat.st_mode = stat.S_IFLNK | 0o777
    elif drift == 'resolved_alias': parent.resolve = lambda strict: object()
    elif drift == 'no_sticky_base': base_stat.st_mode = stat.S_IFDIR | 0o777
    elif drift == 'foreign_base': base_stat.st_uid = 1000
    api = exact_functions(SOURCE / 'admission.py', ['donor_temp_parent'],
        donor_temp_path=lambda c: root, stat=stat, os=SimpleNamespace(getuid=lambda: 1000))
    if drift:
        with pytest.raises(ValueError, match='DONOR_TEMP_PARENT'):
            api['donor_temp_parent']({})
    else:
        assert api['donor_temp_parent']({}) is parent

    env_api, _, run, root = env_model()
    registry = SimpleNamespace(spawn_runtest=Mock(return_value='retained-model-handle'))
    wrapped = ['/model/bwrap', '--', 'exact-prechecked-model-argv']

    status.spawn = lambda kwargs: registry.spawn_runtest(session, wrapped, kwargs, native_owner=native_owner)
    native_owner = SimpleNamespace(prepare_donor_status=Mock(return_value=status))
    session = object()
    holder = {}  # Inert preparation only; no actual native/status evidence.
    require = lambda ok, why: None if ok else (_ for _ in ()).throw(ValueError(why))

        require=require, donor={'sys': SimpleNamespace(executable='/pinned/python'),
            'subprocess': SimpleNamespace(DEVNULL=-3)}, Path=pathlib.Path, unit=unit,
        temproot=root, temp_api=env_api, run=run, clock={'model': True},
        registry=registry, native_owner=native_owner, session=session, holder=holder,
        before={'argv': ['exact-prechecked-model-argv']})

    kwargs = {'env': {'TMPDIR': root, 'PYTEST_DEBUG_TEMPROOT': root, 'OPENAI_API_KEY': 'must-drop'}}

    actual = registry.spawn_runtest.call_args.args[-1]
    assert registry.spawn_runtest.call_args.args[1] == wrapped
    assert holder == {'semantic_argv': ['exact-prechecked-model-argv'], 'wrapped_argv': wrapped, 'status': status}

    assert actual['stdin'] == -3 and actual['close_fds'] is True

def receipt_model(tmp_path):
    api, system, env, manifest = startup_model()
    (tmp_path / 'import-path-pins.json').write_bytes(b'INERT SOURCE CONTROL MANIFEST\n')
    home = str(tmp_path / 'fixtures'); temp = str(tmp_path / 'temp')
    receipt = {'state': 'INDEPENDENT_DONOR_CLEAN_STARTUP_OBSERVED',
        'interpreter_sha256': manifest['interpreter']['sha256'],
        'import_path_pins_sha256': hashlib.sha256((tmp_path / 'import-path-pins.json').read_bytes()).hexdigest(),
        'sys_path': manifest['startup_expected'], 'boot_id': 'inert-source-control-boot',
        'flags': api['startup_flags'](runtime=True), 'candidate_imported': False,
        'backend_terminal_reaped': True,
        'actual_shims': {n: m.__file__ for n, m in system.modules.items()},
        'probe_home': home, 'probe_temporary_root': temp,
        'interpreter_flags_argv': ['-P', '-s', '-B'],
        'pre_interpreter_environment': {'PATH': '/usr/bin:/bin', 'LANG': 'C.UTF-8',
            'LC_ALL': 'C.UTF-8', 'TZ': 'UTC', 'HOME': home, 'TMPDIR': temp,
            'PYTEST_DEBUG_TEMPROOT': temp, 'PYTHONHASHSEED': '0', 'PYTHONUTF8': '1', 'PYTHONDONTWRITEBYTECODE': '1'}}
    api['load'] = lambda source: manifest
    api['pathlib'] = SimpleNamespace(Path=lambda p:
        SimpleNamespace(read_text=lambda: 'inert-source-control-boot\n')
        if str(p) == '/proc/sys/kernel/random/boot_id' else pathlib.Path(p))
    return api, receipt


def test_new_receipt_contract_model_and_old_profile_cannot_be_relabelled(tmp_path):
    api, receipt = receipt_model(tmp_path)
    path = tmp_path / 'inert-receipt.json'
    raw = json.dumps(receipt).encode(); path.write_bytes(raw); path.chmod(0o400)
    digest = hashlib.sha256(raw).hexdigest()
    assert api['verify_owner_startup_receipt'](tmp_path, path, digest, runtime=True)['sha256'] == digest
    with pytest.raises(api['ProbeRefusal']):
        api['verify_owner_startup_receipt'](tmp_path, path, digest)


@pytest.mark.parametrize('gap', ['old_receipt', 'seed_flag', 'pre_env_extra', 'pre_env_seed',
    'argv', 'no_reap', 'candidate_imported', 'foreign_boot', 'user_site'])

    api, receipt = receipt_model(tmp_path)
    if gap == 'old_receipt': receipt['state'] = 'INDEPENDENT_FULL_ISOLATED_STARTUP_OBSERVED'
    elif gap == 'seed_flag': receipt['flags']['hash_randomization'] = 1

    elif gap == 'pre_env_seed': receipt['pre_interpreter_environment']['PYTHONHASHSEED'] = '17'
    elif gap == 'argv': receipt['interpreter_flags_argv'] = ['-I', '-B']
    elif gap == 'no_reap': receipt['backend_terminal_reaped'] = False
    elif gap == 'candidate_imported': receipt['candidate_imported'] = True
    elif gap == 'foreign_boot': receipt['boot_id'] = 'foreign-boot'
    else: receipt['sys_path'] = receipt['sys_path'] + ['<USER_HOME>/.local/site-packages']
    path = tmp_path / 'inert-receipt.json'
    raw = json.dumps(receipt).encode(); path.write_bytes(raw); path.chmod(0o400)
    with pytest.raises(api['ProbeRefusal']):
        api['verify_owner_startup_receipt'](tmp_path, path, hashlib.sha256(raw).hexdigest(), runtime=True)
