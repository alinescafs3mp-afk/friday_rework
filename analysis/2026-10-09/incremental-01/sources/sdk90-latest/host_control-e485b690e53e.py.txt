"""Source-only boundary between the actual FileOwner and donor preparation.

No manager, runner, drainer or Popen lives here. Preparation is never admission.
"""

class HostControlRefusal(RuntimeError):
    pass


def preparation_request(argv, *, phase_identity, required_capabilities):
    """Bind the semantic child before the existing provider prepares its mounts."""
    if (type(argv) is not list or not argv or
            any(type(v) is not str or '\0' in v for v in argv)):
        raise HostControlRefusal('EXACT_DONOR_ARGV_REQUIRED')
    required = {'original_clock_sha256', 'phase_clock_sha256', 'phase',
                'unit', 'boot_id', 'original_deadline'}
    if type(phase_identity) is not dict or set(phase_identity) != required:
        raise HostControlRefusal('ORIGINAL_PHASE_IDENTITY_REQUIRED')
    if (type(required_capabilities) is not list or not required_capabilities or
            any(type(v) is not str or not v for v in required_capabilities) or
            len(set(required_capabilities)) != len(required_capabilities)):
        raise HostControlRefusal('COMPLETE_CAPABILITY_CENSUS_REQUIRED')
    return {'capability': 'host-control-v1', 'semanticArgv': list(argv),
            'phase': dict(phase_identity),
            'requiredCapabilities': list(required_capabilities),
            'runtimeGrant': False}


def refuse_unqualified_preparation(preparation, argv):
    """Even a forged full/QUALIFIED/file-only/custom-runner result cannot spawn."""
    if (type(preparation) is not dict or
            preparation.get('capability') != 'host-control-v1' or
            preparation.get('state') != 'UNQUALIFIED_SOURCE_PREPARATION' or
            preparation.get('runtimeGrant') is not False or
            preparation.get('semanticArgv') != argv):
        raise HostControlRefusal('FOREIGN_OR_PROMOTED_DONOR_PREPARATION')
    raise HostControlRefusal('REACHABLE_LOCAL_NETWORK_CONTROL_UNMEDIATED')
