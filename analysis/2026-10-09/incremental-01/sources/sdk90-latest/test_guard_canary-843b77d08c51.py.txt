"""Source-only proposed guard canaries. Run individually only after owner admission.
Each denial requires exact durable operation identity/reason/exit and independent post-reap evidence; startup failure is INCONCLUSIVE.
Synthetic outside fixtures/descriptors belong to the future owner canary root;
never point these cases at actual credentials or user-owned files.
"""
import os, json, time, errno
from pathlib import Path
import pytest


def run_root():
    return Path(os.environ['FRIDAY_FIXTURE_EVIDENCE'])


def mark(operation, kind='operation_entry'):
    root = run_root()
    clock = json.loads((root / 'clock.json').read_text())
    source = Path(os.environ['FRIDAY_GUARD_SOURCE'])
    phase = json.loads((source / 'canary-plan.json').read_text())[clock['canary']]
    if operation != phase['entry_operation']:
        raise RuntimeError('wrong exact canary operation')
    identity = {k: clock[k] for k in ('assignment', 'generation', 'nonce', 'entry',
                'selection', 'run_name', 'run_root', 'canary', 'reviewed_source_pins_sha256')}
    identity.update(node=phase['node'], operation=operation)
    if identity['selection'] != '@canary':
        raise RuntimeError('wrong selected route')
    value = {'kind': kind, 'identity': identity, 'monotonic': time.monotonic(),
             'pid': os.getpid(), 'pgid': os.getpgrp()}
    fd = os.open(root / (kind + '.json'), os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600)
    with os.fdopen(fd, 'w') as stream:
        stream.write(json.dumps(value) + '\n'); stream.flush(); os.fsync(stream.fileno())
    directory = os.open(root, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
    try: os.fsync(directory)
    finally: os.close(directory)


def after_denial():
    (run_root() / 'DENIAL_WAS_SWALLOWED').write_text('INVALID')
    pytest.fail('boundary denial did not terminate the process')


def test_positive_owned_paths_descriptors_and_pipe():
    root = run_root()
    mark('owned-positive-io')
    with open('fixtures/relative.txt', 'w') as f:
        f.write('owned')
    assert Path('fixtures/relative.txt').read_text() == 'owned'
    fd = os.open(root / 'fixtures', os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
    try:
        # Actual dir_fd differs from cwd and remains the kernel anchor.
        os.chdir(root / 'tmp')
        item = os.open('via-fd.txt', os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600, dir_fd=fd)
        with os.fdopen(item, 'w') as f:
            f.write('descriptor')
        assert (root / 'fixtures/via-fd.txt').read_text() == 'descriptor'
        # O_NOFOLLOW must still reject the final link, not silently resolve it.
        os.symlink('via-fd.txt', 'link', dir_fd=fd)
        with pytest.raises(OSError) as failure:
            os.open('link', os.O_RDONLY | os.O_NOFOLLOW, dir_fd=fd)
        assert failure.value.errno == errno.ELOOP
        os.rename('via-fd.txt', 'renamed.txt', src_dir_fd=fd, dst_dir_fd=fd)
        os.unlink('renamed.txt', dir_fd=fd)
    finally:
        os.close(fd)
        os.chdir(root)
    read, write = os.pipe2(os.O_NONBLOCK | os.O_CLOEXEC)
    try:
        with os.fdopen(write, 'wb', buffering=0, closefd=False) as stream:
            stream.write(b'pipe')
        assert os.read(read, 4) == b'pipe'
    finally:
        os.close(read)
        os.close(write)
    # Real installed optional search entry stays present in sys.path. Each
    # surface receives the genuine anchored kernel ENOENT BEFORE content open.
    import sys, io, importlib.metadata
    source = Path(os.environ['FRIDAY_GUARD_SOURCE'])
    declaration = json.loads((source / 'import-path-pins.json').read_bytes())
    archive = declaration['declared_absent_path']
    assert archive in sys.path
    for function in (lambda: open(archive, 'rb'), lambda: io.open(archive, 'rb'),
                     lambda: Path(archive).open('rb'), lambda: os.open(archive, os.O_RDONLY)):
        with pytest.raises(FileNotFoundError) as failure:
            function()
        assert failure.value.errno == errno.ENOENT and failure.value.filename == Path(archive).name
    anchor = os.open(Path(archive).parent, os.O_PATH | os.O_DIRECTORY | os.O_NOFOLLOW)
    try:
        with pytest.raises(FileNotFoundError) as failure:
            os.open(Path(archive).name, os.O_RDONLY, dir_fd=anchor)
        assert failure.value.errno == errno.ENOENT
    finally:
        os.close(anchor)
    # Actual installed importlib discovery consumes this ENOENT normally;
    # no entry is hidden and no Pydantic plugin setting is changed.
    list(importlib.metadata.distributions())
    assert archive in sys.path
    # Exercise the exact pinned donor module, without adding donor sys.path
    # entries or importing the config/provider chain. Direct reads precede the
    # detector so /.dockerenv or cached predicates cannot hide a missing route.
    import hashlib, runpy
    donor = Path('<SOL_WORK>/FRW026-ORDINARY-SKILLS-FINAL-GATE-native')
    relative = 'hermes_platform/host/runtime.py'
    corpus = json.loads((source / 'corpus-pins.json').read_bytes())
    assert hashlib.sha256((donor / relative).read_bytes()).hexdigest() == corpus['native'][relative]
    host = runpy.run_path(str(donor / relative))
    observed = {}
    for path in ('/proc/1/cgroup', '/proc/self/mountinfo', '/proc/version'):
        with open(path, 'r', encoding='utf-8') as stream:
            observed[path] = stream.read()
        assert observed[path], 'real public host probe returned no data: ' + path
        assert host['_read_proc'](path) == observed[path]
        # Exercise existing descriptor checks for the same exact read route.
        handle = os.open(path, os.O_RDONLY | os.O_CLOEXEC)
        try:
            assert os.read(handle, 1)
        finally:
            os.close(handle)
    assert type(host['_detect_container']()) is bool
    assert type(host['is_container']()) is bool
    assert host['is_wsl']() == ('microsoft' in observed['/proc/version'].lower())
    # A cached call must keep the actual first observation, without a stub.
    assert host['is_wsl']() == ('microsoft' in observed['/proc/version'].lower())
    # Exercise the complete real Linux psutil import, including all three
    # /proc/stat reads. No process/sensors/network API is called. This SOURCE
    # requires new independent admission; prior positive receipts do not apply.
    psutil_root = Path('<PROJECT>/.runtime/hermes-telegram-test/env/lib/python3.14/site-packages/psutil')
    ancillary = json.loads((source / 'ancillary-pins.json').read_bytes())
    for relative in ('__init__.py', '_common.py', '_ntuples.py', '_psposix.py',
                     '_pslinux.py', '_psutil_linux.abi3.so',
                     '__pycache__/__init__.cpython-314.pyc',
                     '__pycache__/_common.cpython-314.pyc',
                     '__pycache__/_ntuples.cpython-314.pyc',
                     '__pycache__/_psposix.cpython-314.pyc',
                     '__pycache__/_pslinux.cpython-314.pyc'):
        path = psutil_root / relative
        assert hashlib.sha256(path.read_bytes()).hexdigest() == ancillary[str(path)]
    with open('/proc/stat', 'rb') as stream:
        assert stream.readline().startswith(b'cpu ')
    handle = os.open('/proc/stat', os.O_RDONLY | os.O_CLOEXEC)
    try:
        assert os.read(handle, 4) == b'cpu '
    finally:
        os.close(handle)
    assert not any(name == 'psutil' or name.startswith('psutil.') for name in sys.modules), 'psutil import must be fresh'
    import psutil
    assert Path(psutil.__file__).resolve() == psutil_root / '__init__.py'
    assert psutil.__version__ == '7.2.2'
    assert psutil.PROCFS_PATH == '/proc'
    # These are constructed at import, not lazily supplied by a later call.
    assert psutil._last_cpu_times and psutil._last_per_cpu_times
    assert all('idle' in value._fields for value in psutil._last_cpu_times.values())
    assert all(values and all('idle' in value._fields for value in values)
               for values in psutil._last_per_cpu_times.values())
    assert archive in sys.path
    mark('owned-positive-io', 'positive_assertions_complete')


@pytest.mark.parametrize('api', ['builtin', 'path'])
def test_relative_write_terminates(api):
    try:
        if api == 'builtin':
            mark('builtin-open-relative-write')
            open('../synthetic-outside.txt', 'w').write('FORBIDDEN')
        else:
            mark('path-write-relative')
            Path('../synthetic-outside.txt').write_text('FORBIDDEN')
    except BaseException:
        pass  # This would swallow the old boundary exception.
    after_denial()

@pytest.mark.parametrize('operation', ['fdopen', 'read', 'fdopen-write', 'write', 'truncate'])
def test_inherited_foreign_descriptor_terminates(operation):
    # Future trusted setup preopens ONLY fake canary files, then binds FD identity
    # in its admission/launch receipt. No real foreign descriptor is acquired here.
    import json
    config = json.loads((run_root() / 'synthetic-fd-launch.json').read_text())['descriptors']
    key = 'synthetic_credential_fd' if operation in ('fdopen', 'read') else 'synthetic_write_fd'
    fd = config[key]['fd']
    try:
        if operation == 'fdopen':
            mark('os-fdopen-foreign-read')
            os.fdopen(fd, 'r', closefd=False).read()
        elif operation == 'read':
            mark('os-read-foreign-fd')
            os.read(fd, 1)
        elif operation == 'fdopen-write':
            mark('os-fdopen-foreign-write')
            os.fdopen(fd, 'w', closefd=False).write('FORBIDDEN')
        elif operation == 'write':
            mark('os-write-foreign-fd')
            os.write(fd, b'FORBIDDEN')
        else:
            mark('os-ftruncate-foreign-fd')
            os.ftruncate(fd, 0)
    except BaseException:
        pass
    after_denial()

@pytest.mark.parametrize('api', ['direct', 'multiprocessing', 'socket'])
def test_actual_installed_bindings_terminate(api):
    # Actual installed module attributes, never the guard's local replacements.
    try:
        if api == 'direct':
            import _posixsubprocess
            mark('installed-fork-exec')
            _posixsubprocess.fork_exec()
        elif api == 'multiprocessing':
            import multiprocessing
            process = multiprocessing.Process(target=lambda: None)
            mark('installed-BaseProcess-start')
            process.start()
        else:
            import socket
            mark('installed-socket-new')
            socket.socket()
    except BaseException:
        pass
    after_denial()


def test_wall_timeout_cannot_be_swallowed():
    import time
    # Whole canary invocation has a separately admitted short hard wall limit.
    mark('inner-wall-loop')
    while True:
        try:
            time.sleep(60)
        except BaseException:
            (run_root() / 'ALARM_WAS_SWALLOWED').write_text('INVALID')


def test_external_timeout_covers_delayed_python_handler():
    import signal
    # Deliberately suspend dispatch in this canary; external timeout SIGKILL must
    # terminate the exact owned process group. No guard counter proves this.
    import time
    mark('block-SIGALRM-then-wall-loop')
    signal.pthread_sigmask(signal.SIG_BLOCK, {signal.SIGALRM})
    mark('block-SIGALRM-then-wall-loop', 'wall_entered')
    while True:
        try:
            time.sleep(60)
        except BaseException:
            (run_root() / 'ALARM_WAS_SWALLOWED').write_text('INVALID')
