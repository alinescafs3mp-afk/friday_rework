def terminate(code, reason):
    # No exception crosses into pytest/product handlers. A receipt failure still exits.
    try:
        record('termination_intent', reason=reason, exit_code=code,
               attempt_log_scope='instrumented routes only; no syscall/effect census')
    finally:
        _hard_exit(code)

def deny(reason):

def deny_unreviewed_file_read(target):
    # target is the canonical path already checked by check_open. Capture only
    # bounded frame metadata; never inspect locals, format traceback/source, or
    # open another path. Diagnostic failures must still reach the same hard exit.
    try:
        frames = []
        frame = sys._getframe(1)
        try:
            for _ in range(16):
                if frame is None:
                    break
                code = frame.f_code
                frames.append({'filename': code.co_filename[:4096],
                               'function': code.co_name[:256],
                               'line': frame.f_lineno,
                               'filename_truncated': len(code.co_filename) > 4096,
                               'function_truncated': len(code.co_name) > 256})
                frame = frame.f_back
            record('read_refusal_diagnostic', reason='UNREVIEWED_FILE_READ',
                   canonical_path=target, frames=frames,
                   frame_order='innermost_first', frame_limit=16,
                   outer_frames_omitted=frame is not None)
        finally:
            del frame
    finally:
        deny('UNREVIEWED_FILE_READ')

def own(p, fd=None):
    if isinstance(p, int):
        return os.path.realpath('/proc/self/fd/' + str(p))
    p = os.fsdecode(p)
    if not os.path.isabs(p):
        if fd not in (None, -1, -100):
            if not stat.S_ISDIR(os.fstat(fd).st_mode):
                raise NotADirectoryError(fd)
            # Keep the directory descriptor anchor, including renamed directories.
            p = '/proc/self/fd/' + str(fd) + '/' + p
        else:
            p = os.path.join(os.getcwd(), p)
    return os.path.realpath(p)

def fd_identity(fd):
    s = os.fstat(fd)
    return (s.st_dev, s.st_ino, s.st_mode, fcntl.fcntl(fd, fcntl.F_GETFL) & os.O_ACCMODE)

# Only actual preexisting stdout/stderr PIPE identities and our subsequently
# created pipes are retained. An arbitrary pipe/socket FD is never blanket-safe.
pipe_fds = {}
for fd in (1, 2):
    if stat.S_ISFIFO(os.fstat(fd).st_mode):
        pipe_fds[fd] = fd_identity(fd)

def check_fd(fd, writing=False, mutation=False):
    identity = fd_identity(fd)
    if pipe_fds.get(fd) == identity and not mutation:
        if (writing and identity[-1] != os.O_RDONLY) or (not writing and identity[-1] != os.O_WRONLY):
            return
    p = own(fd)
    if p == path_manifest['declared_absent_path']:
        deny('DECLARED_LEAF_DESCRIPTOR_REFUSED')
    if not stat.S_ISREG(identity[2]) and not stat.S_ISDIR(identity[2]):
        deny('UNOWNED_DESCRIPTOR')
    if writing and not allowed(p):
        deny('OUTSIDE_DESCRIPTOR_WRITE')
    if credential(p):
        deny('FOREIGN_CREDENTIAL_DESCRIPTOR')
    if not writing and not readable(p):
        deny('UNREVIEWED_DESCRIPTOR_READ')

def check_open(p, flags):
    writing = bool(flags & (os.O_WRONLY | os.O_RDWR | os.O_CREAT | os.O_TRUNC | os.O_APPEND))
    if isinstance(p, int):
        check_fd(p, writing)
        return
    target = own(p)
    try:
        path_api['failure_only_open'](path_manifest, p, flags, target)
    except FileNotFoundError as error:
        record('absent_import_probe_enoent', path=path_manifest['declared_absent_path'],
               errno=error.errno, exception='FileNotFoundError', metadata_only=True, content_open=False)
        raise
    except path_api['ProbeRefusal'] as error:
        deny('ABSENT_IMPORT_PROBE_REFUSED:' + str(error))
    if writing and not allowed(target):
        deny('OUTSIDE_WRITE')
    if credential(target):
        deny('FOREIGN_CREDENTIAL_READ')
    # /dev/null is the only admitted special read path; special writes are not.
    if not writing and not readable(target):
        deny_unreviewed_file_read(target)

def audit(event, args):
    if event.startswith('socket.') or event in ('subprocess.Popen', 'os.system', 'os.posix_spawn', 'os.fork', 'os.forkpty', 'os.exec'):
        deny('OFFLINE_EFFECT:' + event)
    if event == 'open':
        # Builtin/Path/IO and FD paths: relative is checked exactly like absolute.
        check_open(args[0], args[2] or 0)
    if event in ('os.mkdir', 'os.remove', 'os.rmdir', 'os.chmod', 'os.chown', 'os.utime'):
        p = args[0]
        if isinstance(p, int):
            check_fd(p, True, True)
        elif not allowed(own(p, args[-1])):
            deny('OUTSIDE_MUTATION:' + event)
    if event == 'os.truncate':
        if isinstance(args[0], int):
            check_fd(args[0], True, True)
        elif not allowed(own(args[0])):
            deny('OUTSIDE_TRUNCATE')
    if event in ('os.rename', 'os.link'):
        for p, fd in ((args[0], args[2]), (args[1], args[3])):
            if not allowed(own(p, fd)):
                deny('OUTSIDE_MUTATION:' + event)
    if event == 'os.symlink' and not allowed(own(args[1], args[2])):
        deny('OUTSIDE_SYMLINK_DESTINATION')
    if event == 'ctypes.dlopen' and args[0] is not None:
        deny('NATIVE_LIBRARY_LOAD')

orig_open = os.open

def guarded_open(p, flags, mode=0o777, *, dir_fd=None):
    # Audit open does not carry dir_fd. Translate ONLY a relative dir_fd path
    # to its descriptor-anchored absolute /proc path before the actual call.
    # Do not replace it with realpath: keep final O_NOFOLLOW / O_EXCL semantics.
    if not os.path.isabs(os.fsdecode(p)) and dir_fd not in (None, -1, -100):
        if not stat.S_ISDIR(os.fstat(dir_fd).st_mode):
            raise NotADirectoryError(dir_fd)
        p = '/proc/self/fd/' + str(dir_fd) + '/' + os.fsdecode(p)
        dir_fd = None
    check_open(p, flags)
    return orig_open(p, flags, mode, dir_fd=dir_fd)
os.open = guarded_open
orig_read, orig_write = os.read, os.write

def guarded_read(fd, n):
    check_fd(fd)

def guarded_write(fd, data):
    check_fd(fd, True)
    return orig_write(fd, data)

# Actual FD mutation routes (independent of audit-event aliases).
for opname in ('ftruncate', 'fchmod', 'fchown'):
    original = getattr(os, opname)
    def fd_mutation(fd, *args, _original=original):
        check_fd(fd, True, True)
        return _original(fd, *args)
    setattr(os, opname, fd_mutation)
for opname in ('pipe', 'pipe2'):
    original = getattr(os, opname)
    def pipe(*args, _original=original):
        pair = _original(*args)
        for fd in pair:
            pipe_fds[fd] = fd_identity(fd)
        return pair
    setattr(os, opname, pipe)
orig_dup = os.dup

def dup(fd):
    result = orig_dup(fd)
    if pipe_fds.get(fd) == fd_identity(fd):
        pipe_fds[result] = fd_identity(result)
    return result
os.dup = dup

            deny('NATIVE_LIBRARY')
        self.lib = orig_cdll(*args, **kw)

                setattr(fn, n, v)
        return Rename()
