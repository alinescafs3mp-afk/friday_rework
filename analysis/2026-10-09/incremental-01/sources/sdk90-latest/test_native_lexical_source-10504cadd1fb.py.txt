"""SOURCE ONLY / NOT_RUN. Finite changed-area controls for future owner admission.
All mutable paths live in tmp_path. These definitions do not grant execution.
"""
import errno, importlib.util, json, os, pathlib
import pytest


def module():
    path = pathlib.Path(__file__).with_name('import_path_readiness.py')
    spec = importlib.util.spec_from_file_location('native_lexical_source_test', path)
    api = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(api)
    return api


def fixture_alias(tmp_path, monkeypatch):
    api = module()
    native = tmp_path / 'native'
    tools = native / 'tests/tools'
    tools.mkdir(parents=True)
    monkeypatch.setattr(api, 'NATIVE', str(native))
    raw = str(native) + '/tests/tools/../..'
    components = raw.split('/')[1:]
    paths = ['/'] + ['/' + '/'.join(components[:i]) for i in range(1, len(components) + 1)]
    m = {'native_lexical_alias': {'raw': raw, 'target': str(native),
            'steps': [{'path': p, 'identity': api.identity(os.lstat(p))} for p in paths],
            'source_files': {rel: 'a' * 64 for rel in api.NATIVE_ALIAS_SOURCES}},
         'directory_identities': {str(p): api.identity(p.lstat())
                                  for p in (native, native / 'tests', tools)}}
    return api, m, native, raw


def fixture_search(tmp_path, monkeypatch):
    api, m, native, raw = fixture_alias(tmp_path, monkeypatch)
    for name in ('stdlib', 'dynload', 'site', 'source', 'run', 'project/tests'):
        p = tmp_path / name
        p.mkdir(parents=True)
        m['directory_identities'][str(p)] = api.identity(p.lstat())
    m['directory_identities'][str(tmp_path / 'project')] = api.identity((tmp_path / 'project').lstat())
    archive = tmp_path / 'python314.zip'
    monkeypatch.setattr(api, 'ARCHIVE', str(archive))
    m['startup_expected'] = [str(archive)] + [str(tmp_path / x) for x in ('stdlib', 'dynload', 'site')]
    m['declared_absent_path'] = str(archive)
    m['ancestors'] = [{'path': str(p), 'identity': api.identity(p.lstat())}
                      for p in list(reversed(tmp_path.parents)) + [tmp_path]]
    args = (tmp_path / 'source', tmp_path / 'run', tmp_path / 'project', native)
    initial = [str(tmp_path / 'project/tests'), str(native), str(tmp_path / 'project'), str(tmp_path / 'run')] + m['startup_expected']
    return api, m, native, raw, args, initial


def test_exact_alias_real_metadata_every_component_and_fd_closed(tmp_path, monkeypatch):
    api, m, native, raw = fixture_alias(tmp_path, monkeypatch)
    actual_open, actual_close = api._open, api._close
    opened, closed, names = [], [], []
    def opened_metadata(path, flags):
        assert flags == os.O_PATH | os.O_DIRECTORY | os.O_NOFOLLOW | os.O_CLOEXEC
        names.append(path.rsplit('/', 1)[-1])
        fd = actual_open(path, flags)
        opened.append(fd)
        return fd
    def closed_metadata(fd):
        closed.append(fd)
        actual_close(fd)
    monkeypatch.setattr(api, '_open', opened_metadata)
    monkeypatch.setattr(api, '_close', closed_metadata)
    assert api.verify_native_alias(m, raw, native) is None
    assert names[-2:] == ['..', '..']
    assert len(opened) == len(m['native_lexical_alias']['steps'])
    assert closed == list(reversed(opened))
    for fd in opened:
        with pytest.raises(OSError) as error:
            os.fstat(fd)
        assert error.value.errno == errno.EBADF


def test_alias_repeated_insertion_keeps_initial_order_suffix_and_input(tmp_path, monkeypatch):
    api, m, native, raw, args, initial = fixture_search(tmp_path, monkeypatch)
    api.verify_paths(m, initial, '@native', *args, initial=True)
    paths = [raw, raw, raw] + initial
    before = list(paths)
    api.verify_paths(m, paths, '@native', *args)
    assert paths == before
    api.verify_paths(m, initial, '@project', *args)
    api.verify_paths(m, [str(args[0])] + m['startup_expected'], '@canary', *args, initial=True)


@pytest.mark.parametrize('suffix', ['/tests/..', '/tests/tools/./../..', '/tests/tools/../../', '/tests//tools/../..', '/tests/agent/../..', '/tests/tools/../../..'])
def test_other_lexical_spelling_even_same_target_refused(tmp_path, monkeypatch, suffix):
    api, m, native, raw, args, initial = fixture_search(tmp_path, monkeypatch)
    denied = str(native) + suffix
    with pytest.raises(api.ProbeRefusal, match='UNDECLARED_PYTEST_SEARCH_ENTRY') as error:
        api.verify_paths(m, [denied] + initial, '@native', *args)
    detail = json.loads(str(error.value).split(':', 1)[1])
    assert detail['raw'] == denied and detail['index'] == 0 and detail['raw_truncated'] is False

def test_native_alias_never_admitted_to_other_modes(tmp_path, monkeypatch, mode):

def test_alias_before_initial_prepend_and_reordered_suffix_refused(tmp_path, monkeypatch):
    api, m, native, raw, args, initial = fixture_search(tmp_path, monkeypatch)
    with pytest.raises(api.ProbeRefusal, match='POST_PREPEND_PATH_MISMATCH'):
        api.verify_paths(m, [raw] + initial, '@native', *args, initial=True)
    changed = [raw] + initial
    changed[-1], changed[-2] = changed[-2], changed[-1]
    with pytest.raises(api.ProbeRefusal, match='STARTUP_PATH_REMOVED_OR_REORDERED'):
        api.verify_paths(m, changed, '@native', *args)


@pytest.mark.parametrize('which', ['raw', 'target', 'step', 'source', 'root_identity', 'extra_source'])
def test_declaration_mutation_cannot_widen_alias(tmp_path, monkeypatch, which):
    api, m, native, raw = fixture_alias(tmp_path, monkeypatch)
    a = m['native_lexical_alias']
    if which in ('raw', 'target'): a[which] += '/.'
    elif which == 'step': a['steps'][-1]['path'] += '/.'
    elif which == 'source': a['source_files'][api.NATIVE_ALIAS_SOURCES[0]] = 'not-a-sha'
    elif which == 'extra_source': a['source_files']['tests/unreviewed.py'] = 'a' * 64
    else: m['directory_identities'][str(native)][1] += 1
    with pytest.raises(api.ProbeRefusal, match='NATIVE_ALIAS_'):
        api.verify_native_alias(m, raw, native)

@pytest.mark.parametrize('kind', ['symlink_same_directory', 'dangling_symlink', 'file', 'missing', 'replaced_directory'])
def test_lexical_tools_component_cannot_be_hidden_by_dotdot(tmp_path, monkeypatch, kind):
    api, m, native, raw = fixture_alias(tmp_path, monkeypatch)
    tools = native / 'tests/tools'
    former = native / 'tests/former-tools'
    tools.rename(former)
    if kind == 'symlink_same_directory': tools.symlink_to(former, target_is_directory=True)
    elif kind == 'dangling_symlink': tools.symlink_to(native / 'missing', target_is_directory=True)
    elif kind == 'file': tools.write_text('not a directory')
    elif kind == 'replaced_directory': tools.mkdir()
    with pytest.raises(api.ProbeRefusal, match='NATIVE_ALIAS_COMPONENT_'):
        api.verify_native_alias(m, raw, native)

@pytest.mark.parametrize('component', ['native', 'tests', 'tools'])
def test_binding_replacement_after_all_opens_refuses_and_closes(tmp_path, monkeypatch, component):
    api, m, native, raw = fixture_alias(tmp_path, monkeypatch)
    actual_open, actual_close = api._open, api._close
    opened, closed = [], []
    moved = native if component == 'native' else native / ('tests' if component == 'tests' else 'tests/tools')
    def replace_after_last_open(path, flags):
        fd = actual_open(path, flags)
        opened.append(fd)
        if len(opened) == len(m['native_lexical_alias']['steps']):
            moved.rename(moved.with_name('former-' + moved.name))
            moved.mkdir()
        return fd
    def closed_metadata(fd):
        closed.append(fd)
        actual_close(fd)
    monkeypatch.setattr(api, '_open', replace_after_last_open)
    monkeypatch.setattr(api, '_close', closed_metadata)
    with pytest.raises(api.ProbeRefusal, match='NATIVE_ALIAS_BINDING_DRIFT'):
        api.verify_native_alias(m, raw, native)
    assert closed == list(reversed(opened))


def test_component_permission_error_fails_closed(tmp_path, monkeypatch):
    api, m, native, raw = fixture_alias(tmp_path, monkeypatch)
    def refused(*args, **kwargs):
        raise PermissionError(errno.EACCES, 'synthetic metadata denial')
    monkeypatch.setattr(api, '_open', refused)
    with pytest.raises(api.ProbeRefusal, match='NATIVE_ALIAS_COMPONENT_ERROR:13'):
        api.verify_native_alias(m, raw, native)


def test_bounded_diagnostic_escapes_raw_and_does_not_repr_objects(tmp_path, monkeypatch):
    api, m, native, raw, args, initial = fixture_search(tmp_path, monkeypatch)
    for raw in ('/foreign/line\nquote"', 'x' * 5000):
        with pytest.raises(api.ProbeRefusal) as error:
            api.verify_paths(m, [raw] + initial, '@native', *args)
        detail = json.loads(str(error.value).split(':', 1)[1])
        assert detail['raw'] == raw[:4096] and detail['raw_length'] == len(raw)
        assert detail['raw_truncated'] == (len(raw) > 4096)
    class Foreign:
        def __repr__(self):
            pytest.fail('foreign repr must not execute')
    with pytest.raises(api.ProbeRefusal) as error:
        api.verify_paths(m, [Foreign()] + initial, '@native', *args)
    detail = json.loads(str(error.value).split(':', 1)[1])
    assert detail == {'index': 0, 'entry_count': len(initial) + 1, 'raw_is_str': False}
