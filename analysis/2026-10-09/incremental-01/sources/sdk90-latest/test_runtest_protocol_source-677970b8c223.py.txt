"""Source controls, NOT RUN in FRW026 source-only assignment.
Synthetic protocol cases test evidence rules; they do not replace donor tests.
"""
import copy
import pathlib
import runpy
import pytest

API = runpy.run_path(str(pathlib.Path(__file__).with_name('runtest_protocol.py')))


def fixture():
    identity = {'path': '/f.py', 'object_path': '/f.py', 'kind': 'function',
                'type': '_pytest.python.Function', 'nodeid': 'f.py::test_a[p]', 'source_relative_path': 'f.py'}
    collected = dict(identity, nodeid='::test_a[p]')
    definition = {'path': '/f.py', 'sha256': 'a'*64, 'qualified_name': 'test_a',
                  'line': 10, 'end_line': 12, 'definition_id': 'f0::test_a:10'}
    unit = {'unit_id': 'f000', 'selection': '@native', 'entry': 'native_guard.py',
            'file': {'path': '/f.py', 'sha256': 'a'*64},
            'expected': [{'collection_identity': collected, 'definition': definition}]}
    return unit, API['bind_item'](unit, identity, definition)


def events(outcomes=('passed', 'passed', 'passed'), xfail=None, exit_code=0):
    unit, binding = fixture(); rows=[]
    def put(kind, **fields):
        rows.append({'sequence': len(rows), 'kind': kind, 'identity': {'nonce': 'actual'},
                     'phase_clock_sha256': 'clock', **fields})
    put('collection', selected=[binding], deselected=[])
    put('item_start', binding=binding)
    phases = ['setup', 'call', 'teardown'] if len(outcomes)==3 else ['setup', 'teardown']
    for when, outcome in zip(phases, outcomes):
        put('report', binding=binding, report={'nodeid': binding['actual']['nodeid'], 'when': when,
            'outcome': outcome, 'duration': .001, 'wasxfail': xfail if when in ('call','setup') and outcome=='skipped' else None,
            'detail': 'real detail retained' if outcome!='passed' else None})
    put('item_finish', binding=binding);put('session_finish', exit=exit_code)
    return unit,rows


def decide(unit, rows, rc=0):
    return API['census'](unit, rows, {'nonce':'actual'}, 'clock', rc)


@pytest.mark.parametrize('outcomes,xfail,rc,bucket,status', [
    (('passed','passed','passed'),None,0,'passed','RUNTEST_REPORTED'),
    (('passed','failed','passed'),None,1,'failed','RUNTEST_PRODUCT_FAIL'),
    (('failed','passed'),None,1,'failed','RUNTEST_PRODUCT_FAIL'),
    (('skipped','passed'),None,0,'skipped','RUNTEST_REPORTED'),
    (('passed','skipped','passed'),'known failure',0,'xfailed','RUNTEST_REPORTED'),
    (('passed','passed','failed'),None,1,'failed','RUNTEST_PRODUCT_FAIL'),
])
def test_complete_actual_phase_census(outcomes,xfail,rc,bucket,status):
    unit, rows=events(outcomes,xfail,rc);result=decide(unit,rows,rc)
    assert result['status']==status and len(result['partition'][bucket])==1
    assert result['runtime_or_product_accepted'] is False

@pytest.mark.parametrize('mutation', ['nonce','clock','foreign_node','definition','missing_setup','duplicate_setup',
                                       'duplicate_finish','duplicate_collection','phase_order','zero_exit_failed',
                                       'missing_teardown','sequence_gap','negative_duration','nan_duration'])
def test_corrupt_evidence_refuses(mutation):
    unit,rows=events();rc=0
    if mutation=='nonce':rows[2]['identity']={'nonce':'foreign'}
    elif mutation=='clock':rows[2]['phase_clock_sha256']='different'
    elif mutation=='foreign_node':rows[2]['binding']=copy.deepcopy(rows[2]['binding']);rows[2]['binding']['actual']['path']='/foreign.py'
    elif mutation=='definition':rows[2]['binding']=copy.deepcopy(rows[2]['binding']);rows[2]['binding']['definition']['line']=999
    elif mutation=='missing_setup':del rows[2]
    elif mutation=='duplicate_setup':rows.insert(3,copy.deepcopy(rows[2]))
    elif mutation=='duplicate_finish':rows.insert(-1,copy.deepcopy(rows[-2]))
    elif mutation=='duplicate_collection':rows.insert(1,copy.deepcopy(rows[0]))
    elif mutation=='phase_order':rows[2],rows[3]=rows[3],rows[2]
    elif mutation=='zero_exit_failed':rows[3]['report']['outcome']='failed'
    elif mutation=='missing_teardown':del rows[4]
    elif mutation=='sequence_gap':rows[2]['sequence']=99
    elif mutation=='negative_duration':rows[2]['report']['duration']=-1
    elif mutation=='nan_duration':rows[2]['report']['duration']=float('nan')
    if mutation!='sequence_gap':
        for n,row in enumerate(rows):row['sequence']=n
    with pytest.raises(ValueError):decide(unit,rows,rc)

@pytest.mark.parametrize('trace', ['failed_call_exit5','selected_exit5','finish_then_collection',
                                  'finish_then_start','finish_then_report','finish_then_finish',
                                  'finish_then_session','bool_backend_exit','bool_session_exit'])
def test_contradictory_completion_is_not_admissible(trace):
    unit, rows=events(('passed','failed','passed'),exit_code=5)
    rc=5
    if trace=='selected_exit5':unit,rows=events(exit_code=5)
    elif trace.startswith('finish_then_'):
        unit,binding=fixture()
        rows=[{'sequence':0,'kind':'session_finish','exit':5,'identity':{'nonce':'actual'},
               'phase_clock_sha256':'clock'}]
        if trace=='finish_then_collection':extra={'kind':'collection','selected':[],'deselected':[binding]}
        elif trace=='finish_then_start':extra={'kind':'item_start','binding':binding}
        elif trace=='finish_then_report':extra={'kind':'report','binding':binding,'report':{}}
        elif trace=='finish_then_finish':extra={'kind':'item_finish','binding':binding}
        else:extra={'kind':'session_finish','exit':5}
        rows.append(dict(extra,sequence=1,identity={'nonce':'actual'},phase_clock_sha256='clock'))
    elif trace=='bool_backend_exit':rc=True
    elif trace=='bool_session_exit':rows[-1]['exit']=True
    with pytest.raises(ValueError) as failure:decide(unit,rows,rc)
    if trace=='failed_call_exit5':
        assert len(failure.value.failure_observed)==1
        assert failure.value.failure_observed[0]['report']['outcome']=='failed'

def final_fixture(rows, rc):
    """Inert final receipt data; no runtime/plugin-autoload qualification."""
    ref = {'name': 'anyio', 'module': 'anyio.pytest_plugin', 'path': '/p/plugin.py',
           'sha256': 'b'*64, 'entrypoint': '/p/entry_points.txt', 'entrypoint_sha256': 'c'*64}
    pins = {ref['path']: ref['sha256'], ref['entrypoint']: ref['entrypoint_sha256']}
    plan = {'runtime_plugins': [ref], 'runtime_dependency_pins': pins,
            'runtime_import_origins': {ref['module']: ref['path']},
            'runtime_autoload_entrypoints': {ref['entrypoint']: ref['entrypoint_sha256']}}
    result = {'receipt_schema': 3, 'collect_only': False, 'runtime_plugin_closure_validated': True,
              'runtime_events': rows, 'identity': {'nonce': 'actual'}, 'phase_clock_sha256': 'clock',
              'exit': rc, 'inputs': dict(pins), 'inputs_unchanged': True,
              'origins': {ref['module']: ref['path']}}
    return plan, pins, result


@pytest.mark.parametrize('gap', ['absent_result', 'validation_bit', 'body', 'entrypoint',
                                  'origin', 'changed_inputs', 'lexical_join'])
def test_final_failed_trace_refuses_incomplete_input_plugin_closure(gap):
    unit, rows = events(('passed', 'failed', 'passed'), exit_code=1)
    plan, pins, result = final_fixture(rows, 1)
    if gap == 'absent_result': result = None
    elif gap == 'validation_bit': result.pop('runtime_plugin_closure_validated')
    elif gap == 'body': result['inputs'].pop('/p/plugin.py')
    elif gap == 'entrypoint': result['inputs'].pop('/p/entry_points.txt')
    elif gap == 'origin': result['origins']['anyio.pytest_plugin'] = '/foreign.py'
    elif gap == 'changed_inputs': result['inputs_unchanged'] = False
    census = API['census'](unit, rows, {'nonce': 'actual'}, 'clock', 1, result)
    closed = API['final_runtime_decision'](census, result, plan, pins,
        final_input_error='bad join' if gap == 'lexical_join' else None)
    assert closed['status'] == 'RUNTEST_EVIDENCE_STOP'
    assert closed['final_runtime_evidence_validated'] is False
    assert closed['failure_observed'] == census['failure_observed']
    assert len(closed['partition']['failed']) == 1

def test_actual_post_autoload_check_after_failed_session_cannot_admit_without_result():
    # Execute only the exact existing validator body later, with an in-memory
    # incorrect plugin origin. Actual on-disk input bytes are not altered.
    import ast, hashlib, json
    from types import SimpleNamespace
    source = pathlib.Path(__file__).with_name('controls.py')
    plan = json.loads(source.with_name('runtest-plan.json').read_text())
    ref = plan['runtime_plugins'][0]
    digest = hashlib.sha256(pathlib.Path(ref['path']).read_bytes()).hexdigest()
    assert digest == ref['sha256']
    tree = ast.parse(source.read_bytes())
    function = next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == 'verify_runtime_plugin')
    namespace = {'runtest_file': 'f000', 'runtime_plan': plan,
                 'hashlib': hashlib, 'pathlib': pathlib}
    exec(compile(ast.Module(body=[function], type_ignores=[]), str(source), 'exec'), namespace)
    unit, rows = events(('passed', 'failed', 'passed'), exit_code=1)
    assert rows[-1]['kind'] == 'session_finish' and rows[-1]['exit'] == 1
    plugin = SimpleNamespace(__name__=ref['module'], __file__=ref['path'],
                             __spec__=SimpleNamespace(origin='/foreign/in-memory.py'))
    with pytest.raises(ValueError, match='LOADED_PLUGIN_ORIGIN_DRIFT'):
        namespace['verify_runtime_plugin'](plugin, ref['name'])
    assert hashlib.sha256(pathlib.Path(ref['path']).read_bytes()).hexdigest() == digest
    model_plan, pins, _ = final_fixture(rows, 1)
    census = API['census'](unit, rows, {'nonce': 'actual'}, 'clock', 1)
    closed = API['final_runtime_decision'](census, None, model_plan, pins)
    assert closed['status'] == 'RUNTEST_EVIDENCE_STOP' and closed['failure_observed']


@pytest.mark.parametrize('complete', [False, True])
def test_exact_owner_runtest_check_preserves_failure_and_final_evidence_fence(complete):
    import ast
    source = pathlib.Path(__file__).parent.parent/'owner-helper/owner_observer.py'
    function = next(n for n in ast.parse(source.read_bytes()).body
                    if isinstance(n, ast.FunctionDef) and n.name == 'runtest_check')
    unit, rows = events(('passed', 'failed', 'passed'), exit_code=1)
    plan, pins, result = final_fixture(rows, 1)
    plan.update(schema=1, units=[unit], required_not_run=[], missing=[])
    class DataPath:
        def exists(self): return True
        def read_bytes(self):
            return ('\n'.join(__import__('json').dumps(row) for row in rows)+'\n').encode()
    class DataRun:
        def __truediv__(self, name): return DataPath()
    from types import SimpleNamespace
    namespace = {'runpy': SimpleNamespace(run_path=lambda path: API),
                 'SOURCE': pathlib.Path('/inert/source'), 'read': lambda path: plan,
                 'sha': lambda path: 'clock', 'json': __import__('json'),
                 'require': API['require'], 'node_key': lambda actual: actual['nodeid']}
    exec(compile(ast.Module(body=[function], type_ignores=[]), str(source), 'exec'), namespace)
    closed = namespace['runtest_check'](DataRun(),
        {'runtest_file': 'f000', 'selection': '@native', 'run_name': 'model'},
        result if complete else {}, {'identity': {'nonce': 'actual'}}, 1, False, pins)
    assert closed['status'] == ('RUNTEST_PRODUCT_FAIL' if complete else 'RUNTEST_EVIDENCE_STOP')
    assert closed['final_runtime_evidence_validated'] is complete
    assert closed['failure_observed'] and closed['actual_event_prefix'] == rows
