SOURCE = Path(__file__).resolve().parents[1] / 'owner-helper/private_network.py'


def load_model():
    tree = ast.parse(SOURCE.read_text())
    nodes = [n for n in tree.body if isinstance(n, (ast.Assign, ast.FunctionDef, ast.ClassDef))]
    clock = {'now': 100.0}
    os = NS(getuid=lambda:1000, get_inheritable=lambda fd:False,
            uname=lambda:NS(sysname='Linux', machine='x86_64'),
            memfd_create=Mock(side_effect=AssertionError('no real memfd')),
            pread=Mock(side_effect=AssertionError('no real pread')),
            MFD_CLOEXEC=1, MFD_ALLOW_SEALING=2)
    env = dict(Path=Path, ipaddress=ipaddress, fcntl=fcntl, hashlib=hashlib, math=math,
               stat=stat, os=os, time=NS(monotonic=lambda:clock['now']),
               select=NS(select=Mock(return_value=([], [], []))),
               subprocess=NS(Popen=Mock(side_effect=AssertionError('no real subprocess'))))
    exec(compile(ast.Module(body=nodes, type_ignores=[]), str(SOURCE), 'exec'), env)
    obj = env['NetworkAttachment'].__new__(env['NetworkAttachment'])
    obj.failed = obj.closed = obj.released = False
    obj.deadline, obj.startup_end = 200.0, 105.0
    obj.proc=NS(returncode=None); obj.pidfd=11; obj.donor_proc=NS(returncode=None)
    obj.session=NS(process=obj.donor_proc, _runtest_pidfd=12, runtest_deadline=200.0,
                   runtest_stop_requested=False, runtest_persistence_error='', runtest_pidfd_error='',
                   runtest_outer={'unit':'original'})
    obj.owner=NS(deadline=200.0, pre={'reserve_seconds':10}, stop_requested=lambda:False,
                 require_current=Mock())
    obj.block_write=13; obj.block_identity=[1,2]; obj._pipe=Mock(); obj._close=Mock()
    env['os'].write=Mock(return_value=1)
    return env, obj, clock


@pytest.mark.parametrize('cut', ['entry', 'owner', 'stop', 'outer_mismatch', 'good'])
def test_same_startup_anchor_fences_owner_return(cut):
    env, obj, clock=load_model()
    if cut=='entry': clock['now']=105
    elif cut=='owner': obj.owner.require_current.side_effect=lambda outer:clock.update(now=105)
    elif cut=='stop': obj.owner.stop_requested=lambda:True
    elif cut=='outer_mismatch': obj.session.runtest_deadline=201
    if cut=='good': obj._startup_current()
    else:
        with pytest.raises(env['NetworkRefusal']): obj._startup_current()
    assert obj.startup_end==105 and obj.deadline==200


@pytest.mark.parametrize('cut', ['entry', 'owner', 'pipe', 'helper_exit', 'donor_exit',
                                  'select_late', 'write_error', 'short_write', 'good'])
def test_final_release_requires_fresh_both_retained_pidfds_after_owner_check(cut):
    env,obj,clock=load_model(); calls=[]
    def owner(outer):
        calls.append('owner')
        if cut=='owner': clock['now']=105
    obj.owner.require_current.side_effect=owner
    def poll(fds, *args):
        calls.append(('pidfds',list(fds)))
        if cut=='select_late': clock['now']=105
        return ([11] if cut=='helper_exit' else [12] if cut=='donor_exit' else [],[],[])
    env['select'].select.side_effect=poll
    if cut=='entry': clock['now']=105
    if cut=='pipe': obj._pipe.side_effect=lambda *args:clock.update(now=105)
    if cut=='write_error': env['os'].write.side_effect=OSError('inert write failure')
    if cut=='short_write': env['os'].write.return_value=0
    if cut=='good':
        obj._release()
        assert obj.released
        env['os'].write.assert_called_once_with(13,b'1')
        assert calls[-1]==('pidfds',[11,12])
        obj._close.assert_called_once_with('block_write')
    else:
        with pytest.raises((env['NetworkRefusal'],OSError)): obj._release()
        assert not obj.released
        if cut not in ('write_error','short_write'): env['os'].write.assert_not_called()
    assert obj.startup_end==105 and obj.deadline==200


@pytest.mark.parametrize('cut', ['select_late', 'read_late', 'owner_late', 'helper_exit', 'donor_exit', 'good'])
def test_ready_eof_does_not_skip_post_return_fences(cut):
    env,obj,clock=load_model(); obj.ready_read=14; chunks=iter([b'1',b''])
    def poll(*args):
        if cut=='select_late':clock['now']=105
        return ([14,11] if cut=='helper_exit' else [14,12] if cut=='donor_exit' else [14],[],[])
    def read(*args):
        chunk=next(chunks)
        if cut=='read_late' and not chunk:clock['now']=105
        if cut=='owner_late' and not chunk:
            obj.owner.require_current.side_effect=lambda outer:clock.update(now=105)
        return chunk
    env['select'].select.side_effect=poll;env['os'].read=read
    if cut=='good':obj._ready();obj._close.assert_called_once_with('ready_read')
    else:
        with pytest.raises(env['NetworkRefusal']):obj._ready()
        obj._close.assert_not_called()
    env['os'].write.assert_not_called()


@pytest.mark.parametrize('cut', ['binary_hash', 'pidfd_preflight', 'good'])

    env,obj,clock=load_model()

    def sha(path):
        if cut=='binary_hash':clock['now']=105
        return digest
    obj.owner.h=NS(NETWORK_PATCH_SHA256='b'*64,sha=sha)

                               stat=lambda:NS(st_mode=0o555),__str__=lambda self:'/reviewed/slirp')

        with pytest.raises(AssertionError,match='no real subprocess'):obj._start_helper()

        env['subprocess'].Popen.assert_not_called()


@pytest.mark.parametrize('cut', ['write_seal', 'grow_seal', 'shrink_seal', 'seal_seal',
                                  'linked', 'wrong_inode', 'wrong_size', 'content', 'inheritable', 'good'])
def test_resolver_requires_kernel_seals_for_all_fd_aliases(cut):
    env,obj,clock=load_model(); data=b'nameserver 192.168.12.2\n'
    obj.resolver_fd=20;obj.resolver_identity=[3,4];obj.resolver_size=len(data)
    obj.resolver_sha=hashlib.sha256(data).hexdigest()
    # Exact product compatibility ABI, not invented fcntl attributes or kernel proof.
    abi=env['LINUX_SEAL_ABI']; _, get_seals, seals=env['linux_seal_constants']()
    removed={'write_seal':abi['F_SEAL_WRITE'],'grow_seal':abi['F_SEAL_GROW'],
             'shrink_seal':abi['F_SEAL_SHRINK'],'seal_seal':abi['F_SEAL_SEAL']}.get(cut,0)
    def sealed(fd,op):
        assert fd==20 and op==get_seals
        return seals & ~removed
    env['fcntl']=NS(fcntl=sealed)
    env['os'].fstat=lambda fd:NS(st_mode=stat.S_IFREG|0o777, st_uid=1000,
        st_nlink=1 if cut=='linked' else 0,st_dev=3,st_ino=99 if cut=='wrong_inode' else 4,
        st_size=1 if cut=='wrong_size' else len(data))
    env['os'].get_inheritable=lambda fd:cut=='inheritable'
    env['os'].pread=lambda *args:b'changed' if cut=='content' else data
    if cut=='good':obj._resolver_current()
    else:
        with pytest.raises(env['NetworkRefusal']):obj._resolver_current()
