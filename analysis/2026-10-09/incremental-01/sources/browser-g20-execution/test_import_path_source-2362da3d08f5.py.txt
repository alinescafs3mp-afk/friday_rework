import errno, importlib.util, os, pathlib, stat

def module():
    path = pathlib.Path(__file__).with_name('import_path_readiness.py')
    spec = importlib.util.spec_from_file_location('isolated_import_path_source_test', path)
    obj = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(obj)
    return obj


def fixture_probe(tmp_path, monkeypatch):
    api = module()
    parent = tmp_path / 'pinned-parent'
    parent.mkdir()
    leaf = parent / 'python314.zip'
    chain = list(reversed(parent.parents)) + [parent]
    m = {'declared_absent_path': str(leaf),
         'ancestors': [{'path': str(p), 'identity': api.identity(p.lstat())} for p in chain]}
    monkeypatch.setattr(api, 'ARCHIVE', str(leaf))
    return api, m, leaf


def test_same_genuine_kernel_enoent_and_no_content_open(tmp_path, monkeypatch):
    api, m, leaf = fixture_probe(tmp_path, monkeypatch)
    actual_stat, actual_open = api._stat, api._open
    saved = []
    flags_seen = []
    def metadata_stat(*args, **kwargs):
        try:
            return actual_stat(*args, **kwargs)
        except FileNotFoundError as error:
            saved.append(error)
            raise
    def metadata_open(*args, **kwargs):
        flags_seen.append(args[1])
        assert args[1] == os.O_PATH | os.O_DIRECTORY | os.O_NOFOLLOW | os.O_CLOEXEC
        return actual_open(*args, **kwargs)
    monkeypatch.setattr(api, '_stat', metadata_stat)
    monkeypatch.setattr(api, '_open', metadata_open)
    with pytest.raises(FileNotFoundError) as result:
        api.failure_only_open(m, str(leaf), os.O_RDONLY, str(leaf))
    assert result.value is saved[0]
    assert result.value.errno == errno.ENOENT
    assert result.value.filename == leaf.name
    assert len(flags_seen) == len(m['ancestors'])
    assert not leaf.exists()


@pytest.mark.parametrize('kind', ['file', 'directory', 'link', 'dangling', 'fifo'])
def test_any_existing_leaf_refuses_without_open(tmp_path, monkeypatch, kind):
    api, m, leaf = fixture_probe(tmp_path, monkeypatch)
    if kind == 'file': leaf.write_bytes(b'unreviewed bytes')
    elif kind == 'directory': leaf.mkdir()
    elif kind == 'fifo': os.mkfifo(leaf)
    elif kind == 'link': leaf.symlink_to(tmp_path)
    else: leaf.symlink_to(tmp_path / 'missing')
    with pytest.raises(api.ProbeRefusal, match='DECLARED_ABSENT_LEAF_PRESENT'):
        api.failure_only_open(m, str(leaf), os.O_RDONLY, os.path.realpath(leaf))


@pytest.mark.parametrize('flags', [os.O_WRONLY, os.O_RDWR, os.O_CREAT,
                                  os.O_TRUNC, os.O_APPEND, os.O_WRONLY | os.O_CREAT])
def test_every_write_route_refuses_before_metadata(tmp_path, monkeypatch, flags):

def test_appearance_after_metadata_preserves_same_error_and_next_refuses(tmp_path, monkeypatch):
    api, m, leaf = fixture_probe(tmp_path, monkeypatch)
    real_stat = api._stat
    saved = []
    def after_absence(*args, **kwargs):
        try:
            return real_stat(*args, **kwargs)
        except FileNotFoundError as error:
            saved.append(error)
            leaf.write_bytes(b'appeared after kernel negative observation')
            raise
    monkeypatch.setattr(api, '_stat', after_absence)
    with pytest.raises(FileNotFoundError) as result:
        api.failure_only_open(m, str(leaf), os.O_RDONLY, str(leaf))
    assert result.value is saved[0]
    monkeypatch.setattr(api, '_stat', real_stat)
    with pytest.raises(api.ProbeRefusal, match='DECLARED_ABSENT_LEAF_PRESENT'):
        api.failure_only_open(m, str(leaf), os.O_RDONLY, str(leaf))


def test_exact_directory_descriptor_anchor(tmp_path, monkeypatch):
    api, m, leaf = fixture_probe(tmp_path, monkeypatch)
    fd = os.open(leaf.parent, os.O_PATH | os.O_DIRECTORY)
    try:
        with pytest.raises(FileNotFoundError):
            api.failure_only_open(m, '/proc/self/fd/' + str(fd) + '/' + leaf.name,
                                  os.O_RDONLY, str(leaf))
    finally:
        os.close(fd)
