# Exact trusted offline test sources, independently inspected by root. No OS
# sandbox / hostile-code claim. Preserve process denial; one private AF_UNIX
# bind is needed to prove nonregular input rejection. No listen/connect route.
import socket
raw_socket_listen=socket.socket.listen
def deny(event):
 os.write(2,('OFFLINE_EFFECT_DENIED '+event+'\n').encode());os._exit(125)
def no_listen(self,*args,**kwargs):deny('socket.listen')
socket.socket.listen=no_listen
unix_sockets=set()
def audit(event,args):
 if event in ('subprocess.Popen','os.system','os.fork','os.forkpty','os.posix_spawn','os.exec','pty.spawn','os.kill','os.killpg','os.pidfd_open','signal.pidfd_send_signal'):
  deny(event)
 if event=='socket.__new__':
  obj,family,kind,protocol=args
  if family!=socket.AF_UNIX or kind!=socket.SOCK_STREAM or protocol!=0:deny(event)
  unix_sockets.add(id(obj))
 elif event=='socket.bind':
  obj,address=args
  if id(obj) not in unix_sockets or not isinstance(address,str) or not address or address[0]=='\0':deny(event)
  path=Path(address)
  if not path.is_absolute() or not path.parent.resolve().is_relative_to(TMPROOT/'pytesttmp') or path.exists() or path.is_symlink():deny(event)
 elif event.startswith('socket.'):
  deny(event)
sys.addaudithook(audit)
