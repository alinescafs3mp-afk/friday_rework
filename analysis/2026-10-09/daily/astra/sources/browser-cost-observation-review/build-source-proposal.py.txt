"""AST/source transformation only; never imports or executes candidate modules."""
from pathlib import Path
import ast, difflib, hashlib, json, textwrap
W=Path('<ASTRA_WORK>')
OUT=W/'browser-cost-observation-review'
BASE=W/'browser-observer-dispatch-join/native-preparation/positive'

def profile_parts():
 return '''
 _frw_cost_entered=False
 _frw_cost_cpu=None
 _frw_cost_wall=None
 try:
  _frw_cost=fence.FRW_COST
  if _frw_cost['active']:
   if fence.FRW_COST_IDENT()!=_frw_cost['owner']:
    _frw_cost['valid']=False;_frw_cost['error']='FOREIGN_CALLBACK_THREAD'
    _frw_cost['foreign_calls']+=1
   else:
    _frw_cost_entered=True
    _frw_cost['profile_depth']+=1
    _frw_cost['profile'][0]+=1
    if _frw_cost['profile_depth']!=1:
     _frw_cost['valid']=False;_frw_cost['error']='NESTED_PROFILE'
    _frw_cost_cpu=fence.FRW_COST_CPU()
    _frw_cost_wall=fence.FRW_COST_WALL()
 except BaseException:
  fence.FRW_COST['valid']=False;fence.FRW_COST['error']='PROFILE_MEASUREMENT_ENTRY'
 try:
''', '''
 finally:
  if _frw_cost_entered:
   try:
    _frw_cost_dc=fence.FRW_COST_CPU()-_frw_cost_cpu
    _frw_cost_dw=fence.FRW_COST_WALL()-_frw_cost_wall
    if _frw_cost_dc<0 or _frw_cost_dw<0:
     _frw_cost['valid']=False;_frw_cost['error']='NEGATIVE_PROFILE_INTERVAL'
    else:
     _frw_cost_row=_frw_cost['profile']
     _frw_cost_row[1]+=1;_frw_cost_row[2]+=_frw_cost_dc;_frw_cost_row[3]+=_frw_cost_dw
   except BaseException:
    fence.FRW_COST['valid']=False;fence.FRW_COST['error']='PROFILE_MEASUREMENT_EXIT'
   finally:
    _frw_cost['profile_depth']-=1
'''

def audit_parts():
 return '''
 _frw_cost_entered=False
 _frw_cost_outer=False
 _frw_cost_cpu=None
 _frw_cost_wall=None
 try:
  _frw_cost=FRW_COST
  if _frw_cost['active']:
   if FRW_COST_IDENT()!=_frw_cost['owner']:
    _frw_cost['valid']=False;_frw_cost['error']='FOREIGN_CALLBACK_THREAD'
    _frw_cost['foreign_calls']+=1
   else:
    _frw_cost_entered=True
    _frw_cost['audit_calls']+=1
    _frw_cost['audit_depth']+=1
    _frw_cost_outer=_frw_cost['audit_depth']==1
    if _frw_cost_outer:
     _frw_cost_key=event if event in FRW_COST_AUDIT_KEYS else 'other'
     _frw_cost_row=_frw_cost['audit'][_frw_cost_key]
     _frw_cost_row[0]+=1
     _frw_cost_inside_profile=_frw_cost['profile_depth']>0
     _frw_cost_cpu=FRW_COST_CPU()
     _frw_cost_wall=FRW_COST_WALL()
    else:
     _frw_cost['nested_audit_calls']+=1
 except BaseException:
  FRW_COST['valid']=False;FRW_COST['error']='AUDIT_MEASUREMENT_ENTRY'
 try:
''', '''
 finally:
  if _frw_cost_entered:
   try:
    if _frw_cost_outer:
     _frw_cost_dc=FRW_COST_CPU()-_frw_cost_cpu
     _frw_cost_dw=FRW_COST_WALL()-_frw_cost_wall
     if _frw_cost_dc<0 or _frw_cost_dw<0:
      _frw_cost['valid']=False;_frw_cost['error']='NEGATIVE_AUDIT_INTERVAL'
     else:
      _frw_cost_row[1]+=1;_frw_cost_row[2]+=_frw_cost_dc;_frw_cost_row[3]+=_frw_cost_dw
      if _frw_cost_inside_profile:
       _frw_cost['audit_inside_profile'][0]+=1
       _frw_cost['audit_inside_profile'][1]+=_frw_cost_dc
       _frw_cost['audit_inside_profile'][2]+=_frw_cost_dw
   except BaseException:
    FRW_COST['valid']=False;FRW_COST['error']='AUDIT_MEASUREMENT_EXIT'
   finally:
    _frw_cost['audit_depth']-=1
'''

globals_source='''

# SOURCE PROPOSAL: observation counters only; no authority input reads them.
# Activate once at the root-owned collection boundary; never reset after activation.
# Rows are [outer entries, completed outer intervals, thread CPU ns, wall ns].
FRW_COST_IDENT=_thread.get_ident
FRW_COST_CPU=time.thread_time_ns
FRW_COST_WALL=time.perf_counter_ns
FRW_COST_AUDIT_KEYS=('open','shutil.copyfile','shutil.copystat','os.chown','object.__getattr__')
FRW_COST={'active':False,'owner':None,'valid':True,'error':None,
 'foreign_calls':0,'profile_depth':0,'audit_depth':0,
 'profile':[0,0,0,0],'audit_calls':0,'nested_audit_calls':0,
 'audit_inside_profile':[0,0,0],
 'audit':{key:[0,0,0,0] for key in (*FRW_COST_AUDIT_KEYS,'other')}}
'''
checks=[];patches=[]
for filename,name,parts in [('run.py','directory_profile',profile_parts()),('private_effect_fence.py','audit',audit_parts())]:
 original=(BASE/filename).read_text();old=ast.parse(original)
 node=next(n for n in old.body if isinstance(n,ast.FunctionDef) and n.name==name)
 assert not any(isinstance(n,ast.Name) and n.id.startswith('_frw_cost') for n in ast.walk(node))
 lines=original.splitlines(keepends=True)
 pre,post=(part.replace('except BaseException:', 'except Exception:') for part in parts)
 original_body=''.join(lines[node.lineno:node.end_lineno])
 updated=''.join(lines[:node.lineno])+pre.lstrip('\n')+textwrap.indent(original_body,' ')+post.lstrip('\n')+''.join(lines[node.end_lineno:])
 if filename=='private_effect_fence.py':updated+=globals_source
 new=ast.parse(updated);replacement=next(n for n in new.body if isinstance(n,ast.FunctionDef) and n.name==name)
 wrapped=next(n for n in replacement.body if isinstance(n,ast.Try) and n.finalbody)
 assert ast.dump(ast.Module(body=wrapped.body,type_ignores=[]),include_attributes=False)==ast.dump(ast.Module(body=node.body,type_ignores=[]),include_attributes=False)
 before_rest=[n for n in old.body if n is not node]
 after_rest=[n for n in new.body[:len(old.body)] if n is not replacement]
 assert ast.dump(ast.Module(body=before_rest,type_ignores=[]),include_attributes=False)==ast.dump(ast.Module(body=after_rest,type_ignores=[]),include_attributes=False)
 # The original function object is defined once and installed at original sites.
 assert len([n for n in ast.walk(new) if isinstance(n,ast.FunctionDef) and n.name==name])==1
 dest=OUT/(filename+'.source-proposal')
 dest.write_text(updated)
 patches.extend(difflib.unified_diff(original.splitlines(keepends=True),updated.splitlines(keepends=True),fromfile='original/'+filename,tofile='source-proposal/'+filename))
 checks.append({'file':filename,'base_sha256':hashlib.sha256(original.encode()).hexdigest(),'proposal_path':str(dest),'proposal_sha256':hashlib.sha256(updated.encode()).hexdigest(),'ast_parse':True,'original_callback_body_exact_ast':True,'all_other_original_toplevel_nodes_exact_ast':True,'single_callback_definition_no_wrapper':True,'candidate_executed':False})
(OUT/'callback-cost-source-proposal.diff').write_text(''.join(patches))
(OUT/'source-proposal-static-check.json').write_text(json.dumps({'state':'AST_ONLY_NOT_INTEGRATED_OR_RUN','checks':checks},indent=2)+'\n')
print(json.dumps(checks,indent=2))
