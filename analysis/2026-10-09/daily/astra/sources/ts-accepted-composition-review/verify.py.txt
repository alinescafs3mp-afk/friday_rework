def check(name, ok, details=None):
    checks.append({'name': name, 'ok': bool(ok), 'details': details})

def digest(p):
    p = str(p)
    if p not in hashes:
        with open(p, 'rb') as f: hashes[p] = hashlib.file_digest(f, 'sha256').hexdigest()
    return hashes[p]

def read(p): return json.loads(Path(p).read_text())
def ref(p): return {'path': str(p), 'sha256': digest(p)}

def verify_ref(r):
    try: check('SHA ' + r['path'], digest(r['path']) == r['sha256'])
    except (OSError, ValueError) as ex: check('SHA ' + r['path'], False, str(ex))

def refs(value):
    if isinstance(value, dict):
        if isinstance(value.get('path'), str) and isinstance(value.get('sha256'), str):
            yield value
        for sub in value.values(): yield from refs(sub)
    elif isinstance(value, list):
        for sub in value: yield from refs(sub)
