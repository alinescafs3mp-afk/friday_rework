"""Independent successor delta: source/hash/AST only, no candidate execution."""
from pathlib import Path
import ast, hashlib, json, os, resource
os.sched_setaffinity(0, {0, 1})
resource.setrlimit(resource.RLIMIT_AS, (2 * 1024**3, 2 * 1024**3))
W = Path('<ASTRA_WORK>')
OLD = W / 'callback-partial-chronology'
NEW = W / 'callback-partial-chronology-r2'
OUT = W / 'callback-partial-chronology-review'
refs = {}
checks = []
def ref(path):
    path = Path(path); data = path.read_bytes()
    row = {'path': str(path), 'sha256': hashlib.sha256(data).hexdigest(), 'bytes': len(data)}
    refs[str(path)] = row
    return row
def read(path):
    ref(path)
    return json.loads(Path(path).read_bytes())
def checked(label, condition, detail):
    assert condition, (label, detail)
    checks.append({'check': label, 'passed': True, 'detail': detail})
summary = read(NEW / 'result.json')
checked('successor_anchor', ref(NEW / 'result.json')['sha256'] == 'b1609d9308de2a9f7c894c016728784f9839ef84033f2944a0f02347e13f84ec', 'Root supplied continuation within same assignment and original900s.')
checked('predecessor_anchor', ref(OLD / 'result.json')['sha256'] == 'a151395a86b38ca008ba9f933f6fff79d841c597f0fa7d0d599c22a254094abf', 'Original reviewed snapshot retained.')
for item in summary['refs']:
    for key in ('current', 'previous'):
        checked('manifest_ref', ref(item[key]['path']) == item[key], item['relative'] + ':' + key)
    relative = item['relative']
    before = (OLD / relative).read_text().replace(str(OLD), str(NEW))
    after = (NEW / relative).read_text()
    if relative.endswith('.py') and relative != 'qualification/telemetry_projection.py':
        checked('retained_semantics_path_rebase_only', before == after, relative)
        ast.parse(after)
projection = NEW / 'qualification/telemetry_projection.py'
old_projection = (OLD / 'qualification/telemetry_projection.py').read_text()
expected_old = "state['callback_cost_latest'] = fence.callback_cost_snapshot('snapshot')"
expected_new = "fence.callback_cost_store_latest(state, 'snapshot')"
checked('exact_projection_one_line_fix', old_projection.count(expected_old) == 1 and old_projection.replace(expected_old, expected_new) == projection.read_text(), 'One exact seam assertion changed; all other projection source unchanged, including full original AST equality.')
runner = ast.parse((NEW / 'diagnostic-positive/run.py').read_text())
snapshot = next(n for n in runner.body if isinstance(n, ast.FunctionDef) and n.name == 'snapshot')
checked('projection_assert_matches_actual_AST', ast.unparse(snapshot.body[0]) == expected_new, {'actual_line': snapshot.body[0].lineno, 'statement': expected_new})
projection_ast = ast.parse(projection.read_text())
pf = next(n for n in projection_ast.body if isinstance(n, ast.FunctionDef) and n.name == 'projected')
checked('projection_retains_full_original_AST_assertion', any(isinstance(n, ast.Assert) and ast.unparse(n.test) == 'dump(new) == dump(old)' for n in ast.walk(pf)), 'No weakened equality, substituted original body or extra removal introduced by successor.')
for relative in ('diagnostic-positive/launch-pins.json', 'diagnostic-positive/execution-package-pins.json', 'qualification/projection-pins.json', 'qualification/probe-input-pins.json'):
    before = read(OLD / relative); after = read(NEW / relative)
    checked('affected_map_key_rebase_only', set(after) == {p.replace(str(OLD), str(NEW)) for p in before}, relative)
    for path, sha in after.items():
        checked('affected_map_current_hash', ref(path)['sha256'] == sha, relative + ':' + path)
old_inputs = read(OLD / 'qualification/input-pins.json')
new_inputs = read(NEW / 'qualification/input-pins.json')
old_ext = {p: h for p, h in old_inputs.items() if not p.startswith(str(OLD) + '/')}
new_ext = {p: h for p, h in new_inputs.items() if not p.startswith(str(NEW) + '/')}
checked('all_unchanged_external_map_entries', old_ext == new_ext, len(new_ext))
new_local = {p: h for p, h in new_inputs.items() if p.startswith(str(NEW) + '/')}
checked('local_input_set_rebase_only', set(new_local) == {p.replace(str(OLD), str(NEW)) for p in old_inputs if p.startswith(str(OLD) + '/')}, len(new_local))
for path, sha in new_local.items():
    checked('current_local_input_hash', ref(path)['sha256'] == sha, path)
for relative in ('diagnostic-positive/source-contract.json', 'diagnostic-positive/effect-admission-proposal.json', 'qualification/source-contract.json'):
    content = read(NEW / relative)
    def walk(value):
        if isinstance(value, dict):
            if isinstance(value.get('path'), str) and isinstance(value.get('sha256'), str) and value['path'].startswith(str(NEW) + '/'):
                actual = ref(value['path'])
                checked('current_contract_ref', actual['sha256'] == value['sha256'] and ('bytes' not in value or actual['bytes'] == value['bytes']), relative + ':' + value['path'])
            for child in value.values(): walk(child)
        elif isinstance(value, list):
            for child in value: walk(child)
    walk(content)
    checked('no_prior_successor_path_in_contract', str(OLD) + '/' not in json.dumps(content), relative)
diag = read(NEW / 'diagnostic-positive/source-contract.json')
checked('successor_admission_hashes', diag['admission_fields']['source_package_sha256'] == ref(NEW / 'diagnostic-positive/execution-package-pins.json')['sha256'] and diag['admission_fields']['effect_proposal_sha256'] == ref(NEW / 'diagnostic-positive/effect-admission-proposal.json')['sha256'], 'Bindings coherent; not an admission.')
report = {'schema': 'friday.callback-partial-chronology.successor-delta-review.v1', 'method': 'Own source/hash/AST only. No candidate imports, compile, exec, test collection/tests or profiling.',
          'state': 'R1_DEPENDENCY_FIXED_AT_SOURCE_LEVEL_ACTUAL_QUALIFICATION_NOT_RUN', 'checks': checks,
          'current_input_map': {'total': len(new_inputs), 'unchanged_external_entries': len(new_ext), 'verified_current_entries': len(new_local), 'historical_census_repeated': False},
          'refs': sorted(refs.values(), key=lambda r: r['path'])}
(OUT / 'successor-verification.json').write_text(json.dumps(report, indent=2, ensure_ascii=False) + '\n')
print(json.dumps({'checks': len(checks), 'refs': len(refs), 'state': report['state'], 'input_map': report['current_input_map'], 'artifact': ref(OUT / 'successor-verification.json')}))
