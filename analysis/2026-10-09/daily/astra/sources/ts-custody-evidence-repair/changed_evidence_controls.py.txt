"""Prepared NOT_RUN controls. Root imports only after owner qualification.

Uses genuine current positive evidence plus three minimal payload/coverage
mutations. Re-numbering removes the incidental sequence-gap rejection so the
actual semantic join must reject. No candidate/native process is launched here.
"""
import json
from pathlib import Path


def changed_controls(join, genuine_run, budget, owner, max_bytes, destination):
    baseline = Path(genuine_run)
    # Positive is the actual current consumer on the actual current evidence.
    join(baseline, budget, owner, max_bytes)
    originals = {p.name: p.read_bytes() for p in baseline.iterdir()
                 if p.name in ('clock.json', 'pre-exec-os.json', 'progress.json')
                 or p.name.startswith('guard-') and p.suffix == '.jsonl'}
    def mutate(kind):
        streams = {name: [json.loads(line) for line in raw.splitlines()]
                   for name,raw in originals.items() if name.startswith('guard-')}
        rows = [row for stream in streams.values() for row in stream]
        if kind == 'plugin-output-hash':
            row = next(r for r in rows if r['event'] == 'VITE_PLUGIN_OUTPUT')
            row['codeSha256'] = ('0' if row['codeSha256'][0] != '0' else '1') + row['codeSha256'][1:]
        elif kind == 'graph-edge':
            row = next(r for r in rows if r['event'] == 'VITE_TRANSFORM_GRAPH' and r['imports'])
            row['imports'].pop()
        elif kind == 'shared-worker-chain':
            candidates = [r for r in rows if r['event'] == 'VITEST_WORKER_FETCH' and 'canonical' in r]
            worker = lambda r: (r['pid'],r['workerId'],r['poolId'])
            first = next(r for r in candidates if any(x['canonical'] == r['canonical'] and worker(x) != worker(r) for x in candidates))
            for name, stream in streams.items():
                streams[name] = [r for r in stream if not (worker(r) == worker(first) and r.get('canonical') == first['canonical']
                    and r['event'] in ('VITEST_WORKER_FETCH', 'VITE_WORKER_REQUEST', 'VITEST_EVAL_BEGIN',
                                       'VITEST_EVAL_DONE', 'VITE_RUNNER_EVAL_BEGIN', 'VITE_RUNNER_EVAL_DONE'))]
        target=Path(destination)/kind
        target.mkdir(mode=0o700,parents=False,exist_ok=False)
        for name,raw in originals.items():
            if name in streams:
                for seq,row in enumerate(streams[name],1): row['seq']=seq
                raw=(''.join(json.dumps(row)+'\n' for row in streams[name])).encode()
            (target/name).write_bytes(raw)
        try: join(target,budget,owner,max_bytes)
        except ValueError as error:
            expected = {
                'plugin-output-hash': {'actual_plugin_output_next_input_chain_broken'},
                'graph-edge': {'returned_graph_edges_changed_without_delta', 'graph_delta_previous_edges_changed', 'final_graph_edges_not_equal_to_observed_edges'},
                'shared-worker-chain': {'worker_importer_evaluation_missing', 'worker_static_import_chain_incomplete', 'worker_request_local_evaluation_missing'},
            }
            if str(error) not in expected[kind]: raise AssertionError('unrelated refusal: '+str(error)) from error
            return str(error)
        raise AssertionError('semantic mutation accepted: '+kind)
    return {kind:mutate(kind) for kind in ('plugin-output-hash','graph-edge','shared-worker-chain')}
