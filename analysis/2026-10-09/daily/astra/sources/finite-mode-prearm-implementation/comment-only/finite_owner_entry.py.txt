"""Existing Popen caller's preexec seam, BEFORE this finite owner exec.
Prepared source only. Importing this module installs nothing.
The caller must pin this body and actual original runtime clock before Popen.
No new service, worker, cgroup or renewed clock.
"""
import os
import resource
import signal

def finite_owner_entry_limits():
    # Reset sigaction flags as well as disposition; no SA_NOCLDWAIT/auto-reap.
    signal.signal(signal.SIGCHLD, signal.SIG_DFL)
    signal.pthread_sigmask(signal.SIG_UNBLOCK, {signal.SIGXCPU, signal.SIGCHLD})
    os.sched_setaffinity(0, {0, 1})
    resource.setrlimit(resource.RLIMIT_AS, (2 << 30, 2 << 30))
    # Installed here after fork: CPU accounting includes the earlier prefix,
    # but this callback does not retroactively bound it. The actual caller
    # birth envelope remains F4. Soft6 terminates the trusted non-execing root;
    # hard45 leaves room for LOWERED GNU/mode caps. Never raise a hard limit.
    signal.signal(signal.SIGXCPU, signal.SIG_DFL)
    resource.setrlimit(resource.RLIMIT_CPU, (6, 45))
    resource.setrlimit(resource.RLIMIT_CORE, (0, 0))
