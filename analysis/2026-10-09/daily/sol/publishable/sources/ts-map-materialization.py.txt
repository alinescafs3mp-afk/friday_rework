"""One-shot source materialization/hash/AST utility; no candidate import or execution.

Writes only this assignment's evidence directory. Existing pin schemas are reused.
No runtime clocks, admissions, receipts, tests, managers or kernel probes are made.
"""

import hashlib
import json
import os

E = Path(__file__).resolve().parent
C = E / 'candidate'

checks = []
verified = {}


def h(p):
    with Path(p).open('rb') as handle:
        return hashlib.file_digest(handle, 'sha256').hexdigest()


def ref(p):
    return {'path': str(p), 'sha256': h(p)}


def check(label, ok):
    if not ok:
        raise AssertionError(label)
    checks.append(label)


def read(p):
    return json.loads(Path(p).read_bytes())


def write(p, value):
    p = Path(p)
    check('Owned output ' + str(p.relative_to(E)), p.is_relative_to(E))
    p.write_text(json.dumps(value, sort_keys=True, indent=2) + '\n')
    os.chmod(p, 0o600)


def verify(p, sha):
    p = Path(p)
    actual = h(p)
    check('SHA ' + str(p), actual == sha)
    verified[str(p)] = sha


def relocate(path):
    if path == str(O) or path.startswith(str(O) + '/'):
        return str(C) + path[len(str(O)):]
    return path

    verify(W / relative, sha)
check('Exact independent composition accepted, no runtime grant',
      read(W / 'ts-cache-identity-controls-review/result-r2.json')['source_accepted'] is True

check('No candidate symlinks copied or followed', not any(p.is_symlink() for p in O.rglob('*')))

    raw = p.read_bytes()

    elif p.suffix in ('.py', '.js', '.mjs'):
