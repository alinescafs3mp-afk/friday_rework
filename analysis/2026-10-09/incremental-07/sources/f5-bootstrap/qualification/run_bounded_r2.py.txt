def exact_lock():
    fd=os.open(LOCK,os.O_RDONLY|os.O_NOFOLLOW)
    try:
        s=os.fstat(fd)
        identity=(s.st_dev,s.st_ino,s.st_uid,s.st_gid,s.st_nlink,s.st_size,stat.S_IMODE(s.st_mode))
        if identity!=LOCK_ID:raise RuntimeError('LOCK_IDENTITY_CHANGED')
        fcntl.flock(fd,fcntl.LOCK_EX|fcntl.LOCK_NB)
        fcntl.flock(fd,fcntl.LOCK_UN)
        return list(identity)
    finally:
        os.close(fd)

def proc_children():
    return [int(v) for v in Path('/proc/self/task/'+str(os.getpid())+'/children').read_text().split()]

def group_members(group):
    members=[];errors=[]
    for p in Path('/proc').iterdir():
        if not p.name.isdecimal():continue
        try:
            row=proc_stat(int(p.name))
            if row['pgid']==group:members.append(row)
        except (FileNotFoundError,ProcessLookupError):pass
        except (OSError,ValueError,IndexError) as error:
            errors.append({'pid':p.name,'error':repr(error)})
    return members,errors
