#define _GNU_SOURCE
/* GNU-only observer. SOURCE ONLY: never compiled/loaded in this assignment.
 * The exact unmodified GNU/libc/loader and a separately reviewed built ELF
 * must be pinned by the existing owner. No extra process, timer or allowance.
 */
#include <dlfcn.h>
#include <errno.h>
#include <limits.h>
#include <signal.h>
#include <stdint.h>
#include <stdlib.h>
#include <string.h>
#include <sys/stat.h>
#include <time.h>
#include <unistd.h>
#if __BYTE_ORDER__ != __ORDER_LITTLE_ENDIAN__ || UINTPTR_MAX != UINT64_MAX
#error "unqualified pipe/timer identity ABI"
#endif

struct record {
    char magic[8];
    uint64_t pid, ppid, kind, api, timer_id, clock_id, signo, duration_ns, arm_ns;
    char nonce[32];
};
_Static_assert(sizeof(struct record) == 112, "fixed pipe ABI");
_Static_assert(sizeof(struct record) <= PIPE_BUF, "one writer atomic record");
static int fd = -1, active, stage, armed;
static pid_t gnu_pid;
static timer_t created_timer;
static char nonce[32], interpreter[PATH_MAX], provider[PATH_MAX];
static int (*next_create)(clockid_t, struct sigevent *, timer_t *);
static int (*next_set)(timer_t, int, const struct itimerspec *, struct itimerspec *);
static int (*next_delete)(timer_t);
static unsigned int (*next_alarm)(unsigned int);
static pid_t (*next_fork)(void);
static int (*next_execvp)(const char *, char *const []);

static void refuse(void) { _exit(125); }
static uint64_t mono_ns(void) {
    struct timespec t;
    if (clock_gettime(CLOCK_MONOTONIC, &t) || t.tv_sec < 0) refuse();
    return (uint64_t)t.tv_sec * UINT64_C(1000000000) + (uint64_t)t.tv_nsec;
}
static void emit(uint64_t kind, uint64_t api, uint64_t id, uint64_t clock,
                 uint64_t signo, uint64_t duration, uint64_t at) {
    struct record r = { .magic = "FGNUv1", .pid = (uint64_t)getpid(),
        .ppid = (uint64_t)getppid(), .kind = kind, .api = api,
        .timer_id = id, .clock_id = clock, .signo = signo,
        .duration_ns = duration, .arm_ns = at };
    memcpy(r.nonce, nonce, sizeof nonce);
    ssize_t n;
    do { n = write(fd, &r, sizeof r); } while (n < 0 && errno == EINTR);
    /* <=PIPE_BUF, one GNU writer. A partial write never grants release. */
    if (n != (ssize_t)sizeof r) refuse();
}
static void *symbol(const char *name, const char *version) {
    dlerror();
    void *p = dlvsym(RTLD_NEXT, name, version);
    if (dlerror() || !p) refuse();
    Dl_info info;
    struct stat got, wanted;
    if (!dladdr(p, &info) || !info.dli_fname || stat(info.dli_fname, &got) ||
        stat(provider, &wanted) || got.st_dev != wanted.st_dev ||
        got.st_ino != wanted.st_ino) refuse();
    return p;
}
static const char *env(const char *name) {
    const char *v = getenv(name);
    if (!v || !*v) refuse();
    return v;
}
__attribute__((constructor)) static void initialise(void) {
    const char *n = env("FGNU_NONCE"), *f = env("FGNU_READY_FD");
    const char *i = env("FGNU_INTERPRETER"), *l = env("FGNU_LIBC");
    if (strlen(n) != sizeof nonce || strlen(i) >= sizeof interpreter ||
        strlen(l) >= sizeof provider) refuse();
    char *end = NULL;
    errno = 0;
    long value = strtol(f, &end, 10);
    if (errno || !end || *end || value < 3 || value > INT_MAX) refuse();
    fd = (int)value;
    memcpy(nonce, n, sizeof nonce);
    strcpy(interpreter, i); strcpy(provider, l);
    next_create = symbol("timer_create", "GLIBC_2.34");
    next_set = symbol("timer_settime", "GLIBC_2.34");
    next_delete = symbol("timer_delete", "GLIBC_2.34");
    next_alarm = symbol("alarm", "GLIBC_2.2.5");
    next_fork = symbol("fork", "GLIBC_2.2.5");
    next_execvp = symbol("execvp", "GLIBC_2.2.5");
    gnu_pid = getpid(); active = 1;
    emit(2, 0, 0, 0, 0, 0, 0); /* GNU constructor, not timer readiness. */
}
pid_t fork(void) {
    if (!active || getpid() != gnu_pid || stage || armed) refuse();
    pid_t child = next_fork();
    int saved = errno;
    if (child == 0) {
        /* Runs only after the initial kernel stop is released. No preload or
         * observer FD crosses the actual Python exec. GNU is single threaded. */
        active = 0;
        if (unsetenv("LD_PRELOAD") || unsetenv("FGNU_NONCE") ||
            unsetenv("FGNU_READY_FD") || unsetenv("FGNU_INTERPRETER") ||
            unsetenv("FGNU_LIBC")) refuse();
        if (close(fd)) refuse();
        fd = -1;
    } else if (child > 0) stage = 1;
    errno = saved;
    return child;
}
int execvp(const char *file, char *const argv[]) {
    if (active || getpid() == gnu_pid || strcmp(file, interpreter) ||
        getenv("LD_PRELOAD") || getenv("FGNU_NONCE") ||
        getenv("FGNU_READY_FD") || getenv("FGNU_INTERPRETER") ||
        getenv("FGNU_LIBC") || fd != -1) refuse();
    return next_execvp(file, argv);
}
int timer_create(clockid_t clock, struct sigevent *event, timer_t *out) {
    if (!active || getpid() != gnu_pid || stage != 1 || armed ||
        clock != CLOCK_REALTIME || event != NULL || out == NULL) refuse();
    int r = next_create(clock, event, out), saved = errno;
    if (r == 0) { created_timer = *out; stage = 2; }
    else stage = 3; /* Original GNU alarm fallback; no success record yet. */
    errno = saved;
    return r;
}
int timer_settime(timer_t timer, int flags, const struct itimerspec *value,
                  struct itimerspec *old) {
    if (!active || getpid() != gnu_pid || stage != 2 || armed ||
        timer != created_timer || flags != 0 || !value || old != NULL ||
        value->it_interval.tv_sec || value->it_interval.tv_nsec ||
        value->it_value.tv_sec != 60 || value->it_value.tv_nsec) refuse();
    int r = next_set(timer, flags, value, old), saved = errno;
    if (r == 0) {
        armed = 1;
        emit(3, 1, (uint64_t)(uintptr_t)timer, CLOCK_REALTIME, SIGALRM,
             UINT64_C(60000000000), mono_ns());
    } else stage = 4; /* GNU must delete failed timer before alarm fallback. */
    errno = saved;
    return r;
}
int timer_delete(timer_t timer) {
    if (!active || getpid() != gnu_pid || stage != 4 || armed ||
        timer != created_timer) refuse();
    int r = next_delete(timer), saved = errno;
    if (r != 0) refuse();
    stage = 3;
    errno = saved;
    return r;
}
unsigned int alarm(unsigned int seconds) {
    if (!active || getpid() != gnu_pid || stage != 3 || armed || seconds != 60)
        refuse();
    unsigned int prior = next_alarm(seconds);
    int saved = errno;
    if (prior != 0) refuse();
    armed = 1;
    emit(3, 2, 0, CLOCK_REALTIME, SIGALRM, UINT64_C(60000000000), mono_ns());
    errno = saved;
    return prior;
}
