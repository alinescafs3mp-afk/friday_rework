"""Bindings for the one same-process birth seam; SOURCE ONLY, NOT_RUN.

PyDLL holds the GIL; CDLL is deliberately forbidden. The child never returns
through libffi or Python: finite_birth.c performs only syscall -> exec/_exit.
No runtime loading occurs on import. A final reviewed binary/hash is required.
"""
import ctypes
import os
import signal
import sys


class NativeBirth:
    def __init__(self, library, deadline):
        if sys.implementation.name != 'cpython' or sys.version_info[:2] != (3, 14):
            raise RuntimeError('QUALIFIED_CPYTHON_314_REQUIRED')
        if not sys._is_gil_enabled():
            raise RuntimeError('QUALIFIED_GIL_BUILD_REQUIRED')
        for signum in (signal.SIGALRM, signal.SIGXCPU):
            if signal.getsignal(signum) != signal.SIG_DFL:
                raise RuntimeError('NONDEFAULT_PYTHON_SIGNAL_TABLE')
        self.lib = ctypes.PyDLL(str(library), mode=os.RTLD_NOW | os.RTLD_LOCAL)
        self.lib.frw_arm.argtypes = (ctypes.c_longlong, ctypes.c_long)
        self.lib.frw_arm.restype = ctypes.c_int
        self.lib.frw_spawn.argtypes = (ctypes.c_char_p, ctypes.POINTER(ctypes.c_char_p),
                                      ctypes.c_int, ctypes.c_int)
        self.lib.frw_spawn.restype = ctypes.c_long
        for name in ('pid', 'fd', 'expired', 'cpu_hit', 'signal_errno', 'rollback_errno', 'kill'):
            function = getattr(self.lib, 'frw_' + name)
            function.argtypes = ()
            function.restype = ctypes.c_int
        self.lib.frw_close.argtypes = (ctypes.c_int,)
        self.lib.frw_close.restype = ctypes.c_int
        seconds = int(deadline)
        try:
            error = self.lib.frw_arm(seconds, int((deadline-seconds)*1e9))
            if error:
                raise OSError(error, 'NATIVE_ORIGINAL_DEADLINE_ARM')
        except BaseException:
            # A Python pending signal can raise after C armed successfully but
            # before __init__ publishes this wall object. No birth exists yet.
            cleanup_error = self.lib.frw_close(False)
            if cleanup_error:
                raise OSError(cleanup_error, 'NATIVE_ARM_ROLLBACK_UNCONFIRMED')
            raise

    def spawn(self, argv, cwd, gate_read, error_write):
        self._argv_bytes = tuple(os.fsencode(value) for value in argv)
        self._argv = (ctypes.c_char_p * (len(argv)+1))(*self._argv_bytes, None)
        pid = self.lib.frw_spawn(os.fsencode(cwd), self._argv, gate_read, error_write)
        if pid <= 0:
            raise OSError(-pid if pid < 0 else 71, 'NATIVE_SINGLE_BIRTH_FAILED')
        return pid

    @property
    def pid(self):
        return self.lib.frw_pid()

    @property
    def root_fd(self):
        return self.lib.frw_fd()

    @property
    def expired(self):
        return bool(self.lib.frw_expired())

    @property
    def signal_error(self):
        errors = {'signal_errno': self.lib.frw_signal_errno(),
                  'rollback_errno': self.lib.frw_rollback_errno(),
                  'observer_cpu_endpoint': bool(self.lib.frw_cpu_hit())}
        return errors if any(errors.values()) else None

    def kill_root(self):
        error = self.lib.frw_kill()
        if error:
            raise OSError(error, 'NATIVE_BIRTH_PIDFD_KILL')

    def close_after_observation(self, terminal_observed):
        error = self.lib.frw_close(bool(terminal_observed))
        if error:
            raise OSError(error, 'NATIVE_CUSTODY_NOT_CLOSED')
