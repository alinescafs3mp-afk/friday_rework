"""NOT_RUN. One source qualifier call through the existing finite owner.

The external root owns Budget180, a canonical held quality OFD, distinct
reviewed output and actual independent source review. One finite coordinator
and its ordinary bwrap descendants share the existing foreground scope.
This call creates none of those, supplies no grant and never unlocks the parent.
"""
import hashlib
import json
import os
from pathlib import Path
import stat


def digest(path):
    with Path(path).open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()


def evidence_join(run, budget, owner, max_evidence_bytes):
    """Read after native namespace closure; external root must still close scope."""
    if type(max_evidence_bytes) is not int or not 0 < max_evidence_bytes < 2 << 30:
        raise ValueError('original_declared_evidence_bound_required')
    budget.check()
    original_clock=json.loads((run/'clock.json').read_text())
    if original_clock['deadline_monotonic']!=budget.deadline: raise ValueError('foreign_evidence_clock')
    pre_exec = json.loads((run / 'pre-exec-os.json').read_text())
    if (pre_exec['host_custody_sha256'] != owner['host_scope_observation_sha256']
            or pre_exec['host_before'] != owner['host_before']
            or pre_exec['pid'] != 1 or pre_exec['runtimeGrant'] is not False
            or pre_exec['deadline'] != budget.deadline
            or pre_exec['cgroup'] != owner['host_before']['cgroup']):
        raise ValueError('host_namespace_launch_receipt_not_joined')
    progress = json.loads((run / 'progress.json').read_text())
    collected = progress['collected']; results = progress['results']
    ids = {row['id'] for row in collected}
    if (progress['state'] != 'SOURCE_TS24_CASES_PASS_AWAIT_ROOT_CLOSURE'
            or progress.get('closed') is not True or progress['runtimeGrant'] is not False
            or progress['errors'] or len(collected) != 24 or len(ids) != 24
            or len(results) != 24 or {row['id'] for row in results} != ids
            or any(row['result']['state'] != 'passed' for row in results)):
        raise ValueError('actual_unique_ts24_results_not_closed')
    pins = {}; events = []; total = 0
    paths = sorted(run.glob('guard-*.jsonl'))
    if not paths: raise ValueError('actual_node_lifetime_evidence_missing')
    for path in paths:
        budget.check()
        st = path.lstat()
        if not stat.S_ISREG(st.st_mode) or st.st_nlink != 1 or st.st_uid != os.getuid():
            raise ValueError('foreign_guard_evidence_file')
        total += st.st_size
        if total > max_evidence_bytes: raise ValueError('original_evidence_bound_exhausted')
        raw = path.read_bytes(); pins[str(path)] = hashlib.sha256(raw).hexdigest()
        rows = [json.loads(line) for line in raw.splitlines()]
        if not rows or path.name != f'guard-{rows[0]["pid"]}.jsonl' or any(r['pid'] != rows[0]['pid'] for r in rows):
            raise ValueError('guard_file_process_identity_changed')
        events.extend(rows)
    if any(row.get('event') == 'DENIED' for row in events):
        raise ValueError('source_guard_refusal_preserved')
    starts = [row for row in events if row['event'] == 'START']
    exits = [row for row in events if row['event'] == 'EXIT']
    inherited = [row for row in events if row['event'] == 'QUALITY_FD_INHERITED']
    parent_pids = {r['pid'] for r in starts if r['role'] == 'parent'}
    all_pids = {r['pid'] for r in starts}
    exit_pids = {r['pid'] for r in exits}
    if (len(parent_pids) != 1 or not [r for r in starts if r['role'] == 'worker']
            or len(all_pids) != len(starts) or not parent_pids.issubset(exit_pids)
            or not exit_pids.issubset(all_pids)
            or {r['pid'] for r in inherited} != all_pids
            or any(r['code'] != 0 or r['child_still_active'] is not None for r in exits)):
        raise ValueError('whole_node_worker_lifetime_not_reconciled')
    forks = [r for r in events if r['event'] == 'FORK']
    child_exits = [r for r in events if r['event'] == 'CHILD_EXIT']
    worker_pids = {r['pid'] for r in starts if r['role'] == 'worker'}
    if ({r['child_pid'] for r in forks} != worker_pids
            or {r['child_pid'] for r in child_exits} != worker_pids
            or any(not (r['code'] == 0 and r['signal'] is None
                        or r['code'] is None and r['signal'] == 'SIGTERM') for r in child_exits)):
        raise ValueError('fork_exit_and_worker_records_not_joined')
    # Each process file is a contiguous typed stream, not a global hash bag.
    from collections import Counter
    if (Counter(r['child_pid'] for r in forks)!=Counter({pid:1 for pid in worker_pids})
            or Counter(r['child_pid'] for r in child_exits)!=Counter({pid:1 for pid in worker_pids})
            or Counter(r['pid'] for r in inherited)!=Counter({pid:1 for pid in all_pids})
            or any(r['ppid'] not in parent_pids for r in starts if r['role']=='worker')
            or any(r['pid'] not in parent_pids for r in forks+child_exits)
            or any(n!=1 for n in Counter(r['pid'] for r in exits).values())):
        raise ValueError('fork_inheritance_or_exit_multiplicity_changed')
    for r in starts:
        if int(r['birth'].split(' ',1)[0])!=r['pid'] or int(r['birth'].rsplit(')',1)[1].split()[19])<=0:
            raise ValueError('node_birth_identity_missing')
    if any(r['code']==0 and r['child_pid'] not in exit_pids for r in child_exits):
        raise ValueError('normal_worker_exit_record_missing')
    for r in inherited:
        expected_fd=4 if r['pid'] in worker_pids else owner['host_before']['quality_fd']
        if r['fd']!=expected_fd: raise ValueError('quality_fd_inheritance_changed')
    by_pid = {}
    for row in events:
        if row['pid'] not in all_pids: raise ValueError('event_from_unstarted_process')
        by_pid.setdefault(row['pid'], []).append(row)
    for pid, rows in by_pid.items():
        if [r['seq'] for r in rows] != list(range(1,len(rows)+1)):
            raise ValueError('missing_duplicate_or_reordered_process_event')
        start = next(r for r in starts if r['pid']==pid)
        if f'guard-{pid}.jsonl' not in {Path(path).name for path in pins}: raise ValueError('foreign_guard_filename')
        if any(r['role']!=start['role'] or r['ppid']!=start['ppid'] for r in rows):
            raise ValueError('worker_identity_changed')
        if any(not original_clock['started_monotonic'] <= r['at'] <= budget.deadline for r in rows): raise ValueError('event_outside_original_deadline')
    def selected(kind): return [r for r in events if r['event']==kind]
    catalogs=selected('NODE_BUILTIN_CATALOG')
    if Counter(r['pid'] for r in catalogs)!=Counter({pid:1 for pid in all_pids}):
        raise ValueError('actual_node_builtin_catalog_multiplicity_changed')
    builtin_names=set(next(r['names'] for r in catalogs if r['pid'] in parent_pids))
    if not builtin_names or any(set(r['names'])!=builtin_names for r in catalogs):
        raise ValueError('actual_node_builtin_catalog_changed')
    def builtin(ident):
        return isinstance(ident,str) and (ident=='/@vite/client' or ident.removeprefix('node:') in builtin_names)
    root = Path(__file__).resolve().parents[2] / 'ts24'
    view = json.loads((root/'source-view-pins.json').read_text())
    source_pins = {p:ref['sha256'] for p,ref in view.items()}
    for mapping in (json.loads((root/'support-pins.json').read_text())['files'],
                    json.loads((root/'runner-pins.json').read_text())):
        for p,sha in mapping.items():
            if p in source_pins and source_pins[p]!=sha: raise ValueError('conflicting_evidence_source_pin')
            source_pins[p]=sha
    expected = json.loads((root/'expected-cases.json').read_text())
    expected_names=Counter((r['file'],r['name']) for r in expected)
    if (len(expected)!=24 or Counter((r['file'],r['name']) for r in collected)!=expected_names
            or Counter((r['file'],r['name']) for r in results)!=expected_names
            or {r['id']:(r['file'],r['name']) for r in collected}!={r['id']:(r['file'],r['name']) for r in results}):
        raise ValueError('original_two_file_case_identity_changed')
    for row in events:
        if 'canonical' in row and source_pins.get(row['canonical'])!=row.get('sourceSha256'):
            raise ValueError('observed_source_not_current_pin')
    def module_key(r): return r['environment'],r['id'],r['canonical']
    def payload_key(r): return (*module_key(r),r['codeSha256'])
    graphs=selected('VITE_FINAL_GRAPH')
    if len(graphs)!=1 or graphs[0]['pid'] not in parent_pids: raise ValueError('exact_final_graph_missing')
    graph_rows=graphs[0]['graph']; graph={module_key(r):r for r in graph_rows}
    if len(graph)!=len(graph_rows) or not graph: raise ValueError('ambiguous_or_empty_graph')
    inputs=selected('VITE_TRANSFORM_INPUT'); returned=selected('VITE_RETURNED_TRANSFORM')
    plugin_outputs=selected('VITE_PLUGIN_OUTPUT'); next_inputs=selected('VITE_NEXT_TRANSFORM_INPUT')
    next_outputs=selected('VITE_NEXT_TRANSFORM_OUTPUT'); transform_graphs=selected('VITE_TRANSFORM_GRAPH')
    final_rows=selected('VITEST_FINAL_RESULT_SOURCE'); temps_rows=selected('VITEST_TMP_WRITTEN')
    fetch_rows=[r for r in events if r['event'] in ('VITEST_WORKER_FETCH','VITEST_ENV_FETCH')]
    begins=[r for r in events if r['event'] in ('VITEST_EVAL_BEGIN','VITE_RUNNER_EVAL_BEGIN')]
    done=[r for r in events if r['event'] in ('VITEST_EVAL_DONE','VITE_RUNNER_EVAL_DONE')]
    if not begins or not inputs or not returned or not plugin_outputs or not transform_graphs:
        raise ValueError('missing_actual_source_event_chain')
    producer=inputs+returned+plugin_outputs+next_inputs+next_outputs+transform_graphs+temps_rows+final_rows
    if any(r['pid'] not in parent_pids for r in producer): raise ValueError('transform_producer_in_wrong_process')
    if any(r['pid'] not in worker_pids or r['workerId'] is None or r['poolId'] is None for r in fetch_rows if r['event']=='VITEST_WORKER_FETCH'):
        raise ValueError('fetch_outside_original_worker_identity')
    def transform_key(r): return r['pid'], *module_key(r), r['transformId']
    stages=[inputs,plugin_outputs,next_inputs,next_outputs]
    signatures=[Counter(map(transform_key,rows)) for rows in stages]
    if any(c!=signatures[0] for c in signatures[1:]) or any(n!=1 for n in signatures[0].values()):
        raise ValueError('exact_transform_stage_identity_or_multiplicity_changed')
    transform_outputs={transform_key(r):r for r in next_outputs}
    for original in inputs:
        key=transform_key(original)
        post,following,output=(next(r for r in rows if transform_key(r)==key) for rows in stages[1:])
        if (original['codeSha256']!=original['sourceSha256'] or post['codeSha256']!=following['codeSha256']
                or not original['seq']<post['seq']<following['seq']<output['seq']):
            raise ValueError('actual_plugin_output_next_input_chain_broken')
    def request_key(r): return r['pid'],r['environment'],r['id'],r['requestId']
    if (graphs[0]['observedRequestRows']!=len(returned) or graphs[0]['observedTransformRows']!=len(inputs)
            or {r['transformId'] for r in inputs}!={r['transformId'] for r in returned}):
        raise ValueError('observed_transform_request_coverage_changed')
    request_counts=Counter(map(request_key,returned))
    if (any(n!=1 for n in request_counts.values())
            or request_counts!=Counter(map(request_key,transform_graphs))
            or request_counts!=Counter(map(request_key,[r for r in final_rows if 'canonical' in r]))):
        raise ValueError('transform_request_return_graph_final_multiplicity_changed')
    for r in returned:
        observed_graph=next(g for g in transform_graphs if request_key(g)==request_key(r))
        if (observed_graph['transformId']!=r['transformId'] or observed_graph['file']!=r['canonical']
                or observed_graph['seq']<=r['seq']):
            raise ValueError('returned_graph_exact_transform_identity_changed')
        output=transform_outputs.get(transform_key(r))
        if output is None or output['seq']>=r['seq']: raise ValueError('returned_transform_identity_missing')
        # Same-run Vite cache can already contain the observed inline-map bytes.
        # It must have an earlier exact final record for this transform, not a
        # different module/environment or a hash invented by this consumer.
        if r['codeSha256']!=output['codeSha256'] and not any(
                transform_key(f)==transform_key(r) and f['seq']<r['seq'] and f['codeSha256']==r['codeSha256']
                for f in final_rows if 'canonical' in f):
            raise ValueError('returned_transform_or_witnessed_cache_bytes_changed')
    finals={payload_key(r):r for r in final_rows if 'canonical' in r}
    for r in final_rows:
        if 'canonical' not in r: continue
        ret=next(x for x in returned if request_key(x)==request_key(r))
        if (transform_key(ret)!=transform_key(r) or ret['seq']>=r['seq']
                or ret['codeSha256']!=r['returnedTransformSha256']):
            raise ValueError('returned_final_exact_transform_chain_broken')
    final_modules={module_key(r) for r in final_rows if 'canonical' in r}
    if {module_key(g) for g in graph_rows if g['transformedSha256'] is not None} != final_modules:
        raise ValueError('extra_or_missing_transformed_graph_module')
    for key in final_modules:
        last=max((r for r in final_rows if 'canonical' in r and module_key(r)==key),key=lambda r:r['seq'])
        if graph[key]['transformedSha256']!=last['codeSha256']: raise ValueError('latest_final_graph_bytes_changed')
    graph_ids={(g['environment'],g['id']) for g in graph_rows}
    for g in graph_rows:
        if g['canonical'] is None:
            if not builtin(g['id']) or g['transformedSha256'] is not None:
                raise ValueError('unbound_nonfile_graph_entry')
        elif source_pins.get(g['canonical'])!=g['sourceSha256']: raise ValueError('graph_source_pin_changed')
        for imported in g['imports']:
            ident=imported['id']
            if (g['environment'],ident) not in graph_ids:
                if not builtin(ident):
                    raise ValueError('graph_import_not_closed')
    if any(module_key(r) not in graph for rows in stages for r in rows): raise ValueError('extra_transform_outside_graph')
    # Reconcile exact edge sets in process order. A change is admissible only
    # with the actual updateModuleInfo before/after observation at that seam.
    def edges(rows):
        value=Counter((e['id'],e['file'],e['url']) for e in rows)
        if any(n!=1 for n in value.values()): raise ValueError('duplicate_graph_edge')
        return value
    observed_edges={}
    graph_events=selected('VITE_GRAPH_UPDATE')+transform_graphs
    for r in sorted(graph_events,key=lambda r:r['seq']):
        if r['pid'] not in parent_pids: raise ValueError('foreign_graph_producer')
        key=r['environment'],r['id']; current=edges(r['imports'])
        if r['event']=='VITE_GRAPH_UPDATE':
            if r['reason']!='actual-updateModuleInfo': raise ValueError('unwitnessed_graph_delta')
            previous=edges(r['previousImports'])
            if key in observed_edges and previous!=observed_edges[key]: raise ValueError('graph_delta_previous_edges_changed')
        elif key in observed_edges and observed_edges[key]!=current:
            raise ValueError('returned_graph_edges_changed_without_delta')
        observed_edges[key]=current
    for g in graph_rows:
        key=g['environment'],g['id']
        if key in observed_edges and edges(g['imports'])!=observed_edges[key]:
            raise ValueError('final_graph_edges_not_equal_to_observed_edges')
        if g['transformedSha256'] is not None and key not in observed_edges:
            raise ValueError('transformed_graph_edges_unobserved')
    fetch=[r for r in fetch_rows if 'canonical' in r]
    for r in temps_rows+fetch+begins+done:
        if 'canonical' not in r: continue
        final_key=(*module_key(r),r['finalSourceSha256'])
        if final_key not in finals: raise ValueError('payload_not_joined_to_final_source')
        if r.get('tmp') is not None:
            if r['tempSha256']!=r['codeSha256'] or not any(payload_key(t)==payload_key(r) and t['tmp']==r['tmp'] for t in temps_rows):
                raise ValueError('temporary_payload_bytes_changed')
            if r['cacheMetadata']['id']!=r['id'] or r['cacheMetadata']['file']!=r['canonical']:
                raise ValueError('cache_suffix_identity_changed')
    # Pair event kinds, environment, process, worker identity, byte and wrapper
    # hashes one-to-one. Missing inner VM observations cannot hide behind runner.
    def pair_key(r):
        return (r['event'].rsplit('_',1)[0],r['pid'],r['workerId'],r['poolId'],
                *payload_key(r),r['finalSourceSha256'],r['wrappedSha256'])
    if Counter(map(pair_key,begins))!=Counter(map(pair_key,done)):
        raise ValueError('typed_evaluation_begin_done_multiplicity_changed')
    if any(n!=1 for n in Counter(map(pair_key,begins)).values()): raise ValueError('duplicate_evaluation')
    for r in begins:
        end=next(x for x in done if pair_key(x)==pair_key(r))
        if end['seq']<=r['seq']: raise ValueError('completion_precedes_begin')
        matching=[f for f in fetch if f['pid']==r['pid'] and payload_key(f)==payload_key(r)
                  and f['seq']<r['seq'] and f['workerId']==r['workerId'] and f['poolId']==r['poolId']]
        if not matching: raise ValueError('worker_fetch_identity_missing')
        if r['event']=='VITEST_EVAL_BEGIN':
            if (r['wrappedSha256'] is None or r['wrappedSha256']!=r['expectedWrappedSha256']
                    or end['expectedWrappedSha256']!=r['expectedWrappedSha256']): raise ValueError('wrapper_hash_changed')
            outer=[x for x in begins if x['event']=='VITE_RUNNER_EVAL_BEGIN' and x['pid']==r['pid'] and payload_key(x)==payload_key(r)]
            if len(outer)!=1 or outer[0]['seq']>=r['seq']: raise ValueError('inner_vm_not_inside_runner')
            outer_done=next(x for x in done if pair_key(x)==pair_key(outer[0]))
            if outer_done['seq']<=end['seq']: raise ValueError('runner_done_precedes_vm_done')
        elif any(f['event']=='VITEST_WORKER_FETCH' for f in matching):
            if sum(x['event']=='VITEST_EVAL_BEGIN' and x['pid']==r['pid'] and payload_key(x)==payload_key(r) for x in begins)!=1:
                raise ValueError('vitest_inner_evaluation_missing')
    for r in fetch:
        if not any(x['event']=='VITE_RUNNER_EVAL_BEGIN' and x['pid']==r['pid'] and payload_key(x)==payload_key(r) for x in begins):
            raise ValueError('fetched_module_never_evaluated')
    if {module_key(r) for r in fetch} != final_modules: raise ValueError('graph_worker_correspondence_missing')
    for r in fetch_rows:
        if 'externalize' in r:
            url=r['externalize']
            if builtin(url): continue
            from urllib.parse import unquote,urlparse
            canonical=unquote(urlparse(url).path) if url.startswith('file:') else url
            if canonical not in source_pins or not any(x['pid']==r['pid'] and x.get('path')==canonical for x in selected('NODE_MODULE_OR_PIN_CHECK')):
                raise ValueError('external_module_not_explicitly_pinned')
    # Actual assignment + per-import requests anchor each worker's own chain.
    # A shared module in worker B cannot supply worker A's fetch or evaluation.
    assignments=selected('VITEST_WORKER_ASSIGNMENT'); requests=selected('VITE_WORKER_REQUEST')
    def worker_key(r): return r['pid'],r['workerId'],r['poolId']
    expected_files={r['file'] for r in expected}
    if Counter(f for r in assignments for f in r['files'])!=Counter({f:1 for f in expected_files}):
        raise ValueError('original_files_not_assigned_once_to_actual_workers')
    if any(r['pid'] not in worker_pids or r['action']!='run' or None in worker_key(r) for r in assignments):
        raise ValueError('foreign_worker_assignment')
    outer=[r for r in begins if r['event']=='VITE_RUNNER_EVAL_BEGIN']
    for assignment in assignments:
        for file in assignment['files']:
            roots=[r for r in requests if worker_key(r)==worker_key(assignment)
                and r.get('canonical')==file and r['importer'] is None and r['seq']>assignment['seq']]
            if len(roots)!=1: raise ValueError('assigned_worker_test_root_request_missing')
    for r in requests:
        if r['pid'] not in all_pids: raise ValueError('foreign_worker_request')
        if 'externalize' in r:
            if r.get('builtin'):
                catalogs=[x for x in selected('NODE_BUILTIN_CATALOG') if x['pid']==r['pid']]
                if (len(catalogs)!=1 or catalogs[0]['seq']>=r['seq']
                        or r['externalize']!='/@vite/client' and r['externalize'].removeprefix('node:') not in catalogs[0]['names']):
                    raise ValueError('builtin_request_not_in_actual_node_catalog')
            elif (source_pins.get(r.get('externalCanonical'))!=r.get('externalSha256')
                    or not any(x['pid']==r['pid'] and x.get('path')==r.get('externalCanonical') and x['seq']<r['seq']
                               for x in selected('NODE_MODULE_OR_PIN_CHECK'))):
                raise ValueError('external_request_pin_missing')
        else:
            local=[x for x in outer if worker_key(x)==worker_key(r) and payload_key(x)==payload_key(r)]
            if len(local)!=1: raise ValueError('worker_request_local_evaluation_missing')
            if r['cache'] and local[0]['seq']>=r['seq']: raise ValueError('worker_cache_without_prior_local_evaluation')
            if not any(worker_key(f)==worker_key(r) and payload_key(f)==payload_key(r) and f['seq']<r['seq'] for f in fetch):
                raise ValueError('worker_request_local_fetch_missing')
        if r['importer'] is not None:
            parents=[x for x in outer if worker_key(x)==worker_key(r) and x['id']==r['importer']
                     and x['seq']<r['seq'] and (r['environment'] is None or x['environment']==r['environment'])]
            if len(parents)!=1: raise ValueError('worker_importer_evaluation_missing')
    for begin in outer:
        final=finals[(*module_key(begin),begin['finalSourceSha256'])]
        transformed=transform_outputs[transform_key(final)]
        end=next(x for x in done if pair_key(x)==pair_key(begin))
        calls=[r for r in requests if worker_key(r)==worker_key(begin) and r['importer']==begin['id']
               and begin['seq']<r['seq']<end['seq']]
        static={r['requested'] for r in calls if not r['dynamic']}
        if static!=set(transformed['deps']): raise ValueError('worker_static_import_chain_incomplete')
        for request in calls:
            if request['dynamic'] or 'externalize' in request: continue
            if not any(edge['id']==request['id'] and edge['file']==request['canonical']
                       for edge in graph[module_key(begin)]['imports']):
                raise ValueError('worker_static_import_missing_exact_graph_edge')
        if not any(worker_key(r)==worker_key(begin) and r.get('canonical')==begin['canonical']
                   and r['id']==begin['id'] for r in requests):
            raise ValueError('evaluation_without_actual_worker_request')
    final={payload_key(r) for r in final_rows if 'canonical' in r}
    fetch={payload_key(r) for r in fetch}
    mapped = [r for r in events if r['event'] == 'MAPPED_OBJECTS_SNAPSHOT']
    if any(not any(r['pid'] == s['pid'] and r['phase'] == 'pre-framework' for r in mapped) for s in starts):
        raise ValueError('node_initial_native_maps_missing')
    for row in events:
        if row['event'] == 'ADDON_LOAD_REQUEST':
            if not any(m['pid'] == row['pid'] and m['phase'] == 'after-addon:' + row['path'] for m in mapped):
                raise ValueError('native_addon_maps_not_joined')
    if owner['state'] != 'NAMESPACE_TERMINAL_SCOPE_OBSERVER_PENDING':
        raise ValueError('native_namespace_termination_not_joined')
    budget.check()
    return {'pre_exec_sha256': digest(run / 'pre-exec-os.json'),
            'progress_sha256': digest(run / 'progress.json'), 'guard_files': pins,
            'observed_test_ids': sorted(ids), 'node_starts': starts,
            'actual_evaluations': begins, 'actual_completions': done,
            'final_sources': sorted(final), 'worker_payloads': sorted(fetch),
            'native_maps': mapped, 'owner_scope_closure': 'PENDING_EXTERNAL_ROOT', 'meaning': 'Actual trusted source qualification observations only; native hostile-effect/control completeness still requires independent root negatives.',
            'runtimeGrant': False}

def consumed_inputs(root):
    from scripts.qualifier_scope import merged
    pins={};links={}
    for name in ('runner-pins','support-pins','native-runtime-pins','launcher-runtime-pins'):
        value=json.loads((root/(name+'.json')).read_text())
        merged(pins,value.get('files',value))
        if name!='native-runtime-pins': merged(links,value.get('links',{}))
    source=json.loads((root/'source-view-pins.json').read_text())
    if len(source)!=11: raise ValueError('exact_current_ts11_required')
    merged(pins,{p:ref['sha256'] for p,ref in source.items()})
    merged(pins,json.loads((root.parent/'owner-source-pins.json').read_text()))
    return pins,links


def qualify(custody, *, budget, quality_fd, run, unit, tasks, max_evidence_bytes):
    """Existing external root observer; launches one capped finite coordinator."""
    from scripts.install_containment import Containment
    from scripts.dsh_prepare import run as foreground, StopUnconfirmed, finalize_error
    from scripts.qualifier_scope import held, prepare_owner, stop_owner, finish_owner
    if type(custody) is not Containment or custody.budget is not budget:
        raise ValueError('same_existing_containment_and_original_budget_required')
    root=Path(__file__).resolve().parents[2]/'ts24'; run=Path(run)
    held(quality_fd);budget.check(reserve=3)
    # Byte hashing occurs within the scope in the coordinator. Root's trusted
    # call-site and independently reviewed owner pins remain prerequisites.
    initial={name:digest(run/name) for name in ('clock.json','admission.json')}
    option={'unit':unit,'tasks':tasks,'quality_fd':quality_fd,'run':str(run),'ts_root':str(root),
        'host_guard':str(Path(__file__).with_name('ts_scope_guard.py')),
        'coordinator':str(Path(__file__)), 'bwrap_pin':custody.pin,
        'max_evidence_bytes':max_evidence_bytes,
        'mounts':{'path':str(root/'mount-plan.json'),'sha256':digest(root/'mount-plan.json')}}
    args,fds,owner=prepare_owner(budget,option,custody.env)
    receipt={'state':'SOURCE_QUALIFICATION_PENDING','runtimeGrant':False,'original_deadline':budget.deadline,'original_inputs':initial}
    failure=None;observed={}
    try:
        _,observed=foreground(args,run,timeout=min(90,budget.check(reserve=3)),env=custody.env,
            deadline=budget.deadline,pass_fds=fds,cleanup_reserve=3,
            on_stop=lambda process,starttime:stop_owner(budget,owner,process,starttime))
    except BaseException as exc:
        failure=exc;observed=getattr(exc,'observation',{})
    try:
        terminal=finish_owner(budget,owner,observed)
        receipt['execution']=observed;receipt['owner_receipt']=terminal
        if failure: raise failure
        coordinator=json.loads((run/'coordinator.json').read_text())
        if coordinator['state']!='COORDINATOR_NAMESPACE_CLOSED_AWAIT_EXTERNAL_SCOPE' or coordinator['runtimeGrant'] is not False:
            raise ValueError('coordinator_source_evidence_not_complete')
        namespace=coordinator['execution']['owner_receipt']
        if (namespace['state']!='NAMESPACE_TERMINAL_SCOPE_OBSERVER_PENDING'
                or namespace['host_before']!=terminal['host_before']
                or namespace['host_scope_observation_sha256']!=terminal['host_scope_observation_sha256']
                or not coordinator['execution']['namespace_init_exit_verified']):
            raise ValueError('coordinator_native_receipt_not_joined')
        terminal['state']='OWNED_NAMESPACE_AND_SCOPE_TERMINAL_LOCK_STILL_HELD'
        receipt['coordinator_sha256']=digest(run/'coordinator.json')
        receipt['source_evidence']=coordinator['source_evidence']
        if receipt['source_evidence']['owner_scope_closure']!='PENDING_EXTERNAL_ROOT':
            raise ValueError('coordinator_claimed_own_scope_empty')
        if {name:digest(run/name) for name in initial}!=initial: raise ValueError('original_root_input_bytes_changed')
        receipt['quality_parent_still_held']=held(quality_fd)
        receipt['state']='TRUSTED_TS24_SOURCE_RESULTS_JOINED_ROOT_LOCK_RETURN_PENDING'
        budget.check();return receipt
    except BaseException as exc:
        exc=finalize_error(exc, failure, observed)
        receipt['failure']=type(exc).__name__+':'+str(exc)
        receipt['state']='STOP_UNCONFIRMED' if isinstance(exc,StopUnconfirmed) else 'SOURCE_QUALIFICATION_FAILED'
        exc.owner_context=owner
        if hasattr(exc, "custody"): exc.custody.owner_context=owner
        exc.qualifier_receipt=receipt
        raise exc


def coordinator(p):
    from scripts.install_containment import Budget,Containment
    from scripts.dsh_prepare import retain_custody
    from scripts.qualifier_scope import bind_owner,held
    # Reconstitute the SAME original start/deadline, never start another180.
    budget=Budget(180,started=p['started'],deadline=p['deadline'])
    host=bind_owner(budget,p,dict(os.environ))
    root=Path(p['ts_root']);run=Path(p['run'])
    pins,links=consumed_inputs(root)
    option=dict(p,pins=pins,links=links,host_before=host)
    custody=Containment(p['bwrap_pin'],budget,dict(os.environ))
    receipt={'state':'COORDINATOR_SOURCE_PENDING','runtimeGrant':False}
    failure=None;observed={}
    try:
        _,observed=custody.run(['/usr/bin/python3','-I',str(root/'launch.py'),str(run),'FRW_TS_CUSTODY_FD'],
            run,timeout=min(90,budget.check(reserve=3)),qualifier=option)
        receipt.update(state='COORDINATOR_NAMESPACE_CLOSED_AWAIT_EXTERNAL_SCOPE',execution=observed,
            source_evidence=evidence_join(run,budget,observed['owner_receipt'],p['max_evidence_bytes']))
        held(p['quality_fd']);budget.check(reserve=3)
    except BaseException as exc:
        failure=exc
        receipt.update(state='SOURCE_QUALIFICATION_FAILED',failure=type(exc).__name__+':'+str(exc),
            execution=getattr(exc,'observation',None),owner_stop_evidence=getattr(exc,'owner_receipt',None))
        raise
    finally:
        try: (run/'coordinator.json').write_text(json.dumps(receipt,indent=2)+'\n')
        except BaseException as exc:
            if failure is not None:
                failure.receipt_write_error=exc
                raise failure
            raise retain_custody(exc, observed)


if __name__=='__main__':
    import sys
    if len(sys.argv)!=3 or sys.argv[1]!='--coordinator': raise ValueError('exact_finite_coordinator_entry_required')
    sys.path.insert(0,str(Path(__file__).resolve().parents[1]))
    coordinator(json.loads(sys.argv[2]))
