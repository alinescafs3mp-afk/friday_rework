"""Finite existing scope boundary around the coordinator; source only, NOT_RUN.

The external root retains the original clock, Popen, lock and terminal observer.
The coordinator and ordinary bwrap descendants share exactly one capped scope.
No new per-job slice, service, supervisor, clock, lock or admission is created.
"""
import hashlib
import json
import os
from pathlib import Path
import re
import stat
import time

LOCK = Path('<LOCAL_HOME>/.jericho/runtime/friday-quality-ui-exclusive.lock')
LOCK_ID = (2050, 9044963, 1000, 1000, 1, 0, 0o664)
CLEANUP_RESERVE = 3


def identity(st):
    return (st.st_dev, st.st_ino, st.st_uid, st.st_gid,
            st.st_nlink, st.st_size, stat.S_IMODE(st.st_mode))


def held(fd):
    if type(fd) is not int or fd < 3 or identity(os.fstat(fd)) != LOCK_ID:
        raise ValueError('canonical_existing_quality_fd_required')
    if LOCK.resolve() != LOCK or identity(LOCK.lstat()) != LOCK_ID:
        raise ValueError('canonical_quality_path_changed')
    text = Path(f'/proc/self/fdinfo/{fd}').read_text()
    if not any('FLOCK  ADVISORY  WRITE' in line and f'{LOCK_ID[1]} ' in line for line in text.splitlines()):
        raise ValueError('held_quality_open_description_required')
    return text


def birth(pid):
    return int(Path(f'/proc/{pid}/stat').read_text().rsplit(')', 1)[1].split()[19])


def scope_show(unit, budget, env):
    from scripts.dsh_prepare import run
    out, observation = run(['/usr/bin/systemctl', '--user', 'show', unit,
        '--property=Id,LoadState,ActiveState,ControlGroup,InvocationID,ActiveEnterTimestampMonotonic,RuntimeMaxUSec,RuntimeRandomizedExtraUSec,TimeoutStopUSec,KillSignal,FinalKillSignal,SendSIGKILL,KillMode,OOMPolicy'],
        Path('/'), env=env, timeout=min(1, budget.check()), deadline=budget.deadline, cleanup_reserve=0)
    rows = [line.split('=', 1) for line in out.splitlines()]
    if len(rows) != 14 or len({r[0] for r in rows}) != 14:
        raise ValueError('exact_scope_properties_required')
    return dict(rows), observation


def merged(target, mapping):
    for path, sha in mapping.items():
        if path in target and target[path] != sha: raise ValueError('conflicting_consumed_pin')
        target[path] = sha


def prepare_owner(budget, option, env):
    """Called only by the existing external root, before its single foreground launch."""
    if not re.fullmatch(r'friday-qualify-[a-z0-9][a-z0-9-]{0,190}\.scope', option['unit']):
        raise ValueError('unique_caller_owned_scope_required')
    if type(option['tasks']) is not int or not 0 < option['tasks'] < 2**31:
        raise ValueError('original_declared_task_count_required')
    held(option['quality_fd']); budget.check(reserve=CLEANUP_RESERVE)
    run = Path(option['run']); root = Path(option['ts_root'])
    if (run != run.resolve() or root != root.resolve() or not run.name.startswith('ts24-run-')
            or run.is_relative_to(root.parent) or root.is_relative_to(run)
            or run.stat().st_uid != os.getuid() or stat.S_IMODE(run.stat().st_mode) != 0o700):
        raise ValueError('fresh_external_owned_output_required')
    original_source=(root/'source-view-pins.json').resolve().parent
    if run.is_relative_to(original_source.parent): raise ValueError('output_inside_immutable_source')
    if {p.name for p in run.iterdir()}!={'clock.json','admission.json'}:
        raise ValueError('fresh_output_no_old_attempt_replay')
    for name in ('clock.json','admission.json'):
        st=(run/name).lstat()
        if not stat.S_ISREG(st.st_mode) or st.st_nlink!=1 or st.st_uid!=os.getuid():
            raise ValueError('original_inputs_not_owned_regular_files')
    clock = json.loads((run / 'clock.json').read_text())
    if (clock['original_budget_seconds'] != 180 or clock['execution_authorized'] is not False
            or clock['scope'] != 'TS24_SOURCE_OFFLINE_ONLY'
            or clock['deadline_monotonic'] != clock['started_monotonic'] + 180
            or clock['deadline_monotonic'] != budget.deadline
            or clock['boot_id'] != Path('/proc/sys/kernel/random/boot_id').read_text().strip()):
        raise ValueError('same_existing_original_ts180_required')
    before, query = scope_show(option['unit'], budget, env)
    if before['Id'] != option['unit'] or before['LoadState'] != 'not-found' or before['ControlGroup']:
        raise ValueError('scope_name_not_fresh')
    reader, writer = os.pipe2(os.O_NONBLOCK | os.O_CLOEXEC)
    context = dict(option, reader=reader, writer=writer, owner_pid=os.getpid(),
        owner_birth=birth(os.getpid()), deadline=budget.deadline, boot_id=clock['boot_id'],
        started=clock['started_monotonic'], before_scope=before, before_query=query,
        pipe_identity={key:[os.fstat(fd).st_dev,os.fstat(fd).st_ino] for key,fd in [('reader',reader),('writer',writer)]},
        quality_before=held(option['quality_fd']), env=env)
    envelope = {k:v for k,v in context.items() if k not in ('reader','env','before_scope','before_query','quality_before','pipe_identity')}
    # KillSignal=SIGKILL avoids the default TERM grace; all stop/reap work stays
    # inside the original absolute budget. Admission below checks actual expiry.
    maximum = min(90, budget.check(reserve=CLEANUP_RESERVE))
    envelope['requested_runtime_seconds'] = maximum
    args = ['/usr/bin/systemd-run', '--user', '--scope', '--quiet', '--no-ask-password',
        '--expand-environment=no', '--unit=' + option['unit'],
        '--property=MemoryMax=2147483648', '--property=MemorySwapMax=0',
        '--property=CPUQuota=200%', '--property=TasksMax=' + str(option['tasks']),
        '--property=OOMPolicy=kill', '--property=MemoryOOMGroup=yes', '--property=KillMode=control-group',
        '--property=KillSignal=SIGKILL', '--property=FinalKillSignal=SIGKILL',
        '--property=TimeoutStopSec=1s', '--property=SendSIGKILL=yes',
        '--property=RuntimeRandomizedExtraSec=0', '--property=RuntimeMaxSec=' + str(maximum) + 's',
        '/usr/bin/python3', '-I', option['coordinator'], '--coordinator', json.dumps(envelope)]
    return args, (option['quality_fd'], writer), context


def bind_owner(budget, p, env):
    """Actual coordinator facts before hashing, namespace launch or candidate effects."""
    held(p['quality_fd']); budget.check(reserve=CLEANUP_RESERVE)
    if os.getpid()!=os.getpgrp() or os.getsid(0)!=os.getpid(): raise ValueError('original_foreground_session_not_owned')
    if birth(p['owner_pid']) != p['owner_birth']: raise ValueError('root_owner_birth_changed')
    line = Path('/proc/self/cgroup').read_text().strip()
    if not line.startswith('0::/') or '\n' in line: raise ValueError('actual_owner_cgroup_required')
    group = line[3:]; cg = Path('/sys/fs/cgroup') / group.lstrip('/')
    if cg.resolve() != cg or cg.name != p['unit']: raise ValueError('coordinator_outside_owned_scope')
    caps = {k:(cg/k).read_text().strip() for k in ('memory.max','memory.swap.max','cpu.max','pids.max')}
    q,t = caps['cpu.max'].split()
    if (caps['memory.max'] != str(2 << 30) or caps['memory.swap.max'] != '0'
            or caps['pids.max'] != str(p['tasks']) or not q.isdecimal() or not t.isdecimal()
            or int(t) <= 0 or int(q) != 2*int(t) or (cg/'memory.oom.group').read_text().strip() != '1'
            or not 0 < len(os.sched_getaffinity(0)) <= 2):
        raise ValueError('actual_whole_owner_caps_required')
    props, query = scope_show(p['unit'], budget, env)
    if (props['Id'] != p['unit'] or props['LoadState'] != 'loaded' or props['ActiveState'] != 'active'
            or props['ControlGroup'] != group or not re.fullmatch('[0-9a-f]{32}', props['InvocationID'])
            or props['KillSignal'] != '9' or props['FinalKillSignal'] != '9' or props['SendSIGKILL'] != 'yes'
            or props['OOMPolicy'] != 'kill' or props['KillMode'] != 'control-group' or props['RuntimeRandomizedExtraUSec'] != '0'):
        raise ValueError('actual_scope_identity_or_hard_stop_policy_unbound')
    # Parse systemctl's exact finite span grammar (e.g. "1min 30s").
    # This observes installed output; unknown/infinite values refuse.
    def seconds(value):
        pieces = re.findall(r'([0-9]+(?:\.[0-9]+)?)(us|ms|min|s|h)', value)
        if not pieces or ''.join(n+u for n,u in pieces) != value.replace(' ', ''):
            raise ValueError('native_duration_format_unqualified')
        return sum(float(n) * {'us':1e-6,'ms':1e-3,'s':1,'min':60,'h':3600}[u] for n,u in pieces)
    active = int(props['ActiveEnterTimestampMonotonic']) / 1e6
    expiry = active + seconds(props['RuntimeMaxUSec'])
    if not (0 < active <= time.monotonic() and seconds(props['TimeoutStopUSec']) <= 1
            and expiry + 1 <= budget.deadline - 1):
        raise ValueError('actual_native_scope_expiry_exceeds_original_budget')
    snapshot = {k:p[k] for k in ('unit','tasks','quality_fd','owner_pid','owner_birth','boot_id','deadline')}
    snapshot.update(requested_oom_policy='kill', effective_oom_policy=props['OOMPolicy'], invocation=props['InvocationID'], control_group=group, cgroup=str(cg),
        cgroup_identity=[cg.stat().st_dev,cg.stat().st_ino], caps=caps,
        affinity=sorted(os.sched_getaffinity(0)), scope_pid=os.getpid(), scope_birth=birth(os.getpid()),
        host_namespace_inodes={n:Path('/proc/self/ns/'+n).stat().st_ino for n in ('pid','mnt','net')},
        quality_fdinfo=held(p['quality_fd']), native_expiry_monotonic=expiry, runtimeGrant=False)
    raw = json.dumps(snapshot,sort_keys=True).encode()
    if len(raw)>4096 or os.write(p['writer'],raw)!=len(raw): raise ValueError('host_observation_write_failed')
    os.close(p['writer'])
    return snapshot


def prepare(budget, command, status_fd, option, env):
    """Inside the already admitted coordinator; launch only ordinary bwrap child."""
    held(option['quality_fd']); budget.check(reserve=CLEANUP_RESERVE)
    host = option['host_before']; root = Path(option['ts_root'])
    if command != ['/usr/bin/python3','-I',str(root/'launch.py'),option['run'],'FRW_TS_CUSTODY_FD']:
        raise ValueError('exact_qualified_ts24_command_required')
    if status_fd == option['quality_fd'] or not stat.S_ISFIFO(os.fstat(status_fd).st_mode):
        raise ValueError('distinct_status_quality_fd_required')
    for path,sha in option['pins'].items():
        budget.check(reserve=CLEANUP_RESERVE)
        with Path(path).open('rb') as f:
            if hashlib.file_digest(f,'sha256').hexdigest()!=sha: raise ValueError('qualifier_source_or_native_pin_changed')
    for path,target in option['links'].items():
        budget.check(reserve=CLEANUP_RESERVE)
        if os.readlink(path)!=target: raise ValueError('qualifier_link_changed')
    envelope = {k:option[k] for k in ('run','ts_root','mounts','quality_fd')}
    envelope.update(host_before=host, native_status_fd=status_fd, deadline=budget.deadline)
    return ['/usr/bin/python3','-I',option['host_guard'],json.dumps(envelope),*command], (status_fd,option['quality_fd']), option


def finish(budget, context, observation, native_data):
    """Coordinator can attest native child closure only, never its own empty scope."""
    from scripts.dsh_prepare import StopUnconfirmed
    budget.check()
    if not observation.get('namespace_init_exit_verified') or not observation.get('reaped'):
        raise StopUnconfirmed('STOP_UNCONFIRMED: ordinary namespace child')
    pid=observation.get('pid')
    if type(pid) is not int or Path(f'/proc/{pid}').exists(): raise StopUnconfirmed('STOP_UNCONFIRMED: namespace owner retained')
    try: os.killpg(pid,0)
    except ProcessLookupError: pass
    else: raise StopUnconfirmed('STOP_UNCONFIRMED: namespace owner group retained')
    held(context['quality_fd'])
    host=context['host_before']
    return {'state':'NAMESPACE_TERMINAL_SCOPE_OBSERVER_PENDING', 'host_before':host,
        'host_scope_observation_sha256':hashlib.sha256(json.dumps(host,sort_keys=True).encode()).hexdigest(),
        'native_status_sha256':hashlib.sha256(native_data).hexdigest(), 'command':observation,'runtimeGrant':False}


def host_receipt(context):
    if 'host_before' not in context:
        raw=os.read(context['reader'],4097)
        if not raw or len(raw)>4096: raise ValueError('host_scope_observation_missing')
        host=json.loads(raw)
        if any(host[k]!=context[k] for k in ('unit','owner_pid','owner_birth','deadline','boot_id')):
            raise ValueError('foreign_host_scope_observation')
        context['host_before']=host
        context['host_scope_observation_sha256']=hashlib.sha256(raw).hexdigest()
    return context['host_before']


def stop_owner(budget, context, process, starttime):
    """One original-deadline whole-unit kill, only after exact actual ownership join."""
    from scripts.dsh_prepare import run
    host=host_receipt(context)
    if host['scope_pid']!=process.pid or host['scope_birth']!=starttime:
        raise ValueError('original_foreground_owner_not_joined')
    props,query=scope_show(context['unit'],budget,context['env'])
    context['stop_identity_observation']={'properties':props,'query':query}
    if props['LoadState']=='not-found': return
    if props['InvocationID']!=host['invocation'] or props['ControlGroup']!=host['control_group']:
        raise ValueError('owned_scope_replaced_no_signal')
    cg=Path(host['cgroup'])
    if [cg.stat().st_dev,cg.stat().st_ino]!=host['cgroup_identity']: raise ValueError('scope_inode_reused_no_signal')
    _,context['stop_observation']=run(['/usr/bin/systemctl','--user','kill','--signal=SIGKILL','--kill-whom=all',context['unit']],
        Path('/'),env=context['env'],timeout=min(1,budget.check()),deadline=budget.deadline,cleanup_reserve=0)


def finish_owner(budget, context, observation):
    """External root alone observes terminal scope, original PID/PGID and held lock."""
    from scripts.dsh_prepare import StopUnconfirmed, retain_custody
    receipt={'state':'STOP_UNCONFIRMED','runtimeGrant':False,
             'stop_identity_observation':context.get('stop_identity_observation'),
             'stop_observation':context.get('stop_observation')}
    terminal=False
    try:
        host=host_receipt(context)
        props,query=scope_show(context['unit'],budget,context['env'])
        cg=Path(host['cgroup']); events='ABSENT'
        if cg.exists():
            if [cg.stat().st_dev,cg.stat().st_ino]!=host['cgroup_identity']: raise ValueError('scope_inode_reused')
            events=dict(line.split() for line in (cg/'cgroup.events').read_text().splitlines())
            if events.get('populated')!='0': raise ValueError('owned_scope_populated')
        if props['LoadState']!='not-found':
            if (props['InvocationID']!=host['invocation'] or props['ActiveState'] not in ('inactive','failed')
                    or props['ControlGroup'] not in ('',host['control_group'])): raise ValueError('owned_scope_not_terminal')
        elif props['ControlGroup'] or cg.exists(): raise ValueError('missing_scope_retained_group')
        pid=observation.get('pid')
        if (pid!=host['scope_pid'] or observation.get('starttime_ticks')!=host['scope_birth']
                or not observation.get('reaped') or Path(f'/proc/{pid}').exists()): raise ValueError('original_owner_not_reaped')
        try: os.killpg(pid,0)
        except ProcessLookupError: pass
        else: raise ValueError('original_owner_pgid_retained')
        receipt.update(state='OWNED_SCOPE_TERMINAL_LOCK_STILL_HELD',
            host_before=host,host_scope_observation_sha256=context['host_scope_observation_sha256'],
            scope_after=props,cgroup_after=events,after_query=query,quality_after=held(context['quality_fd']))
        budget.check()
        terminal=True
        return receipt
    except BaseException as exc:
        receipt['failure']=type(exc).__name__+':'+str(exc)
        error=StopUnconfirmed('STOP_UNCONFIRMED: external exact scope closure');error.owner_receipt=receipt
        error.owner_context=context
        raise retain_custody(error, observation, exc) from exc
    finally:
        # Pending handoff remains recoverable: retain the original read endpoint
        # and its identity in owner_context until closure is actually established.
        if terminal:
            close_errors=[]
            for key in ('writer','reader'):
                fd=context[key]
                if fd is None: continue
                try:
                    if [os.fstat(fd).st_dev,os.fstat(fd).st_ino]!=context['pipe_identity'][key]:
                        raise ValueError('original_owner_descriptor_replaced')
                    os.close(fd);context[key]=None
                except BaseException as exc: close_errors.append(exc)
            if close_errors:
                error=StopUnconfirmed('STOP_UNCONFIRMED: owner receipt descriptor closure')
                error.owner_receipt=receipt;error.owner_context=context
                raise retain_custody(error, observation) from close_errors[0]
