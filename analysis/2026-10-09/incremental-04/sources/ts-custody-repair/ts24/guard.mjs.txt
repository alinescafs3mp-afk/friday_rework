// The preload is module-cached in the entry; this is a small existing-framework
// evidence seam, not a supervisor or an authority token.
export const guardEvidence = Object.freeze({
  refuse, pinned, emit, moduleResult, transform, request, graphEdges, isBuiltin,
  rawHash: value => crypto.createHash('sha256').update(value).digest('hex'),
})
function pinned(filename) {
  current()
  try {
    const canonical = fs.realpathSync(filename)
    const expected = pins[canonical]
    if (!expected || crypto.createHash('sha256').update(raw.read(canonical)).digest('hex') !== expected) refuse('UNPINNED_MODULE:' + canonical)
    emit({ event: 'NODE_MODULE_OR_PIN_CHECK', path: canonical, sha256: expected })
    return canonical
  } catch (error) { refuse('PIN_READ_ERROR:' + filename) }
}
let transformSequence = 0
function graphEdges(node) {
  return [...node.importedModules].map(m => ({ id: m.id, file: m.file, url: m.url }))
}
function transform(event, code, node, environment, transformId = null, extra = {}) {
  const canonical = pinned(node.file)
  const codeSha256 = crypto.createHash('sha256').update(code).digest('hex')
  if (event === 'VITE_TRANSFORM_INPUT') {
    if (codeSha256 !== pins[canonical]) refuse('TRANSFORM_INPUT_DIFFERS_FROM_PINNED_SOURCE')
    transformId = ++transformSequence
  }
  if (!Number.isInteger(transformId) || transformId <= 0) refuse('MISSING_ACTUAL_TRANSFORM_ID')
  emit({ event, id: node.id, canonical, environment, transformId,
    sourceSha256: pins[canonical], codeSha256, ...extra })
  return transformId
}
function request(event, meta, importer, requested, dynamic = false, cache = false) {
  // Observe cachedRequest itself: this seam sees builtins, externals, cycles and
  // already evaluated modules as well as new VM work in this exact worker.
  const known = moduleContexts.get(meta.id + '\0' + (meta.file ? fs.realpathSync(meta.file) : ''))
  const environment = known?.environment ?? null
  const row = { event, id: meta.id, importer, requested, dynamic, cache, environment }
  if ('externalize' in meta) {
    const externalize = meta.externalize
    row.externalize = externalize
    if (isBuiltin(externalize) || externalize === '/@vite/client') row.builtin = true
    else {
      const resolved = meta.__frwExternalCanonical ?? externalize
      row.externalCanonical = pinned(resolved.startsWith('file:') ? fileURLToPath(resolved) : resolved)
      row.externalSha256 = pins[row.externalCanonical]
    }
  } else {
    row.canonical = pinned(meta.file)
    row.sourceSha256 = pins[row.canonical]
    row.codeSha256 = crypto.createHash('sha256').update(meta.code).digest('hex')
    if (!known) refuse('REQUEST_WITHOUT_LOCAL_FETCH_CONTEXT')
  }
  emit(row)
}
const moduleContexts = new Map()
function moduleResult(event, result, tmp = null, wrapped = null, context = null) {
  current()
  if ('externalize' in result) {
    const url = result.externalize
    if (isBuiltin(url) || url === '/@vite/client') {
      emit({ event, externalize: url, type: result.type, environment: context?.environment ?? null }); return
    }
    if (!url.startsWith('file:') && !path.isAbsolute(url)) refuse('UNBOUND_EXTERNAL_RESULT')
    pinned(url.startsWith('file:') ? fileURLToPath(url) : url)
    emit({ event, externalize: url, type: result.type, environment: context?.environment ?? null }); return
  }
  if (typeof result.code !== 'string' || typeof result.file !== 'string' || typeof result.id !== 'string') refuse('UNBOUND_FINAL_RESULT')
  const canonical = pinned(result.file)
  const contextKey = result.id + '\0' + canonical
  if (context !== null) {
    if (typeof context.environment !== 'string') refuse('UNBOUND_MODULE_ENVIRONMENT')
    const old = moduleContexts.get(contextKey)
    if (old && old.environment !== context.environment) refuse('AMBIGUOUS_MODULE_ENVIRONMENT')
    moduleContexts.set(contextKey, context)
  }
  const effective = context ?? moduleContexts.get(contextKey)
  if (!effective) refuse('MODULE_CONTEXT_NOT_OBSERVED')
  const hash = value => crypto.createHash('sha256').update(value).digest('hex')
  let code = result.code
  let tempSha256 = null
  if (tmp !== null) {
    const canonicalTmp = fs.realpathSync(tmp)
    if (!canonicalTmp.startsWith(run + '/') || !fs.statSync(canonicalTmp).isFile()) refuse('FOREIGN_TRANSFORM_TEMP')
    const bytes = raw.read(canonicalTmp)
    tempSha256 = hash(bytes)
    if (event === 'VITEST_TMP_WRITTEN') code = bytes.toString('utf8')
    else if (tempSha256 !== hash(code)) refuse('WORKER_TEMP_BYTES_CHANGED')
  }
  const marker = '\n//# vitestCache='
  const at = code.lastIndexOf(marker)
  let finalSource = code
  let cacheMetadata = null
  if (at !== -1) {
    const suffix = code.slice(at + marker.length)
    if (!/^[A-Za-z0-9+/]+={0,2}$/.test(suffix)) refuse('UNBOUND_CACHE_SUFFIX')
    cacheMetadata = JSON.parse(Buffer.from(suffix, 'base64').toString('utf8'))
    if (cacheMetadata.id !== result.id || pinned(cacheMetadata.file) !== canonical) refuse('CACHE_METADATA_IDENTITY_CHANGED')
    finalSource = code.slice(0, at)
  }
  if (event === 'VITEST_TMP_WRITTEN' && (cacheMetadata === null || finalSource !== result.code)) refuse('FINAL_TEMP_SOURCE_NOT_JOINED')
  const expectedWrapper = "'use strict';async (__vite_ssr_exports__,__vite_ssr_import_meta__,__vite_ssr_import__,__vite_ssr_dynamic_import__,__vite_ssr_exportAll__,__vite_ssr_exportName__,__filename,__dirname,module,exports,require)=>{{" + code + "\n}}"
  if (wrapped !== null && wrapped !== expectedWrapper) refuse('UNBOUND_EVALUATOR_WRAPPER')
  emit({ event, id: result.id, canonical, environment: effective.environment,
    transformId: effective.transformId ?? null, requestId: effective.requestId ?? null,
    sourceSha256: pins[canonical], returnedTransformSha256: effective.returnedTransformSha256 ?? null,
    codeSha256: hash(code), finalSourceSha256: hash(finalSource), tmp, tempSha256,
    cacheMetadata, wrappedSha256: wrapped === null ? null : hash(wrapped),
    expectedWrappedSha256: wrapped === null ? null : hash(expectedWrapper) })
}
function mappedObjects(phase) {
  const maps = raw.read('/proc/self/maps', 'utf8')
  const objects = [...new Set(maps.split('\n').map(line => line.match(/^\S+\s+\S+\s+\S+\s+\S+\s+\S+\s+(\/.*)$/)?.[1]).filter(Boolean))]
  for (const object of objects) {
    if (object.endsWith(' (deleted)')) refuse('DELETED_MAPPED_OBJECT')
    pinned(object)
  }
  emit({ event: 'MAPPED_OBJECTS_SNAPSHOT', phase, maps, objects })
}

registerHooks({
  resolve(specifier, context, next) {
    const result = next(specifier, context)
    if (!result.url.startsWith('node:')) {
      const url = new URL(result.url)
      if (url.protocol !== 'file:' || url.search || url.hash) refuse('NON_FILE_MODULE')
      pinned(fileURLToPath(url))
    }
    return result
  },
  load(url, context, next) {
    if (url.startsWith('node:')) return next(url, context)
    const filename = pinned(fileURLToPath(url))
    const result = next(url, context)
    // Vite8 may use its exact pinned native compiler: mapped libs still UNBOUND.
    // This is limited trusted source qualification, never generic native safety.
    if (filename.endsWith('.node')) return result
    if (!['module', 'module-typescript', 'commonjs', 'commonjs-typescript', 'json'].includes(result.format)) refuse('MODULE_FORMAT')
    const overlay = overlays[filename]
    if (overlay) {
      if (pins[filename] !== overlay.original_sha256) refuse('OVERLAY_ORIGINAL_CHANGED')
      pinned(overlay.overlay)
      emit({ event: 'EXPLICIT_FRAMEWORK_OVERLAY', original: filename, originalSha256: overlay.original_sha256, overlay: overlay.overlay, overlaySha256: overlay.overlay_sha256 })
      return { ...result, source: raw.read(overlay.overlay) }
    }
    return { ...result, source: raw.read(filename) }
  },
})

  pinned(workerEntry)
  // Retain Vitest's pipes/IPC and deliberately map the same quality OFD to

  handle.once('exit', (code, signal) => { emit({ event: 'CHILD_EXIT', child_pid: handle.pid, code, signal }); active = null })
  return handle
}
