/** Exact ordinary Vitest4 source run. Prepared NOT_RUN; independent review first. */
import fs from 'node:fs'
import path from 'node:path'
import { fileURLToPath, pathToFileURL } from 'node:url'
import { sourceEvidence } from './source-evidence.mjs'

const root = path.dirname(fileURLToPath(import.meta.url))
const donor = '<PROJECT>/.donors/dsh'
const source = "<SOL_WORK>/FRW026-TS-EXISTING-OWNER-INTEGRATION-evidence/ts24/source"
const run = process.env.FRW_TS_RUN
const sourceObservation = sourceEvidence()
if (process.argv.length !== 2 || !run || !process.execArgv.includes(path.join(root, 'guard.mjs'))) throw new Error('EXACT_GUARDED_ENTRY_REQUIRED')
const { startVitest } = await import(pathToFileURL(path.join(donor, 'node_modules/vitest/dist/node.js')).href)
const { default: tsconfigPaths } = await import(pathToFileURL(path.join(donor, 'node_modules/vite-tsconfig-paths/dist/index.js')).href)
const { standardDecoratorPlugin } = await import(pathToFileURL(path.join(donor, 'vitest.shared.ts')).href)
const cases = ['private-network.spec.ts', 'network-start-repair.spec.ts'].map(name => path.join(source, 'packages/sandbox/sandbox-local/tests', name))
const progress = { state: 'RUNNING_SOURCE_ONLY', collected: [], results: [], errors: [], reason: null, runtimeGrant: false }
const save = () => fs.writeFileSync(path.join(run, 'progress.json'), JSON.stringify(progress, null, 2) + '\n', { mode: 0o600 })
const row = test => ({ id: test.id, file: test.module.moduleId, name: test.fullName, result: test.result() })
const reporter = {
  onTestModuleCollected(module) { progress.collected.push(...Array.from(module.children.allTests(), row)); save() },
  onTestCaseResult(test) { progress.results.push(row(test)); save() },
  onTestRunEnd(modules, errors, reason) {
    progress.errors = errors; progress.reason = reason
    const ids = progress.collected.map(r => r.id)
    const final = progress.results.map(r => r.id)
    const valid = reason === 'passed' && errors.length === 0 && ids.length === 24 &&
      new Set(ids).size === 24 && final.length === 24 && new Set(final).size === 24 &&
      final.every(id => ids.includes(id)) && progress.results.every(r => r.result.state === 'passed')
    progress.state = valid ? 'SOURCE_TS24_CASES_PASS_AWAIT_ROOT_CLOSURE' : 'SOURCE_TS24_FAILED'
    if (!valid) process.exitCode = 1
    save()
  },
}
save()
try {
  const ctx = await startVitest('test', cases, {
    root: source, config: false, run: true, watch: false, api: false, ui: false,
    pool: 'forks', maxWorkers: 1, fileParallelism: false, maxConcurrency: 1,
    isolate: true, retry: 0, passWithNoTests: false, allowOnly: false,
    testTimeout: 5000, hookTimeout: 5000, teardownTimeout: 2000,
    cache: false, fsModuleCache: false, coverage: { enabled: false },
    include: cases, exclude: [], setupFiles: [], globalSetup: [], reporters: [reporter],
    env: { FRW_TS_RUN: run, FRW_TS_QUALITY_FD: process.env.FRW_TS_QUALITY_FD, TMPDIR: run, HOME: run, LANG: 'C', LC_ALL: 'C' },
    execArgv: process.execArgv,
  }, {
    configFile: false, envFile: false, cacheDir: path.join(run, 'vite-cache'),
    plugins: [...sourceObservation.plugins, tsconfigPaths({ projects: ["<SOL_WORK>/FRW026-TS-EXISTING-OWNER-INTEGRATION-evidence/ts24/tsconfig.json", path.join(donor, 'tsconfig.base.json')] }), standardDecoratorPlugin()],
    optimizeDeps: { noDiscovery: true, include: [] },
    server: { middlewareMode: true, hmr: false, ws: false, watch: null, open: false },
    resolve: { alias: [
      { find: '@deepseek-ai/dsh-sandbox', replacement: path.join(source, 'packages/sandbox/sandbox/src/index.ts') },
      { find: '@deepseek-ai/node-addon-system/landlock-run', replacement: path.join(source, 'native/system/packages/entry/src/index.ts') },
    ] },
    ssr: { resolve: { conditions: ['node', 'development'] } },
  })
  sourceObservation.snapshot()
  await ctx.close()
  progress.closed = true; save()
} catch (error) {
  progress.state = 'SOURCE_TS24_REFUSED_OR_ERROR'
  progress.errors.push({ name: error.name, message: error.message }); save(); process.exitCode = 1
}
