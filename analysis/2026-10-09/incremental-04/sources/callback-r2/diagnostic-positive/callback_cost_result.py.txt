"""Finite read-only consumer of one fresh diagnostic's actual saved evidence.
No execution, signalling, admission, mutation or synthetic terminal receipt.
"""
from pathlib import Path
import hashlib,json

KEYS=('open','shutil.copyfile','shutil.copystat','os.chown','object.__getattr__','other')
ASSIGNMENT='FRW041-BROWSER-CALLBACK-COST-DIAGNOSTIC-POSITIVE'
def read(p):return json.loads(p.read_bytes())
def digest(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def saved_progress(path):
 try:
  value=read(Path(path))
  if not isinstance(value,dict):return {'available':False,'reason':'NONOBJECT_JSON','cost':'UNKNOWN'}
  return {'available':True,'value':value}
 except (FileNotFoundError,json.JSONDecodeError,UnicodeDecodeError) as error:
  return {'available':False,'reason':type(error).__name__,'cost':'UNKNOWN','unobserved_is_zero':False}
def measured(sample):
 if not isinstance(sample,dict) or sample.get('valid') is not True:
  return {'available':False,'reason':'INVALID_OR_MISSING_SAMPLE','source':sample}
 try:
  assert sample['schema']=='friday.browser.callback-cost.v1'
  assert tuple(sample['audit_keys'])==KEYS and len(sample['audit'])==6
  p=sample['profile'];a=sample['audit'];o=sample['audit_inside_profile']
  assert len(p)==4 and len(o)==3 and all(len(r)==4 for r in a)
  assert all(type(v) is int and v>=0 for r in [p,o,*a] for v in r)
  assert p[0]==p[1] and all(r[0]==r[1] for r in a)
  assert sample['depths']==[0,0] and sample['foreign_calls']==0 and sample['balanced'] is True
  assert sample['first_fault'] is None and sample['fault_count']==0
  assert type(sample['audit_calls']) is int and type(sample['nested_audit_calls']) is int
  assert min(sample['audit_calls'],sample['nested_audit_calls'])>=0
  assert sample['audit_calls']==sum(r[0] for r in a)+sample['nested_audit_calls']
  outer=sum(r[2] for r in a);union=p[2]+outer-o[1]
  assert o[0]<=sum(r[1] for r in a) and o[1]<=outer and o[1]<=p[2]
  baseline=sample['baseline'];assert len(baseline)==3
  assert all(type(v) is int and v>=0 for v in baseline)
  assert all(type(sample[k]) is int and sample[k]>=0 for k in
   ('sample_main_thread_cpu_ns','sample_process_cpu_ns','sample_wall_ns'))
  main=sample['sample_main_thread_cpu_ns']-baseline[1]
  process=sample['sample_process_cpu_ns']-baseline[0]
  residual=main-union;other=process-main
  assert min(union,main,process,residual,other)>=0
  assert (union,main,process,residual,other)==(
   sample['observer_body_union_cpu_ns'],sample['main_thread_cpu_since_activation_ns'],
   sample['process_cpu_since_activation_ns'],sample['unassigned_main_thread_cpu_ns'],
   sample['process_minus_main_thread_cpu_ns'])
  return {'available':True,'union_cpu_ns':union,'main_cpu_ns':main,'process_cpu_ns':process,
   'unassigned_main_cpu_ns':residual,'process_minus_main_thread_cpu_ns':other,
   'process_thread_sampling_skew':'Paired-clock main-thread CPU INCLUDED; exact worker attribution UNKNOWN',
   'audit_by_class_thread_cpu_ns':{k:r[2] for k,r in zip(KEYS,a)},
   'profile_entries_completed':p[1],'outer_audit_entries_completed':sum(r[1] for r in a),
   'profile_overlap_audit_cpu_ns':o[1],
   'instrumentation':'Included in measured bodies and residual; uninstrumented cost UNKNOWN',
   'boundary':sample['counter_boundary'],'after_saved_boundary':'UNKNOWN'}
 except (AssertionError,KeyError,TypeError,ValueError):
  return {'available':False,'reason':'INCONSISTENT_SOURCE_COUNTERS','source':sample}

def continuity(initial,samples):
 """Saved cumulative samples only. No native history/terminal receipt generation."""
 try:
  assert isinstance(initial,dict) and initial.get('valid') is True
  baseline=list(initial['baseline']);assert len(baseline)==3
  assert samples,'NO_ACTUAL_SAVED_SAMPLE'
  previous=None
  for s in samples:
   assert measured(s)['available'] is True,'UNAVAILABLE_SAMPLE'
   assert s['baseline']==baseline,'BASELINE_CHANGED_NO_RESET_ALLOWED'
   counters=[*s['profile'],*(v for row in s['audit'] for v in row),
    *s['audit_inside_profile'],s['audit_calls'],s['nested_audit_calls'],
    s['sample_main_thread_cpu_ns'],s['sample_process_cpu_ns'],s['sample_wall_ns']]
   if previous is not None:assert all(a<=b for a,b in zip(previous,counters)),'NONMONOTONIC_CUMULATIVE_SAMPLE'
   previous=counters
  return {'available':True,'samples':len(samples),'baseline':baseline}
 except (AssertionError,KeyError,TypeError,ValueError) as error:
  return {'available':False,'reason':'INVALID_OR_DISCONTINUOUS_SAVED_SAMPLES',
   'detail':str(error),'cost':'UNKNOWN','unobserved_is_zero':False}

def report_samples(progress):
 """Exact producer JSON roundtrip seam; no successful native receipt is inferred."""
 try:
  reports=progress['reports'];copies=progress['callback_cost_reports']
  assert isinstance(reports,list) and isinstance(copies,list) and len(copies)==len(reports)<=114
  identities=[(r['nodeid'],r['when']) for r in copies]
  assert identities==[(r['nodeid'],r['when']) for r in reports]
  assert len(set(identities))==len(identities) and all(w in ('setup','call','teardown') for n,w in identities)
  sequence=[r['cost'] for r in copies]
  final_present='callback_cost_final_profile_active' in progress
  if 'callback_cost_latest' in progress:
   boundary=progress['callback_cost_latest_boundary']
   count=boundary['report_count'];after_final=boundary['after_final']
   assert type(count) is int and 0<=count<=len(copies),'INVALID_SNAPSHOT_REPORT_BOUNDARY'
   assert type(after_final) is bool,'INVALID_SNAPSHOT_FINAL_BOUNDARY'
   if after_final:
    assert final_present and count==len(copies),'SNAPSHOT_FINAL_BOUNDARY_WITHOUT_FINAL'
    sequence.append(progress['callback_cost_final_profile_active'])
    sequence.append(progress['callback_cost_latest'])
   else:
    # Insert only the explicitly bounded snapshot; never sort report history.
    sequence.insert(count,progress['callback_cost_latest'])
    if final_present:sequence.append(progress['callback_cost_final_profile_active'])
  elif final_present:sequence.append(progress['callback_cost_final_profile_active'])
  return continuity(progress['callback_cost_initial'],sequence)
 except (AssertionError,KeyError,TypeError,ValueError) as error:
  return {'available':False,'reason':'INVALID_REPORT_IDENTITY_OR_PAYLOAD','detail':str(error),
   'cost':'UNKNOWN','unobserved_is_zero':False}

def compare_overhead(original,timed,closures):
 """Actual finite control comparison only, never diagnostic/native admission."""
 try:
  assert original['mode']=='overhead-original' and timed['mode']=='overhead-timed'
  assert original['owner']['clock']==timed['owner']['clock']
  for data in (original,timed):
   mode=data['mode'];closure=closures[mode]
   assert closure['actual_exit_code']==0 and closure['terminal_reaped'] is True
   assert closure['whole_group_closed'] is True and closure['quality_lock_returned'] is True
   assert closure['clock_sha256']==data['owner']['step']['clock_sha256']
   assert data['native_effect_receipt'] is False and data['runtime_grant'] is False
  a=original['matched_control'];b=timed['matched_control']
  assert a['original_full_hooks'] is b['original_full_hooks'] is True
  assert a['timed_hooks'] is False and b['timed_hooks'] is True
  assert a['ordinary_calls']==b['ordinary_calls']==512
  assert a['setup_and_activation_outside_sample'] is b['setup_and_activation_outside_sample'] is True
  assert {k:v for k,v in a['audit_deltas'].items() if v}=={k:v for k,v in b['audit_deltas'].items() if v},'ACTUAL_EVENT_MIX_DIFFERENT'
  for row in (a,b):assert all(type(row[k]) is int and row[k]>=0 for k in ('main_ns','process_ns','wall_ns'))
  return {'available':True,'main_control_difference_ns':b['main_ns']-a['main_ns'],
   'process_control_difference_ns':b['process_ns']-a['process_ns'],
   'wall_control_difference_ns':b['wall_ns']-a['wall_ns'],
   'scope':'Two actual CLOSED full-hook512 controls only; sample noise/order included',
   'diagnostic_cost_calibration':False,'extrapolation_to38_cases':False,
   'dominant_product_cost':'UNKNOWN','runtime_grant':False,'production_browser_grant':False}
 except (AssertionError,KeyError,TypeError,ValueError) as error:
  return {'available':False,'reason':'UNQUALIFIED_OR_UNMATCHED_CONTROL','detail':str(error),
   'overhead_usefulness':'UNKNOWN','runtime_grant':False,'production_browser_grant':False}

def analyze(root):
 root=Path(root).resolve();out=root/'project'
 admission=read(root/'runtime-admission.json');clock=read(root/'runtime-clock.json')
 assert admission['assignment']==clock['assignment']==ASSIGNMENT
 assert admission['generation']==clock['generation']==1
 assert clock['budget_seconds']==180 and clock['deadline_monotonic']-clock['started_monotonic']==180
 assert admission['source_package_sha256']==digest(root/'execution-package-pins.json')
 assert admission['effect_proposal_sha256']==digest(root/'effect-admission-proposal.json')
 assert admission['production_browser_grant'] is False
 for p,h in read(root/'execution-package-pins.json').items():assert digest(Path(p))==h,p
 backend=read(out/'backend-result.json');terminal=backend['terminal']
 assert 'exit_code' in terminal and 'session_id' not in terminal
 cessation=read(out/'cessation.json')
 assert cessation['process_unit_closed'] is True and cessation['within_original_clock'] is True
 assert cessation['assignment']==ASSIGNMENT and cessation['generation']==1
 assert cessation['runtime_clock_sha256']==digest(root/'runtime-clock.json')
 assert cessation['original_deadline_monotonic']==clock['deadline_monotonic']
 assert cessation['backend_terminal_exit']==terminal['exit_code']
 assert cessation['quality_lock_fresh_probe_returned'] is True
 assert not any(cessation.get(k) for k in ('exact_pid_and_group_scan','proc_scan_errors'))
 integrity=cessation['closed'] is True and cessation['module_origin_inventory_complete'] is True and cessation['mapping_inventory_complete'] is True and not any(cessation.get(k) for k in (
  'input_changes','module_origin_changes','mapping_changes','distribution_metadata_changes','capture_errors'))
 if not integrity:
  return {'state':'ACTUAL_UNIT_CLOSED_INTEGRITY_EVIDENCE_INCOMPLETE_NO_PRODUCT_PASS',
   'actual_backend_exit':terminal['exit_code'],'whole_process_closure':cessation,
   'cost':'UNKNOWN','reports':'UNKNOWN','runtime_grant':False,
   'production_browser_grant':False,'unobserved_is_zero':False}
 # Terminal and whole-group cessation are independently verified FIRST.
 # A killed w-mode save can be absent, empty or truncated; never synthesize zero.
 saved=saved_progress(out/'pytest-progress.json')
 if not saved['available']:
  return {'state':'PROGRESS_UNAVAILABLE_PARTIAL_DIAGNOSTIC_NO_PRODUCT_PASS',
   'assignment':ASSIGNMENT,'generation':1,'actual_backend_exit':terminal['exit_code'],
   'whole_process_closure':cessation,'original_clock':clock,
   'progress_reason':saved['reason'],'cost':'UNKNOWN','reports':'UNKNOWN',
   'actual_original38_complete':'UNKNOWN','previous_sample':'NOT_INDEPENDENTLY_RETAINED',
   'after_last_saved_cpu':'UNKNOWN','unobserved_is_zero':False,'runtime_grant':False,
   'production_browser_grant':False,'performance_fix':None}
 progress=saved['value']
 joined=report_samples(progress)
 if not joined['available']:
  return {'state':'SAVED_PROGRESS_INVALID_OR_DISCONTINUOUS_NO_PRODUCT_PASS','cost':'UNKNOWN',
   'sample_join':joined,'actual_backend_exit':terminal['exit_code'],
   'whole_process_closure':cessation,'runtime_grant':False,'production_browser_grant':False}
 reports=progress['reports'];copies=progress.get('callback_cost_reports',[])
 assert len(copies)==len(reports)<=114
 assert [(r['nodeid'],r['when']) for r in copies]==[(r['nodeid'],r['when']) for r in reports]
 observations=[measured(row.get('cost')) for row in copies]
 final=measured(progress.get('callback_cost_final_profile_active'))
 latest=measured(progress.get('callback_cost_latest'))
 nodes=progress['nodes'];full=(len(nodes)==38 and len(set(nodes))==38 and len(reports)==114
  and {(r['nodeid'],r['when']) for r in reports}=={(n,w) for n in nodes for w in ('setup','call','teardown')}
  and all(r['outcome']=='passed' for r in reports) and not progress['deselected']
  and not progress.get('skipped') and terminal['exit_code']==0)
 history=progress.get('ownership_history',[]);metadata=progress.get('copy_metadata_history',[])
 directory=progress.get('directory_sync_history',[])
 history_complete=bool(history and metadata and directory and progress.get('ownership_evidence_verified')
  and progress.get('copy_evidence_verified') and progress.get('directory_sync_evidence_verified'))
 # All original exact syscall/frame/ancestry/history assertions remain in run.py.
 # Timing aggregates cannot replace them or prove a missing syscall succeeded.
 metric_history_join=False
 if final['available'] and history_complete:
  s=progress['callback_cost_final_profile_active']
  metric_history_join=(s['audit'][0][0]>=len(directory)
   and s['audit'][3][0]>=len(history)
   and s['profile'][1]>=len(directory)+len(metadata))
 ready=full and history_complete and metric_history_join and final['available'] and all(x['available'] for x in observations) and joined['available']
 return {'state':'FULL_DIAGNOSTIC_EVIDENCE_REQUIRES_ROOT_REVIEW' if ready else 'PARTIAL_OR_INVALID_DIAGNOSTIC_NO_PRODUCT_PASS',
  'assignment':ASSIGNMENT,'generation':1,'actual_backend_exit':terminal['exit_code'],
  'actual_original38_complete':full,'reports':len(reports),'history_complete':history_complete,
  'metric_history_lower_bounds_met':metric_history_join,'per_report_observations':observations,
  'saved_sample_continuity':joined,
  'final_profile_active':final,'latest_saved':latest,'original_clock':clock,
  'whole_process_closure':cessation,'runtime_grant':False,'production_browser_grant':False,
  'performance_fix':None,'finer_product_attribution':'UNKNOWN',
  'after_last_saved_cpu':'UNKNOWN','historical_G20':'FAILED137_CONSUMED_CLOSED_NO_RETRY',
  'all19_full725_fourcompat_install_live_journeys':'STILL_SEPARATELY_REQUIRED_UNACCEPTED'}
