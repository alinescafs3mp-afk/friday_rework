"""One finite exec wrapper following the existing launcher/GNU pattern.
No spawning, waiting loop, supervisor, retry, clock or admission creation.
Root owns sequence, prior terminal/reap/CPU/cessation and aggregate caps.
"""
from pathlib import Path
import os,sys,json,hashlib,time,resource,math
HERE=Path(__file__).resolve().parent
sys.path.insert(0,str(HERE))
from owner_step_inputs import MODES,proc_stat,digest
assert len(sys.argv)==2 and sys.argv[1] in MODES
mode=sys.argv[1];RUN=HERE/'run'
clock=json.loads((HERE/'clock.json').read_bytes());admission=json.loads((HERE/'admission.json').read_bytes())
root_step=json.loads((RUN/('root-step-'+mode+'.json')).read_bytes())
assert clock['assignment']=='FRW041-CALLBACK-COST-CHANGED-SOURCE-QUALIFICATION' and clock['generation']==1
assert clock['deadline_monotonic']-clock['started_monotonic']==clock['budget_seconds']==180
assert clock['cpu_seconds']==45 and clock['boot_id']==Path('/proc/sys/kernel/random/boot_id').read_text().strip()
assert admission['assignment']==clock['assignment'] and admission['generation']==1
assert admission['execution_authorized'] is True and admission['production_browser_grant'] is False
assert admission['finite_sequence_sha256']==digest(HERE/'finite-sequence.json')
assert admission['input_pins_sha256']==digest(HERE/'input-pins.json')
assert admission['launcher_sha256']==digest(__file__)
pins=json.loads((HERE/'input-pins.json').read_bytes())
for path in (HERE/'owner_step_inputs.py',HERE/'probe-input-pins.json',HERE/'finite-sequence.json',Path(__file__)):
 assert digest(path)==pins[str(path)]
for p,h in json.loads((HERE/'probe-input-pins.json').read_bytes()).items():assert digest(p)==h,p
assert root_step['clock_sha256']==digest(HERE/'clock.json') and root_step['mode']==mode
assert root_step['ordinal']==MODES.index(mode) and root_step['completed_modes']==list(MODES[:MODES.index(mode)])
assert root_step['all_previous_terminal_reaped_groups_closed_lock_returned'] is True
assert root_step['cumulative_parent_and_group_cpu_enforcement_reviewed'] is True
assert root_step['existing_owner_route_review_sha256']==admission['existing_owner_route_review_sha256']
owner=proc_stat(os.getppid());assert owner['pid']==clock['owner_pid'] and owner['start_ticks']==clock['owner_start_ticks']
os.umask(0o077);os.sched_setaffinity(0,{0,1});resource.setrlimit(resource.RLIMIT_AS,(2<<30,2<<30))
if os.getsid(0)!=os.getpid():os.setsid()
assert os.getpgrp()==os.getpid()==os.getsid(0)
stat_=proc_stat(os.getpid())
prior=sum(r['wait4_user_cpu_seconds']+r['wait4_system_cpu_seconds'] for r in root_step['previous_receipts'])
owner_cpu=(owner['cpu_ticks']-clock['owner_cpu_start_ticks'])/os.sysconf('SC_CLK_TCK')
assert owner_cpu>=0
cumulative=prior+owner_cpu+stat_['cpu_ticks']/os.sysconf('SC_CLK_TCK')
child_cpu=math.floor(45-cumulative-3);assert 0<child_cpu<=42
seconds=min(60,math.floor(clock['deadline_monotonic']-time.monotonic()-31));assert seconds>0
python=admission['interpreter'];assert digest(python)==admission['interpreter_sha256']
argv=[str(HERE/'run.py')] if mode=='source-cases' else [str(HERE/'callback_probe.py'),mode,str(RUN/('callback-'+mode+'.json'))]
gnu=['gnutimeout','--signal=KILL','--verbose',str(seconds)+'s',python,'-I','-B','-S',*argv]
step=dict(root_step);step.update(admission_sha256=digest(HERE/'admission.json'),
 gnu_identity={k:stat_[k] for k in ('pid','ppid','pgid','sid','start_ticks')},
 expected_pgid=stat_['pgid'],gnu_argv=gnu,gnu_seconds=seconds,child_cpu_seconds=child_cpu,
 actual_total_cpu_before_step=cumulative,actual_before_exec_monotonic=time.monotonic())
with (RUN/('owner-step-'+mode+'.json')).open('x') as f:json.dump(step,f,indent=2)
os.environ.clear();os.environ.update(PATH='/usr/bin:/bin',LANG='C.UTF-8',LC_ALL='C.UTF-8',
 PYTEST_DISABLE_PLUGIN_AUTOLOAD='1',HOME=str(RUN/'home'),TMPDIR=str(RUN/'tmp'))
os.chdir(RUN)
assert time.monotonic()+seconds+30<clock['deadline_monotonic']
os.execv('/usr/bin/gnutimeout',gnu)
