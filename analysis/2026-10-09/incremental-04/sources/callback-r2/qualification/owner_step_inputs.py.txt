"""Read-only input checks for a step of the existing root-owned finite route.
Does not spawn, wait, signal, admit, renew a clock, or create an owner.
Actual root wait4/cgroup/terminal/cessation receipts are mandatory inputs.
"""
from pathlib import Path
import json,hashlib,os,sys,time,resource,math
HERE=Path(__file__).resolve().parent
MODES=('source-cases','positive','profile-entry','profile-exit','audit-entry','audit-exit',
 'foreign','baseexceptions','deny','overhead-original','overhead-timed')
def digest(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def proc_stat(pid):
 raw=Path('/proc/'+str(pid)+'/stat').read_text();tail=raw[raw.rfind(')')+2:].split()
 return {'pid':pid,'ppid':int(tail[1]),'pgid':int(tail[2]),'sid':int(tail[3]),
  'start_ticks':int(tail[19]),'cpu_ticks':int(tail[11])+int(tail[12])}
def check(mode,argv):
 assert mode in MODES
 clock=json.loads((HERE/'clock.json').read_bytes());admission=json.loads((HERE/'admission.json').read_bytes())
 step=json.loads((HERE/'run'/('owner-step-'+mode+'.json')).read_bytes())
 assert clock['assignment']=='FRW041-CALLBACK-COST-CHANGED-SOURCE-QUALIFICATION' and clock['generation']==1
 assert clock['budget_seconds']==180 and clock['cpu_seconds']==45
 assert clock['deadline_monotonic']-clock['started_monotonic']==180
 assert clock['boot_id']==Path('/proc/sys/kernel/random/boot_id').read_text().strip()
 assert admission['assignment']==clock['assignment'] and admission['generation']==1
 assert admission['execution_authorized'] is True and admission['production_browser_grant'] is False
 assert admission['finite_sequence_sha256']==digest(HERE/'finite-sequence.json')
 assert admission['input_pins_sha256']==digest(HERE/'input-pins.json')
 pins=json.loads((HERE/'input-pins.json').read_bytes())
 for path in (Path(argv[0]),HERE/'owner_step_inputs.py',HERE/'probe-input-pins.json',HERE/'finite-sequence.json'):
  assert digest(path)==pins[str(path)]
 assert step['mode']==mode and step['clock_sha256']==digest(HERE/'clock.json')
 assert step['admission_sha256']==digest(HERE/'admission.json')
 assert step['ordinal']==MODES.index(mode) and step['completed_modes']==list(MODES[:step['ordinal']])
 assert step['all_previous_terminal_reaped_groups_closed_lock_returned'] is True
 assert step['previous_receipts'] and len(step['previous_receipts'])==step['ordinal'] or step['ordinal']==0 and step['previous_receipts']==[]
 prior_children=0.0
 for i,row in enumerate(step['previous_receipts']):
  assert row['mode']==MODES[i] and row['kernel_wait4_rusage_includes_waited_descendants'] is True
  assert row['terminal_reaped'] is True and row['whole_group_closed'] is True and row['lock_returned'] is True
  assert row['clock_sha256']==step['clock_sha256'] and row['quality_scope']==clock['quality_scope']
  for key in ('terminal_ref','cessation_ref','rusage_ref'):
   ref=row[key];assert digest(ref['path'])==ref['sha256']
  u=row['wait4_user_cpu_seconds'];s=row['wait4_system_cpu_seconds']
  assert type(u) in (int,float) and type(s) in (int,float) and min(u,s)>=0 and math.isfinite(u+s)
  prior_children+=u+s
 owner=proc_stat(clock['owner_pid']);assert owner['start_ticks']==clock['owner_start_ticks']
 parent=proc_stat(os.getppid())
 assert {k:parent[k] for k in ('pid','ppid','pgid','sid','start_ticks')}==step['gnu_identity']
 assert parent['ppid']==owner['pid'] and parent['pgid']==step['expected_pgid']
 assert os.readlink('/proc/'+str(parent['pid'])+'/exe')=='/usr/bin/gnutimeout'
 assert digest('/proc/'+str(parent['pid'])+'/exe')=='1ba715580334044dbd32873bddd7f8a19687cda1cf5e9f44be7266af6c048766'
 assert digest('/proc/self/exe')==admission['interpreter_sha256']
 assert sys.flags.isolated and sys.flags.no_site and sys.dont_write_bytecode
 assert sys.argv==argv
 expected=['gnutimeout','--signal=KILL','--verbose',str(step['gnu_seconds'])+'s',admission['interpreter'],'-I','-B','-S',*argv]
 actual=[s.decode() for s in Path('/proc/'+str(parent['pid'])+'/cmdline').read_bytes().split(b'\0') if s]
 assert actual==expected and step['gnu_argv']==expected and 0<step['gnu_seconds']<=60
 assert set(os.environ)<=set(('PATH','HOME','TMPDIR','LANG','LC_ALL','PYTEST_DISABLE_PLUGIN_AUTOLOAD'))
 assert os.getpgrp()==parent['pgid']
 assert os.environ.get('PYTEST_DISABLE_PLUGIN_AUTOLOAD')=='1' and os.environ.get('PATH')=='/usr/bin:/bin'
 assert Path(os.environ['HOME']).resolve()==(HERE/'run'/'home').resolve()
 assert Path(os.environ['TMPDIR']).resolve()==(HERE/'run'/'tmp').resolve()
 # Existing quality cgroup must already bind this owner and GNU child. Never provision here.
 scope=Path(clock['quality_scope']);assert str(scope).startswith('/sys/fs/cgroup/')
 assert (scope/'memory.max').read_text().strip()==str(2<<30)
 assert (scope/'memory.swap.max').read_text().strip()=='0'
 assert owner['pid'] in map(int,(scope/'cgroup.procs').read_text().split())
 assert parent['pid'] in map(int,(scope/'cgroup.procs').read_text().split())
 ticks=os.sysconf('SC_CLK_TCK')
 owner_cpu=(owner['cpu_ticks']-clock['owner_cpu_start_ticks'])/ticks
 assert owner_cpu>=0
 cumulative=owner_cpu+prior_children+parent['cpu_ticks']/ticks
 # Remaining hard child budget includes one-second clock/CPU rounding margin and root closure reserve.
 maximum_child_cpu=math.floor(45-cumulative-3)
 child_cpu=step['child_cpu_seconds']
 assert type(child_cpu) is int and 0<child_cpu<=maximum_child_cpu<=42
 assert abs(step['actual_total_cpu_before_step']-cumulative)<1
 assert time.monotonic()+step['gnu_seconds']+30<clock['deadline_monotonic']
 assert sorted(os.sched_getaffinity(0))==[0,1]
 resource.setrlimit(resource.RLIMIT_AS,(2<<30,2<<30))
 resource.setrlimit(resource.RLIMIT_CPU,(child_cpu,child_cpu))
 return {'clock':clock,'step':step,'cumulative_cpu_lower_bound':cumulative,
  'root_closure_cpu_reserve':3,'actual_remaining_wall':clock['deadline_monotonic']-time.monotonic()}
