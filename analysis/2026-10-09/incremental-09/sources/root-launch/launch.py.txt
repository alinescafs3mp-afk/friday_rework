# Inert complete source export for analysis; not a runtime grant.
"""Root launch binding only: retain the existing quality lock and exec in place.
Not a supervisor or observation receipt. An exact root permit is required.
"""
import fcntl
import hashlib
import json
import os
from pathlib import Path
import stat
import time

ROOT = Path(__file__).resolve().parent
PERMIT = ROOT / 'permit.json'
LAUNCH = ROOT / 'launch-binding.json'
QUALIFICATION = Path('<PRIMITIVE_CANDIDATE>/qualification')
INTERPRETER = '<SELECTED_INTERPRETER>'
os.umask(0o077)
assert not LAUNCH.exists()
p = json.loads(PERMIT.read_bytes())
assert p['source_reviewed_observation_authorized'] is True
assert p['product_mode_admission'] is False
assert p['qualification_directory'] == str(QUALIFICATION)
assert p['interpreter'] == INTERPRETER
assert p['scope'] == 'ONE_PRIMITIVE_BUNDLE_ONE_BIRTH_NO_RETRY'
assert not (QUALIFICATION / 'f6-primitive-run').exists()
fd = os.open('<QUALITY_EXCLUSIVE_LOCK>', os.O_RDONLY | os.O_NOFOLLOW)
s = os.fstat(fd)
identity = [s.st_dev, s.st_ino, s.st_uid, s.st_gid, s.st_nlink, s.st_size, stat.S_IMODE(s.st_mode)]
assert identity == [PUBLIC_LOCK_DEV, PUBLIC_LOCK_INO, PUBLIC_LOCK_UID, PUBLIC_LOCK_GID, 1, 0, 0o664]
fcntl.flock(fd, fcntl.LOCK_EX | fcntl.LOCK_NB)
os.set_inheritable(fd, True)
argv = [INTERPRETER, '-I', '-B', '-S', str(QUALIFICATION / 'invoke_qualification.py'), '--primitive', str(PERMIT)]
environ = {'PATH': '/usr/bin:/bin', 'LANG': 'C', 'LC_ALL': 'C'}
record = {'state': 'LOCK_HELD_BEFORE_SAME_PID_EXEC_NOT_AN_OBSERVATION',
          'pid': os.getpid(), 'proc_stat': Path('/proc/self/stat').read_text(),
          'boot_id': Path('/proc/sys/kernel/random/boot_id').read_text().strip(),
          'monotonic_before_exec': time.monotonic(), 'argv': argv,
          'permit_sha256': hashlib.sha256(PERMIT.read_bytes()).hexdigest(),
          'quality_lock_identity': identity, 'quality_lock_fd': fd,
          'new_supervisor': False, 'new_budget': False}
out = os.open(LAUNCH, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600)
with os.fdopen(out, 'w') as stream:
    json.dump(record, stream, indent=2)
    stream.write('\n')
    stream.flush()
    os.fsync(stream.fileno())
dirfd = os.open(ROOT, os.O_RDONLY | os.O_DIRECTORY)
os.fsync(dirfd)
os.close(dirfd)
os.execve(INTERPRETER, argv, environ)
