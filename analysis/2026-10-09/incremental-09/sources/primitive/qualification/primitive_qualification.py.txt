# Inert complete source export for analysis; not a runtime grant.
"""One F6 primitive bundle in the EXISTING terminal caller, source only.

This branch does not call invoke_existing_owner or admit any product mode.
It reuses final CallerWall/NativeBirth and Custody without another supervisor.
A root-issued, independently source-reviewed observation permit is required;
it is NOT an observed/accepted primitive receipt. No permit is supplied here.
"""
import errno
import hashlib
import json
import math
import os
from pathlib import Path
import resource
import signal
import sys
import time
from types import SimpleNamespace

BUNDLE = ('native-birth-exec', 'timer-create-errno', 'timer-settime-errno',
          'subreaper-set-get', 'live-handle-close-refusal',
          'second-birth-refusal', 'real-stop-continue', 'terminal-wait4-reentry')
LIMITS = {'wall_seconds': 180, 'tree_cpu_seconds': 45, 'root_cpu_seconds': 6,
          'as_bytes': 2 << 30, 'affinity': [0, 1]}
CONTROL_FILES = ('primitive_qualification.py', 'primitive_target.py',
                 'invoke_qualification.py')
REUSED_FILES = ('existing_caller_join.py', 'caller_wall.py', 'native_birth.py',
                'finite_birth.c', 'finite_custody.py', 'finite_kernel.py',
                'owner_step_inputs.py')


def digest(path):
    with Path(path).open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()


def verify_permit(here, path):
    # Reads only. In particular no candidate import, DSO load or clock here.
    permit = json.loads(Path(path).read_bytes())
    assert permit['schema'] == 'friday.f6.primitive-observation-permit.v1'
    assert permit['source_reviewed_observation_authorized'] is True
    assert permit['product_mode_admission'] is False
    assert permit['scope'] == 'ONE_PRIMITIVE_BUNDLE_ONE_BIRTH_NO_RETRY'
    assert permit['budget_authority'] == 'EXPLICIT_ROOT_PERMIT_NOT_INFERRED_FROM_SOURCE_CASE_TASK'
    assert permit['limits'] == LIMITS and permit['bundle'] == list(BUNDLE)
    assert type(permit['generation']) is int and permit['generation'] > 0
    assert permit['assignment'] and permit['qualification_directory'] == str(here)
    assert sys.flags.isolated and sys.flags.no_site and sys.dont_write_bytecode
    assert not sys.flags.optimize
    assert digest('/proc/self/exe') == digest(permit['interpreter']) == permit['interpreter_sha256']
    assert digest(here/'source-manifest.json') == permit['source_manifest_sha256']
    # Use the ACTUAL existing manifest schema: {files:[{path,sha256,...},...]}
    manifest = json.loads((here/'source-manifest.json').read_bytes())
    entries = {row['path']: row for row in manifest['files']}
    assert len(entries) == len(manifest['files'])
    for row in entries.values():
        assert digest(row['path']) == row['sha256']
    for name in (*CONTROL_FILES, *REUSED_FILES):
        assert str(here/name) in entries
    assert digest(here/'finite_birth.so') == permit['native_birth_binary_sha256']
    for path, sha in permit['resolved_runtime_dependencies'].items():
        assert digest(path) == sha
    assert permit['resolved_runtime_dependencies']  # Root must bind actual providers.
    release = Path('/proc/sys/kernel/osrelease').read_text().strip()
    assert release == permit['kernel_release']
    config = Path('/boot/config-'+release)
    assert digest(config) == permit['kernel_config_sha256']
    assert 'CONFIG_POSIX_CPU_TIMERS_TASK_WORK=y' in config.read_text().splitlines()
    review_ref = permit['independent_source_review']
    assert digest(review_ref['path']) == review_ref['sha256']
    review = json.loads(Path(review_ref['path']).read_bytes())
    assert review['state'] == 'SOURCE_REVIEWED_FOR_PRIMITIVE_OBSERVATION'
    assert review['accepted_by_independent_root'] is True
    assert review['source_manifest_sha256'] == permit['source_manifest_sha256']
    assert review['native_birth_binary_sha256'] == permit['native_birth_binary_sha256']
    assert review['kernel_config_sha256'] == permit['kernel_config_sha256']
    assert review['bundle'] == list(BUNDLE)
    # Deliberately no observed=True predicate: that is what this run measures.
    return permit


def cpu(which):
    usage = resource.getrusage(which)
    return usage.ru_utime + usage.ru_stime


def main(here, permit_path):
    here = Path(here).resolve()
    permit = verify_permit(here, permit_path)
    # No renewed control/mode budget: one permit, one directory, one clock,
    # one native attempted birth. Existing directories are never reused.
    run = here/'f6-primitive-run'
    run.mkdir(mode=0o700)
    started = time.monotonic()
    clock = dict(assignment=permit['assignment'], generation=permit['generation'],
                 budget_seconds=180, cpu_seconds=45, started_monotonic=started,
                 deadline_monotonic=started+180,
                 boot_id=Path('/proc/sys/kernel/random/boot_id').read_text().strip(),
                 permit_sha256=digest(permit_path), scope=permit['scope'])
    # Imports and all native loading are inside the original bundle clock.
    from existing_caller_join import (exclusive, direct_children, group_members,
                                      wait_exec_pipe)
    from owner_step_inputs import proc_stat
    from finite_kernel import subreaper
    from finite_custody import Custody, StopUnconfirmed
    from caller_wall import CallerWall
    exclusive(run/'clock.json', clock)
    exclusive(run/'permit.json', permit)
    exclusive(run/'claim.json', {'clock_sha256': digest(run/'clock.json'),
                                 'permit_sha256': digest(permit_path),
                                 'bundle': list(BUNDLE), 'consumed': True})
    assert not direct_children() and len(list(Path('/proc/self/task').iterdir())) == 1
    assert signal.getsignal(signal.SIGCHLD) == signal.SIG_DFL
    assert subreaper() == 1
    previous = cpu(resource.RUSAGE_CHILDREN)
    observer_before = cpu(resource.RUSAGE_SELF)
    wall = proc = custody = None
    gate_read = gate_write = error_read = error_write = None
    failures, events = [], []
    target = None
    result = None
    try:
        gate_read, gate_write = os.pipe2(os.O_CLOEXEC)
        error_read, error_write = os.pipe2(os.O_CLOEXEC)
        argv = [permit['interpreter'], '-I', '-B', '-S', str(here/'primitive_target.py'),
                str(gate_read)]
        wall = CallerWall(here/'finite_birth.so', clock['deadline_monotonic'])
        pid = wall.spawn(argv, here, gate_read, error_write)
        proc = SimpleNamespace(pid=pid, returncode=None)
        custody = Custody(proc, os.getpid(), proc_stat, direct_children, group_members)
        os.close(gate_read); gate_read = None
        os.close(error_write); error_write = None
        wait_exec_pipe(error_read, clock['deadline_monotonic'])
        birth = proc_stat(pid)
        assert birth['ppid'] == os.getpid() and birth['pgid'] == birth['sid'] == pid
        assert digest('/proc/'+str(pid)+'/exe') == permit['interpreter_sha256']
        observed_argv = [p.decode() for p in Path('/proc/'+str(pid)+'/cmdline').read_bytes().split(b'\0') if p]
        assert observed_argv == argv
        exclusive(run/'binding.json', {'child': birth, 'caller': proc_stat(os.getpid()),
                                      'argv': argv, 'clock_sha256': digest(run/'clock.json')})
        # R1: a native CPU/delivery/rollback latch rejects before the gate.
        assert not wall.expired and not wall.signal_error
        assert time.monotonic() < clock['deadline_monotonic']
        assert os.write(gate_write, b'B') == 1
        # This target produces only primitive observations, no original mode.
        while not (run/'target-ready-committed').exists():
            assert time.monotonic() < clock['deadline_monotonic'], 'ORIGINAL_DEADLINE'
            _, scan_errors = custody.discover()
            reap_errors = custody.reap_ready()
            assert not scan_errors and not reap_errors and custody.gnu_reap is None
            time.sleep(min(.01, max(0, clock['deadline_monotonic']-time.monotonic())))
        target = json.loads((run/'target-ready.json').read_bytes())
        assert target['pid'] == pid and target['clock_sha256'] == digest(run/'clock.json')
        assert target['checks_complete'] is True
        # Real negative helper calls; neither can create another process.
        old_fd = wall.root_fd
        try:
            wall.close_after_observation(False)
        except OSError as error:
            events.append({'control': 'live-handle-close-refusal', 'actual_errno': error.errno})
            assert error.errno == errno.EBUSY
        else:
            raise RuntimeError('LIVE_NATIVE_HANDLE_WAS_DISCARDED')
        assert wall.root_fd == old_fd and wall.pid == pid
        try:
            wall.spawn(argv, here, -1, -1)
        except OSError as error:
            events.append({'control': 'second-birth-refusal', 'actual_errno': error.errno})
            assert error.errno == errno.EINVAL
        else:
            raise RuntimeError('SECOND_NATIVE_BIRTH_WAS_NOT_REFUSED')
        assert direct_children() == [pid] and wall.pid == pid and wall.root_fd == old_fd
        # Same retained birth pidfd. No numeric PID/PGID reacquisition.
        signal.pidfd_send_signal(old_fd, signal.SIGSTOP)
        stopped = None
        while stopped is None:
            assert time.monotonic() < clock['deadline_monotonic'], 'ORIGINAL_DEADLINE'
            stopped = os.waitid(os.P_PID, pid, os.WSTOPPED | os.WNOWAIT | os.WNOHANG)
            if stopped is None:
                time.sleep(min(.01, max(0, clock['deadline_monotonic']-time.monotonic())))
        events.append({'control': 'real-stop-continue', 'pid': stopped.si_pid,
                       'si_code': stopped.si_code, 'si_status': stopped.si_status,
                       'waitid_WNOWAIT': True, 'observed_monotonic': time.monotonic()})
        assert stopped.si_pid == pid and stopped.si_code == os.CLD_STOPPED
        assert stopped.si_status == signal.SIGSTOP
        signal.pidfd_send_signal(old_fd, signal.SIGCONT)
        assert not wall.expired and not wall.signal_error
        assert os.write(gate_write, b'C') == 1
        os.close(gate_write); gate_write = None
        custody.finish(clock['deadline_monotonic'])
        retained = json.dumps(custody.evidence(), sort_keys=True)
        charged = cpu(resource.RUSAGE_CHILDREN)
        custody.finish(clock['deadline_monotonic'])
        assert json.dumps(custody.evidence(), sort_keys=True) == retained
        assert cpu(resource.RUSAGE_CHILDREN) == charged
        events.append({'control': 'terminal-wait4-reentry', 'rows_unchanged': True,
                       'children_usage_unchanged': True})
    except BaseException as error:
        failures.append({'phase': 'observation', 'error': repr(error),
                         'errno': getattr(error, 'errno', None)})
        # R2: recovery of retained PID is independent of failed pidfd signaling.
        if wall is not None:
            try:
                wall.kill_root()
            except BaseException as error:
                failures.append({'phase': 'native_signal', 'error': repr(error)})
            try:
                if proc is None and wall.pid:
                    proc = SimpleNamespace(pid=wall.pid, returncode=None)
                if proc is not None and custody is None:
                    custody = Custody(proc, os.getpid(), proc_stat, direct_children, group_members)
            except BaseException as error:
                failures.append({'phase': 'native_custody_recovery', 'error': repr(error)})
        if custody is not None:
            try:
                custody.finish(clock['deadline_monotonic'], terminate=True)
            except BaseException as error:
                failures.append({'phase': 'cleanup', 'error': repr(error)})
                # One final nonblocking harvest; never a new cleanup deadline.
                try:
                    _, scan_errors = custody.discover()
                    harvest = custody.signal_owned() + custody.reap_ready()
                    failures.extend(scan_errors + harvest)
                except BaseException as error:
                    failures.append({'phase': 'harvest', 'error': repr(error)})
    finally:
        for fd in (gate_read, gate_write, error_read, error_write):
            if fd is not None:
                try:
                    os.close(fd)
                except OSError as error:
                    failures.append({'phase': 'pipe_close', 'error': repr(error)})
        # Observation assembly/write errors cannot skip native close handling.
        try:
            rows = custody.rows if custody is not None else []
            delta = cpu(resource.RUSAGE_CHILDREN)-previous
            row_sum = sum(row['user']+row['system'] for row in rows)
            raw_valid = all(math.isfinite(v) and v >= 0 for v in (delta, row_sum))
            accounted = bool(custody and custody.closed and custody.gnu_reap
                             and not custody.anomalies and raw_valid
                             and abs(delta-row_sum) <= 2e-6)
            root = custody.gnu_reap if custody is not None else None
            if wall is not None and wall.signal_error:
                failures.append({'phase': 'native_latches', 'errors': wall.signal_error})
            within_wall = time.monotonic() < clock['deadline_monotonic']
            # This narrow bundle has no GNU/mode descendants: the actual root
            # wait4 is also its own lifetime CPU. Do not infer an endpoint
            # latency qualification merely from this ordinary positive bound.
            complete = bool(accounted and len(rows) == 1 and root['exit_code'] == 0
                            and root['user']+root['system'] <= 6 and row_sum <= 45
                            and target and within_wall and not failures)
            result = {'schema': 'friday.f6.primitive-observation.v1',
                      'state': 'OBSERVED_UNACCEPTED' if complete else 'FAILED_OR_UNKNOWN_NO_RETRY',
                      'bundle': list(BUNDLE), 'clock_sha256': digest(run/'clock.json'),
                      'source_manifest_sha256': permit['source_manifest_sha256'],
                      'native_birth_binary_sha256': permit['native_birth_binary_sha256'],
                      'rows': rows, 'root_wait4': root, 'caller_children_delta': delta,
                      'known_reaped_cpu_lower_bound': row_sum,
                      'accounted_tree_cpu': row_sum if accounted else None,
                      'kernel_accounting_complete': accounted, 'events': events,
                      'target': target, 'errors': failures, 'original_wall_met': within_wall,
                      'custody': custody.evidence() if custody is not None else None,
                      'observer_cpu_lifetime_before': observer_before,
                      'observer_cpu_lifetime_at_record': cpu(resource.RUSAGE_SELF),
                      'observer_cpu_scope': 'SEPARATE_EXISTING_CALLER_NOT_ZERO_NOT_INCLUDED_IN45',
                      'observer_write_exit_cpu': 'NOT_OBSERVED_BY_SELF',
                      'native_teardown': 'PENDING_SEPARATE_CLOSE_RECORD',
                      'independently_accepted': False, 'product_mode_admission': False,
                      'F6_complete': False, 'original11_execution': 'NOT_RUN'}
            exclusive(run/'observation.json', result)
        finally:
            if wall is not None:
                close_error = None
                try:
                    wall.close_after_observation(not wall.pid or bool(custody and custody.closed))
                except BaseException as error:
                    close_error = repr(error)
                close_at = time.monotonic()
                exclusive(run/'native-close.json', {'error': close_error,
                    'closed': close_error is None,
                    'after_close_monotonic': close_at,
                    'original_wall_met': close_at < clock['deadline_monotonic'],
                    'observation_sha256': digest(run/'observation.json') if (run/'observation.json').exists() else None,
                    'caller_exit_still_requires_external_observation': True})
                if close_error:
                    raise StopUnconfirmed(close_error)
                if close_at >= clock['deadline_monotonic']:
                    raise StopUnconfirmed('NATIVE_CLOSE_AFTER_ORIGINAL_DEADLINE_NO_RETRY')
    return 0 if result and result['state'] == 'OBSERVED_UNACCEPTED' else 1
