# Inert complete source export for analysis; not a runtime grant.
"""One real native exec target; no product/source-case import, fork or manager.

Runs only through the existing caller's native birth with one inherited root6,
AS2GiB and [0,1] envelope. Original CPU/wall timers survive until process exit.
Real invalid kernel calls exercise errno without replacing any return value.
"""
import ctypes
import errno
import json
import os
from pathlib import Path
import resource
import signal
import sys
import time


def main():
    here = Path(__file__).resolve().parent
    sys.path.insert(0, str(here))
    from finite_kernel import (arm, checked, libc, subreaper, Sigevent, Itimerspec,
                               Timespec)
    from owner_step_inputs import digest, proc_stat
    from existing_caller_join import exclusive
    run = here/'f6-primitive-run'
    clock = json.loads((run/'clock.json').read_bytes())
    permit = json.loads((run/'permit.json').read_bytes())
    assert sys.flags.isolated and sys.flags.no_site and sys.dont_write_bytecode
    assert not sys.flags.optimize and len(sys.argv) == 2
    assert clock['deadline_monotonic']-clock['started_monotonic'] == 180
    assert clock['budget_seconds'] == 180 and clock['cpu_seconds'] == 45
    assert clock['boot_id'] == Path('/proc/sys/kernel/random/boot_id').read_text().strip()
    assert digest('/proc/self/exe') == permit['interpreter_sha256']
    assert resource.getrlimit(resource.RLIMIT_CPU) == (6, 45)
    assert resource.getrlimit(resource.RLIMIT_AS) == (2 << 30, 2 << 30)
    assert sorted(os.sched_getaffinity(0)) == [0, 1]
    assert signal.getsignal(signal.SIGXCPU) == signal.SIG_DFL
    assert signal.SIGXCPU not in signal.pthread_sigmask(signal.SIG_BLOCK, set())
    # Not a post-import CPU baseline: the absolute6 includes native child setup.
    timers = [arm(time.CLOCK_PROCESS_CPUTIME_ID, 6.0),
              arm(time.CLOCK_MONOTONIC, clock['deadline_monotonic'])]
    gate = int(sys.argv[1])
    assert gate >= 3 and os.read(gate, 1) == b'B'
    binding = json.loads((run/'binding.json').read_bytes())
    identity = proc_stat(os.getpid())
    assert binding['clock_sha256'] == digest(run/'clock.json')
    for key in ('pid', 'ppid', 'pgid', 'sid', 'start_ticks'):
        assert identity[key] == binding['child'][key]
    assert os.getppid() == binding['caller']['pid']
    assert proc_stat(os.getppid())['start_ticks'] == binding['caller']['start_ticks']
    subreaper_value = subreaper()
    assert subreaper_value == 1
    observations = []
    lib = libc()  # Exact final ctypes signatures/structures, not a replacement.
    event = Sigevent(); event.signo = signal.SIGKILL; event.notify = 0
    timer = ctypes.c_void_p()
    ctypes.set_errno(0)
    result = lib.timer_create(-1, ctypes.byref(event), ctypes.byref(timer))
    error = ctypes.get_errno()
    observations.append({'operation': 'timer_create', 'clock': -1,
                         'actual_return': result, 'actual_errno': error})
    assert result == -1 and error == errno.EINVAL
    try:
        checked(result, 'timer_create:real-invalid-clock')
    except OSError as observed:
        observations[-1]['checked_exception_errno'] = observed.errno
        assert observed.errno == error
    else:
        raise RuntimeError('KERNEL_ERRNO_NOT_PROPAGATED')
    # This extra timer is disarmed and deleted; it is an invalid-input control,
    # not another enforcement timer, deadline or budget.
    ctypes.set_errno(0)
    result = lib.timer_create(time.CLOCK_MONOTONIC, ctypes.byref(event), ctypes.byref(timer))
    observations.append({'operation': 'timer_create:disarmed-control',
                         'actual_return': result, 'actual_errno': ctypes.get_errno()})
    checked(result, 'timer_create:disarmed-errno-control')
    lib.timer_delete.argtypes = [ctypes.c_void_p]
    lib.timer_delete.restype = ctypes.c_int
    try:
        invalid = Itimerspec(Timespec(0, 0), Timespec(0, 1_000_000_000))
        ctypes.set_errno(0)
        result = lib.timer_settime(timer, 1, ctypes.byref(invalid), None)
        error = ctypes.get_errno()
        observations.append({'operation': 'timer_settime', 'actual_return': result,
                             'actual_errno': error, 'invalid_tv_nsec': 1_000_000_000})
        assert result == -1 and error == errno.EINVAL
        try:
            checked(result, 'timer_settime:real-invalid-nanoseconds')
        except OSError as observed:
            observations[-1]['checked_exception_errno'] = observed.errno
            assert observed.errno == error
        else:
            raise RuntimeError('KERNEL_ERRNO_NOT_PROPAGATED')
    finally:
        ctypes.set_errno(0)
        result = lib.timer_delete(timer)
        observations.append({'operation': 'timer_delete:disarmed-control',
                             'actual_return': result, 'actual_errno': ctypes.get_errno()})
        checked(result, 'timer_delete:disarmed-errno-control')
    exclusive(run/'target-ready.json', {'pid': os.getpid(), 'identity': identity,
        'clock_sha256': digest(run/'clock.json'), 'timers': timers,
        'actual_subreaper_getter': subreaper_value, 'kernel_calls': observations,
        'cpu_lifetime_at_record': time.clock_gettime(time.CLOCK_PROCESS_CPUTIME_ID),
        'cpu_limits': list(resource.getrlimit(resource.RLIMIT_CPU)),
        'as_limits': list(resource.getrlimit(resource.RLIMIT_AS)),
        'affinity': sorted(os.sched_getaffinity(0)), 'checks_complete': True,
        'timer_delivery_qualified': False, 'accepted': False})
    # The data file is already flushed/fsynced. Presence of this zero-byte
    # O_EXCL marker cannot expose a partially written JSON record to the caller.
    ready_fd = os.open(run/'target-ready-committed', os.O_CREAT | os.O_EXCL | os.O_WRONLY, 0o600)
    os.close(ready_fd)
    # Parent observes an actual stopped child, continues it, then releases this
    # existing gate. No extra child, copied fake status or shortened timer.
    assert os.read(gate, 1) == b'C'
    os.close(gate)
    return 0


if __name__ == '__main__':
    raise SystemExit(main())
