#!/usr/bin/python3
"""Static design only; produces text/JSON and never writes a unit lookup path."""
import hashlib
import json
import os
from pathlib import Path
import resource
import signal
import tarfile

os.umask(0o077)
os.sched_setaffinity(0,[0,1])
resource.setrlimit(resource.RLIMIT_AS,(2*1024**3,2*1024**3))
resource.setrlimit(resource.RLIMIT_CPU,(15,15))
signal.alarm(30)
E=Path(__file__).parent
S=E.parent/'a0-boot-argument-repair/source'
checks=[]
def check(name,condition):
    assert condition,name
    checks.append(name)
def save(name,item):
    (E/name).write_text(json.dumps(item,ensure_ascii=False,indent=2)+'\n')
def sha(data): return hashlib.sha256(data).hexdigest()
def source(name):return (E/name).read_text()

parser=source('unitfile-load-fragment.c.txt')
mapping=source('unitfile-load-fragment-gperf.gperf.in.txt')
unit=source('upstream2595-unit.c.txt')
manager=source('upstream2595-dbus-manager.c.txt')
dropin=source('unitfile-load-dropin.c.txt')
dirs=source('unitfile-dropin.c.txt')
service=source('unitfile-service.c.txt')
check('native Unit.JobTimeoutSec parser mapping', 'Unit.JobTimeoutSec,                           config_parse_job_timeout_sec' in mapping)
branch=parser.split('int config_parse_job_timeout_sec(',1)[1].split('\nint config_parse_job_running_timeout_sec(',1)[0]
check('normal parser assigns both total and default running timeout', 'parse_sec_fix_0(rvalue, &usec)' in branch and 'if (!u->job_running_timeout_set)\n                u->job_running_timeout = usec;' in branch and 'u->job_timeout = usec;' in branch)
check('transient units intentionally accept normal dropins', "we allow drop-ins for transient units" in unit)
check('service loads main fragment and dropins', 'unit_load_fragment_and_dropin(u, true)' in service)
check('transient fragment becomes loaded before dropins', 'if (u->transient) {' in parser and 'u->load_state = UNIT_LOADED;\n                return 0;' in parser)
load=unit.split('int unit_load_fragment_and_dropin(',1)[1].split('\nvoid unit_add_to_target_deps_queue',1)[0]
check('missing main fragment does not load dropin on absent query', load.index('if (u->load_state == UNIT_STUB)')<load.index('r = unit_load_dropin(u);') and 'if (fragment_required)\n                        return -ENOENT;' in load)
check('dropins parsed through official parser', 'config_item_perf_lookup, load_fragment_gperf_lookup' in dropin)
transient=manager.split('static int transient_unit_from_message(',1)[1].split('static int transient_aux_units_from_message(',1)[0]
start=manager.split('static int method_start_transient_unit(',1)[1].split('static int method_get_job(',1)[0]
check('load/dropin dispatch precedes enqueue', 'manager_dispatch_load_queue(m);' in transient and start.index('transient_unit_from_message(')<start.index('return bus_unit_queue_job('))
check('GC only unlinks dropins from transient directory', '/* Only drop transient drop-ins */' in unit and 'if (!path_equal(u->manager->lookup_paths.transient, pp))\n                        continue;' in unit)
check('lookup path cache is material proof obligation', 'unit_find_dropin_paths(u, /* use_unit_path_cache= */ true' in dropin and 'if (!unit_path_cache || set_get(unit_path_cache, path))' in dirs)

with tarfile.open(E/'systemd_259.5-0ubuntu3.4.debian.tar.xz') as archive:
    files={m.name:archive.extractfile(m).read() for m in archive if m.isfile()}
    targets=[b'load-fragment.c',b'load-fragment-gperf.gperf.in',b'load-dropin.c',b'src/shared/dropin.c']
    changes={p:len(b) for p,b in files.items() if p.startswith('debian/patches/') and p.endswith('.patch') and any(x in b for x in targets)}
    check('distribution patches do not change selected parser/dropin files',not changes)

recipe=json.loads((S/'recipe.json').read_bytes())
content='[Unit]\nJobTimeoutSec=20s\n'
(E/'queue-inclusive-timeout.conf.inert.txt').write_text(content)
transfer=[]
for role,argv in recipe['native_start_argv'].items():
    check(role+' exactly one moved timeout property',argv.count('--property=JobTimeoutSec=20s')==1)
    without=[a for a in argv if a!='--property=JobTimeoutSec=20s']
    check(role+' every other constructor token unchanged',len(without)==len(argv)-1 and [a for a in without if a.startswith('--property=')]==[a for a in argv if a.startswith('--property=') and a!='--property=JobTimeoutSec=20s'])
    transfer.append({'role':role,'unit':recipe['prefix']+'-'+role+'.service','remove_only_token':'--property=JobTimeoutSec=20s','configuration_carrier':'exact-name normal runtime drop-in','proposed_path_text_only':'/run/user/<UID>/systemd/user/'+recipe['prefix']+'-'+role+'.service.d/90-frw011-job-timeout.conf','required_bytes_sha256':sha(content.encode()),'expected_JobTimeoutUSec_unchanged':'20s','expected_JobRunningTimeoutUSec':'20s','other_argv_tokens_preserved':True})
check('all eight roles remain represented',len(transfer)==8)
save('normal-unitfile-route.json',{
 'status':'SOURCE_SUPPORTED_DESIGN_WITH_OPEN_ADMISSION_PROOF_NOT_EXECUTABLE_CANDIDATE',

 'not_full_static_unit_conversion':'Full ordinary main unit files would change Transient=yes, expected FragmentPath, and the not-found/GC/re-registration control. That larger conversion is rejected for this minimal route.',
 'queue_bound':{'same_Unit_fields':['job_timeout','job_running_timeout'],'seconds':20,'enforced_by':'unchanged job_start_timer queue origin begin_usec and running origin begin_running_usec','before_job_enqueue':True,'runtime_proven':False},
 'transfer_matrix':transfer,
 'required_harness_delta':[
  'Remove only the single JobTimeoutSec=20s property from each of eight constructors after safe configuration admission exists; never emit an unguarded runnable version.',
  'For loaded transient units require exact singleton owned DropInPaths instead of empty; preserve Transient=yes, exact transient FragmentPath, every other property, all dependency edges and hooks. Also require JobRunningTimeoutUSec=20s as an extra readback check.',
  'Keep common.absent() unchanged: on service without a main fragment, unit_load_fragment_and_dropin returns ENOENT before parsing drop-ins. Keep actual GC/reconstruction, one-use permits, no replay and all eight names unchanged.',
  'Add bounded exact-file ownership/content admission before ANY native start, source/manifest binding to eight file bytes, and durable per-file create/cleanup receipts. Refuse preexisting foreign directories/files, symlinks, wrong metadata, wrong bytes, extra effective dropins or unknown cache/image state.',
  'Reject nonempty journal stderr before issuing an authorization receipt, including rc0 ENOMEM. Empty stdout must not be described as proof of complete historical coverage.',
 ],
 'mandatory_runtime_and_permission_gaps':[
  'No current manager mapped image/library hash was established. The G3 witness proves executable path and stable identity only; package-source facts must not be represented as exact live-image proof.',
  'unit_load_dropin uses the manager unit_path_cache.   'Install/cleanup eight exact runtime drop-in directories/files is outside the current solewrites E grant and absent from consumed G3 authority. Parent needs a separately reviewed finite ownership/effect envelope; no live command or grant is supplied.',
  'Preserve acceptance+5s admission, +35s final submission, +110s driver and +120s whole maximum, including newly authorized setup if performed in that attempt. If setup/cache validation cannot fit, fail closed; do not reset clocks, invent pre-budget execution, or extend limits.',
  'G3 consumed the sentinel name and original single-use source/authorization. Existing stopped trial cannot be retried or renamed by this design.   'Retain runtime drop-ins through transient GC and the deliberate outer-bound reconstruction. Source GC removes only drop-ins under the transient lookup directory, so normal user runtime drop-ins should persist; exact-image behavior needs later independent proof.',
  'After verified cessation, remove only created file/directory identities whose hashes and ownership match the recorded set. Unknown actors/jobs/cleanup keeps STOP_UNCONFIRMED and FD9 retention; no wildcard, manager stop/kill, or silent global reload is authorized.',
  'Independent reviewer must accept the complete final implementation and resource/timing envelope before a concrete execution proposal; source-only checks are not the native gate.',
 ],
 'preserved_original_anchors_seconds':[5,35,110,120],
 'all_native_product_live_gates':'NOT_GRANTED_NOT_RUN',
 'files_installed':0,'manager_calls':0,'new_grants':0,'candidate_ready_for_execution':False,
 'static_checks':checks,'static_check_count':len(checks),
 'resource_peak_rss_kib':resource.getrusage(resource.RUSAGE_SELF).ru_maxrss,
})
print(json.dumps({'normal_unitfile_route':'SUPPORTED_BY_SOURCE_WITH_OPEN_ADMISSION_PROOF','static_checks':len(checks)}))
