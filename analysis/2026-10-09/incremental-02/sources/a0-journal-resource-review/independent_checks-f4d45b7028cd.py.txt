import ast,copy,hashlib,json,os,pathlib,resource,signal,stat,sys,time,types
W=pathlib.Path('<ASTRA_WORK>')
P=W/'a0-journal-resource-repair'; E=W/'a0-journal-resource-review'
os.umask(0o077)
additional=[]
def independent_checks(v):
    common=v['common']; fixture=v['fixture']; native=v['Process']; warning=v['warning']; case_root=v['root']
    def check(name,fn):
        started=time.monotonic()
        try: detail=fn()
        except BaseException as ex:
            additional.append({'name':name,'status':'FAIL','error':type(ex).__name__+': '+str(ex)})
            (E/'independent-additional-checks.json').write_text(json.dumps(additional,indent=2)+'\n')
            raise
        additional.append({'name':name,'status':'PASS','detail':detail,'elapsed_seconds':time.monotonic()-started})
    def malformed(kind):
        f=fixture(setup=False); original_capture=common._bounded_capture; seen=[]; returned=[]
        f.owner.native_observer=seen.append
        def altered(*args,**kw):
            real=original_capture(*args,**kw)
            if kind=='null': bad=None
            elif kind=='list': bad=[]
            elif kind.startswith('missing_'):
                bad=dict(real); del bad[kind[8:]]
            elif kind=='returncode_mismatch': bad={**real,'returncode':3}
            elif kind=='unrelated_warning': bad={**real,'stderr':'unexpected diagnostic\n'}
            else: raise AssertionError(kind)
            returned.append(bad);return bad
        common._bounded_capture=altered
        failed=False
        try:
            try: f.owner.call(common.history_argv(f.owner.original),.75)
            except (common.Refused,TypeError,KeyError):failed=True
        finally:common._bounded_capture=original_capture
        assert failed and len(returned)==1 and not seen
        r=f.owner.records[0]
        assert r['capture'] is returned[0] and r['uncertain'] and f.owner.uncertain
        assert r['reaped'] and r['process'].returncode==0 and not f.owner.busy and f.owner.current is None
        assert f.owner.inventory()[0]['capture'] is returned[0]
        try:f.owner.confirm_closed()
        except common.Refused:pass
        else:raise AssertionError('malformed admitted for cessation')
        return {'capture_retained':True,'owned_reaped':True,'uncertain':True,'observer_calls':0,'cessation_accepted':False}
    for kind in ('null','list','missing_returncode','missing_timed_out','missing_output_truncated','missing_stderr','returncode_mismatch','unrelated_warning'):
        check('malformed_or_refused_capture_'+kind,lambda kind=kind:malformed(kind))
    def capture_raises_after_custody():
        f=fixture(setup=False); original_capture=common._bounded_capture
        def fail(*args,**kw):
            original_capture(*args,**kw)
            raise ValueError('capture failed after owned process reaped')
        common._bounded_capture=fail
        try:
            try:f.owner.call(common.history_argv(f.owner.original),.75)
            except ValueError:pass
            else:raise AssertionError('exception lost')
        finally:common._bounded_capture=original_capture
        r=f.owner.records[0]
        assert 'capture' not in r and r['reaped'] and f.owner.uncertain and r['uncertain']
        return {'no_fabricated_capture':True,'owned_reaped':True,'uncertain':True}
    check('capture_exception_after_custody_preserves_truth',capture_raises_after_custody)
    def disk_failure(kind):
        f=fixture(setup=False); audit=case_root/('independent-'+kind);audit.mkdir(mode=0o700)
        saves=[];calls=[];fsync_calls=[];clock_calls=[]
        class DiagnosticOS:
            def __getattr__(self,k):return getattr(os,k)
            def getpid(self):return 777
            def urandom(self,n):assert n==32;return b'U'*32
            def fsync(self,fd):
                fsync_calls.append(fd)
                if kind=='all_fsync_fail' or (kind=='directory_fsync_fail_once' and len(fsync_calls)==2):
                    raise OSError('synthetic storage failure')
                return os.fsync(fd)
        def readclock(which):
            clock_calls.append(which)
            if kind=='original_clock_failure' and which=='boottime_ns':raise OSError('synthetic clock read failure')
            return f.clock()[which]
        ns={'os':DiagnosticOS(),'json':json,'hashlib':hashlib,'Path':v['VPath'],'E':audit,
            'anchor':None,'parent':None,'attempt_token':None,'boot':None,'observations':[],
            'original_attempt_ref':None,'__transport':None,'__witness':None,'__dropins':None,'__released':False,
            'manifest_sha256':'6'*64,'OWNER_NATIVE_GRANT':{'assignment':'FRW011-A0-DEPENDENCY-DROPIN-G4','independent_review_sha256':'1'*64},
            'recipe':json.loads((P/'candidate/recipe.json.inert.txt').read_text()),'ticks':lambda _:88,
            'time':types.SimpleNamespace(monotonic_ns=lambda:readclock('monotonic_ns'),CLOCK_BOOTTIME=7,
                clock_gettime_ns=lambda _:readclock('boottime_ns'),time_ns=lambda:readclock('realtime_ns')),
            'signal':types.SimpleNamespace(SIGALRM=14,SIGTERM=15,ITIMER_REAL=0,signal=lambda *a:None,setitimer=lambda *a:None),
            'remaining':common.remaining}
        exec(compile(ast.Module(body=[v['assignment']],type_ignores=[]),'resource declaration','exec'),ns)
        exec(compile(v['helpers'],'diagnostics helpers','exec'),ns)
        production_save=ns['save']
        def save(path,value):saves.append((path.name,copy.deepcopy(value)));return production_save(path,value)
        ns['save']=save
        def forbidden(*args,**kw):calls.append('effect');raise AssertionError('unexpected native/transport effect')
        ns['ParentTransport']=forbidden;ns['install_parent_transport']=forbidden;ns['bounded_capture_call']=forbidden
        caught=None
        try:exec(compile(v['parent_fragment'],'exact early prefix/refusal/finally','exec'),ns)
        except OSError as ex:caught=ex
        assert caught and not calls and not f.children and not ns['__released']
        assert saves[-1][0]=='parent-refusal.json' and saves[-1][1]['state']=='STOP_UNCONFIRMED'
        assert saves[-1][1]['original_attempt_ref'] is None
        assert not saves[-1][1]['transport_inventory'] and not saves[-1][1]['original_attempt']['retry_authorized']
        assert saves[-1][1]['canonical_fd9']=='STILL_HELD'
        if kind=='original_clock_failure':
            assert saves[-1][1]['original_attempt']['original_anchor'] is None
            assert clock_calls==['monotonic_ns','boottime_ns']
        else:
            assert clock_calls==['monotonic_ns','boottime_ns','realtime_ns']
            assert saves[0][1]==saves[-1][1]['original_attempt']
        # A failed directory fsync in the unchanged save helper leaves its fd
        # open for original-parent custody. Close only this test's private fd.
        closed_fixture_fds=[]
        for fd in fsync_calls:
            try:
                target=os.readlink('/proc/self/fd/'+str(fd))
                if target==str(audit):os.close(fd);closed_fixture_fds.append(fd)
            except OSError:pass
        return {'native_calls':0,'clock_calls':clock_calls,'save_attempts':[n for n,_ in saves],
                'admission':False,'retry':False,'cessation':False,'storage_durability_claimed':False,
                'closed_only_private_fixture_fds':closed_fixture_fds}
    for kind in ('all_fsync_fail','directory_fsync_fail_once','original_clock_failure'):
        check('early_failure_'+kind,lambda kind=kind:disk_failure(kind))
    def diagnostic_copies():
        ns={'ORDINARY_AS_BYTES':1024**3}
        node=next(n for n in v['parent_tree'].body if isinstance(n,ast.FunctionDef) and n.name=='attempt_diagnostics')
        exec(compile(ast.Module(body=[node],type_ignores=[]),'diagnostic function','exec'),ns)
        anchor={'accepted_monotonic_ns':1,'accepted_boottime_ns':2,'accepted_realtime_ns':3};parent={'pid':777,'starttime_ticks':88,'fd':9}
        result=ns['attempt_diagnostics'](anchor,parent,'5'*64,'boot','6'*64,'1'*64)
        anchor['accepted_monotonic_ns']=999;parent['pid']=999
        assert result['original_anchor']['accepted_monotonic_ns']==1 and result['original_parent']['pid']==777
        assert result['retry_authorized'] is False and result['status']=='ORIGINAL_ATTEMPT_DIAGNOSTICS_ONLY_NOT_ADMISSION'
        return {'original_values_copied':True,'grant_environment_privileged_argv_dumped':False}
    check('diagnostics_are_original_copies_not_authority',diagnostic_copies)
    (E/'independent-additional-checks.json').write_text(json.dumps({'status':'PASS_GUARDED_FUNCTION_CHECKS_NOT_NATIVE','count':len(additional),'cases':additional,'native_acceptance':False},indent=2)+'\n')

def main():
    source=P/'check_changed_functions.py.inert.txt';tree=ast.parse(source.read_text())
    fn=next(n for n in tree.body if isinstance(n,ast.FunctionDef) and n.name=='run_checks')
    # Only helper output location and read-only candidate root change. Candidate
    # AST, checks, fake process/native boundaries and diagnostics stay exact.
    for n in fn.body:
        if isinstance(n,ast.Assign) and any(isinstance(t,ast.Name) and t.id=='E' for t in n.targets):
            n.value=ast.parse('Path('+repr(str(E))+')',mode='eval').body
        if isinstance(n,ast.Assign) and any(isinstance(t,ast.Name) and t.id=='C' for t in n.targets):
            n.value=ast.parse('Path('+repr(str(P/'candidate'))+')',mode='eval').body
    insertion=next(i for i,n in enumerate(fn.body) if isinstance(n,ast.Expr) and ast.unparse(n)=='signal.alarm(0)')
    fn.body.insert(insertion,ast.parse('_independent_checks(locals())').body[0])
    module=ast.fix_missing_locations(ast.Module(body=[fn],type_ignores=[]))
    (E/'review-adapted-helper.py.inert.txt').write_text("raise SystemExit('INERT_REVIEW_FUNCTIONS_ONLY')\n"+ast.unparse(module)+'\n')
    ns={'_independent_checks':independent_checks};exec(compile(module,str(source),'exec'),ns)
    result=ns['run_checks'](); print(json.dumps({'replayed_changed_controls':result,'additional_count':len(additional)}))
if __name__=='__main__': main()
