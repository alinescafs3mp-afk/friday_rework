"""Future finite source/data controls; NOT_RUN by the Sol source-only TASK.

All rows below are explicitly SYNTHETIC_SOURCE_MODELS, one per catalogued
definition, without evaluated parameter values. They are not collection
receipts or observed counts. The canonical four-JSON join is replaced ONLY
inside these inner-check models. No publication, clock creation, collection,
corpus import, runtest, observer, native process or service is invoked.
Root must review/admit this controls file separately before running it.
"""
import ast
import copy
import hashlib
import json
from pathlib import Path
import runpy
import unittest
from unittest.mock import patch

HERE = Path(__file__).resolve().parent
PINS = json.loads((HERE / 'source-pins.json').read_bytes())
INSPECTOR = HERE / 'inspect_native.py'
if hashlib.sha256(INSPECTOR.read_bytes()).hexdigest() != PINS['inspect_native.py']:
    raise RuntimeError('SOURCE_DRIFT')
NS = runpy.run_path(str(INSPECTOR), run_name='future_source_model_controls')
MAP = json.loads(NS['MAP'].read_bytes())


def ref(file, definition=None):
    value = {k: file[k] for k in ('path', 'sha256')}
    if definition:
        value.update({k: definition[k] for k in ('qualified_name', 'line', 'end_line')})
        if 'id' in definition:
            value['definition_id'] = definition['id']
    return value


def source_model():
    """Synthetic inner-guard exercise. Never passed to publish/canonical join."""
    modules = []; kept = []; dropped = []
    rules = MAP['dynamic_rules']
    special = {v['definition_id']: v for v in MAP['special_cases']}
    definitions = {v['id']: (f, v) for f in MAP['files'] for v in f['definitions']}
    for file in MAP['files']:
        if file['path'] not in MAP['selection']['native']:
            continue
        module_identity = {'path': file['path'], 'kind': 'module', 'nodeid': file['path'],
                           'type': '_pytest.python.Module'}
        children = []
        for declaration in file['definitions']:
            if declaration['kind'] != 'test_definition':
                continue
            name = declaration['qualified_name'].split('.')[-1]
            class_names = declaration['qualified_name'].split('.')[:-1]
            class_refs = [ref(file, next(c for c in file['classes'] if c['qualified_name'] ==
                                        '.'.join(class_names[:i + 1]))) for i in range(len(class_names))]
            unittest_class = bool(class_refs) and next(c for c in file['classes'] if
                c['qualified_name'] == '.'.join(class_names))['bases'] == ['unittest.TestCase']
            parents = [{'path': '/SYNTHETIC_SESSION_NOT_ACTUAL', 'kind': 'session',
                        'nodeid': '', 'type': '_pytest.main.Session'}, copy.deepcopy(module_identity)]
            for i, cls in enumerate(class_names):
                parents.append({'path': file['path'], 'kind': 'class',
                                'nodeid': file['path'] + '::' + '::'.join(class_names[:i + 1]),
                                'type': '_pytest.unittest.UnitTestCase' if unittest_class else '_pytest.python.Class'})
            # Parameter modes only, copied from reviewed source declarations.
            direct = declaration.get('direct_parametrize_argument_candidates', [])
            indirect = declaration.get('indirect_parametrize_argument_candidates', [])
            nodeid = parents[-1]['nodeid'] + '::' + name + ('[SYNTHETIC_OPAQUE_ID]' if direct or indirect else '')
            identity = {'path': file['path'], 'kind': 'function', 'nodeid': nodeid,
                        'type': '_pytest.unittest.TestCaseFunction' if unittest_class else '_pytest.python.Function'}
            children.append(identity)
            lookup = MAP['fixture_candidate_sets'][file['fixture_candidates_by_test'][declaration['qualified_name']]]
            fixture_origins = {'request': [rules['intrinsic_request']['origin']]}
            fixture_basis = {'request': 'intrinsic_request'}
            for fixture in direct:
                fixture_origins[fixture] = [ref(file, declaration), rules['direct_parameter']['origin']]
                fixture_basis[fixture] = 'direct_parameter_from_exact_decorator'
            # Exercise one real, exact source fixture candidate per definition;
            # no claim that it is an actually resolved fixturedef is made.
            for fixture, candidates in lookup.items():
                if (len(candidates) == 1 and fixture not in fixture_origins
                        and candidates[0] in definitions and definitions[candidates[0]][1]['kind'] == 'fixture'):
                    ff, fd = definitions[candidates[0]]
                    fixture_origins[fixture] = [ref(ff, fd)]
                    fixture_basis[fixture] = 'exact_source_fixture_lookup_candidate'
                    break
            if unittest_class:
                fixture = '_unittest_setUpClass_fixture_' + '.'.join(class_names)
                fixture_origins[fixture] = [rules['unittest_names']['origin'], class_refs[-1]]
                fixture_basis[fixture] = 'pinned_unittest_factory_and_direct_TestCase_base'
            platforms = []; markers = []
            for decorator in declaration.get('decorators', []):
                node = ast.parse(decorator, mode='eval').body
                fn = node.func if isinstance(node, ast.Call) else node
                text = ast.unparse(fn)
                if text.startswith('pytest.mark.'):
                    markers.append(text.rsplit('.', 1)[-1])
                if text == 'pytest.mark.platforms':
                    platforms.append([ast.literal_eval(v) for v in node.args])
            windows = any(set(v) == {'windows'} for v in platforms)
            if windows:
                markers.append('skip')
            metadata = {'definition': file['path'], 'path': file['path'], 'nodeid': nodeid,
                        'identity': identity, 'parents': parents, 'fixtures': list(fixture_origins),
                        'platforms': platforms, 'markers': markers,
                        'skip_reasons': ['Native Windows-only case: actual Linux host; NOT_APPLICABLE, not safety PASS'] if windows else []}
            scope = special.get(declaration['id'], {}).get('scope', 'COLLECTED_ONLY_ALL_RUNTESTS_NOT_RUN')
            row = {'metadata': metadata, 'definition_basis': ref(file, declaration),
                   'originalname_source_basis': name, 'class_lineage_basis': class_refs,
                   'decorators_source_basis': declaration.get('decorators', []),
                   'direct_parameters': direct, 'indirect_parameters': indirect,
                   'fixture_origins': fixture_origins, 'fixture_basis': fixture_basis,
                   'scope_basis': scope, 'disposition': 'LINUX_NOT_APPLICABLE' if windows else 'COLLECTED_ONLY_NOT_RUN'}
            if scope in ('STOP_REQUIRED_NOT_RUN_NO_REROUTE',
                         'REAL_LOOPBACK_HTTP_AND_OPTIONAL_GIT_PROCESS_FIXTURE_REQUIRED_NOT_RUN'):
                metadata['originalname'] = name
                metadata['reason'] = ('NOT_RUN: actual subprocess outside nochildren task boundary; native process lock second-open proof remains required.'
                                      if scope == 'STOP_REQUIRED_NOT_RUN_NO_REROUTE' else
                                      'NOT_RUN: actual local HTTP server/git subprocess fixture exceeds task no network/nochildren; still required in separately authorized admission')
                if scope.startswith('REAL_LOOPBACK'):
                    fixture = 'served_repo_missing_support' if name == 'test_install_skips_unreachable_support_file_e2e' else 'served_repo'
                    ff, fd = definitions[lookup[fixture][0]]
                    fixture_origins[fixture] = [ref(ff, fd)]
                    fixture_basis[fixture] = 'exact_source_fixture_lookup_candidate'
                    metadata['fixtures'] = list(fixture_origins)
                row['disposition'] = 'NOT_RUN'; dropped.append(row)
            else:
                kept.append(row)
        receipt = {'identity': module_identity, 'children': children, 'outcome': 'passed', 'detail': None}
        modules.append({'path': file['path'], 'source_basis': ref(file), 'receipt': receipt,
                        'module_declarations': file['module_declarations'], 'disposition': 'COLLECTED_ONLY_NOT_RUN'})
    by_path = {v['path']: v for v in modules}
    return {'state': 'DATA_JOIN_RESOLVED_REQUIRES_OWNER_INSPECTION', 'provenance_sha256': NS['MAP_SHA'],
            'unresolved': [], 'all_runtests': 'NOT_RUN', 'acceptance': False,
            'selected_modules': [by_path[p] for p in MAP['selection']['native']],
            'kept': kept, 'deselected_reviews': dropped, 'deselected': [v['metadata'] for v in dropped]}


class NativeInspectionSourceModels(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        cls.model = source_model()

    def check_model(self, basis):
        # This bypass is scoped to inner source checks, never the publication CLI.
        function = NS['inspect_native']
        with patch.dict(function.__globals__, canonical_basis=lambda b, p: copy.deepcopy(b)):
            return function(basis, [])

    def test_positive_source_model_preserves_not_run_and_windows(self):
        checked = self.check_model(copy.deepcopy(self.model))
        self.assertFalse(checked['acceptance'])
        self.assertEqual(checked['all_runtests'], 'NOT_RUN')
        self.assertTrue(any(v['disposition'] == 'LINUX_NOT_APPLICABLE' for v in checked['kept']))
        self.assertEqual(len(checked['deselected_reviews']), 8)
        self.assertTrue(all(v['definition_inspected'] and v['reason_inspected'] for v in checked['deselected_reviews']))

    def test_parameter_names_indirect_are_not_evaluated(self):
        node = ast.parse("@pytest.mark.parametrize('a,b', EXPRESSION_MUST_NOT_RUN, indirect=['b'])\ndef case(a,b): pass").body[0]
        self.assertEqual(NS['parameters'](node), (['a'], ['b']))

    def test_duplicate_json_keys_refused(self):
        with self.assertRaisesRegex(ValueError, 'DUPLICATE_JSON_KEY'):
            NS['read_json']('{"identity":1,"identity":2}')

    def test_negative_inner_checks(self):
        # Every model starts from the same complete source-definition model.
        # No generated row is saved as an actual collection/review artifact.
        mutations = [
            ('module_order', 'FOREIGN_NATIVE_MODULE', lambda b: b['selected_modules'].reverse()),
            ('foreign_module', 'FOREIGN_NATIVE_MODULE', lambda b: b['selected_modules'][0].update(path='/FOREIGN')),
            ('skipped_module', 'UNEXPLAINED_NATIVE_MODULE_SKIP', lambda b: b['selected_modules'][0]['receipt'].update(outcome='skipped', detail='unexplained model skip')),
            ('empty_module_relabel', 'MODULE_DISPOSITION_MISMATCH', lambda b: b['selected_modules'][0]['receipt'].update(children=[])),
            ('module_source', 'UNPINNED_SOURCE_REF', lambda b: b['selected_modules'][0]['source_basis'].update(sha256='0' * 64)),
            ('definition_line', 'DEFINITION_IDENTITY_MISMATCH', lambda b: b['kept'][0]['definition_basis'].update(line=1)),
            ('definition_name', 'DEFINITION_IDENTITY_MISMATCH', lambda b: b['kept'][0]['definition_basis'].update(qualified_name='foreign')),
            ('definition_path_alias', 'WRONG_NATIVE_DEFINITION', lambda b: b['kept'][0]['metadata'].update(definition='/FOREIGN_ALIAS')),
            ('original_name', 'WRONG_NATIVE_DEFINITION', lambda b: b['kept'][0].update(originalname_source_basis='foreign')),
            ('decorator', 'DECORATOR_DRIFT', lambda b: b['kept'][0].update(decorators_source_basis=['pytest.mark.skip'])),
            ('parent_path', 'PARENT_CLASS_MISMATCH', lambda b: b['kept'][0]['metadata']['parents'][-1].update(path='/FOREIGN')),
            ('nodeid', 'ACTUAL_NODEID_DEFINITION', lambda b: b['kept'][0]['metadata'].update(nodeid='FOREIGN')),
            ('parameter_mode', 'PARAMETER_DECORATOR', lambda b: b['kept'][0].update(indirect_parameters=['foreign'])),
            ('platform', 'PLATFORM_SOURCE', lambda b: b['kept'][0]['metadata'].update(platforms=[['windows']])),
            ('skip_reason', 'PLATFORM_SKIP', lambda b: b['kept'][0]['metadata'].update(skip_reasons=['invented'])),
            ('xfail', 'UNEXPLAINED_SKIP_OR_XFAIL', lambda b: b['kept'][0]['metadata']['markers'].append('xfail')),
            ('fixture_gap', 'FIXTURE_COVERAGE', lambda b: b['kept'][0]['fixture_origins'].pop('request')),
            ('intrinsic_ref', 'WRONG_INTRINSIC_REQUEST', lambda b: b['kept'][0]['fixture_origins'].update(request=[b['kept'][0]['definition_basis']])),
            ('unknown_fixture_mode', 'UNREVIEWED_DYNAMIC_FIXTURE', lambda b: b['kept'][0]['fixture_basis'].update(request='made_up_factory')),
            ('stop_reason', 'STOP_INTENT_RELABELED', lambda b: next(v for v in b['deselected_reviews'] if v['scope_basis'].startswith('STOP'))['metadata'].update(reason='PASS')),
            ('HTTP_reason', 'HTTP_GIT_EXCLUSION', lambda b: next(v for v in b['deselected_reviews'] if v['scope_basis'].startswith('REAL_LOOPBACK'))['metadata'].update(reason='PASS')),
            ('deselection_order', 'DESELECTION_ORDER', lambda b: b['deselected'].reverse()),
            ('missing_definition', 'NATIVE_DEFINITION_COVERAGE', lambda b: b['kept'].pop(0)),
            ('kept_STOP', 'PLATFORM_SKIP|TWO_STOP|NATIVE_DEFINITION|DESELECTION', lambda b: b['kept'].append(b['deselected_reviews'].pop(0))),
            ('inline_bash_scope', 'INLINE_BASH_COLLECTION_COVERAGE', lambda b: next(v for v in b['kept'] if v['scope_basis'].startswith('REAL_INLINE')) .update(scope_basis='COLLECTED_ONLY_ALL_RUNTESTS_NOT_RUN')),
        ]
        for name, message, change in mutations:
            with self.subTest(name=name):
                basis = copy.deepcopy(self.model); change(basis)
                with self.assertRaisesRegex((ValueError, KeyError), message):
                    self.check_model(basis)

    def test_unittest_actual_type_and_factory_name_refused_when_changed(self):
        for alteration in ('type', 'name'):
            with self.subTest(alteration=alteration):
                basis = copy.deepcopy(self.model)
                row = next(v for v in basis['kept'] if any(mode == 'pinned_unittest_factory_and_direct_TestCase_base'
                                                          for mode in v['fixture_basis'].values()))
                if alteration == 'type':
                    row['metadata']['identity']['type'] = '_pytest.python.Function'
                else:
                    old = next(k for k, mode in row['fixture_basis'].items()
                               if mode == 'pinned_unittest_factory_and_direct_TestCase_base')
                    row['fixture_origins'][old + '_foreign'] = row['fixture_origins'].pop(old)
                    row['fixture_basis'][old + '_foreign'] = row['fixture_basis'].pop(old)
                    row['metadata']['fixtures'] = list(row['fixture_origins'])
                with self.assertRaisesRegex(ValueError, 'UNITTEST_CLASS_OR_FIXTURE_IDENTITY'):
                    self.check_model(basis)


if __name__ == '__main__':
    unittest.main()
