def extract(path, names, namespace):
    tree = ast.parse(path.read_text())
    body = [n for n in tree.body if isinstance(n, (ast.FunctionDef, ast.ClassDef)) and n.name in names]
    assert {n.name for n in body} == set(names)
    exec(compile(ast.Module(body=body, type_ignores=[]), str(path), 'exec'), namespace)
    return namespace

def lexical_result(f, selected=MODULES):
    result = copy.deepcopy(f.result)
    for module in selected:
        row = f.rows[module]; result['inputs'][row['raw']] = row['sha256']; result['origins'][module] = row['raw']
    return result


def join(f, result=None, selection='@native'):
    return f.owner['observed_input_joins'](f.result if result is None else result, selection, f.expanded)


@pytest.mark.parametrize('selection', ['@native', '@project', '@canary'])
def test_canonical_positive_unchanged(fixture, selection):
    f = fixture; before = copy.deepcopy((f.result, f.expanded))
    assert join(f, selection=selection) == {}
    assert (f.result, f.expanded) == before


@pytest.mark.parametrize('selected', [MODULES, MODULES[:1]])
def test_exact_four_or_subset_positive_preserves_raw_and_canonical(fixture, selected):
    f = fixture; result = lexical_result(f, selected); before = copy.deepcopy((result, f.expanded))
    joined = join(f, result)
    assert joined == {f.rows[m]['raw']: {'module': m, 'canonical': f.rows[m]['canonical'], 'sha256': f.rows[m]['sha256']} for m in selected}
    assert (result, f.expanded) == before
    assert all(f.rows[m]['raw'] not in f.expanded for m in MODULES)

@pytest.mark.parametrize('kind', ['same_target_symlink', 'dangling_symlink', 'directory', 'fifo', 'replacement_same_bytes', 'hash_change', 'missing'])
def test_leaf_substitution_refuses_before_unsafe_content_read(fixture, kind):
    f = fixture; row = f.rows[MODULES[0]]; p = Path(row['canonical']); old = p.with_suffix('.original'); p.rename(old)
    if kind == 'same_target_symlink': p.symlink_to(old)
    elif kind == 'dangling_symlink': p.symlink_to(p.with_suffix('.missing'))
    elif kind == 'directory': p.mkdir()
    elif kind == 'fifo': os.mkfifo(p)
    elif kind == 'replacement_same_bytes': p.write_bytes(old.read_bytes())
    elif kind == 'hash_change':
        old.rename(p); p.write_text('# changed original inode\n')
    opened = []; actual_open = f.os.open
    def tracked_open(path, flags, **kwargs):
        opened.append(path); return actual_open(path, flags, **kwargs)

        assert not any(p.startswith('/proc/self/fd/') for p in opened)

@pytest.mark.parametrize('component', ['native', 'tests', 'tools', 'leaf'])
def test_post_hash_binding_replacement_refuses_and_closes_all(fixture, component):
    f = fixture; row = f.rows[MODULES[0]]; opened = []; closed = []; real_open = f.os.open; real_close = f.os.close; real_read = f.os.read
    def opened_fd(*args, **kwargs):
        fd = real_open(*args, **kwargs); opened.append(fd); return fd
    def closed_fd(fd):
        closed.append(fd); real_close(fd)
    mutated = False
    def mutate_on_eof(fd, count):
        nonlocal mutated
        chunk = real_read(fd, count)
        if not chunk and not mutated:
            mutated = True
            p = Path(row['canonical']) if component == 'leaf' else f.native if component == 'native' else f.native / ('tests' if component == 'tests' else 'tests/tools')
            old = p.with_name(p.name + '-former'); p.rename(old)
            if component == 'leaf': p.write_bytes(old.read_bytes())
            else: p.mkdir()
        return chunk
    f.os.open = opened_fd; f.os.close = closed_fd; f.os.read = mutate_on_eof
    with pytest.raises(ValueError, match='LEXICAL_COMPONENT_BINDING_DRIFT|LEXICAL_LEAF_BINDING_DRIFT'):
        f.owner['verify_lexical_input'](row, f.api, f.manifest)
    assert mutated and closed == list(reversed(opened))
    for fd in opened:
        with pytest.raises(OSError): os.fstat(fd)
