"""Future explicitly admitted native negative-control source; NOT executed.
Call inside the same reviewed unit with its original owner staying alive.
The ordinary root stop must close the whole cgroup, including this setsid child.
No standalone main, daemon, outside-unit dispatch or default build invocation.
"""
import math
import os
from pathlib import Path
import signal
import time


def create_owned_detached_control(clock, typed_root_observation):
    now = time.monotonic()
    def require(value, why):
        if not value:raise RuntimeError(why)
    require(os.getuid() == os.geteuid() == 1000, 'rootless_native_control_required')
    require(clock.get('native_detached_control_authorized_by') == 'Astra', 'separate_native_control_grant_required')
    require(type(clock.get('deadline_monotonic')) in (int, float)
            and math.isfinite(clock['deadline_monotonic']) and now + 60 < clock['deadline_monotonic'], 'original_control_budget')
    require(clock['boot_id'] == Path('/proc/sys/kernel/random/boot_id').read_text().strip(), 'native_control_boot')
    ns = os.readlink('/proc/self/ns/pid')
    require(typed_root_observation['namespaces']['pid'] == ns
            and typed_root_observation['unit'] == clock['unit']
            and typed_root_observation['clock_sha256'] == clock['sha256'], 'exact_observed_unit_control')
    child = os.fork()
    if child == 0:
        # A real detached process, not a mocked status or a literal child count.
        os.setsid()
        for entry in Path('/proc/self/fd').iterdir():
            try:
                fd = int(entry.name)
                if fd > 2:
                    os.close(fd)
            except OSError:
                pass
        signal.alarm(60)  # Supplemental fail-closed ceiling, never stop acceptance.
        signal.pause()
        os._exit(91)
    birth = int(Path('/proc', str(child), 'stat').read_text().rsplit(')', 1)[1].split()[19])
    return dict(pid=child, start_ticks=birth, pid_namespace=ns,
                namespace_scope='same_existing_private_unit',
                original_deadline_monotonic=clock['deadline_monotonic'],
                natural_alarm_seconds=60, native_stop_required=True,
                acceptance='root observes exact host-mapped child birth alive before stop; actual fixed stop before60s, cgroup recursively empty, original births absent and original parent reaped; alarm or namespace-init exit alone cannot PASS')
