def digest(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()


def js(p):
    return json.loads(p.read_bytes())


def pin(p):
    s = p.lstat()
    assert stat.S_ISREG(s.st_mode) and s.st_nlink == 1 and p.resolve() == p
    return dict(path=str(p), sha256=digest(p), bytes=s.st_size, mode=oct(stat.S_IMODE(s.st_mode)))

# Exact byte delta without duplicating 31 MB of unchanged input-map text.
before, after = (B / 'inputs.json').read_bytes(), (C / 'inputs.json').read_bytes()
offset = 0
while before[offset:offset+32768] == after[offset:offset+32768]:
    assert offset < len(before)
    offset += 32768
while offset < min(len(before), len(after)) and before[offset] == after[offset]:
    offset += 1
suffix = 0
while suffix < len(before)-offset and before[-suffix-1] == after[-suffix-1]:
    suffix += 1
deleted = before[offset:len(before)-suffix]
inserted = after[offset:len(after)-suffix]
assert before[:offset] + inserted + before[offset+len(deleted):] == after
delta = dict(format='single_exact_utf8_byte_splice', original=pin(B / 'inputs.json'), final=pin(C / 'inputs.json'),
             offset=offset, delete_utf8=deleted.decode(), insert_utf8=inserted.decode(),
             preserved_entries=101301, added={str(config): graph[str(config)]}, reconstruction_verified=True)
(E / 'inputs.delta.json').write_text(json.dumps(delta, indent=2) + '\n')

def case_count(function):
    count = 1
    for decorator in function.decorator_list:
        if isinstance(decorator, ast.Call) and isinstance(decorator.func, ast.Attribute) and decorator.func.attr == 'parametrize':
            assert isinstance(decorator.args[1], (ast.List, ast.Tuple))
            count *= len(decorator.args[1].elts)
    return count
