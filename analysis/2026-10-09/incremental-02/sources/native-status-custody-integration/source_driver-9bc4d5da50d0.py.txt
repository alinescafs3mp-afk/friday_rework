from pathlib import Path
import hashlib
import json
import os
import stat
import sys

PACKAGE = Path(__file__).resolve().parents[1]
CALLER = PACKAGE / 'candidate/caller'
SUITE = [
    'producer/test_custody_repair_source.py',
    'producer/test_runtest_protocol_source.py',
    'producer/test_startup_temp_source.py',
    'producer/test_native_boundary_source.py',
    'producer/test_cancellation_pidfd_join_source.py',

    'producer/test_host_control_join_source.py',
    'controls/test_pidfd_source_faults.py',
    'producer/test_native_status_source.py',
]


def main():
    if len(sys.argv) != 3 or sys.argv[1] != '--owned-run':
        raise SystemExit('EXACT_EXISTING_WRAPPER_OWNED_RUN_REQUIRED')
    owned = Path(sys.argv[2])
    if not owned.is_absolute() or owned.resolve(strict=True) != owned or owned.is_symlink():
        raise SystemExit('EXACT_PRIVATE_OWNED_RUN_REQUIRED')
    meta = owned.stat()
    if not stat.S_ISDIR(meta.st_mode) or stat.S_IMODE(meta.st_mode) != 0o700 or meta.st_uid != os.getuid():
        raise SystemExit('FOREIGN_SOURCE_OUTPUT_DIRECTORY')
    pins = json.loads((PACKAGE / 'candidate-source-pins.json').read_text())
    for relative, expected in pins.items():
        path = PACKAGE / 'candidate' / relative
        if path.resolve(strict=True) != path or path.is_symlink() or hashlib.sha256(path.read_bytes()).hexdigest() != expected:
            raise SystemExit('SOURCE_DRIFT:' + relative)
    if os.environ.get('PYTEST_ADDOPTS'):
        raise SystemExit('AMBIENT_PYTEST_OPTIONS')
    basetemp = owned / 'pytest-status-source'
    if basetemp.exists() or basetemp.is_symlink():
        raise SystemExit('SOURCE_ATTEMPT_ALREADY_EXISTS_NO_REPLAY')
    sys.path.insert(0, str(CALLER / 'producer'))
    import pytest
    return pytest.main(['--noconftest', '-c', '/dev/null', '-p', 'no:cacheprovider',
                        '--import-mode=importlib', '--basetemp', str(basetemp),
                        *[str(CALLER / file) for file in SUITE]])


if __name__ == '__main__':
    raise SystemExit(main())
