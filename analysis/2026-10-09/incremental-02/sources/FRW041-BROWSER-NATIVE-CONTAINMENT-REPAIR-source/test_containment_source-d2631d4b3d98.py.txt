def test_render_preserves_custody_and_explicit_isolated_contour(tmp_path):
    payload, argv = unit.render(ROOT, tmp_path, 'friday-browser-prerequisite-1234567890abcdef.service')
    text = payload.decode()
    for line in ['PrivateNetwork=yes', 'PrivateUsers=self', 'PrivatePIDs=yes', 'PrivateMounts=yes', 'PrivateIPC=yes',
                 'TemporaryFileSystem=/dev/shm:rw,nosuid,nodev,noexec,mode=1777,size=64M',
                 'ProtectHome=tmpfs', 'ProtectSystem=strict', 'NoNewPrivileges=yes',
                 'RestrictAddressFamilies=AF_UNIX AF_INET AF_INET6', 'CapabilityBoundingSet=',
                 'AmbientCapabilities=', 'MemoryMax=2147483648', 'MemorySwapMax=0', 'TasksMax=128',
                 'CPUQuota=200%', 'CPUAffinity=0 1', 'ExitType=cgroup', 'KillMode=control-group',
                 'RuntimeMaxSec=45s', 'JobTimeoutSec=65s', 'TimeoutStopSec=5s', 'Restart=no']:
        assert line in text.splitlines()
    assert text.count('ExecStart=') == 1 and 'systemd-run' not in text
    assert 'EnvironmentFile=' not in text and 'NetworkNamespacePath=' not in text and 'JoinsNamespaceOf=' not in text
    assert argv[:2] == ['/usr/bin/env', '-i']
    assert 'HOME=' + str(tmp_path / 'runtime-home') in argv
    assert not any('CI=' in x or 'PROXY=' in x or 'PLAYWRIGHT' in x or '--no-sandbox' in x for x in argv)
    assert unit.POST_START_WINDOW == 55 and unit.MIN_START_WINDOW == 20

def test_outer_private_user_mapping_does_not_fake_nested_chrome_sandbox():
    browser = {'argv': [probe.EXECUTABLE, '--remote-debugging-port=0'], 'executable': probe.EXECUTABLE,
        'label': probe.LABEL, 'uid': [1000]*4, 'cap_eff': 0, 'cap_prm': 0, 'pid': 2,
        'nspid': [2], 'uid_map': [[1000,1000,1]], 'seccomp': 2, 'seccomp_filters': 1, 'cgroup': 'own',
        'user_namespace': 'user:[2]', 'network_namespace': 'net:[2]'}
    renderer = {'argv': [probe.EXECUTABLE, '--type=renderer'], 'uid': [1000]*4, 'cap_eff': 0, 'cap_prm': 0,
        'pid': 3, 'nspid': [3,1], 'uid_map': [[1000,1000,1]], 'gid_map': [[1000,1000,1]],
        'no_new_privs': 1, 'seccomp': 2, 'seccomp_filters': 2, 'cgroup': 'own',
        'user_namespace': 'user:[3]', 'network_namespace': 'net:[2]'}
    probe.check_sandbox_records(browser, [renderer], [[1000,1000,1]], ['1'])
    for field, bad in [('user_namespace','user:[2]'),

def test_mount_observation_requires_exact_unambiguous_destination():
    text='99 1 0:999 / /dev/shm rw,nosuid,nodev,noexec - tmpfs tmpfs rw,size=65536k\n'
    row=probe.mount_record(text,'/dev/shm')
    assert row['filesystem']=='tmpfs' and row['device']=='0:999'
    with pytest.raises(RuntimeError,match='missing_or_ambiguous'): probe.mount_record('', '/dev/shm')
    with pytest.raises(RuntimeError,match='missing_or_ambiguous'): probe.mount_record(text+text, '/dev/shm')


def tcp_row(address='0100007F',port=12345,inode=456,state='0A'):
    return ' 0: '+address+':'+format(port,'04X')+' 00000000:0000 '+state+' 00000000:00000000 00:00000000 00000000 1000 0 '+str(inode)+' 1\n'


def test_actual_native_stream_kernel_listener_positive_and_negatives():
    heading='sl local_address rem_address st tx_queue rx_queue tr tm->when retrnsmt uid timeout inode\n'
    assert probe.stream_listener_record(heading+tcp_row(),heading,12345)['inode']==456
    for text,v6,port in [(heading,heading,12345), (heading+tcp_row('00000000'),heading,12345),
                         (heading+tcp_row(state='01'),heading,12345),
                         (heading+tcp_row(inode=0),heading,12345),
                         (heading+tcp_row(),heading+tcp_row('00000000000000000000000001000000'),12345),
                         (heading+tcp_row()+tcp_row(),heading,12345),
                         (heading+tcp_row(),heading,0)]:
        with pytest.raises(RuntimeError): probe.stream_listener_record(text,v6,port)
