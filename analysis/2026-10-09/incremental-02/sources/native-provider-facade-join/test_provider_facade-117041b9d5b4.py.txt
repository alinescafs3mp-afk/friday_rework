Execution requires root's separate source qualification. Mocked command plumbing
below never creates a process; real compiled imports/FD/native cancellation are
separately mandatory in qualification.json.

import copy

import types
import unittest
from unittest.mock import patch

ROOT = Path(__file__).resolve().parents[1] / 'candidate'


class FacadeValues(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        cls.api = runpy.run_path(str(ROOT / 'caller/owner-helper/provider_facade.py'))

    def test_valid_exact_sdk_preparation(self):
        policy = {'requiredCapabilities': ['native-tools', 'network']}
        prepared = {'capability': 'host-control-v1', 'state': 'UNQUALIFIED_SOURCE_PREPARATION',
            'semanticArgv': ['/usr/bin/true'], 'argv': ['/usr/bin/bwrap', '--', '/usr/bin/true'],
            'requiredCapabilities': policy['requiredCapabilities'], 'statusFd': 7,
            'missing': ['FINAL_MAP_UNBOUND'], 'runtimeGrant': False}
        self.assertIs(self.api['validate_prepared'](prepared, ['/usr/bin/true'], 7, policy), prepared)
        for key, value in [('runtimeGrant', True), ('state', 'QUALIFIED'), ('statusFd', True),
                           ('statusFd', 8), ('semanticArgv', ['/usr/bin/false']),
                           ('missing', []), ('requiredCapabilities', ['native-tools'])]:
            changed = dict(prepared, **{key: value})
            with self.subTest(key=key, value=value), self.assertRaises(self.api['PreparationRefusal']):
                self.api['validate_prepared'](changed, ['/usr/bin/true'], 7, policy)

    def test_network_output_must_equal_owned_policy(self):
        policy = {'requiredCapabilities': ['network'], 'privateNetwork': {
            'driver': 'slirp4netns-1.3.3-frw-private-loopback', 'barrierProtocol': 'frw-success-byte-1-v1',
            'blockFd': 9, 'resolverFd': 11, 'resolverSha256': 'a' * 64, 'nameservers': ['9.9.9.9']}}
        prepared = {'capability': 'host-control-v1', 'state': 'UNQUALIFIED_SOURCE_PREPARATION',
            'semanticArgv': ['x'], 'argv': ['/usr/bin/bwrap', '--', 'x'], 'statusFd': 7,
            'requiredCapabilities': ['network'], 'missing': ['UNBOUND'], 'runtimeGrant': False,
            'network': copy.deepcopy(policy['privateNetwork'])}
        self.api['validate_prepared'](prepared, ['x'], 7, policy)
        prepared['network']['blockFd'] = 10
        with self.assertRaises(self.api['PreparationRefusal']):
            self.api['validate_prepared'](prepared, ['x'], 7, policy)

    def test_duplicate_nonfinite_malformed_and_overlimit_json_refuse(self):
        for data in (b'{"x":1,"x":2}', b'{"x":NaN}', b'{', b'\xff', b'12345'):
            with self.subTest(data=data), self.assertRaises(self.api['PreparationRefusal']):
                self.api['decode'](data, 4 if data == b'12345' else 100)

    def test_pure_bind_does_not_start_or_admit(self):
        owner = types.SimpleNamespace()
        bound = self.api['bind'](owner)
        self.assertIs(bound.owner, owner)
        self.assertFalse(bound.used)
        self.assertIsNone(bound.process)
        self.assertEqual(vars(owner), {})

    def test_failed_stop_persistence_retains_fence(self):
        def fail(): raise OSError('controlled storage failure')
        owner = types.SimpleNamespace(persist_stop=fail)
        bound = self.api['bind'](owner)
        bound.fence()
        self.assertTrue(owner._stop_pending)
        self.assertTrue(bound.fence_error)


class Stream:
    def __init__(self, fd): self.fd, self.closed = fd, False
    def fileno(self): return self.fd
    def close(self): self.closed = True


class FakeProcess:
    def __init__(self, exitcode=0):
        self.pid = 4711; self.returncode = None; self.exitcode = exitcode
        self.stdin, self.stdout, self.stderr = Stream(10), Stream(11), Stream(12)
        self.waits = []
    def wait(self, timeout):
        self.waits.append(timeout)
        self.returncode = self.exitcode
        return self.returncode


class Selector:
    def __init__(self): self.rows = {}
    def register(self, stream, event, data):
        self.rows[stream] = types.SimpleNamespace(fileobj=stream, data=data)
    def unregister(self, stream): self.rows.pop(stream)
    def get_map(self): return self.rows
    def select(self, timeout): return [(key, 0) for key in list(self.rows.values())]
    def close(self): self.rows.clear()


class FiniteCommand(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        cls.api = runpy.run_path(str(ROOT / 'project/scripts/dsh_prepare.py'))

    def invoke(self, *, stdout=b'{}\n', stderr=b'', exitcode=0, cancel=False, pidfd_error=False):
        process = FakeProcess(exitcode)
        buffers = {11: bytearray(stdout), 12: bytearray(stderr)}
        events = []
        owner = types.SimpleNamespace(
            retain=lambda p: events.append(('retained', p)),
            started=lambda p: events.append(('started', p)),
            fence=lambda: events.append(('fenced',)),
            record=lambda p, row: events.append(('record', p, row)))
        count = 0
        def guard():
            nonlocal count
            count += 1
            if cancel and count >= 2: raise RuntimeError('owner cancelled')
        owner.guard = guard
        def read(fd, maximum):
            part = bytes(buffers[fd][:maximum]); del buffers[fd][:maximum]
            return part
        def pidfd_open(pid, flags):
            if pidfd_error: raise OSError('pidfd unavailable')
            return 15
        globals_ = self.api['_run_owned_preparation'].__globals__
        fake_os = types.SimpleNamespace(pidfd_open=pidfd_open,
            set_inheritable=lambda *a: None, set_blocking=lambda *a: None,
            write=lambda fd, data: len(data), read=read,
            killpg=lambda pid, sig: events.append(('killpg', pid, process.returncode)),
            close=lambda fd: events.append(('close', fd)))
        fake_popen = types.SimpleNamespace(PIPE=-1, Popen=lambda *a, **k: (
            events.append(('popen', a, k)) or process), TimeoutExpired=TimeoutError)
        with patch.dict(globals_, os=fake_os, subprocess=fake_popen,
                selectors=types.SimpleNamespace(DefaultSelector=Selector, EVENT_READ=1, EVENT_WRITE=2),
                time=types.SimpleNamespace(monotonic=lambda: 1.0)):
            try:
                result = self.api['_run_owned_preparation'](['node', 'entry'], '/private',
                    {'LANG': 'C'}, 1.0, 3.0, b'{}', owner)
                return result, process, events
            except BaseException as error:
                return error, process, events

    def test_success_reaps_separate_process_and_closes_inherited_descriptors(self):
        result, process, events = self.invoke()
        self.assertNotIsInstance(result, BaseException)
        self.assertEqual(result[0], b'{}\n')
        self.assertEqual(process.waits, [0])
        spawn = next(e for e in events if e[0] == 'popen')
        self.assertEqual(spawn[2]['pass_fds'], ())
        self.assertIs(spawn[2]['close_fds'], True)
        self.assertFalse(any(e[0] == 'killpg' for e in events))
        self.assertEqual([e[0] for e in events[:3]], ['popen', 'retained', 'started'])
        self.assertEqual(next(e[2] for e in events if e[0] == 'record')['state'],
                         'COMMAND_REAPED_NOT_VALIDATED')

    def test_overflow_stderr_cancel_and_missing_pidfd_stop_and_reap(self):
        for case in ({'stdout': b'x' * 65537}, {'stderr': b'x' * 4097},
                     {'stderr': b'error'}, {'cancel': True}, {'pidfd_error': True}):
            with self.subTest(case=list(case)):
                error, process, events = self.invoke(**case)
                self.assertIsInstance(error, BaseException)
                self.assertIsNotNone(process.returncode)
                self.assertTrue(any(e[0] == 'fenced' for e in events))
                self.assertTrue(any(e[0] == 'record' for e in events))
                self.assertTrue(all(e[2] is None for e in events if e[0] == 'killpg'))
                self.assertTrue(all(0 <= t <= 2.0 for t in process.waits))
                self.assertTrue(all(s.closed for s in (process.stdin, process.stdout, process.stderr)))

    def test_nonzero_already_reaped_pid_is_never_group_signalled(self):
        error, process, events = self.invoke(exitcode=3)
        self.assertIsInstance(error, BaseException)
        self.assertEqual(process.returncode, 3)
        self.assertFalse(any(e[0] == 'killpg' for e in events))
        self.assertTrue(any(e[0] == 'fenced' for e in events))


if __name__ == '__main__':
    unittest.main()
