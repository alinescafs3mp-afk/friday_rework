def host_control_build_selection(donor, toolchain):
    """Select exact built SDK/launcher bytes; this is never a runtime admission.

    Called by the existing preparer after its source/build checks. The FileOwner
    independently compares sourcePins against its reviewed composed overlay, so
    selecting an old unpatched build cannot bind the newly added SDK method.
    """
    donor = Path(donor)
    if donor != donor.resolve() or os.uname().sysname != 'Linux':
        raise ValueError('host_control_linux_canonical_installation_required')
    machine = {'x86_64': 'x64', 'aarch64': 'arm64'}.get(os.uname().machine)
    if machine is None:
        raise ValueError('host_control_platform_not_selected')
    files, links = {}, {}
    total = 0
    for directory, dirs, names in os.walk(donor, followlinks=False):
        if Path(directory) == donor and '.git' in dirs:
            dirs.remove('.git')
        for name in [*dirs, *names]:
            path = Path(directory) / name
            rel = str(path.relative_to(donor))
            if path.is_symlink():
                target = path.resolve(strict=True)
                if not target.is_relative_to(donor) or '.git' in target.relative_to(donor).parts:
                    raise ValueError('host_control_external_installation_link')
                links[rel] = os.readlink(path)
            elif path.is_file():
                total += path.stat().st_size
                if total > 2 << 30:
                    raise ValueError('host_control_installation_byte_bound')
                # Stream installed files; do not allocate the whole tree in RAM.
                h = hashlib.sha256()
                with path.open('rb') as source:
                    for block in iter(lambda: source.read(65536), b''):
                        h.update(block)
                files[rel] = h.hexdigest()
            elif not path.is_dir():
                raise ValueError('host_control_special_installation_file')
            if len(files) + len(links) > 100000:
                raise ValueError('host_control_installation_file_bound')
    imports = {
        'provider': 'packages/sandbox/sandbox-local/lib/index.js',
        'cordis': 'vendor/cordis/lib/index.js',
        'landlock': 'native/system/packages/entry/lib/index.js',
    }
    source_names = (
        'packages/sandbox/sandbox-local/src/index.ts',
        'packages/sandbox/sandbox-local/src/host-control.ts',
        'packages/sandbox/sandbox-local/src/profiles.ts',
        'packages/sandbox/sandbox/src/index.ts',
        'native/system/packages/entry/src/index.ts',
        'native/system/packages/entry/src/main.c',
    )
    launcher = 'native/system/packages/linux-' + machine + '/bin/landlock-run'
    if any(name not in files for name in [*imports.values(), *source_names, launcher]):
        raise ValueError('host_control_compiled_sdk_or_scoped_launcher_missing')
    entry = Path(__file__).resolve().with_name('dsh_host_control_prepare.mjs')
    node = Path(toolchain['node_executable'])
    if node != node.resolve() or digest(node) != toolchain['node_sha256']:
        raise ValueError('host_control_node_selection_drift')
    return {'schema': 1, 'state': 'BUILT_SOURCE_SELECTION_NOT_ADMISSION',
        'donorCommit': '5badb15009ae1756c3afe0ae0cef1faafc290ccc',
        'root': str(donor), 'profile': 'builtin-linux-host-control-v1',
        'node': {'path': str(node), 'sha256': digest(node)},
        'entry': {'path': str(entry), 'sha256': digest(entry)},
        'commandHelper': {'path': str(Path(__file__).resolve()), 'sha256': digest(__file__)},
        'imports': imports, 'launcher': launcher, 'sourcePins': {n: files[n] for n in source_names},
        'files': files, 'links': links,
        'resolutionFiles': sorted(n for n in files if n == 'package.json' or n.endswith('/package.json')),
        'runtimeGrant': False}

''' + s[pos:]
