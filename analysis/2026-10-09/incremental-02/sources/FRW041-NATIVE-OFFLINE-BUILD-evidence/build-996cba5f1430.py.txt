def require(value, message):
    if not value:
        raise RuntimeError(message)


def unique_json(data):
    def pairs(rows):
        result = {}
        for key, value in rows:
            require(key not in result, 'duplicate_json_key')
            result[key] = value
        return result
    return json.loads(data, object_pairs_hook=pairs,
                      parse_constant=lambda _: require(False, 'nonfinite_json'))


def save(name, value):
    path = LIVE / name
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW | os.O_CLOEXEC, 0o600)
    with os.fdopen(fd, 'w') as stream:
        json.dump(value, stream, separators=(',', ':'))
        stream.write('\n')
        stream.flush()
        os.fsync(stream.fileno())


def read_pin(path, pin):
    path = Path(path)
    require(path.is_absolute() and '..' not in path.parts, 'clean_input_path')
    parent = os.open('/', os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW | os.O_CLOEXEC)
    try:
        for part in path.parts[1:-1]:
            child = os.open(part, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW | os.O_CLOEXEC, dir_fd=parent)
            os.close(parent)
            parent = child
            metadata = os.fstat(parent)
            require(metadata.st_uid in (0, 1000) and not metadata.st_mode & 0o022, 'input_ancestor')
        fd = os.open(path.name, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK | os.O_CLOEXEC, dir_fd=parent)
        with os.fdopen(fd, 'rb') as stream:
            a = os.fstat(stream.fileno())
            require(stat.S_ISREG(a.st_mode) and a.st_nlink == 1 and a.st_uid == 1000
                    and a.st_size == pin['bytes'] and stat.S_IMODE(a.st_mode) == pin['mode'], 'input_metadata')
            content = stream.read(a.st_size + 1)
            b = os.fstat(stream.fileno())
            c = os.stat(path.name, dir_fd=parent, follow_symlinks=False)
            key = lambda s: (s.st_dev, s.st_ino, s.st_size, s.st_mtime_ns, s.st_mode, s.st_nlink)
            require(key(a) == key(b) == key(c) and len(content) == a.st_size
                    and hashlib.sha256(content).hexdigest() == pin['sha256'], 'input_changed')
            return content
    finally:
        os.close(parent)


def clock_valid(clock):
    now = time.monotonic()
    require(clock['assignment'] == 'FRW041-NATIVE-OFFLINE-BUILD-RUN' and clock['unit'] == UNIT
            and clock['nonce'] == NONCE and clock['budget_seconds'] == 1800
            and clock['native_effects_authorized_by'] == 'Astra'
            and all(type(clock[k]) in (int, float) and math.isfinite(clock[k])
                    for k in ('started_monotonic', 'deadline_monotonic'))
            and clock['deadline_monotonic'] == clock['started_monotonic'] + 1800
            and clock['started_monotonic'] <= now < clock['deadline_monotonic']
            and clock['boot_id'] == Path('/proc/sys/kernel/random/boot_id').read_text().strip(), 'original_clock')
    return clock


def boundary_controls(grant):
    """Real future controls in the same unit before any candidate tool runs."""
    observed = {}
    require(os.getuid() == os.geteuid() == os.getgid() == os.getegid() == 1000, 'rootless_payload')
    status = dict(line.split(':', 1) for line in Path('/proc/self/status').read_text().splitlines())
    require(all(int(status[key], 16) == 0 for key in ('CapEff', 'CapPrm', 'CapBnd', 'CapAmb'))
            and int(status['NoNewPrivs']) == 1, 'zero_caps_nnp')
    require(os.readlink('/proc/self/ns/net') == grant['root_observation']['namespaces']['net']
            and os.readlink('/proc/self/ns/net') != grant['host_net'], 'different_host_net_namespace')
    for family in (socket.AF_INET, socket.AF_INET6):
        try:
            handle = socket.socket(family, socket.SOCK_STREAM)
        except OSError as exc:
            require(exc.errno in (errno.EAFNOSUPPORT, errno.EPERM, errno.EACCES), 'unexpected_ip_refusal')
            observed[str(family)] = exc.errno
        else:
            handle.close()
            raise RuntimeError('IP_socket_admitted')
    # Resolution cannot be accepted from local /etc/hosts; invalid DNS name has
    # no valid local answer. Native packet denial remains proven by AF and netns.
    try:
        answer = socket.getaddrinfo('friday-build-denial.invalid', 443)
    except socket.gaierror as exc:
        observed['dns_error'] = exc.errno
    else:
        raise RuntimeError('unexpected_dns_answer:' + str(answer))
    for path in ('/run/dbus/system_bus_socket', '<USER_RUNTIME>/bus', '<PRIVATE_TEMP>'):
        handle = socket.socket(socket.AF_UNIX, socket.SOCK_STREAM)
        try:
            handle.connect(path)
        except OSError as exc:
            require(exc.errno in (errno.ENOENT, errno.EACCES, errno.EPERM), 'host_socket_uncertainty')
            observed[path] = exc.errno
        else:
            raise RuntimeError('host_control_socket_reached')
        finally:
            handle.close()
    a, b = socket.socketpair()
    try:
        os.write(a.fileno(),b'local')
        require(os.read(b.fileno(),5) == b'local', 'local_compiler_IPC_broken')
    finally:
        a.close()
        b.close()
    fds = [int(p.name) for p in Path('/proc/self/fd').iterdir()]
    require(set(fds).issubset({0, 1, 2, 3, 4}), 'ambient_inherited_descriptor')
    for number in fds:
        try:
            target = os.readlink('/proc/self/fd/' + str(number))
        except FileNotFoundError:
            continue  # The bounded directory enumeration descriptor is closed.
        require(number<=3 and not target.startswith('socket:'), 'ambient_descriptor')
        if number==3:require(target==str(LIVE/'runtime.stdout'), 'wrong_result_descriptor')
    observed['local_unix_ipc'] = True
    observed['inherited_network_socket'] = False
    return observed


def prepare(inputs, clock, end):
    require(time.monotonic()<end,'preparation_original_ceiling')
    roots = (Path(inputs['build_root']), Path(inputs['joined_root']) / 'agent-browser')
    copied = []
    for original, pin in inputs['files'].items():
        path = Path(original)
        target = None
        if path.is_relative_to(roots[1]):
            target = LIVE / 'source' / path.relative_to(roots[1])
        elif path.is_relative_to(roots[0] / 'pnpm-store'):
            target = LIVE / 'pnpm-store' / path.relative_to(roots[0] / 'pnpm-store')
        # Full old manifests are retained; tools/crates/fonts are read directly
        # through read-only binds. Verify every manifest entry once, never only
        # a selected output or a relabelled donor build.
        require(time.monotonic() < end, 'preparation_cap_exhausted')
        data = read_pin(path, pin)
        if target is not None:
            target.parent.mkdir(parents=True, exist_ok=True, mode=0o700)
            fd = os.open(target, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600)
            with os.fdopen(fd, 'wb') as stream:
                stream.write(data)
            target.chmod(0o700 if pin['mode'] & 0o111 else 0o600)
            copied.append(str(target.relative_to(LIVE)))
    require(len(copied) > 323 and time.monotonic() < end, 'complete_private_source_store_required')
    save('prepared.json', {'verified_inputs': len(inputs['files']), 'copied': copied,
                          'completed_monotonic': time.monotonic(), 'automatic_retry': False})

def phase(name, argv, cap, clock, env, cwd, ceiling=None):
    started = time.monotonic()
    end = min(started + cap, clock['deadline_monotonic'] - 300)
    if ceiling is not None:
        end = min(end, ceiling)
    require(started < end, 'original_phase_budget_exhausted')
    intent = dict(name=name, argv=argv, started_monotonic=started, deadline_monotonic=end,
                  cap_seconds=cap, original_deadline=clock['deadline_monotonic'], automatic_retry=False)
    save(name + '-intent.json', intent)
    output = os.open(LIVE / (name + '.log'), os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600)
    proc = None
    try:
        with os.fdopen(output, 'wb') as stream:
            proc = subprocess.Popen(argv, cwd=cwd, env=env, stdin=subprocess.DEVNULL,
                                    stdout=stream, stderr=subprocess.STDOUT, close_fds=True, start_new_session=True)
            birth = int(Path('/proc', str(proc.pid), 'stat').read_text().rsplit(')', 1)[1].split()[19])
            save(name + '-handle.json', dict(pid=proc.pid, start_ticks=birth, pgid=proc.pid,
                                            pid_scope='payload_private_pid_namespace',
                                            pid_namespace=os.readlink('/proc/self/ns/pid'), **intent))
            require(time.monotonic() < end, 'phase_admission_late')
            code = proc.wait(timeout=end - time.monotonic())
            require(time.monotonic() < end and code == 0, 'phase_failed_or_late:' + name)
            remaining=[int(p.name) for p in Path('/proc').iterdir() if p.name.isdigit() and int(p.name)!=os.getpid()]
            require(not remaining,'phase_descendants_not_closed:'+str(remaining))
            save(name + '-exit.json', dict(returncode=code, parent_reaped=True, completed_monotonic=time.monotonic(), **intent))
    except BaseException:
        # Local leader termination is supplemental. Full detached descendants
        # remain in the one ordinary unit until native group stop is observed.
        if proc is not None:
            try:
                os.killpg(proc.pid, signal.SIGKILL)
            except ProcessLookupError:
                pass
            proc.wait(timeout=max(0.001, min(5, clock['deadline_monotonic'] - time.monotonic())))
        raise


def dashboard_manifest(inputs):
    root = LIVE / 'source/packages/dashboard/out'
    require(root.is_dir() and (root / 'index.html').is_file(), 'actual_dashboard_missing')
    entries = list(root.rglob('*'))
    require(all(not p.is_symlink() and (p.is_dir() or p.is_file()) for p in entries), 'dashboard_special_or_symlink')
    assets = [p for p in entries if p.is_file()]
    require(any(p.suffix == '.js' for p in assets) and any(p.suffix == '.css' for p in assets)
            and any(p.suffix == '.woff2' for p in assets), 'actual_dashboard_assets_fonts_missing')
    pins = {str(p.relative_to(root)): hashlib.sha256(p.read_bytes()).hexdigest() for p in assets}
    font_pins = {pin['sha256'] for name, pin in inputs['files'].items() if '/font-inputs/' in name and name.endswith('.woff2')}
    for path in assets:
        if path.suffix == '.woff2':
            require(path.read_bytes()[:4] == b'wOF2' and pins[str(path.relative_to(root))] in font_pins, 'real_font_bytes_required')
    class References(HTMLParser):
        def handle_starttag(self, tag, attrs):
            for key, value in attrs:
                if key in ('src', 'href') and value:
                    url = urlsplit(value)
                    if url.scheme or url.netloc:
                        continue
                    name = unquote(url.path)
                    if name.endswith(('.js', '.css', '.woff2')):
                        target = root / name.lstrip('/')
                        require(target.resolve().is_relative_to(root.resolve()) and target.is_file(), 'missing_export_reference:' + name)
    for path in assets:
        if path.suffix=='.css':
            for match in re.finditer(r'url\(\s*["\']?([^\)"\']+)',path.read_text()):
                url=urlsplit(match.group(1).strip())
                if url.scheme=='data':continue
                require(not url.scheme and not url.netloc,'external_export_asset')
                name=unquote(url.path)
                target=(root/name.lstrip('/')) if name.startswith('/') else (path.parent/name)
                require(target.resolve().is_relative_to(root.resolve()) and target.is_file(),'missing_css_asset:'+name)
    parser = References()
    for path in assets:
        if path.suffix == '.html':
            parser.feed(path.read_text())
    # build.rs writes this exact string when dashboard is absent; it cannot count.
    require(b'Dashboard not built' not in (root / 'index.html').read_bytes(), 'placeholder_refused')
    save('dashboard-artifacts.json', {'root': str(root), 'sha256': pins, 'actual_build_required': True})
    return pins


def main():
    require(len(sys.argv) == 1 and Path(__file__) == STAGE / 'build.py', 'fixed_root_staged_payload')
    os.umask(0o077)
    clock = clock_valid(unique_json((LIVE / 'clock.json').read_bytes()))
    inputs = unique_json((STAGE / 'inputs.json').read_bytes())
    # FD3 provenance is observed by the original root control before execution.
    fd = os.open(LIVE / 'runtime.stdout', os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600)
    if fd != 3:
        os.dup2(fd, 3)
        os.close(fd)
    os.set_inheritable(3, False)
    with (LIVE / 'identity-ready.fifo').open('wb', buffering=0) as channel:
        channel.write((NONCE + '\n').encode())
    with (LIVE / 'identity-grant.fifo').open('rb', buffering=0) as channel:
        require(select.select([channel], [], [], 25)[0], 'root_identity_grant_missing')
        require(channel.read(17) == (NONCE + '\n').encode(), 'root_identity_grant_foreign')
    grant = unique_json((LIVE / 'host-current-invocation.json').read_bytes())
    preparation_end=min(time.monotonic()+90,clock['deadline_monotonic']-300)
    require(time.monotonic()<preparation_end,'preparation_original_ceiling')
    def preparation_expired(*_):raise RuntimeError('preparation_original_cap_exhausted')
    old=signal.signal(signal.SIGALRM,preparation_expired)
    signal.setitimer(signal.ITIMER_REAL,preparation_end-time.monotonic())
    try:
        controls = boundary_controls(grant)
        prepare(inputs, clock, preparation_end)
    finally:
        signal.setitimer(signal.ITIMER_REAL,0)
        signal.signal(signal.SIGALRM,old)
    build = Path(inputs['build_root'])
    env = clean_env(build)
    cwd = LIVE / 'source'
    node = str(build / 'node/bin/node')
    pnpm = str(build / 'npm/pnpm-9.15.9/bin/pnpm.cjs')
    # All tool applicability checks are real future execs, never source PASS.
    versions_end = min(time.monotonic() + 30, clock['deadline_monotonic'] - 300)
    versions = [('node', [node, '--version'], 'v22.23.2'),
                ('pnpm', [node, pnpm, '--version'], '9.15.9'),
                ('rustc', [str(build / 'toolchain/bin/rustc'), '--version'], 'rustc 1.94.0'),
                ('cargo', [str(build / 'toolchain/bin/cargo'), '--version'], 'cargo 1.94.0')]
    for name, argv, expected in versions:
        phase('versions-' + name, argv, 30, clock, env, cwd, versions_end)
        require((LIVE / ('versions-' + name + '.log')).read_text().strip().startswith(expected), 'tool_version_mismatch:' + name)
    locks = {name: hashlib.sha256((cwd / name).read_bytes()).hexdigest() for name in ('cli/Cargo.lock', 'pnpm-lock.yaml')}
    phase('install', [node, pnpm, '--filter', 'dashboard...', 'install', '--offline', '--frozen-lockfile',
                      '--ignore-scripts', '--store-dir', str(LIVE / 'pnpm-store'),
                      '--package-import-method', 'copy', '--reporter', 'append-only'], 120, clock, env, cwd)
    phase('dashboard', [node, pnpm, '--filter', 'dashboard', 'build', '--webpack'], 300, clock, env, cwd)
    dashboard = dashboard_manifest(inputs)
    phase('cargo', [str(build / 'toolchain/bin/cargo'), '--config', str(build / 'cargo-offline.toml'),
                    'build', '--offline', '--locked', '--manifest-path', str(cwd / 'cli/Cargo.toml'),
                    '--features', 'rust-embed/debug-embed', '-j', '1'], 900, clock, env, cwd)
    binary = LIVE / 'target/debug/agent-browser'
    require(all(hashlib.sha256((cwd / name).read_bytes()).hexdigest() == value for name, value in locks.items()), 'source_lock_changed')
    generated = list((LIVE / 'target/debug/build').glob('agent-browser-*/out/cdp_generated.rs'))
    require(len(generated) == 1 and generated[0].stat().st_size > 0, 'actual_CDP_generation_missing')
    require(binary.is_file() and binary.read_bytes()[:4] == b'\x7fELF', 'actual_native_binary_missing')
    proof = dict(probe='OFFLINE_NATIVE_BUILD_COMPLETE', native_identity=grant['native_identity'],
                 cgroup='/system.slice/' + UNIT, boundary_controls=controls,
                 binary_path=str(binary), binary_sha256=hashlib.sha256(binary.read_bytes()).hexdigest(),
                 locks=locks, generated_CDP={str(generated[0]): hashlib.sha256(generated[0].read_bytes()).hexdigest()},
                 dashboard=dashboard, build_completed_monotonic=time.monotonic(),
                 installed_journey_accepted=False, source_task_accepted=False)
    require(time.monotonic() < clock['deadline_monotonic'] - 55, 'original_build_clock_exhausted')
    os.write(3, (json.dumps(proof, separators=(',', ':')) + '\n').encode())
    os.fsync(3)
    os.close(3)


if __name__ == '__main__':
    try:
        main()
    finally:
        # Notification only. Original host owner still stops/observes entire unit
        # and rehashes artifacts; this marker never proves cgroup cessation.
        fd=os.open(LIVE/'payload-finished.fifo',os.O_WRONLY|os.O_NONBLOCK|os.O_NOFOLLOW|os.O_CLOEXEC)
        try:os.write(fd,(NONCE+'\n').encode())
        finally:os.close(fd)
