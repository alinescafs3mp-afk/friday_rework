"""Finite controls of the actual installed runner policy, separately admitted.

Negative controls use real descriptors but call the SAME admission predicate
without issuing the forbidden write. They do not suppress the audit hook or
claim a process-terminal denial. Positive uses real os.fdopen and real write.
"""
from pathlib import Path
import os
import pytest


@pytest.fixture
def policy(request):
    return request.config._publication_fds


@pytest.fixture
def target(tmp_path):
    phase = tmp_path / 'collect-native'
    phase.mkdir(mode=0o700)
    return phase / 'classification-review.json'


def rejected(policy, descriptor):
    assert policy.admit_open(descriptor, 'w', os.O_WRONLY) is False


def test_current_exclusive_private_fd_writes_and_closes(policy, target):
    before = policy.admitted
    descriptor = os.open(target, policy.FLAGS, 0o400)
    with os.fdopen(descriptor, 'wb') as stream:
        stream.write(b'private-publication\n')
        stream.flush()
        os.fsync(stream.fileno())
    assert policy.admitted == before + 1
    assert target.read_bytes() == b'private-publication\n'
    assert target.stat().st_mode & 0o777 == 0o400
    assert not policy.pending and not policy.leaks()


def test_foreign_private_descriptor_has_no_provenance(policy, target):
    # Same private naming/mode, but not created by the authorized open route.
    with target.open('xb') as stream:
        target.chmod(0o400)
        rejected(policy, stream.fileno())


def test_outside_read_descriptor_is_refused_without_outside_write(policy):
    descriptor = os.open(Path(__file__), os.O_RDONLY)
    try:
        rejected(policy, descriptor)
    finally:
        os.close(descriptor)


def test_closed_descriptor_is_refused(policy, target):
    descriptor = os.open(target, policy.FLAGS, 0o400)
    os.close(descriptor)
    rejected(policy, descriptor)
    assert not policy.pending and not policy.leaks()


def test_reused_number_same_inode_is_refused(policy, target):
    descriptor = os.open(target, policy.FLAGS, 0o400)
    alias = os.dup(descriptor)
    identity = os.fstat(alias)
    os.close(descriptor)
    os.dup2(alias, descriptor, inheritable=False)
    try:
        assert os.fstat(descriptor).st_ino == identity.st_ino
        rejected(policy, descriptor)
    finally:
        os.close(descriptor)
        os.close(alias)
    assert not policy.leaks()


def test_intervening_open_invalidates_permit(policy, target):
    descriptor = os.open(target, policy.FLAGS, 0o400)
    try:
        with Path(__file__).open('rb') as stream:
            assert stream.read(1)
        rejected(policy, descriptor)
    finally:
        os.close(descriptor)


def test_symlink_rebinding_refused(policy, target):
    descriptor = os.open(target, policy.FLAGS, 0o400)
    saved = target.with_name('original.json')
    target.rename(saved)
    target.symlink_to(saved)
    try:
        rejected(policy, descriptor)
    finally:
        os.close(descriptor)


def test_hardlink_rebinding_refused(policy, target):
    descriptor = os.open(target, policy.FLAGS, 0o400)
    os.link(target, target.with_name('alias.json'))
    try:
        rejected(policy, descriptor)
    finally:
        os.close(descriptor)


def test_mode_drift_refused(policy, target):
    descriptor = os.open(target, policy.FLAGS, 0o400)
    target.chmod(0o600)
    try:
        rejected(policy, descriptor)
    finally:
        os.close(descriptor)


def test_consumed_permit_cannot_be_reused(policy, target):
    descriptor = os.open(target, policy.FLAGS, 0o400)
    with os.fdopen(descriptor, 'wb') as stream:
        rejected(policy, stream.fileno())


def test_duplicate_alias_has_no_permit_and_leak_is_visible(policy, target):
    descriptor = os.open(target, policy.FLAGS, 0o400)
    alias = os.dup(descriptor)
    try:
        rejected(policy, alias)
        assert {row['fd'] for row in policy.leaks()} == {descriptor, alias}
    finally:
        os.close(descriptor)
        os.close(alias)
    assert not policy.leaks()


def test_unconsumed_leak_is_visible_and_range_close_retires(policy, target):
    descriptor = os.open(target, policy.FLAGS, 0o400)
    try:
        assert policy.leaks() == [{'fd': descriptor, 'path': str(target)}]
    finally:
        os.closerange(descriptor, descriptor + 1)
    rejected(policy, descriptor)
    assert not policy.pending and not policy.leaks()


def test_wrong_creation_flags_never_mint_permit(policy, target):
    descriptor = os.open(target, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o400)
    try:
        assert descriptor not in policy.pending
        rejected(policy, descriptor)
    finally:
        os.close(descriptor)


def test_path_identity_replaced_refused(policy, target):
    descriptor = os.open(target, policy.FLAGS, 0o400)
    target.rename(target.with_name('replaced.json'))
    # A second exclusive creation retires the first permit, and identity checks
    # independently observe that the original inode is no longer at its path.
    replacement = os.open(target, policy.FLAGS, 0o400)
    try:
        rejected(policy, descriptor)
        rejected(policy, replacement)
    finally:
        os.close(descriptor)
        os.close(replacement)
