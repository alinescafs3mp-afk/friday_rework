def digest(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def private(path, directory=False):
    path = Path(path)
    require(path == path.resolve(strict=True), 'symlink_path')
    s = path.stat()
    require(s.st_uid == 1000 and not s.st_mode & 0o022, 'foreign_writes_or_owner')
    require(stat.S_ISDIR(s.st_mode) if directory else stat.S_ISREG(s.st_mode), 'wrong_file_type')
    return s



def root_file(path, mode=None):
    path = Path(path)
    require(path == path.resolve(strict=True), 'root_file_symlink')
    s = path.lstat()
    require(stat.S_ISREG(s.st_mode) and s.st_uid == s.st_gid == 0 and s.st_nlink == 1
            and not s.st_mode & 0o022 and (mode is None or stat.S_IMODE(s.st_mode) == mode), 'root_file_boundary')
    for parent in path.parents:
        q=parent.lstat()
        require(parent == parent.resolve(strict=True) and stat.S_ISDIR(q.st_mode)
                and q.st_uid == q.st_gid == 0 and not q.st_mode & 0o022, 'root_parent_boundary')
    return s

def usec(value):
    # systemctl renders timespans, rather than raw uint64 microseconds.
    units = {'us': 1, 'ms': 1000, 's': 1000000, 'min': 60000000}
    pieces = value.split()
    require(bool(pieces), 'missing_timespan')
    total = 0
    for piece in pieces:
        m = re.fullmatch(r'(\d+)(us|ms|s|min)', piece)
        require(m is not None, 'unrecognized_timespan')
        total += int(m[1]) * units[m[2]]
    return total


def affinity(value):
    result = set()
    for token in value.split():
        require(bool(re.fullmatch(r'\d+(?:-\d+)?', token)), 'invalid_affinity')
        ends = [int(x) for x in token.split('-')]
        require(max(ends) < 4096, 'invalid_affinity')
        result.update(range(ends[0], ends[-1] + 1))
    return result

def unit_object_path(unit):
    # systemd D-Bus path escape: every non-ASCII-alphanumeric byte is _hh.
    return '/org/freedesktop/systemd1/unit/' + ''.join(
        chr(b) if 48 <= b <= 57 or 65 <= b <= 90 or 97 <= b <= 122 else '_%02x' % b
        for b in unit.encode('utf-8'))


def typed_reply(stdout, signature):
    def pairs(values):
        result = {}
        for key, value in values:
            require(key not in result, 'duplicate_typed_property')
            result[key] = value
        return result
    value = json.loads(stdout, object_pairs_hook=pairs,
                       parse_constant=lambda value: require(False, 'invalid_typed_constant'))
    require(isinstance(value, dict) and set(value) == {'type', 'data'}
            and value['type'] == signature and isinstance(value['data'], list), 'typed_property_schema')
    return value['data']

class Attempt:
    """Finite single submission through the existing systemd manager."""
    def __init__(self, root, live, clock, transport=None, now=time.monotonic, interruption=None):
        self.root, self.live, self.clock = Path(root), Path(live), clock
        self.unit = unit_name(clock['nonce'])
        self.fragment = UNIT_DIRECTORY / self.unit
        self.payload, self.argv = render(self.root, self.live, self.unit)
        self.cgroup = '/system.slice/' + self.unit
        self.now, self.transport = now, transport or self.native_call
        self.records, self.submitted, self.closed = [], False, False
        self.file_identity, self.file_complete, self.load_attempted, self.cleaned = None, False, False, False
        self.manager = None
        self.publication_intent, self.publication_refused = False, False
        self.publication_fd = None
        self.interruption = interruption or (lambda: None)
        self.suffix_started, self.suffix_transport_seconds = None, 0.0
        self.host_sentinel = None
        self.root_control_handles = {}

    @property
    def host_sentinel_unresolved(self):
        s = self.host_sentinel
        return bool(s and ((s['listener'] is not None and not s['listener_closed']) or
                          (s['intent'] and not s['bind_refused'] and not s['absent_confirmed'])))

    def host_sentinel_evidence(self):
        if self.host_sentinel is None:
            return None
        return {k: v for k, v in self.host_sentinel.items() if k != 'listener'}

    def create_host_sentinel(self):
        require(self.host_sentinel is None, 'host_sentinel_no_replacement_attempt')
        path = Path('/dev/shm/friday-' + self.clock['nonce'] + '.sock')
        require(not path.exists() and not path.is_symlink(), 'host_filesystem_sentinel_exists')
        self.before_effect(); self.remaining()
        # Original Attempt owns the handle and create intent before bind. A
        # later observation failure cannot turn a possible named effect into None.
        s = self.host_sentinel = {'path': str(path), 'parent_pid': os.getpid(),
            'deadline_monotonic': self.clock['deadline_monotonic'], 'intent': False,
            'bind_returned': False, 'bind_refused': False, 'identity': None,
            'identity_verified': False, 'listener': None, 'listener_fd': None,
            'listener_closed': False, 'unlink_returned': False,
            'absent_confirmed': False, 'cleanup_errors': []}
        s['listener'] = socket.socket(socket.AF_UNIX, socket.SOCK_STREAM)
        s['listener_fd'] = s['listener'].fileno()
        require(s['listener_fd'] >= 0, 'host_filesystem_sentinel_descriptor')
        s['intent'] = True
        try:
            s['listener'].bind(str(path))
        except OSError as exc:
            # This precise AF_UNIX pathname refusal created nothing. Other
            # exceptions/unknown outcomes stay latched; never adopt/unlink a path.
            if exc.errno == errno.EADDRINUSE:
                s['bind_refused'] = True
            raise
        s['bind_returned'] = True
        hs = path.lstat()
        s['identity'] = [hs.st_dev, hs.st_ino]
        require(stat.S_ISSOCK(hs.st_mode) and hs.st_uid == 1000 and hs.st_nlink == 1,
                'host_filesystem_sentinel_identity')
        s['identity_verified'] = True
        s['listener'].listen(1)
        s['listener'].setblocking(False)
        return s['listener']

    def close_host_sentinel(self):
        s = self.host_sentinel
        if s is None or not self.host_sentinel_unresolved:
            return
        # This is same-owner cleanup under the SAME deadline, not a new grant.
        self.remaining()
        if s['listener'] is not None and not s['listener_closed']:
            s['listener'].close()
            s['listener_closed'] = True
        if not s['intent'] or s['bind_refused']:
            return
        require(s['bind_returned'] and s['identity'] is not None and s['identity_verified'],
                'host_filesystem_sentinel_identity_unknown')
        path = Path(s['path'])
        if not s['unlink_returned']:
            hs = path.lstat()
            require([hs.st_dev, hs.st_ino] == s['identity'] and stat.S_ISSOCK(hs.st_mode)
                    and hs.st_uid == 1000 and hs.st_nlink == 1, 'host_filesystem_sentinel_replaced')
            self.remaining()
            path.unlink()
            s['unlink_returned'] = True
        # Never erase the identity on unlink alone. Errors or a replacement
        # after unlink preserve custody; only this precise absence closes it.
        try:
            path.lstat()
        except FileNotFoundError:
            self.remaining()
            s['absent_confirmed'] = True
        else:
            raise RuntimeError('host_filesystem_sentinel_absence_unconfirmed')

    @property
    def unresolved(self):
        # Missing inode/fd is uncertainty after intent, never proof of no effect.
        return self.host_sentinel_unresolved or getattr(self, 'root_control_unknown', False) or (((self.publication_intent and not self.publication_refused)
                or self.submitted) and not (self.closed and self.cleaned))

    def before_effect(self):
        require(self.interruption() is None, 'foreground_interrupted')

    def close_publication(self):
        if self.publication_fd is not None:
            os.close(self.publication_fd)
            self.publication_fd = None

    def remaining(self, reserve=0):
        n = self.clock['deadline_monotonic'] - self.now() - reserve
        require(n > 0, 'original_budget_exhausted')
        return n

    def suffix_reserve(self, reserve=10):
        if self.suffix_started is None:
            return reserve
        local_used = self.now() - self.suffix_started - self.suffix_transport_seconds
        require(0 <= local_used <= 10, 'suffix_local_window_exceeded')
        # Keep future full native windows plus the unspent part of local10.
        # Previously spent local time is not reserved a second time.
        return reserve - local_used

    def native_call(self, argv, timeout):
        if argv == [SUDO, '-n', '--', str(ADMIN_DIRECTORY / 'start')]:
            return self.native_start(argv, timeout)
        try:
            p = subprocess.run(argv, stdin=subprocess.DEVNULL, stdout=subprocess.PIPE,
                               stderr=subprocess.PIPE, env=ENV, timeout=timeout, close_fds=True)
        except subprocess.TimeoutExpired as exc:
            # subprocess.run kills/reaps the finite client; native manager outcome remains unknown.
            return {'returncode': None, 'timeout': True, 'client_reaped': True,
                    'stdout': (exc.stdout or b'').decode(errors='replace'), 'stderr': (exc.stderr or b'').decode(errors='replace')}
        return {'returncode': p.returncode, 'timeout': False, 'client_reaped': True,
                'stdout': p.stdout.decode(errors='replace'), 'stderr': p.stderr.decode(errors='replace')}

    def native_start(self, argv, timeout):
        """One existing wait client, finite pipe rendezvous, no start retry/polling.
        Native identity is read by the original host custodian; the isolated
        probe never receives the user's systemd bus or a host network FD.
        """
        started = self.now()
        client = None
        ready_fd = grant_fd = None
        host_tcp = host_abstract = host_filesystem = None
        host_socket_path = Path("/dev/shm/friday-" + self.clock["nonce"] + ".sock")
        identity = None
        answer = None
        try:
            def fifo(name):
                path = self.live / name
                s = path.lstat()
                require(stat.S_ISFIFO(s.st_mode) and s.st_uid == 1000 and stat.S_IMODE(s.st_mode) == 0o600 and s.st_nlink == 1, 'handshake_fifo_identity')
                fd = os.open(path, os.O_RDWR | os.O_NONBLOCK | os.O_NOFOLLOW | os.O_CLOEXEC)
                require((os.fstat(fd).st_dev, os.fstat(fd).st_ino) == (s.st_dev, s.st_ino), 'handshake_fifo_replaced')
                return fd
            ready_fd = fifo('identity-ready.fifo')
            grant_fd = fifo('identity-grant.fifo')

            except subprocess.TimeoutExpired:

        except BaseException as exc:

            if client is not None:
                if client.poll() is None: client.kill()
                out, err = client.communicate()

            answer = {'returncode': None, 'timeout': True, 'client_reaped': True,
                    'stdout': out.decode(errors='replace'), 'stderr': err.decode(errors='replace'),
                    'handshake_error': str(exc), 'host_identity': identity}
            try:
                self.capture_root_control('start'); self.root_controls_closed()
            except BaseException as control_error:
                answer['root_control_unknown'] = str(control_error)

        finally:

            try:
                for fd in (ready_fd, grant_fd):

                # never prevent the owned sentinel cleanup attempt.
                try:
                    self.close_host_sentinel()
                except BaseException as exc:
                    cleanup_errors.append(str(exc))
                    if self.host_sentinel is not None:
                        self.host_sentinel['cleanup_errors'].append(str(exc))
                # Keep the original handshake exception/outcome. Supplemental
                # cleanup failure never substitutes for it or clears custody.
                if answer is not None:
                    answer['host_sentinel'] = self.host_sentinel_evidence()
                    if cleanup_errors:
                        answer['cleanup_errors'] = cleanup_errors
                        answer['timeout'] = True
        return answer

    def completion_status(self, result):
        # Called after all persistence, flushed output and descriptor/handler
        # bookkeeping. No success bytes are published before these checks.
        # The caller must observe the reaped parent exit before this same clock
        # expires; a pending receipt/stdout alone never proves accepted success.
        if self.unresolved or not self.closed or not self.cleaned:
            return 1
        if self.interruption() is not None:
            return 2
        try:
            self.suffix_reserve()
            self.remaining()
        except RuntimeError:
            return 3
        return 0 if result['state'] == 'PASS' else 1

    def reconcile_prestart(self):
        require(not self.submitted and self.publication_intent and not self.publication_refused,
                'post_submission_or_no_owned_publication')
        require(getattr(self,'clock_hash',None) is not None,'admission_incomplete_retains_custody')
        self.remaining(35)
        # Fixed cleanup revalidates root receipt and exact creation journal itself.
        # A failed user-side adoption read cannot suppress this bounded rollback.
        self.cleanup(None)
        self.cleaned,self.closed=True,True


def retain_unknown(attempt):
    # Existing original foreground parent's descriptor remains custody. No new daemon,
    # timer, polling, retry or replacement custodian. Recovery is external and reviewed.
    while attempt.unresolved:
        try:
            code.interact(banner='STOP_UNCONFIRMED: original parent/lock retained; no retry.', local={'attempt': attempt})
        except BaseException:
            pass
        if attempt.unresolved:
            try: signal.pause()
            except BaseException: pass

        require(digest(ROOT / name) == admission['source_sha256'][name], 'source_changed')
    review_path = Path(admission['independent_review_path']); private(review_path)
    require(digest(review_path) == admission['independent_review_sha256'], 'independent_review_changed')
    review = json.loads(review_path.read_text())
    require(review['state'] == 'INDEPENDENT_BROWSER_SYSTEM_SERVICE_SOURCE_ACCEPTED'
            and review['package_pins_sha256'] == digest(ROOT/'package-pins.json')
            and review['source_task'] == SOURCE_TASK and review['runtime_grant'] is False, 'exact_changed_area_review_required')
    clock_path = Path(admission['clock_path']); private(clock_path)
    require(str(clock_path) == IDENTITY['trial_clock_path'] and stat.S_IMODE(clock_path.stat().st_mode)==0o400, 'fixed_immutable_original_clock_path')
    require(digest(clock_path) == admission['clock_sha256'], 'clock_changed')
    clock = json.loads(clock_path.read_text())
    require(clock['budget_seconds'] == 120 and clock['deadline_monotonic'] == clock['started_monotonic'] + 120
            and clock['started_monotonic'] <= time.monotonic() < clock['deadline_monotonic']
            and clock['boot_id'] == Path('/proc/sys/kernel/random/boot_id').read_text().strip()
            and clock['assignment'] == 'FRW041-BROWSER-NATIVE-CONTAINMENT-TRIAL'
            and clock['native_effects_authorized_by'] == 'Astra', 'original_clock_invalid')
    name = unit_name(clock['nonce']); require(clock['unit'] == name == IDENTITY['unit'] and clock['nonce'] == IDENTITY['nonce'], 'unit_clock_mismatch')
    protected_install()  # host root ownership and complete287-file integrity, before unit publication
    manifest = json.loads((ROOT / 'source-pins.json').read_text())
    require(BUSCTL in manifest['pins'], 'typed_transport_pin_required')
    alias = manifest['python_alias']
    require(alias['path'] == '<LOCAL_HOME>/jericho/.venv/bin/python'
            and Path(alias['path']).resolve(strict=True) == Path(alias['resolved'])
            and alias['resolved'] in manifest['pins'], 'python_alias_changed')
    for p, pin in manifest['pins'].items():
        require(time.monotonic() + 90 < clock['deadline_monotonic'], 'pin_budget_exhausted')
        require(Path(p).stat().st_size == pin['size'] and digest(p) == pin['sha256'], 'dependency_changed')
    live = Path(admission['live_directory'])
    require(live.parent == ROOT and not live.exists() and live.name == 'live-' + clock['nonce'], 'fresh_runtime_directory_required')
    return admission, clock, live


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--admission', type=Path, required=True)
    args = parser.parse_args()
    require(os.getuid() == 1000 and os.isatty(0), 'foreground_native_owner_required')
    os.umask(0o077); os.sched_setaffinity(0, {0, 1})
    resource.setrlimit(resource.RLIMIT_AS, (2 * 1024**3, 2 * 1024**3))
    live=Path(IDENTITY['live_directory'])
    fd = os.open(LOCK, os.O_RDONLY | os.O_NOFOLLOW)
    s = os.fstat(fd)
    require((s.st_dev, s.st_ino, s.st_uid, s.st_gid, s.st_nlink, s.st_size, stat.S_IMODE(s.st_mode))
            == (2050, 9044963, 1000, 1000, 1, 0, 0o664), 'quality_lock_changed')
    fcntl.flock(fd, fcntl.LOCK_EX | fcntl.LOCK_NB)
    attempt = None
    interrupt = [None]
    def interrupted(signum, frame):
        # Do not raise asynchronously across bookkeeping/finally boundaries.
        # Admission checks stop pre-start effects; post-start finite settlement
        # proceeds, or the same original foreground parent retains custody.
        interrupt[0] = signum
    handlers = {sig: signal.signal(sig, interrupted) for sig in (signal.SIGINT, signal.SIGHUP, signal.SIGTERM)}
    try:
        attempt = Attempt(ROOT, live,dict(nonce=IDENTITY['nonce'],deadline_monotonic=0),interruption=lambda:interrupt[0])
        attempt.publication_intent=True;attempt.load_attempted=True
        admission,clock,live=admit(args.admission)
        attempt.clock=clock;attempt.clock_hash=admission['clock_sha256']
        attempt.adopt_installed()
        live.mkdir(mode=0o700)
        for directory in ('runtime-tmp', 'runtime-cache', 'runtime-config', 'runtime-home', 'native-sockets', 'chrome-profile'):
            (live / directory).mkdir(mode=0o700)
        for name in ('identity-ready.fifo', 'identity-grant.fifo'):
            os.mkfifo(live / name, 0o600)
        clock_fd = os.open(live/'clock.json', os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW, 0o600)
        with os.fdopen(clock_fd,'wb') as stream:
            original_bytes = Path(admission['clock_path']).read_bytes()
            require(hashlib.sha256(original_bytes).hexdigest() == admission['clock_sha256'], 'original_clock_changed_before_copy')
            stream.write(original_bytes); stream.flush(); os.fsync(stream.fileno())
        result = attempt.execute()
        # A write/flush can itself cross the deadline or receive a signal. Keep
        # every published success-path record explicitly pending. Acceptance is
        # the final checked exit status plus the caller's actual reaped timestamp,
        # never a provisional PASS written before the operation returns.
        pending = dict(result, state='CESSATION_CONFIRMED_PENDING_FINAL_EXIT',
                       probe_verified=bool(result['proof']), final_outcome_accepted=False,
                       clock=clock, required_acceptance='parent_exit_0_reaped_before_original_deadline')
        fresh_json(live / 'result.json', pending)
        print(json.dumps(pending), flush=True)
    except BaseException as exc:
        if attempt and attempt.unresolved and not attempt.submitted:
            try: attempt.reconcile_prestart()
            except BaseException: pass
        # Error reporting is best effort. It cannot control lock custody.
        failure = None
        try:
            unknown = bool(attempt and attempt.unresolved)
            failure = {'state': 'STOP_UNCONFIRMED' if unknown else 'REFUSED_CLOSED',
                       'error': str(exc), 'error_type': type(exc).__name__, 'parent_pid': os.getpid(),
                       'submitted': bool(attempt and attempt.submitted), 'signal': interrupt[0],
                       'records': attempt.records if attempt else [], 'automatic_retry': False,
                       'native_cgroup_closed': bool(attempt and attempt.closed),
                       'cleaned': bool(attempt and attempt.cleaned), 'final_outcome_accepted': False}
            failure['host_sentinel'] = attempt.host_sentinel_evidence() if attempt else None
        except BaseException:
            pass
        if failure is not None:
            try:
                if live.exists(): fresh_json(live / 'failure.json', failure)
            except BaseException:
                pass
            try: print(json.dumps(failure), flush=True)
            except BaseException: pass
        raise
    finally:
        # This finally executes despite simultaneous receipt/serialization/output
        # failures or KeyboardInterrupt. Catchable signal handlers never throw.
        # retain_unknown is itself allowed to suffer terminal/error interruption;
        # only verified original reconciliation can clear unresolved custody.
        if attempt is not None:
            while attempt.unresolved:
                try: retain_unknown(attempt)
                except BaseException: pass
        os.close(fd)
        for sig, old in handlers.items(): signal.signal(sig, old)
    # This is deliberately after finally: even descriptor close/handler restore
    # belong to the original clock. Already confirmed closure stays closed when
    # reporting is late or interrupted; there is no new custody wait or retry.
    return attempt.completion_status(result)


if __name__ == '__main__':
    raise SystemExit(main())
