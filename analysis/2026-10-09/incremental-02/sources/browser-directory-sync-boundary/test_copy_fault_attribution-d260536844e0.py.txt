"""Pure observed-node controls / SOURCE ONLY, never import or run the runner."""
import ast
from pathlib import Path
import pytest

ROOT=Path(__file__).parent
PREFIX='project/tests/test_browser_native_host.py::'
ARMED=PREFIX+'first';ACTUAL=PREFIX+'second'

def predicate():
    tree=ast.parse((ROOT/'copy_fault_attribution_runner.py').read_text())
    node=next(n for n in tree.body if isinstance(n,ast.FunctionDef) and n.name=='copy_fault_node_allowed')
    namespace={}
    exec(compile(ast.Module(body=[node],type_ignores=[]),'<pure-observed-node>','exec'),namespace)
    return namespace['copy_fault_node_allowed']


def test_fault_actual_later_node_is_valid_without_overwriting_armed_node():
    row={'armed_node':ARMED,'node':ARMED}
    assert predicate()(ACTUAL,[ARMED,ACTUAL],row)
    assert row=={'armed_node':ARMED,'node':ARMED}


@pytest.mark.parametrize('node,nodes,row',[
    (None,[ARMED,ACTUAL],{'armed_node':ARMED,'node':ARMED}),
    (ACTUAL,[ARMED],{'armed_node':ARMED,'node':ARMED}),
    ('foreign/test.py::test',[ARMED,'foreign/test.py::test'],{'armed_node':ARMED,'node':ARMED}),
    (ACTUAL,[ARMED,ACTUAL],{'armed_node':'unknown','node':'unknown'}),
    (ACTUAL,[ARMED,ACTUAL],{'armed_node':ARMED,'node':ACTUAL}),
    (ACTUAL,[ARMED,ACTUAL],{'armed_node':ARMED,'node':ARMED,'injection_frame':{}}),
    (ACTUAL,[ARMED,ACTUAL],{'node':ARMED}),
])
def test_fault_unknown_replayed_and_changed_observations_refuse(node,nodes,row):
    assert not predicate()(node,nodes,row)


def test_all_three_actual_fault_branches_record_before_their_effects_and_never_assign_current_node():
    tree=ast.parse((ROOT/'copy_fault_attribution_runner.py').read_text())
    funcs={n.name:n for n in tree.body if isinstance(n,ast.FunctionDef)}
    profile=funcs['copy_fault_profile'];record=funcs['record_copy_fault_node']
    calls=[n for n in ast.walk(profile) if isinstance(n,ast.Call) and ast.unparse(n.func)=='record_copy_fault_node']
    assert len(calls)==3 and all([ast.unparse(a) for a in n.args]==['frame','event'] for n in calls)
    branches=[n for n in ast.walk(profile) if isinstance(n,ast.If) and any(isinstance(s,ast.Expr) and isinstance(s.value,ast.Call) and ast.unparse(s.value.func)=='record_copy_fault_node' for s in n.body)]
    assert len(branches)==3
    for branch in branches:
        statements=[ast.unparse(s) for s in branch.body]
        index=statements.index('record_copy_fault_node(frame, event)')
        assert statements[index+1].startswith("state['negative_triggered'].update(")
        assert all('os.chmod(' not in x and 'os.symlink(' not in x and 'os.set_inheritable(' not in x and 'os.lseek(' not in x for x in statements[:index])
    for node in ast.walk(record):
        if isinstance(node,ast.Subscript) and isinstance(node.ctx,ast.Store):
            assert ast.unparse(node)!="state['current_node']"
    body=ast.unparse(record)
    assert "node = state.get('current_node')" in body
    assert "copy_fault_node_allowed(node, state['nodes'], row)" in body
    assert "row.update(node=node" in body
    assert "fence.COPY_PENDING['node'] == node" in body


def test_fault_overlay_keeps_every_prior_runner_body_except_exact_attribution():
    import json,hashlib
    observation=json.loads((ROOT.parent/'attribution-source-observation.json').read_text())
    path=Path(observation['source_runner']['path'])
    assert hashlib.sha256(path.read_bytes()).hexdigest()==observation['source_runner']['sha256']
    old=ast.parse(path.read_text());new=ast.parse((ROOT/'copy_fault_attribution_runner.py').read_text())
    new.body=[n for n in new.body if not (isinstance(n,ast.FunctionDef) and n.name in ('copy_fault_node_allowed','record_copy_fault_node'))]
    profile=next(n for n in new.body if isinstance(n,ast.FunctionDef) and n.name=='copy_fault_profile')
    for branch in ast.walk(profile):
        if isinstance(branch,ast.If):
            branch.body=[n for n in branch.body if not (isinstance(n,ast.Expr) and isinstance(n.value,ast.Call) and ast.unparse(n.value.func)=='record_copy_fault_node')]
    removals=0
    for node in ast.walk(new):
        if isinstance(node,ast.Dict) and any(isinstance(k,ast.Constant) and k.value=='armed_node' for k in node.keys):
            index=next(i for i,k in enumerate(node.keys) if isinstance(k,ast.Constant) and k.value=='armed_node')
            assert ast.unparse(node.values[index])=='item.nodeid'
            del node.keys[index];del node.values[index];removals+=1
    assert removals==1
    assert ast.dump(new,include_attributes=False)==ast.dump(old,include_attributes=False)
