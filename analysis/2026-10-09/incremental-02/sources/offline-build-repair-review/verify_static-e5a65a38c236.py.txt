def unique(pairs):
    answer = {}
    for k, v in pairs:
        assert k not in answer, ('duplicate_json_key', k)
        answer[k] = v
    return answer

def data(path):
    return json.loads(path.read_bytes(), object_pairs_hook=unique)

def ref(path):
    path = Path(path)
    a = path.lstat()
    raw = path.read_bytes()
    b = path.lstat()
    assert stat.S_ISREG(a.st_mode) and a.st_nlink == 1
    key = lambda s: (s.st_dev, s.st_ino, s.st_size, s.st_mtime_ns, s.st_mode, s.st_uid, s.st_nlink)
    assert key(a) == key(b) and a.st_size == len(raw)
    return dict(path=str(path), sha256=hashlib.sha256(raw).hexdigest(), bytes=len(raw), mode=stat.S_IMODE(a.st_mode), uid=a.st_uid, nlink=a.st_nlink)

def check_ref(expected):
    actual = ref(expected['path'])
    for key in ('sha256', 'bytes', 'mode'):
        if key in expected:
            value = expected[key]
            if key == 'mode' and isinstance(value, str):
                value = int(value, 8)
            assert actual[key] == value, (expected['path'], key, actual[key], value)
    return actual

def counts(tree):
    answer = {}
    for name, node in tests(tree).items():
        count = 1
        for dec in node.decorator_list:
            if isinstance(dec,ast.Call) and isinstance(dec.func,ast.Attribute) and dec.func.attr=='parametrize':
                assert isinstance(dec.args[1], (ast.List,ast.Tuple))
                count *= len(dec.args[1].elts)
        answer[name] = count
    return answer
