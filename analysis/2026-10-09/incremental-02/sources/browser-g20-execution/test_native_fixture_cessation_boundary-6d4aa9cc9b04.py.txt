import ast
from pathlib import Path
from types import SimpleNamespace as NS
import pytest
import private_effect_fence as fence

class Refused(Exception):pass

def refuse(event):raise Refused(event)

SCOPES=[('registry',n) for n in sorted(fence.REGISTRY_NODES)]+[
    (phase,n) for phase in ('native','owner') for n in sorted(fence.PIDFD_EXTRA_NODES[phase])]

def failed_reports(node,stage='call'):
    stages=['setup','teardown'] if stage=='setup' else ['setup','call','teardown']
    return [dict(nodeid=node,when=when,outcome='failed' if when==stage else 'passed',
                 pidfd_failure_frames=[{'path':fence.BASE+'/'+node.split('::')[0],
                     'name':node.split('::')[-1], 'line':fence.PIDFD_LOOP_ENTRY_LINES.get(node,100)-1}]
                 if when==stage else None)
            for when in stages]

def missing_model(monkeypatch,phase,node):
    monkeypatch.setattr(fence,'state',dict(phase=phase,current_node=node,reports=failed_reports(node,'setup' if phase=='owner' else 'call')))
    for name in ('REGISTRY_PIDFD_ROWS','PIDFD_FAILED_BEFORE_PROBE','REGISTRY_ROWS','PAIRS'):
        monkeypatch.setattr(fence,name,{})
    monkeypatch.setattr(fence,'PAIR_NODES',set())
    monkeypatch.setattr(fence,'COUNTS',{k:0 for k in fence.COUNTS})
    monkeypatch.setattr(fence,'REGISTRY_PIDFD_PENDING',None)
    monkeypatch.setattr(fence,'LOCAL',NS())
    monkeypatch.setattr(fence,'deny',refuse)
    item=dict(node=node,phase=phase,bound=False,loop=None,endpoints=None,endpoint_births=None,
              asyncgen_hooks=fence.sys.get_asyncgen_hooks(),
              coroutine_origin_depth=fence.sys.get_coroutine_origin_tracking_depth(),
              probe_start=fence.pidfd_probe_checkpoint())
    monkeypatch.setattr(fence,'REGISTRY_ITEM',item if phase=='registry' else None)
    monkeypatch.setattr(fence,'PIDFD_EXTRA_ITEM',item if phase!='registry' else None)
    return item

def finish(phase,node):
    if phase=='registry':fence.registry_test_finish(node)
    else:fence.pidfd_extra_test_finish(node)

@pytest.mark.parametrize('phase,node',SCOPES)
def test_failed_before_probe_preserves_original_failure_without_manufactured_closure(monkeypatch,phase,node):
    item=missing_model(monkeypatch,phase,node)
    reports=fence.state['reports'];original=[dict(r) for r in reports]
    finish(phase,node)
    summary=fence.registry_pidfd_cessation()
    assert reports==original and any(r['outcome']=='failed' for r in reports)
    assert summary['rows']==[] and summary['attempts']==0
    assert summary['all_observed_probes_closed'] is True
    assert summary['complete_expected_probes'] is False
    assert summary['complete_two_probes'] is False
    assert summary['failed_before_probe']==[dict(node=node,phase=phase,
        classification='FAILED_BEFORE_PROBE',probe_observed=False,closure_claimed=False,test_outcome='failed')]
    assert fence.PIDFD_EXTRA_ITEM is None and fence.REGISTRY_ITEM is None

@pytest.mark.parametrize('problem,allowed',[
 ('call_failure',True),('setup_failure',True),('teardown_failure',False),
 ('all_passed',False),('skipped',False),('incomplete',False),('duplicate',False),
 ('foreign_node',False),('no_frames',False),('empty_frames',False),
 ('runner_failure',False),('selector_failure',False),('opaque_frame',False),
 ('setup_fail_then_call',False),('unknown_outcome',False),
 ('at_loop_call',False),('after_loop_call',False),('foreign_frame',False),('bool_line',False),
 ('missing_test_frame',False)])
def test_failed_probe_exception_requires_complete_failed_reports_and_nonconstructor_trace(monkeypatch,problem,allowed):
    phase,node=SCOPES[2];missing_model(monkeypatch,phase,node)
    reports=fence.state['reports'];failed=reports[1]
    if problem=='setup_failure':fence.state['reports']=failed_reports(node,'setup')
    elif problem=='teardown_failure':fence.state['reports']=failed_reports(node,'teardown')
    elif problem=='all_passed':failed['outcome']='passed'
    elif problem=='skipped':failed['outcome']='skipped'
    elif problem=='incomplete':reports.pop()
    elif problem=='duplicate':reports.append(dict(failed))
    elif problem=='foreign_node':failed['nodeid']='foreign'
    elif problem=='no_frames':failed.pop('pidfd_failure_frames')
    elif problem=='empty_frames':failed['pidfd_failure_frames']=[]
    elif problem=='runner_failure':failed['pidfd_failure_frames'][0]['path']=fence.STDLIB+'/asyncio/runners.py'
    elif problem=='selector_failure':failed['pidfd_failure_frames'][0]['path']=fence.STDLIB+'/selectors.py'
    elif problem=='opaque_frame':failed['pidfd_failure_frames']=[object()]
    elif problem=='setup_fail_then_call':reports[0]['outcome']='failed'
    elif problem=='unknown_outcome':failed['outcome']='unknown'
    elif problem=='at_loop_call':failed['pidfd_failure_frames'][0]['line']=fence.PIDFD_LOOP_ENTRY_LINES[node]
    elif problem=='after_loop_call':failed['pidfd_failure_frames'][0]['line']=fence.PIDFD_LOOP_ENTRY_LINES[node]+1
    elif problem=='foreign_frame':failed['pidfd_failure_frames'][0]['path']='/foreign/test.py'
    elif problem=='bool_line':failed['pidfd_failure_frames'][0]['line']=True
    elif problem=='missing_test_frame':failed['pidfd_failure_frames'][0]['name']='foreign'
    assert fence.pidfd_failed_reports(node) is allowed

@pytest.mark.parametrize('phase',['registry','native'])
@pytest.mark.parametrize('problem',[
 'pending','foreign_probe','pair_created','socketpair_attempt','wrapper_attempt',
 'thread_attempt','signal_change','pair_node','unknown_loop','bound_runtime',
 'unknown_endpoints','unknown_births','pair_creation_pending','native_pair_pending',
 'hooks','origin_depth','no_checkpoint','success_report'])
def test_missing_probe_refuses_changed_or_unknown_custody_even_when_test_failed(monkeypatch,phase,problem):
    node=next(n for p,n in SCOPES if p==phase);item=missing_model(monkeypatch,phase,node)
    if problem=='pending':monkeypatch.setattr(fence,'REGISTRY_PIDFD_PENDING',{})
    elif problem=='foreign_probe':fence.REGISTRY_PIDFD_ROWS['foreign']={}
    elif problem=='pair_created':fence.PAIRS[123]=dict(object=NS(fileno=lambda:7),closed=False)
    elif problem=='socketpair_attempt':fence.COUNTS['socketpair']=1
    elif problem=='wrapper_attempt':fence.COUNTS['pair_wrapping_events']=1
    elif problem=='thread_attempt':fence.COUNTS['threads']=1
    elif problem=='signal_change':fence.COUNTS['sigint_handlers']=1
    elif problem=='pair_node':fence.PAIR_NODES.add(node.split('::')[-1])
    elif problem=='unknown_loop':item['loop']=object()
    elif problem=='bound_runtime':item['bound']=True
    elif problem=='unknown_endpoints':item['endpoints']=(object(),object())
    elif problem=='unknown_births':item['endpoint_births']=((1,2,3,4),)
    elif problem=='pair_creation_pending':fence.LOCAL.registry_pair={}
    elif problem=='native_pair_pending':fence.LOCAL.in_pair=True
    elif problem=='hooks':item['asyncgen_hooks']=object()
    elif problem=='origin_depth':item['coroutine_origin_depth']=-1
    elif problem=='no_checkpoint':item.pop('probe_start')
    elif problem=='success_report':fence.state['reports'][1]['outcome']='passed'
    with pytest.raises(Refused):finish(phase,node)
    assert fence.PIDFD_FAILED_BEFORE_PROBE=={}

@pytest.mark.parametrize('phase',['registry','native','owner'])
@pytest.mark.parametrize('problem',['none','missing','foreign','open','kernel_refused','failed_before_probe','active_scope'])
def test_each_success_scope_still_requires_every_genuine_closed_probe(monkeypatch,phase,problem):
    node=next(n for p,n in SCOPES if p==phase);missing_model(monkeypatch,phase,node)
    for scope_node in fence.pidfd_scope_nodes(phase):
        fence.REGISTRY_PIDFD_ROWS[scope_node]=dict(node=scope_node,phase=phase,closed=True,
            outcome='closed',open_calls=1,close_calls=1,loop_closed=True,hooks_restored=True,fd=19,birth=(19,2,3,4))
    row=fence.REGISTRY_PIDFD_ROWS[node]
    if problem!='active_scope':
        monkeypatch.setattr(fence,'REGISTRY_ITEM',None)
        monkeypatch.setattr(fence,'PIDFD_EXTRA_ITEM',None)
    if problem=='missing':del fence.REGISTRY_PIDFD_ROWS[node]
    elif problem=='foreign':fence.REGISTRY_PIDFD_ROWS['foreign']=dict(row)
    elif problem=='open':row['closed']=False
    elif problem=='kernel_refused':row.update(outcome='kernel_refused',close_calls=0,fd=None,birth=None)
    elif problem=='failed_before_probe':fence.PIDFD_FAILED_BEFORE_PROBE[node]={'closure_claimed':False}
    summary=fence.registry_pidfd_cessation()
    assert summary['complete_expected_probes'] is (problem=='none')
    assert summary['complete_two_probes'] is (phase=='registry' and problem=='none')
    assert summary['all_observed_probes_closed'] is (problem not in ('foreign','open','active_scope'))


def test_fixture_owns_dispatch_traps_without_replacing_direct_product_helpers():
    path=Path(fence.PIDFD_EXTRA_PINS['native'])
    tree=ast.parse(path.read_text());defs={n.name:n for n in tree.body if isinstance(n,ast.FunctionDef)}
    def patches(node):
        return {n.args[1].value for n in ast.walk(node) if isinstance(n,ast.Call)
                and isinstance(n.func,ast.Attribute) and n.func.attr=='setattr'
                and len(n.args)>1 and isinstance(n.args[1],ast.Constant)}
    high={'_ensure_cdp_supervisor','_get_cdp_override'}
    assert high<=patches(defs['blocked_dispatch'])
    assert not high & patches(defs['blocked_browser_policy'])
    assert not high & patches(defs['separate_browser_effects'])
    assert [a.arg for a in defs['separate_browser_effects'].args.args]==['blocked_browser_policy','monkeypatch']
    assert {'_popen_agent_browser','_create_session_for_key','_prepare_session_socket_dir',
            '_start_browser_cleanup_thread','get'}<=patches(defs['blocked_browser_policy'])
    assert {'create_connection','Popen','run','websockets','get_or_start',
            'get_browser_control_broker','_find_cli','_workspace_dir'}<=patches(defs['separate_browser_effects'])
    product=path.parents[2]/'tools'
    for filename,names in {
        'browser_tool_cdp.py':['_resolve_cdp_override','_get_cdp_override','_ensure_cdp_supervisor'],
        'browser_cdp_tool.py':['_browser_cdp_via_supervisor','_cdp_call'],
        'browser_use_cli.py':['_run_cli_killing_process_group','_route_backend','stop_harness_daemons'],
        'browser_vault_tool.py':['_eval_js','_eval_js_secret']}.items():
        functions={n.name:n for n in ast.parse((product/filename).read_text()).body
                   if isinstance(n,(ast.FunctionDef,ast.AsyncFunctionDef))}
        for name in names:
            assert any(isinstance(d,ast.Name) and d.id=='browser_action_entry'
                       for d in functions[name].decorator_list),(filename,name)

@pytest.mark.parametrize('phase',['registry','native','owner'])
@pytest.mark.parametrize('case',['failed_before_probe','failed_open_handle','passed_missing_probe',
                                  'passed_complete','passed_skipped'])
def test_real_runner_finish_preserves_failure_and_requires_complete_success(monkeypatch,phase,case):
    # Execute only the actual final runner block with an inert pytest result and
    # resource records. This is an outcome-propagation model, never live proof.
    import copy
    tree=ast.parse((Path(__file__).parent/'run.py').read_text())
    terminal=tree.body[-1]
    assert isinstance(terminal,ast.Raise)
    blocks=[n for n in tree.body if isinstance(n,ast.Try)]
    assert len(blocks)==1
    rc=1 if case.startswith('failed_') else 0
    complete=case in ('passed_complete','passed_skipped')
    probes=dict(all_observed_probes_closed=case!='failed_open_handle',
                complete_two_probes=complete,complete_expected_probes=complete)
    primitive=dict(unclosed_ipv6=[],unclosed_self_pipe=[],live_threads=[],
        all_thread_count=1,sigint_restored=True,registry_pidfd_probe=probes,
        registry_self_pipe={'complete_two_pairs':complete})
    state=dict(reports=[dict(nodeid='preserved',when='call',outcome=
        'failed' if rc else 'skipped' if case=='passed_skipped' else 'passed')])
    before=copy.deepcopy(state['reports']);snapshots=[];closes=[]
    env=dict(phase=phase,state=state,pytest=NS(main=lambda *a,**k:rc),args=[],Capture=lambda:None,
        snapshot=lambda:snapshots.append(copy.deepcopy(state)),
        fence=NS(closed_primitives=lambda:primitive),save=lambda *a:None,
        REAL_MONOTONIC=lambda:1,c={'deadline_monotonic':10},os=NS(close=closes.append),fd=123)
    expected=SystemExit if case in ('failed_before_probe','passed_complete') else AssertionError
    with pytest.raises(expected) as got:
        exec(compile(ast.Module(body=[blocks[0],terminal],type_ignores=[]),'<runner-finish-model>','exec'),env)
    assert state['reports']==before and state['exit_code']==rc and closes==[123]
    if expected is SystemExit:assert got.value.code==rc
    assert (state['phase_result']=='PASS') is (case=='passed_complete')
    if rc:assert state['phase_result']=='FAIL'
    assert snapshots[0]['exit_code']==rc
