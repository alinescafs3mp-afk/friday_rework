def fixture_probe(tmp_path, monkeypatch):
    api = module()
    parent = tmp_path / 'pinned-parent'
    parent.mkdir()
    leaf = parent / 'python314.zip'
    chain = list(reversed(parent.parents)) + [parent]
    m = {'declared_absent_path': str(leaf),
         'ancestors': [{'path': str(p), 'identity': api.identity(p.lstat())} for p in chain]}
    monkeypatch.setattr(api, 'ARCHIVE', str(leaf))
    return api, m, leaf


def test_same_genuine_kernel_enoent_and_no_content_open(tmp_path, monkeypatch):
    api, m, leaf = fixture_probe(tmp_path, monkeypatch)
    actual_stat, actual_open = api._stat, api._open
    saved = []
    flags_seen = []
    def metadata_stat(*args, **kwargs):
        try:
            return actual_stat(*args, **kwargs)
        except FileNotFoundError as error:
            saved.append(error)
            raise
    def metadata_open(*args, **kwargs):
        flags_seen.append(args[1])
        assert args[1] == os.O_PATH | os.O_DIRECTORY | os.O_NOFOLLOW | os.O_CLOEXEC
        return actual_open(*args, **kwargs)
    monkeypatch.setattr(api, '_stat', metadata_stat)
    monkeypatch.setattr(api, '_open', metadata_open)
    with pytest.raises(FileNotFoundError) as result:
        api.failure_only_open(m, str(leaf), os.O_RDONLY, str(leaf))
    assert result.value is saved[0]
    assert result.value.errno == errno.ENOENT
    assert result.value.filename == leaf.name
    assert len(flags_seen) == len(m['ancestors'])
    assert not leaf.exists()

def test_distribution_same_kernel_enoent_and_metadata_descriptors_only(tmp_path, monkeypatch):
    api, m, leaf = distribution_probe(tmp_path, monkeypatch)
    real_stat, real_open = api._stat, api._open

    def observed_stat(*args, **kwargs):
        try: return real_stat(*args, **kwargs)
        except FileNotFoundError as error:
            saved.append(error)
            raise
    def observed_open(*args, **kwargs):
        flags.append(args[1])
        assert args[1] == os.O_PATH | os.O_DIRECTORY | os.O_NOFOLLOW | os.O_CLOEXEC

    assert len(flags) == len(m['distribution_metadata_probes']['roots'][str(leaf.parent)])

@pytest.mark.parametrize('filename', ['entry_points.txt', 'METADATA', 'PKG-INFO', 'RECORD',
    'installed-files.txt', 'SOURCES.txt', 'requires.txt', 'direct_url.json', 'top_level.txt'])
def test_distribution_real_runtime_query_provenance_and_direct_caller_negative(tmp_path, monkeypatch, filename):
    import importlib.metadata as metadata
    api = module(); root = tmp_path / 'example.dist-info'; root.mkdir()
    dist = metadata.PathDistribution(root)
    m = {'distribution_metadata_probes': {'caller_source': metadata.__file__}}
    seen = []
    def observed_read(path, *args, **kwargs):
        if path.name == filename:
            seen.append(api.distribution_metadata_caller(m, str(path)))
        if path.name == 'METADATA' and filename == 'PKG-INFO': return ''
        if path.name in ('METADATA', 'PKG-INFO'): return 'Name: example\nVersion: 1\n'
        if path.name == 'direct_url.json': return '{}'
        return ''
    monkeypatch.setattr(pathlib.Path, 'read_text', observed_read)
    method = api.METADATA_QUERIES[filename]
    if method == '_top_level_declared': metadata._top_level_declared(dist)
    elif method == '_load_json': dist._load_json(filename)
    elif method in ('metadata', 'entry_points'): getattr(dist, method)
    else: getattr(dist, method)()
    assert seen == [True]
    seen.clear(); dist.read_text(filename)
    assert seen == [False]

def test_distribution_failure_closes_every_metadata_descriptor(tmp_path, monkeypatch):
    api, m, leaf = distribution_probe(tmp_path, monkeypatch)
    opened, closed = [], []; real_open, real_close = api._open, api._close
    def tracked_open(*args, **kwargs):
        fd = real_open(*args, **kwargs); opened.append(fd); return fd
    def tracked_close(fd):
        closed.append(fd); real_close(fd)
    monkeypatch.setattr(api, '_open', tracked_open); monkeypatch.setattr(api, '_close', tracked_close)
    leaf.write_bytes(b'present')
    with pytest.raises(api.ProbeRefusal):
        api.failure_only_metadata_open(m, str(leaf), 0, str(leaf))
    assert opened and closed == list(reversed(opened))
