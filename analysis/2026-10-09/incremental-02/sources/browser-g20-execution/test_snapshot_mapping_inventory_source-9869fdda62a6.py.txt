"""SOURCE ONLY / NOT_RUN. Extract only snapshot after independent admission.

No candidate module import, guard install, mapped-file read or native effect.
The five cases exercise exact snapshot source with finite supplied maps text.
"""
import ast
from pathlib import Path
import pytest

SOURCE = Path(__file__).resolve().parent / 'candidate' / 'run.py'

def snapshot_fixture(mapping_paths, *, denied=None, origins_denied=False):
    tree = ast.parse(SOURCE.read_text())
    nodes = [n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == 'snapshot']
    assert len(nodes) == 1
    events, saved = [], {}
    text = '\n'.join('1000-2000 r--p 0 00:00 0 ' + p for p in mapping_paths)
    text += '\n2000-3000 rw-p 0 00:00 0 [heap]\n3000-4000 rw-p 0 00:00 0\n'
    class Maps:
        def __init__(self, path):
            assert path == '/proc/self/maps'
        def read_text(self):
            events.append(('maps-read', None))
            return text
    def save(name, value):
        events.append(('save', name))
        saved[name] = value.copy()
    def sha(path):
        assert 'loaded-mapping-paths.json' in saved
        events.append(('sha', path))
        if path == denied:
            raise PermissionError('UNCHANGED_EXACT_READ_REFUSAL')
        return 'sha:' + path
    def origins():
        assert 'loaded-mapping-paths.json' in saved
        events.append(('origins', None))
        if origins_denied:
            raise PermissionError('MODULE_HASH_REFUSED')
        return {}
    env = {'Path': Maps, 'save': save, 'sha': sha, 'origins': origins, 'state': {'phase': 'controls'}}
    exec(compile(ast.Module(body=nodes, type_ignores=[]), '<reviewed-snapshot-control>', 'exec'), env)
    return env['snapshot'], events, saved

def test_complete_paths_saved_before_first_hash_and_one_maps_read():
    paths = ['/first', '/second file', '/third', '/first']
    snapshot, events, saved = snapshot_fixture(paths)
    snapshot()
    assert saved['loaded-mapping-paths.json'] == paths
    assert saved['loaded-mappings.json'] == {p: 'sha:' + p for p in paths}
    assert events.count(('maps-read', None)) == 1
    assert events.index(('save', 'loaded-mapping-paths.json')) < events.index(('origins', None))
    assert events.index(('save', 'loaded-mapping-paths.json')) < events.index(('sha', '/first'))

def test_unknown_first_mapping_refuses_without_losing_later_paths():
    paths = ['/unknown', '/later-known', '/later-unknown']
    snapshot, events, saved = snapshot_fixture(paths, denied='/unknown')
    with pytest.raises(PermissionError, match='UNCHANGED_EXACT_READ_REFUSAL'):
        snapshot()
    assert saved['loaded-mapping-paths.json'] == paths
    assert [e for e in events if e[0] == 'sha'] == [('sha', '/unknown')]
    assert 'loaded-mappings.json' not in saved

def test_deleted_mapping_still_refuses_after_inventory_before_hash():
    paths = ['/gone (deleted)', '/later']
    snapshot, events, saved = snapshot_fixture(paths)
    with pytest.raises(AssertionError, match='DELETED_MAPPING'):
        snapshot()
    assert saved['loaded-mapping-paths.json'] == paths
    assert not [e for e in events if e[0] == 'sha']
    assert 'loaded-mappings.json' not in saved

def test_module_content_refusal_also_preserves_complete_mapping_inventory():
    paths = ['/first', '/last']
    snapshot, events, saved = snapshot_fixture(paths, origins_denied=True)
    with pytest.raises(PermissionError, match='MODULE_HASH_REFUSED'):
        snapshot()
    assert saved['loaded-mapping-paths.json'] == paths
    assert not [e for e in events if e[0] == 'sha']

def test_inventory_has_no_authority_mutation_and_empty_inventory_still_written():
    tree = ast.parse(SOURCE.read_text())
    node = next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == 'snapshot')
    assert not {'pins', 'read_paths', 'fence', 'install'} & {n.id for n in ast.walk(node) if isinstance(n, ast.Name)}
    snapshot, events, saved = snapshot_fixture([])
    snapshot()
    assert saved['loaded-mapping-paths.json'] == []
    assert saved['loaded-mappings.json'] == {}
    assert events.count(('maps-read', None)) == 1
