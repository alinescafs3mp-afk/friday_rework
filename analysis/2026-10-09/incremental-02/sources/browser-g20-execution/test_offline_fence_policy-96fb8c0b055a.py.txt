from pathlib import Path
import os
import socket
import pytest
import private_effect_fence as fence

@pytest.mark.parametrize('family,kind,proto,provenance,count', [
    (socket.AF_INET,socket.SOCK_STREAM,0,True,0),
    (socket.AF_INET6,socket.SOCK_DGRAM,0,True,0),
    (socket.AF_INET6,socket.SOCK_STREAM,6,True,0),
    (socket.AF_INET6,socket.SOCK_STREAM,0,False,0),
    (socket.AF_INET6,socket.SOCK_STREAM,0,True,1),
])
def test_ipv6_probe_foreign_or_duplicate_refused(family,kind,proto,provenance,count):
    assert not fence.ipv6_request_allowed(family,kind,proto,provenance,count)

def test_ipv6_original_probe_positive():
    assert fence.ipv6_request_allowed(socket.AF_INET6,socket.SOCK_STREAM,0,True,0)
    assert fence.ipv6_bind_allowed(('::1',0),True,False)

@pytest.mark.parametrize('address,registered,bound', [
    (('::',0),True,False), (('::1',1),True,False),
    (('127.0.0.1',0),True,False), (('::1',0,0,0),True,False),
    (('::1',0),False,False), (('::1',0),True,True),
])
def test_nonexact_bind_refused(address,registered,bound):
    assert not fence.ipv6_bind_allowed(address,registered,bound)

def test_pathname_fd_and_symlink_boundary(tmp_path):
    inside=tmp_path/'file';inside.write_text('synthetic fixture')
    outside=Path(fence.BASE)/'native/tools/browser_tool.py'
    link=tmp_path/'external';link.symlink_to(outside)
    assert fence.path_ok(inside)
    assert not fence.path_ok(link)
    assert not fence.path_ok(outside)
    assert not fence.path_ok('/etc/passwd')
    assert not fence.path_ok('<LOCAL_HOME>/.ssh/id_ed25519')
    # Resolve the symlink before '..': lexical collapse would incorrectly admit.
    directory_link=tmp_path/'directory';directory_link.symlink_to('/etc')
    assert not fence.path_ok(directory_link/'..'/'passwd')
    fd=os.open(inside,os.O_RDONLY)
    try:assert fence.path_ok(fd)
    finally:os.close(fd)
    fd=os.open(tmp_path,os.O_RDONLY|os.O_DIRECTORY)
    try:
        assert fence.path_ok('file',fd)
        assert not fence.path_ok('external',fd)
        assert not fence.path_ok('../../../../../../etc/passwd',fd)
    finally:os.close(fd)

def test_every_external_process_or_signal_path_is_denied_source():
    assert {'subprocess.Popen','os.system','os.fork','os.forkpty','os.posix_spawn','os.exec','os.kill','os.killpg','os.pidfd_open','signal.pthread_kill','signal.pidfd_send_signal'} <= fence.PROCESS_EVENTS
