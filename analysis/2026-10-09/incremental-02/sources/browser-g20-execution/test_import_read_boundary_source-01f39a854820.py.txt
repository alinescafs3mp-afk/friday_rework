ROOT=Path(__file__).parent
PUBLIC={'/proc/1/cgroup','/proc/version','/proc/stat','/proc/self/mountinfo'}
DENIED={'/proc','/proc/1','/proc/statx','/proc/stat/extra','/proc/2/cgroup',
        '/proc/1/environ','/proc/1/cmdline','/proc/1/mem','/proc/1/stat',

class Refused(RuntimeError):pass

def rejected(reason):raise Refused(reason)

def functions(*names, **values):
    tree=ast.parse((ROOT/'private_effect_fence.py').read_text())
    nodes=[n for n in tree.body if isinstance(n,ast.FunctionDef) and n.name in names]
    assert len(nodes)==len(names)
    env={'os':os,'directory_sync_request':lambda path:False,**values}
    exec(compile(ast.Module(body=nodes,type_ignores=[]),'<exact-fence-source>','exec'),env)
    return SimpleNamespace(**{name:env[name] for name in names})

def test_exact_four_public_files_only():
    tree=ast.parse((ROOT/'private_effect_fence.py').read_text())
    values=[ast.literal_eval(n.value) for n in tree.body if isinstance(n,ast.Assign)
            and any(isinstance(t,ast.Name) and t.id=='PUBLIC_HOST_READS' for t in n.targets)]
    assert values==[PUBLIC]
    api=functions('public_read_allowed',PUBLIC_READABLE=PUBLIC)
    assert all(api.public_read_allowed(p) for p in PUBLIC)
    assert not any(api.public_read_allowed(p) for p in DENIED)

def test_self_mountinfo_never_grants_foreign_pid():
    own='/proc/4321/mountinfo'
    api=functions('public_read_allowed',PUBLIC_READABLE=(PUBLIC-{'/proc/self/mountinfo'})|{own})
    assert api.public_read_allowed(own)
    assert not api.public_read_allowed('/proc/4322/mountinfo')
    assert not api.public_read_allowed('/proc/4321/environ')

@pytest.mark.parametrize('write_flag',[os.O_WRONLY,os.O_RDWR,os.O_CREAT,os.O_TRUNC,os.O_APPEND])
def test_public_reads_do_not_bypass_write_policy(write_flag):
    seen=[]
    def fail(*args,**kwargs):raise Refused('outside private write root')
    api=functions('guarded_open',check_import_open=lambda *a:None,WRITE_FLAGS=write_flag,
                  check_path=fail,read_ok=lambda *a:pytest.fail('write routed to read allowlist'),
                  LOCAL=SimpleNamespace(),string_path=str,RAW={'open':lambda *a,**k:seen.append(a)})
    for path in PUBLIC:
        with pytest.raises(Refused):api.guarded_open(path,write_flag)
    assert seen==[]

def import_wrapper(failure):
    state={}; local=SimpleNamespace(absent_import_probe=False)
    helper=SimpleNamespace(ARCHIVE='/exact/python314.zip',failure_only_open=failure,ProbeRefusal=Refused)
    api=functions('check_import_open',fd_path=lambda p:('/exact/python314.zip',None),
                  string_path=str,canonical=lambda p:p,RAW={'getcwd':lambda:'/cwd'},
                  LOCAL=local,import_path=helper,IMPORT_MANIFEST={},state=state,deny=rejected)
    return api,state,local

def test_original_kernel_exception_propagates_and_content_open_unreachable():
    actual=FileNotFoundError(errno.ENOENT,'missing','python314.zip')
    def missing(*a):raise actual
    api,state,local=import_wrapper(missing)
    with pytest.raises(FileNotFoundError) as got:api.check_import_open('/exact/python314.zip',os.O_RDONLY)
    assert got.value is actual
    assert state['absent_import_enoent_count']==1
    assert state['last_absent_import_probe']['content_open'] is False
    assert local.absent_import_probe is False

def test_helper_refusal_remains_hard_refusal_and_clears_reentrancy():
    def refused(*a):raise Refused('DECLARED_ABSENT_LEAF_PRESENT')
    api,state,local=import_wrapper(refused)
    with pytest.raises(Refused,match='ABSENT_IMPORT_PROBE_REFUSED'):api.check_import_open('/exact/python314.zip',0)
    assert not state and local.absent_import_probe is False

def test_dirfd_anchor_reaches_exact_helper():
    seen=[]
    api,state,local=import_wrapper(lambda *a:seen.append(a))
    api.check_import_open('python314.zip',0,42)
    assert seen[0][1:]==('/proc/self/fd/42/python314.zip',0,'/proc/self/fd/42/python314.zip')
    assert not state and local.absent_import_probe is False

def test_process_socket_thread_and_sigint_guards_preserved():
    prior=Path('<ASTRA_WORK>/browser-owner-controls-preparation/private_effect_fence.py')
    def nodes(path):
        return {n.name:n for n in ast.parse(path.read_text()).body
                if isinstance(n,ast.FunctionDef)}
    def dump(node):
        return ast.dump(node,include_attributes=False)
    old,new=nodes(prior),nodes(ROOT/'private_effect_fence.py')
    for name in ('guarded_bind','guarded_thread_start','guarded_joinable','guarded_signal',
                 'check_path','check_fd','path_ok','fd_path','canonical'):
        assert dump(new[name])==dump(old[name]),name

    # Reconstruct each complete function from the legacy AST and only the exact
    # registry additions. Equality also rejects altered signatures, legacy guards,
    # extra branches and reordered checks; no registry branch is stripped away.
    additions={
        'guarded_close':(
            (0,"entry=PAIRS.get(id(obj))\n"
               "if entry and 'registry_owner' in entry:registry_endpoint_check(obj,'_close_self_pipe')"),
            (3,"if entry and 'registry_owner' in entry and obj.fileno()!=-1:registry_refuse('socket.close:registry_endpoint_not_closed','close')")),
        'guarded_socketpair':(
            (1,"if state['phase']=='registry':return registry_create_pair(family,type,proto)"),),
        'guarded_send':(
            (0,"entry=PAIRS.get(id(obj))\n"
               "if entry and 'registry_owner' in entry:\n"
               " registry_endpoint_check(obj,'_write_to_self')\n"
               " if type(data) is not bytes or data!=b'\\0' or args or kwargs:deny('socket.send:registry_self_pipe_payload')\n"
               " row=REGISTRY_ROWS[entry['registry_owner']]\n"
               " if sum(e['send'] for e in row['endpoints'])>=16:deny('socket.send:registry_pair_budget')\n"
               " entry['send']+=1;COUNTS['self_send']+=1\n"
               " return SOCKET_SEND(obj,data)"),),
        'guarded_recv':(
            (0,"entry=PAIRS.get(id(obj))\n"
               "if entry and 'registry_owner' in entry:\n"
               " registry_endpoint_check(obj,'_read_from_self')\n"
               " if type(size) is not int or size!=4096 or args or kwargs:deny('socket.recv:registry_self_pipe_payload')\n"
               " row=REGISTRY_ROWS[entry['registry_owner']]\n"
               " if sum(e['recv'] for e in row['endpoints'])>=32:deny('socket.recv:registry_pair_budget')\n"
               " entry['recv']+=1;COUNTS['self_recv']+=1\n"
               " return SOCKET_RECV(obj,size)"),),
    }
    for name,insertions in additions.items():
        expected=old[name]
        for index,source in insertions:
            expected.body[index:index]=ast.parse(source).body
        assert dump(new[name])==dump(expected),name

    expected=old['closed_primitives']
    summary=expected.body[0].value
    summary.keys.insert(0,ast.Constant(value='registry_self_pipe'))
    summary.values.insert(0,ast.parse('registry_cessation()',mode='eval').body)
    summary.keys.insert(0,ast.Constant(value='registry_pidfd_probe'))

    summary.keys.insert(0,ast.Constant(value='directory_sync'))
    summary.values.insert(0,ast.parse('directory_sync_cessation()',mode='eval').body)
    assert dump(new['closed_primitives'])==dump(expected),'closed_primitives'
