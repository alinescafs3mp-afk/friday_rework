kernel denial or real endpoint cessation. The unchanged seven registry behavior
cases must later prove the actual two pairs/four closed endpoints under the fence.

@pytest.mark.parametrize('kind',['signal','thread','executor','process','ctypes'])
def test_registry_separate_pair_admission_never_grants_escalation(monkeypatch,kind):
    monkeypatch.setattr(fence,'state',{'phase':'registry','current_node':NODES[0]})
    monkeypatch.setattr(fence,'deny',refusal)
    calls=[]
    monkeypatch.setattr(fence,'SIGNAL',lambda *a: calls.append(a))
    monkeypatch.setattr(fence,'THREAD_START',lambda *a: calls.append(a))
    assert not fence.async_active()
    if kind=='signal':
        with pytest.raises(Refused,match='outside_asyncio_own_SIGINT_handler'):
            fence.guarded_signal(fence.signal.SIGINT,fence.signal.default_int_handler)
    elif kind in ('thread','executor'):
        target=(lambda: None)
        # Thread construction is inert; start is guarded and never reaches RAW.
        thread=threading.Thread(target=target,name='foreign-executor' if kind=='executor' else 'foreign')
        with pytest.raises(Refused,match='outside_four_required_threads'):
            fence.guarded_thread_start(thread)
    else:
        with pytest.raises(Refused,match='subprocess.Popen|ctypes.dlopen'):
            fence.audit('subprocess.Popen' if kind=='process' else 'ctypes.dlopen',())
    assert calls==[]

class ModelRaw:
    def __init__(self,fd):self.fd=fd
    def fileno(self):return self.fd

class ModelPython(ModelRaw):
    pass

class ModelFrame:
    def __init__(self,code,values,next_frame=None):
        self.f_code=code;self.f_globals=socket.__dict__;self.f_locals=values;self.f_back=next_frame

def model_birth(fd):
    return (fd,1,100+fd,fence.stat.S_IFSOCK|0o600) if type(fd) is int and fd>=0 else None

def model_setup(monkeypatch):
    node=NODES[0];loop=object()
    monkeypatch.setattr(fence,'state',{'phase':'registry','current_node':node})
    monkeypatch.setattr(fence,'REGISTRY_ITEM',{'node':node,'path':FIXTURE})
    monkeypatch.setattr(fence,'REGISTRY_PINS',{'fixture':FIXTURE})
    monkeypatch.setattr(fence,'REGISTRY_NATIVE_BOUND',True)
    monkeypatch.setattr(fence,'REGISTRY_ROWS',{})
    monkeypatch.setattr(fence,'PAIRS',{})
    monkeypatch.setattr(fence,'IPV6',{})
    monkeypatch.setattr(fence,'COUNTS',dict.fromkeys(fence.COUNTS,0))
    monkeypatch.setattr(fence,'registry_stack',lambda *a,**kw:loop)
    monkeypatch.setattr(fence,'deny',refusal)
    monkeypatch.setattr(fence,'REGISTRY_RAW_SOCKET_CLASS',ModelRaw)
    monkeypatch.setattr(fence,'SOCKET_CLASS',ModelPython)
    monkeypatch.setattr(fence,'registry_fd_identity',model_birth)
    monkeypatch.setattr(fence.LOCAL,'registry_pair',None,raising=False)
    return node,loop

@pytest.mark.parametrize('problem',[
    'none','pair_frame','init_frame','self','raw_type','raw_duplicate',
    'first_raw_attached','fd_bool','fd_negative','same_fd','nonsocket','current_node',
    'owner_loop','foreign_object','duplicate_object','third_wrapper',
])
def test_registry_first_audit_handoff_refuses_before_charge(monkeypatch,problem):
    node,loop=model_setup(monkeypatch)
    raw0,raw1=ModelRaw(-1),ModelRaw(11);obj=ModelPython(-1)
    pending={'node':node,'loop':loop,'objects':[],'raw':(),'births':()}
    pair_frame=ModelFrame(fence.SOCKET_PAIR.__code__,{'a':raw0,'b':raw1})
    init_frame=ModelFrame(fence.REGISTRY_SOCKET_INIT.__code__,{'self':obj,'fileno':10},pair_frame)
    monkeypatch.setattr(fence.LOCAL,'registry_pair',pending)
    monkeypatch.setattr(fence,'READ_GETFRAME',lambda depth:init_frame)
    if problem=='none':monkeypatch.setattr(fence.LOCAL,'registry_pair',None)
    elif problem=='pair_frame':pair_frame.f_code=(lambda:None).__code__
    elif problem=='init_frame':init_frame.f_code=(lambda:None).__code__
    elif problem=='self':init_frame.f_locals['self']=object()
    elif problem=='raw_type':pair_frame.f_locals['a']=obj
    elif problem=='raw_duplicate':pair_frame.f_locals['b']=raw0
    elif problem=='first_raw_attached':raw0.fd=10
    elif problem=='fd_bool':init_frame.f_locals['fileno']=True
    elif problem=='fd_negative':init_frame.f_locals['fileno']=-1
    elif problem=='same_fd':raw1.fd=10
    elif problem=='nonsocket':monkeypatch.setattr(fence,'registry_fd_identity',lambda fd:(fd,1,fd,fence.stat.S_IFREG))
    elif problem=='current_node':pending['node']=NODES[1]
    elif problem=='owner_loop':pending['loop']=object()
    elif problem=='foreign_object':obj=raw0
    elif problem=='duplicate_object':pending['objects']=[obj]
    elif problem=='third_wrapper':pending['objects']=[ModelPython(10),ModelPython(11)]
    before=list(pending['objects'])
    with pytest.raises(Refused,match='registry_foreign_native_wrapper|registry_handoff_identity'):
        fence.registry_audit_wrapper(obj,socket.AF_UNIX,socket.SOCK_STREAM,0)
    assert pending['objects']==before
    assert fence.COUNTS['pair_wrapping_events']==0
    assert fence.REGISTRY_ROWS=={} and fence.PAIRS=={}
    assert fence.state['registry_wrapper_diagnostic']['schema']=='friday.registry-wrapper-refusal.v1'

@pytest.mark.parametrize('problem',['none','first_wrapper','second_raw','raw_attached','first_birth','second_birth'])
def test_registry_second_handoff_and_exact_two_audits_model(monkeypatch,problem):
    node,loop=model_setup(monkeypatch)
    raw0,raw1=ModelRaw(-1),ModelRaw(11);first,second=ModelPython(-1),ModelPython(-1)
    pending={'node':node,'loop':loop,'objects':[],'raw':(),'births':()}
    pair_frame=ModelFrame(fence.SOCKET_PAIR.__code__,{'a':raw0,'b':raw1})
    init_frame=ModelFrame(fence.REGISTRY_SOCKET_INIT.__code__,{'self':first,'fileno':10},pair_frame)
    monkeypatch.setattr(fence.LOCAL,'registry_pair',pending)
    monkeypatch.setattr(fence,'READ_GETFRAME',lambda depth:init_frame)
    fence.registry_audit_wrapper(first,socket.AF_UNIX,socket.SOCK_STREAM,0)
    assert pending['raw']==(raw0,raw1) and pending['births']==(model_birth(10),model_birth(11))
    first.fd=10;raw1.fd=-1
    pair_frame.f_locals['a']=first
    init_frame.f_locals={'self':second,'fileno':11}
    if problem=='first_wrapper':pair_frame.f_locals['a']=ModelPython(10)
    elif problem=='second_raw':pair_frame.f_locals['b']=ModelRaw(-1)
    elif problem=='raw_attached':raw1.fd=11
    elif problem=='first_birth':first.fd=12
    elif problem=='second_birth':init_frame.f_locals['fileno']=12

        fence.registry_audit_wrapper(second,socket.AF_UNIX,socket.SOCK_STREAM,0)
        assert pending['objects']==[first,second] and fence.COUNTS['pair_wrapping_events']==2

        with pytest.raises(Refused,match='registry_handoff_identity'):
            fence.registry_audit_wrapper(second,socket.AF_UNIX,socket.SOCK_STREAM,0)
        assert pending['objects']==[first] and fence.COUNTS['pair_wrapping_events']==1

@pytest.mark.parametrize('problem',[
    'none','raise_before','raise_partial','not_tuple','missing_return','missing_audit',
    'extra_audit','wrong_first','wrong_second','same_wrapper','raw_attached',
    'lost_birth','reused_fd','changed_inode',
])
def test_registry_pair_return_failure_never_publishes_or_retries_model(monkeypatch,problem):
    node,loop=model_setup(monkeypatch);calls=[]
    first,second=ModelPython(10),ModelPython(11);raw0,raw1=ModelRaw(-1),ModelRaw(-1)
    def native_pair(*args):
        calls.append(args);pending=fence.LOCAL.registry_pair
        if problem=='raise_before':raise OSError('model')
        pending.update(objects=[first,second],raw=(raw0,raw1),births=(model_birth(10),model_birth(11)))
        if problem=='raise_partial':
            pending['objects']=[first];raise OSError('model')
        if problem=='not_tuple':return [first,second]
        if problem=='missing_return':return (first,)
        if problem=='missing_audit':pending['objects']=[first]
        if problem=='extra_audit':pending['objects'].append(ModelPython(12))
        if problem=='wrong_first':return (ModelPython(10),second)
        if problem=='wrong_second':return (first,ModelPython(11))
        if problem=='same_wrapper':pending['objects']=[first,first];return (first,first)
        if problem=='raw_attached':raw1.fd=11
        if problem=='lost_birth':second.fd=-1
        if problem=='reused_fd':second.fd=10
        if problem=='changed_inode':pending['births']=(model_birth(10),(11,1,999,fence.stat.S_IFSOCK|0o600))
        return first,second
    monkeypatch.setattr(fence,'SOCKET_PAIR',native_pair)
    if problem=='none':
        assert fence.registry_create_pair(socket.AF_UNIX,socket.SOCK_STREAM,0)==(first,second)
        assert fence.COUNTS['socketpair']==1 and len(fence.PAIRS)==2
        assert fence.REGISTRY_ROWS[node]['wrapper_count']==2
    else:
        with pytest.raises(Refused,match='registry_creation_failed'):
            fence.registry_create_pair(socket.AF_UNIX,socket.SOCK_STREAM,0)
        assert fence.COUNTS['socketpair']==0 and fence.PAIRS=={} and fence.REGISTRY_ROWS=={}

    other={'object':second,'closed':False,'registry_owner':node,'birth':model_birth(11),'send':0,'recv':0}
    row={'node':node,'loop':loop,'wrapper_count':2,'endpoints':[entry,other]}
    fence.REGISTRY_ROWS[node]=row;fence.PAIRS[id(first)]=entry;fence.PAIRS[id(second)]=other
    monkeypatch.setattr(fence,'registry_stack',lambda *a,**kw:loop)
    calls=[]
    def close(obj):calls.append(obj);obj.fd=-1
    monkeypatch.setattr(fence,'SOCKET_CLOSE',close)
    if problem=='wrong_owner':entry['registry_owner']=NODES[1]
    elif problem=='wrong_object':entry['object']=ModelPython(10)
    elif problem=='closed':entry['closed']=True
    elif problem=='reused_fd':first.fd=12

    elif problem=='foreign_frame':monkeypatch.setattr(fence,'registry_stack',lambda *a,**kw:None)
    if problem=='none':
        fence.guarded_close(first)
        assert calls==[first] and entry['closed'] is True and first.fileno()==-1
    else:

def test_registry_close_without_actual_fd_cessation_never_marks_closed(monkeypatch):
    model_setup(monkeypatch);obj=ModelPython(10);calls=[]
    entry={'object':obj,'closed':False,'registry_owner':NODES[0]}
    fence.PAIRS[id(obj)]=entry

    with pytest.raises(Refused,match='registry_endpoint_not_closed'):
