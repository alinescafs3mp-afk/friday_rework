import ast

import pytest

ROOT = Path(__file__).parent


def functions(*names, **values):
    tree = ast.parse((ROOT / 'private_effect_fence.py').read_text())
    nodes = [n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name in names]
    assert len(nodes) == len(names)
    env = {'json': json, 'sys': sys, 'READ_GETFRAME': sys._getframe,
           'READ_FRAME_TYPE': type(sys._getframe()),
           'READ_DIAGNOSTIC_MAX_BYTES': 65536, 'READ_DIAGNOSTIC_MAX_FRAMES': 8,
           'READ_DIAGNOSTIC_PATH_CHARS': 512, 'READ_DIAGNOSTIC_MAX_WRITES': 8,
           **values}
    exec(compile(ast.Module(body=nodes, type_ignores=[]), '<reviewed-diagnostic-control>', 'exec'), env)
    return SimpleNamespace(**{name: env[name] for name in names})


def diagnostics(**values):
    return functions('diagnostic_text', 'read_diagnostic', 'safe_read_diagnostic', **values)


class CallbackTrap:
    def __fspath__(self):
        pytest.fail('diagnostic invoked fspath')

    def __str__(self):
        pytest.fail('diagnostic invoked str')

    def __repr__(self):
        pytest.fail('diagnostic invoked repr')


def test_original_path_and_native_stack_metadata_without_source_or_locals():
    api = diagnostics()
    secret_local = 'PRIVATE_LOCAL_MUST_NOT_ENTER_METADATA'
    got = api.read_diagnostic('../outside', '/private/../outside', '/outside', 'authority_decision')
    assert got['raw_path']['value'] == '../outside'
    assert got['evaluated_path']['value'] == '/private/../outside'
    assert got['canonical_path']['value'] == '/outside'
    assert got['resolution_stage'] == 'authority_decision'
    assert any(row['function']['value'] == 'test_original_path_and_native_stack_metadata_without_source_or_locals'
               for row in got['frames'])
    assert secret_local not in json.dumps(got)
    assert all(set(row) == {'filename', 'function', 'line'} for row in got['frames'])
    assert all(type(row['line']) is int for row in got['frames'])

@pytest.mark.parametrize('raw', ['\U0001f600' * 10000, b'\xff' * 10000])

def test_diagnostic_has_no_syscalls_or_arbitrary_frame_access():
    syscall_calls = []
    frame_callbacks = []

    def never(*args, **kwargs):
        syscall_calls.append((args, kwargs))
        pytest.fail('extra diagnostic file/FD read')

    class FrameTrap:
        def __getattribute__(self, key):
            frame_callbacks.append(key)
            pytest.fail('arbitrary frame callback')

    api = diagnostics(RAW={name: never for name in ('open', 'read', 'stat', 'readlink', 'getcwd')},
                      READ_GETFRAME=lambda depth: FrameTrap())
    assert api.safe_read_diagnostic('/outside', '/outside', '/outside', 'authority_decision')['status'] == 'UNAVAILABLE'
    assert syscall_calls == [] and frame_callbacks == []

class Refusal(Exception):
    pass


def read_api(canonical=lambda text: text, diagnostic=lambda *args: {'status': 'CONTROL_METADATA'}):
    seen = []

    def denied(event, **kwargs):
        seen.append((event, kwargs))
        raise Refusal(event)

    api = functions('read_ok', string_path=lambda value: value, canonical=canonical,
                    READABLE={'/pinned'}, writable=lambda value: value.startswith('/private/'),
                    public_read_allowed=lambda value: value == '/public',
                    RAW={'getcwd': lambda: '/private'}, fd_path=lambda fd: ('/dirfd', None),
                    safe_read_diagnostic=diagnostic, deny=denied)
    return api, seen


@pytest.mark.parametrize('path', ['/pinned', '/private/file', '/dev/null', '/public', 42])
def test_original_positive_decisions_never_invoke_diagnostics(path):
    def forbidden(*args):
        pytest.fail('diagnostics on an allowed action')

    api, seen = read_api(diagnostic=forbidden)
    assert api.read_ok(path) is None and seen == []


@pytest.mark.parametrize('path', ['/etc/shadow', '/private/../outside', '/public/extra'])
def test_refused_decision_never_becomes_permission(path):
    api, seen = read_api(canonical=lambda text: '/outside' if '..' in text else text)
    with pytest.raises(Refusal, match='outside_pinned_or_private_read_roots'):
        api.read_ok(path)
    assert len(seen) == 1 and seen[0][1]['diagnostic']['status'] == 'CONTROL_METADATA'


def test_resolution_failure_retains_already_available_path_and_unknown_canonical():
    captured = []

    def failed(text):
        raise OSError('fixture')

        captured.append(args)
        return {'status': 'UNAVAILABLE'}

    api, seen = read_api(canonical=failed, diagnostic=diagnostic)
    with pytest.raises(Refusal):
        api.read_ok('leaf', dir_fd=42)

    assert len(seen) == 1


def test_read_denial_precedes_both_raw_open_and_builtin_audit_return():
    raw_calls = []

    def refused(*args, **kwargs):
        raise Refusal('original refusal')

    local = SimpleNamespace(absent_import_probe=False, checked_open=None)
    api = functions('guarded_open', 'audit', RAW={'open': lambda *a, **k: raw_calls.append(a)},
                    check_import_open=lambda *a: None, WRITE_FLAGS=os.O_WRONLY | os.O_RDWR,
                    read_ok=refused, LOCAL=local, string_path=lambda p: p, PROCESS_EVENTS=set())
    with pytest.raises(Refusal):
        api.guarded_open('/outside', os.O_RDONLY)
    with pytest.raises(Refusal):
        api.audit('open', ('/outside', 'r', os.O_RDONLY))
    assert raw_calls == [] and local.checked_open is None


def test_diagnostic_failure_in_actual_read_gate_keeps_raw_open_unreachable():
    raw_calls = []
    denials = []

    def broken(depth):
        raise OSError('synthetic diagnostic failure')

    def denied(event, **kwargs):
        denials.append((event, kwargs))
        raise Refusal(event)

    api = functions('diagnostic_text', 'read_diagnostic', 'safe_read_diagnostic', 'read_ok', 'guarded_open',
                    READ_GETFRAME=broken, READABLE=set(), writable=lambda p: False,
                    public_read_allowed=lambda p: False, string_path=lambda p: p,
                    canonical=lambda p: p, check_import_open=lambda *a: None,
                    WRITE_FLAGS=os.O_WRONLY | os.O_RDWR, LOCAL=SimpleNamespace(checked_open=None),
                    RAW={'open': lambda *a, **k: raw_calls.append(a)}, deny=denied)
    with pytest.raises(Refusal):
        api.guarded_open('/outside', os.O_RDONLY)
    assert raw_calls == [] and len(denials) == 1
    assert denials[0][1]['diagnostic'] == {
        'schema': 'friday.offline.read-refusal-metadata.v1', 'status': 'UNAVAILABLE'}


class Terminal(BaseException):
    pass


@pytest.mark.parametrize('write_behavior', ['full', 'one_byte', 'zero', 'error'])
def test_same_private_fd_bounded_writes_and_unconditional_exit125(write_behavior):
    calls = []
    syncs = []
    exits = []
    shared = {'current_node': None}

    def write(fd, data):
        calls.append((fd, data))
        if write_behavior == 'error':
            raise OSError('fixture write failure')
        return {'full': len(data), 'one_byte': 1, 'zero': 0}[write_behavior]

    def exit_(code):
        exits.append(code)
        raise Terminal()

    api = functions('deny', state=shared, MONOTONIC=lambda: 100, COUNTS={}, DENIAL_FD=97,
                    RAW={'write': write, 'fsync': lambda fd: syncs.append(fd), '_exit': exit_})
    with pytest.raises(Terminal):
        api.deny('open:outside_pinned_or_private_read_roots', diagnostic={'status': 'CONTROL_METADATA'})
    assert exits == [125]
    assert sum(fd == 97 for fd, _ in calls) <= 8
    assert all(fd in (97, 2) for fd, _ in calls)
    assert all(fd == 97 for fd in syncs)
    assert shared['denied']['event'] == 'open:outside_pinned_or_private_read_roots'
    if write_behavior == 'full':
        assert json.loads(calls[0][1]) == shared


def test_policy_and_whole_original_suite_pins_are_unchanged_source():
    tree = ast.parse((ROOT / 'private_effect_fence.py').read_text())
    forbidden_names = {'open', 'read', 'stat', 'readlink', 'canonical', 'string_path', 'fspath',
                       'repr', 'str', 'extract_stack', 'format_stack', 'getenv'}
    for node in tree.body:
        if isinstance(node, ast.FunctionDef) and node.name in ('diagnostic_text', 'read_diagnostic', 'safe_read_diagnostic'):
            for call in (n for n in ast.walk(node) if isinstance(n, ast.Call)):
                name = call.func.id if isinstance(call.func, ast.Name) else call.func.attr if isinstance(call.func, ast.Attribute) else ''
                assert name not in forbidden_names
    policy = json.loads((ROOT / 'semantic-delta.json').read_text())
    assert policy['authority_changed'] is False and policy['old_generation2_replay'] is False
    assert policy['original_suite_counts'] == {'controls': 21, 'native': 116, 'project': 38, 'owner': 71}
