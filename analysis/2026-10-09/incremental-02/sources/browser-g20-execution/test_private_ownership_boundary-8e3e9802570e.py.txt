def file_stat(**changes):
    values=dict(st_dev=1,st_ino=2,st_mode=0o100600,st_uid=1000,st_gid=1000,st_nlink=1)
    values.update(changes)
    return SimpleNamespace(**values)


@pytest.mark.parametrize('change',[
    {'st_mode':0o040600},{'st_mode':0o120600},{'st_mode':0o010600},
    {'st_mode':0o020600},{'st_mode':0o060600},{'st_mode':0o140600},
    {'st_mode':0o104600},{'st_mode':0o102600},{'st_mode':0o101600},
    {'st_mode':0o100700},{'st_mode':0o100640},{'st_mode':0o100644},
    {'st_uid':1001},{'st_gid':1001},{'st_nlink':2},{'st_nlink':0},
])
def test_ownership_file_rejects_special_alias_shared_foreign(change):
    assert not fence.ownership_file_allowed(file_stat(**change),1000,1000)


def test_ownership_file_allows_only_exact_private_regular_current_owner():
    assert fence.ownership_file_allowed(file_stat(),1000,1000)


@pytest.mark.parametrize('uid,gid',[(True,1000),(1000,True),(-1,1000),(1000,-1),('1000',1000),(1000,None)])
def test_ownership_ids_are_exact_nonnegative_integers(uid,gid):
    assert not fence.ownership_file_allowed(file_stat(),uid,gid)


@pytest.mark.parametrize('path',['relative','/a/../b','/a/./b','//a','/a/','/a\0b','/' ,'/'+'x'*4096,b'/a',1,None,'/'+'/'.join(['x']*65)])
def test_ownership_path_rejects_alias_conversion_and_unbounded_input(path):

@pytest.mark.parametrize('failure',['still_open','foreign_error'])
def test_ownership_no_optimistic_descriptor_closure(monkeypatch,failure):
    monkeypatch.setitem(fence.RAW,'close',lambda fd:None)
    def observed(fd):
        if failure=='foreign_error':raise OSError(5,'failure')
        return file_stat()
    monkeypatch.setitem(fence.RAW,'fstat',observed)
    with pytest.raises((RuntimeError,OSError)):fence.ownership_close([10])

def test_ownership_descriptor_chain_cleanup_on_partial_open_failure(monkeypatch):
    closed=[];seen=[]
    def opened(name,flags,dir_fd=None):
        seen.append(name)
        if len(seen)==3:raise OSError(40,'symlink')
        return 10+len(seen)
    monkeypatch.setattr(fence,'ownership_open',opened)
    monkeypatch.setattr(fence,'ownership_close',lambda fds:closed.extend(fds))
    monkeypatch.setitem(fence.RAW,'fstat',lambda fd:file_stat(st_mode=0o040700))
    with pytest.raises(OSError):fence.ownership_chain('/a/b')
    assert seen==['/','a','b'] and closed==[11,12]


def test_ownership_changed_named_ancestor_is_never_same(monkeypatch):
    closed=[]
    monkeypatch.setattr(fence,'ownership_chain',lambda path:([11],((1,2,0o040700,1000,1000,2),)))
    monkeypatch.setattr(fence,'ownership_close',lambda fds:closed.extend(fds))
    assert not fence.ownership_chain_same('/private',((1,3,0o040700,1000,1000,2),))
    assert closed==[11]


def test_ownership_kernel_effect_is_single_captured_fchown_and_never_fake_success():
    tree=ast.parse(Path(fence.__file__).read_text())
    funcs={n.name:n for n in tree.body if isinstance(n,ast.FunctionDef)}
    node=funcs['guarded_ownership_chown']
    calls=[n for n in ast.walk(node) if isinstance(n,ast.Call) and ast.unparse(n.func)=="OWNERSHIP_RAW['fchown']"]
    assert len(calls)==1 and [ast.unparse(x) for x in calls[0].args]==['leaf','uid','gid']
    assert not any(isinstance(n,ast.Return) and isinstance(n.value,ast.Constant) and n.value.value is True for n in ast.walk(node))
    assert any(isinstance(n,ast.Raise) and isinstance(n.exc,ast.Name) and n.exc.id=='kernel_error' for n in ast.walk(node))
    assert ast.dump(funcs['deny'])==ast.dump(next(n for n in ast.parse(Path('<ASTRA_WORK>/browser-import-runtime-join/candidate/private_effect_fence.py').read_text()).body if isinstance(n,ast.FunctionDef) and n.name=='deny'))
