"""Supplementary SOURCE controls, NOT_RUN. No independent runtime acceptance.

AST facades check pure predicates and native binding semantics when separately
admitted. Full actual compose/import/native suites remain mandatory.
"""
import ast
from pathlib import Path
import sys
from types import SimpleNamespace

import pytest

ROOT = Path(__file__).parent / 'source'


def leaf(relative):
    source = (ROOT / relative).read_text()
    tree = ast.parse(source)
    # Only reviewed pure leaves: no substituted donor/runtime implementation.
    assert not any(isinstance(n, (ast.Import, ast.ImportFrom)) and
                   any(x in ast.unparse(n) for x in ('hermes_cli.config ', 'web_server', 'ctypes'))
                   for n in ast.walk(tree))
    env = {'__name__': 'source_policy_control'}
    exec(compile(tree, str(ROOT / relative), 'exec'), env)
    return SimpleNamespace(**env)


def function(relative, name, **env):
    tree = ast.parse((ROOT / relative).read_text())
    nodes = [n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == name]
    assert len(nodes) == 1
    exec(compile(ast.Module(body=nodes, type_ignores=[]), '<native-facade-source>', 'exec'), env)
    return env[name]


@pytest.mark.parametrize('key', ['FRIDAY_LLM_API_KEY', 'EXA_API_KEY', 'HERMES_LANGFUSE_PUBLIC_KEY'])
def test_pure_env_policy_keeps_integration_names(key):
    leaf('native/hermes_cli/config_env_policy.py').validate_env_var_name_for_write(key)


@pytest.mark.parametrize('key', ['PATH', 'LD_AUDIT', 'DYLD_FRAMEWORK_PATH', 'GIT_CONFIG_KEY_17',
                               'PYTHONPATH', 'NODE_OPTIONS', 'HERMES_HOME', 'EDITOR'])
def test_pure_env_policy_retains_subprocess_and_location_refusal(key):

@pytest.mark.parametrize('key', ['', '1BAD', 'A-B', 'A=B', 'A\nB'])
def test_pure_env_policy_retains_exact_name_validation(key):
    with pytest.raises(ValueError):
        leaf('native/hermes_cli/config_env_policy.py').validate_env_var_name_for_write(key)


@pytest.mark.parametrize('windows,key,refused', [(True, 'Path', True), (True, 'pythonpath', True),
                                              (False, 'Path', False), (False, 'pythonpath', False)])
def test_native_facade_keeps_live_platform_override(windows, key, refused):
    api = leaf('native/hermes_cli/config_env_policy.py')
    fn = function('native/hermes_cli/config.py', 'validate_env_var_name_for_write',
                  _IS_WINDOWS=windows, _env_write_policy=api)
    if refused:
        with pytest.raises(ValueError): fn(key)
    else:
        fn(key)

def test_native_dashboard_facade_keeps_late_config_and_host_patch_seams():
    api = leaf('native/hermes_cli/dashboard_auth_policy.py')
    seen = []
    fn = function('native/hermes_cli/web_server.py', 'should_require_dashboard_auth',
        Optional=__import__('typing').Optional,
        _explicit_dashboard_auth_required=lambda: seen.append('config') or False,
        _dashboard_public_hosts=lambda: seen.append('public') or frozenset({'localhost'}),
        should_require_auth=lambda host: seen.append(('host', host)) or True,
        _LOOPBACK_HOST_VALUES=api.LOOPBACK_HOST_VALUES,
        _dashboard_auth_required=api.dashboard_auth_required)
    assert fn('localhost') is True
    assert seen == ['config', 'public', ('host', 'localhost')]


def test_renderer_reuses_exact_pure_defaults_and_no_full_server_or_config():
    tree = ast.parse((ROOT / 'project/tools/configure_product.py').read_text())
    compose = next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == 'compose_product')
    dashboard = next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == '_dashboard')
    imports = [ast.unparse(n) for n in ast.walk(ast.Module(body=[compose, dashboard], type_ignores=[]))
               if isinstance(n, (ast.Import, ast.ImportFrom))]
    assert 'from hermes_cli.config_defaults import DEFAULT_CONFIG' in imports
    assert 'from hermes_cli.config_env_policy import validate_env_var_name_for_write' in imports
    assert 'from hermes_cli.dashboard_auth_policy import dashboard_auth_required' in imports
    assert not any('hermes_cli.config import' in s or 'hermes_cli.web_server import' in s for s in imports)
    # materialize_product keeps the existing effectful native write API.
    materialize = next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == 'materialize_product')
    assert any(isinstance(n, ast.ImportFrom) and n.module == 'hermes_cli.config' for n in ast.walk(materialize))


def test_coupled_template_validator_uses_same_pure_write_policy():
    tree = ast.parse((ROOT / 'project/plugins/friday_rework/onboarding.py').read_text())
    validator = next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == 'validate_template')
    imports = [n.module for n in ast.walk(validator) if isinstance(n, ast.ImportFrom)]
    assert 'hermes_cli.config_env_policy' in imports and 'hermes_cli.config' not in imports
    # Effectful onboarding config/credential publication retains native writers.
    assert any(isinstance(n, ast.ImportFrom) and n.module == 'hermes_cli.config' for n in ast.walk(tree))


def test_gateway_value_declarations_use_canonical_native_home_symbol():
    tree = ast.parse((ROOT / 'native/gateway/config.py').read_text())
    assert any(isinstance(n, ast.ImportFrom) and n.module == 'hermes_constants' and
               any(a.name == 'get_hermes_home' for a in n.names) for n in tree.body)
    assert not any(isinstance(n, ast.ImportFrom) and n.module == 'hermes_cli.config' for n in tree.body)
    # Actual config loading/saving stays in the native consumer.
    assert any(isinstance(n, ast.ImportFrom) and n.module == 'hermes_cli.config' for n in ast.walk(tree))


def test_pure_policy_annotations_resolve_in_actual_module_globals():
    # Python3.14 defers annotations; resolution still catches a missing import
    # that also breaks module import on supported Python3.11-3.13.
    import typing
    api = leaf('native/hermes_cli/config_env_policy.py')
    assert typing.get_type_hints(api._env_var_policy_name) == {
        'key': str, 'is_windows': typing.Optional[bool], 'return': str}
