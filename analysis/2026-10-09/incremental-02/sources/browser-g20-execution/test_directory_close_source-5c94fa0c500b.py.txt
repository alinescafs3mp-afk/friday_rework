def tree(path):
    actual = ast.parse(path.read_text())
    # Exact reviewed instrumentation inversion, solely for the original complete
    # historical AST oracles below. New bodies are separately qualified.
    if path == RUNNER:
        import copy
        repair = HERE.parents[1] / 'browser-observer-dispatch-repair/positive-run-v2.py'
        assert hashlib.sha256(repair.read_bytes()).hexdigest() == '06675c41cbde3533afe0030235024d455ce85d890a00aad4d6e1017f3bddf1be'
        expected = repair.read_text().replace('G19', 'G20')
        assert ast.dump(actual, include_attributes=False) == ast.dump(ast.parse(expected), include_attributes=False)
        prior = ast.parse((HERE.parents[1] / 'browser-directory-close-repair/candidate/positive-run.py').read_text())
        def byname(nodes, name):
            matches = [n for n in nodes if getattr(n, 'name', None) == name]
            assert len(matches) == 1
            return matches[0]
        for name in ('save', 'snapshot'):
            observed = byname(actual.body, name)
            index = actual.body.index(observed)
            actual.body[index] = copy.deepcopy(byname(prior.body, name))
        capture = byname(actual.body, 'Capture')
        report = byname(capture.body, 'pytest_runtest_logreport')
        capture.body[capture.body.index(report)] = copy.deepcopy(byname(byname(prior.body, 'Capture').body, 'pytest_runtest_logreport'))
        # Bindings are checked by whole approved current AST above, then exactly
        # relocated back to the historical source template for original oracles.
        for n in ast.walk(actual):
            if isinstance(n, ast.Constant) and isinstance(n.value, str):
                n.value = n.value.replace(str(HERE.parents[1] / 'browser-native-context-join/source'), str(HERE.parents[1] / 'browser-import-composed-source/source')).replace('G20', 'G17')
    elif path == GUARD:
        old_fixture = HERE.parents[1] / 'browser-import-composed-source/source/project/tests/test_browser_native_host.py'
        new_fixture = HERE.parents[1] / 'browser-native-context-join/source/project/tests/test_browser_native_host.py'
        old_hash = 'e18a07fbc382eb8632f48140ee9e1378dbefa2647815783b5db1ab2c39ac050c'
        new_hash = '90a82a0870bdcb42e48c4c667c4ab4a3ca39ea14aace98970979e5dda890dafc'
        assert hashlib.sha256(old_fixture.read_bytes()).hexdigest() == old_hash
        assert hashlib.sha256(new_fixture.read_bytes()).hexdigest() == new_hash
        matches = [n for n in actual.body if isinstance(n, ast.Assign) and len(n.targets) == 1 and isinstance(n.targets[0], ast.Name) and n.targets[0].id == 'OWNERSHIP_TEST_SHA']
        assert len(matches) == 1 and isinstance(matches[0].value, ast.Constant) and matches[0].value.value == new_hash
        matches[0].value = ast.Constant(old_hash)
    return actual


def functions(path):
    return {n.name: n for n in tree(path).body if isinstance(n, ast.FunctionDef)}


def text(node):
    return ast.unparse(node)


def dump(node):
    return ast.dump(node, include_attributes=False)


def calls(node, name):
    return [n for n in ast.walk(node) if isinstance(n, ast.Call) and text(n.func) == name]


def unchanged_guard_tree(candidate):
    changed = {'directory_sync_open', 'directory_sync_open_audit',
               'directory_sync_fsync', 'directory_sync_close', 'directory_sync_diagnostic'}
    before = tree(OLD / 'candidate/private_effect_fence.py')
    return [dump(n) for n in before.body if getattr(n, 'name', None) not in changed] == [
        dump(n) for n in candidate.body if getattr(n, 'name', None) not in changed]

@pytest.mark.parametrize('name', ['guarded_open', 'read_ok', 'check_fd', 'audit',

def test_real_exception_diagnostic_is_bounded_and_does_not_claim_kernel_success():
    fn = functions(GUARD)['directory_sync_diagnostic']
    namespace = {'sys': sys, 'state': {}}
    # The isolated pure diagnostic observes sys.exception(); no guard import,
    # fabricated frame, raw-call replacement, or filesystem operation occurs.
    exec(compile(ast.Module(body=[fn], type_ignores=[]), '<directory-diagnostic>', 'exec'), namespace)
    try:
        raise OSError(5, 'source-only actual Python exception')
    except OSError as error:
        for _ in range(12):
            namespace['directory_sync_diagnostic']('kernel_open', None)
        assert sys.exception() is error
    history = namespace['state']['directory_sync_diagnostics']
    assert len(history) == 8
    assert all(row['exception']['type'] == 'OSError' and row['exception']['errno'] == 5 for row in history)
    assert all('closed' not in row for row in history)

def test_runner_phase_execution_and_full_acceptance_are_unchanged():
    before = tree(OLD / 'native-preparation/positive/run.py')
    after = tree(RUNNER)
    changed = {'current_copy_snapshot', 'copy_metadata_profile', 'record_copy_fault_node',
               'copy_fault_profile', 'fault_profile', 'directory_profile',
               'directory_profile_event', 'OwnershipEvidence'}
    # Capture's only change routes the same real event through the fail-closed
    # observer dispatcher. All native collection/reporting and source IDs stay.
    capture = next(n for n in after.body if isinstance(n, ast.ClassDef) and n.name == 'Capture')
    rewritten = 0
    for call in calls(capture, 'sys.setprofile'):
        if len(call.args) == 1 and text(call.args[0]) == 'fault_profile':
            call.args[0] = ast.parse('directory_fault_profile if SCENARIO in DIRECTORY_SCENARIOS else copy_fault_profile', mode='eval').body
            rewritten += 1
    assert rewritten == 1
    assert [dump(n) for n in before.body if getattr(n, 'name', None) not in changed] == [
        dump(n) for n in after.body if getattr(n, 'name', None) not in changed]
