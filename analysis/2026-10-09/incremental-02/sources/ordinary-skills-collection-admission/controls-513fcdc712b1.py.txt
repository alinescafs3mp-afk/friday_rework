# Offline-only event-loop wakeup: an OS pipe instead of asyncio's socketpair.
# Native coroutines/dispatcher remain real; no socket is ever admitted.
def make_pipe(loop):
 r,w=os.pipe2(os.O_NONBLOCK|os.O_CLOEXEC)
 loop._ssock=os.fdopen(r,'rb',buffering=0);loop._csock=os.fdopen(w,'wb',buffering=0)
 loop._internal_fds+=1;loop._add_reader(r,loop._read_from_self)
def read_pipe(loop):
 try:
  while os.read(loop._ssock.fileno(),4096):pass
 except BlockingIOError:pass
def write_pipe(loop):
 try:
  if loop._csock:os.write(loop._csock.fileno(),b'\0')
 except (BlockingIOError,OSError):pass
def close_pipe(loop):
 loop._remove_reader(loop._ssock.fileno());loop._ssock.close();loop._csock.close()
 loop._ssock=loop._csock=None;loop._internal_fds-=1
asyncio.SelectorEventLoop._make_self_pipe=make_pipe
asyncio.SelectorEventLoop._read_from_self=read_pipe
asyncio.SelectorEventLoop._write_to_self=write_pipe
asyncio.SelectorEventLoop._close_self_pipe=close_pipe
