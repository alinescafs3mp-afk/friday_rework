observations = []

def sha(data):
    return hashlib.sha256(data).hexdigest()

def check_file(path, expected, size=None):
    path = Path(path)
    data = path.read_bytes()
    actual = sha(data)
    assert actual == expected, (str(path), actual, expected)
    if size is not None:
        assert len(data) == size, str(path)
    observations.append({'path': str(path), 'sha256': actual, 'bytes': len(data),
                         'mode': oct(stat.S_IMODE(path.stat().st_mode))})
    return data

def write(name, data):
    path = OUT / name
    with path.open('x') as stream:
        stream.write(json.dumps(data, indent=2) + '\n')
    path.chmod(0o400)

def check_refs(value):
    if isinstance(value, dict):
        if {'path', 'sha256'} <= value.keys():
            check_file(value['path'], value['sha256'], value.get('bytes'))
        for nested in value.values():
            check_refs(nested)
    elif isinstance(value, list):
        for nested in value:
            check_refs(nested)

def reconstruct(group):
    patch = (SOURCE / (group + '.patch')).read_bytes().splitlines(keepends=True)
    base = Path(pins['baseline_' + group])
    items = []
    index = 0
    while index < len(patch):
        assert patch[index].startswith(b'diff --git '), (group, index)
        end = index + 1
        while end < len(patch) and not patch[end].startswith(b'diff --git '):
            end += 1
        section = patch[index:end]
        old_header = next(i for i, line in enumerate(section) if line.startswith(b'--- '))
        old_name = section[old_header][4:].strip().decode()
        new_name = section[old_header + 1][4:].strip().decode()
        assert new_name.startswith('b/')
        rel = new_name[2:]
        assert old_name in ('/dev/null', 'a/' + rel)
        before = b'' if old_name == '/dev/null' else (base / rel).read_bytes()
        if old_name == '/dev/null':
            assert not (base / rel).exists(), rel
        lines = before.splitlines(keepends=True)
        output = []
        cursor = 0
        j = old_header + 2
        while j < len(section):
            match = re.fullmatch(rb'@@ -(\d+)(?:,(\d+))? \+(\d+)(?:,(\d+))? @@[^\n]*\n', section[j])
            assert match, (group, rel, j, section[j])
            old_start, old_count, new_start, new_count = (int(match[k] or b'1') for k in range(1,5))
            start = old_start - 1 if old_count else old_start
            assert cursor <= start <= len(lines)
            output.extend(lines[cursor:start]); cursor = start
            assert len(output) == (new_start - 1 if new_count else new_start)
            consumed = produced = 0
            j += 1
            while consumed < old_count or produced < new_count:
                line = section[j]
                sign, payload = line[:1], line[1:]
                assert sign in (b' ', b'-', b'+'), (group, rel, line)
                if sign in (b' ', b'-'):
                    assert cursor < len(lines) and lines[cursor] == payload, (group, rel, cursor)
                    cursor += 1; consumed += 1
                if sign in (b' ', b'+'):
                    output.append(payload); produced += 1
                j += 1
            assert consumed == old_count and produced == new_count
        output.extend(lines[cursor:])
        after = b''.join(output)
        assert after == (SOURCE / group / rel).read_bytes(), (group, rel)
        items.append({'group': group, 'path': rel, 'before_sha256': sha(before), 'sha256': sha(after), 'bytes': len(after), 'exact': True})
        index = end
    expected = {rel[len(group) + 1:] for rel in pins['files'] if rel.startswith(group + '/')}
    assert {item['path'] for item in items} == expected, (group, expected)
    return items
