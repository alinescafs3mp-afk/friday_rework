def unique(pairs):
    out = {}
    for k, v in pairs:
        assert k not in out, ('duplicate_json_key', k)
        out[k] = v
    return out

def js(p):
    return json.loads(Path(p).read_bytes(), object_pairs_hook=unique)

def ref(p):
    p = Path(p)
    b = p.read_bytes()
    s = p.lstat()
    assert stat.S_ISREG(s.st_mode) and s.st_uid == 1000 and s.st_nlink == 1
    assert p.resolve(strict=True) == p and not p.is_symlink()
    return dict(path=str(p), sha256=hashlib.sha256(b).hexdigest(), bytes=len(b),
                mode=stat.S_IMODE(s.st_mode), uid=s.st_uid, nlink=s.st_nlink)

def pinned(p, sha):
    x = ref(p)
    assert x['sha256'] == sha, (str(p), 'hash_mismatch')
    return x

            n = 1
            for deco in node.decorator_list:
                if isinstance(deco,ast.Call) and isinstance(deco.func,ast.Attribute) and deco.func.attr == 'parametrize':
                    assert isinstance(deco.args[1],(ast.List,ast.Tuple))

            counts[node.name] = n
    casecounts[name] = sum(counts.values())
    functions[name] = counts
