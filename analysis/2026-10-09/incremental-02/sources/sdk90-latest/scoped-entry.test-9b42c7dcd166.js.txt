/** Future parser/API tests only; fake launchers never qualify the kernel. */
import assert from 'node:assert/strict';
import fs from 'node:fs';
import os from 'node:os';
import path from 'node:path';
import { scopedGrantArgs, probeScoped } from '@deepseek-ai/node-addon-system/landlock-run';

assert.deepEqual(scopedGrantArgs({ readOnly: ['/usr'], readWrite: ['/work'] }),
  ['--scope-ipc', '--ro', '/usr', '--rw', '/work']);
for (const timeoutMs of [0, -1, Infinity, NaN, 0.5]) {
  assert.throws(() => probeScoped('/absent', { timeoutMs }), /positive finite/);
}
assert.equal(probeScoped('/absent', { timeoutMs: 20 }), 'unusable');
const dir = fs.mkdtempSync(path.join(os.tmpdir(), 'scoped-report-source-'));
try {
  for (const [name, line] of [
    ['old-full', 'landlock: fully enforced'],
    ['old-partial', 'landlock: partially enforced (older ABI)'],
    ['extra', 'landlock: ipc scopes enforced (abi>=6)\nextra'],
    ['blank', ''],
  ]) {
    const fake = path.join(dir, name);
    fs.writeFileSync(fake, '#!/bin/sh\nprintf "%s\\n" ' + "'" + line + "'" + '\n', { mode: 0o755 });
    assert.equal(probeScoped(fake, { timeoutMs: 100 }), 'unusable');
  }
} finally { fs.rmSync(dir, { recursive: true }); }
