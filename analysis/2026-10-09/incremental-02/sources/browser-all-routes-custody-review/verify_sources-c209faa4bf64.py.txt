def digest(data):
    return hashlib.sha256(data).hexdigest()

def check_refs(value):
    if isinstance(value, dict):
        if 'path' in value and 'sha256' in value:
            path = value['path']
            data = pathlib.Path(path).read_bytes()
            assert digest(data) == value['sha256'], path
            if 'bytes' in value:
                assert len(data) == value['bytes'], path
            report['refs'][path] = {'sha256': digest(data), 'bytes': len(data)}
        for item in value.values():
            check_refs(item)
    elif isinstance(value, list):
        for item in value:
            check_refs(item)

def reconstruct(tree):
    lines = (P/(tree+'.patch')).read_bytes().splitlines(keepends=True)
    i = 0
    names = []
    while i < len(lines):
        assert lines[i].startswith(b'--- '), (i, lines[i])
        before = lines[i][4:].strip().decode()
        after = lines[i+1][4:].strip().decode()
        assert lines[i+1].startswith(b'+++ b/'), after
        name = after[2:]
        original = [] if before == '/dev/null' else (BASE[tree]/before[2:]).read_bytes().splitlines(keepends=True)
        output, cursor = [], 0
        i += 2
        while i < len(lines) and lines[i].startswith(b'@@ '):
            match = re.match(rb'@@ -(\d+)(?:,(\d+))? \+(\d+)(?:,(\d+))? @@', lines[i])
            assert match, (name, lines[i])
            old_start, old_count, new_start, new_count = [int(v) if v is not None else 1 for v in match.groups()]
            begin = old_start - 1 if old_start else 0
            assert begin >= cursor, (name, cursor, begin)
            output.extend(original[cursor:begin])
            cursor = begin
            seen_old = seen_new = 0
            i += 1
            while seen_old < old_count or seen_new < new_count:
                line = lines[i]
                prefix, text = line[:1], line[1:]
                assert prefix in (b' ', b'-', b'+'), (name, line)
                if prefix in (b' ', b'-'):
                    assert original[cursor] == text, (name, cursor)
                    cursor += 1
                    seen_old += 1
                if prefix in (b' ', b'+'):
                    output.append(text)
                    seen_new += 1
                i += 1
            assert seen_old == old_count and seen_new == new_count, name
        output.extend(original[cursor:])
        final = b''.join(output)
        assert final == (P/'candidate'/tree/name).read_bytes(), name
        names.append(name)
    expected = sorted(row['file'] for row in pins['files'] if row['tree'] == tree)
    assert sorted(names) == expected
    return names

for tree in ('native', 'project'):
    report['reconstructed'][tree] = reconstruct(tree)

tests = 0
for row in pins['files']:
    path = pathlib.Path(row['after']['path'])
    if path.suffix == '.py':
        source = path.read_text()
        parsed = ast.parse(source, filename=str(path))
        compile(parsed, str(path), 'exec')  # compilation only; never executes
        report['AST'].append(str(path))
        tests += sum(isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)) and node.name.startswith('test_') for node in ast.walk(parsed))
report['test_function_sources_not_run'] = tests

class StripFence(ast.NodeTransformer):
    def visit_ImportFrom(self, node):
        if node.module == 'hermes_cli.browser_runtime' and len(node.names) == 1 and node.names[0].name == 'browser_action_entry':
            return None
        return node
    def visit_FunctionDef(self, node):
        node.decorator_list = [d for d in node.decorator_list if not (isinstance(d, ast.Name) and d.id == 'browser_action_entry')]
        return self.generic_visit(node)
    visit_AsyncFunctionDef = visit_FunctionDef

for name in ['hermes_cli/browser_connect.py', 'hermes_cli/cli_commands_mixin.py', 'tools/browser_cdp_tool.py', 'tools/browser_dialog_tool.py', 'tools/browser_extension_router.py', 'tools/browser_tool_cdp.py', 'tools/browser_use_cli.py', 'tools/browser_vault_tool.py']:
