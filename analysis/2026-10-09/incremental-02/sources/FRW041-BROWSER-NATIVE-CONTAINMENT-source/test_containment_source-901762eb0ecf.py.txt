        ('SubState', 's', 'running'), ('MainPID', 'u', 1234), ('ControlPID', 'u', 0),

        ('NeedDaemonReload', 'b', False), ('DropInPaths', 'as', []), ('InvocationID', 'ay', [1] * 16)]}

             'pid': 1234, 'pid_start_time': 8, 'cgroup': cg, 'timeout_seconds': 5, 'timeout': False,
             'returncode': 0, 'client_reaped': True, 'stderr': '', 'started_monotonic': 1.0,
             'completed_monotonic': 2.0, 'stdout': json.dumps({'type': 'a{sv}', 'data': [props]})}

def test_render_preserves_custody_and_explicit_isolated_contour(tmp_path):
    payload, argv = unit.render(ROOT, tmp_path, 'friday-browser-prerequisite-1234567890abcdef.service')
    text = payload.decode()
    for line in ['PrivateNetwork=yes', 'PrivateUsers=self', 'PrivatePIDs=yes', 'PrivateMounts=yes',
                 'ProtectHome=tmpfs', 'ProtectSystem=strict', 'NoNewPrivileges=yes',
                 'RestrictAddressFamilies=AF_UNIX AF_INET AF_INET6', 'CapabilityBoundingSet=',
                 'AmbientCapabilities=', 'MemoryMax=2147483648', 'MemorySwapMax=0', 'TasksMax=128',
                 'CPUQuota=200%', 'CPUAffinity=0 1', 'ExitType=cgroup', 'KillMode=control-group',
                 'RuntimeMaxSec=45s', 'JobTimeoutSec=65s', 'TimeoutStopSec=5s', 'Restart=no']:
        assert line in text.splitlines()
    assert text.count('ExecStart=') == 1 and 'systemd-run' not in text
    assert 'EnvironmentFile=' not in text and 'NetworkNamespacePath=' not in text and 'JoinsNamespaceOf=' not in text
    assert argv[:2] == ['/usr/bin/env', '-i']
    assert 'HOME=' + str(tmp_path / 'runtime-home') in argv
    assert not any('CI=' in x or 'PROXY=' in x or 'PLAYWRIGHT' in x or '--no-sandbox' in x for x in argv)
    assert unit.POST_START_WINDOW == 55 and unit.MIN_START_WINDOW == 20


def test_actual_current_typed_invocation_positive():
    owner, cg, fragment, value, _ = fixture_identity()
    assert probe.validate_native_identity(value, owner, cg, fragment, 0.0, 3.0, pid=1234) == '01' * 16


@pytest.mark.parametrize('mutation', ['nonce', 'boot', 'pid', 'stale', 'timeout', 'exit', 'not_reaped', 'dropin', 'inactive', 'zero_invocation'])
def test_foreign_stale_or_ambiguous_invocation_refused(mutation):
    owner, cg, fragment, value, props = fixture_identity()
    if mutation == 'nonce': value['nonce'] = 'fedcba0987654321'
    if mutation == 'boot': value['boot_id'] = 'foreign'
    if mutation == 'pid': props['MainPID']['data'] = 1235
    if mutation == 'stale': value['started_monotonic'] = -1
    if mutation == 'timeout': value['timeout'] = True
    if mutation == 'exit': value['returncode'] = 1
    if mutation == 'not_reaped': value['client_reaped'] = False
    if mutation == 'dropin': props['DropInPaths']['data'] = ['/foreign.conf']
    if mutation == 'inactive': props['ActiveState']['data'] = 'inactive'
    if mutation == 'zero_invocation': props['InvocationID']['data'] = [0] * 16
    value['stdout'] = json.dumps({'type': 'a{sv}', 'data': [props]})
    with pytest.raises(RuntimeError): probe.validate_native_identity(value, owner, cg, fragment, 0.0, 3.0, pid=1234)


def test_outer_private_user_mapping_does_not_fake_nested_chrome_sandbox():
    browser = {'argv': [probe.EXECUTABLE, '--remote-debugging-port=0'], 'executable': probe.EXECUTABLE,
        'label': probe.LABEL, 'uid': [1000]*4, 'cap_eff': 0, 'cap_prm': 0, 'pid': 2,
        'nspid': [2], 'uid_map': [[1000,1000,1]], 'seccomp': 2, 'seccomp_filters': 1, 'cgroup': 'own',
        'user_namespace': 'user:[2]', 'network_namespace': 'net:[2]'}
    renderer = {'argv': [probe.EXECUTABLE, '--type=renderer'], 'uid': [1000]*4, 'cap_eff': 0, 'cap_prm': 0,
        'pid': 3, 'nspid': [3,1], 'uid_map': [[1000,1000,1]], 'gid_map': [[1000,1000,1]],
        'no_new_privs': 1, 'seccomp': 2, 'seccomp_filters': 2, 'cgroup': 'own',
        'user_namespace': 'user:[3]', 'network_namespace': 'net:[2]'}
    probe.check_sandbox_records(browser, [renderer], [[1000,1000,1]], ['1'])
    for field, bad in [('user_namespace','user:[2]'),
                       ('seccomp_filters',1), ('nspid',[3]), ('cap_eff',1)]:
        changed = dict(renderer, **{field:bad})
        with pytest.raises(RuntimeError): probe.check_sandbox_records(browser, [changed], [[1000,1000,1]], ['1'])

def test_unconfirmed_cgroup_or_unit_not_terminal():
    name = 'friday-browser-prerequisite-1234567890abcdef.service'
    path = '<USER_RUNTIME>/systemd/user/' + name
    state = {'Id': name, 'FragmentPath': path, 'Transient': 'no', 'DropInPaths': '',
             'ActiveState': 'inactive', 'MainPID': '0', 'ControlPID': '0', 'Job': '', 'InvocationID': '', 'ControlGroup': ''}
    assert unit.cessation(state, name, Path(path), 'own', True) == ''
    with pytest.raises(RuntimeError): unit.cessation(state, name, Path(path), 'own', False)
    with pytest.raises(RuntimeError): unit.cessation(dict(state, MainPID='1234'), name, Path(path), 'own', True)
    with pytest.raises(RuntimeError): unit.cessation(dict(state, DropInPaths='/foreign'), name, Path(path), 'own', True)


@pytest.mark.parametrize('grant', [
    {'runtime_authorized': False, 'allow_one_browser_prerequisite': True, 'source_task': unit.SOURCE_TASK, 'independent_review_sha256': 'a'*64},
    {'runtime_authorized': True, 'allow_one_browser_prerequisite': True, 'source_task': 'FRW041-BROWSER-NATIVE-INVOCATION-REPAIR#1', 'independent_review_sha256': 'a'*64},
    {'runtime_authorized': True, 'allow_one_browser_prerequisite': True, 'source_task': unit.SOURCE_TASK, 'independent_review_sha256': ''}])
def test_source_clock_or_historical_admission_does_not_authorize_runtime(grant):
    def read(path, *args, **kwargs):
        if str(path) == '/proc/1/cgroup': return '0::/owner-service'
        if str(path) == '/test-admission.json': return json.dumps(grant)
        raise AssertionError('unexpected input/effect before admission refusal')
    with patch.dict(unit.os.environ, {}, clear=True), patch.object(unit.os, 'geteuid', return_value=1000), \
         patch.object(unit, 'private'), patch.object(Path, 'exists', return_value=False), \
         patch.object(Path, 'read_text', read):
        with pytest.raises(RuntimeError, match='fresh_root_review_required'):
            unit.admit(Path('/test-admission.json'))
