"""One factual closure after the existing channel's single RESULT submission."""
import asyncio, datetime, hashlib, json, os, runpy, time
from pathlib import Path

os.umask(0o077)
E = Path(__file__).parent
S = E.parent
R = S.parent
A = 'FRW026-ORDINARY-SKILLS-PROVENANCE-REPAIR-SOL'
EV = 'SOL-FRW026-ORDINARY-SKILLS-PROVENANCE-REPAIR-R1'
clock = json.loads((E / 'clock.json').read_text())
assert Path('/proc/sys/kernel/random/boot_id').read_text().strip() == clock['boot_id']


def ref(p):
    return {'path': str(p), 'sha256': hashlib.sha256(Path(p).read_bytes()).hexdigest()}


receipt_path = R / 'sol-link-native-receipts' / (EV + '.json')
receipt = json.loads(receipt_path.read_text())
result = S / 'FRW026-ORDINARY-SKILLS-PROVENANCE-REPAIR-RESULT.json'
assert receipt['state'] == 'ENQUEUED' and receipt['sha256'] == ref(result)['sha256']
rows = {n: json.loads((R / 'subagent-lifecycle/state' / n / 'registry.json').read_text())['peer_assignments'][A + '#1'] for n in ('sol', 'astra')}
assert all(x['state'] == 'CLOSED' and x['last_event_id'] == EV and x['last_action'] == 'complete' for x in rows.values())
N = runpy.run_path(str(R / 'sol-link-native-control.py'))['NativeControl']


async def inspect():
    async with N('sol') as c:
        return await c.inspect()


native = asyncio.run(inspect())
assert native['goal_status'] is None
owned = []
for p in Path('/proc').iterdir():
    if not p.name.isdecimal() or int(p.name) in (os.getpid(), os.getppid()):
        continue
    try:
        if p.stat().st_uid != os.getuid():
            continue
        args = [a.decode(errors='replace') for a in (p / 'cmdline').read_bytes().split(b'\0') if a]
        if any(str(E) in a for a in args):
            owned.append({'pid': int(p.name), 'argv': args,
                          'start_ticks': (p / 'stat').read_text().rsplit(')', 1)[1].split()[19]})
    except OSError:
        pass
assert not owned, owned
manifest = json.loads((E / 'FINAL-MANIFEST.json').read_text())
assert all(ref(x['path'])['sha256'] == x['sha256'] for x in manifest['source_files'])
assert all(ref(x['origin'])['sha256'] == x['sha256'] for x in json.loads((E / 'initial-source-pins.json').read_text()))
for name, expected in [('provenance-final', 18), ('native-dependencies', 35)]:
    checks = json.loads((E / (name + '-results.json')).read_text())
    assert checks['exit'] == 0 and len(checks['reports']) == expected
    assert all(ref(p)['sha256'] == h for p, h in checks['inputs'].items())
now = time.monotonic()
assert now < clock['deadline_monotonic']
out = {
    'assignment': A, 'generation': 1, 'state': 'CLOSED_SOURCE_REPAIRED_INDEPENDENT_ACCEPTANCE_PENDING',
    'finished_msk': datetime.datetime.now(datetime.timezone(datetime.timedelta(hours=3))).isoformat(),
    'elapsed_seconds': now - clock['start_monotonic'],
    'remaining_original_seconds': clock['deadline_monotonic'] - now, 'budget_reset': False,
    'manifest': ref(E / 'FINAL-MANIFEST.json'), 'result': ref(result), 'native_receipt': ref(receipt_path),
    'delivery_state': receipt['state'], 'peer_read': 'UNCONFIRMED; ENQUEUED never proves read',
    'peer_rows': rows, 'native_ownership': native, 'owned_process_inventory': owned, 'owned_model_children': 0,
    'controls': '18 prompt+35 native dependency final offline PASS; exact process sessions92730/71557 exited0 and reaped; canonical lock released, no model/native/network/credentials effects.',
    'source_pins_unchanged_after_freeze': True, 'parent_immutable_source31_unchanged': True,
    'source_changes': 'Only two files in new private31 snapshot; no parent worktree/main/source integration, commit or push.',
    'product_live_effects': 'NOT_RUN', 'independent_acceptance': 'PENDING_ASTRA',
    'next': 'Astra independently review/integrate exact source and run whole required final gate/installer/installed/live admission. Sol ready/end, no new work without verified current TASK; no replay/resend/reset.'
}
p = E / 'FINAL-CLOSURE.json'
fd = os.open(p, os.O_CREAT | os.O_EXCL | os.O_WRONLY | os.O_NOFOLLOW, 0o600)
with os.fdopen(fd, 'w') as f:
    json.dump(out, f, ensure_ascii=False, indent=2); f.write('\n'); f.flush(); os.fchmod(f.fileno(), 0o400); os.fsync(f.fileno())
fd = os.open(E, os.O_RDONLY | os.O_DIRECTORY)
try:
    os.fsync(fd)
finally:
    os.close(fd)
for p in E.iterdir():
    if p.is_file() and not p.is_symlink():
        p.chmod(0o400)
print(json.dumps({k: out[k] for k in ('state', 'finished_msk', 'elapsed_seconds', 'remaining_original_seconds', 'delivery_state', 'peer_read', 'owned_process_inventory')}))
