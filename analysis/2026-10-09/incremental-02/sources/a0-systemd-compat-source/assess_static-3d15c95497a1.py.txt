#!/usr/bin/python3
"""Finite source-only assessment. Never import/execute a target module or native client."""
import ast
import datetime
import difflib
import hashlib
import json
import os
from pathlib import Path
import resource
import signal
import tarfile
from types import SimpleNamespace

os.umask(0o077)
os.sched_setaffinity(0, [0, 1])
resource.setrlimit(resource.RLIMIT_AS, (2 * 1024**3, 2 * 1024**3))
resource.setrlimit(resource.RLIMIT_CPU, (20, 20))
signal.alarm(40)
E = Path(__file__).parent
W = E.parent
S = W / 'a0-boot-argument-repair/source'
checks = []

def check(name, predicate):
    assert predicate, name
    checks.append(name)

def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def save(name, value):
    (E / name).write_text(json.dumps(value, indent=2, ensure_ascii=False) + '\n')

pins = {
    W/'a0-owner-g3-reconciliation/source-diagnosis.json': '8f63589901fed26e25b14ad41da52b2f720dae0cb40e87b79af8a077ea267b13',
    W/'a0-owner-g3-reconciliation/release.json': 'afb623d9cb0870b7eaf558078fc0781af279d69c75a39828c06a80cc8916c458',
    S/'manifest.json': 'edcb427f081791159ba064ed4d429d816ca63ea04c25d39ba836e06621c41b9c',
    S/'native-evidence/native-result.json': '9d8792a0866fe721343fd12b915ec7d117a3db972c2357b9dab182239888acef',
}
manifest = json.loads((S/'manifest.json').read_bytes())
pins.update({S/name: pin for name, pin in manifest['files'].items()})
check('all original pinned inputs match', all(digest(p) == h for p,h in pins.items()))
save('verified-original-inputs.json', {str(p): h for p,h in pins.items()})

recipe = json.loads((S/'recipe.json').read_bytes())
commands = recipe['native_start_argv']
roles = ['outer-bound','dependent-bound','dependent-late','outer-negative','dependent-negative','outer-timeout','dependent-timeout','sentinel']
check('exact eight constructor roles', list(commands) == roles)
fixed = {
    'Restart':'no', 'KillMode':'control-group','SendSIGKILL':'yes','FinalKillSignal':'SIGKILL',
    'TimeoutStartSec':'8s','TimeoutStopSec':'2s','TimeoutStartFailureMode':'kill',
    'TimeoutStopFailureMode':'kill','RuntimeRandomizedExtraSec':'0','JobTimeoutSec':'20s',
    'JobTimeoutAction':'none','CollectMode':'inactive-or-failed','CPUQuota':'200%',
    'CPUAffinity':'0 1','MemoryMax':'128M','TasksMax':'8','OOMPolicy':'stop','UMask':'0077',
    'NoNewPrivileges':'yes','NotifyAccess':'main',
}
bound = {'dependent-bound':'outer-bound','dependent-late':'outer-bound','dependent-timeout':'outer-timeout'}
matrix = []
for role, argv in commands.items():
    props = dict(arg.removeprefix('--property=').split('=',1) for arg in argv if arg.startswith('--property='))
    unit = recipe['prefix']+'-'+role+'.service'
    check(role+' fixed manager/resource properties', all(props.get(k)==v for k,v in fixed.items()))
    check(role+' bounded constructor and identity', argv[:4] == ['/usr/bin/systemd-run','--user','--quiet','--no-block'] and argv[4:6] == ['--unit='+unit,'--service-type=notify'])
    runtime = '35s' if role=='sentinel' else '8s' if role.startswith('outer-') else '12s'
    check(role+' original active runtime', props['RuntimeMaxSec']==runtime)
    check(role+' original dependency edges', all(props.get(k)==(recipe['prefix']+'-'+bound[role]+'.service' if role in bound else None) for k in ['BindsTo','After']))
    check(role+' exact condition/main/stop hook', props['ExecCondition']==f'/usr/bin/python3 -I -S -B {S}/payload.py condition {role}' and props['ExecStopPost']==f'/usr/bin/python3 -I -S -B {S}/observer.py stop-post {role}' and argv[argv.index('--')+1:]==['/usr/bin/python3','-I','-S','-B',str(S/'payload.py'),'payload',role])
    check(role+' no alternate running-only bound', 'JobRunningTimeoutSec' not in props)
    matrix.append({'role':role,'unit':unit,'properties':props,'argv_sha256':hashlib.sha256(json.dumps(argv,separators=(',',':')).encode()).hexdigest(),'queue_inclusive_timeout_seconds':20,'active_runtime_seconds':int(runtime[:-1])})
save('all-eight-constructor-audit.json', {'constructor_source':str(S/'recipe.json'),'sha256':digest(S/'recipe.json'),'roles':matrix,'potential_submission_count':9,'note':'Eight names; outer-bound has one deliberate reconstruction attempt after consumed one-use permit. No constructor or name changed.'})

driver = (S/'driver.py').read_text()
common = (S/'common.py').read_text()
parent = (S/'parent_integration.py').read_text()
check('original recipe whole/driver/submit anchors', [recipe[k] for k in ['maximum_whole_grant_seconds','driver_finish_deadline_seconds','submission_cutoff_seconds']]==[120,110,35])
check('admission remains original5seconds', "require(within(r, 5), 'stale admission; never renew original anchor')" in driver)
check('receipt deadlines cannot be reset', all(s in common for s in ['a + 35_000_000_000','a + 110_000_000_000','a + 120_000_000_000']))
check('no late or uncertain start retained', "require(not self.no_more_starts and within(self.r, 35, self.clock()), 'submission closed')" in driver and "require(not self.submission_uncertain, 'no replay after uncertain start')" in driver)
check('attempt count before transport retained', driver.index('self.starts[role] += 1')<driver.index("return self.manager(self.recipe['native_start_argv'][role], effect=True)"))
check('FD9 unknown cessation retention intact', "'parent_retains_canonical_lock_on_unknown':True" in parent and "'canonical_fd9':'STILL_HELD; release requires verified cessation'" in parent and 'if (self.submission_uncertain or not self.last_snapshot_valid' in driver)
check('expected queue bound remains manager-observed20s', "'JobTimeoutUSec': '20s'" in driver)

dsc = (E/'systemd_259.5-0ubuntu3.4.dsc').read_text()
for line in dsc.split('Checksums-Sha256:',1)[1].split('Files:',1)[0].strip().splitlines():
    sha,size,name = line.split()
    check('DSC source checksum '+name, digest(E/name)==sha and (E/name).stat().st_size==int(size))
equality = json.loads((E/'ubuntu-orig-source-equality.json').read_bytes())
check('Ubuntu orig exact five upstream source files', len(equality)==5 and all(x['upstream_raw_identical'] for x in equality))
with tarfile.open(E/'systemd_259.5-0ubuntu3.4.debian.tar.xz') as archive:
    debian = {m.name:archive.extractfile(m).read() for m in archive if m.isfile()}
    series = [l.split()[0] for l in debian['debian/patches/series'].decode().splitlines() if l and not l.startswith('#')]
    check('all40 applied Debian patches scanned', len(series)==40 and all('debian/patches/'+p in debian for p in series))
    relevant = {p:debian['debian/patches/'+p].decode() for p in series if b'dbus-unit.c' in debian['debian/patches/'+p] or b'JobTimeout' in debian['debian/patches/'+p] or b'bus-unit-util.c' in debian['debian/patches/'+p]}
    check('Ubuntu applied patches leave setter/parser unchanged', not relevant)
    metadata_mentions = [p for p,b in debian.items() if b'JobTimeout' in b or b'dbus-unit.c' in b]
    save('ubuntu-patch-compatibility.json', {'series_entries':len(series),'relevant_setter_parser_patches':relevant,'all_debian_files_mentioning_setter_or_JobTimeout':metadata_mentions,'orig_equality':equality,'source_checksums_verified_against_https_DSC':True,'DSC_signature_cryptographically_verified':False})

core = (E/'upstream2595-dbus-unit.c.txt').read_text()
branch = core.split('if (streq(name, "JobTimeoutUSec")) {',1)[1].split('\n        }',1)[0]
check('actual upstream JobTimeout branch lacks return', 'bus_set_transient_usec_fix_0' in branch and 'return ' not in branch)
check('two pass validation fails before mutation pass', 'bool for_real = false;' in core and 'f = for_real ? flags : (flags & ~(UNIT_RUNTIME|UNIT_PERSISTENT));' in core and 'Cannot set property %s, or unknown property.' in core)
needle = '                        u->job_running_timeout = u->job_timeout;\n        }'
check('unique minimal core edit target', core.count(needle)==1)
fixed_core = core.replace(needle, '                        u->job_running_timeout = u->job_timeout;\n\n                return r;\n        }')
core_diff = ''.join(difflib.unified_diff(core.splitlines(True),fixed_core.splitlines(True),fromfile='a/src/core/dbus-unit.c',tofile='b/src/core/dbus-unit.c'))
(E/'manager-missing-return.inert.patch.txt').write_text(core_diff)
check('minimal core edit restores return and no other bytes', fixed_core.replace('\n\n                return r;\n        }\n\n        if (streq(name, "JobRunningTimeoutUSec"))','\n        }\n\n        if (streq(name, "JobRunningTimeoutUSec"))')==core)
client = (E/'upstream2595-bus-unit-util.c.txt').read_text()
check('official client maps JobTimeoutSec via rename', '{ "JobTimeoutSec",                         bus_append_parse_sec_rename' in client)
job = (E/'upstream2595-job.c.txt').read_text()
check('job total and running timers use distinct origins', 'usec_add(j->begin_running_usec, j->unit->job_running_timeout)' in job and 'usec_add(j->begin_usec, j->unit->job_timeout)' in job)

# A diagnostic-only delta. It is not an executable candidate, owner entry, grant,
# retry, or compatibility workaround. Original source and all names stay intact.
old = '     assert result.returncode in allowed\n     return result.stdout.decode()'
new = "     assert result.returncode in allowed\n     if argv[0]=='/usr/bin/journalctl':\n      assert not result.stderr,'Journal coverage unconfirmed: '+result.stderr.decode()\n     return result.stdout.decode()"
check('unique journal hardening target', parent.count(old)==1)
fixed_parent = parent.replace(old,new)
ast.parse(fixed_parent)
(E/'journal-stderr-denial.inert.patch.txt').write_text(''.join(difflib.unified_diff(parent.splitlines(True),fixed_parent.splitlines(True),fromfile='a/parent_integration.py',tofile='b/parent_integration.py')))
old_ast = ast.parse(parent)
new_ast = ast.parse(fixed_parent)
old_read = next(n for n in ast.walk(old_ast) if isinstance(n,ast.FunctionDef) and n.name=='read')
new_read = next(n for n in ast.walk(new_ast) if isinstance(n,ast.FunctionDef) and n.name=='read')
# Only compile the selected transport-result validation function with a stub
# subprocess object. No subprocess, filesystem, manager, or target module runs.
def run_read(node, result):
    observations = []
    ns = {'subprocess':SimpleNamespace(run=lambda *a,**k:result),'env':{},'observations':observations}
    selected = ast.fix_missing_locations(ast.Module(body=[node],type_ignores=[]))
    exec(compile(selected,'<selected-read-function-only>','exec'),ns)
    return ns['read'](['/usr/bin/journalctl']), observations
g3 = json.loads((E/'g3-selective-evidence.json').read_bytes())
observed = next(x for x in g3['preflight_observations'] if x['argv'][0]=='/usr/bin/journalctl')
actual = SimpleNamespace(returncode=observed['returncode'],stdout=observed['stdout'].encode(),stderr=observed['stderr'].encode())
check('actual G3 warning reached with rc0 empty stdout', actual.returncode==0 and not actual.stdout and b'Cannot allocate memory' in actual.stderr)
check('original preflight incorrectly accepts actual warning', run_read(old_read,actual)[0]=='')
vectors = [
    ('clean empty',0,b'',b'',True),
    ('actual G3 ENOMEM',actual.returncode,actual.stdout,actual.stderr,False),
    ('any stderr including whitespace',0,b'',b'\n',False),
    ('rc1 empty',1,b'',b'',False),
    ('rc2 empty',2,b'',b'',False),
    ('negative return code',-9,b'',b'',False),
    ('oversize result',0,b'x'*65537,b'',False),
    ('prior entry',0,b'{"USER_UNIT":"already-used"}\n',b'',False),
]
vector_results=[]
for label,rc,out,err,expected in vectors:
    try:
        output, obs = run_read(new_read,SimpleNamespace(returncode=rc,stdout=out,stderr=err))
        accepted = not output.strip() # unchanged caller absence predicate
    except AssertionError:
        accepted = False
    check('journal boundary '+label, accepted==expected)
    vector_results.append({'case':label,'accepted':accepted,'expected':expected})
check('journal delta leaves entire surrounding parent unchanged', fixed_parent.replace(new,old)==parent)
save('journal-boundary-static-vectors.json', {'vectors':vector_results,'limitation':'Selected-function stub tests only; no journal query. No stderr plus empty output still cannot prove complete historical retention or authorize reuse of consumed names.'})

check('original files remain unchanged after static checks', all(digest(p)==h for p,h in pins.items()))
save('static-checks.json', {'status':'PASS_SOURCE_ONLY_NOT_NATIVE_PROOF','checks':checks,'count':len(checks),'inert_deltas':['manager-missing-return.inert.patch.txt','journal-stderr-denial.inert.patch.txt'],'candidate_modules_executed':0,'selected_definition_executed':'parent read() transport-result validator with an in-memory subprocess stub only','resources':{'affinity':sorted(os.sched_getaffinity(0)),'RLIMIT_AS':resource.getrlimit(resource.RLIMIT_AS),'RLIMIT_CPU':resource.getrlimit(resource.RLIMIT_CPU),'wall_alarm_seconds':40,'peak_rss_kib':resource.getrusage(resource.RUSAGE_SELF).ru_maxrss}})
print(json.dumps({'checks':len(checks),'status':'PASS_SOURCE_ONLY_NOT_NATIVE_PROOF','peak_rss_kib':resource.getrusage(resource.RUSAGE_SELF).ru_maxrss}))
