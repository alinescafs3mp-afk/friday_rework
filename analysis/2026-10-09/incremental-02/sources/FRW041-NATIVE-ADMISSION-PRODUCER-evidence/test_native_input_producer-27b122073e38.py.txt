@pytest.fixture(scope='module')
def exact_consumer_view():
    """Actual pinned sources/bytes, virtual fresh targets and original clock.

    No candidate module import, actual grant/clock creation or runtime consumer.
    Binary dependencies read once; this is data compatibility, not native execution.
    """
    contract = p.unique_json((Path(p.__file__).parent / 'consumer-pins.json').read_bytes())
    source = Path(contract['source'])
    package = (source / 'package-pins.json').read_bytes()
    assert p.sha(package) == p.PACKAGE_SHA
    pins = p.unique_json(package)
    files, stats, resolution, source_bytes = {}, {}, {}, {}
    for name in set(pins) | {'package-pins.json'}:
        path = source / name
        data = path.read_bytes()
        assert p.sha(data) == (p.PACKAGE_SHA if name == 'package-pins.json' else pins[name]['sha256'])
        files[str(path)] = data
        stats[str(path)] = SimpleNamespace(st_uid=1000, st_mode=stat.S_IFREG | 0o400)
        resolution[str(path)] = str(path)
        source_bytes[name] = data
    identity = p.unique_json(source_bytes['source-identity.json'])
    dependencies = p.unique_json(source_bytes['source-pins.json'])
    for path, pin in dependencies['pins'].items():
        data = Path(path).read_bytes()
        assert p.sha(data) == pin['sha256'] and len(data) == pin['size']
        files[path] = data
        stats[path] = SimpleNamespace(st_size=len(data))
    resolution[dependencies['python_alias']['path']] = dependencies['python_alias']['resolved']
    live_parent = str(Path(identity['live_directory']).parent)
    stats[live_parent] = SimpleNamespace(st_uid=1000, st_mode=stat.S_IFDIR | 0o700)
    resolution[live_parent] = live_parent
    clock = sample_clock()
    clock_bytes = p.encode(clock)
    hashes = p.string_map(pins, package)
    review_path = '/modeled/independent-source-review.json'
    review = dict(state='INDEPENDENT_OFFLINE_BUILD_SYSTEM_SERVICE_SOURCE_ACCEPTED',
                  package_pins_sha256=p.PACKAGE_SHA, source_task=p.SOURCE_TASK, runtime_grant=False)
    review_bytes = p.encode(review)
    admission = p.admission_data(identity, hashes, review_path, p.sha(review_bytes), clock_bytes)
    for path, data in ((review_path, review_bytes), (identity['trial_clock_path'], clock_bytes),
                       (identity['runtime_admission_path'], p.encode(admission))):
        files[path] = data
        stats[path] = SimpleNamespace(st_uid=1000, st_mode=stat.S_IFREG | 0o400)
        resolution[path] = path
    files['/proc/1/cgroup'] = b'0::/\n'
    files['/proc/sys/kernel/random/boot_id'] = b'test-boot\n'
    receipt = dict(payloads=hashes, independent_review_path=review_path,
                   independent_review_sha256=p.sha(review_bytes))
    host = dict(uid=1000, environment={}, exists={'/.dockerenv': False, '/run/.containerenv': False,
                                                identity['live_directory']: False})
    return dict(source_bytes=source_bytes, contract=contract, identity=identity, files=files, stats=stats,
                resolution=resolution, admission=admission, clock=clock, receipt=receipt,
                now=1001.0, boot='test-boot', host=host)

@pytest.mark.parametrize('mutation', ['nested-source-map', 'source-missing', 'source-wrong',
    'receipt-nested', 'receipt-missing', 'receipt-review', 'clock-hash', 'clock-path', 'live-path',
    'existing-live', 'writable-clock', 'foreign-uid', 'CI', 'container', 'unobserved-read', 'AST-drift'])
def test_actual_consumer_negatives_refuse_before_publish(exact_consumer_view, mutation):
    view = copy.deepcopy(exact_consumer_view)
    admission, identity = view['admission'], view['identity']
    if mutation == 'nested-source-map':
        admission['source_sha256'] = {k: {'sha256': v, 'size': 0} for k, v in admission['source_sha256'].items()}
    if mutation == 'source-missing': admission['source_sha256'].pop('build.py')
    if mutation == 'source-wrong': admission['source_sha256']['build.py'] = '0'*64
    if mutation == 'receipt-nested': view['receipt']['payloads'] = {'build.py': {'sha256': 'a'*64}}
    if mutation == 'receipt-missing': view['receipt']['payloads'] = {}
    if mutation == 'receipt-review': view['receipt']['independent_review_sha256'] = '0'*64
    if mutation == 'clock-hash': admission['clock_sha256'] = '0'*64
    if mutation == 'clock-path': admission['clock_path'] = '/foreign/clock.json'
    if mutation == 'live-path': admission['live_directory'] += '-foreign'
    if mutation == 'existing-live': view['host']['exists'][identity['live_directory']] = True
    if mutation == 'writable-clock': view['stats'][identity['trial_clock_path']].st_mode = stat.S_IFREG | 0o600
    if mutation == 'foreign-uid': view['host']['uid'] = 0
    if mutation == 'CI': view['host']['environment']['CI'] = '1'
    if mutation == 'container': view['host']['exists']['/.dockerenv'] = True
    if mutation == 'unobserved-read': view['files'].pop(str(p.SOURCE / 'build.py'))
    if mutation == 'AST-drift': view['contract']['functions']['run-live-unit.py:admit'] = '0'*64
    view['files'][identity['runtime_admission_path']] = p.encode(admission)
    with pytest.raises((RuntimeError, KeyError)):
        p.actual_consumers(**view)

def test_grant_last_original_bytes_and_reaped_fds(monkeypatch):
    writes, closed = [], []
    original = b'{ "original": "bytes and spacing" }\n'
    clock = sample_clock()
    identity = dict(runtime_admission_path='/fixed/input/admission.json', live_directory='/fixed/live',
                    stage='/fixed/root', nonce=clock['nonce'], unit=clock['unit'])
    prepared = dict(identity=identity, clock=clock, clock_bytes=original, admission={'pinned': True},
                    observations={}, stage_members=[], checks={'model': True}, authorization_sha256='a'*64)
    monkeypatch.setattr(p, 'absent', lambda *args: None)
    monkeypatch.setattr(p, 'clock_contract', lambda *args: None)
    monkeypatch.setattr(p.Path, 'read_text', lambda *args: 'test-boot')
    monkeypatch.setattr(p, 'parent_fd', lambda *args: (91, 'input', []))
    monkeypatch.setattr(p.os, 'listdir', lambda *args: [])
    monkeypatch.setattr(p.os, 'mkdir', lambda *args, **kwargs: writes.append(('mkdir', args)))
    monkeypatch.setattr(p.os, 'open', lambda *args, **kwargs: 92)
    sample = SimpleNamespace(st_dev=2, st_ino=9, st_uid=1000, st_gid=1000, st_mode=stat.S_IFDIR | 0o700,
                             st_nlink=2, st_size=0, st_mtime_ns=1, st_ctime_ns=1)
    monkeypatch.setattr(p.os, 'fstat', lambda *args: sample)
    monkeypatch.setattr(p.os, 'stat', lambda *args, **kwargs: sample)
    monkeypatch.setattr(p.os, 'fsync', lambda fd: None)
    monkeypatch.setattr(p.os, 'close', closed.append)
    monkeypatch.setattr(p, 'write_new', lambda fd, name, data: writes.append((name, data)))
    result = p.publish(prepared)
    assert [row[0] for row in writes] == ['mkdir', 'clock.json', 'admission.json']
    assert writes[1][1] == original and closed == [92, 91]
    assert result['original_clock_bytes_preserved'] is True and result['native_executed'] is False


def test_partial_write_preserves_one_shot_and_closes_fd(monkeypatch):
    calls, closed = [], []
    monkeypatch.setattr(p.os, 'open', lambda *args, **kwargs: calls.append((args, kwargs)) or 94)
    monkeypatch.setattr(p.os, 'write', lambda *args: (_ for _ in ()).throw(OSError('modeled failure')))
    monkeypatch.setattr(p.os, 'close', closed.append)
    with pytest.raises(OSError):
        p.write_new(90, 'clock.json', b'original')
    assert calls[0][0][1] & p.os.O_EXCL and calls[0][0][1] & p.os.O_NOFOLLOW
    assert closed == [94]

def file_stat(**changes):
    values = dict(st_dev=2, st_ino=10, st_mode=stat.S_IFREG | 0o400, st_uid=1000, st_gid=1000,
                  st_nlink=1, st_size=5, st_mtime_ns=1, st_ctime_ns=1)
    values.update(changes)
    return SimpleNamespace(**values)


@pytest.mark.parametrize('change', [{'st_uid': 0}, {'st_nlink': 2},
    {'st_mode': stat.S_IFREG | 0o600}, {'st_mode': stat.S_IFIFO | 0o400},
    {'st_mode': stat.S_IFLNK | 0o400}, {'st_size': 100000000}])
def test_actual_reader_refuses_leaf_metadata_and_releases_fds(change, monkeypatch):
    closed, flags = [], []
    monkeypatch.setattr(p, 'parent_fd', lambda *args: (97, 'input', [(2, 9)]))
    monkeypatch.setattr(p.os, 'open', lambda name, value, **kwargs: flags.append(value) or 98)
    monkeypatch.setattr(p.os, 'fstat', lambda *args: file_stat(**change))
    monkeypatch.setattr(p.os, 'close', closed.append)
    with pytest.raises(RuntimeError, match='input_metadata_shape'):
        p.read_file('/modeled/input')
    assert closed == [98, 97]
    assert flags[0] & p.os.O_NOFOLLOW and flags[0] & p.os.O_NONBLOCK


@pytest.mark.parametrize('key,value', [('st_dev', 3), ('st_ino', 11), ('st_mode', stat.S_IFREG | 0o600),
    ('st_uid', 0), ('st_gid', 0), ('st_nlink', 2), ('st_size', 6), ('st_mtime_ns', 2), ('st_ctime_ns', 2)])
def test_reader_refuses_every_observed_metadata_change(key, value, monkeypatch):
    closed = []
    observations = iter([file_stat(), file_stat(**{key: value})])
    reads = iter([b'bytes', b''])
    monkeypatch.setattr(p, 'parent_fd', lambda *args: (97, 'input', [(2, 9)]))
    monkeypatch.setattr(p.os, 'open', lambda *args, **kwargs: 98)
    monkeypatch.setattr(p.os, 'fstat', lambda *args: next(observations))
    monkeypatch.setattr(p.os, 'read', lambda *args: next(reads))
    monkeypatch.setattr(p.os, 'stat', lambda *args, **kwargs: file_stat())
    monkeypatch.setattr(p.os, 'close', closed.append)
    with pytest.raises(RuntimeError, match='input_metadata_drift'):
        p.read_file('/modeled/input')
    assert closed == [98, 97]


def test_reader_positive_closed_snapshot_and_ancestor_recheck(monkeypatch):
    closed = []
    reads = iter([b'bytes', b''])
    parents = iter([(97, 'input', [(2, 9)]), (96, 'input', [(2, 9)])])
    monkeypatch.setattr(p, 'parent_fd', lambda *args: next(parents))
    monkeypatch.setattr(p.os, 'open', lambda *args, **kwargs: 98)
    monkeypatch.setattr(p.os, 'fstat', lambda *args: file_stat())
    monkeypatch.setattr(p.os, 'read', lambda *args: next(reads))
    monkeypatch.setattr(p.os, 'stat', lambda *args, **kwargs: file_stat())

    actual, observed = p.read_file('/modeled/input')
    assert actual == b'bytes' and observed.st_ino == 10 and closed == [96, 98, 97]
