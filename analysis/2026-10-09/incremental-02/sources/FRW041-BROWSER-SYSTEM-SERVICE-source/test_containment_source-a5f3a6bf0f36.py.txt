def mount_overlay():
    return ('99 1 0:888 / /dev/shm rw - tmpfs tmpfs rw\n'
            '100 99 0:999 / /dev/shm rw,nosuid,nodev,noexec - tmpfs tmpfs rw,size=65536k\n')


@pytest.mark.parametrize('destination',['/dev/shm','/dev/mqueue'])
def test_mount_observation_selects_actual_reached_overlay_in_either_row_order(destination):
    overlay=mount_overlay().replace('/dev/shm',destination)

@pytest.mark.parametrize('destination,mode',[('/dev/shm',0o1777),('/dev/mqueue',0)])
def test_owned_opath_fd_resolves_mount_id_and_matches_current_stat(destination,mode):
    import io, os, stat
    from types import SimpleNamespace
    observed = SimpleNamespace(st_dev=os.makedev(0,999), st_ino=42, st_mode=stat.S_IFDIR|mode)
    def read(path, *args, **kwargs):
        if str(path) == '/proc/self/fdinfo/17': return io.StringIO('flags: 012000000\nmnt_id:\t100\n')
        if str(path) == '/proc/self/mountinfo': return io.StringIO(mount_overlay().replace('/dev/shm',destination))
        raise AssertionError(str(path))
    with patch.object(probe.os,'open',return_value=17) as opened, \
         patch.object(probe.os,'close') as closed, patch.object(probe.os,'fstat',return_value=observed), \
         patch.object(probe.os,'stat',return_value=observed), patch.object(Path,'open',read):
        row = probe.visible_mount_record(destination)
        assert row['mount_id']==100 and row['identity']==[observed.st_dev,42] and row['mode']==mode
        opened.assert_called_once_with(destination, os.O_PATH|os.O_DIRECTORY|os.O_NOFOLLOW|os.O_CLOEXEC)
        closed.assert_called_once_with(17)

def sentinel_attempt(now=None):
    clock={'nonce':'1234567890abcdef','deadline_monotonic':120.0}
    return unit.Attempt(ROOT, ROOT/'live-1234567890abcdef', clock, now=now or (lambda:10.0))


class SentinelListener:
    def __init__(self, attempt, faults=()):
        self.attempt=attempt; self.faults=set(faults); self.events=[]; self.named=False
    def fileno(self): return 63
    def bind(self,path):
        self.events.append('bind')
        s=self.attempt.host_sentinel
        assert s['intent'] is True and s['listener'] is self
        if 'refuse' in self.faults: raise OSError(unit.errno.EADDRINUSE,'precise refusal')
        self.named=True
        if 'unknown_bind' in self.faults: raise RuntimeError('unknown_bind_outcome')
    def listen(self,n):
        assert self.attempt.host_sentinel['identity']==[3,200]
        self.events.append('listen')
        if 'listen' in self.faults: raise RuntimeError('original_listen_failure')
    def setblocking(self,value):
        if 'setblocking' in self.faults: raise RuntimeError('original_setblocking_failure')
    def close(self):
        self.events.append('close')
        if 'close' in self.faults: raise RuntimeError('listener_close_failure')


def sentinel_patches(attempt,faults=()):
    from contextlib import ExitStack
    from types import SimpleNamespace
    import stat
    listener=SentinelListener(attempt,faults); stack=ExitStack()
    def lstat(path):
        if not listener.named: raise FileNotFoundError(str(path))
        if 'identity' in listener.faults: raise RuntimeError('original_postbind_identity_failure')
        return SimpleNamespace(st_dev=3,st_ino=201 if 'replacement' in listener.faults else 200,
                               st_mode=stat.S_IFSOCK|0o600,st_uid=1001 if 'foreign_owner' in listener.faults else 1000,st_nlink=1)
    def unlink(path):
        listener.events.append('unlink')
        if 'unlink' in listener.faults: raise PermissionError('owned_unlink_failed')
        listener.named='post_unlink_replacement' in listener.faults
    stack.enter_context(patch.object(Path,'exists',return_value=False))
    stack.enter_context(patch.object(Path,'is_symlink',return_value=False))
    stack.enter_context(patch.object(Path,'lstat',lstat))
    stack.enter_context(patch.object(Path,'unlink',unlink))
    stack.enter_context(patch.object(unit.socket,'socket',return_value=listener))
    return stack,listener


def test_host_sentinel_exact_creation_identity_close_unlink_absence():
    attempt=sentinel_attempt(); stack,listener=sentinel_patches(attempt)
    with stack:
        assert attempt.create_host_sentinel() is listener
        assert attempt.host_sentinel_unresolved and attempt.unresolved
        assert attempt.host_sentinel['identity']==[3,200]
        assert attempt.host_sentinel['identity_verified'] and attempt.host_sentinel['listener_fd']==63

def test_native_start_preserves_original_postbind_failure_and_unknown_latch():
    from types import SimpleNamespace
    import stat
    attempt=sentinel_attempt()
    class InertOtherListener:
        def bind(self,*args): pass
        def listen(self,*args): pass
        def setblocking(self,*args): pass
        def close(self): pass
    owned=SentinelListener(attempt,['identity'])
    fifo=SimpleNamespace(st_mode=stat.S_IFIFO|0o600,st_uid=1000,st_nlink=1,st_dev=3,st_ino=77)
    def lstat(path):
        if str(path).endswith('.fifo'): return fifo
        raise RuntimeError('original_postbind_identity_failure')
    with patch.object(Path,'exists',return_value=False),patch.object(Path,'is_symlink',return_value=False), \
         patch.object(Path,'lstat',lstat),patch.object(unit.os,'open',side_effect=[10,11]), \
         patch.object(unit.os,'fstat',return_value=fifo),patch.object(unit.os,'close'), \
         patch.object(unit.socket,'socket',side_effect=[InertOtherListener(),InertOtherListener(),owned]), \
         patch.object(unit.subprocess,'Popen',side_effect=AssertionError('no native start')), \
         patch.object(Path,'unlink',side_effect=AssertionError('no blind unlink')):
        row=attempt.native_start(['controlled-start'],20)
        assert row['handshake_error']=='original_postbind_identity_failure'
        assert row['cleanup_errors']==['host_filesystem_sentinel_identity_unknown']
        assert row['host_sentinel']['intent'] and row['host_sentinel']['identity'] is None
        assert attempt.host_sentinel['listener'] is owned and attempt.unresolved
        assert row['timeout'] is True and attempt.clock['deadline_monotonic']==120.0


# Explicit controlled system-manager fixture derived from pinned installed
# typed schema. Historical user-manager JSON separately preserved byte-for-byte.
# No actual system observation/runtime acceptance is claimed.
def controlled_system_prestart_fixture():
    row=json.loads((ROOT/'prestart-observation.json').read_text())
    state=dict(line.split('=',1) for line in row['loaded_stdout'].splitlines() if '=' in line)
    live=ROOT/Path(row['old_live']).name
    def translate(value):
        if isinstance(value,str): return value.replace(row['old_root'],str(ROOT))
        if isinstance(value,list): return [translate(v) for v in value]
        if isinstance(value,dict): return {k:translate(v) for k,v in value.items()}
        return value
    state=translate(state)
    typed=translate(json.loads(row['typed_stdout']))
    evidence=dict(typed,unit=row['unit'],manager_scope='system',object_path=unit.unit_object_path(row['unit']))
    _,argv=unit.render(ROOT,live,row['unit'])
    return state,row['unit'],Path(state['FragmentPath']),live,argv,evidence
