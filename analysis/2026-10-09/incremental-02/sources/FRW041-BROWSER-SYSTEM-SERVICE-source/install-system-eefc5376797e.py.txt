"""Exact administrator staging/install procedure. Never called by source task.
Run only as the checked ROOT-owned bootstrap copied by INSTALL-ENTRY.sh.
Normal PID1 dependency installation; no start, enable, policy/profile relaxations.
Failure preserves owned root artifacts and concrete identities for reconciliation.
"""
import hashlib
import json
import os
from pathlib import Path
import re
import resource
import stat
import subprocess
import sys
import time

SOURCE=Path('<SOL_WORK>/FRW041-BROWSER-SYSTEM-SERVICE-source')
STAGE=Path('/run/friday-browser-3f4901af6bb7a95f')
UNIT='friday-browser-prerequisite-3f4901af6bb7a95f.service'
LIVE=Path('<SOL_WORK>/FRW041-BROWSER-SYSTEM-SERVICE-source/live-3f4901af6bb7a95f')
POLICY=Path('/etc/sudoers.d/friday-browser-3f4901af6bb7a95f')
FRAGMENT=Path('/run/systemd/system')/UNIT
ENV={'PATH':'/usr/bin:/bin','LANG':'C.UTF-8','HOME':'/root'}
BUNDLE_SHA='336dec20868c2ddeb9253cb4f807c4528cc1100175f4eeb94b916fe862b61a0b'


def require(value,message):
    if not value: raise RuntimeError(message)


def data(path,owner,mode=None,maximum=2000000,single_link=True):
    p=Path(path);require(p==p.resolve(strict=True),'symlink_input')
    fd=os.open(p,os.O_RDONLY|os.O_NOFOLLOW|os.O_CLOEXEC)
    try:
        s=os.fstat(fd)
        require(stat.S_ISREG(s.st_mode) and s.st_uid==owner and (owner!=0 or s.st_gid==0) and (not single_link or s.st_nlink==1) and not s.st_mode&0o022
                and s.st_size<=maximum and (mode is None or stat.S_IMODE(s.st_mode)==mode),'input_boundary')
        b=os.read(fd,maximum+1);t=os.fstat(fd);u=p.lstat()
        key=lambda s:(s.st_dev,s.st_ino,s.st_mtime_ns,s.st_size,s.st_mode,s.st_uid,s.st_gid)
        require(key(s)==key(t)==key(u) and len(b)==s.st_size,'input_changed')
        return b,s
    finally:os.close(fd)


def digest(b):return hashlib.sha256(b).hexdigest()


def rootdir(path):
    for q in [path,*path.parents]:
        s=q.lstat();require(q==q.resolve(strict=True) and stat.S_ISDIR(s.st_mode)
            and s.st_uid==s.st_gid==0 and not s.st_mode&0o022,'root_directory')


def absent(path):
    try:path.lstat()
    except FileNotFoundError:return
    raise RuntimeError('collision:'+str(path))


def create(path,b,mode):
    fd=os.open(path,os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW|os.O_CLOEXEC,mode)
    try:
        os.fchmod(fd,mode);os.fchown(fd,0,0)
        with os.fdopen(fd,'wb',closefd=False) as f:f.write(b);f.flush();os.fsync(f.fileno())
        s=os.fstat(fd);return [s.st_dev,s.st_ino]
    finally:os.close(fd)


def main():
    require(os.getuid()==os.geteuid()==0 and len(sys.argv)==2,'root_bootstrap_review_path_required')
    os.umask(0o077);os.sched_setaffinity(0,{0,1});resource.setrlimit(resource.RLIMIT_AS,(2147483648,2147483648))
    rootdir(STAGE);rootdir(FRAGMENT.parent);rootdir(POLICY.parent)
    bootstrap=Path(__file__)
    require(bootstrap==STAGE/'bootstrap.py','root_staged_bootstrap_only')
    own,s=data(bootstrap,0,0o444)
    require(set(p.name for p in STAGE.iterdir())=={'bootstrap.py'},'initial_stage_collision')
    # External final review path only selects a nonexecuted data record. The
    # installed commands themselves expose no caller path/argv parameters.
    review_path=Path(sys.argv[1]);b,_=data(review_path,1000,0o400);review=json.loads(b)
    require(review['state']=='INDEPENDENT_BROWSER_SYSTEM_SERVICE_SOURCE_ACCEPTED'
            and review['source_task']=='FRW041-BROWSER-SYSTEM-SERVICE-SOL#1'
            and review['runtime_grant'] is False,'independent_final_source_review_required')
    b,_=data(SOURCE/'package-pins.json',1000,0o400);pins=json.loads(b)
    require(digest(b)==review['package_pins_sha256'],'reviewed_package_drift')
    payloads={}
    for name,pin in pins.items():
        require(re.fullmatch(r'[A-Za-z0-9_.-]+',name),'package_leaf_required')
        b,_=data(SOURCE/name,1000,0o400);require(digest(b)==pin['sha256'],'source_pin_changed:'+name);payloads[name]=b
    require(digest(own)==pins['install-system.py']['sha256'],'bootstrap_pin_changed')
    b,_=data(SOURCE/'install-bundle.json',1000,0o400);require(digest(b)==BUNDLE_SHA,'install_bundle_changed');bundle=json.loads(b)
    require(bundle['stage']==str(STAGE) and bundle['unit']==UNIT and bundle['fragment']==str(FRAGMENT)
            and bundle['policy']==str(POLICY) and bundle['source']==str(SOURCE) and bundle['live']==str(LIVE),'fixed_bundle_contour')
    require(set(bundle['root_assets'])=={'admin-control.py','preflight','start','stop','cleanup'},'fixed_asset_inventory')
    for p in (FRAGMENT, POLICY, Path('/etc/systemd/system')/UNIT,FRAGMENT.with_name(UNIT+'.d'),Path('/etc/systemd/system')/(UNIT+'.d')):absent(p)
    dependency_spec=json.loads(payloads['source-pins.json'])['administrative_executables']
    for alias,pin in dependency_spec.items():
        q=Path(alias).resolve(strict=True);require(str(q)==pin['resolved'],'admin_dependency_alias')
        rootdir(q.parent);b,metadata=data(q,0,maximum=20000000,single_link=False)
        require([metadata.st_uid,metadata.st_gid,stat.S_IMODE(metadata.st_mode),metadata.st_nlink]
            == [pin['uid'],pin['gid'],pin['mode'],pin['nlink']],'admin_dependency_boundary')
        require(digest(b)==pin['sha256'] and len(b)==pin['size'],'admin_dependency_pin')
    started=time.monotonic()
    # Administrative preparation has its own bounded call graph, no trial
    # clock or browser execution. No normal task budget is reset here.
    end=started+60
    def run(argv,seconds=3,allowed=(0,)):
        require(time.monotonic()+seconds<end,'admin_preparation_clock_exhausted')
        p=subprocess.run(argv,stdin=subprocess.DEVNULL,stdout=subprocess.PIPE,stderr=subprocess.PIPE,env=ENV,close_fds=True,timeout=seconds)
        require(len(p.stdout)+len(p.stderr)<1048576 and p.returncode in allowed and not p.stderr,'administration_command_failed')
        return p.stdout.decode()
    def manager():
        pre=['/usr/bin/busctl','--system','--no-pager','--json=short','--auto-start=no','--allow-interactive-authorization=no','--timeout=2',
             'call','org.freedesktop.DBus','/org/freedesktop/DBus','org.freedesktop.DBus']
        def one(text,sig):
            x=json.loads(text);require(x['type']==sig and len(x['data'])==1,'manager_reply');return x['data'][0]
        name=one(run(pre+['GetNameOwner','s','org.freedesktop.systemd1'],2),'s')
        require(re.fullmatch(r':[0-9]+\.[0-9]+',name),'unique_bus_owner')
        props=one(run(pre+['GetConnectionCredentials','s',name],2),'a{sv}')
        require(props['ProcessID']=={'type':'u','data':1} and props['UnixUserID']=={'type':'u','data':0},'PID1_system_bus_required')
        return dict(scope='system',pid=1,uid=0,bus_owner=name,boot_id=Path('/proc/sys/kernel/random/boot_id').read_text().strip(),
                    start_ticks=int(Path('/proc/1/stat').read_text().rsplit(')',1)[1].split()[19]))
    before=run(['/usr/bin/systemctl','--system','--no-pager','show',UNIT,'--property=Id,LoadState,ActiveState,MainPID,ControlPID,Job,FragmentPath,DropInPaths'],allowed=(0,4))
    state=dict(x.split('=',1) for x in before.splitlines())
    require(state['Id']==UNIT and state['LoadState']=='not-found' and state['ActiveState']=='inactive'
            and state['MainPID']==state['ControlPID']=='0' and state['Job'] in ('','0') and not state['FragmentPath'] and not state['DropInPaths'],'existing_unit_collision')
    owner=manager()
    require(run(['/usr/bin/id','-u','jericho']).strip()=='1000' and run(['/usr/bin/id','-g','jericho']).strip()=='1000','payload_account_changed')
    fragment=payloads['browser.service'];require(digest(fragment)==bundle['fragment_sha256'],'fragment_hash')
    # Parse-only native validation. It does not start a service.
    run(['/usr/bin/systemd-analyze','verify','--man=no',str(SOURCE/'browser.service')],5)
    assets={}
    assets['bootstrap.py']=dict(sha256=digest(own),identity=[s.st_dev,s.st_ino],mode=0o444)
    for name,pin in bundle['root_assets'].items():
        require(digest(payloads[name])==pin['sha256'] and pin['mode'] in (0o444,0o555),'root_asset_hash')
        identity=create(STAGE/name,payloads[name],pin['mode']);assets[name]=dict(pin,identity=identity)
    create(STAGE/'control.lock',b'',0o600)
    fragment_id=create(FRAGMENT,fragment,0o444)
    # visudo validates exact isolated policy before publication; whole policy
    # parse afterward and exact user no-argument command admission checked.
    policy=payloads['sudoers'];require(digest(policy)==bundle['policy_sha256'],'policy_hash')
    staged_policy=STAGE/'policy.pending';create(staged_policy,policy,0o440)
    run(['/usr/sbin/visudo','-c','-f',str(staged_policy)])
    policy_id=create(POLICY,policy,0o440);staged_policy.unlink()
    run(['/usr/sbin/visudo','-c'])
    run(['/usr/bin/systemctl','--system','--no-pager','daemon-reload'],5)
    out=run(['/usr/bin/systemctl','--system','--no-pager','show',UNIT,'--property=Id,LoadState,ActiveState,MainPID,ControlPID,Job,InvocationID,ControlGroup,FragmentPath,DropInPaths,Transient,NeedDaemonReload'])
    loaded=dict(x.split('=',1) for x in out.splitlines())
    require(loaded['Id']==UNIT and loaded['LoadState']=='loaded' and loaded['ActiveState']=='inactive'
            and loaded['MainPID']==loaded['ControlPID']=='0' and loaded['Job'] in ('','0')
            and not loaded['InvocationID'] and not loaded['ControlGroup']
            and loaded['FragmentPath']==str(FRAGMENT) and not loaded['DropInPaths']
            and loaded['Transient']=='no' and loaded['NeedDaemonReload']=='no','loaded_exact_ordinary_unit_required')
    for op in ('preflight','start','stop','cleanup'):
        run(['/usr/sbin/runuser','-u','jericho','--','/usr/bin/sudo','-n','-l','--',str(STAGE/op)])
    require(manager()==owner,'manager_changed_during_install')
    receipt=dict(state='EXACT_ROOT_DEPENDENCIES_INSTALLED_NOT_STARTED',unit=UNIT,stage=str(STAGE),source=str(SOURCE),live=str(LIVE),
        manager=owner,assets=assets,administrative_executables=dependency_spec,fragment_sha256=digest(fragment),fragment_identity=fragment_id,
        policy_sha256=digest(policy),policy_identity=policy_id,
        payloads={**{name:pin['sha256'] for name,pin in pins.items()},'package-pins.json':review['package_pins_sha256']},
        independent_review_path=str(review_path),independent_review_sha256=digest(review_path.read_bytes()),
        loaded=loaded,installation_elapsed_seconds=time.monotonic()-started,trial_started=False)
    create(STAGE/'receipt.json',(json.dumps(receipt,indent=2)+'\n').encode(),0o444)
    require(time.monotonic()<end,'installation_late_unknown')
    print(json.dumps(receipt),flush=True)


if __name__=='__main__':main()
