#!/usr/bin/bash
# SOURCE PROPOSAL ONLY. Later independently reviewed administration; NOT runtime.
set -euo pipefail
[[ "$#" == 2 && "$UID" == 1000 ]] || exit 2
source_directory='<SOL_WORK>/FRW041-BROWSER-SYSTEM-SERVICE-source'
stage='/run/friday-browser-3f4901af6bb7a95f'
review_path="$1"
review_sha="$2"
/usr/bin/env -i PATH=/usr/bin:/bin LANG=C.UTF-8 /usr/bin/python3 -I -B - "$source_directory" "$review_path" "$review_sha" <<'CHECK'
from pathlib import Path
import hashlib,json,os,stat,sys
root,review=Path(sys.argv[1]),Path(sys.argv[2]);expected=sys.argv[3]
assert len(expected)==64 and all(x in '0123456789abcdef' for x in expected)
assert review==review.resolve(strict=True)
s=review.stat();assert s.st_uid==1000 and s.st_nlink==1 and stat.S_IMODE(s.st_mode)==0o400
assert hashlib.sha256(review.read_bytes()).hexdigest()==expected
r=json.loads(review.read_text());assert r['state']=='INDEPENDENT_BROWSER_SYSTEM_SERVICE_SOURCE_ACCEPTED'
assert r['source_task']=='FRW041-BROWSER-SYSTEM-SERVICE-SOL#1' and r['runtime_grant'] is False
assert hashlib.sha256((root/'package-pins.json').read_bytes()).hexdigest()==r['package_pins_sha256']
for n,v in json.loads((root/'package-pins.json').read_text()).items():
 p=root/n;assert p==p.resolve(strict=True);s=p.stat()
 assert s.st_uid==1000 and s.st_nlink==1 and stat.S_IMODE(s.st_mode)==0o400
 assert hashlib.sha256(p.read_bytes()).hexdigest()==v['sha256']
assert hashlib.sha256((root/'install-system.py').read_bytes()).hexdigest()=='1bda682ae3160787f9cd33fe3da23381edd6217c5921ad2fdfa589323ec3ff4f'
for p in (Path('/run/friday-browser-3f4901af6bb7a95f'),Path('/run/systemd/system/friday-browser-prerequisite-3f4901af6bb7a95f.service')):
 try:p.lstat()
 except FileNotFoundError:continue
 raise RuntimeError('existing administrative contour; retain and reconcile')
CHECK
# Supported ordinary administration, exact root-staged path, no service start.
/usr/bin/sudo -- /usr/bin/mkdir --mode=0755 -- "$stage"
/usr/bin/sudo -- /usr/bin/install -o root -g root -m 0444 -- "$source_directory/install-system.py" "$stage/bootstrap.py"
/usr/bin/env -i PATH=/usr/bin:/bin LANG=C.UTF-8 /usr/bin/python3 -I -B - "$stage/bootstrap.py" <<'PIN'
from pathlib import Path
import hashlib,stat,sys
p=Path(sys.argv[1]);assert p==p.resolve(strict=True);s=p.stat()
assert s.st_uid==s.st_gid==0 and s.st_nlink==1 and stat.S_IMODE(s.st_mode)==0o444
assert hashlib.sha256(p.read_bytes()).hexdigest()=='1bda682ae3160787f9cd33fe3da23381edd6217c5921ad2fdfa589323ec3ff4f'
PIN
/usr/bin/sudo -- /usr/bin/env -i PATH=/usr/bin:/bin LANG=C.UTF-8 HOME=/root /usr/bin/python3 -I -B "$stage/bootstrap.py" "$review_path"
