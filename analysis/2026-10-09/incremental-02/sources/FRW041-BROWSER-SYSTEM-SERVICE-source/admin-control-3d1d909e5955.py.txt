"""Fixed one-unit administrative dependency; root-staged only, no daemon.
No arbitrary path/unit/property/command arguments; called by three fixed wrappers.
Source proposal only. Install/review/runtime grants are separate.
"""
import fcntl
import hashlib
import json
import math
import os
from pathlib import Path
import re
import resource
import stat
import subprocess
import sys
import time

# Filled as literal reviewed paths by source packaging, never caller arguments.
STAGE = Path('/run/friday-browser-3f4901af6bb7a95f')
SOURCE = Path('<SOL_WORK>/FRW041-BROWSER-SYSTEM-SERVICE-source')
LIVE = Path('<SOL_WORK>/FRW041-BROWSER-SYSTEM-SERVICE-source/live-3f4901af6bb7a95f')
UNIT = 'friday-browser-prerequisite-3f4901af6bb7a95f.service'
NONCE = '3f4901af6bb7a95f'
FRAGMENT = Path('/run/systemd/system') / UNIT
POLICY = Path('/etc/sudoers.d/friday-browser-3f4901af6bb7a95f')
CGROUP = '/system.slice/' + UNIT
ENV = {'PATH':'/usr/bin:/bin', 'LANG':'C.UTF-8', 'HOME':'/root'}
STATE_KEYS = 'Id,LoadState,ActiveState,SubState,MainPID,ControlPID,ControlGroup,InvocationID,Job,Result,Transient,FragmentPath,DropInPaths,NeedDaemonReload'


def require(value, message):
    if not value: raise RuntimeError(message)


def sha(data):
    return hashlib.sha256(data).hexdigest()


def read_exact(path, uid, mode=None, maximum=1048576, single_link=True):
    path = Path(path)
    require(path.is_absolute() and path == path.resolve(strict=True), 'canonical_path_required')
    fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_CLOEXEC)
    try:
        a = os.fstat(fd)
        require(stat.S_ISREG(a.st_mode) and a.st_uid == uid and (uid != 0 or a.st_gid == 0) and (not single_link or a.st_nlink == 1)
                and not a.st_mode & 0o022 and a.st_size <= maximum
                and (mode is None or stat.S_IMODE(a.st_mode) == mode), 'input_owner_mode_type')
        data = os.read(fd, maximum + 1)
        b = os.fstat(fd); c = path.lstat()
        key = lambda x: (x.st_dev,x.st_ino,x.st_size,x.st_mtime_ns,x.st_mode,x.st_uid,x.st_gid)
        require(key(a) == key(b) == key(c) and len(data) == a.st_size, 'input_changed_or_overflow')
        return data, a
    finally: os.close(fd)


def root_directory(path):
    path = Path(path)
    for q in [path, *path.parents]:
        s = q.lstat()
        require(q == q.resolve(strict=True) and stat.S_ISDIR(s.st_mode)
                and s.st_uid == s.st_gid == 0 and not s.st_mode & 0o022, 'root_parent_boundary')


def unique_json(data):
    def pairs(rows):
        value = {}
        for k,v in rows:
            require(k not in value, 'duplicate_json_key'); value[k] = v
        return value
    return json.loads(data, object_pairs_hook=pairs,
                      parse_constant=lambda x: require(False, 'nonfinite_json'))


def clock_valid(clock):
    require(isinstance(clock,dict) and clock['unit'] == UNIT and clock['nonce'] == NONCE
            and clock['assignment'] == 'FRW041-BROWSER-NATIVE-CONTAINMENT-TRIAL'
            and clock['native_effects_authorized_by'] == 'Astra' and clock['budget_seconds'] == 120
            and all(type(clock[k]) in (int,float) and math.isfinite(clock[k]) for k in ('started_monotonic','deadline_monotonic'))
            and clock['deadline_monotonic'] == clock['started_monotonic'] + 120
            and clock['boot_id'] == Path('/proc/sys/kernel/random/boot_id').read_text().strip(), 'original_clock_invalid')
    return clock



def runtime_admission_valid(value, receipt, clock_hash):
    require(value.get('runtime_authorized') is True and value.get('allow_one_browser_prerequisite') is True
            and value.get('source_task') == 'FRW041-BROWSER-SYSTEM-SERVICE-SOL#1'
            and value.get('independent_review_sha256') == receipt['independent_review_sha256']
            and value.get('independent_review_path') == receipt['independent_review_path']
            and value.get('source_sha256') == receipt['payloads']
            and value.get('clock_sha256') == clock_hash and value.get('live_directory') == str(LIVE),
            'separate_exact_runtime_admission_required')
    return True


def kernel_closed():
    p = Path('/sys/fs/cgroup') / CGROUP.lstrip('/')
    try: p.lstat()
    except FileNotFoundError: return True
    require(p == p.resolve(strict=True), 'cgroup_symlink')
    events = dict(row.split() for row in (p/'cgroup.events').read_text().splitlines())
    return events.get('populated') == '0' and not (p/'cgroup.procs').read_text().strip()


def fixed_argv(operation):
    allowed = {
        'start':['/usr/bin/systemctl','--system','--no-pager','--wait','start',UNIT],
        'stop':['/usr/bin/systemctl','--system','--no-pager','--job-mode=replace','stop',UNIT],
        'reset':['/usr/bin/systemctl','--system','--no-pager','reset-failed',UNIT],
        'reload':['/usr/bin/systemctl','--system','--no-pager','daemon-reload'],
        'show':['/usr/bin/systemctl','--system','--no-pager','show',UNIT,'--property='+STATE_KEYS]}
    require(operation in allowed, 'fixed_operation_required')
    return allowed[operation]


def manager_valid(value):
    require(value['scope']=='system' and type(value['pid']) is int and value['pid']==1
            and type(value['uid']) is int and value['uid']==0
            and type(value['start_ticks']) is int and value['start_ticks']>0
            and re.fullmatch(r':[0-9]+\.[0-9]+',value['bus_owner']), 'PID1_required')
    return value


class Control:
    def __init__(self, operation):
        require(operation in ('preflight','start','stop','cleanup'), 'no_arbitrary_operation')
        self.operation = operation
        root_directory(STAGE); root_directory(FRAGMENT.parent); root_directory(POLICY.parent)
        b,_ = read_exact(STAGE/'receipt.json',0,0o444)
        self.receipt = unique_json(b)
        require(self.receipt['unit']==UNIT and self.receipt['stage']==str(STAGE)
                and self.receipt['source']==str(SOURCE) and self.receipt['live']==str(LIVE), 'fixed_receipt_contour')
        self.verify_assets()
        # The root snapshot wins after a start intent. Mutable user clock can
        # never revoke stop authority or introduce a replacement deadline.
        consumed = STAGE/'start-intent.json'
        if consumed.exists():
            b,_ = read_exact(consumed,0,0o600)
            saved = unique_json(b); self.clock_bytes=bytes.fromhex(saved['clock_hex'])
            require(sha(self.clock_bytes)==saved['clock_sha256'], 'root_clock_snapshot')
        else:
            self.clock_bytes,_ = read_exact(SOURCE/('trial-clock-'+NONCE+'.json'),1000,0o400)
        self.clock=clock_valid(unique_json(self.clock_bytes))
        self.started=time.monotonic()
        cap = {'preflight':10,'start':65,'stop':10,'cleanup':15}[operation]
        # Same existing 55-second post-start suffix; no new grace period.
        self.end=min(self.clock['deadline_monotonic']-(57 if operation=='start' else 0),self.started+cap)
        self.check(4)
        for alias,pin in self.receipt['administrative_executables'].items():
            q=Path(alias).resolve(strict=True)
            require(str(q)==pin['resolved'], 'admin_executable_alias_changed')
            root_directory(q.parent)
            b,metadata=read_exact(q,0,maximum=20000000,single_link=False)
            require([metadata.st_uid,metadata.st_gid,stat.S_IMODE(metadata.st_mode),metadata.st_nlink]
                    == [pin['uid'],pin['gid'],pin['mode'],pin['nlink']], 'admin_executable_boundary')
            require(sha(b)==pin['sha256'] and len(b)==pin['size'], 'admin_executable_changed')
        self.records=[]
        birth=int(Path('/proc/self/stat').read_text().rsplit(')',1)[1].split()[19])
        self.operation_identity=dict(operation=operation,unit=UNIT,nonce=NONCE,boot_id=self.clock['boot_id'],
            pid=os.getpid(),start_ticks=birth,started_monotonic=self.started,deadline_monotonic=self.clock['deadline_monotonic'])
        fd=os.open(STAGE/('operation-'+operation+'.json'),os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o444)
        with os.fdopen(fd,'w') as stream:
            os.fchmod(stream.fileno(),0o444);json.dump(self.operation_identity,stream);stream.flush();os.fsync(stream.fileno())

    def check(self, reserve=0):
        require(self.clock['started_monotonic'] <= time.monotonic() < self.end-reserve, 'original_control_clock_exhausted')

    def verify_assets(self):
        require(set(self.receipt['assets']) == {'admin-control.py','preflight','start','stop','cleanup','bootstrap.py'}, 'exact_root_inventory')
        for name,pin in self.receipt['assets'].items():
            data,s=read_exact(STAGE/name,0,pin['mode'])
            require(sha(data)==pin['sha256'] and [s.st_dev,s.st_ino]==pin['identity'], 'root_control_asset_changed')
        data,s=read_exact(FRAGMENT,0,0o444)
        require(sha(data)==self.receipt['fragment_sha256'] and [s.st_dev,s.st_ino]==self.receipt['fragment_identity'], 'foreign_fragment')
        data,s=read_exact(POLICY,0,0o440)
        require(sha(data)==self.receipt['policy_sha256'] and [s.st_dev,s.st_ino]==self.receipt['policy_identity'], 'foreign_policy')
        require(set(p.name for p in STAGE.iterdir()) <= set(self.receipt['assets']) | {'receipt.json','control.lock','start-intent.json','operation-preflight.json','operation-start.json','operation-stop.json','operation-cleanup.json'}, 'foreign_stage_artifact')

    def run(self, argv, seconds):
        self.check(seconds)
        started=time.monotonic()
        try:
            p=subprocess.run(argv,stdin=subprocess.DEVNULL,stdout=subprocess.PIPE,stderr=subprocess.PIPE,
                             env=ENV,close_fds=True,timeout=seconds)
            row={'argv':argv,'returncode':p.returncode,'stdout':p.stdout.decode(),'stderr':p.stderr.decode(),'timeout':False,'client_reaped':True}
        except subprocess.TimeoutExpired:
            # This finite root-owned client is killed/reaped by run(); unit
            # and sudo transport remain separately reconciled, never retried.
            raise RuntimeError('root_native_control_timeout_unknown')
        require(len(p.stdout)+len(p.stderr)<=1048576,'control_output_limit')
        self.check();row.update(started_monotonic=started,completed_monotonic=time.monotonic());self.records.append(row)
        require(p.returncode in ((0,4) if 'show' in argv else (0,)) and not p.stderr,'root_control_error')
        return row

    def manager(self):
        prefix=['/usr/bin/busctl','--system','--no-pager','--json=short','--auto-start=no','--allow-interactive-authorization=no','--timeout=2',
                'call','org.freedesktop.DBus','/org/freedesktop/DBus','org.freedesktop.DBus']
        def reply(row,sig):
            x=unique_json(row['stdout']);require(set(x)=={'type','data'} and x['type']==sig and len(x['data'])==1,'manager_reply');return x['data'][0]
        name=reply(self.run(prefix+['GetNameOwner','s','org.freedesktop.systemd1'],2),'s')
        require(isinstance(name,str) and re.fullmatch(r':[0-9]+\.[0-9]+',name),'bus_owner')
        p=reply(self.run(prefix+['GetConnectionCredentials','s',name],2),'a{sv}')
        def field(k):
            require(set(p[k])=={'type','data'} and p[k]['type']=='u' and type(p[k]['data']) is int,'manager_credential');return p[k]['data']
        return manager_valid(dict(scope='system',pid=field('ProcessID'),uid=field('UnixUserID'),bus_owner=name,
             start_ticks=int(Path('/proc/1/stat').read_text().rsplit(')',1)[1].split()[19]),
             boot_id=Path('/proc/sys/kernel/random/boot_id').read_text().strip()))

    def show(self):
        row=self.run(fixed_argv('show'),3)
        lines=row['stdout'].splitlines();require(all('=' in line for line in lines),'state_format')
        result=dict(line.split('=',1) for line in lines);require(len(result)==len(lines),'state_duplicates');return result

    def stopped(self, state):
        require(state.get('Id')==UNIT and state.get('FragmentPath')==str(FRAGMENT)
                and state.get('Transient')=='no' and not state.get('DropInPaths')
                and state.get('ActiveState') in ('inactive','failed')
                and state.get('MainPID')==state.get('ControlPID')=='0'
                and state.get('Job') in ('','0') and state.get('ControlGroup') in ('',CGROUP)
                and kernel_closed(), 'actual_cgroup_stop_required')

    def execute(self):
        manager=self.manager();require(manager==self.receipt['manager'],'manager_changed')
        self.verify_assets()
        if self.operation=='preflight':
            state=self.show();self.stopped(state)
            require(state['ActiveState']=='inactive' and not state['InvocationID'] and not state['ControlGroup'], 'prestart_preflight_execution')
            return dict(state='EXACT_ADMINISTRATIVE_PREFLIGHT_PASS',unit=UNIT,manager=manager,clock_sha256=sha(self.clock_bytes),receipt=self.receipt,records=self.records)
        if self.operation=='start':
            b,_=read_exact(SOURCE/('runtime-admission-'+NONCE+'.json'),1000,0o400)
            runtime_admission_valid(unique_json(b),self.receipt,sha(self.clock_bytes))
            for name,pin in self.receipt['payloads'].items():
                b,s=read_exact(SOURCE/name,1000,0o400,maximum=2000000)
                require(sha(b)==pin,'reviewed_source_drift')
            # Intent consumed before any systemd start call; no second start.
            fd=os.open(STAGE/'start-intent.json',os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o600)
            with os.fdopen(fd,'w') as f:
                json.dump(dict(clock_hex=self.clock_bytes.hex(),clock_sha256=sha(self.clock_bytes),manager=manager),f)
                f.flush();os.fsync(f.fileno())
            self.check(20)
            row=self.run(fixed_argv('start'),min(65,self.end-time.monotonic()-0.1))
            return dict(state='WAIT_CLIENT_ENDED',unit=UNIT,manager=manager,clock_sha256=sha(self.clock_bytes),records=self.records)
        if self.operation=='stop':
            # Native TimeoutStopSec=5 remains; full root command branch<=10.
            self.run(fixed_argv('stop'),5.9)
            return dict(state='STOP_REQUEST_CLIENT_ENDED',unit=UNIT,manager=manager,clock_sha256=sha(self.clock_bytes),records=self.records)
        final=self.show();self.stopped(final)
        if final['ActiveState']=='failed': self.run(fixed_argv('reset'),2)
        self.verify_assets();self.check(6)
        FRAGMENT.unlink()
        self.run(fixed_argv('reload'),2)
        after=self.show()
        require(after.get('Id')==UNIT and after.get('LoadState')=='not-found'
                and after.get('ActiveState')=='inactive' and after.get('SubState')=='dead'
                and after.get('MainPID')==after.get('ControlPID')=='0'
                and after.get('Job') in ('','0') and not after.get('InvocationID')
                and not after.get('ControlGroup') and not after.get('FragmentPath') and kernel_closed(), 'reload_cleanup_unknown')
        # Remove own policy last, after actual stopped state and reload. Existing
        # loaded Python owns immutable bytes; no replacement cleanup worker.
        # Retain policy until exact stage cleanup is complete. The current
        # root process and lock FD retain original cleanup ownership.
        self.check()
        operation_handles={}
        for op in ('preflight','start','stop','cleanup'):
            q=STAGE/('operation-'+op+'.json')
            if q.exists():
                b,_=read_exact(q,0,0o444);operation_handles[op]=unique_json(b)
        for p in STAGE.iterdir():
            require(p.name in set(self.receipt['assets'])|{'receipt.json','control.lock','start-intent.json','operation-preflight.json','operation-start.json','operation-stop.json','operation-cleanup.json'}, 'cleanup_foreign_artifact')
            p.unlink()
        STAGE.rmdir();self.check()
        b,t=read_exact(POLICY,0,0o440)
        require(sha(b)==self.receipt['policy_sha256'] and [t.st_dev,t.st_ino]==self.receipt['policy_identity'],'cleanup_policy_replaced')
        POLICY.unlink();self.check()
        for q in (STAGE,POLICY,FRAGMENT):
            try:q.lstat()
            except FileNotFoundError:continue
            raise RuntimeError('root_artifact_absence_unknown')
        return dict(absence_observations=[dict(path=str(q),errno=2,observed_monotonic=time.monotonic()) for q in (STAGE,POLICY,FRAGMENT)],state='EXACT_ROOT_ARTIFACTS_REMOVED',unit=UNIT,manager=manager,clock_sha256=sha(self.clock_bytes),after=after,records=self.records,operation_handles=operation_handles)


def main():
    require(os.getuid()==os.geteuid()==0 and len(sys.argv)==2 and sys.argv[1] in ('preflight','start','stop','cleanup'), 'fixed_root_entry_required')
    os.umask(0o077);os.sched_setaffinity(0,{0,1});resource.setrlimit(resource.RLIMIT_AS,(2147483648,2147483648))
    root_directory(STAGE)
    fd=os.open(STAGE/'control.lock',os.O_RDWR|os.O_NOFOLLOW|os.O_CLOEXEC)
    try:
        s=os.fstat(fd);require(s.st_uid==s.st_gid==0 and stat.S_ISREG(s.st_mode) and s.st_nlink==1 and s.st_size==0 and stat.S_IMODE(s.st_mode)==0o600,'root_lock_identity')
        fcntl.flock(fd,fcntl.LOCK_EX|fcntl.LOCK_NB)
        control=Control(sys.argv[1]);value=control.execute()
        value.update(control_pid=os.getpid(),control_start_ticks=int(Path('/proc/self/stat').read_text().rsplit(')',1)[1].split()[19]),completed_monotonic=time.monotonic())
        control.check();print(json.dumps(value),flush=True);control.check()
    finally: os.close(fd)


if __name__=='__main__':
    main()
