def test_actual_default_inventory_tracks_every_retained_phase(installed, monkeypatch):
    s=installed;events=installer_transport(s,monkeypatch)
    phases=[];inventory=install_bootstrap.installed_inventory
    def observe(home):
        answer=inventory(home)
        state=entry.read_json(home/install_bootstrap.FOLDER/'state.json')
        phases.append(state['phase'])
        assert answer in ({'bootstrap'},{'bootstrap','probe-plan.json'})
        return answer
    monkeypatch.setattr(install_bootstrap,'installed_inventory',observe)
    original=copy.deepcopy(s.marker['original_attempt'])
    with native_home(s.home):
        result=friday_native.qualify(s.value,'f'*64,None,s.budget)
        assert result['state']=='DEPLOYMENTS_QUALIFIED'
        assert entry.inspect(s.value,'f'*64)['per_job_authority']=='NOT_GRANTED'
    assert {'CLAIMED','START_PENDING','OWNED_PROBE_RUNNING','STOPPED'}<=set(phases),phases
    assert events==['PREFLIGHT','ROUTE','START','OBSERVE_A0','STOP','RETIRE']
    assert not s.running and s.marker['original_attempt']==original


def test_constructor_failure_keeps_inspection_without_replay_or_readiness(installed,monkeypatch):
    s=installed;events=installer_transport(s,monkeypatch)
    original=copy.deepcopy(s.marker['original_attempt']);called=[]
    def unavailable(plan,*,budget):
        budget();called.append(copy.deepcopy(plan))
        raise OSError('independent unavailable runtime before native create')
    monkeypatch.setattr(native,'Runtime',unavailable)
    with native_home(s.home),pytest.raises(OSError,match='independent unavailable'):
        friday_native.qualify(s.value,'f'*64,None,s.budget)
    assert len(called)==1 and events==['PREFLIGHT','ROUTE'] and not s.running
    state=entry.read_json(s.home/install_bootstrap.FOLDER/'state.json')
    assert state['phase']=='FAILED_STOPPED_NO_REPLAY'
    assert state['retry_authorized'] is False if 'retry_authorized' in state else True
    assert not (s.home/qualification.PROOF).exists()
    with native_home(s.home):
        observed=entry.inspect(s.value,'f'*64)
        assert observed['state']=='TEMPLATE_INCOMPLETE' and observed['per_job_authority']=='NOT_GRANTED'
        before=list(events)
        with pytest.raises(ValueError):friday_native.qualify(s.value,'f'*64,None,s.budget)
    assert events==before and len(called)==1 and s.marker['original_attempt']==original
    original_probe=entry.read_json(s.home/install_bootstrap.FOLDER/'original.json')
    assert original_probe['retry_authorized'] is False
