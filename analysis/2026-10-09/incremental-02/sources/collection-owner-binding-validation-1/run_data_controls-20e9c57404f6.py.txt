def sha(path):

def save():

def owned(raw):
    if isinstance(raw, int):
        return raw in (1, 2) or Path('/proc/self/fd/' + str(raw)).resolve().is_relative_to(ROOT)
    return Path(os.fsdecode(raw)).absolute().is_relative_to(ROOT) and Path(os.fsdecode(raw)).resolve().is_relative_to(ROOT)

def deny(event):
    state['denied'].append(event)
    save()
    os.write(2, ('DATA_CONTROL_DENIED ' + event + '\n').encode())
    os._exit(125)

def audit(event, args):
    if event in ('subprocess.Popen', 'os.system', 'os.fork', 'os.forkpty', 'os.posix_spawn', 'os.exec', 'pty.spawn', 'os.kill', 'os.killpg', 'signal.pidfd_send_signal', 'os.pidfd_open') or event.startswith('socket.'):
        deny(event)
    if event == 'open' and args[2] & (os.O_WRONLY | os.O_RDWR | os.O_CREAT | os.O_TRUNC | os.O_APPEND) and not owned(args[0]):
        deny(event + ':outside_write')
    if event in ('os.mkdir', 'os.remove', 'os.rmdir', 'os.chmod', 'os.chown', 'os.utime', 'os.truncate') and not owned(args[0]):
        deny(event + ':outside_mutation')
    if event in ('os.rename', 'os.link') and not (owned(args[0]) and owned(args[1])):
        deny(event + ':outside_pair')
    if event == 'os.symlink' and not owned(args[1]):
        deny(event + ':outside_link')
