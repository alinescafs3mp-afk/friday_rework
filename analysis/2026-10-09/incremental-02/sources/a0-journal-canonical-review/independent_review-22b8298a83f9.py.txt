import ast
import copy
import datetime
import difflib
import hashlib
import json
import os
from pathlib import Path
import re
import resource
import signal
import time
import types

os.umask(0o077)
E = Path(__file__).parent
W = E.parent
C = W / 'a0-journal-canonical-integration/candidate'
os.sched_setaffinity(0, sorted(os.sched_getaffinity(0))[:2])
resource.setrlimit(resource.RLIMIT_AS, (1 << 30, 1 << 30))
resource.setrlimit(resource.RLIMIT_CPU, (20, 20))
resource.setrlimit(resource.RLIMIT_CORE, (0, 0))
signal.signal(signal.SIGALRM, lambda *_: (_ for _ in ()).throw(TimeoutError('finite independent review')))
signal.setitimer(signal.ITIMER_REAL, 30)
clock = json.loads((E / 'original-clock.json').read_text())
assert time.monotonic() < clock['deadline_monotonic']
checked = {}
checks = []
observations = []

def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def pinned(path, expected):
    path = Path(path)
    raw = path.read_bytes()
    assert sha(raw) == expected, 'pin mismatch: ' + str(path)
    checked[str(path)] = {'path': str(path), 'sha256': expected, 'bytes': len(raw)}
    return raw

def read_ref(ref):
    return json.loads(pinned(ref['path'], ref['sha256']))

def note(name, ok, detail=None):
    checks.append({'name': name, 'pass': bool(ok), 'detail': detail})
    assert ok, name

def write(path, obj):
    path.write_text(json.dumps(obj, indent=2, sort_keys=True) + '\n')

contract = '<LOCAL_HOME>/.jericho/runtime/subagent-lifecycle/state/astra/contracts/FRW011-A0-JOURNAL-CANONICAL-REVIEW__1.md'
pinned(contract, '5d2f78eccea5fe27c86cc5e7c2cb46ddccd89dfb0f82029076cbc4a312e816c5')
integration = json.loads(pinned(W/'a0-journal-canonical-integration/result.json', '94eb521bfbd82406e90f4735b48b6638cb3728c8662d72f1590f6b4254ea5182'))
pins = read_ref(integration['candidate_pins'])
root_controls = read_ref(integration['changed_controls'])
prior = read_ref(integration['prior_review'])
producer = read_ref(integration['prior_producer'])
prior_checks = read_ref(prior['independent_checks']['evidence'])
validation = read_ref(producer['validation'])
producer_checks = read_ref(validation['final_checks'])
aliases = read_ref(prior['defects'][0]['evidence'])
old_sources = {}
new_sources = {}
for name, ref in pins.items():
    new_sources[name] = pinned(ref['path'], ref['sha256'])
    old = prior['candidate_exact_pins'][name]
    old_sources[name] = pinned(old['path'], old['sha256'])
note('nine_exact_candidate_files', set(pins) == {'common.py', 'driver.py', 'dropins.py', 'observer.py', 'owner_launcher.py', 'parent_integration.py', 'payload.py', 'recipe.json', 'witness.py'})
line = b"            require(path == parent + '/' + Path(path).name, 'noncanonical diagnostic file path')\n"
note('common_delta_exactly_one_require', new_sources['common.py'].count(line) == 1 and new_sources['common.py'].replace(line, b'') == old_sources['common.py'])
note('other_eight_files_byteidentical', all(new_sources[name] == old_sources[name] for name in pins if name != 'common.py'))
(E/'common.py.diff').write_text(''.join(difflib.unified_diff(old_sources['common.py'].decode().splitlines(True), new_sources['common.py'].decode().splitlines(True), fromfile=prior['candidate_exact_pins']['common.py']['path'], tofile=pins['common.py']['path'])))
note('prior_independent217pass_twoD1fail_evidence_verified', sum(x['pass'] for x in prior_checks['checks']) == 217 and len(prior_checks['checks']) == 219 and prior_checks['all_clients_reaped'] is True)
note('prior_producer182pass_evidence_verified', len(producer_checks['checks']) == producer_checks['pass_count'] == 182 and all(x['status'] == 'PASS' for x in producer_checks['checks']) and producer_checks['all_owned_clients_closed'] is True)
note('root11affected_controls_evidence_verified', root_controls['pass'] == len(root_controls['checks']) == 11)

# Compile only pure grammar declarations, not the product or module top level.
tree = ast.parse(new_sources['common.py'])
names = {'PREFIX', 'ROLES', 'NAMES', 'ENV', 'HISTORY_ENV', 'HISTORY_PROFILE', 'CAPTURE_LIMIT'}
functions = {'Refused', 'require', 'history_argv', 'history_diagnostics'}
nodes = [n for n in tree.body if (isinstance(n, ast.Assign) and any(isinstance(t, ast.Name) and t.id in names for t in n.targets)) or (isinstance(n, (ast.FunctionDef, ast.ClassDef)) and n.name in functions)]
ns = {'Path': Path, 're': re}
exec(compile(ast.Module(body=nodes, type_ignores=[]), str(C/'common.py'), 'exec'), ns)
grammar = ns['history_diagnostics']
Refused = ns['Refused']
source = new_sources['common.py'].decode()
guard_at = source.index(line.decode())
note('raw_guard_precedes_admission_dedup_and_count', guard_at < source.index('path not in files', guard_at) < source.index('files.add(path)', guard_at))
note('raw_path_not_rewritten_and_existing_parent_name_checks_retained', 'path = m[1]; parent = str(Path(path).parent)' in source and source[guard_at:source.index('files.add(path)', guard_at)].count('require(') == 3)
parent_tree = ast.parse(new_sources['parent_integration.py'])
read_nodes = [n for n in ast.walk(parent_tree) if isinstance(n, ast.FunctionDef) and n.name == 'read']
assert len(read_nodes) == 1
reader_code = compile(ast.Module(body=read_nodes, type_ignores=[]), str(C/'parent_integration.py'), 'exec')

def call_grammar(cap):
    original = {'boot_id': cap['argv'][2].split('=', 1)[1]}
    try:
        return True, grammar(cap, original)
    except Refused as exc:
        return False, str(exc)

def parent_control(label, cap, transport_refuses=False, expect_accept=False):
    before = copy.deepcopy(cap)
    transport = types.SimpleNamespace(records=[])
    saved = []
    seen = []
    original = {'boot_id': cap['argv'][2].split('=', 1)[1]}
    outdir = E/'parent-captures'/label
    outdir.mkdir(parents=True)
    def capture(argv, timeout):
        assert argv == before['argv'] and timeout == 1.0
        transport.records.append({'capture': cap})
        if transport_refuses:
            grammar(cap, original)
            raise AssertionError('expected classifier refusal')
        return cap
    def persist(path, value):
        assert path.parent == outdir
        write(path, value)
        saved.append(json.loads(path.read_text()))
    env = {'remaining': lambda *_: 4.0, 'anchor': original, '__transport': transport,
           'bounded_capture_call': capture, 'observations': seen, 'save': persist,
           'E': outdir, 'history_argv': ns['history_argv'], 'history_diagnostics': grammar,
           'boot': original['boot_id']}
    exec(reader_code, env)
    try:
        value = env['read'](cap['argv'])
        accepted = True
    except Refused as exc:
        value = str(exc)
        accepted = False
    note(label+'_parent_outcome', accepted == expect_accept and (value == '' if expect_accept else value == 'noncanonical diagnostic file path'), value)
    note(label+'_parent_exact_raw_saved', saved == [before] and seen == [before] and cap == before)
    observations.append({'case': label, 'mode': 'exact read AST / saved capture / transport dependency supplied',
                         'transport_refuses': transport_refuses, 'accepted': accepted, 'result': value,
                         'saved': str(outdir/'preflight-read-01.json'), 'capture_unchanged': cap == before,
                         'native_execution': False})

note('all_four_saved_D1_aliases_present', {x['case'] for x in aliases['cases']} == {'trailing_slash', 'internal_double_slash', 'internal_dot', 'same_file_alias_doublecount'})
for item in aliases['cases']:
    cap = copy.deepcopy(item['capture'])
    before = copy.deepcopy(cap)
    accepted, value = call_grammar(cap)
    note('saved_'+item['case']+'_classifier_refuses_raw_unchanged', not accepted and value == 'noncanonical diagnostic file path' and cap == before, value)
    parent_control(item['case']+'_returned', cap)
    parent_control(item['case']+'_transport_refused', cap, transport_refuses=True)

# Affected positive domain: each original filename form, both original journal
# roots, and both already-supported directory levels. This is source grammar
# replay, not a query or an assertion that each host directory/file exists.
base = copy.deepcopy(next(x['capture'] for x in aliases['cases'] if x['case'] == 'trailing_slash'))
directory = '/var/log/journal/'+'0'*32
filename = 'user-1000.journal'
base['stderr'] = base['stderr'].replace(directory+'/'+filename+'/', directory+'/'+filename)
base['stderr_raw_hex'] = base['stderr'].encode().hex()
forms = ['user-1000.journal', 'user-1000.journal~', 'user-1000@'+'a'*32+'-'+'b'*16+'-'+'c'*16+'.journal', 'user-1000@'+'a'*32+'-'+'b'*16+'-'+'c'*16+'.journal~']
for root in ('/run/log/journal', '/var/log/journal'):
    for depth in ('root', 'machine'):
        for index, form in enumerate(forms):
            cap = copy.deepcopy(base)
            lines = cap['stderr'].splitlines()
            if root == '/run/log/journal':
                lines = [x.replace('/var/log/journal', '/run/log/journal') for x in lines if x != "Considering root directory '/var/log/journal'."]
                found = next(i for i,x in enumerate(lines) if x.startswith('Found boot '))
                lines.insert(found, "Considering root directory '/var/log/journal'.")
            selected_dir = root+'/'+'0'*32
            if depth == 'root':
                lines = [x for x in lines if not (x.startswith('Considering directory ') or x.startswith('Directory '))]
                lines = [x.replace(selected_dir+'/', root+'/') for x in lines]
            lines = [x.replace(filename, form) for x in lines]
            cap['stderr'] = '\n'.join(lines)+'\n'
            cap['stderr_raw_hex'] = cap['stderr'].encode().hex()
            accepted, value = call_grammar(cap)
            label = root.split('/')[1]+'_'+depth+'_name'+str(index)
            note('valid_'+label+'_grammar', accepted and value['files_observed'] == 1 and value['physical_corpus_certified'] is False and value['global_history_certified'] is False, value)
            if root == '/var/log/journal' and depth == 'machine':
                parent_control('valid_'+label, cap, expect_accept=True)

# D1-adjacent duplicate and separator controls: raw equality must not replace
# existing unique-file or valid-basename checks.

duplicate['stderr_raw_hex'] = duplicate['stderr'].encode().hex()

for label, badpath in [('leading_double', '//var/log/journal/'+'0'*32+'/'+filename), ('parent_dotdot', directory+'/../'+filename), ('suffix_dot', directory+'/'+filename+'/.'), ('three_slashes', directory+'///'+filename), ('bad_basename', directory+'/user-1001.journal')]:
    cap = copy.deepcopy(base)
    cap['stderr'] = cap['stderr'].replace(directory+'/'+filename, badpath)
    cap['stderr_raw_hex'] = cap['stderr'].encode().hex()
    accepted, value = call_grammar(cap)
    note('D1_adjacent_'+label+'_refused', not accepted, value)

for path, ref in checked.items():
    assert sha(Path(path).read_bytes()) == ref['sha256'], 'input changed during review: '+path
note('all_input_and_candidate_pins_unchanged_at_control_completion', True, len(checked))
resources = {'affinity': sorted(os.sched_getaffinity(0)), 'address_space_limit_bytes': resource.getrlimit(resource.RLIMIT_AS)[0],
             'cpu_time_limit_seconds': resource.getrlimit(resource.RLIMIT_CPU)[0], 'model_children': 0, 'process_clients': 0,
             'maxrss_kib': resource.getrusage(resource.RUSAGE_SELF).ru_maxrss,
             'available_memory': [x for x in Path('/proc/meminfo').read_text().splitlines() if x.startswith('MemAvailable:')],
             'pressure': {x: Path('/proc/pressure', x).read_text() for x in ('cpu','io','memory')}}
write(E/'verified-pins.json', list(checked.values()))
write(E/'independent-controls.json', {'checks': checks, 'passed': sum(x['pass'] for x in checks), 'failed': sum(not x['pass'] for x in checks),
      'parent_read_observations': observations, 'resources': resources, 'original_clock': clock,
      'elapsed_seconds': time.monotonic()-clock['accepted_monotonic'], 'native_qualification': 'NOT_RUN',
      'full_battery_rerun': False, 'owned_execution_remaining': False})
print(json.dumps({'passed': len(checks), 'failed': 0, 'parent_saved_capture_cases': len(observations), 'pins_checked': len(checked),
                  'native_qualification': 'NOT_RUN', 'elapsed_seconds': time.monotonic()-clock['accepted_monotonic']}))
