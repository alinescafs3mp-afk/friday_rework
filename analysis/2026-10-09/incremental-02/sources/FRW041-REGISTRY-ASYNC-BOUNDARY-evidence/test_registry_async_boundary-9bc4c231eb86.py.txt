"""Finite decision controls; SOURCE ONLY, NOT RUN.

These models exercise the actual guard predicates and refusal branches without
opening sockets, signalling, spawning or creating threads. They do not establish
kernel denial or real endpoint cessation. The unchanged seven registry behavior
cases must later prove the actual two pairs/four closed endpoints under the fence.
"""
import socket
import threading
import pytest
import private_effect_fence as fence

NODES=tuple(sorted(fence.REGISTRY_NODES))
FIXTURE='/pinned/source/'+fence.REGISTRY_TEST_REL

def request(**changes):
    values=dict(family=socket.AF_UNIX,kind=socket.SOCK_STREAM,proto=0,
                phase='registry',node=NODES[0],fixture=FIXTURE,expected_fixture=FIXTURE,
                pinned=True,main_thread=True,provenance=True,previous=0,total=0)
    values.update(changes)
    return fence.registry_request_allowed(**values)

@pytest.mark.parametrize('change', [
    {'family':socket.AF_INET}, {'family':socket.AF_INET6}, {'family':True},
    {'kind':socket.SOCK_DGRAM}, {'kind':True}, {'proto':1}, {'proto':False},
    {'phase':'native'}, {'phase':'owner'}, {'phase':'controls'},
    {'node':'test_registry_async_failure_keeps_original_error'},
    {'node':'foreign/'+NODES[0]}, {'node':NODES[0]+'[foreign]'}, {'node':None},
    {'fixture':'/foreign/source/'+fence.REGISTRY_TEST_REL}, {'expected_fixture':'/foreign'},
    {'pinned':False}, {'pinned':1}, {'main_thread':False}, {'provenance':False},
    {'previous':1}, {'previous':True}, {'previous':-1}, {'total':2}, {'total':-1},
    {'total':True},
])
def test_registry_request_refuses_foreign_or_excess(change):
    assert not request(**change)

@pytest.mark.parametrize('node,total', [(NODES[0],0),(NODES[1],1)])
def test_registry_request_accepts_each_exact_owned_pair_model(node,total):
    assert request(node=node,total=total)

@pytest.mark.parametrize('key,missing',[(key,missing) for key in ('fixture','helper','selector','socket') for missing in (True,False)])
def test_registry_fixed_source_pin_missing_or_changed_refused(monkeypatch,key,missing):
    monkeypatch.setattr(fence,'REGISTRY_PINS',{})
    pins={((fence.BASE if k in ('fixture','helper') else fence.STDLIB)+'/'+suffix):h
          for k,(suffix,h) in fence.REGISTRY_PIN_SUFFIXES.items()}
    path=(fence.BASE if key in ('fixture','helper') else fence.STDLIB)+'/'+fence.REGISTRY_PIN_SUFFIXES[key][0]
    if missing:del pins[path]
    else:pins[path]='0'*64
    with pytest.raises(RuntimeError,match='registry_source_pin_mismatch:'+key):
        fence.registry_bind_pins(pins)

def test_registry_fixed_source_pin_exact_membership_binds_only_four_leaves(monkeypatch):
    monkeypatch.setattr(fence,'REGISTRY_PINS',{})
    pins={((fence.BASE if k in ('fixture','helper') else fence.STDLIB)+'/'+suffix):h
          for k,(suffix,h) in fence.REGISTRY_PIN_SUFFIXES.items()}
    fence.registry_bind_pins(pins)
    assert set(fence.REGISTRY_PINS.values())==set(pins)

class Refused(Exception):
    pass

def refusal(event,*args,**kwargs):
    raise Refused(event)

@pytest.mark.parametrize('kind',['dup','dup2','detach','fcntl_dup','fcntl_dup_cloexec'])
def test_registry_duplicate_fd_or_detach_cannot_escape_pair_custody(monkeypatch,kind):
    monkeypatch.setattr(fence,'state',{'phase':'registry','current_node':NODES[0]})
    monkeypatch.setattr(fence,'deny',refusal)
    calls=[]
    monkeypatch.setattr(fence,'REGISTRY_DUP_ORIGINALS',{'dup':lambda *a: calls.append(a),'dup2':lambda *a: calls.append(a)})
    monkeypatch.setattr(fence,'REGISTRY_DETACH_ORIGINAL',lambda *a: calls.append(a))
    with pytest.raises(Refused,match='registry_fd_duplication|registry_endpoint_custody'):
        if kind in ('dup','dup2'):fence.registry_guarded_dup(kind)(9)
        elif kind=='detach':fence.registry_guarded_detach(object())
        else:fence.audit('fcntl.fcntl',(9,fence.fcntl.F_DUPFD if kind=='fcntl_dup' else fence.fcntl.F_DUPFD_CLOEXEC,0))
    assert calls==[]

@pytest.mark.parametrize('kind',['signal','thread','executor','process','ctypes'])
def test_registry_separate_pair_admission_never_grants_escalation(monkeypatch,kind):
    monkeypatch.setattr(fence,'state',{'phase':'registry','current_node':NODES[0]})
    monkeypatch.setattr(fence,'deny',refusal)
    calls=[]
    monkeypatch.setattr(fence,'SIGNAL',lambda *a: calls.append(a))
    monkeypatch.setattr(fence,'THREAD_START',lambda *a: calls.append(a))
    assert not fence.async_active()
    if kind=='signal':
        with pytest.raises(Refused,match='outside_asyncio_own_SIGINT_handler'):
            fence.guarded_signal(fence.signal.SIGINT,fence.signal.default_int_handler)
    elif kind in ('thread','executor'):
        target=(lambda: None)
        # Thread construction is inert; start is guarded and never reaches RAW.
        thread=threading.Thread(target=target,name='foreign-executor' if kind=='executor' else 'foreign')
        with pytest.raises(Refused,match='outside_four_required_threads'):
            fence.guarded_thread_start(thread)
    else:
        with pytest.raises(Refused,match='subprocess.Popen|ctypes.dlopen'):
            fence.audit('subprocess.Popen' if kind=='process' else 'ctypes.dlopen',())
    assert calls==[]
