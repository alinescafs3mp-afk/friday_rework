@pytest.mark.parametrize('phase',['compile','import','handoff','install','alarm_compile','alarm_handoff','confirmation'])
def test_R3_original_owner_covers_bootstrap_failures(loader,owner_fixture,monkeypatch,phase):
    context,events,timers=owner_fixture
    class Fault(RuntimeError):pass
    def fault(*args,**kwargs):raise Fault('original_install_clock_exhausted' if phase.startswith('alarm') else phase)
    if phase in ('compile','alarm_compile'):monkeypatch.setattr(loader,'compile',fault,raising=False)
    def execute(compiled,scope):
        if phase=='import':
            namespace=dict(vars(builtins));namespace['__import__']=fault
            scope['__builtins__']=namespace
        builtins.exec(compiled,scope)
        if phase in ('handoff','alarm_handoff'):
            # A module-initialization/handoff failure before main can be called.
            fault()
        scope['rollback']=lambda ctx:events.append('installer_rollback')
        scope['main']=fault if phase=='install' else lambda:{'state':'fixture_prepared'}
    monkeypatch.setattr(loader,'exec',execute,raising=False)
    if phase=='confirmation':monkeypatch.setattr(loader,'confirm_prepared',fault)
    with pytest.raises(Fault):loader.original_owner(context)
    branch='installer_rollback' if phase in ('install','confirmation') else 'loader_rollback'
    assert events.index('bootstrap')<events.index(branch)<events.index('journal_removed')
    assert 'retained' not in events and 'confirmed' not in events
    assert timers[:2]==[30.0,50.0] and timers[-1]==0
    assert context['deadline_monotonic']==60.0 and context['started_monotonic']==0.0

@pytest.mark.parametrize('failure',[BrokenPipeError('closed stdout'),RuntimeError('serialization/flush failure')])
def test_R3_retention_is_mandatory_even_if_reporting_raises(loader,monkeypatch,failure):
    events=[]
    monkeypatch.setattr(loader.signal,'setitimer',lambda *args:None)
    monkeypatch.setattr(loader.signal,'signal',lambda *args:None)
    def reporting(*args):events.append('failed_report');raise failure
    monkeypatch.setattr(loader,'best_effort_report',reporting)
    # A trace observes the exact infinite retention branch without starting a
    # process/thread, waiting, or introducing a production escape parameter.
    node=next(n for n in ast.parse(loader.__source__).body if isinstance(n,ast.FunctionDef) and n.name=='retain_custody')
    loop=next(n for n in ast.walk(node) if isinstance(n,ast.While))
    previous_trace=sys.gettrace()
    def trace(frame,event,arg):
        if frame.f_code is loader.retain_custody.__code__ and event=='line' and frame.f_lineno==loop.lineno:
            events.append('original_process_retained')
            raise RetainedForTest()
        return trace
    try:
        sys.settrace(trace)
        with pytest.raises(RetainedForTest):loader.retain_custody({'deadline_monotonic':60.0},failure,failure)
    finally:sys.settrace(previous_trace)
    assert events==['failed_report','original_process_retained']

@pytest.mark.parametrize('failure',[None,'journal','chmod','replacement'])
def test_R2_real_umask_mode_and_durable_identity_before_chmod(loader,monkeypatch,tmp_path,failure):
    stage=tmp_path/'stage';events=[];real_fstat=os.fstat;real_stat=os.stat;real_chmod=os.fchmod
    monkeypatch.setattr(loader,'STAGE',stage)
    monkeypatch.setattr(loader.os,'fstat',lambda fd:root_fixture_metadata(real_fstat(fd)))
    replacement=[False]
    def stated(*args,**kwargs):
        s=root_fixture_metadata(real_stat(*args,**kwargs))
        if replacement[0]:values=list(s);values[1]+=1;s=os.stat_result(values)
        return s
    monkeypatch.setattr(loader.os,'stat',stated)
    def record(row):
        events.append(row)
        if row['kind']=='created':
            assert stat.S_IMODE(real_stat(stage).st_mode)==0o700
            assert row['identity']==[real_stat(stage).st_dev,real_stat(stage).st_ino]
            if failure=='journal':raise OSError('journal fsync failed')
    monkeypatch.setattr(loader,'journal_append',record)
    def chmod(fd,mode):
        assert events[-1]['kind']=='created' and mode==0o755
        if failure=='chmod':raise OSError('chmod failed')
        real_chmod(fd,mode)
        if failure=='replacement':replacement[0]=True
    monkeypatch.setattr(loader.os,'fchmod',chmod)
    fd=os.open(tmp_path,os.O_RDONLY|os.O_DIRECTORY);previous=os.umask(0o077)
    try:
        if failure:
            with pytest.raises((OSError,RuntimeError)):loader.prepare_stage(fd)
        else:
            loader.prepare_stage(fd)
            assert stat.S_IMODE(real_stat(stage).st_mode)==0o755
            # Actual descriptor file mode plus directory traversal bits; no root,
            # setuid or system path. A later live gate must still prove UID1000.
            asset=stage/'receipt.json';asset.write_bytes(b'private');asset.chmod(0o444)
            assert stat.S_IMODE(real_stat(asset).st_mode)==0o444 and asset.read_bytes()==b'private'
        if failure in ('journal','chmod'):assert stat.S_IMODE(real_stat(stage).st_mode)==0o700
    finally:os.umask(previous);os.close(fd)
