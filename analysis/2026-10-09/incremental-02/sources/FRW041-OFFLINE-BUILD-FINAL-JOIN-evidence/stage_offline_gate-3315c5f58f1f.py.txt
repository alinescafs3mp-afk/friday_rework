"""INERT staging renderer for the exact final joined source and bounded runner.
NOT imported/executed in preparation. Needs a separate exact source-staging
assignment after final independent source/runner review. No CLI/clock/admission.
"""
import hashlib
import json
import os
from pathlib import Path
import stat

E = Path('<SOL_WORK>/FRW041-OFFLINE-BUILD-FINAL-JOIN-evidence')
SOURCE = E / 'candidate'
COPY = E / 'source-copy'
QUAL = Path('<ASTRA_WORK>/offline-build-final-join-qualification')
PRIVATE = QUAL / 'private'
OUT = QUAL / 'output'
SOURCE_TASK = 'FRW041-OFFLINE-BUILD-FINAL-JOIN-SOL#1'
PACKAGE_SHA = '904a5fb865d9db1cf6c6158bcc96d45c15294d9e0c7125abbc3ed70d8dbf1030'


def read(path, expected_sha=None, size=None, mode=0o400):
    path = Path(path)
    assert path.is_absolute() and '..' not in path.parts
    assert path == path.resolve(strict=True) and not path.is_symlink()
    for parent in path.parents:
        s = parent.lstat()
        assert stat.S_ISDIR(s.st_mode) and s.st_uid in (0, 1000) and not s.st_mode & 0o022
    fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK | os.O_CLOEXEC)
    try:
        a = os.fstat(fd)
        assert stat.S_ISREG(a.st_mode) and a.st_uid == 1000 and a.st_nlink == 1
        assert stat.S_IMODE(a.st_mode) == mode and a.st_size <= 128000000
        chunks = []
        while True:
            block = os.read(fd, 131072)
            if not block:
                break
            chunks.append(block)
        data = b''.join(chunks)
        b = os.fstat(fd)
        c = path.lstat()
        key = lambda s: (s.st_dev, s.st_ino, s.st_size, s.st_mtime_ns, s.st_mode, s.st_uid, s.st_nlink)
        assert key(a) == key(b) == key(c) and len(data) == a.st_size
        assert size is None or len(data) == size
        assert expected_sha is None or hashlib.sha256(data).hexdigest() == expected_sha
        return data
    finally:
        os.close(fd)


def create(path, data):
    assert path.is_relative_to(OUT)
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW | os.O_CLOEXEC, 0o600)
    with os.fdopen(fd, 'wb') as f:
        f.write(data)
        f.flush()
        os.fsync(f.fileno())
    path.chmod(0o400)


def render(review_path, review_sha256, publication_sha256):
    """Future exact fresh staging only, never execution or a runtime grant.
    Explicit review bytes must use the existing installer/runtime consumer
    contract on THIS final package. The old repair receipt cannot satisfy it.
    Caller owns the later independently assigned staging operation/budget.
    """
    assert os.getuid() == os.geteuid() == 1000
    review = json.loads(read(review_path, review_sha256))
    assert review['state'] == 'INDEPENDENT_OFFLINE_BUILD_SYSTEM_SERVICE_SOURCE_ACCEPTED'
    assert review['source_task'] == SOURCE_TASK and review['runtime_grant'] is False
    assert review['package_pins_sha256'] == PACKAGE_SHA
    publication = json.loads(read(E / 'PUBLICATION-MANIFEST.json', publication_sha256))
    assert publication['state'] == 'SOURCE_ONLY_FINAL_INDEPENDENT_REVIEW_REQUIRED'
    assert publication['source_package_sha256'] == PACKAGE_SHA
    assert publication['runtime_grant'] is False
    # Review of the whole publication covers the runner and this renderer too;
    # root supplies its exact accepted publication hash separately.
    assert review['publication_manifest_sha256'] == publication_sha256
    assert review['offline_runner_source_accept'] is True
    for name, pin in publication['refs'].items():
        read(pin['path'], pin['sha256'], pin['bytes'], pin['mode'])
    pins_data = read(SOURCE / 'package-pins.json', PACKAGE_SHA)
    assert read(COPY / 'package-pins.json', PACKAGE_SHA) == pins_data
    pins = json.loads(pins_data)
    input_pins = {}
    for tree in (SOURCE, COPY):
        st = tree.lstat()
        assert stat.S_ISDIR(st.st_mode) and st.st_uid == 1000 and stat.S_IMODE(st.st_mode) == 0o500
        assert set(p.name for p in tree.iterdir()) == set(pins) | {'package-pins.json'}
        for leaf, pin in pins.items():
            assert Path(leaf).name == leaf and leaf not in ('.', '..')
            data = read(tree / leaf, pin['sha256'], pin['size'])
            input_pins[str(tree / leaf)] = dict(sha256=pin['sha256'], size=len(data))
        input_pins[str(tree / 'package-pins.json')] = dict(sha256=PACKAGE_SHA, size=len(pins_data))
    runner_pin = publication['refs']['runner']
    runner = read(runner_pin['path'], runner_pin['sha256'], runner_pin['bytes'])
    # O_EXCL equivalent directory creation is the original no-adoption marker.
    # Resolve/check existing protected parents before any creation.
    assert QUAL.parent == QUAL.parent.resolve(strict=True)
    for parent in (QUAL.parent, *QUAL.parent.parents):
        s = parent.lstat()
        assert stat.S_ISDIR(s.st_mode) and s.st_uid in (0, 1000) and not s.st_mode & 0o022
    assert not QUAL.exists() and not QUAL.is_symlink(), 'fresh_staging_only_no_replay'
    QUAL.mkdir(mode=0o700)
    PRIVATE.mkdir(mode=0o700)
    OUT.mkdir(mode=0o700)
    for leaf in ('results', 'tmp', 'home', 'pytesttmp'):
        (PRIVATE / leaf).mkdir(mode=0o700)
    create(OUT / 'run.py', runner)
    input_pins[str(OUT / 'run.py')] = dict(size=len(runner), sha256=hashlib.sha256(runner).hexdigest())
    identity = PRIVATE.stat()
    create(OUT / 'private-temp.json', (json.dumps(dict(device=identity.st_dev, inode=identity.st_ino)) + '\n').encode())
    create(OUT / 'input-pins.json', (json.dumps(input_pins, indent=2) + '\n').encode())
    create(OUT / 'staging.json', (json.dumps(dict(
        state='EXACT_OFFLINE_SOURCE_STAGED_NOT_EXECUTED',
        source_task=SOURCE_TASK, review_path=str(review_path), review_sha256=review_sha256,
        publication_manifest_sha256=publication_sha256, source_package_sha256=PACKAGE_SHA,
        runner_sha256=hashlib.sha256(runner).hexdigest(), runtime_grant=False,
        clock_created=False, tests_collected=False, tests_executed=False,
        next='Separate finite offline assignment and original180s clock; GNU KILL70s required'), indent=2) + '\n').encode())
    assert not (OUT / 'runtime-clock.json').exists()
    return dict(out=str(OUT), private=str(PRIVATE), runner=str(OUT / 'run.py'),
                state='SOURCE_STAGED_NOT_EXECUTED', runtime_grant=False, clock_created=False)
