def digest(data):
    return hashlib.sha256(data).hexdigest()


def js(path):
    return json.loads(path.read_bytes())


def ref(path):
    data = path.read_bytes()
    s = path.lstat()
    assert path == path.resolve(strict=True) and stat.S_ISREG(s.st_mode)
    assert s.st_uid == 1000 and s.st_nlink == 1
    return dict(path=str(path), sha256=digest(data), bytes=len(data), mode=stat.S_IMODE(s.st_mode))


def parse(path):
    data = path.read_bytes()
    tree = ast.parse(data, filename=str(path))
    compile(tree, str(path), 'exec')  # Syntax only, code object never executed.
    return tree


def count(fn):
    value = 1
    for d in fn.decorator_list:
        if isinstance(d, ast.Call) and isinstance(d.func, ast.Attribute) and d.func.attr == 'parametrize':
            assert isinstance(d.args[1], (ast.List, ast.Tuple))
            value *= len(d.args[1].elts)
    return value
