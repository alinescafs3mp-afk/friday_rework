def sha(data):
    return hashlib.sha256(data).hexdigest()


def encoded(value):
    return (json.dumps(value, indent=2) + '\n').encode()


def exact(path, pin):
    data = path.read_bytes()
    s = path.lstat()
    assert path.resolve(strict=True) == path and stat.S_ISREG(s.st_mode)
    assert s.st_uid == 1000 and s.st_nlink == 1 and stat.S_IMODE(s.st_mode) == 0o400
    assert len(data) == pin.get('bytes', pin.get('size')) and sha(data) == pin['sha256']
    return data


def ref(path):
    data = path.read_bytes()
    return dict(path=str(path), sha256=sha(data), bytes=len(data))


def put(path, data):
    assert path.is_relative_to(E)
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600)
    with os.fdopen(fd, 'wb') as f:
        f.write(data)
        f.flush()
        os.fsync(f.fileno())
    path.chmod(0o400)
