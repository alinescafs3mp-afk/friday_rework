def join_model(monkeypatch):
    item=model(); item.records.feed(START)
    item.semantic=['/env', '-i', '/python', '-P', '-s', '-B', '/source/invocation.py', '/run', 'unit', 'guard.py', '@native', '--runtest-file=f000']
    monkeypatch.setattr(a.time,'monotonic',lambda:100)
    ns = {name:[5, 200+i] for i,name in enumerate(a.NS_NAMES)}
    ns['pid']=[5,201]; ns['mnt']=[5,202]
    def row(pid, ppid, inner, pgid, sid):
        return dict(pid=pid,ppid=ppid,start_ticks=50+pid,pgid=pgid,sid=sid,
                    namespaces=copy.deepcopy(ns), cgroup=['0::/unit'],
                    vectors=dict(NSpid=[pid,inner],NSpgid=[pgid,2],NSsid=[sid,2]),
                    exe='/timeout' if pid==12 else '/python',cmdline=[])
    rows={10:row(10,1,10,10,10),11:row(11,10,1,10,10),
          12:row(12,11,2,12,12),13:row(13,12,3,12,12)}
    rows[10]['namespaces']['pid']=[5,100]
    contour=dict(pid_namespace=[5,201],external_deadline_monotonic=150.0,timeout=dict(pid=2,ppid=1,pgid=2,sid=2,start_ticks=62,exe='/timeout'),
                 guard=dict(pid=3,ppid=2,pgid=2,sid=2,start_ticks=63,exe='/python'))
    expected_timeout,expected_guard=a.expected_lineage(item.semantic,contour,60)
    rows[12]['cmdline']=[os.fsencode(v) for v in expected_timeout]
    rows[13]['cmdline']=[os.fsencode(v) for v in expected_guard]
    class P:
        def __init__(self, value): self.value=str(value)
        def __truediv__(self, other): return P(self.value+'/'+str(other))
        def stat(self): return NS(st_dev=5,st_ino=100) if self.value=='/proc/self/ns/pid' else NS(st_dev=3,st_ino=4)
        def resolve(self, **kw): return '/timeout'
        @property
        def parent(self): return P(self.value.rsplit('/',1)[0])
        def __str__(self): return self.value
    monkeypatch.setattr(a,'Path',P)
    monkeypatch.setattr(a,'read_bounded',lambda p,n: b'boot\n' if str(p)=='/proc/sys/kernel/random/boot_id' else b'10\n11\n12\n13\n')
    monkeypatch.setattr(a,'identity',lambda pid: copy.deepcopy(rows[pid]))
    return item,rows,contour


def test_live_join_uses_host_census_not_inner_numeric_targets(monkeypatch):
    item,rows,contour=join_model(monkeypatch)
    result=item._join_live(rows[10],contour,'/python')
    assert result['timeout_host']['pid']==12 and contour['timeout']['pid']==2
    assert result['guard_host']['pid']==13 and result['sandbox_child']['pid']==11
    assert item.session.process is None  # mapping cannot mutate registry custody
