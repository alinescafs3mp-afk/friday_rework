"""Finite source/data verification only; never imports or executes target code."""
import ast
import copy
import hashlib
import json
import os
import resource
from pathlib import Path

os.sched_setaffinity(0, {0, 1})
resource.setrlimit(resource.RLIMIT_AS, (2 << 30, 2 << 30))
OUT = Path(__file__).parent
B = OUT.parent / 'registry-native-wrapper-repair'
OLD = Path('<SOL_WORK>/FRW041-REGISTRY-ASYNC-BOUNDARY-evidence/candidate')
def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()
def data(path):
    return json.loads(Path(path).read_text())
def functions(path):
    return {n.name:n for n in ast.parse(Path(path).read_text()).body if isinstance(n, ast.FunctionDef)}
def dump(node):
    return ast.dump(node, include_attributes=False)
assert sha(B/'result.json') == '864c30661d3c19ebd85986ee9add5aae3fe534fe5e33d879e28ff1de1c4b2ec5'
leaves = data(B/'replacement-leaves.json')
verified = []
for item in leaves['replacements']:
    for role in ('old','new','patch'):
        ref = item[role]
        content = Path(ref['path']).read_bytes()
        assert hashlib.sha256(content).hexdigest() == ref['sha256'], ref['path']
        assert len(content) == ref['bytes'], ref['path']
    verified.append(item['new'])
assert len(verified) == 9
for ref in leaves['cumulative_delta_replacements']:
    assert sha(ref['path']) == ref['sha256']
    assert Path(ref['path']).stat().st_size == ref['bytes']
new, old = functions(B/'private_effect_fence.py'), functions(OLD/'private_effect_fence.py')
changed = sorted(n for n in old if dump(new[n]) != dump(old[n]))
assert changed == ['guarded_close','registry_audit_wrapper','registry_bind_pins','registry_cessation','registry_create_pair','registry_endpoint_check','registry_fd_birth','registry_row_closed','registry_test_start','registry_wrapper_allowed']
assert sorted(set(new)-set(old)) == ['registry_births_allowed','registry_fd_identity','registry_native_allowed','registry_refuse']
legacy = functions(OUT.parent/'browser-owner-controls-preparation/private_effect_fence.py')
f1 = functions(B/'test_import_read_boundary_source.py')['test_process_socket_thread_and_sigint_guards_preserved']
adds = ast.literal_eval(next(n.value for n in f1.body if isinstance(n,ast.Assign) and any(isinstance(t,ast.Name) and t.id=='additions' for t in n.targets)))
f1_compared = []
for name in ('guarded_bind','guarded_thread_start','guarded_joinable','guarded_signal','check_path','check_fd','path_ok','fd_path','canonical'):
    assert dump(new[name]) == dump(legacy[name]), name
    f1_compared.append(name)
for name, operations in adds.items():
    expected = copy.deepcopy(legacy[name])
    for index, source in operations:
        expected.body[index:index] = ast.parse(source).body
    assert dump(new[name]) == dump(expected), name
    f1_compared.append(name)
expected = copy.deepcopy(legacy['closed_primitives'])
expected.body[0].value.keys.insert(0,ast.Constant(value='registry_self_pipe'))
expected.body[0].value.values.insert(0,ast.parse('registry_cessation()',mode='eval').body)
assert dump(new['closed_primitives']) == dump(expected)
f1_compared.append('closed_primitives')
assert len(f1_compared) == 14
expected = data(B/'expected-suite-cases.json')
previous = data(OLD/'expected-suite-cases.json')
for phase in ('native','project','owner','registry'):
    assert expected[phase] == previous[phase]
original_control = {k:v for k,v in previous['controls'].items() if not k.startswith('test_registry_')}
assert sum(original_control.values()) == 21
assert all(expected['controls'][k] == v for k,v in original_control.items())
control_functions = functions(B/'test_registry_async_boundary.py')
counts = {}
for name,node in control_functions.items():
    if not name.startswith('test_'):continue
    count = 1
    for decorator in node.decorator_list:
        assert isinstance(decorator,ast.Call) and isinstance(decorator.func,ast.Attribute) and decorator.func.attr=='parametrize'
        values = decorator.args[1]
        if isinstance(values,(ast.List,ast.Tuple)):size=len(values.elts)
        elif isinstance(values,ast.Name) and values.id=='NODES':size=2
        elif isinstance(values,ast.ListComp):
            size=1
            for generator in values.generators:
                assert isinstance(generator.iter,(ast.List,ast.Tuple)) and not generator.ifs
                size *= len(generator.iter.elts)
        else:raise AssertionError(ast.dump(values))
        count *= size
    counts[name] = count
assert counts == {k:v for k,v in expected['controls'].items() if k.startswith('test_registry_')}
assert sum(counts.values()) == 171
assert sum(sum(v.values()) for v in expected.values()) == 424
nested = set()
def walk(value):
    if isinstance(value,dict):
        if {'path','sha256','bytes'} <= set(value) and value['path'].startswith(str(B)+'/'):
            assert sha(value['path']) == value['sha256']
            assert Path(value['path']).stat().st_size == value['bytes']
            nested.add(value['path'])
        for item in value.values():walk(item)
    elif isinstance(value,list):
        for item in value:walk(item)
for item in verified:
    if item['path'].endswith('.json'):walk(data(item['path']))
pins = data(OLD/'runtime-flat-pins.json')
assert pins['<PROJECT>/.runtime/hermes-telegram-test/env/bin/python'] == '8dfa9757a52b9c3edf1dedaaa2a7a8c40ea4beb058f20e90bdd47b48f3b1b176'
report = {'kind':'INDEPENDENT_SOURCE_DATA_ONLY','candidate_or_test_import_or_execution':False,'verified_replacement_leaves':verified,'cumulative_delta_hashes_verified':2,'changed_existing_fence_functions':changed,'F1_complete_function_reconstructions':f1_compared,'finite_registry_source_counts':counts,'finite_registry_total':171,'source_phase_totals':{p:sum(v.values()) for p,v in expected.items()},'source_total':424,'original253_expectations_preserved':True,'active_nested_repair_refs_verified':sorted(nested),'binary_pin_verified_in_existing_map_only':True,'affinity':sorted(os.sched_getaffinity(0)),'address_space_limit':resource.getrlimit(resource.RLIMIT_AS)}
(OUT/'static-review.json').write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps({'status':'PASS_SOURCE_DATA_ONLY','leaves':len(verified),'F1':len(f1_compared),'registry_source_cases':sum(counts.values()),'full_source_cases':424,'nested_repair_refs':len(nested)}))
