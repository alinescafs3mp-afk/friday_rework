# Data expressions only: never import/evaluate/exec a candidate module or function.
def data(node,names):
 if isinstance(node,ast.Constant):return node.value
 if isinstance(node,ast.Name):return names[node.id]
 if isinstance(node,ast.Dict):return {data(k,names):data(v,names) for k,v in zip(node.keys,node.values)}
 if isinstance(node,(ast.List,ast.Tuple)):return [data(v,names) for v in node.elts]
 if isinstance(node,ast.BinOp) and isinstance(node.op,ast.Div):return data(node.left,names)/data(node.right,names)
 if isinstance(node,ast.BinOp) and isinstance(node.op,ast.Add):return data(node.left,names)+data(node.right,names)
 if isinstance(node,ast.Call) and isinstance(node.func,ast.Name) and node.func.id=='str' and len(node.args)==1 and not node.keywords:return str(data(node.args[0],names))
 raise AssertionError('Non-data expression refused: '+ast.dump(node))
