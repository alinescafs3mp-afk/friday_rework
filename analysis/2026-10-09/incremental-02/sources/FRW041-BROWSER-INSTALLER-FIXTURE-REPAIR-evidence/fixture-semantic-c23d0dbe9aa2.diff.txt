+def test_controlled_prestart_null_logging_matches_fragment_and_preserves_snapshot():
+    snapshots={name:(ROOT/name).read_bytes() for name in (
+        'prestart-observation.json','historical-user-prestart-observation.json')}
+    row=json.loads(snapshots['prestart-observation.json'])
+    legacy=dict(line.split('=',1) for line in row['loaded_stdout'].splitlines() if '=' in line)
+    legacy_typed=json.loads(row['typed_stdout'])['data'][0]
+    state,*rest=controlled_system_prestart_fixture()
+    current_typed=rest[-1]['data'][0]
+    fragment=(ROOT/'browser.service').read_text().splitlines()
+    for name in ('StandardOutput','StandardError'):
+        assert legacy[name]=='append'
+        assert legacy_typed[name]=={'type':'s','data':'append'}

+        assert state[name]=='null'
+        assert current_typed[name]=={'type':'s','data':'null'}
+    unit.effective(state,*rest)
