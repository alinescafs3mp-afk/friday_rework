            raise RetainedForTest()
        return trace
    try:
        sys.settrace(trace)
        with pytest.raises(RetainedForTest):loader.retain_custody({'deadline_monotonic':60.0},failure,failure)
    finally:sys.settrace(previous_trace)
    assert events==['failed_report','original_process_retained']

@pytest.mark.parametrize('failure',[None,'journal','chmod','replacement'])
def test_R2_real_umask_mode_and_durable_identity_before_chmod(loader,monkeypatch,tmp_path,failure):
    stage=tmp_path/'stage';events=[];real_fstat=os.fstat;real_stat=os.stat;real_chmod=os.fchmod
    monkeypatch.setattr(loader,'STAGE',stage)
    monkeypatch.setattr(loader.os,'fstat',lambda fd:root_fixture_metadata(real_fstat(fd)))
    replacement=[False]
    def stated(*args,**kwargs):
        s=root_fixture_metadata(real_stat(*args,**kwargs))
        if replacement[0]:values=list(s);values[1]+=1;s=os.stat_result(values)
        return s
    monkeypatch.setattr(loader.os,'stat',stated)
    def record(row):
        events.append(row)
        if row['kind']=='created':
            assert stat.S_IMODE(real_stat(stage).st_mode)==0o700
            assert row['identity']==[real_stat(stage).st_dev,real_stat(stage).st_ino]
            if failure=='journal':raise OSError('journal fsync failed')
    monkeypatch.setattr(loader,'journal_append',record)
    def chmod(fd,mode):
        assert events[-1]['kind']=='created' and mode==0o755
        if failure=='chmod':raise OSError('chmod failed')
        real_chmod(fd,mode)
        if failure=='replacement':replacement[0]=True
    monkeypatch.setattr(loader.os,'fchmod',chmod)
    fd=os.open(tmp_path,os.O_RDONLY|os.O_DIRECTORY);previous=os.umask(0o077)
    try:
        if failure:
            with pytest.raises((OSError,RuntimeError)):loader.prepare_stage(fd)
        else:
            loader.prepare_stage(fd)
            assert stat.S_IMODE(real_stat(stage).st_mode)==0o755
            # Actual descriptor file mode plus directory traversal bits; no root,
            # setuid or system path. A later live gate must still prove UID1000.
            asset=stage/'receipt.json';asset.write_bytes(b'private');asset.chmod(0o444)
            assert stat.S_IMODE(real_stat(asset).st_mode)==0o444 and asset.read_bytes()==b'private'
        if failure in ('journal','chmod'):assert stat.S_IMODE(real_stat(stage).st_mode)==0o700
    finally:os.umask(previous);os.close(fd)
