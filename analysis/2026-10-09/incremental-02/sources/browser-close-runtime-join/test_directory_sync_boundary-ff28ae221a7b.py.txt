"""SOURCE CONTROLS / NOT_RUN. Pure positives and refusal-only wrapper controls.

These do not impersonate native callers, install the fence, or claim a native
fsync success. The genuine short qualification is specified separately.
"""
import ast
import os
import sys
from pathlib import Path
import pytest
import private_effect_fence as fence

FLAGS=os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW

class Refused(RuntimeError):pass

def refuse(*args,**kwargs):raise Refused(str(args))


def test_directory_sync_exact_native_open_shape_is_accepted_purely():
    assert fence.directory_sync_open_allowed(FLAGS,0o777,None,166)

    (FLAGS,0o777,-1,166),(FLAGS,0o777,0,166),(FLAGS,0o777,True,166),

@pytest.mark.parametrize('fd',[9,8,True,'9'])
def test_directory_sync_direct_or_foreign_fsync_refuses_before_kernel(monkeypatch,fd):
    calls=[]
    monkeypatch.setattr(fence,'DIRECTORY_SYNC_PENDING',{'fd':9,'closed':False,'stage':'opened','fsync_calls':0})
    monkeypatch.setattr(fence,'deny',refuse)
    monkeypatch.setitem(fence.RAW,'fsync',lambda value:calls.append(value))
    with pytest.raises(Refused):fence.directory_sync_fsync(fd)
    assert calls==[]

def test_directory_sync_pending_cannot_claim_cessation(monkeypatch):
    monkeypatch.setattr(fence,'DIRECTORY_SYNC_PENDING',{})
    with pytest.raises(RuntimeError,match='pending_at_cessation'):fence.directory_sync_cessation()


@pytest.mark.parametrize('key,value',[('open_audits',0),('open_audits',2),('fsync_calls',0),
                                      ('fsync_calls',2),('closed',False),('ancestry_closed',False)])
def test_directory_sync_incomplete_closure_never_passes(monkeypatch,key,value):
    row=dict(open_audits=1,fsync_calls=1,closed=True,ancestry_closed=True)
    row[key]=value
    monkeypatch.setattr(fence,'DIRECTORY_SYNC_PENDING',None)
    monkeypatch.setattr(fence,'DIRECTORY_SYNC_ROWS',[row])
    with pytest.raises(RuntimeError,match='incomplete_row'):fence.directory_sync_cessation()


def test_directory_sync_completed_primitive_is_not_product_acceptance(monkeypatch):
    row=dict(open_audits=1,fsync_calls=1,closed=True,ancestry_closed=True,fsync_errno=5)
    monkeypatch.setattr(fence,'DIRECTORY_SYNC_PENDING',None)
    monkeypatch.setattr(fence,'DIRECTORY_SYNC_ROWS',[row])
    result=fence.directory_sync_cessation()
    assert result['all_closed'] and result['rows'][0]['fsync_errno']==5
    assert result['scope']=='ANCESTOR_DIRECTORY_DURABILITY_ONLY'


def test_directory_sync_complete_old_function_bodies_and_privileges_are_preserved():
    from directory_sync_source_comparator import directory_sync_predecessor,PREDECESSOR
    current=Path(fence.__file__).read_text()
    tree=ast.parse(current)
    functions={n.name:n for n in tree.body if isinstance(n,ast.FunctionDef)}
    assert len(functions)==sum(isinstance(n,ast.FunctionDef) for n in tree.body)
    old=directory_sync_predecessor(functions)
    old_tree=ast.parse(PREDECESSOR.read_text())
    old_assign=[ast.dump(n,include_attributes=False) for n in old_tree.body if not isinstance(n,ast.FunctionDef)]
    new_assign=[ast.dump(n,include_attributes=False) for n in tree.body if not isinstance(n,ast.FunctionDef)
                and not (isinstance(n,ast.Assign) and all(isinstance(t,ast.Name) and t.id.startswith('DIRECTORY_SYNC_') for t in n.targets))]
    assert new_assign==old_assign
    assert ast.dump(functions['read_ok'])==ast.dump(old['read_ok'])
    assert ast.dump(functions['check_fd'])==ast.dump(old['check_fd'])


def test_directory_sync_has_real_fsync_and_final_closure_error_paths():
    tree=ast.parse(Path(fence.__file__).read_text());functions={n.name:n for n in tree.body if isinstance(n,ast.FunctionDef)}
    fsync=functions['directory_sync_fsync'];opening=functions['directory_sync_open']
    assert sum(isinstance(n,ast.Call) and ast.unparse(n.func)=="RAW['fsync']" for n in ast.walk(fsync))==2
    assert any(isinstance(n,ast.Raise) and n.exc is None for n in ast.walk(fsync))
    assert sum(isinstance(n,ast.Call) and ast.unparse(n.func)=="RAW['open']" for n in ast.walk(opening))==1
    assert "ownership_close([fd])" in ast.unparse(opening)
    assert "ownership_close(chain)" in ast.unparse(opening)
    assert "ownership_close([fd])" in ast.unparse(functions['directory_sync_close'])
    assert "ownership_close(row['chain'])" in ast.unparse(functions['directory_sync_close'])


@pytest.mark.parametrize('condition',['pending','budget','direct'])
def test_directory_sync_open_rejects_pending_exhausted_and_direct_requests_without_syscall(monkeypatch,condition):
    calls=[]
    monkeypatch.setattr(fence,'DIRECTORY_SYNC_PENDING',{} if condition=='pending' else None)
    monkeypatch.setattr(fence,'DIRECTORY_SYNC_ROWS',[{}]*fence.DIRECTORY_SYNC_MAX_CALLS if condition=='budget' else [])
    monkeypatch.setattr(fence,'deny',refuse)
    monkeypatch.setitem(fence.RAW,'open',lambda *a,**kw:calls.append((a,kw)))
    with pytest.raises(Refused):fence.directory_sync_open(Path('/'),FLAGS,0o777,None,sys._getframe())
    assert calls==[]
