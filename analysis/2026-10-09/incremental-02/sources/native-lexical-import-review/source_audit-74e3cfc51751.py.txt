"""Independent source/data audit; never import or execute candidate functions."""
from pathlib import Path
import ast
import difflib
import hashlib
import json
import stat

W = Path('<ASTRA_WORK>')
OUT = W / 'native-lexical-import-review'
REPAIR = W / 'native-lexical-import-repair'
NEW = REPAIR / 'candidate'
OLD = W / 'collection-ancestor-metadata-repair/candidate'


def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def read(path):
    return json.loads(path.read_bytes())


checks = {}


def check(key, value):
    checks[key] = bool(value)


result = read(REPAIR / 'result.json')
check('task_result_hash', digest(REPAIR / 'result.json') == '1c4ecfea2af33e348b33c507f951b98b14ed247a100eabdf93cea8e2883073a2')
check('task_pair_hash', digest(NEW / 'PAIR-MANIFEST.json') == '7cdeb9b4b54c2568f8186d5a95e11060143bbf8a6cc3ab077bcce024796d17de')
pins = read(REPAIR / 'candidate-pins.json')
new_files = {str(p.relative_to(NEW)): p for p in NEW.rglob('*') if p.is_file()}
old_files = {str(p.relative_to(OLD)): p for p in OLD.rglob('*') if p.is_file()}
check('new_inventory_exact', set(new_files) == set(pins))
check('new_all31_hashes', len(pins) == 31 and all(digest(NEW / n) == h for n, h in pins.items()))
check('new_all31_0400_regular_nonsymlink', all(not p.is_symlink() and stat.S_ISREG(p.lstat().st_mode) and stat.S_IMODE(p.lstat().st_mode) == 0o400 for p in new_files.values()))
check('artifact_hashes', all(digest(REPAIR / n) == h for n, h in read(REPAIR / 'artifact-pins.json').items()))
delta = read(REPAIR / 'delta.json')
changed = sorted(n for n in set(old_files) & set(new_files) if old_files[n].read_bytes() != new_files[n].read_bytes())
added = sorted(set(new_files) - set(old_files))
deleted = sorted(set(old_files) - set(new_files))
unchanged = sorted(n for n in set(old_files) & set(new_files) if old_files[n].read_bytes() == new_files[n].read_bytes())
check('delta_changed_exact', changed == sorted(delta['changed']))
check('delta_added_exact', added == sorted(delta['added']))
check('delta_unchanged_exact', unchanged == sorted(delta['unchanged_exact_bytes']))
check('no_deletions', not deleted)
diff = ''.join(''.join(difflib.unified_diff((OLD / n).read_text().splitlines(keepends=True) if n in old_files else [], (NEW / n).read_text().splitlines(keepends=True) if n in new_files else [], fromfile='prior/' + n, tofile='candidate/' + n)) for n in sorted(set(old_files) | set(new_files)))
check('full_diff_exact', diff == (REPAIR / 'candidate.diff').read_text())
check('prior_pair_hash', digest(OLD / 'PAIR-MANIFEST.json') == delta['prior_pair']['sha256'])
for root, prefix in ((OLD, 'old'), (NEW, 'new')):
    pair = read(root / 'PAIR-MANIFEST.json')
    for kind in ('producer', 'owner-helper'):
        manifest = read(root / kind / 'source-pins.json')
        check(prefix + '_' + kind + '_inventory', {p.name for p in (root / kind).iterdir()} == set(manifest) | {'source-pins.json'})
        check(prefix + '_' + kind + '_pins', all(digest(root / kind / n) == h for n, h in manifest.items()))
    check(prefix + '_pair_source_pin', digest(root / 'producer/source-pins.json') == pair['source_pins_sha256'])
    check(prefix + '_pair_helper_pin', digest(root / 'owner-helper/source-pins.json') == pair['helper_pins_sha256'])

pair = read(NEW / 'PAIR-MANIFEST.json')
spec = read(NEW / 'owner-helper/input-spec.json')
check('pair_exact_roots', pair['source'] == str(NEW / 'producer') and pair['helper'] == str(NEW / 'owner-helper'))
for field, relative in (('helper_sha256', 'owner-helper/owner_observer.py'), ('input_spec_sha256', 'owner-helper/input-spec.json'), ('receipt_schema_sha256', 'owner-helper/receipt-schema.json'), ('import_path_pins_sha256', 'producer/import-path-pins.json')):
    check(field + '_propagation', digest(NEW / relative) == pair[field] == result.get(field, pair[field]))
check('spec_source_and_import_paths', spec['source'] == pair['source'] and spec['source_pins_sha256'] == pair['source_pins_sha256'] and spec['import_path_manifest'] == str(NEW / 'producer/import-path-pins.json'))
clean = read(NEW / 'producer/clean-invocation.json')
check('invocation_template_no_old_root', str(OLD) not in (NEW / 'producer/clean-invocation.json').read_text())
check('invocation_template_current_root_count', (NEW / 'producer/clean-invocation.json').read_text().count(str(NEW / 'producer/invocation.py')) == 3)
check('invocation_import_path_current', clean['import_path_manifest'] == spec['import_path_manifest'])
check('no_runtime_grant', pair['runtime_grant'] is False and pair['independent_source_acceptance'] is False)

ast_files = []
for path in NEW.rglob('*.py'):
    ast.parse(path.read_text(), filename=str(path))
    ast_files.append(str(path.relative_to(NEW)))
ast.parse((REPAIR / 'test_guard_alias_source.py').read_text())


def functions(path):
    source = path.read_text()
    return {n.name: ast.get_source_segment(source, n) for n in ast.parse(source).body if isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef))}


oldfuncs = functions(OLD / 'producer/import_path_readiness.py')
newfuncs = functions(NEW / 'producer/import_path_readiness.py')
for name in delta['protected_helper_functions_exact_bytes']:
    check('protected_function_' + name, oldfuncs[name] == newfuncs[name])
policy_functions = ('own', 'allowed', 'credential', 'readable', 'check_open', 'guarded_open', 'audit')
guard_funcs = functions(NEW / 'producer/guard.py')
native_funcs = functions(NEW / 'producer/native_guard.py')
check('guard_policies_same', all(guard_funcs[n] == native_funcs[n] for n in policy_functions))
trace = read(REPAIR / 'guard-boundary-review.json')
check('guard_trace_exact_extracted_sources', all(guard_funcs[n] == body for n, body in trace['guard_policy_exact_sources'].items()))

before = read(OLD / 'producer/import-path-pins.json')
after = read(NEW / 'producer/import-path-pins.json')
alias = after.pop('native_lexical_alias')
for field in ('directory_identities', 'metadata_inventories'):
    newmeta = after[field].pop(str(NEW / 'producer'))
    oldmeta = before[field].pop(str(OLD / 'producer'))
    if field == 'metadata_inventories':
        check('relocated_producer_metadata_same', newmeta == oldmeta == {})
    else:
        s = (NEW / 'producer').lstat()
        check('relocated_producer_identity_actual', newmeta == [s.st_dev, s.st_ino, s.st_mode, s.st_uid, s.st_gid])
check('import_manifest_other_fields_unchanged', before == after)
native = spec['roots']['native']
raw = native + '/tests/tools/../..'
parts = raw.split('/')[1:]
paths = ['/'] + ['/' + '/'.join(parts[:i]) for i in range(1, len(parts) + 1)]
check('alias_exact_11_components', alias['raw'] == raw and alias['target'] == native and len(paths) == 11 and [r['path'] for r in alias['steps']] == paths)
expected = read(NEW / 'producer/import-path-pins.json')['directory_identities']
check('alias_last5_existing_roots_only', all(r['identity'] == expected[p] for r, p in zip(alias['steps'][-5:], [native, native + '/tests', native + '/tests/tools', native + '/tests', native])))
check('alias_not_canonical_entry', raw not in expected)
corpus = read(NEW / 'producer/corpus-pins.json')
gate = read(NEW / 'producer/gate-plan.json')
source_evidence = []
for rel, sha in alias['source_files'].items():
    path = Path(native) / rel
    text = path.read_text()
    tree = ast.parse(text)
    inserts = [n for n in ast.walk(tree) if isinstance(n, ast.Call) and ast.unparse(n.func) == 'sys.path.insert']
    source_evidence.append({'path': str(path), 'sha256': digest(path), 'insertions': [{'line': n.lineno, 'source': ast.get_source_segment(text, n)} for n in inserts]})
    check('alias_source_' + rel, sha == corpus['native'][rel] == digest(path) and str(path) in gate['native'])

control_inventory = []
for path in (NEW / 'producer/test_import_path_source.py', NEW / 'producer/test_native_lexical_source.py', REPAIR / 'test_guard_alias_source.py'):
    definitions = []
    for node in ast.parse(path.read_text()).body:
        if not isinstance(node, ast.FunctionDef) or not node.name.startswith('test_'):
            continue
        cases = 1
        for dec in node.decorator_list:
            if isinstance(dec, ast.Call) and ast.unparse(dec.func) == 'pytest.mark.parametrize':
                assert isinstance(dec.args[1], (ast.List, ast.Tuple))
                cases *= len(dec.args[1].elts)
        definitions.append({'name': node.name, 'line': node.lineno, 'declared_cases': cases})
    control_inventory.append({'path': str(path), 'sha256': digest(path), 'declared_count': sum(x['declared_cases'] for x in definitions), 'definitions': definitions, 'execution': 'NOT_RUN'})
check('control_counts35_31_10', [x['declared_count'] for x in control_inventory] == [35, 31, 10])

# Reconstruct ONLY the owner's expanded hash dictionary from immutable maps.
# Do not import owner_observer, traverse corpora, or rehash payloads.
expanded = {str(NEW / 'producer/source-pins.json'): pair['source_pins_sha256']}
expanded.update({str(NEW / 'producer' / n): h for n, h in read(NEW / 'producer/source-pins.json').items()})
expanded.update({p: row['sha256'] for p, row in read(NEW / 'producer/runtime-pins.json').items()})
expanded.update(read(NEW / 'producer/ancillary-pins.json'))
for kind, root in spec['roots'].items():
    expanded.update({str(Path(root) / rel): h for rel, h in corpus[kind].items()})
for field in ('baseline_source_review', 'observer_review', 'preprocessing_map', 'prior_root_r1_review'):
    expanded[spec[field]['path']] = spec[field]['sha256']
expanded[str(NEW / 'owner-helper/source-pins.json')] = pair['helper_pins_sha256']
expanded.update({str(NEW / 'owner-helper' / n): h for n, h in read(NEW / 'owner-helper/source-pins.json').items()})
provenance_path = W / 'collection-classification-metadata-join/provenance.json'
provenance = read(provenance_path)
refs = {k: provenance[k] for k in ('current_pair', 'current_producer', 'current_observer')}
check('classification_refs_bound_to_old_pair', refs['current_pair']['path'] == str(OLD / 'PAIR-MANIFEST.json') and refs['current_producer']['path'] == str(OLD / 'producer/controls.py') and refs['current_observer']['path'] == str(OLD / 'owner-helper/owner_observer.py'))
check('classification_current_producer_equal', refs['current_producer']['sha256'] == digest(NEW / 'producer/controls.py') == digest(Path(refs['current_producer']['path'])))
check('classification_current_observer_equal', refs['current_observer']['sha256'] == digest(NEW / 'owner-helper/owner_observer.py') == digest(Path(refs['current_observer']['path'])))
origin_report = []
for rule, row in provenance['dynamic_rules'].items():
    ref = row['origin']
    origin_report.append({'rule': rule, 'path': ref['path'], 'sha256': ref['sha256'], 'new_expanded_pinmap_match': expanded.get(ref['path']) == ref['sha256']})
origin_report.append({'rule': 'asyncio_emitted_current_producer', **refs['current_producer'], 'new_expanded_pinmap_match': expanded.get(refs['current_producer']['path']) == refs['current_producer']['sha256']})
origin_report.append({'rule': 'asyncio_rebound_current_producer', 'path': str(NEW / 'producer/controls.py'), 'sha256': refs['current_producer']['sha256'], 'new_expanded_pinmap_match': expanded.get(str(NEW / 'producer/controls.py')) == refs['current_producer']['sha256']})
catalog_missing_from_expanded = [{'path': f['path'], 'sha256': f['sha256']} for f in provenance['files'] if expanded.get(f['path']) != f['sha256']]

out = {'state': 'SOURCE_HASH_AST_DATA_ONLY', 'checks': checks, 'failed_checks': [k for k,v in checks.items() if not v], 'changed': changed, 'added': added, 'unchanged': unchanged, 'candidate_files_hashed': len(new_files), 'prior_candidate_files_hashed': len(old_files), 'ast_sources': ast_files, 'alias_source_evidence': source_evidence, 'controls': control_inventory, 'expanded_pinmap_count_data_only': len(expanded), 'classification_refs': refs, 'dynamic_origin_membership': origin_report, 'catalog_files_not_in_current_expanded_pinmap': catalog_missing_from_expanded, 'candidate_imports': 0, 'tests_executed': 0, 'collection_executed': 0, 'broad_payload_rehash': False}
(OUT / 'source-audit.json').write_text(json.dumps(out, indent=2) + '\n')
print(json.dumps({'checks': len(checks), 'failed_checks': out['failed_checks'], 'changed': changed, 'added': added, 'control_counts': [x['declared_count'] for x in control_inventory], 'expanded_pinmap_count': len(expanded), 'dynamic_origin_membership': origin_report, 'catalog_missing_count': len(catalog_missing_from_expanded)}, indent=2))
