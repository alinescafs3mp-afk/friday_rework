"""INERT renderer for the existing reviewed finite offline runner.
Call only in a separate explicit ROOT preparation assignment after both packages
stabilize. Does not create clock/admission, take a lock or run/import any source.
The rendered source needs independent review and external original180/70s guard.
"""
import hashlib
import json
import os
from pathlib import Path
import re
import stat

ROOT = Path(__file__).parent
BASE = Path('<ASTRA_WORK>/browser-installer-fixture-offline-qualification/run.py')


def exact(path, expected):
    path = Path(path)
    assert path.is_absolute() and '..' not in path.parts
    assert path == path.resolve(strict=True) and not path.is_symlink()
    s = path.stat()
    assert stat.S_ISREG(s.st_mode) and s.st_nlink == 1 and s.st_uid == 1000
    assert not s.st_mode & 0o022
    value = path.read_bytes()
    assert len(value) == expected['bytes'] and hashlib.sha256(value).hexdigest() == expected['sha256']
    return value


def replace_once(source, old, new):
    assert source.count(old) == 1, ('reviewed_runner_interface_drift', old)
    return source.replace(old, new)


def render(contract_path, contract_sha256, private_path, out_path):
    """SOURCE preparation API only; no CLI or automatic current selection.
    final contract carries actual immutable paths/hashes from ROOT, not templates.
    Caller owns a fresh0700 private tree and OUT; this never adopts an old run.
    """
    raw = Path(contract_path).read_bytes()
    assert hashlib.sha256(raw).hexdigest() == contract_sha256
    contract = json.loads(raw)
    assert contract['state'] == 'EXACT_FINAL_OFFLINE_SOURCE_SELECTED'
    assert contract['source_accept'] is True and contract['runtime_grant'] is False
    assert contract['OB1_OB2_closed_by_independent_review'] is True
    assert contract['source_preparation_authorized_by'] == 'Astra'
    # Exact independent accepted-source receipt, not self-declared booleans.
    review = json.loads(exact(contract['independent_review']['path'], contract['independent_review']))
    assert review['source_accept'] is True and review['runtime_grant'] is False
    final = Path(contract['final_source'])
    final_pin = contract['final_package']
    package = json.loads(exact(final/'package-pins.json', final_pin))
    assert review['package']['sha256'] == final_pin['sha256']
    rejected = '/FRW041-NATIVE-OFFLINE-BUILD-evidence/candidate-v2'
    assert not str(final).endswith(rejected), 'old_rejected_candidate_is_not_final'
    expected_assets = {'admin-control.py','preflight','start','observe','stop','cleanup','build.py','inputs.json','phase-plan.json'}
    files = {}
    for leaf,pin in package.items():
        assert re.fullmatch(r'[A-Za-z0-9_.-]+', leaf)
        assert leaf not in ('FINAL-SOURCE-SELECTION.json','gate_support.py','render_offline_gate.py')
        files[leaf] = exact(final/leaf, {'bytes':pin.get('size',pin.get('bytes')), 'sha256':pin['sha256']})
    assert expected_assets | {'install-system.py','runtime_binding.py','source-identity.json','INSTALL-ENTRY.sh','install-bundle.json','run-live-unit.py'} <= set(files)
    bundle = json.loads(files['install-bundle.json'])
    assert set(bundle['root_assets']) == expected_assets
    for leaf,pin in bundle['root_assets'].items():
        assert hashlib.sha256(files[leaf]).hexdigest() == pin['sha256']
    # Exact current gate source is independently pinned in contract; file presence
    # or counts never imply testing. No changing worktree, inferred latest input.
    for leaf,pin in contract['gate_files'].items():
        assert re.fullmatch(r'[A-Za-z0-9_.-]+',leaf)
        files[leaf] = exact(ROOT/leaf,pin)
    tests = ['test_offline_build_source.py','test_boundary_source.py',
             'test_original1800_settlement.py','test_control_clock_snapshot.py',
             'test_dashboard_phase_order.py']
    assert contract['fix_specific_tests'], 'OB1_OB2_actual_source_controls_required'
    for ref in contract['fix_specific_tests']:
        leaf=ref['leaf']
        assert re.fullmatch(r'test_[A-Za-z0-9_]+\.py',leaf) and leaf not in tests
        assert leaf not in files or files[leaf] == exact(ref['path'],ref)
        files[leaf] = exact(ref['path'],ref); tests.append(leaf)
    # The immutable 32-case legacy file is retained separately in this gate
    # publication; its one fixture addition supplies the repaired errno field.
    assert all(name in files for name in tests)
    selection = dict(contract)
    selection['files']={leaf:dict(bytes=len(value),sha256=hashlib.sha256(value).hexdigest()) for leaf,value in files.items()}
    files['FINAL-SOURCE-SELECTION.json']=(json.dumps(selection,indent=2)+'\n').encode()
    private=Path(private_path);out=Path(out_path)
    for path in (private,out):
        s=path.stat();assert path==path.resolve(strict=True) and stat.S_ISDIR(s.st_mode)
        assert s.st_uid==1000 and stat.S_IMODE(s.st_mode)==0o700
    assert not tuple(out.iterdir()), 'fresh_output_only'
    source=private/'source';copy=private/'copy'
    assert not source.exists() and not copy.exists(), 'no_adopt_or_replay'
    combined={leaf:dict(size=len(value),sha256=hashlib.sha256(value).hexdigest()) for leaf,value in files.items()}
    pins=(json.dumps(combined,indent=2)+'\n').encode()
    files['package-pins.json']=pins
    for tree in (source,copy):
        tree.mkdir(mode=0o700)
        for leaf,value in files.items():
            fd=os.open(tree/leaf,os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o600)
            with os.fdopen(fd,'wb') as stream:stream.write(value);stream.flush();os.fsync(stream.fileno())
            (tree/leaf).chmod(0o400)
        tree.chmod(0o500)
    for leaf in ('results','tmp','home','pytesttmp'):(private/leaf).mkdir(mode=0o700)
    base_pin=contract['reviewed_bounded_runner']
    text=exact(BASE,base_pin).decode()
    # Deliberately only literal data/selection adaptation. Audit/write/native
    # denial implementation and external GNU KILL70s, resources/lock unchanged.
    changes={
        "OUT = '<ASTRA_WORK>/browser-installer-fixture-offline-qualification'": 'OUT = '+repr(str(out)),
        "PRIVATE = '<PRIVATE_TEMP>'": 'PRIVATE = '+repr(str(private)),
        "SOURCE = '<SOL_WORK>/FRW041-BROWSER-INSTALLER-FIXTURE-REPAIR-evidence/candidate'": 'SOURCE = '+repr(str(source)),
        "COPY = PRIVATE + '/source'": "COPY = PRIVATE + '/copy'",
        "TEST_NAMES = ('test_administration_source.py', 'test_boundary_source.py', 'test_containment_source.py')": 'TEST_NAMES = '+repr(tuple(tests)),
        "PIN_SHA = '60edadce916a46071d1f0bcff4cf9a73e07622fdd0215117c3211c13ab68400d'": 'PIN_SHA = '+repr(hashlib.sha256(pins).hexdigest()),
        "clock['assignment'] == 'FRW041-BROWSER-INSTALLER-OFFLINE-RUN'": "clock['assignment'] == 'FRW041-OFFLINE-BUILD-GATE-RUN'",
        "clock['source_root_mode'] == 'ORIGINAL_PRIVATE_READ_ONLY_ROOT'": "clock['source_root_mode'] == 'FINAL_PRIVATE_READ_ONLY_TEST_VIEW'",
        "'source_root_mode':'ORIGINAL_PRIVATE_READ_ONLY_ROOT'": "'source_root_mode':'FINAL_PRIVATE_READ_ONLY_TEST_VIEW'",
        "clock['original_source_exception_accepted'] is True": "clock['final_readonly_test_view_accepted'] is True",
        "require(len(pins) == 24, 'complete25file_package_required')": "require(len(pins) == "+str(len(combined))+", 'complete_exact_final_test_view_required')",
        "require(sys.modules['probe'].__file__ == SOURCE+'/probe.py', 'probe_import_origin_changed')": "require(sys.modules['runtime_binding'].__file__ == SOURCE+'/runtime_binding.py' and sys.modules['gate_support'].__file__ == SOURCE+'/gate_support.py', 'final_dependency_import_origin_changed')",
        "require(state['private_unix_socket_count'] == 1 and len(bound_sockets) == 1, 'one_private_unix_bind_not_observed')": "require(state['private_unix_socket_count'] == 0 and len(bound_sockets) == 0, 'no_socket_admitted_in_selected_gate')",
    }
    for old,new in changes.items():text=replace_once(text,old,new)
    (out/'run.py').write_text(text);(out/'run.py').chmod(0o400)
    s=private.stat()
    (out/'private-temp.json').write_text(json.dumps(dict(device=s.st_dev,inode=s.st_ino))+'\n')
    input_pins={str(source/leaf):dict(size=len(value),sha256=hashlib.sha256(value).hexdigest()) for leaf,value in files.items()}
    # The rendered runner snapshots both identical views; its own source, final
    # selection and runtime_binding are among exact independently checked inputs.
    input_pins[str(out/'run.py')]=dict(size=len(text.encode()),sha256=hashlib.sha256(text.encode()).hexdigest())
    (out/'input-pins.json').write_text(json.dumps(input_pins,indent=2)+'\n')
    (out/'runner-adaptation.json').write_text(json.dumps({'baseline':base_pin,'changes':changes,'state':'SOURCE_ONLY_NEW_REVIEW_REQUIRED','runtime_grant':False},indent=2)+'\n')
    return dict(source=str(source),copy=str(copy),runner=str(out/'run.py'),tests=tests,
                gate_execution='NOT_RUN',runtime_grant=False,clock_created=False,
                original_future_offline_seconds=180,external_GNU_seconds=70,
                root_must_verify_full_parent_cgroup_reaping_and_original_budget=True)
