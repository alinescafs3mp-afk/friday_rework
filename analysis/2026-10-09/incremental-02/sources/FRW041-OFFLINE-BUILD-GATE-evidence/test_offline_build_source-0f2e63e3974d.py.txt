def exact_clock(module):
    start = time.monotonic()
    return dict(unit=module.UNIT, nonce=module.NONCE, assignment='FRW041-NATIVE-OFFLINE-BUILD-RUN',
                native_effects_authorized_by='Astra', budget_seconds=1800,
                started_monotonic=start, deadline_monotonic=start + 1800,
                boot_id=Path('/proc/sys/kernel/random/boot_id').read_text().strip())


@pytest.mark.parametrize('index', [1, 2])
def test_original_clock_positive_both_consumers(modules, index):
    module = modules[index]
    assert module.clock_valid(exact_clock(module))['budget_seconds'] == 1800


@pytest.mark.parametrize('field,value', [('budget_seconds', 120), ('deadline_monotonic', float('inf')),
    ('started_monotonic', True), ('nonce', '3f4901af6bb7a95f'), ('boot_id', 'foreign'),
    ('assignment', 'FRW041-BROWSER-NATIVE-CONTAINMENT-TRIAL'), ('native_effects_authorized_by', 'Sol')])
def test_no_old_clock_or_identity_transfer(modules, field, value):
    for module in modules[1:]:
        clock = exact_clock(module)
        clock[field] = value
        with pytest.raises(RuntimeError):
            module.clock_valid(clock)


def test_clock_cannot_shift_deadline_alone(modules):
    for module in modules[1:]:
        clock = exact_clock(module)
        clock['deadline_monotonic'] += 1
        with pytest.raises(RuntimeError):
            module.clock_valid(clock)
