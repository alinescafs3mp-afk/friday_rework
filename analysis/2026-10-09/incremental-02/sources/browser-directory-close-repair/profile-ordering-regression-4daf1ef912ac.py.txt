"""NOT RUN. Run once in a fresh bounded pinned CPython process, never in pytest.

An inert audit event exercises interpreter ordering only. No product import,
native syscall surrogate, descriptor, authority, or native effect receipt exists.
The hook dies with this process and is never installed in the parent runtime.
"""
import json
import sys

assert sys.implementation.name == 'cpython' and sys.version_info[:3] == (3, 14, 7)
assert sys.getprofile() is None
EVENT = 'friday.directory_close_repair.inert'
TOKEN = object()
seen = []
published = []


def hook_helper(token):
    assert token is TOKEN
    assert sys.getprofile() is observer
    # Synchronous publication remains observable without a profile callback
    # for this function: these are real frames, never manufactured frames.
    frame = sys._getframe()
    try:
        assert frame.f_back.f_code is hook.__code__
        assert frame.f_back.f_back.f_code is emit.__code__
        published.append('actual_hook_helper')
    finally:
        del frame


def hook(event, args):
    if event == EVENT:
        assert args == (TOKEN,) and not published
        hook_helper(args[0])


def emit():
    sys.audit(EVENT, TOKEN)
    # c_return must have been observed after the actual hook and before this.
    assert published == ['actual_hook_helper']
    assert ('emit', 'c_return', 'sys.audit') in seen


marker = OSError(5, 'pure ordering marker; no failing syscall was performed')
closed = []


def raises_with_finally():
    try:
        raise marker
    finally:
        closed.append('finally')


def c_failure():
    len(None)  # Genuine C exception event, no syscall or product substitution.


def observer(frame, event, arg):
    if frame.f_code in (hook.__code__, hook_helper.__code__):
        seen.append(('unexpected_hook_profile', event, None))
    elif frame.f_code is emit.__code__ and arg is sys.audit:
        if event == 'c_return':
            assert published == ['actual_hook_helper']
        seen.append(('emit', event, 'sys.audit'))
    elif frame.f_code is raises_with_finally.__code__ and event == 'return':
        assert arg is None and closed == ['finally']
        seen.append(('python_error_unwind', event, None))
    elif frame.f_code is c_failure.__code__ and arg is len:
        seen.append(('builtin_error', event, 'len'))
    assert len(seen) <= 16


sys.addaudithook(hook)
assert not getattr(hook, '__cantrace__', False)
sys.setprofile(observer)
try:
    emit()
    try:
        raises_with_finally()
    except OSError as error:
        assert error is marker
    else:
        raise AssertionError('Original exception did not propagate')
    try:
        c_failure()
    except TypeError:
        pass
    else:
        raise AssertionError('Genuine builtin exception did not propagate')
finally:
    sys.setprofile(None)
assert not any(row[0] == 'unexpected_hook_profile' for row in seen)
assert seen.count(('emit', 'c_call', 'sys.audit')) == 1
assert seen.count(('emit', 'c_return', 'sys.audit')) == 1
assert seen.count(('python_error_unwind', 'return', None)) == 1
assert seen.count(('builtin_error', 'c_call', 'len')) == 1
assert seen.count(('builtin_error', 'c_exception', 'len')) == 1
assert sys.getprofile() is None
print(json.dumps({'scope': 'REAL_CPYTHON_PROFILE_ORDERING_ONLY',
                  'events': seen, 'synchronous_hook_observation': published,
                  'original_exception_preserved': True,
                  'product_or_directory_native_qualification': False}))
