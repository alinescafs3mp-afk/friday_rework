def read(p): return json.loads(Path(p).read_text())
def sha(p):
    with Path(p).open('rb') as f: return hashlib.file_digest(f, 'sha256').hexdigest()
def ref(p): return {'path': str(p), 'sha256': sha(p), 'bytes': Path(p).stat().st_size}

def historical(k):
    return k.startswith(('baseline', 'prior_', 'old_', 'historical_', 'generation7_', 'generation8_', 'generation9_', 'generation10_', 'generation11_', 'precedent_'))
def pathmap(v):
    for old, new in [(P, C), (P.parent/'supplementary-f1', F), (P.parent/'changed-area', M), (S0, S),
        *[(P.parent/n, O/n) for n in ('root-runtime-contract.json','source-reconstruction.json','source-path-identities.json','source-census.json','known-runtime-prerequisite.json')]]:
        v = re.sub(re.escape(str(old)) + r'''(?=$|/|[\s'"])''', lambda _: str(new), v)
    return v
def remap(v):
    if isinstance(v, dict): return {pathmap(k): x if historical(k) else remap(x) for k,x in v.items()}
    if isinstance(v, list): return [remap(x) for x in v]
    return pathmap(v) if isinstance(v, str) else v
def refresh(v):
    if isinstance(v, dict):
        d = {k: x if historical(k) else refresh(x) for k,x in v.items()}
        if isinstance(d.get('path'), str) and Path(d['path']).is_file():
            p = Path(d['path'])
            if p.is_relative_to(O) or p.is_relative_to(S):
                if 'sha256' in d: d['sha256'] = sha(p)
                if 'bytes' in d: d['bytes'] = p.stat().st_size
                if 'entries' in d: d['entries'] = len(read(p))
        for k,x in d.items():
            if k.startswith('/') and isinstance(x,str) and re.fullmatch('[a-f0-9]{64}', x):
                if Path(k).is_relative_to(O) or Path(k).is_relative_to(S): d[k] = sha(k)
        return d
    if isinstance(v, list): return [refresh(x) for x in v]
    return v
def imports(p):
    rows = []
    def visit(n, scope='module'):
        if isinstance(n, (ast.Import, ast.ImportFrom)):
            rows.append({'line': n.lineno, 'scope': scope, 'source': ast.unparse(n)})
        scope += ('.class:' + n.name if isinstance(n, ast.ClassDef) else '.' + n.name if isinstance(n,(ast.FunctionDef,ast.AsyncFunctionDef)) else '')
        for child in ast.iter_child_nodes(n): visit(child, scope)
    visit(ast.parse(Path(p).read_text()))
    return rows
def counts(p):
    tree = ast.parse(Path(p).read_text()); constants = {}; out = {}
    for n in tree.body:
        if isinstance(n, ast.Assign) and isinstance(n.value,(ast.List,ast.Tuple)):
            for t in n.targets:
                if isinstance(t,ast.Name): constants[t.id] = len(n.value.elts)
    for n in tree.body:
        if not isinstance(n,(ast.FunctionDef,ast.AsyncFunctionDef)) or not n.name.startswith('test_'): continue
        count = 1
        for d in n.decorator_list:
            if isinstance(d,ast.Call) and isinstance(d.func,ast.Attribute) and d.func.attr == 'parametrize':
                seq = d.args[1]
                if isinstance(seq,(ast.List,ast.Tuple)): size = len(seq.elts)
                elif isinstance(seq,ast.Name):
                    size = {'SCOPES':5,'NODES':2}.get(seq.id, constants.get(seq.id)); assert size is not None
                elif isinstance(seq,ast.ListComp):
                    size = 1
                    for g in seq.generators:
                        assert isinstance(g.iter,(ast.List,ast.Tuple)) and not g.ifs
                        size *= len(g.iter.elts)
                else: raise AssertionError(ast.dump(seq))
                count *= size
        assert n.name not in out
        out[n.name] = count
    return out
