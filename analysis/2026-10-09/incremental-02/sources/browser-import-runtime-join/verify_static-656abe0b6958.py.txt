def counts(p):
    tree=ast.parse(Path(p).read_text());const={};out={}
    for n in tree.body:
        if isinstance(n,ast.Assign) and isinstance(n.value,(ast.List,ast.Tuple)):
            for t in n.targets:
                if isinstance(t,ast.Name):const[t.id]=len(n.value.elts)
    for n in tree.body:
        if not isinstance(n,(ast.FunctionDef,ast.AsyncFunctionDef)) or not n.name.startswith('test_'):continue
        count=1
        for d in n.decorator_list:
            if isinstance(d,ast.Call) and isinstance(d.func,ast.Attribute) and d.func.attr=='parametrize':
                seq=d.args[1]
                if isinstance(seq,(ast.List,ast.Tuple)):size=len(seq.elts)
                elif isinstance(seq,ast.Name):size={'SCOPES':5,'NODES':2}.get(seq.id,const.get(seq.id));assert size is not None
                elif isinstance(seq,ast.ListComp):
                    size=1
                    for g in seq.generators:assert isinstance(g.iter,(ast.List,ast.Tuple)) and not g.ifs;size*=len(g.iter.elts)
                else:raise AssertionError(ast.dump(seq))
                count*=size
        assert n.name not in out;out[n.name]=count
    return out

def walk(v,file,loc=''):
    if isinstance(v,dict):
        if isinstance(v.get('path'),str) and 'sha256' in v:
            assert checked(v['path'])==v['sha256'],('STRUCTURED_REF',str(file),loc,v['path'])
            if 'bytes' in v:assert Path(v['path']).stat().st_size==v['bytes']
            if 'entries' in v:assert len(read(v['path']))==v['entries']
            refs.append((str(file),loc))
        for k,x in v.items():
            if k.startswith('/') and isinstance(x,str) and re.fullmatch('[a-f0-9]{64}',x):assert checked(k)==x,('NESTED_MAP',str(file),loc,k)
            elif isinstance(x,str) and re.fullmatch('[a-f0-9]{64}',x) and k!='sha256':
                target=scalar_targets.get((str(file),loc+'.'+k))
                if target is None:unknown.append((str(file),loc+'.'+k,x))
                else:assert checked(target)==x;scalars.append((str(file),loc+'.'+k,str(target)))
            walk(x,file,loc+'.'+k)
    elif isinstance(v,list):
        for i,x in enumerate(v):walk(x,file,loc+'.'+str(i))
