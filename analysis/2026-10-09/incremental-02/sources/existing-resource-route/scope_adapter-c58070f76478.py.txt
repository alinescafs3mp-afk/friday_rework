"""Source proposal: only compose the existing finite foreground scope route.

No entry point, subprocess, scope creation, clock/admission constructor, lock
acquisition, retry, cleanup, or supervisor lives here. The caller already owns
the original Budget, unique scope name, quality-lock FD and namespace status
pipe. This function cannot establish those lifecycle facts on the caller's

Return values belong to the EXISTING dsh_prepare.run call inside Containment:

receipt of effective resources, inherited-FD survival or final cessation.
"""
import json
import math
import os
from pathlib import Path
import re
import stat


LOCK = Path('<LOCAL_HOME>/.jericho/runtime/friday-quality-ui-exclusive.lock')
LOCK_ID = (2050, 9044963, 1000, 1000, 1, 0, 0o664)


def _identity(value):
    return (value.st_dev, value.st_ino, value.st_uid, value.st_gid,
            value.st_nlink, value.st_size, stat.S_IMODE(value.st_mode))


def require_held_quality_fd(fd):
    """Inspect the caller's existing open-file description, without flock()."""
    if type(fd) is not int or fd < 3 or os.getuid() != 1000 or os.geteuid() != 1000:
        raise ValueError('existing_unprivileged_quality_owner_required')
    value = os.fstat(fd)
    if (not stat.S_ISREG(value.st_mode) or _identity(value) != LOCK_ID
            or LOCK.resolve() != LOCK or _identity(LOCK.lstat()) != LOCK_ID):
        raise ValueError('canonical_quality_lock_identity_changed')
    rows = [line.split() for line in Path(f'/proc/self/fdinfo/{fd}').read_text().splitlines()
            if line.startswith('lock:')]
    held = False
    for row in rows:
        if len(row) != 9 or row[2:5] != ['FLOCK', 'ADVISORY', 'WRITE'] or row[7:] != ['0', 'EOF']:
            continue
        device = row[6].split(':')
        try:
            held = held or (len(device) == 3
                           and (int(device[0], 16), int(device[1], 16), int(device[2]))
                           == (os.major(value.st_dev), os.minor(value.st_dev), value.st_ino))
        except ValueError:
            continue
    if not held:
        raise ValueError('existing_quality_open_description_not_exclusively_held')


def wrap_existing_namespace(budget, namespace_argv, *, unit, tasks, seconds, quality_fd):
    """Adapt already constructed Containment argv; never execute it here.

    The caller MUST retain this same budget/deadline, status-pipe owner and
    lock through native cessation and outer scope observation. Current
    Containment.run needs the small explicit integration described in result;
    it does not yet call this helper or forward quality_fd. No mutable global
    monkeypatch, subclass substitution or old-attempt reuse is intended.
    """
    from scripts.install_containment import Budget
    from scripts.worker_qualification import DSH_SCOPE_GUARD

    if type(budget) is not Budget:
        raise ValueError('existing_original_budget_required')
    remaining = budget.check(reserve=1)
    if (type(seconds) not in (int, float) or not math.isfinite(seconds)
            or not 0 < seconds <= remaining):
        raise ValueError('existing_command_limit_inside_original_budget_required')
    if (not isinstance(unit, str)
            or re.fullmatch(r'friday-qualify-[a-z0-9][a-z0-9-]{0,190}\.scope', unit) is None
            or type(tasks) is not int or not 0 < tasks < 2**31):
        raise ValueError('owned_unique_scope_and_original_task_limit_required')
    require_held_quality_fd(quality_fd)
    if (not isinstance(namespace_argv, list) or len(namespace_argv) < 16
            or namespace_argv[:2] != ['/usr/bin/bwrap', '--json-status-fd']
            or namespace_argv[3:7] != ['--unshare-pid', '--die-with-parent', '--new-session', '--as-pid-1']
            or namespace_argv[7] not in ('--bind', '--ro-bind')
            or namespace_argv[8:15] != ['/', '/', '--dev', '/dev', '--proc', '/proc', '--']
            or not isinstance(namespace_argv[15], str) or not namespace_argv[15].startswith('/')):
        raise ValueError('existing_containment_namespace_argv_required')
    status_fd = int(namespace_argv[2])
    if status_fd < 3 or status_fd == quality_fd or not stat.S_ISFIFO(os.fstat(status_fd).st_mode):
        raise ValueError('existing_distinct_native_status_pipe_required')
    envelope = {'unit': unit, 'memory_bytes': 2 << 30, 'cpu_percent': 200,
                'tasks': tasks,
                'boot_id': Path('/proc/sys/kernel/random/boot_id').read_text().strip(),
                'deadline': budget.deadline - 1}
    argv = ['/usr/bin/systemd-run', '--user', '--scope', '--quiet', '--no-ask-password',
            '--expand-environment=no', '--unit=' + unit,
            '--property=MemoryMax=2147483648', '--property=MemorySwapMax=0',
            '--property=CPUQuota=200%', '--property=TasksMax=' + str(tasks),
            '--property=KillMode=control-group', '--property=SendSIGKILL=yes',
            '--property=RuntimeMaxSec=' + str(seconds) + 's',
            '/usr/bin/python3', '-I', '-c', DSH_SCOPE_GUARD, json.dumps(envelope),
            *namespace_argv]
    budget.check(reserve=1)
    return argv, (status_fd, quality_fd), envelope
