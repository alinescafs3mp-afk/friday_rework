"""Independent host census: never import or execute the submitted candidate."""
from pathlib import Path
import datetime
import hashlib
import json
import os

DONOR = Path('<PROJECT>/.donors/dsh')
OUTPUT = Path(__file__).resolve().parent / 'dependency-inventory.json'
pending = [DONOR / 'packages/sandbox/sandbox-local', DONOR / 'vendor/cordis',
           DONOR / 'native/system/packages/entry']
packages, edges, absent, manifests = {}, [], [], {}
while pending:
    directory = pending.pop(0).resolve(strict=True)
    relative = str(directory.relative_to(DONOR))
    if relative in packages:
        continue
    raw = (directory / 'package.json').read_bytes()
    package = json.loads(raw)
    packages[relative] = {'name': package['name'], 'version': package['version']}
    manifests[relative + '/package.json'] = hashlib.sha256(raw).hexdigest()
    for section in ('dependencies', 'peerDependencies', 'optionalDependencies'):
        for name, version in sorted(package.get(section, {}).items()):
            location, found, searched = directory, None, []
            while location.is_relative_to(DONOR):
                if location.name != 'node_modules':
                    candidate = location / 'node_modules' / name
                    searched.append(str(candidate.relative_to(DONOR)))
                    if os.path.lexists(candidate):
                        found = candidate
                        break
                if location == DONOR:
                    break
                location = location.parent
            edge = {'from': relative, 'section': section, 'name': name, 'declared': version}
            optional = section == 'optionalDependencies' or (
                section == 'peerDependencies' and
                package.get('peerDependenciesMeta', {}).get(name, {}).get('optional') is True)
            if found is None:
                absent.append({**edge, 'optional': optional, 'searched': searched})
                continue
            target = found.resolve(strict=True)
            edges.append({**edge, 'via': str(found.relative_to(DONOR)),
                          'target': str(target.relative_to(DONOR)),
                          'link': os.readlink(found) if found.is_symlink() else None})
            pending.append(target)

files, links, sizes = {}, {}, {}
for relative in packages:
    package = DONOR / relative
    for directory, dirs, names in os.walk(package, followlinks=False):
        dirs[:] = [name for name in dirs if name != '.git' and
                   not (name == 'node_modules' and Path(directory) == package)]
        for name in dirs + names:
            source = Path(directory) / name
            rel = str(source.relative_to(DONOR))
            if source.is_symlink():
                links[rel] = os.readlink(source)
            elif source.is_file():
                files[rel] = '0' * 64
                sizes[rel] = source.stat().st_size
for edge in edges:
    if edge['link'] is not None:
        links[edge['via']] = edge['link']
for rel in list(files):
    parent = (DONOR / rel).parent
    while parent.is_relative_to(DONOR):
        metadata = parent / 'package.json'
        if metadata.is_file():
            name = str(metadata.relative_to(DONOR))
            files[name] = '0' * 64
            sizes[name] = metadata.stat().st_size
        if parent == DONOR:
            break
        parent = parent.parent
for rel in ('pnpm-lock.yaml', 'pnpm-workspace.yaml', 'LICENSE', 'THIRD_PARTY_NOTICES.md'):
    files[rel] = '0' * 64
    sizes[rel] = (DONOR / rel).stat().st_size
metadata = sorted(rel for rel in files if rel == 'package.json' or rel.endswith('/package.json'))
graph = {'method': 'STAGED_INSTALLED_RUNTIME_DEPENDENCY_CLOSURE_V1',
         'sourceRoot': str(DONOR), 'packages': packages, 'edges': edges,
         'optionalAbsent': absent, 'manifestPins': manifests}
result = {
    'state': 'INDEPENDENT_READ_ONLY_HISTORICAL_INVENTORY_NOT_EXECUTED_SELECTION',
    'root': str(DONOR), 'observed_msk': datetime.datetime.now(
        datetime.timezone(datetime.timedelta(hours=3))).isoformat(),
    'packages': packages, 'edges': edges, 'optional_absent': absent, 'manifest_pins': manifests,
    'package_count': len(packages), 'dependency_edges': len(edges),
    'required_absent': [row for row in absent if not row['optional']],
    'regular_files': len(files), 'symlinks': len(links), 'rows': len(files) + len(links),
    'regular_file_bytes': sum(sizes.values()),
    'resolution_metadata_files': len(metadata),
    'resolution_metadata_bytes': sum(sizes[rel] for rel in metadata),
    'files_links_indent_json_bytes': len(json.dumps(
        {'files': files, 'links': links}, sort_keys=True, indent=2).encode()),
    'files_links_graph_indent_json_bytes': len(json.dumps(
        {'files': files, 'links': links, 'sdkClosure': graph}, sort_keys=True, indent=2).encode()),
    'measurement': 'All installed dependencies/peers/optionals, including foreign platform packages; no devDependency seeds. Complete package bodies, installed edges, ancestor package scopes, root lock/workspace/notices. 64 ASCII zeroes stand for uncomputed file SHA256. Manifest bytes alone hashed. No candidate function/import/build/test/Node invoked.',
    'row_limit': 100000, 'map_limit_bytes': 8388608, 'file_bytes_limit': 2147483648,
    'bound_stage': 'UNBOUND', 'source_to_built_acceptance': 'UNBOUND',
    'runtime_grant': False,
}
OUTPUT.write_text(json.dumps(result, indent=2) + '\n')
print(json.dumps({key: result[key] for key in ('package_count', 'dependency_edges',
    'regular_files', 'symlinks', 'rows', 'regular_file_bytes',
    'resolution_metadata_files', 'resolution_metadata_bytes',
    'files_links_indent_json_bytes', 'files_links_graph_indent_json_bytes', 'required_absent')}))
