"""Focused controls. NOT RUN; not an execution grant or containment proof.

Models compile exact reviewed function bodies with explicit inert data facades.
They exercise exception ordering and evidence rules; they cannot prove kernel
stop, reap, power-loss durability, containment or actual plugin autoload.
Actual fault cuts remain NOT_RUN in fault-qualification.json; a separate
reviewed finite native owner is required for those executions.
"""

@pytest.mark.parametrize('fault', ['write', 'missing', 'malformed', 'readback'])
def test_stop_attempt_precedes_failing_checkpoint(source_overlay_model, fault):
    registry = source_overlay_model.registry()
    session = source_overlay_model.identity_known_session()
    order = []
    registry._terminate_host_pid = lambda pid, born: order.append(('terminate', pid, born))
    def failure(_):
        order.append(('checkpoint', fault))
        raise OSError(fault)
    registry._runtest_checkpoint = failure
    outcome = registry.stop_runtest(session)
    assert order == [('terminate', session.pid, session.host_start_time), ('checkpoint', fault)]
    assert session.runtest_stop_requested and session.process is not None
    assert outcome['status'] == 'STOP_UNCONFIRMED' and outcome['persistence_error'] == 'OSError'
    assert session.id in registry._running


def test_one_failed_stop_does_not_skip_next_owned_target(source_overlay_model):
    registry = source_overlay_model.registry()
    first = source_overlay_model.identity_known_session('first')
    second = source_overlay_model.identity_known_session('second')
    registry._running = {s.id: s for s in (first, second)}
    attempted = []
    def stop(session_id, **kwargs):
        attempted.append(session_id)
        if session_id == first.id:
            raise OSError('diagnostic failure')
        return {'status': 'STOP_REQUESTED'}
    registry.kill_process = stop
    assert registry.kill_all() == 0
    assert attempted == [first.id, second.id] and first._scope_stop_pending


@pytest.mark.parametrize('cut', ['return_transfer', 'birth', 'pgid', 'sid', 'checkpoint',
                              'stat', 'backend_start', 'runtest_custody'])
def test_post_popen_model_has_one_retained_donor_reader(source_overlay_model, monkeypatch, cut):
    donor = source_overlay_model.donor
    proc = Mock(pid=811, returncode=None)
    order = []
    def communicate(timeout):
        order.append('communicate')
        proc.returncode = -15
        return 'partial original output', None
    proc.communicate = communicate
    def spawn(*args, **kwargs):
        order.append('popen')
        if cut == 'return_transfer':
            raise OSError(cut)
        return proc
    def setup(p):
        assert p is proc
        order.append(cut)
        raise OSError(cut)
    saved = {}
    def capture(p, output, error, cuts):
        assert p is proc
        saved.update(output=output, error=type(error).__name__, cuts=list(cuts))
        order.append('capture')
    custody = {'spawn': spawn, 'after_spawn': setup, 'retained_handle': lambda: proc,
               'retained_pgid': lambda: 811, 'temporary_root': lambda: '/qualified/private/root',
               'remaining': lambda: 1, 'stop': lambda *args: order.append('stop'),
               'capture': capture, 'settle': lambda *args: order.append('settle'),
               'terminal': lambda code: order.append('terminal')}
    monkeypatch.setitem(donor, '_rmtree_force', lambda path: pytest.fail('failed-start evidence removed'))
    with pytest.raises(OSError, match=cut):
        donor['_run_one_file_once'](pathlib.Path('/qualified/file.py'), [],
                                    pathlib.Path('/qualified'), 1, custody=custody)
    assert order.count('popen') == order.count('communicate') == order.count('stop') == order.count('settle') == 1
    assert order.index('stop') < order.index('communicate') < order.index('capture') < order.index('settle')
    assert saved['output'] == 'partial original output' and saved['error'] == 'OSError'


@pytest.mark.parametrize('cut', ['stop', 'capture', 'settle', 'drain_reap'])
def test_cleanup_failure_retains_handle_and_attempts_later_cuts(source_overlay_model, monkeypatch, cut):
    donor = source_overlay_model.donor
    proc = Mock(pid=811, returncode=None)
    order = []
    def fail(operation):
        order.append(operation)
        if cut == operation:
            raise OSError(cut)
    def communicate(timeout):
        fail('drain_reap')
        proc.returncode = -15
        return 'retained', None
    proc.communicate = communicate
    def after(p):
        raise OSError('startup failure')
    custody = {'spawn': lambda *args, **kwargs: proc, 'after_spawn': after,
               'retained_handle': lambda: proc, 'retained_pgid': lambda: 811,
               'temporary_root': lambda: '/qualified/private/root', 'remaining': lambda: 1,
               'stop': lambda *args: fail('stop'), 'capture': lambda *args: fail('capture'),
               'settle': lambda *args: fail('settle'), 'terminal': lambda code: None}
    monkeypatch.setitem(donor, '_rmtree_force', lambda path: pytest.fail('unknown custody evidence removed'))
    with pytest.raises(RuntimeError, match='STOP_UNCONFIRMED'):
        donor['_run_one_file_once'](pathlib.Path('/qualified/file.py'), [],
                                    pathlib.Path('/qualified'), 1, custody=custody)
    assert order == ['stop', 'drain_reap', 'capture', 'settle']


@pytest.mark.parametrize('fsync_cut', ['file', 'directory_open', 'directory_fsync', 'readback'])
def test_strict_failure_fences_admission_without_losing_session(source_overlay_model, fsync_cut):
    # Exact source body, inert data facades. This is not real filesystem/Popen evidence.
    popen = source_overlay_model.popen
    healthy = source_overlay_model.registry()
    baseline = source_overlay_model.prelaunch_session('healthy')
    healthy._runtest_checkpoint(baseline)
    handle = healthy.spawn_runtest(baseline, ['inert'],
        {'start_new_session': True, 'cwd': baseline.cwd})
    popen.assert_called_once()
    assert baseline.process is handle and baseline.runtest_launch_pending is False
    popen.reset_mock()
    registry = source_overlay_model.registry()
    session = source_overlay_model.prelaunch_session('failed')
    assert session.process is None and session.runtest_launch_pending and session.cwd
    with source_overlay_model.inject_checkpoint_failure(registry, fsync_cut):
        with pytest.raises((OSError, RuntimeError, ValueError)):
            registry._runtest_checkpoint(session)
    assert session.runtest_stop_requested and session.runtest_persistence_error
    assert registry._running[session.id] is session
    with pytest.raises(RuntimeError, match='SPAWN_FENCED'):
        registry.spawn_runtest(session, ['inert'], {'start_new_session': True, 'cwd': session.cwd})
    popen.assert_not_called()
    assert session.process is None and session.runtest_launch_pending
    # Focused mutation sensitivity: removing ONLY the two persistence/stop
    # terms now reaches Popen. No unrelated no-replay condition masks the cut.
    source_overlay_model.unfenced_spawn(registry, session, ['inert'],
        {'start_new_session': True, 'cwd': session.cwd})
    popen.assert_called_once()


def test_already_owned_actual_handle_still_refuses_second_popen(source_overlay_model):
    registry = source_overlay_model.registry()
    session = source_overlay_model.identity_known_session()
    registry._runtest_checkpoint(session)
    actual = session.process
    with pytest.raises(RuntimeError, match='SPAWN_FENCED'):
        registry.spawn_runtest(session, ['no-replay'], {'start_new_session': True, 'cwd': session.cwd})
    source_overlay_model.popen.assert_not_called()
    assert session.process is actual


@pytest.mark.parametrize('transition', ['reserve', 'pid', 'stop'])
def test_generic_old_snapshot_cannot_overwrite_new_custody(source_overlay_model, transition):
    registry = source_overlay_model.registry()
    first = source_overlay_model.identity_known_session('old')
    registry._running = {first.id: first}
    entered = threading.Event(); release = threading.Event(); newer_started = threading.Event()
    errors = []
    def stalled_write(path, rows, **kwargs):
        entered.set()
        if not release.wait(1):raise RuntimeError('model barrier timeout')
        source_overlay_model.state['rows'] = json.loads(json.dumps(rows))
    source_overlay_model.utils.atomic_json_write = stalled_write
    def old():
        try:registry._write_checkpoint(strict=True)
        except BaseException as error:errors.append(error)
    def newer():
        try:
            newer_started.set()
            session = source_overlay_model.identity_known_session('new')
            if transition == 'pid':session.pid = 812
            if transition == 'stop':session.runtest_stop_requested = True
            with registry._lock:registry._running[session.id] = session
            registry._write_checkpoint(strict=True)
        except BaseException as error:errors.append(error)
    older = threading.Thread(target=old); newer = threading.Thread(target=newer)
    older.start()
    try:
        assert entered.wait(1)
        newer.start(); assert newer_started.wait(1)
    finally:
        release.set(); older.join(2)
        if newer.ident is not None:newer.join(2)
    assert not older.is_alive() and not newer.is_alive() and not errors
    rows = source_overlay_model.state['rows']
    assert {row['session_id'] for row in rows} == {'old','new'}
    row = next(r for r in rows if r['session_id'] == 'new')
    assert row['runtest_custody'] and row['runtest_stop_requested'] == (transition == 'stop')
    assert row['pid'] == (812 if transition == 'pid' else 811)
    # These are inert calls in the MODEL, not actual file/directory durability.
    assert source_overlay_model.state['directory_fsync_calls'] == 2


@pytest.mark.parametrize('drift_cut', ['body', 'entrypoint', 'missing_origin', 'autoload_map'])
def test_runtime_plugin_closure_refuses_declaration_gap(source_overlay_model, drift_cut):
    plan = json.loads(pathlib.Path(__file__).with_name('runtest-plan.json').read_text())
    ref = plan['runtime_plugins'][0]
    if drift_cut == 'body':plan['runtime_dependency_pins'].pop(ref['path'])
    elif drift_cut == 'entrypoint':plan['runtime_dependency_pins'].pop(ref['entrypoint'])
    elif drift_cut == 'missing_origin':plan['runtime_import_origins'][ref['module']]='/unreviewed/plugin.py'
    else:plan['runtime_autoload_entrypoints'][ref['entrypoint']]='f'*64
    with pytest.raises(ValueError):source_overlay_model.api['runtime_dependency_pins'](plan)
