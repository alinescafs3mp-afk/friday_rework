checked = {}
errors = []

def check(value, message):
    if not value:
        errors.append(message)

def ref(path):
    path = Path(path)
    data = path.read_bytes()
    row = {'path':str(path), 'sha256':hashlib.sha256(data).hexdigest(), 'bytes':len(data)}
    checked[str(path)] = row
    return row

def verify(row):
    got = ref(row['path'])
    for key in ('sha256', 'bytes'):
        if key in row:
            check(got[key] == row[key], f'{key} mismatch: {row["path"]}')
    st = Path(row['path']).lstat()
    check(stat.S_ISREG(st.st_mode), 'non-regular reference: ' + row['path'])
    if 'mode' in row:
        check(stat.S_IMODE(st.st_mode) == row['mode'], 'mode mismatch: ' + row['path'])
    if 'links' in row:
        check(st.st_nlink == row['links'], 'links mismatch: ' + row['path'])
    return got

def load(path):
    return json.loads(Path(path).read_bytes())

def refs_in(value):
    if isinstance(value, dict):
        if 'path' in value and 'sha256' in value and isinstance(value['path'], str):
            yield value
        for child in value.values():
            yield from refs_in(child)
    elif isinstance(value, list):
        for child in value:
            yield from refs_in(child)

def non_functions(tree,skip_new=False):
    rows=[]

        if skip_new and isinstance(n,ast.Assign) and all(isinstance(t,ast.Name) and t.id in added_assignments for t in n.targets):continue
        rows.append(ast.dump(n,include_attributes=False))
    return rows
