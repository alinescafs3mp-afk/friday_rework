def exact_clock(module):
    start = time.monotonic()
    return dict(unit=module.UNIT, nonce=module.NONCE, assignment='FRW041-NATIVE-OFFLINE-BUILD-RUN',
                native_effects_authorized_by='Astra', budget_seconds=1800,
                started_monotonic=start, deadline_monotonic=start + 1800,
                boot_id=Path('/proc/sys/kernel/random/boot_id').read_text().strip())

def damage_cargo_config(inputs, config, damage):
    if damage == 'changed':
        config.chmod(0o600)
        config.write_bytes(config.read_bytes().replace(b'true', b'nope'))
        config.chmod(0o400)
    elif damage == 'missing':
        config.unlink()
    elif damage == 'unlisted':
        del inputs['files'][str(config)]
    else:
        raise AssertionError(damage)


@pytest.mark.parametrize('damage', ['changed', 'missing', 'unlisted'])
def test_bad_config_stops_preparation_before_other_inputs(modules, cargo_config_input, damage):
    build = modules[2]
    inputs, config = cargo_config_input
    # The valid control reads actual bytes with production read_pin.
    assert build.verified_cargo_config(inputs) == config
    damage_cargo_config(inputs, config, damage)
    # No joined_root: reaching copy setup would fail with KeyError, not this
    # config-specific exception. No phase creation is reachable after failure.
    with pytest.raises((RuntimeError, FileNotFoundError), match='input_changed|cargo_config_not_pinned|cargo-offline.toml'):
        build.prepare(inputs, {}, time.monotonic() + 30)

@pytest.mark.parametrize('mutation,reason', [
    ('supported', None),
    ('source-derived-live', 'fresh_runtime_directory_required'),
    ('wrong-live-name', 'fresh_runtime_directory_required'),
    ('existing-live', 'fresh_runtime_directory_required'),
    ('dangling-live', 'fresh_runtime_directory_required'),
    ('writable-live-parent', 'foreign_writes_or_owner'),
    ('aliased-live-parent', 'symlink_path'),
    ('source-root', 'fixed_source_identity_required'),
    ('source-hash', 'source_changed'),
    ('source-inventory', 'complete_source_pin_set'),
    ('admission-path', 'fixed_runtime_admission_path'),
    ('admission-mode', 'immutable_runtime_admission_required'),
    ('review-hash', 'independent_review_changed'),
    ('review-package', 'exact_changed_area_review_required'),
    ('clock-path', 'fixed_immutable_original_clock_path'),
    ('clock-mode', 'fixed_immutable_original_clock_path'),
    ('clock-hash', 'clock_changed'),
    ('clock-old120', 'original_clock_invalid'),
    ('clock-expired', 'original_clock_invalid'),
    ('clock-deadline-reset', 'original_clock_invalid'),
    ('clock-boot', 'original_clock_invalid'),
    ('clock-nonce', 'unit_clock_mismatch'),
    ('dependency-hash', 'dependency_changed'),
])
def test_actual_admit_uses_exact_pinned_identity_without_source_parent_inference(modules, monkeypatch, mutation, reason):
    """Real admit/private/digest; only inert host/runtime/dependency observations.

    Exact source bytes/pins and IDENTITY are read from the actual published
    candidate. No runtime files, clocks, native commands, or grants are created.
    External executables are modeled by their declared pins, not rehashed per
    parameter; the new gate still has its separate real input/origin checks.
    """
    import hashlib
    import stat
    from types import SimpleNamespace
    runner = modules[0]
    real_path = Path
    source = real_path(runner.IDENTITY['root_source'])
    live = real_path(runner.IDENTITY['live_directory'])
    admission_path = real_path(runner.IDENTITY['runtime_admission_path'])
    clock_path = real_path(runner.IDENTITY['trial_clock_path'])
    review_path = admission_path.parent / 'modeled-independent-review.json'
    assert live.parent != source.parent.parent  # actual rebind regression
    pins = json.loads((source / 'package-pins.json').read_text())
    source_hashes = {name: pin['sha256'] for name, pin in pins.items()}
    source_hashes['package-pins.json'] = hashlib.sha256((source / 'package-pins.json').read_bytes()).hexdigest()
    dependencies = json.loads((source / 'source-pins.json').read_text())
    clock = dict(assignment='FRW041-NATIVE-OFFLINE-BUILD-RUN', native_effects_authorized_by='Astra',
                 unit=runner.IDENTITY['unit'], nonce=runner.IDENTITY['nonce'], budget_seconds=1800,
                 started_monotonic=1000.0, deadline_monotonic=2800.0, boot_id='modeled-boot')
    review = dict(state='INDEPENDENT_OFFLINE_BUILD_SYSTEM_SERVICE_SOURCE_ACCEPTED',
                  source_task=runner.SOURCE_TASK, runtime_grant=False,
                  package_pins_sha256=source_hashes['package-pins.json'])
    if mutation == 'review-package': review['package_pins_sha256'] = '0' * 64
    if mutation == 'clock-old120': clock.update(budget_seconds=120, deadline_monotonic=1120.0)
    if mutation == 'clock-expired': clock.update(started_monotonic=-800.0, deadline_monotonic=1000.0)
    if mutation == 'clock-deadline-reset': clock['deadline_monotonic'] += 1
    if mutation == 'clock-boot': clock['boot_id'] = 'foreign-boot'
    if mutation == 'clock-nonce': clock['nonce'] = '0000000000000000'
    if mutation == 'clock-path': clock_path = clock_path.with_name('foreign-clock.json')
    if mutation == 'admission-path': admission_path = admission_path.with_name('foreign-admission.json')
    encode = lambda value: (json.dumps(value) + '\n').encode()
    payloads = {str(clock_path): encode(clock), str(review_path): encode(review)}
    admission = dict(allow_one_offline_native_build=True, runtime_authorized=True,
                     source_task=runner.SOURCE_TASK, source_sha256=dict(source_hashes),
                     independent_review_path=str(review_path),
                     independent_review_sha256=hashlib.sha256(payloads[str(review_path)]).hexdigest(),
                     clock_path=str(clock_path), clock_sha256=hashlib.sha256(payloads[str(clock_path)]).hexdigest(),
                     live_directory=str(live))
    if mutation == 'source-derived-live': admission['live_directory'] = str(source.parent.parent / live.name)
    if mutation == 'wrong-live-name': admission['live_directory'] = str(live.with_name(live.name + '-foreign'))
    if mutation == 'source-hash': admission['source_sha256']['build.py'] = '0' * 64
    if mutation == 'source-inventory': admission['source_sha256'].pop('build.py')
    if mutation == 'review-hash': admission['independent_review_sha256'] = '0' * 64
    if mutation == 'clock-hash': admission['clock_sha256'] = '0' * 64
    payloads[str(admission_path)] = encode(admission)
    observed = {'source': [], 'dependencies': [], 'parent': []}

    class ModeledPath(type(real_path())):
        def read_bytes(self):
            if str(self) in payloads: return payloads[str(self)]
            return super().read_bytes()
        def read_text(self, *args, **kwargs):
            if str(self) in payloads: return payloads[str(self)].decode()
            if str(self) == '/proc/1/cgroup': return '0::/\n'
            if str(self) == '/proc/sys/kernel/random/boot_id': return 'modeled-boot\n'
            return super().read_text(*args, **kwargs)
        def resolve(self, strict=False):
            if str(self) in payloads: return self
            if str(self) == dependencies['python_alias']['path']:
                return ModeledPath(dependencies['python_alias']['resolved'])
            if self == live.parent and mutation == 'aliased-live-parent':
                return ModeledPath('/modeled-foreign-parent')
            return super().resolve(strict=strict)
        def stat(self, *args, **kwargs):
            if str(self) in payloads:
                mode = 0o400
                if (self == admission_path and mutation == 'admission-mode') or (self == clock_path and mutation == 'clock-mode'):
                    mode = 0o600
                return SimpleNamespace(st_uid=1000, st_mode=stat.S_IFREG | mode)
            if str(self) in dependencies['pins']:
                return SimpleNamespace(st_size=dependencies['pins'][str(self)]['size'])
            if self == live.parent:
                observed['parent'].append(str(self))
                if mutation == 'writable-live-parent':
                    return SimpleNamespace(st_uid=1000, st_mode=stat.S_IFDIR | 0o777)
            return super().stat(*args, **kwargs)
        def exists(self):
            if str(self) in ('/.dockerenv', '/run/.containerenv'): return False
            if self == live: return mutation == 'existing-live'
            return super().exists()
        def is_symlink(self):
            if self == live: return mutation == 'dangling-live'
            return super().is_symlink()

    original_digest = runner.digest
    def observed_digest(path):
        key = str(path)
        if key in dependencies['pins']:
            observed['dependencies'].append(key)
            return '0' * 64 if mutation == 'dependency-hash' else dependencies['pins'][key]['sha256']
        result = original_digest(path)
        if real_path(key).parent == source: observed['source'].append(real_path(key).name)
        return result
    os_proxy = SimpleNamespace(**vars(runner.os))
    os_proxy.geteuid = lambda: 1000
    os_proxy.environ = {}
    monkeypatch.setattr(runner, 'os', os_proxy)
    monkeypatch.setattr(runner, 'time', SimpleNamespace(monotonic=lambda: 1001.0))
    monkeypatch.setattr(runner, 'Path', ModeledPath)
    monkeypatch.setattr(runner, 'ROOT', ModeledPath(source if mutation != 'source-root' else source.parent / 'foreign-source'))
    monkeypatch.setattr(runner, 'digest', observed_digest)
    if reason is not None:
        with pytest.raises(RuntimeError, match='^' + reason + '$'):
            runner.admit(ModeledPath(admission_path))
    else:
        actual_admission, actual_clock, actual_live = runner.admit(ModeledPath(admission_path))
        assert actual_admission == admission and actual_clock == clock and actual_live == live
        assert set(observed['source']) == set(source_hashes)
        assert observed['dependencies'] == list(dependencies['pins'])
        assert observed['parent'] == [str(live.parent)]
    if mutation in ('source-derived-live', 'wrong-live-name', 'existing-live', 'dangling-live',
                    'writable-live-parent', 'aliased-live-parent'):
        assert set(observed['source']) == set(source_hashes)
        assert observed['dependencies'] == list(dependencies['pins'])
    if mutation == 'source-hash': assert 'build.py' in observed['source']
    if mutation == 'dependency-hash': assert observed['dependencies'] == [next(iter(dependencies['pins']))]
