@pytest.mark.parametrize('point', ['before-ready', 'observation', 'after-grant', 'build',
                                  'completion-notification', 'foreign-completion',
                                  'client-error', 'root-still-live', 'no-ready'])
def test_original_native_start_outcome_is_one_submission(runner, tmp_path, monkeypatch, point):
    """Exercise real native_start including its catch/finally. No real Popen.
    Pipe readiness/observation/client are finite models; not kernel observations.
    """
    state = {'cancel': None, 'launches': 0, 'writes': [], 'closed': [], 'killed': 0,
             'reaped': 0, 'controls': [], 'selector_count': 0, 'opened': []}
    a = attempt(runner, tmp_path, interrupt=lambda: state['cancel'])
    a.cancel_fd = 999
    for leaf in ('identity-ready.fifo', 'identity-grant.fifo', 'payload-finished.fifo'):
        (tmp_path / leaf).touch()
    # Use a module-local os proxy: mocks must not replace pytest's process os.
    real_os = runner.os
    proxy = SimpleNamespace(**vars(real_os))
    metadata = SimpleNamespace(st_mode=0o010600, st_uid=1000, st_nlink=1, st_dev=1, st_ino=2)
    actual_lstat = Path.lstat
    monkeypatch.setattr(Path, 'lstat', lambda p: metadata if p.name.endswith('.fifo') else actual_lstat(p))
    def open_fd(*args, **kw):
        fd = 11 + len(state['opened']); state['opened'].append(fd); return fd
    proxy.open = open_fd
    proxy.fstat = lambda fd: metadata
    proxy.readlink = lambda path: 'host:[' + Path(path).name + ']'
    proxy.read = lambda *args: (('foreign' if point == 'foreign-completion' and state['selector_count'] >= 2 else a.clock['nonce']) + '\n').encode()
    def written(fd, data):
        state['writes'].append((fd, data))
        if point == 'after-grant': state['cancel'] = 'cancel'
        return len(data)
    proxy.write = written
    proxy.close = lambda fd: state['closed'].append(fd)
    monkeypatch.setattr(runner, 'os', proxy)
    class Client:
        returncode = 0
        def poll(self): return None if state['reaped'] == 0 else 0
        def kill(self): state['killed'] += 1
        def communicate(self, **kwargs):
            state['reaped'] += 1
            if point == 'build': state['cancel'] = 'cancel'
            if point == 'client-error' and state['reaped'] == 1:
                raise RuntimeError('original client failed')
            return b'', b''
    def popen(*args, **kwargs):
        state['launches'] += 1
        assert state['launches'] == 1
        if point == 'before-ready': state['cancel'] = 'cancel'
        return Client()
    monkeypatch.setattr(runner.subprocess, 'Popen', popen)
    class Selector:
        def __init__(self): self.keys = []; state['selector_count'] += 1
        def register(self, fd, events, data): self.keys.append(SimpleNamespace(data=data))
        def select(self, seconds):
            assert 0 < seconds <= 1530
            if point == 'no-ready': return []
            key = next(k for k in self.keys if k.data == ('cancel' if state['cancel'] else ('ready' if state['selector_count'] == 1 else 'payload')))
            return [(key, 1)]
        def close(self): pass
    monkeypatch.setattr(runner.selectors, 'DefaultSelector', Selector)
    monkeypatch.setattr(a, 'capture_root_control', lambda op: state['controls'].append(op))
    def controls_closed():
        if point == 'root-still-live': raise RuntimeError('root_control_still_live')
        return True
    monkeypatch.setattr(a, 'root_controls_closed', controls_closed)
    def observation(*args, **kwargs):
        if point == 'observation':
            state['cancel'] = 'cancel'
            raise RuntimeError('cancel_during_observation')
        return dict(timeout=False, returncode=0, stderr='', stdout=json.dumps(obs))
    a.root_control_handles['observe'] = {'pid': 42}
    obs = dict(state='EXACT_ROOT_NAMESPACE_OBSERVATION', unit=a.unit, nonce=a.clock['nonce'],
               clock_sha256=a.clock_hash, manager=a.manager, operation_identity={'pid': 42},
               pid=1234, start_ticks=7, native_identity={'pid_start_time': 7, 'stdout': '{}'},
               namespaces={x: 'private:[' + x + ']' for x in ('net','user','ipc','mnt')},
               invocation_id='01' * 16, activation_monotonic=10)
    monkeypatch.setattr(a, 'client', observation)
    monkeypatch.setattr(runner, 'read_exact', lambda *a, **k: (json.dumps(obs).encode(), None))
    monkeypatch.setattr(runner, 'validate_native_identity', lambda *a, **k: '01' * 16)
    monkeypatch.setattr(runner, 'typed_reply', lambda *a, **k: [{}])
    monkeypatch.setattr(runner, 'validate_execution_properties', lambda *a, **k: None)
    monkeypatch.setattr(a, 'own_file', lambda: None)
    monkeypatch.setattr(runner, 'fresh_json', lambda *a, **k: None)
    answer = a.native_start([runner.SUDO, '-n', '--', str(runner.ADMIN_DIRECTORY/'start')], 1530)
    assert answer['timeout'] is (point != 'completion-notification')
    assert answer['client_reaped'] is True
    assert state['launches'] == 1 and state['reaped'] >= 1
    assert sorted(state['closed']) == sorted(state['opened']) and len(set(state['closed'])) == 3
    assert not a.closed and not a.cleaned
    # A caller's real execute()/settlement must still stop/observe/clean. A
    # caught client error or notification is never a cgroup-stop observation.

@pytest.mark.parametrize('populated,read_error', [(True, None), (False, PermissionError), (False, OSError)])
def test_zero_leader_never_clears_pending_or_unknown_cgroup(runner, tmp_path, monkeypatch, populated, read_error):
    a = attempt(runner, tmp_path)
    real_lstat, real_resolve, real_read = Path.lstat, Path.resolve, Path.read_text
    prefix = '/sys/fs/cgroup' + a.cgroup
    monkeypatch.setattr(Path, 'lstat', lambda p: object() if str(p) == prefix else real_lstat(p))
    monkeypatch.setattr(Path, 'resolve', lambda p, **k: p if str(p) == prefix else real_resolve(p, **k))
    def read(p, *args, **kw):
        if str(p).startswith(prefix):
            if read_error: raise read_error('unknown')
            return 'populated 1\n' if p.name == 'cgroup.events' else '1235\n'
        return real_read(p, *args, **kw)
    monkeypatch.setattr(Path, 'read_text', read)
    if read_error:
        with pytest.raises(read_error): a.group_closed()
    else:
        assert a.group_closed() is False
        with pytest.raises(RuntimeError, match='cgroup_not_stopped'):
            runner.cessation(stopped(a), a.unit, a.fragment, a.cgroup, a.group_closed())


@pytest.mark.parametrize('change,expected', [
    ('unresolved', 1), ('not_closed', 1), ('not_cleaned', 1),
    ('cancel', 2), ('deadline', 3), ('suffix', 3), ('failed_result', 1), ('none', 0)])
def test_final_completion_consumer_requires_all_original_settlement(runner, tmp_path, change, expected):
    state = {'now': 10.0, 'cancel': None}
    a = attempt(runner, tmp_path, interrupt=lambda: state['cancel'])
    a.now = lambda: state['now']
    a.publication_intent = True
    a.closed = a.cleaned = True
    result = {'state': 'PASS'}
    if change == 'unresolved': a.root_control_unknown = True
    if change == 'not_closed': a.closed = False
    if change == 'not_cleaned': a.cleaned = False
    if change == 'cancel': state['cancel'] = 'SIGTERM'
    if change == 'deadline': state['now'] = 1800.0
    if change == 'suffix': a.suffix_started = 0.0; state['now'] = 11.0
    if change == 'failed_result': result['state'] = 'FAILED_OR_REJECTED_CESSATION_CONFIRMED'
    assert a.completion_status(result) == expected
    assert a.clock['started_monotonic'] == 0 and a.clock['deadline_monotonic'] == 1800

@pytest.mark.parametrize('mutation', [None,'foreign-nonce','foreign-boot','shifted-deadline',
                                    'missing-handle','root-client-live','group-populated',
                                    'foreign-absence','late-absence','root-artifact-remains',
                                    'output-failed','client-not-reaped'])
def test_real_cleanup_receipt_consumer_retains_every_uncertainty(runner, tmp_path, monkeypatch, mutation):
    """Finite transport receipt model passed through actual cleanup().
    No mock receipt is a real ROOT/cgroup/absence/parent observation.
    """
    monkeypatch.setattr(runner,'UNIT_DIRECTORY',tmp_path/'systemd')
    monkeypatch.setattr(runner,'ADMIN_DIRECTORY',tmp_path/'stage')
    monkeypatch.setattr(runner,'INSTALL_JOURNAL',tmp_path/'journal')
    monkeypatch.setattr(runner,'SUDOERS',tmp_path/'policy')
    a=attempt(runner,tmp_path/'live');a.publication_intent=True;a.closed=True
    handle=dict(unit=a.unit,nonce=a.clock['nonce'],boot_id=a.clock['boot_id'],
                deadline_monotonic=1800,pid=1234,start_ticks=7)
    after=dict(Id=a.unit,LoadState='not-found',ActiveState='inactive',SubState='dead',MainPID='0',ControlPID='0',
               Job='',FragmentPath='',Transient='no',DropInPaths='',ControlGroup='',InvocationID='')
    paths=(a.fragment,runner.ADMIN_DIRECTORY,runner.SUDOERS,runner.INSTALL_JOURNAL)
    receipt=dict(state='EXACT_ROOT_ARTIFACTS_REMOVED',unit=a.unit,manager=a.manager,
                 clock_sha256=a.clock_hash,after=after,control_pid=1234,control_start_ticks=7,
                 operation_handles={'cleanup':handle},
                 absence_observations=[dict(path=str(p),errno=2,observed_monotonic=10) for p in paths])
    if mutation=='foreign-nonce':handle['nonce']='foreign'
    if mutation=='foreign-boot':handle['boot_id']='foreign'
    if mutation=='shifted-deadline':handle['deadline_monotonic']=1801
    if mutation=='missing-handle':receipt['operation_handles']={}
    if mutation=='foreign-absence':receipt['absence_observations'][0]['path']='/foreign'
    if mutation=='late-absence':receipt['absence_observations'][0]['observed_monotonic']=1800
    if mutation=='root-artifact-remains':runner.INSTALL_JOURNAL.write_bytes(b'remaining fixture')
    if mutation=='output-failed':receipt['state']='UNKNOWN'
    response=dict(returncode=0,timeout=False,client_reaped=mutation!='client-not-reaped',
                  stdout=json.dumps(receipt),stderr='')
    a.transport=lambda *args:dict(response)
    monkeypatch.setattr(a,'group_closed',lambda:mutation!='group-populated')
    def birth(*args):
        if mutation=='root-client-live':return 7
        raise FileNotFoundError('fixture root client absent')
    monkeypatch.setattr(runner,'start_time',birth)
    if mutation:
        with pytest.raises((RuntimeError,KeyError)):a.cleanup(stopped(a))
        assert a.unresolved and not a.cleaned
        assert a.completion_status({'state':'PASS'})==1
    else:
        assert a.cleanup(stopped(a))==after
        assert a.cleaned and not a.unresolved
        assert a.completion_status({'state':'PASS'})==0


@pytest.mark.parametrize('failure', ['deadline','pending-descendant','bad-output','foreign-result-fd','foreign-invocation'])
def test_execute_settles_or_preserves_custody_without_accepting_failed_output(runner,tmp_path,monkeypatch,failure):
    """Real execute() ordering, finite source-level transport/proof model.
    The first stop request precedes output acceptance. No execution proof here.
    """
    a=attempt(runner,tmp_path);events=[];a.publication_intent=True
    before=dict(Id=a.unit,LoadState='loaded',ActiveState='inactive',MainPID='0',ControlPID='0',
                Job='',InvocationID='',ControlGroup='')
    final=stopped(a)
    if failure=='foreign-invocation':final['InvocationID']='02'*16
    observation=dict(invocation_id='01'*16,result_identity=[1,2],native_identity={'fixture':'native'})
    a.root_observation=observation
    def show(props=runner.STATE_KEYS,**kwargs):
        if kwargs.get('system'):return manager()
        if props==runner.EFFECTIVE_KEYS:return {}
        if 'start' not in events:return before
        return final
    monkeypatch.setattr(a,'show',show)
    monkeypatch.setattr(a,'publish',lambda:events.append('publish'))
    monkeypatch.setattr(a,'own_file',lambda:None)
    monkeypatch.setattr(a,'environment_files',lambda:{})
    monkeypatch.setattr(runner,'effective',lambda *a:None)
    monkeypatch.setattr(runner,'fresh_json',lambda path,value:events.append(path.name))
    monkeypatch.setattr(a,'checked',lambda *a,**kw:events.append('stop'))
    monkeypatch.setattr(a,'group_closed',lambda:failure!='pending-descendant' or 'start' not in events)
    def client(*args,**kwargs):
        events.append('start')
        return dict(started_monotonic=10,completed_monotonic=10,timeout=False,returncode=0)
    monkeypatch.setattr(a,'client',client)
    def read_result(*args,**kwargs):
        assert 'stop' in events
        if failure=='bad-output':return b'not-json',SimpleNamespace(st_dev=1,st_ino=2)
        return json.dumps({'native_identity':observation['native_identity']}).encode(),SimpleNamespace(st_dev=9,st_ino=9)
    monkeypatch.setattr(runner,'read_exact',read_result)
    if failure=='deadline':a.now=lambda:1800
    with pytest.raises((RuntimeError,ValueError)):a.execute()
    assert not a.closed and not a.cleaned and a.unresolved
    assert a.completion_status({'state':'PASS'})==1
    assert events.count('start') <= 1
    if failure!='deadline':assert events.index('start')<events.index('stop')


@pytest.mark.parametrize('failure',['late-output','failed-output','failed-persist','cleanup-unknown'])
def test_actual_foreground_main_final_output_cannot_publish_accepted_PASS(runner,tmp_path,monkeypatch,failure):
    """Actual main()/finally, modeled native suffix; no real locks/signals/FDS.
    Unknown branch is observed with a finite trace, never a wait or poll.
    """
    state={'now':10.0,'attempt':None,'closed_fds':[],'outputs':[],'flocks':[], 'stages':[]}
    live=tmp_path/'live';clock_path=tmp_path/'clock.json'
    value=clock(runner);clock_path.write_text(json.dumps(value))
    monkeypatch.setattr(runner,'IDENTITY',dict(runner.IDENTITY,live_directory=str(live)))
    original_attempt=runner.Attempt
    def create(root,path,c,**kwargs):
        a=original_attempt(root,path,c,now=lambda:state['now'],**kwargs)
        state['stages'].append('constructed')
        a.adopt_installed=lambda:state['stages'].append('adopted')
        def execute():
            state['stages'].append('executed')
            a.publication_intent=a.submitted=True;a.closed=True
            a.cleaned=failure!='cleanup-unknown'
            if failure=='cleanup-unknown':raise RuntimeError('cleanup transport unknown')
            return dict(state='PASS',proof={'FIXTURE_ONLY':True},cleaned=True)
        a.execute=execute;state['attempt']=a;return a
    monkeypatch.setattr(runner,'Attempt',create)
    monkeypatch.setattr(runner,'admit',lambda path:({'clock_sha256':hashlib.sha256(clock_path.read_bytes()).hexdigest(),
                                                 'clock_path':str(clock_path)},value,live))
    proxy=SimpleNamespace(**vars(runner.os))
    proxy.getuid=lambda:1000;proxy.isatty=lambda *a:True
    proxy.umask=lambda *a:None;proxy.sched_setaffinity=lambda *a:None
    proxy.open=lambda *a,**kw:19;proxy.pipe2=lambda *a:(20,21)
    proxy.fstat=lambda fd:SimpleNamespace(st_dev=2050,st_ino=9044963,st_uid=1000,st_gid=1000,
                                        st_nlink=1,st_size=0,st_mode=0o100664)
    proxy.close=lambda fd:state['closed_fds'].append(fd)
    proxy.mkfifo=lambda *a:None
    class Sink(io.BytesIO):
        def fileno(self):return 22
    proxy.fdopen=lambda *a,**kw:Sink();proxy.fsync=lambda *a:None
    monkeypatch.setattr(runner,'os',proxy)
    monkeypatch.setattr(runner.fcntl,'flock',lambda *args:state['flocks'].append(args))
    monkeypatch.setattr(runner.resource,'setrlimit',lambda *a:None)
    monkeypatch.setattr(runner.signal,'signal',lambda *a:None)
    # argparse.parse_args reads this interpreter's sys.argv; runner has no sys export.
    monkeypatch.setattr(sys,'argv',['run','--admission',str(tmp_path/'admission-fixture')])
    def persist(path,data):
        state['outputs'].append((path.name,dict(data)))
        if path.name=='result.json' and failure=='failed-persist':raise RuntimeError('pending persist failed')
    monkeypatch.setattr(runner,'fresh_json',persist)
    def print_data(text,**kwargs):
        if failure=='failed-output':raise BrokenPipeError('pending output failed')
        if failure=='late-output':state['now']=1800.0
    monkeypatch.setattr(runner,'print',print_data,raising=False)
    if failure=='cleanup-unknown':
        class RetainedForTest(BaseException):pass
        node=next(n for n in ast.parse((ROOT/'run-live-unit.py').read_bytes()).body
                  if isinstance(n,ast.FunctionDef) and n.name=='main')
        loop=next(n for n in ast.walk(node) if isinstance(n,ast.While))
        previous=sys.gettrace()
        def trace(frame,event,arg):
            if frame.f_code is runner.main.__code__ and event=='line' and frame.f_lineno==loop.lineno:
                assert state['attempt'].unresolved
                raise RetainedForTest()
            return trace
        try:
            sys.settrace(trace)
            with pytest.raises(RetainedForTest):runner.main()
        finally:sys.settrace(previous)
        assert not state['closed_fds']
        assert any(d['state']=='STOP_UNCONFIRMED' for _,d in state['outputs'])
    elif failure=='late-output':
        assert runner.main()==3
        assert sorted(state['closed_fds'])==[19,20,21]
    else:
        error,reason=(BrokenPipeError,'pending output failed') if failure=='failed-output' else (RuntimeError,'pending persist failed')
        with pytest.raises(error,match='^'+reason+'$'):runner.main()
        assert sorted(state['closed_fds'])==[19,20,21]
    assert state['stages']==['constructed','adopted','executed']
    assert state['outputs']
    if failure=='cleanup-unknown':
        assert any(d.get('error')=='cleanup transport unknown' for _,d in state['outputs'])
    else:
        assert state['outputs'][0][0]=='result.json'
        assert state['outputs'][0][1]['state']=='CESSATION_CONFIRMED_PENDING_FINAL_EXIT'
    assert state['flocks']==[(19,runner.fcntl.LOCK_EX|runner.fcntl.LOCK_NB)]
    assert all(d.get('final_outcome_accepted') is False for _,d in state['outputs'])
    assert state['attempt'].clock['deadline_monotonic']==1800
