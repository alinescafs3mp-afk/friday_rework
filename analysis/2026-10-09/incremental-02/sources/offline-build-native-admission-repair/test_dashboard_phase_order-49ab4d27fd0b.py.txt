Full main ordering is exercised with finite tool/IPC doubles; NOT a real build.
No candidate imports, collection or tests were performed in source preparation.
"""
import hashlib
import io
import json
from pathlib import Path
from types import SimpleNamespace
import sys
import pytest
ROOT = Path(__file__).parent
sys.path.insert(0, str(ROOT))
from gate_support import load, clock


@pytest.fixture
def build():
    return load('build')


def font_input():
    # Explicit final input graph, not a synthetic wOF2 prefix or ambient font.
    inputs = json.loads((ROOT / 'inputs.json').read_bytes())
    name, pin = next((name, pin) for name, pin in inputs['files'].items()
                     if '/font-inputs/' in name and name.endswith('.woff2'))
    path = Path(name)
    data = path.read_bytes()
    assert not path.is_symlink() and path.stat().st_nlink == 1
    assert len(data) == pin['bytes'] and hashlib.sha256(data).hexdigest() == pin['sha256']
    assert data[:4] == b'wOF2' and len(data) > 1024
    return name, pin, data


def actual_dashboard(live):
    name, pin, data = font_input()
    root = live / 'source/packages/dashboard/out'
    (root / '_next/static').mkdir(parents=True)
    (root / 'index.html').write_text('<html><head><link href="/_next/static/app.css" rel="stylesheet"></head><body><script src="/_next/static/app.js"></script></body></html>')
    (root / '_next/static/app.js').write_text('document.documentElement.dataset.offline="fixture";')
    (root / '_next/static/app.css').write_text('@font-face{font-family:Geist;src:url("./geist.woff2")}body{font-family:Geist}')
    (root / '_next/static/geist.woff2').write_bytes(data)
    return root, {'files': {name: pin}}


def test_positive_actual_HTML_JS_CSS_and_exact_real_font(build, tmp_path, monkeypatch):
    monkeypatch.setattr(build, 'LIVE', tmp_path)
    root, inputs = actual_dashboard(tmp_path)
    result = build.dashboard_manifest(inputs)
    assert set(result) == {'index.html','_next/static/app.js','_next/static/app.css','_next/static/geist.woff2'}
    assert result['_next/static/geist.woff2'] == next(iter(inputs['files'].values()))['sha256']
    observed = json.loads((tmp_path / 'dashboard-artifacts.json').read_bytes())
    assert observed['sha256'] == result and observed['actual_build_required'] is True


@pytest.mark.parametrize('mutation', [
    'missing-font', 'replaced-font', 'untrusted-font-pin', 'missing-CSS-file',
    'missing-CSS-reference', 'external-CSS-reference', 'escaping-CSS-reference',
    'missing-HTML-reference', 'placeholder-with-assets', 'missing-JS', 'font-symlink'])
def test_positive_inventory_corruption_refused_before_artifact_report(build, tmp_path, monkeypatch, mutation):
    monkeypatch.setattr(build, 'LIVE', tmp_path)
    root, inputs = actual_dashboard(tmp_path)
    font = root / '_next/static/geist.woff2'
    css = root / '_next/static/app.css'
    if mutation == 'missing-font': font.unlink()
    if mutation == 'replaced-font': font.write_bytes(b'wOF2' + b'changed'*1000)
    if mutation == 'untrusted-font-pin': next(iter(inputs['files'].values()))['sha256'] = '0'*64
    if mutation == 'missing-CSS-file': css.unlink()
    if mutation == 'missing-CSS-reference': css.write_text('@font-face{src:url(missing.woff2)}')
    if mutation == 'external-CSS-reference': css.write_text('@font-face{src:url(https://invalid.example/font.woff2)}')
    if mutation == 'escaping-CSS-reference': css.write_text('@font-face{src:url(../../../../outside.woff2)}')
    if mutation == 'missing-HTML-reference': (root/'index.html').write_text('<script src="/missing.js"></script>')
    if mutation == 'placeholder-with-assets': (root/'index.html').write_text('Dashboard not built')
    if mutation == 'missing-JS': (root/'_next/static/app.js').unlink()
    if mutation == 'font-symlink':
        data = font.read_bytes(); font.unlink(); other = tmp_path/'outside.woff2'
        other.write_bytes(data); font.symlink_to(other)
    with pytest.raises(RuntimeError): build.dashboard_manifest(inputs)
    assert not (tmp_path/'dashboard-artifacts.json').exists()


@pytest.mark.parametrize('failure', [None, 'install', 'dashboard', 'inventory', 'cargo', 'lock'])
def test_real_main_dispatch_order_and_no_Cargo_after_dashboard_refusal(build, tmp_path, monkeypatch, failure):
    """Invoke final main, preserving its actual dashboard and lock consumers.
    Simulated phases write real private artifacts. No tool/version/native proof.
    """
    live = tmp_path/'live'; live.mkdir()
    stage = tmp_path/'stage'; stage.mkdir()
    inputs = json.loads((ROOT/'inputs.json').read_bytes())
    # Same selected final graph is passed to main. prepare verification belongs
    # to T2, not this mocked private phase-order control.
    (stage/'inputs.json').write_text(json.dumps(inputs))
    (live/'clock.json').write_text(json.dumps(clock(build)))
    (live/'host-current-invocation.json').write_text(json.dumps({'native_identity':{'fixture':True}}))
    events=[]; writes=[]; cargo_reads=[]
    nonce = build.NONCE
    monkeypatch.setattr(build, 'LIVE', live); monkeypatch.setattr(build,'STAGE',stage)
    monkeypatch.setattr(build, '__file__', str(stage/'build.py'))
    monkeypatch.setattr(build.sys, 'argv', [str(stage/'build.py')])
    monkeypatch.setattr(build.time, 'monotonic', lambda:10.0)
    monkeypatch.setattr(build,'clock_valid',lambda value:value)
    real_os=build.os
    os_proxy=SimpleNamespace(**vars(real_os))
    real_fds=set()
    def open_fd(path,flags,*a,**kw):
        if path==live/'runtime.stdout':
            assert flags==real_os.O_WRONLY|real_os.O_CREAT|real_os.O_EXCL|real_os.O_NOFOLLOW
            return 3
        fd=real_os.open(path,flags,*a,**kw);real_fds.add(fd);return fd
    def close_fd(fd):
        if fd in real_fds:
            real_fds.remove(fd);return real_os.close(fd)
        assert fd==3, 'only the modeled runtime output may be closed without a real descriptor'
    def fdopen(fd,*a,**kw):
        assert fd in real_fds
        stream=real_os.fdopen(fd,*a,**kw)
        real_fds.remove(fd)  # stream now owns and closes the real descriptor
        return stream
    os_proxy.umask=lambda *a:None;os_proxy.open=open_fd;os_proxy.fdopen=fdopen
    os_proxy.set_inheritable=lambda *a:None;os_proxy.close=close_fd
    os_proxy.fsync=lambda *a:None; os_proxy.write=lambda fd,value:writes.append(value) or len(value)
    monkeypatch.setattr(build,'os',os_proxy)
    actual_read_pin=build.read_pin
    def observed_read_pin(path,pin):
        content=actual_read_pin(path,pin)
        cargo_reads.append((str(path),dict(pin),hashlib.sha256(content).hexdigest()))
        return content
    monkeypatch.setattr(build,'read_pin',observed_read_pin)
    # main owns a model FD3 in this fixture. Never let save() fdopen a real
    # process descriptor through that double; use ordinary private file output.
    monkeypatch.setattr(build,'save',lambda name,value:(live/name).write_text(json.dumps(value)))
    monkeypatch.setattr(build.signal,'signal',lambda *a:None)
    monkeypatch.setattr(build.signal,'setitimer',lambda *a:None)
    real_open=Path.open
    def opened(path,*a,**kw):
        if path.name == 'identity-ready.fifo': return io.BytesIO()
        if path.name == 'identity-grant.fifo': return io.BytesIO((nonce+'\n').encode())
        return real_open(path,*a,**kw)
    monkeypatch.setattr(Path,'open',opened)
    monkeypatch.setattr(build.select,'select',lambda channel,*a: (channel,[],[]))
    monkeypatch.setattr(build,'boundary_controls',lambda grant:events.append('boundary') or {'FIXTURE_ONLY':True})
    def prepare(*a):
        events.append('prepare')
        (live/'source/cli').mkdir(parents=True)
        (live/'source/cli/Cargo.lock').write_text('fixture exact cargo lock')
        (live/'source/pnpm-lock.yaml').write_text('fixture exact pnpm lock')
    monkeypatch.setattr(build,'prepare',prepare)
    versions={'node':'v22.23.2','pnpm':'9.15.9','rustc':'rustc 1.94.0','cargo':'cargo 1.94.0'}
    def phase(name,argv,cap,value,env,cwd,ceiling=None):
        events.append(name)
        if name == failure: raise RuntimeError('phase failed:'+name)
        if name.startswith('versions-'):
            (live/(name+'.log')).write_text(versions[name.removeprefix('versions-')])
        if name == 'dashboard':
            root,_ = actual_dashboard(live)
            if failure == 'inventory': (root/'_next/static/geist.woff2').unlink()
        if name == 'cargo':
            assert 'dashboard-artifacts.json' in [p.name for p in live.iterdir()]
            assert '--offline' in argv and '--locked' in argv and '-j' in argv
            binary=live/'target/debug/agent-browser';binary.parent.mkdir(parents=True)
            binary.write_bytes(b'\x7fELFfixture only, not a compiler output')
            generated=live/'target/debug/build/agent-browser-fixture/out/cdp_generated.rs'
            generated.parent.mkdir(parents=True); generated.write_text('// private fixture')
            if failure == 'lock': (live/'source/cli/Cargo.lock').write_text('changed')
    monkeypatch.setattr(build,'phase',phase)
    if failure:
        reason={'install':'phase failed:install','dashboard':'phase failed:dashboard',
                'inventory':'actual_dashboard_assets_fonts_missing','cargo':'phase failed:cargo',
                'lock':'source_lock_changed'}[failure]
        with pytest.raises(RuntimeError,match='^'+reason+'$'): build.main()
        assert not writes
        if failure in ('install','dashboard','inventory'): assert 'cargo' not in events
    else:
        build.main()
        assert events == ['boundary','prepare','versions-node','versions-pnpm','versions-rustc','versions-cargo','install','dashboard','cargo']
        proof=json.loads(writes[0]);assert proof['installed_journey_accepted'] is False
        assert proof['source_task_accepted'] is False
    prefix=['boundary','prepare','versions-node','versions-pnpm','versions-rustc','versions-cargo','install']
    expected_events=prefix+([] if failure=='install' else ['dashboard'])
    if failure not in ('install','dashboard','inventory'):expected_events+=['cargo']
    assert events==expected_events
    config=str(Path(inputs['build_root'])/'cargo-offline.toml')
    expected_reads=[] if failure in ('install','dashboard','inventory') else [(config,inputs['files'][config],inputs['files'][config]['sha256'])]
    assert cargo_reads==expected_reads and not real_fds
    # A fake ELF and phase double deliberately cannot count as real build PASS.
