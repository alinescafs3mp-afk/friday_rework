+def test_actual_admit_uses_exact_pinned_identity_without_source_parent_inference(modules, monkeypatch, mutation, reason):
+    """Real admit/private/digest; only inert host/runtime/dependency observations.
+
+    Exact source bytes/pins and IDENTITY are read from the actual published
+    candidate. No runtime files, clocks, native commands, or grants are created.
+    External executables are modeled by their declared pins, not rehashed per
+    parameter; the new gate still has its separate real input/origin checks.
+    """
+    import hashlib
+    import stat
+    from types import SimpleNamespace
+    runner = modules[0]
+    real_path = Path
+    source = real_path(runner.IDENTITY['root_source'])
+    live = real_path(runner.IDENTITY['live_directory'])
+    admission_path = real_path(runner.IDENTITY['runtime_admission_path'])
+    clock_path = real_path(runner.IDENTITY['trial_clock_path'])
+    review_path = admission_path.parent / 'modeled-independent-review.json'
+    assert live.parent != source.parent.parent  # actual rebind regression
+    pins = json.loads((source / 'package-pins.json').read_text())
+    source_hashes = {name: pin['sha256'] for name, pin in pins.items()}
+    source_hashes['package-pins.json'] = hashlib.sha256((source / 'package-pins.json').read_bytes()).hexdigest()
+    dependencies = json.loads((source / 'source-pins.json').read_text())
+    clock = dict(assignment='FRW041-NATIVE-OFFLINE-BUILD-RUN', native_effects_authorized_by='Astra',
+                 unit=runner.IDENTITY['unit'], nonce=runner.IDENTITY['nonce'], budget_seconds=1800,
+                 started_monotonic=1000.0, deadline_monotonic=2800.0, boot_id='modeled-boot')
+    review = dict(state='INDEPENDENT_OFFLINE_BUILD_SYSTEM_SERVICE_SOURCE_ACCEPTED',
+                  source_task=runner.SOURCE_TASK, runtime_grant=False,
+                  package_pins_sha256=source_hashes['package-pins.json'])
+    if mutation == 'review-package': review['package_pins_sha256'] = '0' * 64
+    if mutation == 'clock-old120': clock.update(budget_seconds=120, deadline_monotonic=1120.0)
+    if mutation == 'clock-expired': clock.update(started_monotonic=-800.0, deadline_monotonic=1000.0)
+    if mutation == 'clock-deadline-reset': clock['deadline_monotonic'] += 1
+    if mutation == 'clock-boot': clock['boot_id'] = 'foreign-boot'
+    if mutation == 'clock-nonce': clock['nonce'] = '0000000000000000'

+    if mutation == 'admission-path': admission_path = admission_path.with_name('foreign-admission.json')

+                     clock_path=str(clock_path), clock_sha256=hashlib.sha256(payloads[str(clock_path)]).hexdigest(),
+                     live_directory=str(live))

+    if mutation == 'wrong-live-name': admission['live_directory'] = str(live.with_name(live.name + '-foreign'))
+    if mutation == 'source-hash': admission['source_sha256']['build.py'] = '0' * 64
+    if mutation == 'source-inventory': admission['source_sha256'].pop('build.py')
+    if mutation == 'review-hash': admission['independent_review_sha256'] = '0' * 64
+    if mutation == 'clock-hash': admission['clock_sha256'] = '0' * 64

+    class ModeledPath(type(real_path())):
+        def read_bytes(self):
+            if str(self) in payloads: return payloads[str(self)]
+            return super().read_bytes()
+        def read_text(self, *args, **kwargs):
+            if str(self) in payloads: return payloads[str(self)].decode()
+            if str(self) == '/proc/1/cgroup': return '0::/\n'
+            if str(self) == '/proc/sys/kernel/random/boot_id': return 'modeled-boot\n'
+            return super().read_text(*args, **kwargs)
+        def resolve(self, strict=False):
+            if str(self) in payloads: return self

+                return ModeledPath(dependencies['python_alias']['resolved'])
