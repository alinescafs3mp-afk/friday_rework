def require(ok, message):
    if not ok:
        raise ValueError(message)


def sha(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as f:
        for chunk in iter(lambda: f.read(1 << 20), b''):
            h.update(chunk)
    return h.hexdigest()


def read(path):
    return json.loads(Path(path).read_bytes())


def put(path, value, mode=0o400):
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, mode)
    with os.fdopen(fd, 'w') as f:
        json.dump(value, f, indent=2); f.write('\n'); f.flush(); os.fsync(f.fileno())
    fd = os.open(Path(path).parent, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
    try:
        os.fsync(fd)
    finally:
        os.close(fd)


def private(path, mode):
    s = Path(path).lstat()
    require(not Path(path).is_symlink() and s.st_uid == os.getuid() and
            stat.S_IMODE(s.st_mode) == mode, 'FOREIGN_OR_MUTABLE_PATH:' + str(path))
    return s


def finite(value):
    return type(value) in (int, float) and math.isfinite(value) and value > 0

                 runtest_file=u['unit_id']) for u in plan['units']]

                clock['deadline_monotonic'] == clock['start_monotonic'] + clock['original_budget_seconds'],
                'EXPIRED_OR_RESET_ORIGINAL_RUNTIME_CLOCK')
        require(clock.get('boot_id') == Path('/proc/sys/kernel/random/boot_id').read_text().strip() and
                isinstance(clock.get('assignment'), str) and bool(clock['assignment']) and
                type(clock.get('generation')) is int and clock['generation'] >= 1, 'RUNTIME_OWNER_BOOT_IDENTITY')
        require(clock.get('execution_authorized') is True and clock.get('independent_source_review') == 'PASS' and
                clock.get('independent_runtime_review') == 'PASS' and

                clock.get('runtime_plan_sha256') == sha(SOURCE / 'runtest-plan.json') and
                clock.get('resources') == plan['resources'], 'NO_SEPARATE_RUNTIME_COHORT_ADMISSION')
        require(plan.get('runtime_custody_integrated') is True and not plan.get('unresolved_effect_boundaries'),
                'RUNTIME_SOURCE_NOT_ADMISSIBLE')

    require(all(finite(clock.get(k)) for k in CORE[3:]), 'INVALID_ORIGINAL_CLOCK')
    require(clock['start_monotonic'] <= time.monotonic() < clock['deadline_monotonic'] and

            clock.get('reviewed_source_pins_sha256') == SPEC['source_pins_sha256'], 'WRONG_COHORT_SCOPE')
    require(clock.get('resources') == {'cpu': 2, 'memory_bytes': 2 << 30, 'children': 0, 'max_phases': 4},
            'WRONG_RESOURCES_OR_PHASE_COUNT')


def lock_probe():
    fd = os.open(SPEC['canonical_lock']['path'], os.O_RDONLY | os.O_NOFOLLOW)
    try:
        s = os.fstat(fd)
        require([s.st_dev, s.st_ino, s.st_uid, s.st_gid, s.st_nlink, s.st_size,
                 stat.S_IMODE(s.st_mode)] == SPEC['canonical_lock']['identity'], 'LOCK_IDENTITY_CHANGED')
        fcntl.flock(fd, fcntl.LOCK_EX | fcntl.LOCK_NB)
        fcntl.flock(fd, fcntl.LOCK_UN)
    finally:
        os.close(fd)


def file_inventory(root):
    """Frozen corpora contain regular files and directories; no symlinks at intake."""
    found = set()
    for base, dirs, files in os.walk(root, followlinks=False):
        for name in dirs + files:
            p = Path(base) / name; s = p.lstat()
            require(not stat.S_ISLNK(s.st_mode), 'UNREVIEWED_CORPUS_SYMLINK:' + str(p))
            if stat.S_ISREG(s.st_mode):
                found.add(str(p.relative_to(root)))
            else:
                require(stat.S_ISDIR(s.st_mode), 'UNREVIEWED_CORPUS_TYPE:' + str(p))
    return found


def lexical_input_declarations(expanded, path_api, manifest):
    """Four observed leaf spellings, tied to the existing source-derived alias.
    Raw evidence is never rewritten and aliases never enter the canonical map.
    """
    modules = ('hermes_time', 'hermes_state_ids', 'hermes_state_pidns', 'hermes_bootstrap')
    native = SPEC['roots']['native']
    alias = path_api['native_alias_declaration'](manifest, native)
    require(sha(SOURCE / 'import-path-pins.json') ==
            SPEC['native_lexical_input_basis']['import_path_pins_sha256'], 'LEXICAL_IMPORT_MANIFEST_DRIFT')
    for rel, digest in alias['source_files'].items():
        source = native + '/' + rel
        require(expanded.get(source) == digest and sha(source) == digest, 'LEXICAL_SOURCE_PROVENANCE_DRIFT')
    rows = SPEC['native_lexical_inputs']
    require(set(rows) == set(modules), 'LEXICAL_LEAF_SET_DRIFT')
    for module, row in rows.items():
        require(set(row) == {'raw', 'canonical', 'sha256', 'identity'} and
                row['raw'] == alias['raw'] + '/' + module + '.py' and
                row['canonical'] == native + '/' + module + '.py' and
                row['raw'] not in expanded and expanded.get(row['canonical']) == row['sha256'] and
                type(row['sha256']) is str and len(row['sha256']) == 64 and
                len(row['identity']) == 5 and all(type(v) is int for v in row['identity']) and
                stat.S_ISREG(row['identity'][2]), 'LEXICAL_LEAF_DECLARATION_DRIFT')
    return rows


def verify_lexical_input(row, path_api, manifest):
    """Extend the pinned alias walk to one finite no-follow regular leaf.
    Hold all lexical directory descriptors through leaf hashing and binding
    rechecks. This is post-reap readiness, not hostile concurrent containment.
    """
    alias = path_api['native_alias_declaration'](manifest, SPEC['roots']['native'])
    leaf = row['canonical'].rsplit('/', 1)[-1]
    require(row['raw'] == alias['raw'] + '/' + leaf and
            row['canonical'] == alias['target'] + '/' + leaf and
            leaf in ('hermes_time.py', 'hermes_state_ids.py', 'hermes_state_pidns.py', 'hermes_bootstrap.py'),
            'LEXICAL_EXACT_LEAF_REQUIRED')
    identity = path_api['identity']; handles = []
    try:
        # Same literal steps and identity function as import_path_readiness;
        # retain them here because its verify_native_alias closes its handles.
        for index, step in enumerate(alias['steps']):
            name = '/' if index == 0 else step['path'].rsplit('/', 1)[-1]
            fd = os.open(name, os.O_PATH | os.O_DIRECTORY | os.O_NOFOLLOW | os.O_CLOEXEC,
                         **({} if index == 0 else {'dir_fd': handles[-1]}))
            handles.append(fd)
            require(identity(os.fstat(fd)) == step['identity'], 'LEXICAL_COMPONENT_IDENTITY_DRIFT')
        directories = list(handles)
        leaf_fd = os.open(leaf, os.O_PATH | os.O_NOFOLLOW | os.O_CLOEXEC, dir_fd=directories[-1])
        handles.append(leaf_fd)
        require(identity(os.fstat(leaf_fd)) == row['identity'] and stat.S_ISREG(os.fstat(leaf_fd).st_mode),
                'LEXICAL_LEAF_IDENTITY_DRIFT')
        # Reopen only this checked regular descriptor, never a possibly replaced
        # pathname or special device. The intentional proc fd link is not an
        # input alias; its returned identity must equal the retained leaf.
        fd = os.open('/proc/self/fd/' + str(leaf_fd), os.O_RDONLY | os.O_NONBLOCK | os.O_CLOEXEC)
        handles.append(fd)
        require(identity(os.fstat(fd)) == row['identity'], 'LEXICAL_READ_DESCRIPTOR_DRIFT')
        h = hashlib.sha256()
        for chunk in iter(lambda: os.read(fd, 1 << 20), b''):
            h.update(chunk)
        require(h.hexdigest() == row['sha256'], 'LEXICAL_LEAF_HASH_DRIFT')
        for index, step in enumerate(alias['steps']):
            name = '/' if index == 0 else step['path'].rsplit('/', 1)[-1]
            current = os.stat(name, **({} if index == 0 else {'dir_fd': directories[index - 1]}),
                              follow_symlinks=False)
            require(identity(current) == step['identity'] and
                    identity(os.fstat(directories[index])) == step['identity'], 'LEXICAL_COMPONENT_BINDING_DRIFT')
        # The final '..' descriptor and the forward canonical native descriptor
        # must still bind the same pinned regular leaf, including after hashing.
        require(all(identity(os.stat(leaf, dir_fd=parent, follow_symlinks=False)) == row['identity']
                    for parent in (directories[-1], directories[-5])) and
                identity(os.fstat(fd)) == row['identity'], 'LEXICAL_LEAF_BINDING_DRIFT')
    finally:
        for fd in reversed(handles):
            os.close(fd)


def observed_input_joins(result, selection, expanded):
    """Preserve exact canonical joins; admit only evidenced native leaf pairs."""
    extra = set(result['inputs']) - set(expanded)
    foreign_origins = {p for p in result.get('origins', {}).values() if p not in expanded}
    joined = {}
    if extra or foreign_origins:
        require(selection == '@native', 'LEXICAL_INPUT_OUTSIDE_NATIVE')
        path_api = runpy.run_path(str(SOURCE / 'import_path_readiness.py'))
        manifest = path_api['load'](SOURCE)
        rows = lexical_input_declarations(expanded, path_api, manifest)
        by_raw = {row['raw']: (module, row) for module, row in rows.items()}
        require(extra == foreign_origins and extra <= set(by_raw), 'UNDECLARED_LEXICAL_INPUT_OR_ORIGIN')
        for raw in sorted(extra):
            module, row = by_raw[raw]
            require(result['inputs'][raw] == row['sha256'] and
                    result['origins'].get(module) == raw and
                    [name for name, p in result['origins'].items() if p == raw] == [module],
                    'LEXICAL_INPUT_HASH_OR_MODULE_PROVENANCE_DRIFT')
            verify_lexical_input(row, path_api, manifest)
            joined[raw] = {'module': module, 'canonical': row['canonical'], 'sha256': row['sha256']}
    for p, h in result['inputs'].items():
        require(p in joined or expanded.get(p) == h and sha(p) == h, 'UNRESOLVED_EXTRA_INPUT:' + p)
    for p in result.get('origins', {}).values():
        require(p in joined or p in expanded and sha(p) == expanded[p], 'UNCLASSIFIED_OBSERVED_ORIGIN:' + p)
    return joined

def cohort(out):
    private(out, 0o700); c = read(out / 'original-clock.json'); clock_check(c)
    private(out / 'original-clock.json', 0o400)
    private(out / 'input-pins.json', 0o400); private(out / 'preflight.json', 0o400)
    pre = read(out / 'preflight.json')
    require(sha(out / 'original-clock.json') == pre['clock_sha256'], 'ORIGINAL_CLOCK_BYTES_CHANGED')
    require(sha(pre['admission']) == pre['admission_sha256'], 'OWNER_ADMISSION_CHANGED')
    source_review(pre['source_review'], pre['source_review_sha256'])
    path_api = runpy.run_path(str(SOURCE / 'import_path_readiness.py'))
    path_api['verify_owner_startup_receipt'](SOURCE, pre['startup_observation']['path'], pre['startup_observation']['sha256'],
                                             runtime=c.get('scope') == 'FINITE_SERIAL_DONOR_FILE_RUNTESTS')
    require(inputs() == read(out / 'input-pins.json'), 'FULL_INPUT_OR_PATH_DRIFT')
    return c, pre


def source_review(path, digest):
    # Final review is supplied externally AFTER publication; baseline review cannot
    # accept these changed bytes. No placeholder hash or self-issued grant.
    require(isinstance(path, str) and Path(path).is_absolute() and
            Path(path).resolve(strict=True) == Path(path), 'EXACT_EXTERNAL_SOURCE_REVIEW_REQUIRED')
    private(path, 0o400)
    require(isinstance(digest, str) and len(digest) == 64 and sha(path) == digest, 'SOURCE_REVIEW_HASH_MISMATCH')
    review = read(path)
    require(review.get('state') in ('INDEPENDENT_COLLECTION_RECEIPT_AND_OWNER_HELPER_SOURCE_ACCEPTED',
                                    'INDEPENDENT_RUNTEST_PRODUCER_DONOR_REGISTRY_OWNER_SOURCE_ACCEPTED') and
            review.get('source') == str(SOURCE) and review.get('helper') == str(HERE) and
            review.get('source_pins_sha256') == SPEC['source_pins_sha256'] and
            review.get('helper_pins_sha256') == sha(HERE / 'source-pins.json') and
            review.get('helper_sha256') == sha(HERE / 'owner_observer.py') and
            review.get('input_spec_sha256') == sha(HERE / 'input-spec.json') and
            review.get('receipt_schema') == (3 if review.get('state') == 'INDEPENDENT_RUNTEST_PRODUCER_DONOR_REGISTRY_OWNER_SOURCE_ACCEPTED' else 2) and review.get('runtime_grant') is False and
            review.get('import_path_pins_sha256') == sha(SOURCE / 'import-path-pins.json'), 'CHANGED_PAIR_NOT_INDEPENDENTLY_REVIEWED')


def initialize(out, original, original_sha, admission, admission_sha):
    """Consume an existing separately granted clock; never creates an execution budget."""
    require(not out.exists(), 'COHORT_ALREADY_EXISTS_NO_RETRY')
    private(original, 0o400); private(admission, 0o400)
    require(sha(original) == original_sha and sha(admission) == admission_sha, 'OWNER_INPUT_PIN_MISMATCH')
    require(out.resolve(strict=False) == out, 'CANONICAL_FRESH_OWNER_ROOT_REQUIRED')
    c = read(original); clock_check(c); grant = read(admission)
    phases = phases_for(c); runtime = c.get('scope') == 'FINITE_SERIAL_DONOR_FILE_RUNTESTS'
    require(grant.get('state') == ('INDEPENDENT_OWNER_DONOR_FILE_RUNTEST_ADMITTED' if runtime else 'INDEPENDENT_OWNER_HELPER_AND_COLLECTION_ADMITTED') and
            grant.get('original_clock_sha256') == original_sha and grant.get('output_root') == str(out) and
            grant.get('helper_sha256') == sha(HERE / 'owner_observer.py') and
            grant.get('input_spec_sha256') == sha(HERE / 'input-spec.json') and
            grant.get('helper_pins_sha256') == sha(HERE / 'source-pins.json') and
            grant.get('source_pins_sha256') == SPEC['source_pins_sha256'] and
            grant.get('phases') == phases and grant.get('execution_authorized') is True,
            'INDEPENDENT_ROOT_ADMISSION_REQUIRED')
    source_review(grant.get('source_review'), grant.get('source_review_sha256'))
    if runtime:
        review = read(grant['source_review'])
        require(review.get('state') == 'INDEPENDENT_RUNTEST_PRODUCER_DONOR_REGISTRY_OWNER_SOURCE_ACCEPTED' and
                review.get('runtime_plan_sha256') == sha(SOURCE / 'runtest-plan.json') and
                review.get('runtime_overlay') == SPEC['runtime_overlay'], 'RUNTEST_CHANGED_PAIR_REVIEW_REQUIRED')
    path_api = runpy.run_path(str(SOURCE / 'import_path_readiness.py'))
    startup_observation = path_api['verify_owner_startup_receipt'](SOURCE, grant.get('startup_observation'), grant.get('startup_observation_sha256'), runtime=runtime)
    reserve = grant.get('reconciliation_reserve_seconds')
    require(finite(reserve), 'REVIEWED_RECONCILIATION_RESERVE_REQUIRED')
    out.mkdir(mode=0o700)
    # Exclusive original byte copy, not a reset clock or helper-issued permission.
    fd = os.open(out / 'original-clock.json', os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o400)
    with os.fdopen(fd, 'wb') as f:
        f.write(original.read_bytes()); f.flush(); os.fsync(f.fileno())
    expanded = inputs(); lock_probe(); clock_check(c)
    require(c['deadline_monotonic'] - time.monotonic() > (phases[0]['outer'] + reserve if runtime else sum(p['outer'] + reserve for p in phases)), 'INSUFFICIENT_WHOLE_COHORT_BUDGET')
    put(out / 'input-pins.json', expanded)
    put(out / 'preflight.json', {'clock_sha256': original_sha, 'admission': str(admission),
        'admission_sha256': admission_sha, 'reserve_seconds': reserve,
        'source_review': grant['source_review'], 'source_review_sha256': grant['source_review_sha256'],
        'startup_observation': startup_observation,
        'canonical_lock': 'exact acquire/unlock/close returned', 'preflight_monotonic': time.monotonic(),
        'complete_expanded_input_count': len(expanded['hashes']), 'corpus_counts': SPEC['corpus_counts'],
        'affinity_available': sorted(os.sched_getaffinity(0)),
        'pressure': {k: Path('/proc/pressure/' + k).read_text() for k in ('cpu', 'memory', 'io')}})
    print(json.dumps({'prepared_cohort': str(out), 'launch_performed': False}))


def prepare(out, index):
    c, pre = cohort(out); phases = phases_for(c)
    require(0 <= index < len(phases), 'EXACT_PINNED_PHASE_ONLY')
    require(sha(pre['admission']) == pre['admission_sha256'], 'ADMISSION_CHANGED')
    for i, p in enumerate(phases):
        dest = out / p['name']
        if i < index:
            require(dest.is_dir(), 'PREVIOUS_PHASE_MISSING')
            done = read(dest / 'cessation.json')
            require(done['status'] in ('CANARY_PHASE_EVIDENCED', 'COLLECTION_EVIDENCED', 'RUNTEST_REPORTED', 'RUNTEST_PRODUCT_FAIL') and
                    done['next_phase_admissible'] is True, 'PREVIOUS_PHASE_NOT_CLOSED_NO_RETRY')
            if 'runtest_file' in p:
                require(not (dest / 'native-stop.json').exists() and
                        not (dest / 'native-stop.json').is_symlink() and
                        not (dest / 'native-owner/stop.json').exists() and
                        not (dest / 'native-owner/stop.json').is_symlink(),
                        'PREVIOUS_RUNTIME_STOP_INTENT_RETAINED')
                returned = read(dest / 'registry-release.json')
                require(returned.get('status') == 'EXACT_REGISTRY_HANDLE_RETURNED' and

def scan_group(contour):
    remaining, known, errors = [], [], []
    for p in Path('/proc').iterdir():
        if not p.name.isdigit():
            continue
        try:
            raw = (p / 'stat').read_text(); fields = raw[raw.rfind(')') + 2:].split()
            pid, born, pgid = int(p.name), int(fields[19]), int(fields[2])
            if pgid == contour['timeout']['pgid'] or (contour['identity'].get('operation') == 'pytest_runtest_file' and
                                                    int(fields[3]) == contour['timeout']['sid']):
                remaining.append({'pid': pid, 'start_ticks': born, 'state': fields[0]})
            if any(pid == v['pid'] and born == v['start_ticks'] for v in (contour['timeout'], contour['guard'])):
                known.append(pid)
        except FileNotFoundError:
            pass
        except Exception as e:
            errors.append({'path': str(p), 'error': type(e).__name__})
    return {'remaining': remaining, 'known': known, 'errors': errors}


def node_key(identity):
    require(isinstance(identity, dict) and set(identity) ==
            {'path', 'object_path', 'kind', 'type', 'nodeid', 'source_relative_path'}, 'MISSING_OR_EXTRA_NODE_IDENTITY')
    path = identity['path']; object_path = identity['object_path']; kind = identity['kind']
    require(isinstance(path, str) and isinstance(object_path, str) and Path(object_path).is_absolute() and
            os.path.normpath(object_path) == object_path and Path(object_path).resolve(strict=True) == Path(path) and
            Path(path).is_absolute() and Path(path).resolve(strict=True) == Path(path),
            'NONCANONICAL_NODE_PATH')
    require(kind in ('session', 'directory', 'module', 'class', 'function', 'item') and
            isinstance(identity['type'], str) and '.' in identity['type'] and
            isinstance(identity['nodeid'], str) and
            identity['source_relative_path'] == os.path.relpath(path, SOURCE), 'NODE_KIND_OR_RELATIVE_PATH')
    # Empty and colliding raw nodeids are legitimate. Never parse a file from one.
    return object_path, kind, identity['nodeid']

    private(launch_path, 0o400); launch = read(launch_path)

            resources[0]['process_group'] == contour['timeout']['pgid'] and len(set(resources[0]['affinity'])) == 2 and
            all(type(v) is int and v >= 0 for v in resources[0]['affinity']) and
            resources[0]['as_limit'] == [2 << 30, 2 << 30] and resources[0]['cpu_limit'] == [600, 600] and
            resources[0]['lock_identity'] == SPEC['canonical_lock']['identity'][:2] and
            resources[0]['original_deadline'] == c['deadline_monotonic'] and (True if runtime else bool(normal)) and
            all(n['lock_unlock_call_returned'] is True and n['lock_fd_close_returned'] is True for n in normal), 'RESOURCE_OR_NORMAL_FINALLY_RECEIPT_GAP')
    require([a['contour'] for a in attempts if a['kind'] == 'native_contour'] == [contour], 'CONTOUR_ATTEMPT_MISMATCH')
    result_path = run / (c['run_name'] + '-results.json')
    final_input_error = None
    try:
        result = read(result_path) if result_path.exists() else {}
    except Exception as error:

            raise

    child_expected = {p: h for p, h in expanded.items() if p.startswith(str(SOURCE) + '/') and p != str(SOURCE / 'source-pins.json') or
                      p in read(SOURCE / 'runtime-pins.json') or p in read(SOURCE / 'ancillary-pins.json')}

    if runtime:

    if c['selection'] == '@canary':
        require(all(sha(parent / n) == h for n, h in before['outside'].items()), 'SYNTHETIC_OUTSIDE_CHANGED')
        decision = None
    elif runtime:

                'phase_clock_sha256': before['phase_clock_sha256'],

        except Exception as error:

        decision = runtest_check(run, c, result, contour, terminal['exit_code'], guard_stopped,
                                 child_expected, final_input_error=final_input_error)
    else:
        decision = collection_check(parent, run, c, result, contour, expanded)
    # Repeat expanded verification after classification so all supporting sources are bound.
    source_review(pre['source_review'], pre['source_review_sha256'])
    require(sha(pre['admission']) == pre['admission_sha256'], 'POST_REVIEW_ADMISSION_DRIFT')
    require(inputs() == read(out / 'input-pins.json') and sha(out / 'original-clock.json') == pre['clock_sha256'] and
            sha(run / 'clock.json') == before['phase_clock_sha256'], 'POST_REVIEW_INPUT_OR_CLOCK_DRIFT')
    group = scan_group(contour); require(not any(group.values()), 'STOP_UNCONFIRMED:' + json.dumps(group))
    lock_probe(); upper = time.monotonic(); require(0 < upper < c['deadline_monotonic'], 'CESSATION_NOT_EVIDENCED_BEFORE_ORIGINAL_DEADLINE')
    backend_time = back.get('observed_terminal_monotonic')
    require(backend_time is None or finite(backend_time) and backend_time <= upper, 'INVALID_BACKEND_TERMINAL_TIMESTAMP')
    obs = {'identity': identity, 'guard_identity': contour['guard'], 'timeout_identity': contour['timeout'],
           'lexical_input_joins': lexical_joins,
           'backend_reaped': True, 'exit_code': terminal['exit_code'], 'exact_known_pids_absent': True,
           'exact_group_scan_quiescent': True, 'outside_synthetic_hashes_unchanged': True,
           'phase_clock_unchanged': True, 'all_input_pins_unchanged': True, 'canonical_lock_identity_verified': True,
           'fresh_lock_acquire_unlock_close_returned': True, 'post_reap_cessation_upper_bound_monotonic': upper,
           'original_deadline_cessation_evidenced': True, 'exact_kernel_terminal_time_exposed': False,
           'timing_provenance': 'Owner sample AFTER actual backend terminal/reap, whole-group known identity scan, full hashes, classification and fresh exact lock return; upper bound only',
           'backend_start_sha256': sha(parent / 'backend-start.json'), 'backend_result_sha256': sha(parent / 'backend-result.json')}
    if c['selection'] == '@canary':
        reports = result['reports']; positive = read(SOURCE / 'canary-plan.json')['positive']
        obs.update(after_denial_sentinel_present=(run / 'DENIAL_WAS_SWALLOWED').exists(),
            alarm_swallowed_sentinel_present=(run / 'ALARM_WAS_SWALLOWED').exists(),
            later_sentinel_present=(run / 'LATER_TEST_EXECUTED').exists(),
            later_sentinel_bytes=(run / 'LATER_TEST_EXECUTED').read_text() if (run / 'LATER_TEST_EXECUTED').exists() else None,
            selected_test_call_passed=any(r['when'] == 'call' and r['outcome'] == 'passed' and r['nodeid'].endswith('::' + positive['node']) for r in reports),
            later_test_call_passed=any(r['when'] == 'call' and r['outcome'] == 'passed' and r['nodeid'].endswith('::test_later_test_sentinel') for r in reports))
        # Unchanged data-only oracle is used ONLY for this actual @canary identity.
        oracle = runpy.run_path(str(SOURCE / 'canary_oracle.py'))
        decision = oracle['evaluate'](c, positive, read(run / 'operation_entry.json'), attempts,
            dict(obs, terminal_monotonic=upper), contour, read(run / 'positive_assertions_complete.json'), None)
    good = (decision['status'] in ('CANARY_PHASE_EVIDENCED', 'COLLECTION_EVIDENCED', 'RUNTEST_REPORTED', 'RUNTEST_PRODUCT_FAIL') and
            (not runtime or decision.get('final_runtime_evidence_validated') is True))
    put(parent / 'cessation.json', {'status': decision['status'], 'next_phase_admissible': good,
        'decision': decision, 'observation': obs, 'group_scan': group, 'phase_clock_sha256': before['phase_clock_sha256'],
        'original_clock_sha256': pre['clock_sha256'], 'scope': phase['scope'], 'attempts': attempts,
        'runtime_or_product_accepted': False})
    print(json.dumps({'phase': phase['name'], 'status': decision['status'], 'launch_performed': False}))


def runtest_check(run, clock, result, contour, exit_code, guard_stopped, child_expected, *, final_input_error=None):
    """Receipt evidence can fail while independent cessation still gets checked."""
    api = runpy.run_path(str(SOURCE / 'runtest_protocol.py'))
    plan = read(SOURCE / 'runtest-plan.json')
    unit = api['unit_for'](plan, clock['runtest_file'], clock['selection'])
    progress = run / (clock['run_name'] + '-progress.ndjson')
    events = []
    try:
        if progress.exists():
            data = progress.read_bytes()
            require(not data or data.endswith(b'\n'), 'TORN_RUNTIME_PROGRESS')
            events = [json.loads(line) for line in data.splitlines()]
        for event in events:
            bindings = event.get('selected', []) + event.get('deselected', [])
            if 'binding' in event: bindings.append(event['binding'])
            for binding in bindings: node_key(binding['actual'])
        decision = api['census'](unit, events, contour['identity'], sha(run / 'clock.json'),
                                 exit_code, result or None)
        decision = api['final_runtime_decision'](decision, result, plan, child_expected,
                                                  final_input_error=final_input_error)
        decision['actual_event_prefix'] = events
        if guard_stopped:
            decision['status'] = 'RUNTEST_CONTAINMENT_STOP'
        decision['required_effectful_not_run'] = plan['required_not_run']
        decision['missing_original_files'] = plan['missing']
        return decision
    except Exception as error:
        return {'status': 'RUNTEST_EVIDENCE_STOP', 'reason': str(error),
                'failure_observed': getattr(error, 'failure_observed', []),
                'actual_event_prefix': events, 'required_identities': unit['expected'],
                'required_effectful_not_run': plan['required_not_run'],
                'unclassified_items': 'UNKNOWN_NOT_PASS; malformed evidence cannot allocate runtime success',
                'runtime_or_product_accepted': False}


def run_donor_file(out, index, registry, *, native_owner):
    """Finite single-file dispatch through the pinned donor and EXISTING registry.

    The caller owns the native backend/outer kill boundary across this call. It
    supplies actual ownership, not a boolean approval. This function is source
    only until runtime cohort/source/effect admission and overlays are resolved.
    It never loops over files or retries a failed call; prepare/observe serializes.
    """
    original, pre = cohort(out); phases = phases_for(original)
    require(0 <= index < len(phases) and 'runtest_file' in phases[index], 'EXACT_RUNTIME_PHASE_REQUIRED')
    phase = phases[index]; parent = out / phase['name']; run = parent / 'run'
    before = read(parent / 'prelaunch.json'); clock = read(run / 'clock.json')
    template = read(SOURCE / 'clean-invocation.json')['runtest_argv_templates'][phase['selection']]
    expected_argv = [v.replace('{UNIT_ID}', phase['runtest_file']).replace('{RUN_NAME}', phase['name']).replace('{RUN}', str(run)).replace('{DONOR_TEMP}', clock['donor_temporary_root']['path']) for v in template]
    require(before['argv'] == expected_argv, 'EXACT_RUNTIME_PRELAUNCH_ARGV_REQUIRED')
    require(sha(run / 'clock.json') == before['phase_clock_sha256'] and
            {k: clock[k] for k in CORE} == {k: original[k] for k in CORE}, 'RUNTEST_CLOCK_RESET')
    require(native_owner.out == out and native_owner.index == index, 'FOREIGN_NATIVE_RUNTIME_OWNER')
    binding = native_owner.reserve_binding()
    owner_info = {'original_clock_sha256': binding['original_clock_sha256'], 'phase_index': index,
        'backend_handle': binding['unit'] + '#' + binding['invocation_id'],
        'native_backend_pid': binding['main']['pid'],
        'session_key': original['assignment'] + '#' + str(original['generation']) + '/' + phase['name']}
    require(owner_info['original_clock_sha256'] == pre['clock_sha256'], 'ACTUAL_NATIVE_OWNER_REQUIRED')
    # No reused/old registry or donor module. Changed files are imported only
    # AFTER independent final overlay integration; source pins bind their paths.
    import inspect
    require(sha(inspect.getsourcefile(type(registry))) == SPEC['runtime_overlay']['registry']['sha256'],
            'RUNTIME_REGISTRY_OVERLAY_NOT_INTEGRATED')
    donor_ref = SPEC['runtime_overlay']['runner']; require(sha(donor_ref['path']) == donor_ref['sha256'], 'DONOR_RUNNER_DRIFT')
    donor = runpy.run_path(donor_ref['path'])
    unit = read(SOURCE / 'runtest-plan.json')['units'][index]
    require(unit['unit_id'] == phase['runtest_file'], 'RUNTEST_ORDER_DRIFT')
    # All file effects stay in the original allowance, with owner reconciliation
    # reserve. The private fixture dirs/clock were prepared by the same helper.
    remaining = lambda: max(0.001, original['deadline_monotonic'] - time.monotonic() - pre['reserve_seconds'])
    require(remaining() > phase['outer'] + 1, 'INSUFFICIENT_ORIGINAL_FILE_BUDGET')
    session = registry.reserve_runtest(command=' '.join(before['argv']), cwd=unit['root'],
        task_id=original['assignment'], session_key=owner_info['session_key'],
        unit_id=unit['unit_id'], deadline=original['deadline_monotonic'], native_owner=native_owner)
    lower = time.monotonic()
    put(parent / 'backend-prelaunch.json', {'state': 'OWNER_PRELAUNCH_LOWER_BOUND_RECORDED',
        'argv': before['argv'], 'phase_index': index, 'phase_clock_sha256': before['phase_clock_sha256'],
        'original_clock_sha256': pre['clock_sha256'], 'native_backend_pid': owner_info['native_backend_pid'],
        'boot_id': original['boot_id'], 'prelaunch_lower_bound_monotonic': lower})
    temp_api = runpy.run_path(str(SOURCE / 'admission.py'))
    temproot = str(temp_api['verify_donor_temporary_root'](clock))
    holder = {}
    def spawn(cmd, **kwargs):
        require(cmd == [donor['sys'].executable, '-m', 'pytest', str(Path(unit['file']['path']))],
                'DONOR_FILE_COMMAND_DRIFT')
        # Reuse donor Popen lifecycle but make the existing admitted guard the
        # real per-file interpreter. It calls pytest with donor config/conftests.
        # Donor kwargs own stdout and a new SID. env-i in before.argv owns the
        # clean interpreter environment, never the caller's credential env.
        require(kwargs['env'].get('TMPDIR') == temproot and
                kwargs['env'].get('PYTEST_DEBUG_TEMPROOT') == temproot,
                'DONOR_TEMP_ENVIRONMENT_DRIFT')
        kwargs['env'] = temp_api['clean_environment'](run, clock=clock)
        kwargs['stdin'] = donor['subprocess'].DEVNULL
        kwargs['close_fds'] = True
        # Registry retains the actual handle and acquires a native pidfd before
        # return; its failure path is covered by donor retained_handle/finally.
        # Identity/checkpoint/backend writes may still fail after transfer.
        wrapped = native_owner.confined_donor_argv(before['argv'], kwargs)
        # Only an independently qualified original owner may ever return here.
        # Its real host monitor handle stays with this registry and donor waiter.
        holder['semantic_argv'] = list(before['argv'])
        holder['wrapped_argv'] = list(wrapped)
        return registry.spawn_runtest(session, wrapped, kwargs, native_owner=native_owner)
    def after_spawn(proc):
        holder['proc'] = proc
        registry.attach_runtest(session, proc)
        raw = Path('/proc/' + str(proc.pid) + '/stat').read_text()
        put(parent / 'backend-start.json', {'argv': holder['wrapped_argv'], 'semantic_argv': holder['semantic_argv'], 'pid': proc.pid, 'stat': raw,
            'boot_id': original['boot_id'], 'backend_handle': owner_info['backend_handle'],
            'native_backend_pid': owner_info['native_backend_pid'],
            'owner_prelaunch_sha256': sha(parent / 'backend-prelaunch.json')})
        put(parent / 'runtest-custody.json', {'session_id': session.id, 'pid': session.pid,
            'start_ticks': session.host_start_time, 'pgid': session.runtest_pgid, 'sid': session.runtest_sid,
            'boot_id': session.runtest_boot_id, 'unit_id': unit['unit_id'],
            'original_deadline': session.runtest_deadline, 'owner_task_id': session.owner_task_id,
            'phase_clock_sha256': before['phase_clock_sha256'], 'native_outer': session.runtest_outer})
    def capture(proc, output, error, cuts):
        require(proc is session.process, 'FOREIGN_DONOR_OUTPUT')
        # Keep output before fallible diagnostics/FD close. Root reap and local
        # fd release do not release registry or native whole-cgroup custody.
        try:
            session.append_output(output)
            holder['proc'] = proc
        finally:
            registry.release_runtest_pidfd(session, proc)
        put(parent / 'runtest-drain.json', {'session_id': session.id, 'pid': proc.pid,
            'exit_code': proc.returncode, 'output': output,
            'primary_error': type(error).__name__ if error is not None else None,
            'cleanup_errors': list(cuts), 'registry_persistence_error': session.runtest_persistence_error,
            'registry_pidfd_error': session.runtest_pidfd_error,
            'sole_drainer': 'donor communicate', 'reaped': proc.returncode is not None,
            'runtime_or_product_accepted': False})
    def stop(proc, pgid):
        require(proc is session.process and pgid == session.runtest_pgid, 'FOREIGN_DONOR_STOP')
        # On natural exit, do not signal a historical numeric PGID. The owner
        # scans its exact SID/PGID; any survivor retains custody for outer stop.
        if proc.returncode is None:
            stop_result = registry.stop_runtest(session, source='donor_file_timeout_or_abort')
            require(stop_result['status'] == 'STOP_REQUESTED' and not stop_result['persistence_error'],
                    'RUNTEST_STOP_DIAGNOSTIC_OR_CESSATION_UNCONFIRMED')
    def settle(proc, pgid, temproot):
        require(temproot == str(temp_api['verify_donor_temporary_root'](clock)), 'FOREIGN_DONOR_TEMP_SETTLEMENT')
        require(proc is session.process and pgid == session.runtest_pgid, 'FOREIGN_DONOR_SETTLEMENT')
        require(not session.runtest_pidfd_error, 'RUNTEST_PIDFD_FAILURE_RETAINED_NO_ACCEPTANCE')
        require(proc.returncode is not None, 'DONOR_NOT_REAPED')
        require(type(session.pid) is int and session.pid == proc.pid and
                type(session.host_start_time) is int and session.host_start_time > 0 and
                session.runtest_pgid == proc.pid and session.runtest_sid == proc.pid and
                Path('/proc/sys/kernel/random/boot_id').read_text().strip() == session.runtest_boot_id,
                'STOP_UNCONFIRMED_DONOR_IDENTITY_GAP')
        remaining_group = []
        for path in Path('/proc').iterdir():
            if not path.name.isdigit(): continue
            try:
                raw = (path / 'stat').read_text(); fields = raw[raw.rfind(')') + 2:].split()
            except FileNotFoundError: continue
            if int(fields[2]) == session.runtest_pgid or int(fields[3]) == session.runtest_sid:
                remaining_group.append({'pid': int(path.name), 'start_ticks': int(fields[19]), 'state': fields[0]})
        require(not remaining_group, 'STOP_UNCONFIRMED_DONOR_SID_PGID:' + json.dumps(remaining_group))
        require(time.monotonic() < original['deadline_monotonic'], 'DONOR_CESSATION_AFTER_DEADLINE')
        put(parent / 'backend-result.json', {'backend_handle': owner_info['backend_handle'],
            'native_backend_pid': owner_info['native_backend_pid'],
            'tool_result': {'exit_code': proc.returncode}, 'observed_terminal_monotonic': time.monotonic(),
            'donor_temproot': temproot})
    def terminal(code):
        require(code == holder['proc'].returncode, 'DONOR_EXIT_NORMALIZATION_DRIFT')
    # No call to main(), discovery, duration cache, retry, parallel executor or
    # cross-run sweep. This exact donor one-attempt primitive is reused serially.
    donor['_run_one_file_once'](Path(unit['file']['path']), [], Path(unit['root']),
        min(float(phase['outer']) + 1, remaining()),
        custody={'spawn': spawn, 'after_spawn': after_spawn, 'retained_handle': lambda: session.process,
                 'retained_pgid': lambda: session.runtest_pgid, 'capture': capture,
                 'stop': stop, 'settle': settle, 'remaining': remaining, 'terminal': terminal,
                 'temporary_root': lambda: str(temp_api['verify_donor_temporary_root'](clock))})
    temp_api['verify_donor_temporary_root'](clock, removed=True)
    put(parent / 'donor-temp-cleanup.json', {'state': 'EXACT_DONOR_TEMP_ABSENCE_OBSERVED',
        'phase_clock_sha256': before['phase_clock_sha256'],
        'donor_temporary_root': clock['donor_temporary_root'], 'backend_pid': holder['proc'].pid})
    # Retain the registry handle until observe() and actual OUTER owner custody
    # are reconciled. A runner return or parsed console summary cannot release it.
    return {'session_id': session.id, 'phase': phase['name'], 'observer_required': True,
            'exit_code': holder['proc'].returncode, 'runtime_or_product_accepted': False}


def finish_runtime_file(out, index, registry, *, native_owner):
    """Close the original durable session after actual whole-unit cessation.

    Recovery may finish custody after an expired clock, but never admits another
    phase or invents a prior cessation timestamp, exit code or donor Popen.
    """
    require(native_owner.out == out and native_owner.index == index, 'FOREIGN_NATIVE_CLOSER')
    original, pre = native_owner.original, native_owner.pre
    phase = phases_for(original)[index]; parent = out / phase['name']
    require(not (parent / 'registry-release.json').exists(), 'REGISTRY_ALREADY_RETURNED_NO_RETRY')
    with registry._lock:
        sessions = list(registry._running.values())
    require(len(sessions) == 1 and sessions[0].runtest_custody and
            sessions[0].runtest_outer == native_owner.binding, 'EXACT_RETAINED_REGISTRY_HANDLE_REQUIRED')
    session = sessions[0]
    # Must consume the immutable original disposition BEFORE finish/recovery
    # safety flags can obscure cancellation or uncertainty. No registry edit.
    original_allows_next = native_owner.original_row_allows_next_phase(session)
    done_path = parent / 'cessation.json'; custody_path = parent / 'runtest-custody.json'
    done = read(done_path) if done_path.exists() else {}
    custody = read(custody_path) if custody_path.exists() else {}
    obs = done.get('observation', {})
    inner_complete = (custody.get('session_id') == session.id and
        custody.get('native_outer') == session.runtest_outer and
        custody.get('pid') == session.pid and custody.get('start_ticks') == session.host_start_time and
        custody.get('pgid') == session.runtest_pgid and custody.get('sid') == session.runtest_sid and
        custody.get('boot_id') == session.runtest_boot_id and
        custody.get('original_deadline') == session.runtest_deadline and
        custody.get('unit_id') == session.runtest_unit and
        custody.get('owner_task_id') == session.owner_task_id and
        obs.get('identity') == expected_identity(read(parent / 'run/clock.json')) and
        all(obs.get(k) is True for k in ('backend_reaped', 'exact_known_pids_absent',
            'exact_group_scan_quiescent', 'fresh_lock_acquire_unlock_close_returned',
            'original_deadline_cessation_evidenced')))
    returncode = None
    drain_allows_next = False
    if inner_complete:
        drain = read(parent / 'runtest-drain.json'); back = read(parent / 'backend-result.json')
        require(drain['session_id'] == session.id and drain['reaped'] is True and
                type(drain['exit_code']) is int and drain['exit_code'] == back['tool_result']['exit_code'] and
                obs.get('backend_result_sha256') == sha(parent / 'backend-result.json') and
                obs.get('backend_start_sha256') == sha(parent / 'backend-start.json'),
                'RETAINED_DONOR_RESULT_DRIFT')
        returncode = drain['exit_code']
        # The drain can retain a persistence/cleanup failure that never reached
        # the original checkpoint. Keep the genuine result, fence continuation.
        drain_allows_next = ('primary_error' in drain and drain['primary_error'] is None and
                            type(drain.get('cleanup_errors')) is list and drain['cleanup_errors'] == [] and
                            type(drain.get('registry_persistence_error')) is str and
                            drain['registry_persistence_error'] == '' and
                            type(drain.get('registry_pidfd_error')) is str and
                            drain['registry_pidfd_error'] == '')
    # Persist actual native closure BEFORE the checkpoint row may be removed.
    # A crash after removal can then reconcile custody without replay or PASS.
    prior = native_owner.closure_for(session)
    closure_path = parent / 'outer-closure.json'
    if closure_path.exists():
        private(closure_path, 0o400); existing = read(closure_path)
        require(existing.get('registry_session_id') == session.id and
                existing.get('binding') == prior['binding'] and existing.get('returncode') == returncode,
                'FOREIGN_RETAINED_NATIVE_CLOSURE')
    else:
        put(closure_path, dict(prior, registry_session_id=session.id, returncode=returncode,
            state='ACTUAL_NATIVE_RUNTEST_FILE_BOUNDARY_QUIESCENT', runtime_or_product_accepted=False))
    # Registry re-observes native state, then strictly saves closure before release.
    proof = registry.finish_runtest(session, native_owner=native_owner, returncode=returncode)
    next_phase = (original_allows_next and drain_allows_next and inner_complete and
                  proof['cessation_before_original_deadline'] and
                  done.get('next_phase_admissible') is True and
                  not native_owner.stop_requested())
    put(parent / 'registry-release.json', {'session_id': session.id,
        'outer_receipt': str(parent / 'outer-closure.json'),
        'outer_receipt_sha256': sha(parent / 'outer-closure.json'),
        'cessation_sha256': sha(done_path) if done_path.exists() else None,
        'status': 'EXACT_REGISTRY_HANDLE_RETURNED', 'next_phase_admissible': next_phase,
        'recovered_same_durable_session': session.detached, 'exit_code': returncode,
        'final_native_closure': proof,
        'original_registry_disposition': str(parent / 'registry-disposition.json'),
        'original_registry_disposition_sha256': sha(parent / 'registry-disposition.json'),
        'recovery_reentered': native_owner.disposition_reentered,
        'disposition': ('INNER_RESULT_PRESERVED' if next_phase else
                        'STOP_INCONCLUSIVE_NOT_RUN_REMAINS'),
        'runtime_or_product_accepted': False})


def native_runtime_owner(out, index, *, retained=False):
    """Load only final pinned native support; active admission remains cohort()."""
    import types
    if not retained:
        cohort(out)
    expected = read(HERE / 'source-pins.json')['runtest_outer.py']
    require(sha(HERE / 'runtest_outer.py') == expected, 'NATIVE_OUTER_HELPER_CHANGED')
    # These exact primary native files/binaries join the final input map. A
    # missing spec is a prerequisite failure, never permission to use defaults.
    dependencies = SPEC['native_outer']['dependency_pins']
    require({'/usr/bin/systemctl', '/usr/bin/systemd-run', '/usr/bin/busctl', '/usr/bin/env',
             str(Path(SPEC['roots']['project']) / 'plugins/friday_rework/supervision.py'),
             str(Path(SPEC['roots']['project']) / 'plugins/friday_rework/associations.py'),
             str(Path(SPEC['roots']['project']) / 'plugins/friday_rework/boundary.py'),
             str(Path(SPEC['roots']['project']) / 'plugins/friday_rework/__init__.py'),
             str(Path(SPEC['roots']['project']) / 'plugins/friday_rework/admission.py')} <= set(dependencies),
            'INCOMPLETE_NATIVE_CONTROL_INPUT_PINS')
    for path, digest in dependencies.items():
        require(sha(path) == digest, 'NATIVE_OUTER_DEPENDENCY_CHANGED:' + path)
    sys.path.insert(0, SPEC['roots']['project'])
    api = runpy.run_path(str(HERE / 'runtest_outer.py'))
    supervisor_source = __import__('inspect').getsourcefile(api['NativeSupervisor'])
    require(supervisor_source in SPEC['native_outer']['dependency_pins'] and
            sha(supervisor_source) == SPEC['native_outer']['dependency_pins'][supervisor_source],
            'UNPINNED_NATIVE_SUPERVISOR_IMPORT')
    return api['FileOwner'](types.SimpleNamespace(**globals()), out, index, retained=retained)


def runtime_command(command, out, index):
    """Concrete finite caller replacing the consumed collection phase-caller.js."""
    outer = native_runtime_owner(out, index, retained=command in ('close-runtime', 'cancel-runtime'))
    if command == 'unit-runtime':
        outer.unit_entry()
        return
    if command == 'submit-runtime':
        outer.submit()
        outer = native_runtime_owner(out, index, retained=True)
    proof = outer.close(stop=command == 'cancel-runtime')
    # Recovery imports occur only AFTER the entire live registry/test subtree is
    # quiescent. No registry/test process is launched by this recovery path.
    checkpoint = outer.control / 'profile/processes.json'
    require(checkpoint.is_file(), 'NO_DURABLE_REGISTRY_ROW_AFTER_NATIVE_CLOSE_NO_RETRY')
    checkpoint_bytes = checkpoint.read_bytes()
    rows = json.loads(checkpoint_bytes)
    if rows == [] and (outer.parent / 'outer-closure.json').exists():
        # Prior durable closure followed by checkpoint removal and a controller
        # crash: finish only the stopped custody record, never resume this file.
        prior = read(outer.parent / 'outer-closure.json')
        private(outer.parent / 'outer-closure.json', 0o400)
        require(prior.get('binding') == proof['binding'] and
                prior.get('state') == 'ACTUAL_NATIVE_RUNTEST_FILE_BOUNDARY_QUIESCENT' and
                bool(prior.get('registry_session_id')), 'UNPROVED_REMOVED_REGISTRY_ROW')
        put(outer.parent / 'registry-release.json', {'session_id': prior['registry_session_id'],
            'status': 'EXACT_REGISTRY_HANDLE_RETURNED', 'next_phase_admissible': False,
            'outer_receipt_sha256': sha(outer.parent / 'outer-closure.json'),
            'cessation_sha256': sha(outer.parent / 'cessation.json') if (outer.parent / 'cessation.json').exists() else None,
            'disposition': 'RECONCILED_REMOVED_ROW_STOPPED_NO_RETRY',
            'final_native_closure': proof, 'runtime_or_product_accepted': False})
        return
    require(isinstance(rows, list) and len(rows) == 1 and rows[0].get('runtest_custody') is True and
            rows[0].get('runtest_outer') == proof['binding'], 'FOREIGN_OR_AMBIGUOUS_NATIVE_CHECKPOINT')
    # The unit is actually closed and the registry is not loaded yet. Persist
    # the original flags with exact checkpoint hash BEFORE recover mutates them.
    outer.retain_registry_disposition(rows, hashlib.sha256(checkpoint_bytes).hexdigest())
    old_home = os.environ.get('HERMES_HOME'); old_profile = os.environ.pop('HERMES_PROFILE', None)
    os.environ['HERMES_HOME'] = str(outer.control / 'profile')
    try:
        registry = outer.registry()
        require(registry.recover_from_checkpoint() == 1, 'ORIGINAL_REGISTRY_RECOVERY_UNCONFIRMED')
        finish_runtime_file(out, index, registry, native_owner=outer)
    finally:
        if old_home is None: os.environ.pop('HERMES_HOME', None)
        else: os.environ['HERMES_HOME'] = old_home
        if old_profile is not None: os.environ['HERMES_PROFILE'] = old_profile


if __name__ == '__main__':
    try:
        command = sys.argv[1]; out = Path(sys.argv[2]); require(out.is_absolute(), 'ABSOLUTE_OWNER_OUTPUT_REQUIRED')
        if command == 'initialize' and len(sys.argv) == 7:
            initialize(out, Path(sys.argv[3]), sys.argv[4], Path(sys.argv[5]), sys.argv[6])
        elif command == 'prepare' and len(sys.argv) == 4:
            prepare(out, int(sys.argv[3]))
        elif command in ('submit-runtime', 'unit-runtime', 'close-runtime', 'cancel-runtime') and len(sys.argv) == 4:
            runtime_command(command, out, int(sys.argv[3]))
        elif command == 'observe' and len(sys.argv) == 4:
            index = int(sys.argv[3])
            try:
                observe(out, index)
            except Exception as error:
                phases = phases_for(read(out / 'original-clock.json'))
                if 0 <= index < len(phases) and (out / phases[index]['name']).is_dir() and not (out / phases[index]['name'] / 'cessation.json').exists():
                    put(out / phases[index]['name'] / 'cessation.json', {'status': 'INCONCLUSIVE', 'next_phase_admissible': False,
                        'reason': str(error), 'actual_owned_backend_handle': 'Retain original backend-start/backend-result; helper does not stop/retry/release custody',
                        'observation_sample_monotonic': time.monotonic(), 'cessation_evidenced': False, 'runtime_or_product_accepted': False})
                raise
        else:
            raise ValueError('EXACT_PINNED_OWNER_API_ONLY')
    except Exception as error:
        raise SystemExit('INCONCLUSIVE_NO_RETRY: ' + str(error))
