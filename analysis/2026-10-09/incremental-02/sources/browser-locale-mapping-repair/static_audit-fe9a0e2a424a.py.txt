"""Data/hash/AST audit only; no candidate imports, collection or tests."""
import ast
import datetime as dt
import difflib
import hashlib
import json
import os
from pathlib import Path
import stat

O = Path(__file__).resolve().parent
C = O / 'candidate'
B = Path('<SOL_WORK>/FRW041-BROWSER-READ-DIAGNOSTIC-evidence/candidate')
cache = {}

def digest(p):
    p = str(p)
    if p not in cache:
        with Path(p).open('rb') as f:
            cache[p] = hashlib.file_digest(f, 'sha256').hexdigest()
    return cache[p]

def ref(p):
    p = Path(p)
    return {'path': str(p), 'sha256': digest(p), 'bytes': p.stat().st_size}

def read(p):
    return json.loads(Path(p).read_text())

inventory = read(O / 'baseline-source-inventory.json')
names = sorted(inventory)
assert len(names) == 27
assert sorted(p.name for p in C.iterdir()) == names
for name, row in inventory.items():
    assert digest(row['path']) == row['sha256'], ('BASELINE_CHANGED', name)
    s = (C / name).lstat()
    assert stat.S_ISREG(s.st_mode) and s.st_uid == os.getuid() and s.st_nlink == 1
assert not any((C / name).exists() for name in ('runtime-clock.json', 'runtime-admission.json', 'controls', 'native', 'project', 'owner'))

python = sorted(n for n in names if n.endswith('.py'))
ast_checks = {}
for name in python:
    old, new = (B / name).read_text(), (C / name).read_text()
    old_tree, new_tree = ast.parse(old), ast.parse(new)
    if name in ('run.py', 'launch.py'):
        assert old.count("admission['generation']==3") == 1
        assert new == old.replace("admission['generation']==3", "admission['generation']==4")
        normalized = ast.parse(new.replace("admission['generation']==4", "admission['generation']==3"))
        assert ast.dump(normalized) == ast.dump(old_tree)
        expected_diff = ''.join(difflib.unified_diff(old.splitlines(True), new.splitlines(True), fromfile=str(B / name), tofile=str(C / name)))
        assert (C / (name + '.delta.patch')).read_text() == expected_diff
        ast_checks[name] = 'Exactly one inert admission generation literal3->4; all other AST/source bytes identical'
    else:
        assert old == new
        assert ast.dump(old_tree) == ast.dump(new_tree)
        ast_checks[name] = 'Byte-identical and AST-identical'

map_checks = {}
for name in ('input-pins.json', 'selected-pins.json'):
    assert (B / name).read_bytes() == (C / name).read_bytes()
    map_checks[name] = {'old_entries': len(read(B / name)), 'new_entries': len(read(C / name)), 'byte_identical': True}
old_runtime, new_runtime = read(B / 'runtime-flat-pins.json'), read(C / 'runtime-flat-pins.json')
locale = read(O / 'locale-provenance.json')['locale']
assert all(new_runtime.get(p) == h for p, h in old_runtime.items())
assert set(new_runtime) - set(old_runtime) == {locale['path']}
assert new_runtime[locale['path']] == locale['sha256'] == digest(locale['path'])
assert len(new_runtime) == len(old_runtime) + 1 == 13293
map_checks['runtime-flat-pins.json'] = {'old_entries': len(old_runtime), 'new_entries': len(new_runtime), 'old_exact_subset': True, 'added': {locale['path']: locale['sha256']}, 'removed_or_changed': []}
old_launch, new_launch = read(B / 'launch-pins.json'), read(C / 'launch-pins.json')
assert {Path(p).name for p in old_launch} == {Path(p).name for p in new_launch}
assert len(old_launch) == len(new_launch) == 10
assert all(Path(p).parent == C and digest(p) == h for p, h in new_launch.items())
map_checks['launch-pins.json'] = {'old_entries': 10, 'new_entries': 10, 'same_names_with_relocated_current_hashes': True}

counts = {k: sum(v.values()) for k, v in read(C / 'expected-suite-cases.json').items()}
assert counts == {'controls': 21, 'native': 116, 'project': 38, 'owner': 71}, counts
diagnostic = ast.parse((C / 'test_read_refusal_diagnostic_source.py').read_text())
function_count = case_count = 0
for node in diagnostic.body:
    if isinstance(node, ast.FunctionDef) and node.name.startswith('test_'):
        function_count += 1
        cases = 1
        for decorator in node.decorator_list:
            if isinstance(decorator, ast.Call) and ast.unparse(decorator.func) == 'pytest.mark.parametrize':
                assert isinstance(decorator.args[1], (ast.List, ast.Tuple))
                cases *= len(decorator.args[1].elts)
        case_count += cases
assert (function_count, case_count) == (14, 27)
assert read(C / 'semantic-delta.json')['authority_changed'] is False
assert read(C / 'semantic-delta.json')['old_generation2_replay'] is False

# Source consumers already share the augmented map. The exact code is unchanged
# apart from generation guards, so no candidate function is executed to check it.
consumer_lines = {}
for name in ('launch.py', 'run.py', 'cessation.py'):
    lines = (C / name).read_text().splitlines()
    consumer_lines[name] = [{'line': i, 'source': line} for i, line in enumerate(lines, 1)
                            if 'runtime-flat-pins.json' in line or 'read_paths=' in line or
                            'fence.install(' in line or 'allowed_origins=' in line or
                            'mapping_changes.append(' in line or "mapped[path]=sha(path)" in line]
    assert any('runtime-flat-pins.json' in row['source'] for row in consumer_lines[name])
assert "mapped[path]=sha(path)" in (C / 'run.py').read_text()
assert "str(Path(row['path']).resolve()) not in allowed_origins" in (C / 'cessation.py').read_text()
assert "str(Path(path).resolve()) not in allowed_origins:mapping_changes.append(path)" in (C / 'cessation.py').read_text()

baseline_package = read(B / 'execution-package-pins.json')
package = read(C / 'execution-package-pins.json')
assert all(package.get(p) == h for p, h in baseline_package.items())
assert str(C / 'execution-package-pins.json') not in package
assert set(str(C / name) for name in names if name != 'execution-package-pins.json') <= set(package)
assert all(digest(p) == h for p, h in package.items())

references = []
edges = {}
owned = {str(C / n) for n in names}
owned |= {str(O / n) for n in ('locale-provenance.json', 'elf-readiness.json', 'mapping-input-delta.json')}

def record(source, path, expected, bytes_count=None):
    assert digest(path) == expected, ('STALE_REFERENCE', source, path)
    if bytes_count is not None:
        assert Path(path).stat().st_size == bytes_count
    references.append({'source': source, 'target': path, 'sha256': expected})
    if path in owned:
        edges.setdefault(source, set()).add(path)

def refs(source, node):
    if isinstance(node, dict):
        if isinstance(node.get('path'), str) and isinstance(node.get('sha256'), str):
            record(source, node['path'], node['sha256'], node.get('bytes'))
        for value in node.values():
            refs(source, value)
    elif isinstance(node, list):
        for value in node:
            refs(source, value)

for p in sorted(owned):
    if not p.endswith('.json'):
        continue
    name = Path(p).name
    if name in ('input-pins.json', 'selected-pins.json', 'runtime-flat-pins.json', 'import-path-pins.json'):
        continue  # whole unchanged leaf graph was already reviewed; map equality established above
    data = read(p)
    if name in ('launch-pins.json', 'execution-package-pins.json'):
        for target, expected in data.items():
            record(p, target, expected)
    else:
        refs(p, data)
seen, active = set(), set()

def visit(node):
    assert node not in active, ('CIRCULAR_CURRENT_REFERENCE', node)
    if node in seen:
        return
    active.add(node)
    for target in edges.get(node, ()):
        visit(target)
    active.remove(node)
    seen.add(node)

for p in sorted(owned):
    visit(p)

expected_all_diff = []
changed, unchanged = [], []
for name in names:
    old, new = (B / name).read_text(), (C / name).read_text()
    if old == new:
        unchanged.append(name)
    else:
        changed.append(name)
        expected_all_diff.append(''.join(difflib.unified_diff(old.splitlines(True), new.splitlines(True), fromfile=str(B / name), tofile=str(C / name))))
assert (O / 'all-source-changes.patch').read_text() == ''.join(expected_all_diff)

report = {
    'status': 'PASS_SOURCE_DATA_HASH_AST_ONLY', 'candidate': str(C), 'package': ref(C / 'execution-package-pins.json'),
    'baseline_source_files_unchanged': 27, 'current_top_level_files': len(names), 'python_ast_checks': ast_checks,
    'map_checks': map_checks, 'original_suite_source_counts': counts,
    'diagnostic_test_functions': function_count, 'diagnostic_static_cases': case_count,
    'consumer_source_lines': consumer_lines, 'baseline_package_entries_retained': len(baseline_package),
    'current_package_entries': len(package), 'current_metadata_reference_count': len(references),
    'current_graph_acyclic': True, 'stale_hashes': [], 'references': references,
    'diffs_match_actual_bytes': True, 'changed_files': changed, 'unchanged_files': unchanged,
    'distinct_files_hashed': len(cache), 'whole_runtime_selected_corpus_leaf_rehash': False,
    'baseline_graph_reuse_basis': 'Exact byte equality input/selected maps; exact old-flat subset plus1 locale; original runtime graph flattened equals old runtime map; prior immutable review retained.',
    'candidate_imports_collection_tests': False, 'runtime_clock_or_admission_created': False,
    'phase_directories_present': False, 'actual_generation4_assigned': False,
    'finished_msk': dt.datetime.now(dt.timezone(dt.timedelta(hours=3))).isoformat(),
}
(O / 'static-audit.json').write_text(json.dumps(report, indent=2) + '\n')
print(json.dumps({k: report[k] for k in ('status', 'package', 'current_top_level_files', 'current_package_entries', 'current_metadata_reference_count', 'distinct_files_hashed', 'changed_files', 'unchanged_files')}, indent=2))
