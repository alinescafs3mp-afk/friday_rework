"""Finite source publication hash refresh, no execution."""
from pathlib import Path
import json,hashlib,difflib,os,resource
os.sched_setaffinity(0,{0,1});resource.setrlimit(resource.RLIMIT_AS,(2<<30,2<<30))
O=Path(__file__).resolve().parent;C=O/'candidate';F=O/'supplementary-f1'
def read(p):return json.loads(Path(p).read_text())
def sha(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def ref(p):
 p=Path(p);return {'path':str(p),'sha256':sha(p),'bytes':p.stat().st_size}
def write(p,d):
 assert Path(p).is_relative_to(O);Path(p).write_text(json.dumps(d,indent=2,ensure_ascii=False)+'\n')
m=read(C/'execution-package-pins.json')
for p in m:
 if Path(p).parent==C:m[p]=sha(p)
write(C/'execution-package-pins.json',m)
f=read(F/'source-contract.json');f['admission_fields_for_root']['source_package_sha256']=sha(C/'execution-package-pins.json');write(F/'source-contract.json',f)
r=read(O/'root-runtime-contract.json');r['source_package']=ref(C/'execution-package-pins.json');r['admission_fields']['source_package_sha256']=sha(C/'execution-package-pins.json');r['supplementary_F1_before_controls']={**f,'contract':ref(F/'source-contract.json'),'allowance':'Separate root source-qualification180; never charged/transferred as G9 runtime900.'};write(O/'root-runtime-contract.json',r)
d=read(O/'integration-delta.json')
for row in d['leaves']:
 p=Path(row['current']['path']);row['current']=ref(p)
 for key,basekey in [('delta','baseline'),('integration_delta','reviewed_replacement')]:
  if row.get(key):
   before=Path(row[basekey]['path']) if row.get(basekey) else None
   patch=''.join(difflib.unified_diff((before.read_text() if before else '').splitlines(True),p.read_text().splitlines(True),fromfile=str(before) if before else '/dev/null',tofile=str(p)))
   pp=Path(row[key]['path']);pp.write_text(patch);row[key]=ref(pp)
write(O/'integration-delta.json',d)
print(json.dumps({'package':ref(C/'execution-package-pins.json'),'root_contract':ref(O/'root-runtime-contract.json')}))
