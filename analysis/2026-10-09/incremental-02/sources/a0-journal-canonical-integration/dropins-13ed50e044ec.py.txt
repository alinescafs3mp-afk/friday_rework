"""Inert candidate library; no CLI, imports have no effects.

Only a separately reviewed, freshly bound parent may supply the explicit permit,
original clock/nonce/manager tuple and bounded native client. No permit defaults.
The stdlib filesystem implementation is exercised ONLY under the offline sandbox.
"""
import copy
import hashlib
import os
from pathlib import Path
import re
import stat

ROLES = ('outer-bound', 'dependent-bound', 'dependent-late', 'outer-negative',
         'dependent-negative', 'outer-timeout', 'dependent-timeout', 'sentinel')
BODY = b'[Unit]\nJobTimeoutSec=20s\n'
BODY_SHA256 = hashlib.sha256(BODY).hexdigest()
BASENAME = '90-frw011-job-timeout.conf'
LIVE_ROOT = '<USER_RUNTIME>/systemd/user'
OLD_PREFIX = 'frw011a0ndp20261007g2mw21'
NS = 10**9
RELOAD_ARGV = ['/usr/bin/systemctl', '--user', '--no-ask-password', 'daemon-reload']
UNIT_PATH_ARGV = ['/usr/bin/systemctl', '--user', 'show', '--property=UnitPath', '--value', '--no-pager']

class Refused(RuntimeError):
    pass


def need(ok, why):
    if not ok:
        raise Refused(why)


def pin(value):
    need(type(value) is str and re.fullmatch('[0-9a-f]{64}', value), 'SHA256/nonce required')
    return value


def identity(s):
    return {k: getattr(s, k) for k in ('st_dev', 'st_ino', 'st_mode', 'st_uid', 'st_gid', 'st_nlink')}


def file_identity(s):
    return {**identity(s), **{k: getattr(s, k) for k in ('st_size', 'st_mtime_ns', 'st_ctime_ns')}}


def dir_identity(s):
    # nlink changes when this session creates/removes its child directories.
    return {k: v for k, v in identity(s).items() if k != 'st_nlink'}


def check_dir(s, uid, private=False):
    need(stat.S_ISDIR(s.st_mode) and s.st_uid == uid and s.st_gid == uid and
         not stat.S_IMODE(s.st_mode) & 0o022 and
         (not private or stat.S_IMODE(s.st_mode) == 0o700), 'directory ownership/mode')


def open_directory(path):
    """No symlink in any component. Never create an ancestor."""
    p = Path(path)
    need(p.is_absolute() and '..' not in p.parts, 'absolute canonical directory')
    fd = os.open('/', os.O_RDONLY | os.O_DIRECTORY | os.O_CLOEXEC)
    try:
        for part in p.parts[1:]:
            nxt = os.open(part, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW | os.O_CLOEXEC,
                          dir_fd=fd)
            os.close(fd)
            fd = nxt
        return fd
    except BaseException:
        os.close(fd)
        raise


def elapsed(session, now):
    values = [now[k] - session['accepted_' + k] for k in ('monotonic_ns', 'boottime_ns')]
    need(all(type(n) is int and n >= 0 for n in values), 'clock backwards/future anchor')
    return max(values)


def admission_left(original, clock, lock_check, seconds):
    """Admit the NEXT effect under original clocks, after the blocking FD9 check.

    This cannot cancel a syscall already entered. Audit/identity/proof calls must
    finish BEFORE this last check; never reuse its answer across blocking work.
    The FD9 callback is the original ownership predicate, not a new lock/lease.
    """
    need(elapsed(original, clock()) < seconds*NS, 'original budget expired')
    need(lock_check() is True, 'original FD9 ownership unknown')
    left = seconds*NS - elapsed(original, clock())
    need(left > 0, 'original budget expired')
    return left / NS


def transport(result, argv):
    need(type(result) is dict and result.get('argv') == argv and
         result.get('returncode') == 0 and result.get('timed_out') is False and
         result.get('client_reaped') is True and result.get('output_truncated') is False and
         type(result.get('stdout')) is str and type(result.get('stderr')) is str and
         not result['stderr'] and
         len(result['stdout'].encode()) <= 65536, 'incomplete transport/reload or journal warning')
    return result['stdout']


def journal_empty(result, argv):
    # rc0 + ignored archive / ENOMEM is NOT complete historical evidence.
    need(argv[0] == '/usr/bin/journalctl', 'journal scope')
    need(not transport(result, argv).strip(), 'prior exact-name history')
    return True


def validate_permit(permit, session, names, *, runtime_root=LIVE_ROOT):
    required = {'version', 'assignment', 'generation', 'prefix', 'manifest_sha256',
                'independent_review_sha256', 'owner_grant_id', 'runtime_root',
                'runtime_root_identity', 'unit_names', 'files', 'effects',
                'first_use_basis_sha256', 'fresh_generation_attested',
                'source_image_gap_accepted_for_diagnostic_only'}
    need(type(permit) is dict and set(permit) == required, 'exact future effect permit schema')
    need(type(permit['version']) is int and permit['version'] == 1 and type(permit['generation']) is int and permit['generation'] > 0,
         'explicit future generation')
    need(permit['generation']==session['generation'], 'original generation')
    need(type(permit['assignment']) is str and permit['assignment'] == session['assignment'] and
         permit['assignment'] not in ('FRW011-A0-DEPENDENCY-WITNESS-G3', 'UNBOUND_FUTURE_ASSIGNMENT'),
         'consumed/unbound assignment refused')
    prefix = permit['prefix']
    need(type(prefix) is str and re.fullmatch('[a-z][a-z0-9]{7,47}', prefix) and prefix != OLD_PREFIX,
         'fresh explicitly selected simple prefix; no automatic choice or old-name retry')
    need(names == {r: prefix + '-' + r + '.service' for r in ROLES} and permit['unit_names'] == names,
         'exact eight externally named units')
    need(permit['manifest_sha256'] == session['manifest_sha256'], 'source manifest binding')
    for key in ('manifest_sha256', 'independent_review_sha256', 'owner_grant_id', 'first_use_basis_sha256'):
        pin(permit[key])
    pin(session['attempt_token'])
    need(permit['fresh_generation_attested'] is True and
         permit['source_image_gap_accepted_for_diagnostic_only'] is True,
         'fresh history/source-image uncertainty require explicit owner decision')
    need(permit['runtime_root'] == runtime_root and type(permit['runtime_root_identity']) is dict,
         'exact existing runtime directory pin; no mkdir parents')
    need(permit['files'] == {names[r]+'.d/'+BASENAME: BODY_SHA256 for r in ROLES}, 'exact eight bytes/paths')
    need(permit['effects'] == {
        'create_exact_eight_dropin_dirs_files': True,
        'whole_user_manager_daemon_reload_after_install': 1,
        'whole_user_manager_daemon_reload_after_verified_cleanup': 1,
        'remove_only_created_identities_after_verified_cessation': True,
        'all_effects_inside_original_120s': True,
        'all_setup_inside_original_5s': True}, 'explicit whole-manager install/cleanup reload authority')
    for key, seconds in [('submit_cutoff_ns', 35), ('driver_deadline_ns', 110), ('whole_deadline_ns', 120)]:
        need(session[key] == session['accepted_monotonic_ns'] + seconds*NS, 'original deadlines binding')
    return permit


def candidate_dirs(name):
    need(re.fullmatch('[a-z][a-z0-9]*-(?:[a-z]+-)?[a-z]+[.]service', name), 'simple exact service name')
    stem = name[:-8]
    variants = {name, 'service'}
    variants.update(stem[:i+1]+'.service' for i,c in enumerate(stem) if c == '-')
    return sorted(variants)


def parse_unit_paths(text, root):
    paths = text.strip().split()
    need(0 < len(paths) <= 64 and len(paths) == len(set(paths)) and root in paths,
         'complete bounded UnitPath containing approved runtime root')
    need(all(x.startswith('/') and '..' not in Path(x).parts and
             re.fullmatch(r'/[A-Za-z0-9_./@+-]+', x) for x in paths), 'unambiguous UnitPath')
    return paths


def assert_no_foreign_configuration(paths, names, root, installed, gate):
    """Refuse every potentially effective conf/dependency/main fragment.

    No wildcard deletion. Empty foreign *.d directories also refuse. Aliases and
    templates are unavailable because the supplied eight names are simple,
    never-used services with no main fragment before StartTransientUnit.
    Lookup root symlinks (e.g. /lib) are read-only realpath resolved; roots and
    applicable entries are rechecked after reload and before every submission.
    """
    checked = 0
    for base in paths:
        gate()
        for name in names.values():
            main = Path(base)/name
            need(not os.path.lexists(main), 'preexisting main fragment/mask/alias')
        variants = sorted({v for name in names.values() for v in candidate_dirs(name)})
        for variant in variants:
            for suffix in ('.d', '.wants', '.requires', '.upholds'):
                checked += 1
                need(checked <= 16384, 'configuration inspection budget')
                gate()
                p = Path(base)/(variant+suffix)
                ours = base == root and variant in names.values() and suffix == '.d'
                if installed and ours:
                    continue  # strict descriptor/content checks below, not a path exemption alone
                need(not os.path.lexists(p), 'foreign applicable configuration: '+str(p))
    return checked


class Session:
    """One attempted install/reload; irreversible uncertainty, no retries.

    Retains directory FDs until exact verified cleanup; never closes FD9. Every
    mutation has a durable intent/receipt. Final admission/cleanup intent records
    are not success receipts: their return still requires the original-time gate.
    A crash/unknown record is reconciliation
    work, NOT permission to rediscover or delete by basename.
    """
    def __init__(self, permit, original, names, clock, audit, lock_check, manager_check,
                 *, runtime_root=LIVE_ROOT, uid=1000):
        validate_permit(permit, original, names, runtime_root=runtime_root)
        self.permit, self.original, self.names = copy.deepcopy(permit), copy.deepcopy(original), copy.deepcopy(names)
        self.clock, self.audit, self.lock_check, self.manager_check = clock, audit, lock_check, manager_check
        self.root, self.uid = runtime_root, uid
        self.rootfd = None
        self.created = {}
        self.failed = False
        self.reload_uncertain = False
        self.reload_count = 0
        self.cleanup_reload_count = 0
        self.installed = self.admitted = self.cleaned = False
        self.cleanup_attempted = False
        self.paths = None
        self.sequence = 0
        self.root_pin = copy.deepcopy(permit['runtime_root_identity'])
        self.events = []

    def record(self, event, **detail):
        item = {'sequence': self.sequence, 'event': event, 'nonce': self.original['attempt_token'],
                'manifest_sha256': self.original['manifest_sha256'], 'at': self.clock(), **detail}
        self.sequence += 1
        self.events.append(item)
        try:
            self.audit(item)  # caller uses exclusive creation + fsync file and directory
        except BaseException:
            self.failed = True
            raise

    def gate(self, seconds=5, *, allow_failed=False):
        need(allow_failed or not self.failed, 'session latched closed')
        return admission_left(self.original, self.clock, self.lock_check, seconds)

    def verify_root(self):
        need(self.rootfd is not None, 'runtime root not opened')
        check_dir(os.fstat(self.rootfd), self.uid)
        need(dir_identity(os.fstat(self.rootfd)) == self.root_pin, 'runtime root FD changed')
        fd = open_directory(self.root)
        try:
            need(dir_identity(os.fstat(fd)) == self.root_pin, 'runtime root path replaced')
        finally:
            os.close(fd)

    def begin(self):
        self.gate()
        need(self.rootfd is None and not self.created, 'single setup attempt')
        self.rootfd = open_directory(self.root)
        self.verify_root()
        self.record('setup-begin', root=self.root, root_identity=self.root_pin)

    def verify_owned(self, *, seconds=5):
        self.gate(seconds)
        self.verify_root()
        for role, entry in self.created.items():
            self.gate(seconds)
            fd, dirname = entry['fd'], self.names[role]+'.d'
            need(entry.get('file_complete') is True, 'partial/unknown file creation')
            need(dir_identity(os.fstat(fd)) == entry['directory_identity'] and
                 dir_identity(os.stat(dirname, dir_fd=self.rootfd, follow_symlinks=False)) == entry['directory_identity'],
                 'owned directory replaced')
            need(os.listdir(fd) == [BASENAME], 'extra/unknown drop-in directory content')
            f = os.open(BASENAME, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK | os.O_CLOEXEC, dir_fd=fd)
            try:
                before = os.fstat(f)
                need(file_identity(before) == entry['file_identity'] and stat.S_ISREG(before.st_mode) and
                     before.st_nlink == 1 and before.st_size == len(BODY), 'owned file identity/size changed')
                data = os.read(f, len(BODY)+1)
                need(file_identity(os.fstat(f)) == file_identity(before) and data == BODY and
                     hashlib.sha256(data).hexdigest() == entry['sha256'], 'owned file bytes changed')
                need(file_identity(os.stat(BASENAME, dir_fd=fd, follow_symlinks=False)) == entry['file_identity'],
                     'owned file path replacement during read')
            finally:
                os.close(f)
        self.gate(seconds)

    def install(self):
        try:
            self.gate()
            need(self.rootfd is not None and not self.installed and not self.created, 'single install after begin')
            for role in ROLES:
                self.gate()
                self.verify_root()
                dirname = self.names[role]+'.d'
                self.record('mkdir-intent', role=role, relative=dirname)
                self.gate()
                os.mkdir(dirname, 0o700, dir_fd=self.rootfd)  # EEXIST refuses; no adoption
                fd = os.open(dirname, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW | os.O_CLOEXEC,
                             dir_fd=self.rootfd)
                s = os.fstat(fd)
                check_dir(s, self.uid, private=True)
                self.created[role] = {'fd': fd, 'directory_identity': dir_identity(s), 'file_complete': False}
                self.record('mkdir-created', role=role, identity=dir_identity(s))
                self.gate()
                self.record('file-create-intent', role=role, relative=dirname+'/'+BASENAME, sha256=BODY_SHA256)
                self.gate()
                f = os.open(BASENAME, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW | os.O_CLOEXEC,
                            0o600, dir_fd=fd)
                try:
                    self.created[role]['initial_file_identity'] = file_identity(os.fstat(f))
                    self.record('file-created-empty', role=role, identity=self.created[role]['initial_file_identity'])
                    self.gate()
                    need(os.write(f, BODY) == len(BODY), 'short drop-in write')
                    self.gate()
                    os.fsync(f)
                    s = os.fstat(f)
                    need(stat.S_ISREG(s.st_mode) and s.st_uid == s.st_gid == self.uid and
                         stat.S_IMODE(s.st_mode) == 0o600 and s.st_nlink == 1 and s.st_size == len(BODY),
                         'created file metadata')
                    self.created[role].update(file_identity=file_identity(s), sha256=BODY_SHA256, file_complete=True)
                finally:
                    os.close(f)
                self.gate()
                os.fsync(fd)
                self.gate()
                os.fsync(self.rootfd)
                self.record('file-complete', role=role, identity=self.created[role]['file_identity'], sha256=BODY_SHA256)
            self.verify_owned()
            self.record('exact-eight-installed')
            self.gate()
            self.installed = True
        except BaseException:
            self.failed = True
            raise

    def setup(self, bounded_call):
        try:
            self.begin()
            self.gate()
            self.paths = parse_unit_paths(transport(bounded_call(UNIT_PATH_ARGV, self.left(5)), UNIT_PATH_ARGV), self.root)
            assert_no_foreign_configuration(self.paths, self.names, self.root, False, self.gate)
            self.install()
            self.gate()
            need(self.manager_check() == self.original['user_manager'], 'manager changed before reload')
            self.record('whole-user-manager-reload-intent', argv=RELOAD_ARGV)
            timeout = self.left(5)
            # No unknown effect before the transport is actually attempted.
            self.reload_uncertain = True
            self.reload_count += 1
            result = bounded_call(RELOAD_ARGV, timeout, expected='reload')
            self.record('whole-user-manager-reload-return', result=result)
            need(not transport(result, RELOAD_ARGV).strip(), 'unexpected reload output')
            self.gate()
            need(self.manager_check() == self.original['user_manager'], 'manager changed across reload')
            paths = parse_unit_paths(transport(bounded_call(UNIT_PATH_ARGV, self.left(5)), UNIT_PATH_ARGV), self.root)
            need(paths == self.paths, 'manager search path changed during reload')
            assert_no_foreign_configuration(paths, self.names, self.root, True, self.gate)
            self.verify_owned()
            self.gate()
            self.record('pre-enqueue-source-route-admission-intent', proof='completed normal reload; exact source route, not mapped-image equivalence')
            self.gate()
            self.reload_uncertain = False
            self.admitted = True
            return self.receipt_binding()
        except BaseException:
            self.failed = True
            raise

    def left(self, seconds):
        return min(1.0, self.gate(seconds))

    def receipt_binding(self):
        self.gate()
        need(self.admitted and not self.failed and not self.reload_uncertain and len(self.created) == 8,
             'no pre-enqueue admission')
        return {'version': 1, 'nonce': self.original['attempt_token'],
                'manifest_sha256': self.original['manifest_sha256'],
                'manager': self.original['user_manager'], 'root': self.root,
                'root_identity': self.root_pin, 'unit_path': self.paths,
                'files': {self.names[r]+'.d/'+BASENAME: {k:v for k,v in item.items() if k != 'fd'}
                          for r,item in self.created.items()},
                'reload_completed': True, 'setup_finished': self.clock(),
                'source_supported_not_exact_image_proven': True}

    def cleanup(self, cessation, bounded_call):
        """No cleanup from a report alone: parent must independently verify cessation.

        The explicit callback must rerun the original exact actor/job/cgroup and
        witness/client predicates, under the existing sole-writer ownership contract.
        It is a review/authority boundary, not a supplied permissive fallback.
        """
        try:
            self.gate(120)
            need(not self.cleanup_attempted and self.admitted and not self.reload_uncertain,
                 'cleanup refused for partial/unknown setup; separate reconciliation required')
            self.cleanup_attempted = True
            self.parent_decision(cessation, 'before_cleanup')
            self.verify_owned(seconds=120)
            self.record('cleanup-begin')
            for role in reversed(ROLES):
                self.gate(120)
                self.verify_owned(seconds=120)
                item = self.created[role]
                dirname = self.names[role]+'.d'
                self.record('unlink-exact-intent', role=role, identity=item['file_identity'], sha256=BODY_SHA256)
                # Existing task contract assigns sole ownership of these exact paths.
                # Preserve every observed concurrent change; no adoption or wildcard
                # removal. This is not an adversarial arbitrary-same-UID sandbox.
                need(file_identity(os.stat(BASENAME, dir_fd=item['fd'], follow_symlinks=False)) == item['file_identity'],
                     'file replaced before unlink')
                self.gate(120)
                os.unlink(BASENAME, dir_fd=item['fd'])
                item['file_removed'] = True
                self.gate(120)
                os.fsync(item['fd'])
                self.record('rmdir-exact-intent', role=role, identity=item['directory_identity'])
                need(os.listdir(item['fd']) == [], 'directory gained foreign contents')
                need(dir_identity(os.stat(dirname, dir_fd=self.rootfd, follow_symlinks=False)) == item['directory_identity'],
                     'directory replaced before rmdir')
                self.gate(120)
                os.rmdir(dirname, dir_fd=self.rootfd)
                item['directory_removed'] = True
                self.gate(120)
                os.fsync(self.rootfd)
                os.close(item['fd'])
                del self.created[role]
                self.record('exact-file-directory-removed', role=role)
            self.gate(120)
            self.record('whole-user-manager-cleanup-reload-intent', argv=RELOAD_ARGV)
            timeout = self.left(120)
            self.reload_uncertain = True
            self.cleanup_reload_count += 1
            result = bounded_call(RELOAD_ARGV, timeout, expected='reload')
            self.record('whole-user-manager-cleanup-reload-return', result=result)
            need(not transport(result,RELOAD_ARGV).strip(), 'cleanup reload output')
            self.gate(120)
            self.parent_decision(cessation, 'after_cleanup_reload')
            self.record('cleanup-finalization-intent-parent-FD9-still-held')
            self.gate(120)
            os.close(self.rootfd)
            self.rootfd = None
            self.gate(120)
            self.reload_uncertain = False
            self.cleaned = True
            return {'status':'DROPINS_CLEANED', 'parent_fd9':'STILL_HELD; separate owner release decision'}
        except BaseException:
            self.failed = True
            raise

    def parent_decision(self, provider, phase):
        """Consume the existing parent's independent cessation-decision interface.

        Provider receives phase and current original-clock sample, and returns
        (decision, independently_pinned_sha256). It is the parent integration
        authority, never driver JSON or a boolean success stub in native work.
        Old G3 decision cannot satisfy this session's nonce/source/name binding.
        """
        import json
        requested = self.clock()
        decision, independent_pin = provider(phase, copy.deepcopy(requested))
        need(type(decision) is dict, 'independent parent decision object')
        encoded = (json.dumps(decision,sort_keys=True,separators=(',',':'))+'\n').encode()
        need(hashlib.sha256(encoded).hexdigest() == pin(independent_pin), 'independent parent decision pin')
        need(decision.get('status') == 'PARENT_CESSATION_CONFIRMED_SAFEGUARD_STOP_RETAINED' and
             decision.get('retry_authorized') is False and decision.get('phase') == phase,
             'parent decision scope/phase')
        need(decision.get('session_binding') == {
            'assignment':self.original['assignment'],'generation':self.original['generation'],'boot_id':self.original['boot_id'],
            'attempt_token':self.original['attempt_token'],
            'manifest_sha256':self.original['manifest_sha256'],
            'accepted_monotonic_ns':self.original['accepted_monotonic_ns'],
            'accepted_boottime_ns':self.original['accepted_boottime_ns'],
            'unit_names':self.names,'user_manager':self.original['user_manager']},
            'fresh parent cessation session/name/manager binding')
        need(decision.get('parent') == self.original['canonical_lock_holder'], 'original independent parent FD9 identity')
        need(decision.get('checks') == {
            'own_exact_units_jobs_cgroups_absent':True,
            'all_original_actor_identities_ceased':True,
            'all_native_clients_driver_witness_reaped':True,
            'no_pending_or_uncertain_submission':True,
            'no_more_starts':True,'original_FD9_still_held':True},
            'complete independent parent cessation, never child-report-only')
        need(type(decision.get('own_observation_refs')) is list and 1 <= len(decision['own_observation_refs']) <= 32,
             'independent parent evidence references')
        for ref in decision['own_observation_refs']:
            need(type(ref) is dict and set(ref)=={'path','sha256'} and
                 type(ref['path']) is str and ref['path'].startswith('/'), 'parent evidence ref')
            pin(ref['sha256'])
        pin(decision.get('original_native_report_sha256'))
        now=self.clock()
        for axis in ('monotonic_ns','boottime_ns'):
            need(type(decision.get('observed_at',{}).get(axis)) is int and
                 requested[axis] <= decision['observed_at'][axis] <= now[axis],
                 'parent decision predates requested independent verification')
        self.gate(120)
        self.record('independent-parent-cessation-decision', phase=phase, sha256=independent_pin,
                    evidence_refs=decision['own_observation_refs'])
        self.gate(120)

    def stop_state(self):
        return {'status': 'STOP_UNCONFIRMED', 'parent_must_retain_lock': True,
                'no_more_starts': True, 'failed': self.failed, 'reload_uncertain': self.reload_uncertain,
                'cleanup_attempted': self.cleanup_attempted, 'created_roles': list(self.created),
                'reconciliation':'exact durable intents/identities only; no retry/adoption/wildcard cleanup'}


def verify_receipt_binding(binding, original, names):
    need(type(binding) is dict and binding.get('version') == 1 and binding.get('reload_completed') is True and
         binding.get('nonce') == original['attempt_token'] and
         binding.get('manifest_sha256') == original['manifest_sha256'] and
         binding.get('manager') == original['user_manager'] and binding.get('root') == LIVE_ROOT and
         binding.get('source_supported_not_exact_image_proven') is True,
         'drop-in receipt binding')
    need(elapsed(original,binding['setup_finished']) < 5*NS, 'setup after original admission')
    need(set(binding['files']) == {names[r]+'.d/'+BASENAME for r in ROLES}, 'eight receipt files')
    for item in binding['files'].values():
        need(item.get('sha256') == BODY_SHA256 and item.get('file_complete') is True, 'receipt bytes')
    return binding


def before_submit(binding, original, names, clock, lock_check):
    """Repeat exact owned bytes before every start, including GC reconstruction.

    Generic configuration is checked again; transient main fragments from the
    SAME attempt are permitted only because the original driver separately
    validates its current identities/properties and absence/reconstruction gate.
    """
    verify_receipt_binding(binding, original, names)
    admission_left(original, clock, lock_check, 35)
    rootfd = open_directory(binding['root'])
    opened = []
    try:
        need(dir_identity(os.fstat(rootfd)) == binding['root_identity'], 'root replaced before submit')
        for role in ROLES:
            need(elapsed(original,clock()) < 35*NS, 'submission inspection deadline')
            rel = names[role]+'.d/'+BASENAME
            item = binding['files'][rel]
            fd = os.open(names[role]+'.d', os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW|os.O_CLOEXEC, dir_fd=rootfd)
            opened.append(fd)
            need(dir_identity(os.fstat(fd)) == item['directory_identity'] and os.listdir(fd) == [BASENAME],
                 'pre-submit directory changed')
            f = os.open(BASENAME,os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK|os.O_CLOEXEC,dir_fd=fd)
            try:
                need(file_identity(os.fstat(f)) == item['file_identity'], 'pre-submit file identity changed')
                raw = os.read(f,len(BODY)+1)
                need(raw == BODY and file_identity(os.fstat(f)) == item['file_identity'] and
                     file_identity(os.stat(BASENAME,dir_fd=fd,follow_symlinks=False)) == item['file_identity'],
                     'pre-submit bytes/path changed')
            finally:
                os.close(f)
        for base in binding['unit_path']:
            for variant in sorted({v for n in names.values() for v in candidate_dirs(n)}):
                for suffix in ('.d','.wants','.requires','.upholds'):
                    need(elapsed(original,clock()) < 35*NS, 'pre-submit foreign check deadline')
                    if base == binding['root'] and variant in names.values() and suffix == '.d':
                        continue
                    need(not os.path.lexists(Path(base)/(variant+suffix)), 'new foreign effective configuration')
        finalroot = open_directory(binding['root'])
        try:
            need(dir_identity(os.fstat(finalroot)) == binding['root_identity'], 'root replaced during pre-submit inspection')
        finally:
            os.close(finalroot)
    finally:
        for fd in opened:
            os.close(fd)
        os.close(rootfd)
    admission_left(original, clock, lock_check, 35)
