#!/usr/bin/python3
# Entry bootstrap deliberately precedes every project import. -B alone is insufficient.
if __name__ == '__main__':
    import sys
    if not (sys.flags.isolated and sys.flags.no_site):
        raise SystemExit('REFUSED: require python -I -S -B; no project imports performed')
    import os, json, hashlib, types
    _root = os.path.dirname(os.path.realpath(__file__))
    _manifest_raw = open(os.path.join(_root, 'manifest.json'), 'rb').read()
    if '--describe' in sys.argv and os.path.basename(__file__) == 'driver.py':
        print(json.dumps({'status': 'SOURCE_ONLY_NATIVE_NOT_RUN', 'manifest_sha256': hashlib.sha256(_manifest_raw).hexdigest()}))
        raise SystemExit(0)
    _driver = os.path.basename(__file__) == 'driver.py'
    if _driver:
        try:
            _receipt_path = sys.argv[sys.argv.index('--receipt') + 1]
            _receipt_sha = sys.argv[sys.argv.index('--receipt-sha256') + 1]
        except (ValueError, IndexError):
            raise SystemExit('REFUSED: separate pinned receipt required before project imports')
        if _receipt_path != os.path.join(_root, 'authorization.json'):
            raise SystemExit('REFUSED: receipt path')
    else:
        _receipt_path = os.path.join(_root, 'native-evidence', 'receipt.json')
    _fd = os.open(_receipt_path, os.O_RDONLY | os.O_NOFOLLOW)
    try:
        _st = os.fstat(_fd)
        if _st.st_uid != os.getuid() or _st.st_nlink != 1 or (_st.st_mode & 0o177777) != 0o100600:
            raise SystemExit('REFUSED: receipt metadata')
        _raw = os.read(_fd, 1048576)
    finally:
        os.close(_fd)
    if _driver and hashlib.sha256(_raw).hexdigest() != _receipt_sha:
        raise SystemExit('REFUSED: receipt bytes hash')
    _r = json.loads(_raw)
    if _r.get('execution_authorized') is not True or _r.get('manifest_sha256') != hashlib.sha256(_manifest_raw).hexdigest():
        raise SystemExit('REFUSED: source manifest authority')
    _manifest = json.loads(_manifest_raw)
    _verified = {}
    for _name, _pin in _manifest['files'].items():
        if _name not in ('common.py', 'payload.py', 'observer.py', 'driver.py', 'recipe.json', 'witness.py', 'owner_launcher.py', 'parent_integration.py', 'dropins.py'):
            raise SystemExit('REFUSED: source allowlist')
        _bytes = open(os.path.join(_root, _name), 'rb').read()
        if hashlib.sha256(_bytes).hexdigest() != _pin:
            raise SystemExit('REFUSED: source drift')
        _verified[_name] = _bytes
    if set(_verified) != {'common.py', 'payload.py', 'observer.py', 'driver.py', 'recipe.json', 'witness.py', 'owner_launcher.py', 'parent_integration.py', 'dropins.py'}:
        raise SystemExit('REFUSED: missing source pin')
    # Compile precisely the bytes just verified. No SourceFileLoader/import of
    # project paths, no pyc lookup, no deletion/ignoring a discovered cache file.
    for _module_name in ('witness', 'dropins', 'common', 'payload', 'observer'):
        _name = _module_name + '.py'
        _module = types.ModuleType(_module_name)
        _module.__file__ = os.path.join(_root, _name)
        sys.modules[_module_name] = _module
        exec(compile(_verified[_name], _module.__file__, 'exec'), _module.__dict__)

"""Bounded native ExecStopPost recorder plus pure evidence evaluator.

Stop-post executes after main termination, before unit completion/GC. The
native latched ActiveEnterTimestamp catches active pulses missed by polling.
The hook is evidence collection only: no start/stop/reset or READY notification.
"""
import os
import re
import signal
import time
import sys
from common import (ROLES, Refused, emit, identity, own_cgroup, parse_show,
                    receipt, remaining, require, show_argv)


def late_ready_case(outer, dependent, sent, bound):
    """Inputs must already be invocation-pinned by the driver."""
    if not outer or not dependent or not sent:
        return {'status': 'INCONCLUSIVE', 'reason': 'missing native terminal or READY-send evidence'}
    try:
        op, dp = outer['properties'], dependent['properties']
        end = int(op['ExecMainExitTimestampMonotonic']) * 1000
        active = int(dp['ActiveEnterTimestampMonotonic']) * 1000
        dep_end = int(dp['ExecMainExitTimestampMonotonic']) * 1000
        if end <= 0 or dep_end <= 0 or sent['send_started_monotonic_ns'] <= end:
            return {'status': 'INCONCLUSIVE', 'reason': 'outer native death not established before READY send'}
        if bound:
            return {'status': 'PASS' if active == 0 else 'FAIL',
                    'reason': 'native latched active-enter timestamp (not polling)',
                    'active_enter_ns': active, 'outer_exit_ns': end, 'ready_sent_ns': sent['monotonic_ns']}
        return {'status': 'PASS' if end < active <= dep_end and active + 1000 >= sent['send_started_monotonic_ns'] else 'FAIL',
                'reason': 'unbound native accepted-active positive control', 'active_enter_ns': active,
                'outer_exit_ns': end, 'ready_sent_ns': sent['monotonic_ns']}
    except (KeyError, ValueError, TypeError):
        return {'status': 'INCONCLUSIVE', 'reason': 'incomplete native timestamps'}


def main():
    require(len(sys.argv) == 3 and sys.argv[1] == 'stop-post' and sys.argv[2] in ROLES, 'hook argv')
    role = sys.argv[2]
    r = receipt()
    invocation = os.environ.get('INVOCATION_ID', '')
    require(re.fullmatch('[0-9a-f]{32}', invocation) is not None, 'hook invocation')
    require(own_cgroup(role, identity(os.getpid())), 'hook cgroup')
    # Independent process alarm bounds the hook, including local file work.
    # Default SIGALRM would orphan a client: raise instead so call() reaps it.
    def alarm(signum, frame):
        raise TimeoutError('stop-post local alarm')
    signal.signal(signal.SIGALRM, alarm)
    signal.setitimer(signal.ITIMER_REAL, min(1.5, remaining(r, 109)))
    emit(role, 'stop-post', {'service_result':os.environ.get('SERVICE_RESULT'),
         'exit_code':os.environ.get('EXIT_CODE'),'exit_status':os.environ.get('EXIT_STATUS'),
         'properties':{},'native_capture_valid':False,
         'native_capture_owner':'ORIGINAL_PARENT_ONLY'})
    # No native Popen or request/reply channel in a payload/hook. Parent misses
    # this finite window -> INCONCLUSIVE, never a fabricated native timestamp.
    time.sleep(max(0,min(.75,remaining(r,108))))
    signal.setitimer(signal.ITIMER_REAL, 0)
    return 0


if __name__ == '__main__':
    try:
        sys.exit(main())
    except (Refused, OSError, ValueError, KeyError, TimeoutError):
        sys.exit(1)  # missing hook evidence must be INCONCLUSIVE, never fabricated
