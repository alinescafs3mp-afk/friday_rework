def command(session, reviewed_manifest_sha256, reviewed_witness_sha256):
    """Return the complete owner reviewable argv, with exact hashes and anchors.

    The caller must supply independently reviewed pins, not hashes just read
    from an untrusted receipt. Root compiles only the exact verified bytes.
    The code string itself is what the owner approves through normal sudo.
    """
    validate_session(session, reviewed_manifest_sha256,
                     {k: session['accepted_'+k] for k in ('monotonic_ns', 'boottime_ns')})
    need(len(reviewed_witness_sha256) == 64, 'reviewed witness pin required')
    loader = (BOOTSTRAP_PREFIX % session) + BOOTSTRAP_GUARD + (BOOTSTRAP_SUFFIX % (
        str(ROOT/'manifest.json'), reviewed_manifest_sha256, reviewed_manifest_sha256,
        reviewed_witness_sha256, str(ROOT/'witness.py'), str(ROOT/'witness.py'),
        reviewed_manifest_sha256))
    return ['/usr/bin/sudo', '-n', '--', '/usr/bin/python3', '-I', '-S', '-B', '-c', loader]


class ParentWitness:
    def __init__(self, process, client):
        from common import identity
        self.process, self.client = process, client
        self.owner_pid=os.getpid()
        self.identity=None
        self.finished=None
        self.producer_identity=None
        try: self.identity=identity(process.pid)
        except OSError: pass  # owned Popen retained; no success on missing identity

    def preflight(self):
        observation = self.client.observe()
        from common import identity
        self.producer_identity=identity(self.client.producer_pid)
        return self.client.pinned, self.client.receipt_binding(), observation

    def finish(self):
        # Close once after all observation replies; EOF makes the existing root
        # reader return normally. No root kill, handoff, extra sample or FD9 copy.
        if self.finished is not None: return self.finished
        from common import remaining, within, clocks
        self.client.channel.close()
        budget=min(.5,remaining(self.client.s,22))
        try:
            need(budget>0 and not self.client.failed,'witness phase exceeded/unknown')
            code=self.process.wait(timeout=budget)
            need(code==0 and self.identity is not None and within(self.client.s,22),
                 'witness failed/late reaping')
            self.finished={'state':'WITNESS_REAPED','returncode':code,'identity':self.identity,
                           'producer_identity':self.producer_identity,'finished_at':clocks(),'cessation_of_native_payloads':'NOT_ESTABLISHED_BY_WITNESS_EXIT'}
        except BaseException as error:
            self.finished={'state':'STOP_UNCONFIRMED','parent_must_retain_lock':True,
                           'reason':type(error).__name__+': '+str(error)}
        return self.finished


def launch(session, reviewed_manifest_sha256, reviewed_witness_sha256, owner_grant):
    need(os.getresuid() == (1000,1000,1000), 'existing UID1000 parent required')
    need(owner_grant == {'separate_owner_privileged_execution': True,
                         'assignment': ASSIGNMENT, 'manifest_sha256': reviewed_manifest_sha256,
                         'witness_sha256': reviewed_witness_sha256},
         'separate exact owner authorization required')
    # Existing parent owns validation/acquisition of canonical FD9. This helper
    # neither acquires nor releases it, and never passes it to privileged code.
    need(os.fstat(9).st_ino == 9044963, 'original canonical FD9 required')
    for fd in (FD,):
        try:
            os.fstat(fd)
        except OSError:
            pass
        else:
            raise RuntimeError('reserved inherited witness FD occupied')
    argv = command(session, reviewed_manifest_sha256, reviewed_witness_sha256)
    left, right = socket.socketpair(socket.AF_UNIX, socket.SOCK_SEQPACKET)
    copies = []
    try:
        # Move via high private copies, avoiding descriptor alias/clobber races.
        for s in (left,right):
            copies.append(fcntl.fcntl(s.fileno(), fcntl.F_DUPFD_CLOEXEC, 20))
    except BaseException:
        for fd in copies:
            os.close(fd)
        raise
    finally:
        left.close(); right.close()
    client_fd = root_fd = None
    try:
        os.dup2(copies[0], FD, inheritable=False)
        client_fd = FD
        # Root endpoint is passed as STDIN. No sudo close-from policy override.
        root_fd = os.dup(copies[1])
    except BaseException:
        if client_fd is not None:
            os.close(client_fd)
        if root_fd is not None:
            os.close(root_fd)
        raise
    finally:
        for fd in copies:
            os.close(fd)
    sock = inherited(FD)
    # Enable receipt of credentials on the root endpoint BEFORE any queued send.
    root_sock = inherited(root_fd)
    try:
        child = subprocess.Popen(argv, stdin=root_sock, stdout=subprocess.DEVNULL,
                                 stderr=subprocess.DEVNULL, close_fds=True,
                                 env={'PATH':'/usr/bin:/bin','LC_ALL':'C'})
    except BaseException:
        sock.close()
        raise
    finally:
        root_sock.close()
    return ParentWitness(child, Client(session, Channel(sock)))
