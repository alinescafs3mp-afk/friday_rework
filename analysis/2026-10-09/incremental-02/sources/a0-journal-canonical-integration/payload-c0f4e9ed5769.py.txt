#!/usr/bin/python3
# Entry bootstrap deliberately precedes every project import. -B alone is insufficient.
if __name__ == '__main__':
    import sys
    if not (sys.flags.isolated and sys.flags.no_site):
        raise SystemExit('REFUSED: require python -I -S -B; no project imports performed')
    import os, json, hashlib, types
    _root = os.path.dirname(os.path.realpath(__file__))
    _manifest_raw = open(os.path.join(_root, 'manifest.json'), 'rb').read()
    if '--describe' in sys.argv and os.path.basename(__file__) == 'driver.py':
        print(json.dumps({'status': 'SOURCE_ONLY_NATIVE_NOT_RUN', 'manifest_sha256': hashlib.sha256(_manifest_raw).hexdigest()}))
        raise SystemExit(0)
    _driver = os.path.basename(__file__) == 'driver.py'
    if _driver:
        try:
            _receipt_path = sys.argv[sys.argv.index('--receipt') + 1]
            _receipt_sha = sys.argv[sys.argv.index('--receipt-sha256') + 1]
        except (ValueError, IndexError):
            raise SystemExit('REFUSED: separate pinned receipt required before project imports')
        if _receipt_path != os.path.join(_root, 'authorization.json'):
            raise SystemExit('REFUSED: receipt path')
    else:
        _receipt_path = os.path.join(_root, 'native-evidence', 'receipt.json')
    _fd = os.open(_receipt_path, os.O_RDONLY | os.O_NOFOLLOW)
    try:
        _st = os.fstat(_fd)
        if _st.st_uid != os.getuid() or _st.st_nlink != 1 or (_st.st_mode & 0o177777) != 0o100600:
            raise SystemExit('REFUSED: receipt metadata')
        _raw = os.read(_fd, 1048576)
    finally:
        os.close(_fd)
    if _driver and hashlib.sha256(_raw).hexdigest() != _receipt_sha:
        raise SystemExit('REFUSED: receipt bytes hash')
    _r = json.loads(_raw)
    if _r.get('execution_authorized') is not True or _r.get('manifest_sha256') != hashlib.sha256(_manifest_raw).hexdigest():
        raise SystemExit('REFUSED: source manifest authority')
    _manifest = json.loads(_manifest_raw)
    _verified = {}
    for _name, _pin in _manifest['files'].items():
        if _name not in ('common.py', 'payload.py', 'observer.py', 'driver.py', 'recipe.json', 'witness.py', 'owner_launcher.py', 'parent_integration.py', 'dropins.py'):
            raise SystemExit('REFUSED: source allowlist')
        _bytes = open(os.path.join(_root, _name), 'rb').read()
        if hashlib.sha256(_bytes).hexdigest() != _pin:
            raise SystemExit('REFUSED: source drift')
        _verified[_name] = _bytes
    if set(_verified) != {'common.py', 'payload.py', 'observer.py', 'driver.py', 'recipe.json', 'witness.py', 'owner_launcher.py', 'parent_integration.py', 'dropins.py'}:
        raise SystemExit('REFUSED: missing source pin')
    # Compile precisely the bytes just verified. No SourceFileLoader/import of
    # project paths, no pyc lookup, no deletion/ignoring a discovered cache file.
    for _module_name in ('witness', 'dropins', 'common', 'payload', 'observer'):
        _name = _module_name + '.py'
        _module = types.ModuleType(_module_name)
        _module.__file__ = os.path.join(_root, _name)
        sys.modules[_module_name] = _module
        exec(compile(_verified[_name], _module.__file__, 'exec'), _module.__dict__)

"""Derived harmless fixture; original pinned prep payload remains untouched."""
import os
import re
import signal
import socket
import sys
import time
from common import EVIDENCE, NAMES, ROLES, Refused, emit, identity, own_cgroup, read_json, receipt, require, within, write_once, private_bytes


def consume_once(directory, role, expected):
    path = directory / (role + '.permit')
    try:
        raw = private_bytes(path)
    except FileNotFoundError:
        return False
    if raw != expected:
        return False
    try:
        os.unlink(path)
    except FileNotFoundError:
        return False
    return True


def token(r, role):
    return (r['attempt_token'] + ':' + role + '\n').encode()


def admit(mode, role, r, invocation, *, directory=EVIDENCE, now=None):
    require(within(r, 35, now), 'late payload admission')
    require(re.fullmatch('[0-9a-f]{32}', invocation or '') is not None, 'invocation')
    if mode == 'condition':
        if not consume_once(directory, role, token(r, role)):
            return False
        write_once(directory / (role + '.admission.json'),
                   {'invocation_id': invocation, 'attempt_token': r['attempt_token']})
        return True
    admission = read_json(directory / (role + '.admission.json'))
    require(admission == {'invocation_id': invocation, 'attempt_token': r['attempt_token']}, 'condition provenance')
    # Immutable second once-gate prevents accidental second main admission too.
    require(within(r, 35, now), 'late main admission')
    write_once(directory / (role + '.main-once'), invocation.encode())
    return True


def ready(role):
    address = os.environ['NOTIFY_SOCKET']
    if address.startswith('@'):
        address = '\0' + address[1:]
    with socket.socket(socket.AF_UNIX, socket.SOCK_DGRAM) as sock:
        try:
            before = time.monotonic_ns()
            sent = sock.sendto(b'READY=1', address)
            after = time.monotonic_ns()
            emit(role, 'ready-sent', {'bytes': sent, 'accepted_by_manager': 'UNPROVEN',
                                      'send_started_monotonic_ns': before, 'send_finished_monotonic_ns': after})
        except OSError as error:
            emit(role, 'ready-error', {'errno': error.errno})


def descendant(role):
    child = os.fork()
    if child == 0:
        os.setsid()
        signal.signal(signal.SIGTERM, signal.SIG_IGN)
        emit(role, 'child')
        time.sleep(110)
        os._exit(0)
    emit(role, 'fork', {'child_pid': child})
    return child


def main():
    require(len(sys.argv) == 3 and sys.argv[1] in ('condition', 'payload'), 'argv')
    mode, role = sys.argv[1:]
    require(role in ROLES, 'role')
    r = receipt()
    require(own_cgroup(role, identity(os.getpid())), 'foreign cgroup')
    accepted = admit(mode, role, r, os.environ.get('INVOCATION_ID'))
    if mode == 'condition':
        emit(role, 'condition-accepted' if accepted else 'condition-skipped')
        return 0 if accepted else 1
    emit(role, 'entered')
    if role in ('outer-bound', 'outer-negative'):
        ready(role)
        while not (EVIDENCE / (role + '.exit-now')).exists():
            time.sleep(0.02)
        emit(role, 'exiting')
        return 0
    signal.signal(signal.SIGTERM, signal.SIG_IGN)
    child = None
    if role in ('dependent-bound', 'dependent-negative'):
        outer = 'outer-bound' if role == 'dependent-bound' else 'outer-negative'
        while not tuple(EVIDENCE.glob(outer + '.exiting.*.json')):
            time.sleep(0.02)
        time.sleep(0.65)
        ready(role)
    else:
        if role in ('outer-timeout', 'dependent-timeout'):
            child = descendant(role)
        ready(role)
    # These intentionally TERM-ignoring processes are reaped by their native
    # owner after native cgroup SIGKILL. No caller wait/kill claims ownership.
    time.sleep(110)
    if child is not None:
        os.waitpid(child, 0)
    return 0


if __name__ == '__main__':
    try:
        sys.exit(main())
    except (Refused, OSError, ValueError, KeyError):
        sys.exit(1)  # denied condition is a skip, no payload retry
