def leaf_parent(tmp_path):
    def parent(path,uid):
        assert Path(path).parent==tmp_path
        return os.open(tmp_path,os.O_RDONLY|os.O_DIRECTORY),Path(path).name,[]
    return parent


@pytest.mark.parametrize('kind',['fifo','socket','symlink','directory'])
def test_B4_nonregular_leaf_rejected_without_blocking_read(tmp_path,kind):
    path=tmp_path/'input';listener=None
    if kind=='fifo':os.mkfifo(path,0o400)
    if kind=='socket':listener=socket.socket(socket.AF_UNIX,socket.SOCK_STREAM);listener.bind(str(path))
    if kind=='symlink':path.symlink_to('/etc/passwd')
    if kind=='directory':path.mkdir()
    try:
        with patch.object(admin,'safe_parent',side_effect=leaf_parent(tmp_path)),patch.object(admin.os,'read',side_effect=AssertionError('nonregular input was read')):
            with pytest.raises((OSError,RuntimeError)):admin.read_exact(path,os.getuid(),0o400)
    finally:
        if listener is not None:listener.close()
