def sha(data):
    return hashlib.sha256(data).hexdigest()


def data(value):
    return (json.dumps(value, indent=2) + '\n').encode()


def put(path, value):
    assert path.is_relative_to(P)
    with path.open('xb') as stream:
        stream.write(value if isinstance(value, bytes) else data(value))
    path.chmod(0o400)


def ref(path):
    raw = path.read_bytes()
    s = path.lstat()
    return dict(path=str(path), sha256=sha(raw), bytes=len(raw),
                mode=stat.S_IMODE(s.st_mode), uid=s.st_uid, nlink=s.st_nlink)


def snapshot(root):
    rows = {}
    for path in [root, *sorted(root.rglob('*'))]:
        s = path.lstat()
        assert stat.S_ISREG(s.st_mode) or stat.S_ISDIR(s.st_mode)
        row = dict(mode=s.st_mode, uid=s.st_uid, gid=s.st_gid, nlink=s.st_nlink,
                   device=s.st_dev, inode=s.st_ino)
        if stat.S_ISREG(s.st_mode):
            raw = path.read_bytes()
            row.update(sha256=sha(raw), size=len(raw), mtime_ns=s.st_mtime_ns)
        rows[str(path)] = row
    return rows


def functions(raw):
    return {n.name: ast.dump(n, include_attributes=False)
            for n in ast.parse(raw).body if isinstance(n, ast.FunctionDef)}

def replace(raw, old, new):
    assert old != new and old in raw
    return raw.replace(old, new)


def save_successor(relative, replacements):
    old = (OLD / relative).read_bytes()
    new = old
    deltas = []
    for a, b in replacements:
        a, b = a.encode(), b.encode()
        count = new.count(a)
        new = replace(new, a, b)
        deltas.append(dict(before=a.decode(), after=b.decode(), occurrences=count))
    put(P / relative, new)
    if new != old:
        changes.append(dict(relative=relative, before=ref(OLD / relative),
                            after=ref(P / relative), exact_replacements=deltas))
    return new
