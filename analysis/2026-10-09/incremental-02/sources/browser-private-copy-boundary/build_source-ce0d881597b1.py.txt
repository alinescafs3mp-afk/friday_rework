import ast
import difflib
import hashlib
import json
from pathlib import Path

W=Path('<ASTRA_WORK>')
OUT=W/'browser-private-copy-boundary'
OLD=W/'browser-ownership-runtime-join/candidate'
source=(OLD/'private_effect_fence.py').read_text()
assert hashlib.sha256(source.encode()).hexdigest()=='283631feb0ddfc4eb75b087d607c865f7c65d8632ad6f14142f5245f6cfb31fd'
extension=(OUT/'copy_extension.py.txt').read_text()
source=source.replace('\ndef deny(event,diagnostic=None):','\n'+extension+'\n\ndef deny(event,diagnostic=None):',1)
source=source.replace(" if event=='open':\n  path,mode,flags=args", " if event=='open':\n  copy_open_audit(args)\n  path,mode,flags=args",1)
source=source.replace(" elif event=='os.truncate':check_path(event,args[0])", " elif event=='os.truncate':check_path(event,args[0])\n elif event=='shutil.copyfile' and state['phase']=='project':copy_begin(args)\n elif event=='shutil.copystat' and state['phase']=='project':copy_finish(args)",1)
source=source.replace(' ownership_bind(source_pins)\n',' ownership_bind(source_pins)\n copy_bind(source_pins)\n',1)
source=source.replace(' os.chown=guarded_ownership_chown\n',' os.copy_file_range=guarded_copy_file_range\n os.sendfile=guarded_copy_sendfile\n os.chown=guarded_ownership_chown\n',1)
source=source.replace(" return {'registry_pidfd_probe':registry_pidfd_cessation()", " return {'private_copy':copy_cessation(),'registry_pidfd_probe':registry_pidfd_cessation()",1)
ast.parse(source)
(OUT/'candidate/private_effect_fence.py').write_text(source)

# Exact reconstruction inserted ahead of the complete existing comparator.
normalizer='''    # Validate only the exact private-copy insertions, then keep every older
    # full-function comparator and its old source hashes unchanged.
    install=g13_after['install'].body
    for statement,following in (
        ('copy_bind(source_pins)', 'IMPORT_MANIFEST=import_manifest;IMPORT_HELPER=import_path.__file__'),
        ('os.copy_file_range=guarded_copy_file_range', 'os.sendfile=guarded_copy_sendfile'),
        ('os.sendfile=guarded_copy_sendfile', 'os.chown=guarded_ownership_chown')):
        wanted=ast.parse(statement).body[0]
        matches=[i for i,n in enumerate(install) if dump(n)==dump(wanted)]
        assert len(matches)==1
        index=matches[0]
        assert dump(install[index+1])==dump(ast.parse(following).body[0])
        del install[index]
    branch=g13_after['audit'].body[-1]
    while not (isinstance(branch.test,ast.Compare) and dump(branch.test)==dump(ast.parse("event == 'open'",mode='eval').body)):
        assert len(branch.orelse)==1 and isinstance(branch.orelse[0],ast.If)
        branch=branch.orelse[0]
    assert dump(branch.body[0])==dump(ast.parse('copy_open_audit(args)').body[0])
    del branch.body[0]
    for expression,statement in (
        ("event=='shutil.copyfile' and state['phase']=='project'",'copy_begin(args)'),
        ("event=='shutil.copystat' and state['phase']=='project'",'copy_finish(args)')):
        previous=g13_after['audit'].body[-1]
        while not (len(previous.orelse)==1 and isinstance(previous.orelse[0],ast.If)
                   and dump(previous.orelse[0].test)==dump(ast.parse(expression,mode='eval').body)):
            assert len(previous.orelse)==1 and isinstance(previous.orelse[0],ast.If)
            previous=previous.orelse[0]
        added=previous.orelse[0]
        assert [dump(n) for n in added.body]==[dump(ast.parse(statement).body[0])]
        previous.orelse=added.orelse
    mapping=g13_after['closed_primitives'].body[0].value
    assert isinstance(mapping,ast.Dict) and dump(mapping.keys[0])==dump(ast.Constant('private_copy'))
    assert dump(mapping.values[0])==dump(ast.parse('copy_cessation()',mode='eval').body)
    del mapping.keys[0];del mapping.values[0]
'''
(OUT/'exact-copy-normalizer.txt').write_text(normalizer)
comparator=(OLD/'test_registry_pidfd_boundary.py').read_text()

    (OUT/(name+'.patch')).write_text(''.join(difflib.unified_diff((OLD/name).read_text().splitlines(True),
        (OUT/'candidate'/name).read_text().splitlines(True),fromfile=str(OLD/name),tofile=str(OUT/'candidate'/name))))
