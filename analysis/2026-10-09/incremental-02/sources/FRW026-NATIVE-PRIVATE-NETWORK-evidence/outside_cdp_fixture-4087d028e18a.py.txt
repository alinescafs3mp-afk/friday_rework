"""Benign outside-loopback fixture for future EXISTING native qualification.

NOT RUN. Not production CDP and not a network manager. Root supplies the exact
already-owned supervisor/budget; this code cannot admit/start a native unit.
Root must establish direct unconfined TCP/UDP positives, keep this SAME process
alive through the child checks, then independently observe cessation/counts.
"""
import json
import selectors
import socket
import time


def serve(owner, nonce, output):
    owner.require_current(owner.binding)
    if type(nonce) is not str or not nonce.isascii() or not 1 <= len(nonce) <= 64:
        raise ValueError('FIXTURE_NONSECRET_NONCE_REQUIRED')
    if owner.deadline <= time.monotonic(): raise ValueError('ORIGINAL_FIXTURE_DEADLINE_EXHAUSTED')
    sockets = []; rows = []; counts = {}; selector = selectors.DefaultSelector()
    try:
        for family, address in [(socket.AF_INET, '127.0.0.1'), (socket.AF_INET6, '::1')]:
            tcp = socket.socket(family, socket.SOCK_STREAM); sockets.append(tcp)
            if family == socket.AF_INET6: tcp.setsockopt(socket.IPPROTO_IPV6, socket.IPV6_V6ONLY, 1)
            tcp.bind((address, 0)); tcp.listen(16); tcp.setblocking(False)
            port = tcp.getsockname()[1]
            udp = socket.socket(family, socket.SOCK_DGRAM); sockets.append(udp)
            if family == socket.AF_INET6: udp.setsockopt(socket.IPPROTO_IPV6, socket.IPV6_V6ONLY, 1)
            udp.bind((address, port)); udp.setblocking(False)
            selector.register(tcp, selectors.EVENT_READ, ('tcp', family, port))
            selector.register(udp, selectors.EVENT_READ, ('udp', family, port))
            rows.append({'family': family, 'bind': address, 'port': port})
            counts[str(family)] = {'tcp_accepts': 0, 'udp_received': 0}
        owner.h.put(output / 'outside-listener-start.json', {
            'fixture': 'BENIGN_CDP_LIKE_TCP_AND_UDP', 'listeners': rows,
            'native_outer': owner.binding, 'original_deadline': owner.deadline,
            'at_monotonic': time.monotonic(), 'actual_counts': counts,
            'real_browser': False, 'runtimeGrant': False})
        while time.monotonic() < owner.deadline - owner.pre['reserve_seconds']:
            owner.require_current(owner.binding)
            for key, _ in selector.select(min(1.0, max(0.0, owner.deadline - time.monotonic()))):
                kind, family, _ = key.data
                if kind == 'tcp':
                    client, _ = key.fileobj.accept()
                    counts[str(family)]['tcp_accepts'] += 1
                    with client:
                        client.settimeout(min(1.0, max(0.001, owner.deadline - time.monotonic())))
                        # A short fixed HTTP discovery result; no arbitrary execution
                        # or auth/logging surface and no input-controlled response.
                        body = json.dumps({'Browser': 'benign outside fixture', 'nonce': nonce}).encode()
                        client.sendall(b'HTTP/1.1 200 OK\r\nContent-Type: application/json\r\nConnection: close\r\nContent-Length: ' +
                                       str(len(body)).encode() + b'\r\n\r\n' + body)
                else:
                    data, peer = key.fileobj.recvfrom(1024)
                    counts[str(family)]['udp_received'] += 1
                    if data == nonce.encode(): key.fileobj.sendto(data, peer)
        owner.h.put(output / 'outside-listener-end.json', {
            'listeners': rows, 'actual_counts': counts, 'original_deadline': owner.deadline,
            'at_monotonic': time.monotonic(), 'runtimeGrant': False})
        return counts
    finally:
        selector.close()
        for obj in sockets: obj.close()
