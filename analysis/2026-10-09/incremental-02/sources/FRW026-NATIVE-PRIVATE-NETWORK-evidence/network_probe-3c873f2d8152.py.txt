"""Real future child probe, NOT RUN. No sandbox or runtime grant is created.

Root invokes this through the FINAL independently admitted FileOwner path, after

labels denials only after own loopback, DNS, TLS/public web and BOTH native local

import errno
import hashlib
import json
import os
from pathlib import Path
import socket
import ssl
import time
import urllib.request


def require(value, reason):
    if not value:
        raise RuntimeError(reason)


def before(deadline, maximum=2.0):
    left = deadline - time.monotonic()
    require(left > 0, 'ORIGINAL_NETWORK_PROBE_DEADLINE_EXHAUSTED')
    return min(maximum, left)


def own_loopback(deadline, family):
    host = '127.0.0.1' if family == socket.AF_INET else '::1'
    with socket.socket(family, socket.SOCK_STREAM) as listener:
        listener.settimeout(before(deadline))
        listener.bind((host, 0)); listener.listen(1)
        with socket.create_connection((host, listener.getsockname()[1]), timeout=before(deadline)) as client:
            with listener.accept()[0] as accepted:
                accepted.settimeout(before(deadline)); accepted.sendall(b'OWN_LOOPBACK_POSITIVE')
                require(client.recv(64) == b'OWN_LOOPBACK_POSITIVE', 'OWN_LOOPBACK_UNAVAILABLE')
        return {'family': family, 'actual_bound_port': listener.getsockname()[1]}


def own_udp(deadline, family):
    host = '127.0.0.1' if family == socket.AF_INET else '::1'
    with socket.socket(family, socket.SOCK_DGRAM) as listener, socket.socket(family, socket.SOCK_DGRAM) as client:
        listener.bind((host, 0)); listener.settimeout(before(deadline)); client.settimeout(before(deadline))
        client.sendto(b'OWN_UDP_POSITIVE', listener.getsockname())
        data, peer = listener.recvfrom(64); require(data == b'OWN_UDP_POSITIVE', 'OWN_UDP_UNAVAILABLE')
        listener.sendto(data, peer)
        require(client.recvfrom(64)[0] == data, 'OWN_UDP_RETURN_UNAVAILABLE')
        return {'family': family, 'actual_bound_port': listener.getsockname()[1]}


def udp_unreachable(host, port, nonce, deadline):
    family = socket.AF_INET6 if ':' in host else socket.AF_INET
    with socket.socket(family, socket.SOCK_DGRAM) as client:
        client.settimeout(before(deadline))
        try:
            client.sendto(nonce.encode(), (host, port))
            data, _ = client.recvfrom(1024)
            raise RuntimeError('FOREIGN_HOST_UDP_ROUTE_RESPONDED')
        except OSError as error:
            require(isinstance(error, TimeoutError) or error.errno in
                    (errno.ECONNREFUSED, errno.ENETUNREACH, errno.EHOSTUNREACH, errno.ETIMEDOUT),
                    'UDP_NEGATIVE_CAUSE_UNQUALIFIED')
            return {'destination': host, 'port': port, 'errno': error.errno, 'exception': type(error).__name__}


def https_web(url, deadline):
    require(url.startswith('https://'), 'EXPLICIT_HTTPS_WEB_ROUTE_REQUIRED')
    request = urllib.request.Request(url, method='GET')
    with urllib.request.urlopen(request, timeout=before(deadline, 15.0), context=ssl.create_default_context()) as response:
        data = response.read(65537)
        require(response.status == 200 and 0 < len(data) <= 65536, 'WEB_POSITIVE_NOT_ESTABLISHED')
        return {'status': response.status, 'body_bytes': len(data), 'sha256': hashlib.sha256(data).hexdigest()}


def native_model(base, operation, deadline, key_name):
    # Keys are approved per-job native provider inputs; operator/dashboard/A0
    # control credentials must never be supplied to this network probe.
    payload = ({'model': 'dispatcher', 'messages': [{'role': 'user', 'content': 'Reply OK.'}],
                'max_tokens': 4} if operation == 'chat/completions' else
               {'model': 'qwen3-embedding-0.6b', 'input': 'finite network qualification'})
    headers = {'Content-Type': 'application/json'}
    if os.environ.get(key_name): headers['Authorization'] = 'Bearer ' + os.environ[key_name]
    request = urllib.request.Request(base + '/' + operation, data=json.dumps(payload).encode(), headers=headers)
    with urllib.request.urlopen(request, timeout=before(deadline, 30.0)) as response:
        data = response.read(65537)
        require(response.status == 200 and len(data) <= 65536, 'NATIVE_MODEL_POSITIVE_NOT_ESTABLISHED')
        value = json.loads(data)
        require(type(value) is dict and ('choices' in value if operation == 'chat/completions' else 'data' in value),
                'NATIVE_MODEL_SCHEMA_NOT_ESTABLISHED')
        return {'status': response.status, 'body_bytes': len(data), 'operation': operation}


def unreachable(host, port, deadline):
    try:
        with socket.create_connection((host, port), timeout=before(deadline)):
            raise RuntimeError('FOREIGN_LOOPBACK_ROUTE_REACHABLE')
    except OSError as error:
        # Failures such as EMFILE, ENOMEM, EPERM from a different containment
        # boundary do NOT qualify a network-specific negative.
        require(error.errno in (errno.ECONNREFUSED, errno.ENETUNREACH, errno.EHOSTUNREACH,
                                errno.ETIMEDOUT) or isinstance(error, TimeoutError),
                'NETWORK_DENIAL_CAUSE_UNQUALIFIED')
        return {'destination': host, 'port': port, 'errno': error.errno, 'exception': type(error).__name__}


def probe(contract):
    require(contract['state'] == 'INDEPENDENT_NETWORK_QUALIFICATION_ONLY' and
            contract['runtime_grant_source'] == 'FINAL_EXISTING_FILEOWNER_NATIVE_ADMISSION' and
            contract['outside_tcp_unconfined_positive'] is True and contract['outside_udp_unconfined_positive'] is True,
            'EXACT_OUTSIDE_FIXTURE_POSITIVE_REQUIRED_NOT_A_GRANT')
    deadline = contract['original_deadline_monotonic']
    require(deadline == contract['original_start_monotonic'] + contract['original_budget_seconds'],
            'ORIGINAL_CLOCK_NOT_RESET')
    ports = contract['outside_benign_cdp_fixture_ports']
    require(type(ports) is list and ports and all(type(v) is int and 0 < v < 65536 for v in ports),
            'EXACT_OUTSIDE_DYNAMIC_PORTS_REQUIRED')
    positives = {'own_loopback4': own_loopback(deadline, socket.AF_INET),
                 'own_loopback6': own_loopback(deadline, socket.AF_INET6),
                 'own_udp4': own_udp(deadline, socket.AF_INET),
                 'own_udp6': own_udp(deadline, socket.AF_INET6),
                 'web_tls': https_web(contract['public_web_https_url'], deadline),
                 'local_chat': native_model('http://192.168.1.78:8001/v1', 'chat/completions', deadline, 'FRIDAY_LLM_API_KEY'),
                 'local_embedding': native_model('http://192.168.1.78:8002/v1', 'embeddings', deadline, 'FRIDAY_EMBEDDINGS_API_KEY')}
    # HTTPS hostname resolution is a same-child DNS positive; explicit remote
    # DNS/TCP/UDP fixture controls and packet observations remain required.
    negatives = []
    for port in ports:
        for address in ('127.0.0.1', '::1', '10.0.2.2', '10.0.2.3', 'fd00::2', 'fd00::3', '::ffff:127.0.0.1'):
            negatives.append({'tcp': unreachable(address, port, deadline),
                              'udp': udp_unreachable(address, port, contract['outside_nonce'], deadline)})
    return {'state': 'FOCUSED_NETWORK_PROBE_OBSERVED_NOT_PRODUCT_ACCEPTANCE',
            'positives': positives, 'negatives': negatives,
            'negative_count': len(negatives), 'runtimeGrant': False,
            'outside_listener_receipt_sha256': contract['outside_listener_receipt_sha256'],
            'original_deadline': deadline, 'finished_monotonic': time.monotonic()}
