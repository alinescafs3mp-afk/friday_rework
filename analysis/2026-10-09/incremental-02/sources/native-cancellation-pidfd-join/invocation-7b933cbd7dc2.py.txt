"""Read-only launcher followed by ONE existing installed timeout. NOT_RUN.
Exact owner launch is env -i ... pinned-python -I -B invocation.py R NAME ENTRY @canary.
Exact collection adds --collect-only; runtest uses the separate reviewed
env-i/-P/-s/-B/seed0 profile and adds --runtest-file=fNNN.
Separate source readiness and independent runtime admission are mandatory.
No bootstrap here, no clock mutation, no Python/loader/key inheritance or new manager.
"""
import os, pathlib, runpy, sys, time, re


def main():
    tail = sys.argv[5:]
    runtime_match = re.fullmatch(r'--runtest-file=(f[0-9]{3})', tail[0]) if len(tail) == 1 else None
    if len(sys.argv) not in (5, 6) or (tail not in ([], ['--collect-only']) and runtime_match is None):
        raise SystemExit('EXACT_ADMISSION_ARGUMENTS_REQUIRED')
    source = pathlib.Path(__file__).resolve().parent
    run = pathlib.Path(sys.argv[1]).resolve(strict=True)
    name, entry, selection = sys.argv[2:5]
    collect_only = sys.argv[5:] == ['--collect-only']
    api = runpy.run_path(str(source / 'admission.py'))
    runtest_file = runtime_match[1] if runtime_match else None
    clock, phase, _ = api['verify'](source, run, name, entry, selection, collect_only, runtest_file)
    # Keep the planned wall duration or refuse an insufficient ORIGINAL budget.
    # 1s is startup reserve, not a claim of realtime absolute-deadline enforcement.
    now = time.monotonic()
    duration = float(phase['outer_wall_seconds'])
    if clock['deadline_monotonic'] - now <= duration + 1.0:
        raise SystemExit('INSUFFICIENT_ORIGINAL_PHASE_BUDGET')
    external_deadline = now + duration
    interpreter = '<PROJECT>/.runtime/hermes-telegram-test/env/bin/python'
    flags = ['-P', '-s', '-B'] if runtest_file is not None else ['-I', '-B']
    os.execv('/usr/bin/gnutimeout', ['timeout', '--signal=KILL', '--verbose', f'{duration:.6f}s',
             interpreter, *flags, str(source / entry), name, str(run), selection,
             *tail,
             '--external-deadline', repr(external_deadline)])


if __name__ == '__main__':
    main()
