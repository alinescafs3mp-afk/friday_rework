def inventory(p):

  for d in f.decorator_list:
   if isinstance(d,ast.Call) and isinstance(d.func,ast.Attribute) and d.func.attr=='parametrize':

    factors.append(len(d.args[1].elts));params.append(ast.dump(d,include_attributes=False))
  n=1
  for fac in factors:n*=fac
  result.append({'name':f.name,'literal_cases':n,'parameter_decorators':params})
 return result
