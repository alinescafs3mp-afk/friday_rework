async function inspectProjectThenObserve(phase, helper, out) {
  // This step is only used for a fresh, actually collected project phase.
  const root = "<ASTRA_WORK>";
  if (phase !== out + "/collect-project") throw Error("foreign phase");
  const command = "/usr/bin/python3 -I -B " + root +
    "/lexical-qualification-binding/classification/inspect_project.py " +
    phase + "/classification-source-basis.json";
  let inspect = await tools.exec_command({cmd: command, yield_time_ms: 1000, max_output_tokens: 1200});
  let inspectionOutput = inspect.output;
  while (inspect.session_id && inspect.exit_code === undefined) {
    inspect = await tools.write_stdin({session_id: inspect.session_id, chars: "", yield_time_ms: 1000, max_output_tokens: 1200});
    inspectionOutput += inspect.output;
  }
  if (inspect.exit_code !== 0 || inspect.session_id) {
    text({stage:"owner-inspection", result:inspect});
    return false;
  }
  const review = JSON.parse(inspectionOutput.trim());
  if (review.state !== "INDEPENDENT_COLLECTION_METADATA_RECONCILED" ||
      review.path !== phase + "/classification-review.json" ||
      review.runtime_acceptance !== false ||
      review.owner_inspection?.definitions !== 627 || review.owner_inspection?.modules !== 17 ||
      !/^[a-f0-9]{64}$/.test(review.sha256)) throw Error("invalid publication receipt");
  // The observer call is unreachable after inspection refusal or parse failure.
  let observed = await tools.exec_command({cmd: "/usr/bin/python3 -I -B " + helper +
    " observe " + out + " 2", yield_time_ms: 1000, max_output_tokens: 1200});
  while (observed.session_id && observed.exit_code === undefined) {
    observed = await tools.write_stdin({session_id: observed.session_id, chars: "", yield_time_ms: 1000, max_output_tokens: 1200});
  }
  text({stage:"project-observe", result:observed});
  return observed.exit_code === 0 && !observed.session_id;
}
