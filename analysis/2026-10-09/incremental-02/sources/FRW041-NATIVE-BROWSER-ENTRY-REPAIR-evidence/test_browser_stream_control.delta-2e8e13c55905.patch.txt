+@pytest.mark.parametrize('publication', ['none', 'launch', 'headers'])
+@pytest.mark.parametrize('exhausted', ['current', 'original_call', 'original_owner'])
+def test_real_bind_clock_failure_permanently_latches_original_control(native_stream, publication, exhausted):
+    n = native_stream
+    c = stream_control(n.key, n.info)
+    token, bound = c.token, c.original_deadline
+    if publication == 'launch': c.launch_environment(n.owner, 'daemon', {})
+    elif publication == 'headers': c.headers(n.owner, 'daemon')
+    if exhausted == 'current': deadline, n.clock.now = 105.0, 106.0
+    elif exhausted == 'original_call':
+        deadline, n.clock.now = 120.0, 106.0
+        n.call.run_deadline = 105.0
+    else:
+        deadline, n.clock.now = 130.0, 121.0
+        n.call.base_deadline = n.call.run_deadline = 130.0
+        # Native calls remain live, but the retained original owner's 120
+        # is exhausted: this reaches the real BrowserOwnerRefused branch.
+    with n.caller(deadline), pytest.raises((TimeoutError, BrowserOwnerRefused)):
+        stream_control(n.key, n.info)  # Real bind/current_web_call failure.
+    assert n.info['_friday_stream_control'] is c and c._refused
+    assert c.owner is n.owner and c.token == token and c.original_deadline == bound
+    n.call.base_deadline = n.call.run_deadline = 120.0
+    n.clock.now = 107.0 if exhausted != 'original_owner' else 100.0
+    # Clock rollback only stresses sticky refusal; it is never a runtime reset.
+    with n.caller(120.0):
+        for operation in (lambda: stream_control(n.key, n.info),
+                          lambda: c.launch_environment(n.owner, 'daemon', {}),
+                          lambda: c.headers(n.owner, 'daemon')):
+            with pytest.raises(BrowserOwnerRefused, match='permanently_refused'):

+@pytest.mark.parametrize('axis', ['principal', 'generation', 'task_id', 'session_key',
+                                'turn_id', 'turn', 'membership', 'session', 'route', 'scope'])
+@pytest.mark.parametrize('operation', ['entry', 'headers', 'launch'])
+def test_expired_foreign_caller_cannot_poison_retained_control(native_stream, axis, operation):
+    n = native_stream
+    c = stream_control(n.key, n.info)
+    args = {}; info = n.info; key = n.key
+    if axis == 'principal': args['scope'] = replace(n.cap, principal=('telegram', 'x', 'y', 'foreign'))
+    elif axis == 'generation': args['scope'] = replace(n.cap, admission_generation=4)
+    elif axis in ('task_id', 'session_key', 'turn_id'): args['fields'] = {axis: 'foreign'}
+    elif axis == 'turn': args['foreign_turn'] = True
+    elif axis == 'membership': args['member'] = False
+    elif axis == 'session': info = {**n.info, 'session_name': 'foreign'}
+    elif axis == 'route': key = 'foreign'
+    else: info = {**n.info, '_friday_user_scope': replace(n.cap, admission_generation=4)}
+    n.clock.now = 106.0
+    with n.caller(105.0, **args), pytest.raises((RuntimeError, PermissionError, TimeoutError)):
+        if operation == 'entry': stream_control(key, info)
+        elif axis in ('session', 'route', 'scope'):
+            # Dictionary/route identities are entry-only surfaces. Other public
+            # methods have an explicit owner/session instead, without a route.
+            c.headers(replace(n.owner), 'foreign')
+        elif operation == 'headers': c.headers(n.owner, 'daemon')
+        else: c.launch_environment(n.owner, 'daemon', {})
+    assert not c._refused and c._retained_deadline == 120.0
+    assert n.info['_friday_stream_control'] is c
+    with n.caller(120.0):
+        assert stream_control(n.key, n.info) is c
