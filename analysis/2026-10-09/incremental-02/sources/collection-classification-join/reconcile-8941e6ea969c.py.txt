#!/usr/bin/env python3
"""Read actual schema2 JSON; print SOURCE BASIS, never an acceptance/review.
Usage: python3 -I -B reconcile.py RESULT CLASSIFIED DESELECTED COLLECTION
Owner separately verifies execution/receipt graph and inspects this basis before
writing the helper's classification-review.json. No candidate imports or effects.
Exit 2: unresolved or invalid. Exit 0: data join only, all runtests NOT_RUN.
"""
import ast
import hashlib
import json
from pathlib import Path
import sys

MAP = Path(__file__).with_name('provenance.json')
MAP_SHA = 'b7fda73759ac44f851e3029e8e3361d865b0928205ada1a2746392b1528cb332'
ALIAS = 'Imported test alias; not a distinct test defined in the requested file. Definition remains in original dependency evidence; no count inflation.'
STOP = 'NOT_RUN: actual subprocess outside nochildren task boundary; native process lock second-open proof remains required.'
HTTP = 'NOT_RUN: actual local HTTP server/git subprocess fixture exceeds task no network/nochildren; still required in separately authorized admission'
WIN_SKIP = 'Native Windows-only case: actual Linux host; NOT_APPLICABLE, not safety PASS'


def require(ok, message):
    if not ok:
        raise ValueError(message)


def digest(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def read(p):
    def unique(pairs):
        out = {}
        for k, v in pairs:
            require(k not in out, 'duplicate JSON key: ' + k)
            out[k] = v
        return out
    return json.loads(Path(p).read_text(), object_pairs_hook=unique)


def key(identity):
    require(identity['object_path'] == identity['path'], 'noncanonical object_path')
    return (identity['object_path'], identity['kind'], identity['nodeid'])


def ref(f, d=None):
    r = {k: f[k] for k in ('path', 'sha256')}
    if d is not None:
        r.update({k: d[k] for k in ('qualified_name', 'line', 'end_line')})
        if 'id' in d:
            r['definition_id'] = d['id']
    return r


def main(args):
    require(len(args) == 4, 'expected RESULT CLASSIFIED DESELECTED COLLECTION')
    require(digest(MAP) == MAP_SHA, 'provenance map drift')
    m = read(MAP)
    for r in [m[k] for k in ('catalog', 'prior_source_inspection_result', 'current_pair', 'current_producer', 'current_observer')] + m['files']:
        require(digest(r['path']) == r['sha256'], 'source drift: ' + r['path'])
    files = {f['path']: f for f in m['files']}
    definitions = {d['id']: (f, d) for f in m['files'] for d in f['definitions']}
    rules = m['dynamic_rules']
    result, kept, dropped, collection = map(read, args)
    require(result['receipt_schema'] == 2 and result['collect_only'] is True and result['exit'] == 0 and result['reports'] == [], 'not successful collection-only schema2')
    selection = result['selection']
    require(selection in ('@project', '@native'), 'unknown selection')
    selected = m['selection'][selection[1:]]
    require(result['selected'] == selected and result['missing'] == ['tests/agent/test_skill_preprocessing.py'], 'selection/missing mismatch')
    require([r['nodeid'] for r in kept] == collection == result['collection'], 'kept order mismatch')
    require([r['identity'] for r in kept] == result['collection_identities'], 'kept identity mismatch')
    discovered = result['discovered_identities']
    require([r['nodeid'] for r in discovered] == result['discovered'], 'discovery order mismatch')
    dk = list(map(key, discovered)); kk = [key(r['identity']) for r in kept]; xk = [key(r['identity']) for r in dropped]
    require(len(set(dk)) == len(dk) and len(set(kk)) == len(kk) and len(set(xk)) == len(xk), 'duplicate compound identity')
    require(not set(kk) & set(xk) and set(dk) == set(kk) | set(xk), 'compound partition mismatch')
    require([k for k in dk if k in set(kk)] == kk and [k for k in dk if k in set(xk)] == xk, 'compound partition order mismatch')
    by_discovered = dict(zip(dk, discovered))
    require(all(by_discovered[key(r['identity'])] == r['identity'] for r in kept + dropped), 'partition metadata drift')
    reports = result['collection_reports']; report_map = {key(r['identity']): r for r in reports}
    require(len(report_map) == len(reports), 'duplicate collector report')
    module_reports = {}
    for report in reports:
        if report['identity']['kind'] == 'module':
            module_reports.setdefault(report['identity']['path'], []).append(report)
    require(set(module_reports) == set(selected) and all(len(v) == 1 for v in module_reports.values()), 'missing/duplicate module receipt')
    unresolved = []; modules = []; reviewed = []; drop_reviews = []
    for path in selected:
        f = files[path]; receipt = module_reports[path][0]
        require(receipt['outcome'] in ('passed', 'skipped'), 'failed module receipt')
        disposition = ('SKIPPED_COLLECTION_NOT_RUN' if receipt['outcome'] == 'skipped' else 'EMPTY_MODULE_COLLECTED_NO_TESTS_NOT_RUN' if not receipt['children'] else 'COLLECTED_ONLY_NOT_RUN')
        modules.append({'path': path, 'receipt': receipt, 'disposition': disposition, 'source_basis': ref(f), 'source_catalog_file_id': f['id'], 'module_declarations': f['module_declarations']})
        if receipt['outcome'] == 'skipped':
            unresolved.append({'module': path, 'kind': 'ACTUAL_MODULE_SKIP_REQUIRES_OWNER_DETAIL_RECONCILIATION', 'detail': receipt['detail']})

    def join(row, is_drop):
        ik = key(row['identity']); path = ik[0]
        require(path in selected and ik[1] == 'function' and row['path'] == path and row['nodeid'] == ik[2], 'unknown item/path/kind')
        f = files[path]; parents = row['parents']
        require(parents and parents[0]['kind'] == 'session', 'missing actual session ancestry')
        for index, parent in enumerate(parents):
            report = report_map.get(key(parent))
            require(report is not None and report['identity'] == parent and report['parents'] == parents[:index], 'parent receipt mismatch')
        module_parents = [p for p in parents if p['kind'] == 'module']
        require(len(module_parents) == 1 and module_parents[0]['path'] == path, 'exact module parent missing')
        previous = module_parents[0]; class_names = []; class_basis = []
        for p in [p for p in parents if p['kind'] == 'class']:
            prefix = previous['nodeid'] + '::'
            require(p['path'] == path and p['nodeid'].startswith(prefix), 'class lineage mismatch')
            class_names.append(p['nodeid'][len(prefix):]); q = '.'.join(class_names)
            matches = [c for c in f['classes'] if c['qualified_name'] == q]
            require(len(matches) == 1, 'unknown/ambiguous class lineage')
            class_basis.append(matches[0]); previous = p
        scope = '.'.join(class_names) or None
        definition_path = row['definition']
        require(definition_path in files, 'definition file outside inspected catalog')
        source = files[definition_path]
        candidates = []
        for d in source['definitions']:
            if d['kind'] != 'test_definition' or d['enclosing_class'] != scope:
                continue
            name = d['qualified_name'].rsplit('.', 1)[-1]
            prefix = previous['nodeid'] + '::' + name
            tail = row['nodeid'][len(prefix):] if row['nodeid'].startswith(prefix) else None
            if tail == '' or (tail is not None and tail.startswith('[') and tail.endswith(']') and bool(d.get('parametrize'))):
                if not is_drop or row.get('originalname') == name:
                    candidates.append(d)
        require(len(candidates) == 1, 'definition/originalname unknown or ambiguous')
        d = candidates[0]; special = next((s for s in m['special_cases'] if s['definition_id'] == d['id']), None)
        if definition_path != path:
            require(is_drop and row['reason'] == ALIAS, 'foreign kept definition')
            edges = [i for i in f['imported_test_alias_candidates'] if i['source_candidate'] == definition_path and i['original'] == d['qualified_name'] and i['alias'] == row['originalname']]
            require(len(edges) == 1, 'unknown/ambiguous imported alias origin')
            require(False, 'imported alias needs importing-module fixture lookup; no such source alias exists in pinned selected snapshot')
        if is_drop:
            require(special is not None, 'unexpected deselection without exact source scope')
            if special['scope'] == 'STOP_REQUIRED_NOT_RUN_NO_REROUTE':
                require(row['reason'] == STOP, 'stopped reason mismatch')
            else:
                require(special['scope'] == 'REAL_LOOPBACK_HTTP_AND_OPTIONAL_GIT_PROCESS_FIXTURE_REQUIRED_NOT_RUN' and row['reason'] == HTTP, 'HTTP/process exclusion mismatch')
        else:
            require(not special or special['scope'] not in ('STOP_REQUIRED_NOT_RUN_NO_REROUTE', 'REAL_LOOPBACK_HTTP_AND_OPTIONAL_GIT_PROCESS_FIXTURE_REQUIRED_NOT_RUN'), 'required exclusion retained')
        decorators = d.get('decorators', [])
        platforms = []; loop_scope = None
        for dec in decorators:
            node = ast.parse(dec, mode='eval').body
            call = node.func if isinstance(node, ast.Call) else node
            name = ast.unparse(call)
            if name == 'pytest.mark.platforms':
                require(isinstance(node, ast.Call), 'dynamic platforms')
                platforms.append([ast.literal_eval(a) for a in node.args])
            if name == 'pytest.mark.asyncio':
                opts = {k.arg: ast.literal_eval(k.value) for k in node.keywords} if isinstance(node, ast.Call) else {}
                loop_scope = opts.get('loop_scope', opts.get('scope', 'function'))
        disposition = 'COLLECTED_ONLY_NOT_RUN'
        if not is_drop:
            require(row['platforms'] == platforms, 'actual/source platforms mismatch')
            if any(set(p) == {'windows'} for p in platforms):
                require(row['skip_reasons'] == [WIN_SKIP], 'Windows skip reason mismatch')
                disposition = 'LINUX_NOT_APPLICABLE'
            else:
                require(row['skip_reasons'] == [], 'unexpected skip reason needs source inspection')
        fixture_origins = {}; fixture_basis = {}
        candidates_by_name = m['fixture_candidate_sets'][f['fixture_candidates_by_test'][d['qualified_name']]]
        require(len(set(row['fixtures'])) == len(row['fixtures']), 'duplicate actual fixture name')
        for name in row['fixtures']:
            refs = []; explanation = None
            ids = candidates_by_name.get(name, [])
            if name == 'request':
                refs = [rules['intrinsic_request']['origin']]; explanation = 'intrinsic_request'
            elif name in d.get('direct_parametrize_argument_candidates', []):
                require(name not in d.get('indirect_parametrize_argument_candidates', []), 'ambiguous direct/indirect parameter')
                refs = [ref(f, d), rules['direct_parameter']['origin']]; explanation = 'direct_parameter_from_exact_decorator'
            elif len(ids) == 1 and ids[0] in definitions:
                ff, fd = definitions[ids[0]]; refs = [ref(ff, fd)]; explanation = 'exact_source_fixture_lookup_candidate'
            elif name.startswith('_') and name.endswith('_scoped_runner'):
                expected = '_' + str(loop_scope) + '_scoped_runner'
                require(loop_scope in rules['asyncio_runner_fixture']['possible_scopes'] and name == expected and d['async_function'] and row['identity']['type'] == 'pytest_asyncio.plugin.Coroutine', 'runner lacks exact scope/type/source basis')
                refs = [rules['asyncio_runner_fixture']['origin'], ref(f, d), m['current_producer']]; explanation = 'actual_name_and_coroutine_type_match_pinned_factory_and_decorator'
            elif scope and name == '_unittest_setUpClass_fixture_' + scope:
                require(class_basis[-1]['bases'] == ['unittest.TestCase'] and parents[-1]['type'] == '_pytest.unittest.UnitTestCase' and row['identity']['type'] == '_pytest.unittest.TestCaseFunction', 'unittest fixture lacks exact class/source basis')
                refs = [rules['unittest_names']['origin'], ref(f, class_basis[-1])]; explanation = 'pinned_unittest_factory_and_direct_TestCase_base'
            require(bool(refs), 'unknown/ambiguous fixture origin: ' + name)
            fixture_origins[name] = refs; fixture_basis[name] = explanation
        if {'served_repo', 'served_repo_missing_support'} & set(row['fixtures']):
            require(is_drop and row['reason'] == HTTP, 'HTTP fixture unexpectedly retained')
        return {'metadata': row, 'definition_basis': ref(source, d), 'originalname_source_basis': d['qualified_name'].rsplit('.', 1)[-1], 'class_lineage_basis': [ref(f, c) for c in class_basis], 'decorators_source_basis': decorators, 'direct_parameters': d.get('direct_parametrize_argument_candidates', []), 'indirect_parameters': d.get('indirect_parametrize_argument_candidates', []), 'mock_injected_arguments': d.get('mock_injected_argument_candidates', []), 'fixture_origins': fixture_origins, 'fixture_basis': fixture_basis, 'disposition': 'NOT_RUN' if is_drop else disposition, 'scope_basis': special['scope'] if special else 'COLLECTED_ONLY_ALL_RUNTESTS_NOT_RUN'}

    for group, destination, is_drop in ((kept, reviewed, False), (dropped, drop_reviews, True)):
        for row in group:
            try:
                destination.append(join(row, is_drop))
            except (KeyError, TypeError, ValueError, SyntaxError) as exc:
                unresolved.append({'identity': row.get('identity'), 'kind': 'ROW_SOURCE_JOIN_UNRESOLVED', 'detail': str(exc)})
    expected_exclusions = {s['definition_id'] for s in m['special_cases'] if s['path'] in selected and s['scope'] in ('STOP_REQUIRED_NOT_RUN_NO_REROUTE', 'REAL_LOOPBACK_HTTP_AND_OPTIONAL_GIT_PROCESS_FIXTURE_REQUIRED_NOT_RUN')}
    actual_exclusions = {r['definition_basis']['definition_id'] for r in drop_reviews}
    if actual_exclusions != expected_exclusions:
        unresolved.append({'kind': 'EIGHT_SOURCE_EXCLUSIONS_NOT_FULLY_ACCOUNTED', 'missing': sorted(expected_exclusions - actual_exclusions), 'extra': sorted(actual_exclusions - expected_exclusions)})
    out = {'state': 'DATA_JOIN_UNRESOLVED' if unresolved else 'DATA_JOIN_RESOLVED_REQUIRES_OWNER_INSPECTION', 'receipt_schema': 2, 'identity': result['identity'], 'input_refs': [{'path': str(Path(p)), 'sha256': digest(p)} for p in args], 'provenance_sha256': MAP_SHA, 'classified_sha256': digest(args[1]), 'module_reports': module_reports, 'selected_modules': modules, 'kept': reviewed, 'deselected': dropped, 'deselected_reviews': drop_reviews, 'unresolved': unresolved, 'all_runtests': 'NOT_RUN', 'acceptance': False, 'owner_remaining': 'Verify full actual receipt graph/backend/immutable pins and inspect every source basis. Only owner may write helper classification-review.json; no inspected flags are supplied here.'}
    print(json.dumps(out, indent=2))
    return 2 if unresolved else 0


if __name__ == '__main__':
    try:
        sys.exit(main(sys.argv[1:]))
    except (KeyError, TypeError, ValueError, OSError, SyntaxError) as exc:
        print(json.dumps({'state': 'DATA_JOIN_REFUSED', 'unresolved': [str(exc)], 'acceptance': False, 'all_runtests': 'NOT_RUN'}))
        sys.exit(2)
