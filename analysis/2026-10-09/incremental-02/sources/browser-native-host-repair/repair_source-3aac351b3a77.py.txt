def command_deadline(timeout):
    if isinstance(timeout, bool) or not isinstance(timeout, (int, float)) or not math.isfinite(timeout) or timeout <= 0:
        raise BrowserPolicyError('browser_finite_command_budget_required')
    proposed = time.monotonic() + timeout
    retained = _BROWSER_DEADLINE.get()
    return min(proposed, retained) if retained is not None else proposed


def check_command_deadline(deadline=None):
    retained = _BROWSER_DEADLINE.get()
    if retained is not None:
        deadline = min(retained, deadline) if deadline is not None else retained
    if (isinstance(deadline, bool) or not isinstance(deadline, (int, float))
            or not math.isfinite(deadline)):
        raise BrowserPolicyError('browser_original_deadline_required')
    if time.monotonic() >= deadline:
        raise BrowserPolicyError('browser_original_deadline_exhausted')
    return deadline
