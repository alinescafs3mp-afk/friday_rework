+def _pinned_executable(path, expected, *, root_owned=False):
+    path = Path(path)
+    if not path.is_absolute() or path.resolve() != path:
+        raise BrowserPolicyError('browser_executable_path_refused')
+    uid = 0 if root_owned else os.geteuid()
+    for parent in path.parents:
+        st = parent.lstat()

+            raise BrowserPolicyError('browser_executable_ancestor_refused')
+    fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK)
+    try:
+        st = os.fstat(fd)
+        if (not stat.S_ISREG(st.st_mode) or st.st_uid not in (0, uid)
+                or st.st_nlink != 1 or st.st_mode & 0o022 or not st.st_mode & 0o111
+                or not 0 < st.st_size <= 512 * 1024 * 1024):
+            raise BrowserPolicyError('browser_executable_protection_refused')
+        digest = hashlib.sha256()
+        with os.fdopen(fd, 'rb', closefd=False) as stream:

+                digest.update(block)
