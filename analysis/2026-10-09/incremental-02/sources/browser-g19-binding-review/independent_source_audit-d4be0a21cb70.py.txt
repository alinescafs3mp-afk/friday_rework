def read(p):
    return json.loads(Path(p).read_text())

def sha(p):
    with Path(p).open('rb') as f:
        return hashlib.file_digest(f, 'sha256').hexdigest()

def ref(p):
    p = Path(p)
    r = dict(path=str(p), sha256=sha(p), bytes=p.stat().st_size)
    refs[str(p)] = r
    return r

def check(key, condition, detail=None):
    checks.append(dict(key=key, passed=bool(condition), detail=detail))

def tree(p):
    return ast.parse(Path(p).read_text())

def dump(t):
    return ast.dump(t, include_attributes=False)

def equivalent(text, p):
    return dump(ast.parse(text)) == dump(tree(p))

def remap_text(s):
    return s.replace(str(P / 'candidate'), str(C)).replace(str(OLD), str(S)).replace('G18', 'G19').replace("admission['generation']==18", "admission['generation']==19")

def literal_assignment(t, name):
    a = [n for n in t.body if isinstance(n, ast.Assign) and any(isinstance(v, ast.Name) and v.id == name for v in n.targets)]
    assert len(a) == 1
    return ast.literal_eval(a[0].value)

def functions(t):
    return {n.name: dump(n) for n in t.body if isinstance(n, ast.FunctionDef) and n.name.startswith('test_')}
