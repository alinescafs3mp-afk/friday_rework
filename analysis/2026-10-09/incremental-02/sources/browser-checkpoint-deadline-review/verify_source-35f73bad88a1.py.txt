def sha(data): return hashlib.sha256(data).hexdigest()
def check(path,expected,size=None,mode=None):

def reconstruct(group):
    lines=(SOURCE/(group+'.patch')).read_bytes().splitlines(keepends=True)
    base=Path(pins['baseline_'+group]); index=0; records=[]
    while index<len(lines):
        assert lines[index].startswith(b'diff --git ')
        end=index+1
        while end<len(lines) and not lines[end].startswith(b'diff --git '): end+=1
        block=lines[index:end]
        header=next(i for i,line in enumerate(block) if line.startswith(b'--- '))
        old=block[header][4:].strip().decode(); new=block[header+1][4:].strip().decode()
        assert new.startswith('b/'); rel=new[2:]
        assert '..' not in Path(rel).parts and old in ('a/'+rel,'/dev/null')
        if old=='/dev/null': assert not (base/rel).exists()
        before=(base/rel).read_bytes() if old!='/dev/null' else b''
        original=before.splitlines(keepends=True); output=[]; cursor=0; pos=header+2
        while pos<len(block):
            match=re.fullmatch(rb'@@ -(\d+)(?:,(\d+))? \+(\d+)(?:,(\d+))? @@[^\n]*\n',block[pos]); assert match
            start,count,new_start,new_count=[int(match[k] or b'1') for k in range(1,5)]
            offset=start-1 if count else start
            assert cursor<=offset<=len(original)
            output.extend(original[cursor:offset]); cursor=offset
            assert len(output)==(new_start-1 if new_count else new_start)
            consumed=produced=0; pos+=1
            while consumed<count or produced<new_count:
                sign,payload=block[pos][:1],block[pos][1:]; assert sign in (b' ',b'-',b'+')
                if sign in (b' ',b'-'):
                    assert original[cursor]==payload,(group,rel,cursor)
                    cursor+=1; consumed+=1
                if sign in (b' ',b'+'): output.append(payload); produced+=1
                pos+=1
            assert consumed==count and produced==new_count
        output.extend(original[cursor:]); after=b''.join(output)
        assert after==(SOURCE/group/rel).read_bytes(),(group,rel)
        records.append({'group':group,'path':rel,'before_sha256':sha(before),'sha256':sha(after),'bytes':len(after),'exact':True})
        index=end
    assert {r['path'] for r in records}=={p[len(group)+1:] for p in pins['files'] if p.startswith(group+'/')}
    return records
