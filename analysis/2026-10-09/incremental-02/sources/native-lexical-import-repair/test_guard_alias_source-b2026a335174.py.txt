"""SOURCE ONLY / NOT_RUN. Exact extracted guard wrapper/audit integration controls.
No guard top-level body is loaded. Explicit audit dispatch models the real open
hook; this is NOT an actual guarded native launch or independent acceptance.
Separate owner admission is required to execute even these private-FD controls.
"""
import ast, importlib.util, os, pathlib, stat
import pytest

PRODUCER = pathlib.Path(__file__).parent / 'candidate/producer'


def alias_fixture(tmp_path, monkeypatch):
    path = PRODUCER / 'test_native_lexical_source.py'
    spec = importlib.util.spec_from_file_location('lexical_fixture_source', path)
    obj = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(obj)
    return obj.fixture_alias(tmp_path, monkeypatch)


class GuardRefusal(Exception):
    pass


def extracted_guard(api, m, native, tmp_path, entry):
    tree = ast.parse((PRODUCER / entry).read_text())
    names = {'own', 'allowed', 'credential', 'readable', 'check_open', 'guarded_open', 'audit'}
    body = [n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name in names]
    assert {n.name for n in body} == names
    def deny(reason):
        raise GuardRefusal(reason)
    def unexpected(*args, **kwargs):
        pytest.fail('alias traversal must not use absent archive or descriptor-content routes')
    namespace = {'os': os, 'stat': stat, 'pathlib': pathlib, 'R': tmp_path / 'run',
        'read_roots': (str(native),), 'read_files': set(), 'path_manifest': m,
        'path_api': {'failure_only_open': api.failure_only_open, 'ProbeRefusal': api.ProbeRefusal},
        'deny': deny, 'deny_unreviewed_file_read': lambda target: deny('UNREVIEWED_FILE_READ'),
        'check_fd': unexpected, 'record': unexpected}
    exec(compile(ast.Module(body=body, type_ignores=[]), str(PRODUCER / entry), 'exec'), namespace)
    flags_seen, canonical_seen = [], []
    real_open = os.open
    def audit_then_real_open(path, flags, mode=0o777, *, dir_fd=None):
        assert dir_fd is None  # Production alias supplies absolute descriptor anchors.
        namespace['audit']('open', (path, None, flags))
        flags_seen.append(flags)
        canonical_seen.append(namespace['own'](path))
        return real_open(path, flags, mode)
    namespace['orig_open'] = audit_then_real_open
    return namespace, flags_seen, canonical_seen


@pytest.mark.parametrize('entry', ['guard.py', 'native_guard.py'])
@pytest.mark.parametrize('route', ['audit_only', 'guarded_wrapper_and_audit'])
def test_alias_all_components_survive_existing_guard_policy(tmp_path, monkeypatch, entry, route):
    api, m, native, raw = alias_fixture(tmp_path, monkeypatch)
    ns, flags, targets = extracted_guard(api, m, native, tmp_path, entry)
    def forbidden_absence(*args, **kwargs):
        pytest.fail('alias cannot borrow failure-only absent-leaf permission')
    monkeypatch.setattr(api, 'absent_leaf', forbidden_absence)
    monkeypatch.setattr(api, '_open', ns['orig_open'] if route == 'audit_only' else ns['guarded_open'])
    api.verify_native_alias(m, raw, native)
    assert len(flags) == len(m['native_lexical_alias']['steps'])
    assert all(f == os.O_PATH | os.O_DIRECTORY | os.O_NOFOLLOW | os.O_CLOEXEC for f in flags)
    assert targets[-5:] == [str(native), str(native / 'tests'), str(native / 'tests/tools'), str(native / 'tests'), str(native)]
    assert all(os.path.isdir(p) for p in targets) and api.ARCHIVE not in targets


@pytest.mark.parametrize('entry', ['guard.py', 'native_guard.py'])
@pytest.mark.parametrize('kind', ['same_target_symlink', 'binding_replacement'])
def test_guard_does_not_hide_alias_symlink_or_binding_drift(tmp_path, monkeypatch, entry, kind):
    api, m, native, raw = alias_fixture(tmp_path, monkeypatch)
    ns, flags, targets = extracted_guard(api, m, native, tmp_path, entry)
    tools = native / 'tests/tools'
    if kind == 'same_target_symlink':
        tools.rename(tools.with_name('former-tools'))
        tools.symlink_to(tools.with_name('former-tools'), target_is_directory=True)
        monkeypatch.setattr(api, '_open', ns['guarded_open'])
    else:
        actual = ns['guarded_open']
        def replace_after_open(path, openflags):
            fd = actual(path, openflags)
            if len(flags) == len(m['native_lexical_alias']['steps']):
                tools.rename(tools.with_name('former-tools'))
                tools.mkdir()
            return fd
        monkeypatch.setattr(api, '_open', replace_after_open)
    with pytest.raises(api.ProbeRefusal, match='NATIVE_ALIAS_(COMPONENT_ERROR|BINDING_DRIFT)'):
        api.verify_native_alias(m, raw, native)


@pytest.mark.parametrize('entry', ['guard.py', 'native_guard.py'])
def test_alias_metadata_permission_never_grants_adjacent_content_or_write(tmp_path, monkeypatch, entry):
    api, m, native, raw = alias_fixture(tmp_path, monkeypatch)
    ns, flags, targets = extracted_guard(api, m, native, tmp_path, entry)
    unknown = tmp_path / 'unreviewed-content'
    unknown.write_text('private control data')
    with pytest.raises(GuardRefusal, match='UNREVIEWED_FILE_READ'):
        ns['guarded_open'](unknown, os.O_RDONLY)
    for mode in (os.O_WRONLY, os.O_RDWR, os.O_CREAT, os.O_TRUNC, os.O_APPEND):
        with pytest.raises(GuardRefusal, match='OUTSIDE_WRITE'):
            ns['guarded_open'](native, mode)
    assert flags == []
