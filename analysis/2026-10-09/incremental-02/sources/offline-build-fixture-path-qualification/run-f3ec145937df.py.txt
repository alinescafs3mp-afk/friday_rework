def read_bytes(path):
    fd = RAW['open'](path, os.O_RDONLY | os.O_NOFOLLOW | os.O_CLOEXEC)
    try:
        require(stat.S_ISREG(RAW['fstat'](fd).st_mode), 'nonregular_pinned_input')
        chunks = []
        while True:
            block = RAW['read'](fd, 131072)
            if not block:
                return b''.join(chunks)
            chunks.append(block)
    finally:
        RAW['close'](fd)

def write_json(fd, value):
    data = (json.dumps(value,indent=2)+'\n').encode()
    while data:
        n = RAW['write'](fd,data)
        require(n > 0,'short_result_write')
        data = data[n:]
    RAW['fsync'](fd)

def string_path(value):
    value = RAW['fspath'](value) if not isinstance(value,(str,bytes)) else value
    return value.decode(FSENC,'surrogateescape') if isinstance(value,bytes) else value


def canonical(path, follow_leaf=True):
    """Resolve using captured syscalls, including symlinks before '..'."""
    require(path.startswith('/') and '\0' not in path, 'invalid_guard_path')
    todo = path.split('/')[1:]
    resolved = []
    links = 0
    while todo:
        part = todo.pop(0)
        if part in ('','.'):
            continue
        if part == '..':
            if resolved:
                resolved.pop()
            continue
        current = '/' + '/'.join(resolved+[part])
        if follow_leaf or any(x not in ('','.') for x in todo):
            try:
                s = RAW['stat'](current,follow_symlinks=False)
            except FileNotFoundError:
                s = None
            if s is not None and stat.S_ISLNK(s.st_mode):
                links += 1
                require(links <= 40, 'guard_symlink_cycle')
                target = RAW['readlink'](current)
                if target.startswith('/'):
                    resolved = []
                todo = target.split('/') + todo
                continue
        resolved.append(part)
    return '/' + '/'.join(resolved)


def writable(path):
    return any(path == root or path.startswith(root+'/') for root in WRITABLE)


def fd_path(fd):
    require(isinstance(fd,int) and not isinstance(fd,bool) and fd >= 0, 'invalid_guard_fd')
    st = RAW['fstat'](fd)
    path = RAW['readlink']('/proc/self/fd/'+str(fd))
    if path.endswith(' (deleted)') and st.st_nlink == 0:
        path = path[:-10]
    require(path.startswith('/'), 'nonfilesystem_guard_fd')
    return canonical(path), st


def path_ok(raw, dir_fd=None, follow_leaf=True):
    if isinstance(raw,int):
        path, _ = fd_path(raw)
        return writable(path)
    path = string_path(raw)
    if not path.startswith('/'):
        if dir_fd is None or dir_fd == -1:
            base = RAW['getcwd']()
        else:
            base, st = fd_path(dir_fd)
            require(stat.S_ISDIR(st.st_mode), 'guard_dirfd_not_directory')
        path = base+'/'+path
    return writable(canonical(path,follow_leaf=follow_leaf))


def check_path(event, raw, dir_fd=None, follow_leaf=True):
    try:
        allowed = path_ok(raw,dir_fd,follow_leaf)
    except BaseException:
        allowed = False
    if not allowed:
        deny(event+':outside_private_write_roots')


def check_fd(event, fd, stdout=False):
    if stdout and fd in stdout_ids:
        s = RAW['fstat'](fd)
        if (s.st_dev,s.st_ino,s.st_mode) == stdout_ids[fd]:
            return
    check_path(event,fd)


# audit(open) omits dir_fd. Check it in os.open's wrapper and suppress only
# that wrapper's immediately corresponding audit event; retain real dir_fd.
checked_open = []
def guarded_open(path, flags, mode=0o777, *, dir_fd=None):
    if flags & WRITE_FLAGS:
        check_path('os.open',path,dir_fd)
    checked_open.append((string_path(path),flags))
    try:
        return RAW['open'](path,flags,mode,dir_fd=dir_fd)
    finally:
        checked_open.pop()
os.open = guarded_open


def fd_wrapper(name, stdout=False):
    original = RAW[name]
    def wrapped(fd,*args,**kwargs):
        check_fd('os.'+name,fd,stdout)
        return original(fd,*args,**kwargs)
    return wrapped
for name in ('fsync','fchmod','ftruncate','pwrite','pwritev'):
    setattr(os,name,fd_wrapper(name))
for name in ('write','writev'):
    setattr(os,name,fd_wrapper(name,stdout=True))
# mkfifo is not consistently covered by a CPython audit event.
def guarded_mkfifo(path,mode=0o666,*,dir_fd=None):
    check_path('os.mkfifo',path,dir_fd,False)
    return RAW['mkfifo'](path,mode,dir_fd=dir_fd)
os.mkfifo = guarded_mkfifo

def guarded_bind(self,address):
    if (id(self) not in unix_sockets or id(self) in bound_sockets or not isinstance(address,str)
            or not address.startswith(PRIVATE+'/pytesttmp/') or len(address.encode(FSENC)) >= 108):
        deny('socket.bind:invalid_private_path')
    check_path('socket.bind',address,follow_leaf=False)
    try:
        RAW['stat'](address,follow_symlinks=False)
    except FileNotFoundError:
        pass
    else:
        deny('socket.bind:existing_leaf')
    bound_sockets.add(id(self))

PROCESS_EVENTS = {'subprocess.Popen','os.system','os.fork','os.forkpty','os.posix_spawn','os.exec',
                  'pty.spawn','os.kill','os.killpg','os.pidfd_open','signal.pthread_kill','signal.pidfd_send_signal'}

def audit(event,args):
    if event in PROCESS_EVENTS:
        deny(event)
    if event == 'open':
        path, mode, flags = args
        if flags & WRITE_FLAGS:
            if checked_open and not isinstance(path,int) and string_path(path) == checked_open[-1][0]:
                return
            check_path('open',path)
    elif event in ('os.mkdir','os.remove','os.rmdir'):
        check_path(event,args[0],args[-1],False)
    elif event == 'os.chmod':
        check_path(event,args[0],args[2])
    elif event in ('os.chown','os.setxattr','os.removexattr','os.putenv','os.unsetenv'):
        # pytest and mock.patch.dict use environment changes without an external
        # process/network channel. Filesystem metadata writes remain guarded.
        if event in ('os.chown','os.setxattr','os.removexattr'):
            deny(event)
    elif event == 'os.utime':
        check_path(event,args[0],args[-1])
    elif event == 'os.truncate':
        check_path(event,args[0])
    elif event in ('os.rename','os.link'):
        check_path(event+':source',args[0],args[2],False)
        check_path(event+':destination',args[1],args[3],False)
    elif event == 'os.symlink':
        check_path(event,args[1],args[2],False)
    elif event == 'socket.__new__':
        obj,family,kind,protocol = args
        if (family != socket.AF_UNIX or kind != socket.SOCK_STREAM or protocol != 0
                or state['active_node'] != 'test_boundary_source.py::test_B4_nonregular_leaf_rejected_without_blocking_read[socket]'
                or state['private_unix_socket_count'] != 0):
            deny(event)
        state['private_unix_socket_count'] += 1
        unix_sockets.add(id(obj))
    elif event == 'socket.bind':
        obj,address = args
        if id(obj) not in bound_sockets:
            deny('socket.bind:unguarded')
        check_path(event,address,follow_leaf=False)
    elif event.startswith('socket.'):
        deny(event)
sys.addaudithook(audit)
