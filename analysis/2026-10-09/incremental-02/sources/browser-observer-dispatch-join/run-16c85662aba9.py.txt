 def read_bytes(path,hasher=None):
  handle=os.open(path,os.O_RDONLY|os.O_NOFOLLOW)
  try:
   data=bytearray()
   while True:
    chunk=os.read(handle,65536)
    if not chunk:return bytes(data) if hasher is None else hasher.hexdigest()
    if hasher is None:data.extend(chunk)
    else:hasher.update(chunk)
  finally:os.close(handle)
