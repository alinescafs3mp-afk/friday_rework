def require(ok, reason):
    if not ok: raise ValueError(reason)


def read_bytes(path):
    path = Path(path)
    fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK | os.O_CLOEXEC)
    try:
        s = os.fstat(fd)
        require(stat.S_ISREG(s.st_mode) and s.st_size <= 128000000, 'NONREGULAR_OR_OVERSIZE_INPUT')
        chunks = []
        while True:
            block = os.read(fd, 131072)
            if not block: break
            chunks.append(block)
        result = b''.join(chunks)
        t = os.fstat(fd); current = path.lstat()
        key = lambda x: (x.st_dev,x.st_ino,x.st_size,x.st_mtime_ns,x.st_ctime_ns,x.st_mode,x.st_uid,x.st_nlink)
        require(key(s) == key(t) == key(current) and len(result) == s.st_size, 'INPUT_READ_DRIFT')
        return result
    finally: os.close(fd)


def digest(data): return hashlib.sha256(data).hexdigest()


def read(path): return json.loads(read_bytes(path))


def put(path, value):
    data = value if isinstance(value, bytes) else (json.dumps(value, indent=2)+'\n').encode()
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW | os.O_CLOEXEC, 0o400)
    with os.fdopen(fd, 'wb') as f:
        f.write(data); f.flush(); os.fsync(f.fileno())


def clock_check(clock, minimum=0):
    require(clock['assignment'] == ASSIGNMENT and type(clock['generation']) is int and clock['generation'] > 0
        and clock['offline_authorized'] is True and clock['budget_seconds'] == 180
        and clock['source_root_mode'] == 'FINAL_PRIVATE_READ_ONLY_TEST_VIEW'
        and clock['final_readonly_test_view_accepted'] is True
        and clock['resources'] == {'cpu':2, 'memory_bytes':2<<30, 'cpu_seconds':60, 'children':0, 'launches':1},
        'EXTERNAL_ORIGINAL180_EXACT_RESOURCE_GRANT_REQUIRED')
    start, end = clock['started_monotonic'], clock['deadline_monotonic']
    require(all(type(x) in (int,float) and math.isfinite(x) for x in (start,end))
        and end-start == 180 and start <= time.monotonic() < end-minimum, 'ORIGINAL180_EXPIRED_OR_RESET')
    require(clock['boot_id'] == Path('/proc/sys/kernel/random/boot_id').read_text().strip(), 'BOOT_CHANGED')


def lock_probe():
    fd = os.open(LOCK, os.O_RDONLY | os.O_NOFOLLOW)
    try:
        s = os.fstat(fd)
        require([s.st_dev,s.st_ino,s.st_uid,s.st_gid,s.st_nlink,s.st_size,stat.S_IMODE(s.st_mode)] == LOCK_ID,
                'LOCK_IDENTITY_CHANGED')
        fcntl.flock(fd, fcntl.LOCK_EX | fcntl.LOCK_NB)
        fcntl.flock(fd, fcntl.LOCK_UN)
    finally: os.close(fd)


def proc_stat(raw):
    fields = raw[raw.rfind(')')+2:].split()
    return dict(pid=int(raw.split(' ',1)[0]), ppid=int(fields[1]), pgid=int(fields[2]),
                session=int(fields[3]), starttime_ticks=int(fields[19]), state=fields[0])


def scan_group(contour):
    remaining, known, errors = [], [], []
    for p in Path('/proc').iterdir():
        if not p.name.isdigit(): continue
        try:
            row = proc_stat((p/'stat').read_text())
            if row['pgid'] == contour['pgid']: remaining.append(row)
            if any(row['pid'] == v['pid'] and row['starttime_ticks'] == v['starttime_ticks'] for v in contour['known']):
                known.append(row)
        except FileNotFoundError: pass
        except Exception as exc: errors.append(dict(path=str(p), error=type(exc).__name__))
    return dict(remaining=remaining, known=known, errors=errors)


def publication(path_sha):
    data = read_bytes(E/'PUBLICATION-MANIFEST.json'); require(digest(data) == path_sha, 'PUBLICATION_DRIFT')
    pub = json.loads(data)
    require(pub['runtime_grant'] is False and pub['source_task'] == 'FRW041-NATIVE-PRODUCER-CONTROLS-REPAIR#1', 'WRONG_PUBLICATION')
    for pin in pub['refs'].values():
        data = read_bytes(pin['path'])
        require(digest(data) == pin['sha256'] and len(data) == pin['bytes'], 'PUBLICATION_REF_DRIFT:'+pin['path'])
    return pub


def compare_inputs(original_bytes, read_bytes):
    pins = json.loads(original_bytes)
    if not isinstance(pins, dict) or not pins:
        raise ValueError('input_pin_inventory')
    rows = {}
    for path, expected in sorted(pins.items()):
        try:
            data = read_bytes(path)
            actual = dict(sha256=hashlib.sha256(data).hexdigest(), size=len(data))
            rows[path] = dict(expected=expected, actual=actual, unchanged=actual == expected)
        except Exception as exc:
            rows[path] = dict(expected=expected, actual=None, unchanged=False,
                              error=type(exc).__name__ + ':' + str(exc))
    return dict(original_pins_sha256=hashlib.sha256(original_bytes).hexdigest(),
                leaf_count=len(rows), rows=rows,
                unchanged=all(row['unchanged'] for row in rows.values()))

def validate_reports(result, expected_count):
    """Fail closed on partial setup/call/teardown, skips or collection errors."""
    errors = []
    nodes = result.get('nodes', [])
    if (len(nodes) != expected_count or len(set(nodes)) != expected_count
            or any(not n.startswith('test_native_input_producer.py::test_') for n in nodes)):
        errors.append('exact_unique_collection')
    collections = result.get('collections', [])
    if not collections or any(r.get('outcome') != 'passed' for r in collections):
        errors.append('collection_incomplete_failed_or_skipped')
    collection_nodes = [r.get('nodeid') for r in collections]
    if (len(collection_nodes) != 2 or set(collection_nodes) != {'', 'test_native_input_producer.py'}):
        errors.append('collection_missing_duplicate_or_foreign')
    reports = result.get('reports', [])
    expected = {(node, when) for node in nodes for when in ('setup', 'call', 'teardown')}
    actual = [(r.get('nodeid'), r.get('when')) for r in reports]
    if len(actual) != len(expected) or len(set(actual)) != len(actual) or set(actual) != expected:
        errors.append('terminal_reports_missing_duplicate_or_foreign')
    if any(r.get('outcome') != 'passed' for r in reports):
        errors.append('test_failed_or_skipped')
    if result.get('exit_code') != 0 or result.get('wrapper_error') or result.get('denied'):
        errors.append('runner_failure_or_denial')
    if result.get('candidate_unchanged') is not True or result.get('all_inputs_unchanged') is not True:
        errors.append('input_integrity_unconfirmed')
    return dict(accepted=not errors, errors=errors, nodes=len(nodes), reports=len(reports),
                classification='PASS' if not errors else 'FAIL')

def contracts():
    # Identical pure-function AST is verified in static-evidence.json.
    # No candidate import occurs in this external post-pytest owner process.
    return compare_inputs, validate_reports

def runtime_bindings():
    runtime = read(E/'runner-runtime-pins.json')
    for alias, resolved in runtime['aliases'].items():
        require(str(Path(alias).resolve(strict=True)) == resolved, 'RUNTIME_ALIAS_DRIFT')
    return runtime


def prepare(clock_sha, publication_sha, review_path, review_sha):
    require(not (OUT/'owner-prelaunch.json').exists(), 'ONE_LAUNCH_ONLY_NO_REPLAY')
    clock_bytes = read_bytes(OUT/'runtime-clock.json'); require(digest(clock_bytes) == clock_sha, 'CLOCK_PIN_DRIFT')
    clock = json.loads(clock_bytes); clock_check(clock, 120)
    pub = publication(publication_sha)
    require(clock['publication_manifest_sha256'] == publication_sha and clock['review_sha256'] == review_sha, 'CLOCK_SOURCE_REVIEW_BINDING')
    rb = read_bytes(review_path); require(digest(rb) == review_sha, 'REVIEW_REF_DRIFT')
    review = json.loads(rb)
    require(review['state'] == 'INDEPENDENT_NATIVE_ADMISSION_PRODUCER_SOURCE_ACCEPTED'
        and review['source_task'] == pub['source_task'] and review['runtime_grant'] is False
        and review['publication_manifest_sha256'] == publication_sha
        and review['producer_package_sha256'] == pub['source_package_sha256']
        and review['offline_runner_source_accept'] is True, 'EXACT_OWNER_CALLER_SOURCE_REVIEW_REQUIRED')
    staging = read(OUT/'staging.json')
    require(staging['publication_manifest_sha256'] == publication_sha
        and staging['review_sha256'] == review_sha and staging['review_path'] == review_path
        and staging['source_package_sha256'] == pub['source_package_sha256'], 'STAGING_BINDING_DRIFT')
    original = read_bytes(OUT/'input-pins.json')
    require(digest(original) == staging['input_pins_sha256'], 'ORIGINAL_INPUT_PIN_BYTES_CHANGED')
    compare, _ = contracts(); before = compare(original, read_bytes)
    require(before['leaf_count'] == LEAVES and before['unchanged'], 'ALL_INPUTS_REQUIRED_BEFORE_LAUNCH')
    lock_probe(); clock_check(clock, 120)
    put(OUT/'input-pins.original.json', original)
    put(OUT/'runtime-clock.original.json', clock_bytes)
    pre = dict(state='OWNER_PRELAUNCH_LOWER_BOUND_RECORDED', argv=ARGV,
        original_clock_sha256=clock_sha, original_input_pins_sha256=digest(original),
        publication_manifest_sha256=publication_sha, review_path=review_path, review_sha256=review_sha,
        boot_id=clock['boot_id'], clock=clock, before=before, launch_limit=1,
        runtime_bindings=runtime_bindings(),
        prelaunch_lower_bound_monotonic=time.monotonic())
    put(OUT/'owner-prelaunch.json', pre)
    print(json.dumps(dict(argv=ARGV, output=str(OUT), deadline_monotonic=clock['deadline_monotonic'])))


def observe(payload):
    """Always reconcile cessation and hashes, even on test failure/timeout.

    Classification and physical closure are distinct. No kill, restart, retry,
    clock grant, successful-result fabrication or lock transfer is performed.
    """
    require(not (OUT/'owner-closure.json').exists(), 'CLOSURE_ALREADY_RECORDED_NO_REPLAY')
    pre = read(OUT/'owner-prelaunch.json')
    back = json.loads(payload)
    back['owner_received_backend_result_monotonic'] = time.monotonic()
    put(OUT/'backend-result.json', back)
    closure_errors, qualification_errors = [], []
    evidence = dict(backend=back, original_clock_sha256=pre['original_clock_sha256'])
    def attempt(name, fn, errors):
        try: evidence[name] = fn()
        except Exception as exc: errors.append(name+':'+type(exc).__name__+':'+str(exc))
    def terminal():
        result = back['terminal']; initial = back['initial']
        require(type(result.get('exit_code')) is int and not result.get('session_id'), 'BACKEND_NOT_TERMINAL_REAPED')
        handle = 'session:'+str(initial['session_id']) if initial.get('session_id') else 'chunk:'+str(initial['chunk_id'])
        require(back['backend_handle'] == handle and back['argv'] == pre['argv'] == ARGV, 'BACKEND_HANDLE_OR_COMMAND_MISMATCH')
        return dict(backend_handle=handle, exit_code=result['exit_code'], terminal_reaped=True)
    attempt('terminal', terminal, closure_errors)
    def identity_group():
        raw = read_bytes(OUT/'backend-start-stat.txt').decode(); launch = proc_stat(raw)
        parent = proc_stat(read_bytes(OUT/'backend-parent-stat.txt').decode())
        require(launch['ppid'] == parent['pid'] and launch['pid'] > 1 and launch['starttime_ticks'] > 0, 'BACKEND_BIRTH_MISMATCH')
        birth = read(RESULTS/'process-birth.json'); timeout = birth['parent']; guard = birth['self']
        require(birth['self_executable'] == pre['runtime_bindings']['aliases'][ARGV[3]]
            and birth['parent_executable'] == ARGV[0] and birth['parent_argv'] == ARGV
            and birth['boot_id'] == pre['boot_id']
            and timeout['pid'] == launch['pid'] and timeout['starttime_ticks'] == launch['starttime_ticks']
            and timeout['ppid'] == parent['pid'] and timeout['pgid'] == timeout['pid']
            and guard['ppid'] == timeout['pid'] and guard['pgid'] == timeout['pgid']
            and pre['prelaunch_lower_bound_monotonic'] <= birth['started_monotonic'] <= birth['completed_monotonic']
            < pre['clock']['deadline_monotonic'], 'GNU_GROUP_BIRTH_UNCONFIRMED')
        evidence['process_birth'] = birth
        scan = scan_group(dict(pgid=timeout['pgid'], known=[timeout,guard]))
        require(not any(scan.values()), 'WHOLE_GROUP_OR_KNOWN_IDENTITY_REMAINS:'+json.dumps(scan))
        return dict(scan=scan, backend_parent=parent, backend_shell_before_exec=launch)
    attempt('group', identity_group, closure_errors)
    def integrity():
        original = read_bytes(OUT/'input-pins.original.json')
        require(digest(original) == pre['original_input_pins_sha256'], 'RETAINED_ORIGINAL_PIN_BYTES_DRIFT')
        compare, _ = contracts(); after = compare(original, read_bytes)
        evidence['input_after'] = after
        require(after['leaf_count'] == LEAVES and after['unchanged'], 'POST_RUN_ALL_INPUT_INTEGRITY_FAILURE')
        require(read_bytes(OUT/'input-pins.json') == original, 'INPUT_PIN_FILE_REWRITTEN')
        require(digest(read_bytes(OUT/'runtime-clock.json')) == pre['original_clock_sha256']
            and read_bytes(OUT/'runtime-clock.json') == read_bytes(OUT/'runtime-clock.original.json'), 'ORIGINAL_CLOCK_BYTES_DRIFT')
        publication(pre['publication_manifest_sha256'])
        require(runtime_bindings() == pre['runtime_bindings'], 'RUNTIME_BINDINGS_DRIFT')
        return True
    attempt('integrity', integrity, qualification_errors)
    def reports():
        _, validate = contracts(); result = read(RESULTS/'pytest-results.json')
        require(result['clock'] == pre['clock'] and result['process_birth_snapshot'] == evidence.get('process_birth'), 'RUNNER_IDENTITY_MISMATCH')
        require(result['resources'] == {'affinity':[0,1], 'as':[2<<30,2<<30], 'cpu':[60,60]}, 'RESOURCES_UNCONFIRMED')
        require(result['input_pins_sha256'] == pre['original_input_pins_sha256'], 'RUNNER_ORIGINAL_PINS_MISMATCH')
        verdict = validate(result, COUNT)
        evidence['report_validation'] = verdict
        require(verdict['accepted'] and back['terminal']['exit_code'] == 0, 'QUALIFICATION_FAIL')
        return True
    attempt('reports', reports, qualification_errors)
    attempt('lock_return', lambda: (lock_probe(), True)[1], closure_errors)
    attempt('original_deadline', lambda: (clock_check(pre['clock']), time.monotonic())[1], closure_errors)
    if back.get('caller_error'): closure_errors.append('CALLER_ERROR:'+back['caller_error'])
    upper = time.monotonic()
    if not pre['clock']['started_monotonic'] <= upper < pre['clock']['deadline_monotonic']:
        closure_errors.append('CESSATION_NOT_EVIDENCED_BEFORE_ORIGINAL180')
    state = 'STOP_UNCONFIRMED' if closure_errors else 'CLOSED'
    passed = state == 'CLOSED' and not qualification_errors
    result = dict(state=state, qualification='PASS' if passed else 'FAIL', runtime_grant=False,
        closure_errors=closure_errors, qualification_errors=qualification_errors, evidence=evidence,
        completed_monotonic=upper, replay_allowed=False, native_executed=False)
    put(OUT/'owner-closure.json', result)
    print(json.dumps(dict(state=state, qualification=result['qualification'], path=str(OUT/'owner-closure.json'))))
    return 0 if passed else 1


if __name__ == '__main__':
    require(os.getuid() == os.geteuid() == 1000, 'NONROOT_ONLY')
    os.sched_setaffinity(0, {0,1}); resource.setrlimit(resource.RLIMIT_AS, (2<<30,2<<30))
    resource.setrlimit(resource.RLIMIT_CPU, (60,60))
    if sys.argv[1] == 'prepare': prepare(*sys.argv[2:])
    elif sys.argv[1] == 'observe': raise SystemExit(observe(sys.argv[2]))
    else: raise ValueError('EXACT_PREPARE_OR_OBSERVE_ONLY')
