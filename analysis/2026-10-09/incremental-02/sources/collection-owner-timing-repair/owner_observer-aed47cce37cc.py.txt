def require(ok, message):
    if not ok:
        raise ValueError(message)


def sha(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as f:
        for chunk in iter(lambda: f.read(1 << 20), b''):
            h.update(chunk)
    return h.hexdigest()


def read(path):
    return json.loads(Path(path).read_bytes())


def put(path, value, mode=0o400):
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, mode)
    with os.fdopen(fd, 'w') as f:
        json.dump(value, f, indent=2); f.write('\n'); f.flush(); os.fsync(f.fileno())
    fd = os.open(Path(path).parent, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
    try:
        os.fsync(fd)
    finally:
        os.close(fd)


def private(path, mode):
    s = Path(path).lstat()
    require(not Path(path).is_symlink() and s.st_uid == os.getuid() and
            stat.S_IMODE(s.st_mode) == mode, 'FOREIGN_OR_MUTABLE_PATH:' + str(path))
    return s


def finite(value):
    return type(value) in (int, float) and math.isfinite(value) and value > 0


def clock_check(clock):
    require(all(finite(clock.get(k)) for k in CORE[3:]), 'INVALID_ORIGINAL_CLOCK')
    require(clock['start_monotonic'] <= time.monotonic() < clock['deadline_monotonic'] and
            clock['deadline_monotonic'] == clock['start_monotonic'] + clock['original_budget_seconds'] and
            clock['original_budget_seconds'] == 900, 'EXPIRED_OR_RESET_ORIGINAL900')
    require(clock.get('boot_id') == Path('/proc/sys/kernel/random/boot_id').read_text().strip(), 'BOOT_CHANGED')
    require(isinstance(clock.get('assignment'), str) and clock['assignment'] and
            type(clock.get('generation')) is int and clock['generation'] >= 1, 'OWNER_IDENTITY_REQUIRED')
    require(clock.get('execution_authorized') is True and clock.get('independent_source_review') == 'PASS',
            'NO_SEPARATE_EXECUTION_GRANT')
    require(clock.get('scope') == 'FINITE_TWO_POSITIVE_CANARIES_AND_EXACT_COLLECTION_ONLY' and
            clock.get('reviewed_source_pins_sha256') == SPEC['source_pins_sha256'], 'WRONG_COHORT_SCOPE')
    require(clock.get('resources') == {'cpu': 2, 'memory_bytes': 2 << 30, 'children': 0, 'max_phases': 4},
            'WRONG_RESOURCES_OR_PHASE_COUNT')


def lock_probe():
    fd = os.open(SPEC['canonical_lock']['path'], os.O_RDONLY | os.O_NOFOLLOW)
    try:
        s = os.fstat(fd)
        require([s.st_dev, s.st_ino, s.st_uid, s.st_gid, s.st_nlink, s.st_size,
                 stat.S_IMODE(s.st_mode)] == SPEC['canonical_lock']['identity'], 'LOCK_IDENTITY_CHANGED')
        fcntl.flock(fd, fcntl.LOCK_EX | fcntl.LOCK_NB)
        fcntl.flock(fd, fcntl.LOCK_UN)
    finally:
        os.close(fd)


def file_inventory(root):
    """Frozen corpora contain regular files and directories; no symlinks at intake."""
    found = set()
    for base, dirs, files in os.walk(root, followlinks=False):
        for name in dirs + files:
            p = Path(base) / name; s = p.lstat()
            require(not stat.S_ISLNK(s.st_mode), 'UNREVIEWED_CORPUS_SYMLINK:' + str(p))
            if stat.S_ISREG(s.st_mode):
                found.add(str(p.relative_to(root)))
            else:
                require(stat.S_ISDIR(s.st_mode), 'UNREVIEWED_CORPUS_TYPE:' + str(p))
    return found

def cohort(out):
    private(out, 0o700); c = read(out / 'original-clock.json'); clock_check(c)
    private(out / 'original-clock.json', 0o400)
    private(out / 'input-pins.json', 0o400); private(out / 'preflight.json', 0o400)
    pre = read(out / 'preflight.json')
    require(sha(out / 'original-clock.json') == pre['clock_sha256'], 'ORIGINAL_CLOCK_BYTES_CHANGED')
    require(sha(pre['admission']) == pre['admission_sha256'], 'OWNER_ADMISSION_CHANGED')
    source_review(pre['source_review'], pre['source_review_sha256'])
    require(inputs() == read(out / 'input-pins.json'), 'FULL_INPUT_OR_PATH_DRIFT')
    return c, pre


def source_review(path, digest):
    # Final review is supplied externally AFTER publication; baseline review cannot
    # accept these changed bytes. No placeholder hash or self-issued grant.
    require(isinstance(path, str) and Path(path).is_absolute() and
            Path(path).resolve(strict=True) == Path(path), 'EXACT_EXTERNAL_SOURCE_REVIEW_REQUIRED')
    private(path, 0o400)
    require(isinstance(digest, str) and len(digest) == 64 and sha(path) == digest, 'SOURCE_REVIEW_HASH_MISMATCH')
    review = read(path)
    require(review.get('state') == 'INDEPENDENT_COLLECTION_RECEIPT_AND_OWNER_HELPER_SOURCE_ACCEPTED' and
            review.get('source') == str(SOURCE) and review.get('helper') == str(HERE) and
            review.get('source_pins_sha256') == SPEC['source_pins_sha256'] and
            review.get('helper_pins_sha256') == sha(HERE / 'source-pins.json') and
            review.get('helper_sha256') == sha(HERE / 'owner_observer.py') and
            review.get('input_spec_sha256') == sha(HERE / 'input-spec.json') and
            review.get('receipt_schema') == 2, 'CHANGED_PAIR_NOT_INDEPENDENTLY_REVIEWED')


def initialize(out, original, original_sha, admission, admission_sha):
    """Consume an existing separately granted clock; never creates an execution budget."""
    require(not out.exists(), 'COHORT_ALREADY_EXISTS_NO_RETRY')
    private(original, 0o400); private(admission, 0o400)
    require(sha(original) == original_sha and sha(admission) == admission_sha, 'OWNER_INPUT_PIN_MISMATCH')
    require(out.resolve(strict=False) == out, 'CANONICAL_FRESH_OWNER_ROOT_REQUIRED')
    c = read(original); clock_check(c); grant = read(admission)
    require(grant.get('state') == 'INDEPENDENT_OWNER_HELPER_AND_COLLECTION_ADMITTED' and
            grant.get('original_clock_sha256') == original_sha and grant.get('output_root') == str(out) and
            grant.get('helper_sha256') == sha(HERE / 'owner_observer.py') and
            grant.get('input_spec_sha256') == sha(HERE / 'input-spec.json') and
            grant.get('helper_pins_sha256') == sha(HERE / 'source-pins.json') and
            grant.get('source_pins_sha256') == SPEC['source_pins_sha256'] and
            grant.get('phases') == PHASES and grant.get('execution_authorized') is True,
            'INDEPENDENT_ROOT_ADMISSION_REQUIRED')
    source_review(grant.get('source_review'), grant.get('source_review_sha256'))
    reserve = grant.get('reconciliation_reserve_seconds')
    require(finite(reserve), 'REVIEWED_RECONCILIATION_RESERVE_REQUIRED')
    out.mkdir(mode=0o700)
    # Exclusive original byte copy, not a reset clock or helper-issued permission.
    fd = os.open(out / 'original-clock.json', os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o400)
    with os.fdopen(fd, 'wb') as f:
        f.write(original.read_bytes()); f.flush(); os.fsync(f.fileno())
    expanded = inputs(); lock_probe(); clock_check(c)
    require(c['deadline_monotonic'] - time.monotonic() > sum(p['outer'] + reserve for p in PHASES), 'INSUFFICIENT_WHOLE_COHORT_BUDGET')
    put(out / 'input-pins.json', expanded)
    put(out / 'preflight.json', {'clock_sha256': original_sha, 'admission': str(admission),
        'admission_sha256': admission_sha, 'reserve_seconds': reserve,
        'source_review': grant['source_review'], 'source_review_sha256': grant['source_review_sha256'],
        'canonical_lock': 'exact acquire/unlock/close returned', 'preflight_monotonic': time.monotonic(),
        'complete_expanded_input_count': len(expanded['hashes']), 'corpus_counts': SPEC['corpus_counts'],
        'affinity_available': sorted(os.sched_getaffinity(0)),
        'pressure': {k: Path('/proc/pressure/' + k).read_text() for k in ('cpu', 'memory', 'io')}})
    print(json.dumps({'prepared_cohort': str(out), 'launch_performed': False}))

            done = read(dest / 'cessation.json')
            require(done['status'] in ('CANARY_PHASE_EVIDENCED', 'COLLECTION_EVIDENCED') and
                    done['next_phase_admissible'] is True, 'PREVIOUS_PHASE_NOT_CLOSED_NO_RETRY')
        else:
            require(not dest.exists(), 'PHASE_ALREADY_PREPARED_NO_RETRY')
    require(c['deadline_monotonic'] - time.monotonic() > sum(p['outer'] + pre['reserve_seconds'] for p in PHASES[index:]),
            'INSUFFICIENT_ORIGINAL_REMAINING_WITH_RECONCILIATION')

    run = parent / 'run'; run.mkdir(mode=0o700)

        phase_clock.pop('collect_only', None)

def scan_group(contour):
    remaining, known, errors = [], [], []
    for p in Path('/proc').iterdir():
        if not p.name.isdigit():
            continue
        try:
            raw = (p / 'stat').read_text(); fields = raw[raw.rfind(')') + 2:].split()
            pid, born, pgid = int(p.name), int(fields[19]), int(fields[2])
            if pgid == contour['timeout']['pgid']:
                remaining.append({'pid': pid, 'start_ticks': born, 'state': fields[0]})
            if any(pid == v['pid'] and born == v['start_ticks'] for v in (contour['timeout'], contour['guard'])):
                known.append(pid)
        except FileNotFoundError:
            pass
        except Exception as e:
            errors.append({'path': str(p), 'error': type(e).__name__})
    return {'remaining': remaining, 'known': known, 'errors': errors}


def node_key(identity):
    require(isinstance(identity, dict) and set(identity) ==
            {'path', 'object_path', 'kind', 'type', 'nodeid', 'source_relative_path'}, 'MISSING_OR_EXTRA_NODE_IDENTITY')
    path = identity['path']; object_path = identity['object_path']; kind = identity['kind']
    require(isinstance(path, str) and isinstance(object_path, str) and Path(object_path).is_absolute() and
            os.path.normpath(object_path) == object_path and Path(object_path).resolve(strict=True) == Path(path) and
            Path(path).is_absolute() and Path(path).resolve(strict=True) == Path(path),
            'NONCANONICAL_NODE_PATH')
    require(kind in ('session', 'directory', 'module', 'class', 'function', 'item') and
            isinstance(identity['type'], str) and '.' in identity['type'] and
            isinstance(identity['nodeid'], str) and
            identity['source_relative_path'] == os.path.relpath(path, SOURCE), 'NODE_KIND_OR_RELATIVE_PATH')
    # Empty and colliding raw nodeids are legitimate. Never parse a file from one.
    return object_path, kind, identity['nodeid']

def observe(out, index):
