"""Prepared source only. No install/start/stop, clock creation, retry or execution.

Future explicit owner invocation copies separately authorized original clock
bytes and publishes one exact admission. Existing/partial/consumed paths refuse.
Pinned consumer functions run only against a closed in-memory observation view.
This preflight proves data compatibility, never kernel/native acceptance.
"""
import argparse
import ast
import hashlib
import json
import math
import os
from pathlib import Path, PurePosixPath
import re
import stat
import time
from types import SimpleNamespace

HERE = Path(__file__).absolute().parent
SOURCE = Path('<ASTRA_WORK>/offline-build-native-admission-repair/candidate')
OWNER_ROOT = Path('<ASTRA_WORK>')
SOURCE_TASK = 'FRW041-OFFLINE-BUILD-FINAL-JOIN-SOL#1'
ASSIGNMENT = 'FRW041-NATIVE-OFFLINE-BUILD-RUN'
PACKAGE_SHA = 'd881baa2158b5522fbba4802c54368a6422b889f56c67c2401f94979d58896b4'
READ_FLAGS = os.O_RDONLY | os.O_NONBLOCK | os.O_NOFOLLOW | os.O_CLOEXEC


def require(condition, reason):
    if not condition:
        raise RuntimeError(reason)


def sha(data):
    return hashlib.sha256(data).hexdigest()


def encode(value):
    return (json.dumps(value, indent=2, sort_keys=True, allow_nan=False) + '\n').encode()


def unique_json(data):
    def pairs(rows):
        result = {}
        for key, value in rows:
            require(key not in result, 'duplicate_json_key')
            result[key] = value
        return result
    return json.loads(data, object_pairs_hook=pairs,
                      parse_constant=lambda _: require(False, 'nonfinite_json'))


def hash_string(value):
    return type(value) is str and re.fullmatch('[0-9a-f]{64}', value) is not None


def clean_path(value):
    require(type(value) is str and value.startswith('/') and str(PurePosixPath(value)) == value
            and '..' not in PurePosixPath(value).parts, 'absolute_clean_path_required')
    return Path(value)


def parent_fd(path, uid):
    path = clean_path(str(path))
    fd = os.open('/', READ_FLAGS | os.O_DIRECTORY)
    identities = []
    try:
        for part in path.parts[1:-1]:
            nxt = os.open(part, READ_FLAGS | os.O_DIRECTORY, dir_fd=fd)
            os.close(fd)
            fd = nxt
            s = os.fstat(fd)
            require(stat.S_ISDIR(s.st_mode) and s.st_uid in (0, uid)
                    and not s.st_mode & 0o022, 'unsafe_input_ancestor')
            identities.append((s.st_dev, s.st_ino))
        return fd, path.name, identities
    except BaseException:
        os.close(fd)
        raise


def metadata(s):
    return (s.st_dev, s.st_ino, s.st_mode, s.st_uid, s.st_gid, s.st_nlink,
            s.st_size, s.st_mtime_ns, s.st_ctime_ns)


def read_file(path, uid=1000, mode=0o400, maximum=40 << 20, single_link=True):
    parent, leaf, ancestors = parent_fd(path, uid)
    fd = None
    try:
        fd = os.open(leaf, READ_FLAGS, dir_fd=parent)
        a = os.fstat(fd)
        require(stat.S_ISREG(a.st_mode) and a.st_uid == uid and (uid != 0 or a.st_gid == 0)
                and (not single_link or a.st_nlink == 1) and stat.S_IMODE(a.st_mode) == mode
                and a.st_size <= maximum, 'input_metadata_shape')
        chunks = []
        left = a.st_size + 1
        while left:
            block = os.read(fd, min(left, 65536))
            if not block:
                break
            chunks.append(block)
            left -= len(block)
        data = b''.join(chunks)
        require(metadata(a) == metadata(os.fstat(fd)) == metadata(os.stat(leaf, dir_fd=parent, follow_symlinks=False))
                and len(data) == a.st_size, 'input_metadata_drift')
        check, _, current = parent_fd(path, uid)
        os.close(check)
        require(current == ancestors, 'input_ancestor_drift')
        return data, a
    finally:
        if fd is not None:
            os.close(fd)
        os.close(parent)


def absent(path):
    parent, leaf, _ = parent_fd(path, 1000)
    try:
        try:
            os.stat(leaf, dir_fd=parent, follow_symlinks=False)
        except FileNotFoundError:
            return
        raise RuntimeError('existing_or_consumed_path:' + str(path))
    finally:
        os.close(parent)


def pin_rows(pins):
    require(type(pins) is dict and bool(pins), 'package_inventory')
    for name, row in pins.items():
        require(type(name) is str and name not in ('', '.', '..', 'package-pins.json')
                and '/' not in name and type(row) is dict and set(row) == {'size', 'sha256'}
                and type(row['size']) is int and row['size'] >= 0 and hash_string(row['sha256']),
                'package_metadata_shape')


def string_map(pins, package_bytes):
    pin_rows(pins)
    return {**{name: row['sha256'] for name, row in pins.items()},
            'package-pins.json': sha(package_bytes)}


def admission_data(identity, source_hashes, review_path, review_hash, clock_bytes):
    require(type(source_hashes) is dict and all(hash_string(v) for v in source_hashes.values()),
            'string_sha_map_required')
    return dict(allow_one_offline_native_build=True, runtime_authorized=True, source_task=SOURCE_TASK,
                independent_review_path=str(review_path), independent_review_sha256=review_hash,
                source_sha256=source_hashes, clock_path=identity['trial_clock_path'],
                clock_sha256=sha(clock_bytes), live_directory=identity['live_directory'])


def clock_contract(clock, identity, now, boot):
    require(type(clock) is dict and clock.get('assignment') == ASSIGNMENT
            and type(clock.get('generation')) is int and clock['generation'] > 0
            and clock.get('native_effects_authorized_by') == 'Astra'
            and type(clock.get('budget_seconds')) is int and clock['budget_seconds'] == 1800
            and clock.get('nonce') == identity['nonce'] and clock.get('unit') == identity['unit']
            and clock.get('boot_id') == boot, 'original_clock_contour')
    require(all(type(clock.get(k)) in (int, float) and math.isfinite(clock[k])
                for k in ('started_monotonic', 'deadline_monotonic'))
            and clock['deadline_monotonic'] == clock['started_monotonic'] + 1800
            and clock['started_monotonic'] <= now < clock['deadline_monotonic']
            and now + 1665 < clock['deadline_monotonic'], 'original_clock_window')


def actual_consumers(source_bytes, contract, identity, files, stats, resolution, admission, clock, receipt, now, boot, host):
    """No module import or native API. Exact pinned AST, closed data facades only.

    Missing snapshot paths refuse; there is no real filesystem fallback.
    Observations are actual collected inputs, except not-yet-published exact
    clock/admission stat400 and fresh-path absence. These are publication targets.
    """
    class DataPath(PurePosixPath):
        def read_bytes(self):
            require(str(self) in files, 'consumer_unobserved_read:' + str(self))
            return files[str(self)]
        def read_text(self, *args, **kwargs):
            return self.read_bytes().decode()
        def stat(self):
            require(str(self) in stats, 'consumer_unobserved_stat:' + str(self))
            return stats[str(self)]
        def resolve(self, strict=False):
            require(str(self) in resolution, 'consumer_unobserved_resolution:' + str(self))
            return DataPath(resolution[str(self)])
        def exists(self):
            require(str(self) in host['exists'], 'consumer_unobserved_existence')
            return host['exists'][str(self)]
        def is_symlink(self):
            require(str(self) == identity['live_directory'], 'consumer_unobserved_symlink')
            return False
    def load(name, functions, extra):
        tree = ast.parse(source_bytes[name].decode(), filename=str(SOURCE / name))
        selected = []
        for function in functions:
            nodes = [n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == function]
            require(len(nodes) == 1 and sha(ast.dump(nodes[0], include_attributes=False).encode())
                    == contract['functions'][name + ':' + function], 'consumer_AST_drift')
            selected.append(nodes[0])
        env = dict(Path=DataPath, ROOT=DataPath(SOURCE), IDENTITY=identity, SOURCE_TASK=SOURCE_TASK,
                   BUSCTL='/usr/bin/busctl',
                   UNIT=identity['unit'], NONCE=identity['nonce'], LIVE=DataPath(identity['live_directory']),
                   json=json, hashlib=hashlib, re=re, stat=stat, math=math,
                   time=SimpleNamespace(monotonic=lambda: now),
                   os=SimpleNamespace(geteuid=lambda: host['uid'], environ=host['environment']))
        env.update(extra)
        exec(compile(ast.Module(body=selected, type_ignores=[]), str(SOURCE / name), 'exec'), env)
        return env
    runner = load('run-live-unit.py', ('require', 'private', 'digest', 'unit_name', 'admit'), {})
    admin = load('admin-control.py', ('require', 'clock_valid', 'runtime_admission_valid'), {})
    observed = runner['admit'](DataPath(identity['runtime_admission_path']))
    require(observed == (admission, clock, DataPath(identity['live_directory'])), 'actual_owner_contract')
    require(admin['clock_valid'](clock) == clock
            and admin['runtime_admission_valid'](admission, receipt, sha(files[identity['trial_clock_path']])),
            'actual_root_contract')
    return {'owner_admit': 'PASS_DATA_PREFLIGHT', 'root_clock_valid': 'PASS_DATA_PREFLIGHT',
            'root_runtime_admission_valid': 'PASS_DATA_PREFLIGHT', 'native_executed': False}


def verified_ref(ref, root_only=False):
    require(type(ref) is dict and set(ref) == {'path', 'sha256'} and hash_string(ref['sha256']), 'external_ref_shape')
    path = clean_path(ref['path'])
    if root_only:
        require(path.is_relative_to(OWNER_ROOT), 'separate_root_authority_path')
    data, s = read_file(path)
    require(sha(data) == ref['sha256'], 'external_ref_changed')
    return data, s


def authority_schema(auth):
    require(auth.get('state') == 'EXACT_NATIVE_INPUT_PUBLICATION_AUTHORIZED'
            and auth.get('authorized_by') == 'Astra' and auth.get('publish_inputs_only') is True
            and auth.get('execute_native') is False and auth.get('prior_owner_state') == 'CLOSED'
            and auth.get('assignment') == ASSIGNMENT and type(auth.get('generation')) is int
            and auth['generation'] > 0 and auth.get('source_package_sha256') == PACKAGE_SHA,
            'separate_root_authorization_required')
    return auth


def prepare(authorization_path, authorization_sha):
    require(os.getuid() == os.geteuid() == 1000 and hash_string(authorization_sha), 'owner_invocation_required')
    auth_bytes, _ = verified_ref({'path': str(authorization_path), 'sha256': authorization_sha}, True)
    auth = authority_schema(unique_json(auth_bytes))
    # External lead owns actual parent recovery. A byte-pinned CLOSED receipt
    # plus explicit current authorization is prerequisite, never our reap proof.
    settlement_bytes, _ = verified_ref(auth['prior_settlement'], True)
    settlement = unique_json(settlement_bytes)
    state = settlement.get('state', '')
    require(type(state) is str and (state == 'CLOSED' or state.startswith('CLOSED_'))
            and not any(x in state for x in ('UNCONFIRMED', 'UNKNOWN', 'PENDING')), 'prior_owner_not_reconciled')
    producer_package, _ = read_file(HERE / 'package-pins.json')
    require(sha(producer_package) == auth['producer_package_sha256'], 'producer_package_changed')
    producer_pins = unique_json(producer_package)
    pin_rows(producer_pins)
    require(set(os.listdir(HERE)) == set(producer_pins) | {'package-pins.json'}, 'producer_membership_drift')
    for name, row in producer_pins.items():
        data, _ = read_file(HERE / name)
        require(len(data) == row['size'] and sha(data) == row['sha256'], 'producer_source_drift')
    accepted_bytes, _ = verified_ref(auth['producer_review'], True)
    accepted = unique_json(accepted_bytes)
    require(accepted.get('state') == 'INDEPENDENT_NATIVE_ADMISSION_PRODUCER_SOURCE_ACCEPTED'
            and accepted.get('producer_package_sha256') == sha(producer_package)
            and accepted.get('runtime_grant') is False, 'producer_independent_review_required')
    contract_bytes, _ = read_file(HERE / 'consumer-pins.json')
    contract = unique_json(contract_bytes)
    require(contract['source'] == str(SOURCE) and contract['package_sha256'] == PACKAGE_SHA, 'fixed_source_contract')
    package, _ = read_file(SOURCE / 'package-pins.json')
    require(sha(package) == PACKAGE_SHA, 'source_package_changed')
    pins = unique_json(package)
    pin_rows(pins)
    require(pins == contract['source_leaves'] and set(os.listdir(SOURCE)) == set(pins) | {'package-pins.json'},
            'source_inventory_drift')
    files, stats, resolution, source_bytes = {}, {}, {}, {}
    def add(path, uid=1000, mode=0o400, maximum=40 << 20, single_link=True):
        data, s = read_file(path, uid, mode, maximum, single_link)
        files[str(path)], stats[str(path)], resolution[str(path)] = data, s, str(path)
        return data, s
    for name, row in pins.items():
        data, _ = add(SOURCE / name)
        require(len(data) == row['size'] and sha(data) == row['sha256'], 'source_bytes_drift')
        source_bytes[name] = data
    add(SOURCE / 'package-pins.json')
    identity = unique_json(source_bytes['source-identity.json'])
    schema = unique_json(source_bytes['RUNTIME-ADMISSION-SCHEMA.json'])
    require(identity['root_source'] == str(SOURCE) and identity['nonce'] == '010d839888625d6d'
            and schema['clock_path'] == identity['trial_clock_path']
            and schema['live_directory'] == identity['live_directory'], 'source_schema_identity_drift')
    destination = clean_path(identity['runtime_admission_path']).parent
    require(clean_path(identity['trial_clock_path']).parent == destination, 'clock_admission_parent_mismatch')
    absent(destination)
    live = clean_path(identity['live_directory'])
    absent(live)
    # Exact stage membership excludes every saved binding/start/observation intent.
    stage = clean_path(identity['stage'])
    bundle = unique_json(source_bytes['install-bundle.json'])
    receipt_bytes, receipt_stat = add(stage / 'receipt.json', 0, 0o444)
    require(sha(receipt_bytes) == auth['root_receipt_sha256'], 'root_receipt_changed')
    receipt = unique_json(receipt_bytes)
    require(receipt.get('state') == 'EXACT_ROOT_DEPENDENCIES_INSTALLED_NOT_STARTED'
            and receipt.get('trial_started') is False and receipt.get('source') == str(SOURCE)
            and receipt.get('unit') == identity['unit'] and receipt.get('stage') == str(stage)
            and receipt.get('live') == str(live), 'root_receipt_contour')
    expected_assets = set(bundle['root_assets']) | {'bootstrap.py'}
    require(set(receipt['assets']) == expected_assets
            and set(os.listdir(stage)) == expected_assets | {'receipt.json', 'control.lock', 'observe.lock'},
            'consumed_or_foreign_root_stage')
    source_hashes = string_map(pins, package)
    require(receipt['payloads'] == source_hashes
            and all(hash_string(v) for v in receipt['payloads'].values()), 'root_string_payload_map')
    for name, pin in receipt['assets'].items():
        original = 'install-system.py' if name == 'bootstrap.py' else name
        expected_mode = 0o444 if name == 'bootstrap.py' else bundle['root_assets'][name]['mode']
        data, s = add(stage / name, 0, expected_mode)
        require(pin['mode'] == expected_mode and sha(data) == pin['sha256'] == source_hashes[original]
                and [s.st_dev, s.st_ino] == pin['identity'], 'installed_asset_drift')
    data, s = add(clean_path(bundle['fragment']), 0, 0o444)
    require(sha(data) == bundle['fragment_sha256'] == receipt['fragment_sha256']
            and [s.st_dev, s.st_ino] == receipt['fragment_identity'], 'installed_fragment_drift')
    # UID1000 has no read authority for root0440 sudoers. The unchanged root
    # preflight verifies that asset before effects; never add a privileged route.
    require(receipt['policy_sha256'] == bundle['policy_sha256'], 'root_policy_receipt_pin')
    for name in ('control.lock', 'observe.lock'):
        s = (stage / name).lstat()
        require(stat.S_ISREG(s.st_mode) and s.st_uid == s.st_gid == 0
                and s.st_nlink == 1 and s.st_size == 0 and stat.S_IMODE(s.st_mode) == 0o600,
                'root_lock_metadata_drift')
        stats[str(stage / name)] = s
    journal, s = add(clean_path(identity['installation_journal']), 0, 0o444)
    require(receipt['journal_path'] == identity['installation_journal']
            and [s.st_dev, s.st_ino] == receipt['journal_identity'], 'installation_journal_drift')
    review_path = clean_path(receipt['independent_review_path'])
    review_bytes, _ = add(review_path)
    require(sha(review_bytes) == receipt['independent_review_sha256'], 'source_review_changed')
    review = unique_json(review_bytes)
    require(review.get('state') == 'INDEPENDENT_OFFLINE_BUILD_SYSTEM_SERVICE_SOURCE_ACCEPTED'
            and review.get('source_task') == SOURCE_TASK and review.get('runtime_grant') is False
            and review.get('package_pins_sha256') == PACKAGE_SHA, 'source_review_contour')
    original_path = clean_path(auth['original_clock']['path'])
    require(original_path.is_relative_to(OWNER_ROOT), 'separate_original_clock_path')
    clock_bytes, _ = verified_ref(auth['original_clock'], True)
    clock = unique_json(clock_bytes)
    boot_bytes = Path('/proc/sys/kernel/random/boot_id').read_bytes()
    now = time.monotonic()
    clock_contract(clock, identity, now, boot_bytes.decode().strip())
    require(clock['generation'] == auth['generation'], 'original_generation_changed')
    dependencies = unique_json(source_bytes['source-pins.json'])
    dependency_meta = unique_json((HERE / 'consumer-pins.json').read_bytes())['dependency_metadata']
    require(set(dependency_meta) == set(dependencies['pins']), 'dependency_metadata_inventory')
    for path, pin in dependencies['pins'].items():
        expected = dependency_meta[path]
        data, s = add(clean_path(path), 0, expected['mode'], single_link=False)
        require(len(data) == pin['size'] and sha(data) == pin['sha256'], 'consumer_dependency_drift')
        require([s.st_uid, s.st_gid, s.st_nlink] == [expected['uid'], expected['gid'], expected['nlink']],
                'consumer_dependency_metadata_drift')
    alias = dependencies['python_alias']
    require(alias['path'] == '/usr/bin/python3' and str(Path(alias['path']).resolve(strict=True)) == alias['resolved']
            and alias['resolved'] in dependencies['pins'], 'consumer_python_alias_drift')
    resolution[alias['path']] = alias['resolved']
    ps = live.parent.lstat()
    require(live.parent.resolve(strict=True) == live.parent and stat.S_ISDIR(ps.st_mode)
            and ps.st_uid == 1000 and not ps.st_mode & 0o022, 'live_parent_metadata')
    stats[str(live.parent)], resolution[str(live.parent)] = ps, str(live.parent)
    admission = admission_data(identity, source_hashes, review_path, sha(review_bytes), clock_bytes)
    for path, data in ((identity['trial_clock_path'], clock_bytes), (identity['runtime_admission_path'], encode(admission))):
        files[path], stats[path], resolution[path] = data, SimpleNamespace(st_uid=1000, st_mode=stat.S_IFREG | 0o400), path
    files['/proc/sys/kernel/random/boot_id'] = boot_bytes
    files['/proc/1/cgroup'] = Path('/proc/1/cgroup').read_bytes()
    host = {'uid': os.geteuid(), 'environment': {'CI': os.environ['CI']} if 'CI' in os.environ else {},
            'exists': {'/.dockerenv': os.path.lexists('/.dockerenv'), '/run/.containerenv': os.path.lexists('/run/.containerenv'), str(live): False}}
    now = time.monotonic()
    clock_contract(clock, identity, now, boot_bytes.decode().strip())
    checks = actual_consumers(source_bytes, contract, identity, files, stats, resolution, admission,
                              clock, receipt, now, boot_bytes.decode().strip(), host)
    # Recheck every actual observation, including source, dependency and receipt.
    for path, s in stats.items():
        if path in (identity['trial_clock_path'], identity['runtime_admission_path']):
            continue
        require(metadata(Path(path).lstat()) == metadata(s), 'snapshot_metadata_drift')
    authorities = {str(authorization_path): authorization_sha,
                   auth['prior_settlement']['path']: auth['prior_settlement']['sha256'],
                   auth['producer_review']['path']: auth['producer_review']['sha256'],
                   auth['original_clock']['path']: auth['original_clock']['sha256']}
    for path, expected in authorities.items():
        data, s = read_file(clean_path(path))
        require(sha(data) == expected, 'authority_changed_before_publish')
        stats[path] = s
    return dict(identity=identity, admission=admission, clock_bytes=clock_bytes,
                clock=clock, boot=boot_bytes.decode().strip(), checks=checks,
                authorization_sha256=sha(auth_bytes), root_receipt_sha256=sha(receipt_bytes),
                observations={path: metadata(s) for path, s in stats.items()
                              if path not in (identity['trial_clock_path'], identity['runtime_admission_path'])},
                stage_members=sorted(os.listdir(stage)))


def write_new(directory_fd, name, content):
    fd = os.open(name, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW | os.O_CLOEXEC, 0o600, dir_fd=directory_fd)
    try:
        remaining = memoryview(content)
        while remaining:
            count = os.write(fd, remaining)
            require(count > 0, 'publication_short_write')
            remaining = remaining[count:]
        os.fchmod(fd, 0o400)
        os.fsync(fd)
    finally:
        os.close(fd)


def publish(prepared):
    """Explicit future file effect only; no rollback/retry after partial publish."""
    identity = prepared['identity']
    destination = clean_path(identity['runtime_admission_path']).parent
    absent(destination)
    absent(clean_path(identity['live_directory']))
    for path, previous in prepared['observations'].items():
        require(metadata(clean_path(path).lstat()) == previous, 'publish_snapshot_drift')
    require(sorted(os.listdir(identity['stage'])) == prepared['stage_members'], 'publish_consumed_root_stage')
    clock_contract(prepared['clock'], identity, time.monotonic(),
                   Path('/proc/sys/kernel/random/boot_id').read_text().strip())
    parent, leaf, _ = parent_fd(destination, 1000)
    directory = None
    try:
        os.mkdir(leaf, 0o700, dir_fd=parent)  # atomic one-shot reservation
        directory = os.open(leaf, READ_FLAGS | os.O_DIRECTORY, dir_fd=parent)
        birth = metadata(os.fstat(directory))
        require(birth[3] == 1000 and stat.S_IMODE(birth[2]) == 0o700, 'publication_directory_mode')
        write_new(directory, 'clock.json', prepared['clock_bytes'])
        os.fsync(directory)
        clock_contract(prepared['clock'], identity, time.monotonic(),
                       Path('/proc/sys/kernel/random/boot_id').read_text().strip())
        write_new(directory, 'admission.json', encode(prepared['admission']))  # grant last
        os.fsync(directory)
        os.fsync(parent)
        final = os.stat(leaf, dir_fd=parent, follow_symlinks=False)
        require((final.st_dev, final.st_ino, final.st_uid, stat.S_IMODE(final.st_mode))
                == (birth[0], birth[1], 1000, 0o700), 'published_parent_identity_changed')
        return dict(state='EXACT_NATIVE_INPUTS_PUBLISHED_NOT_EXECUTED', clock_sha256=sha(prepared['clock_bytes']),
                    admission_sha256=sha(encode(prepared['admission'])), consumer_checks=prepared['checks'],
                    native_executed=False, original_clock_bytes_preserved=True,
                    original_budget_seconds=1800, authorization_sha256=prepared['authorization_sha256'])
    finally:
        if directory is not None:
            os.close(directory)
        os.close(parent)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--authorization', type=Path, required=True)
    parser.add_argument('--authorization-sha256', required=True)
    parser.add_argument('--publish', action='store_true', help='explicit separately reviewed file effect')
    args = parser.parse_args()
    prepared = prepare(args.authorization, args.authorization_sha256)
    result = publish(prepared) if args.publish else dict(state='EXACT_NATIVE_INPUT_DATA_PREFLIGHT_ONLY',
        clock_sha256=sha(prepared['clock_bytes']), admission_sha256=sha(encode(prepared['admission'])),
        consumer_checks=prepared['checks'], runtime_grant_published=False, native_executed=False)
    print(json.dumps(result, sort_keys=True))


if __name__ == '__main__':
    main()
