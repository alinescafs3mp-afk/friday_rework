@pytest.mark.parametrize('mutation', ['nested-source-map', 'source-missing', 'source-wrong',
    'receipt-nested', 'receipt-missing', 'receipt-review', 'clock-hash', 'clock-path', 'live-path',
    'existing-live', 'writable-clock', 'foreign-uid', 'CI', 'container', 'unobserved-read', 'AST-drift'])
def test_actual_consumer_negatives_refuse_before_publish(exact_consumer_view, mutation):
    view = copy.deepcopy(exact_consumer_view)
    admission, identity = view['admission'], view['identity']
    if mutation == 'nested-source-map':
        admission['source_sha256'] = {k: {'sha256': v, 'size': 0} for k, v in admission['source_sha256'].items()}
    if mutation == 'source-missing': admission['source_sha256'].pop('build.py')
    if mutation == 'source-wrong': admission['source_sha256']['build.py'] = '0'*64
    if mutation == 'receipt-nested': view['receipt']['payloads'] = {'build.py': {'sha256': 'a'*64}}
    if mutation == 'receipt-missing': view['receipt']['payloads'] = {}
    if mutation == 'receipt-review': view['receipt']['independent_review_sha256'] = '0'*64
    if mutation == 'clock-hash': admission['clock_sha256'] = '0'*64
    if mutation == 'clock-path': admission['clock_path'] = '/foreign/clock.json'
    if mutation == 'live-path': admission['live_directory'] += '-foreign'
    if mutation == 'existing-live': view['host']['exists'][identity['live_directory']] = True
    if mutation == 'writable-clock': view['stats'][identity['trial_clock_path']].st_mode = stat.S_IFREG | 0o600
    if mutation == 'foreign-uid': view['host']['uid'] = 0
    if mutation == 'CI': view['host']['environment']['CI'] = '1'
    if mutation == 'container': view['host']['exists']['/.dockerenv'] = True
    if mutation == 'unobserved-read': view['files'].pop(str(p.SOURCE / 'build.py'))
    if mutation == 'AST-drift': view['contract']['functions']['run-live-unit.py:admit'] = '0'*64
    view['files'][identity['runtime_admission_path']] = p.encode(admission)
    with pytest.raises((RuntimeError, KeyError)):
        p.actual_consumers(**view)

    original = b'{ "original": "bytes and spacing" }\n'
    clock = sample_clock()
    identity = dict(runtime_admission_path='/fixed/input/admission.json', live_directory='/fixed/live',
                    stage='/fixed/root', nonce=clock['nonce'], unit=clock['unit'])
    prepared = dict(identity=identity, clock=clock, clock_bytes=original, admission={'pinned': True},
                    observations={}, stage_members=[], checks={'model': True}, authorization_sha256='a'*64)
    monkeypatch.setattr(p, 'absent', lambda *args: None)

    monkeypatch.setattr(p.os, 'listdir', lambda *args: [])
    monkeypatch.setattr(p.os, 'mkdir', lambda *args, **kwargs: writes.append(('mkdir', args)))
    monkeypatch.setattr(p.os, 'open', lambda *args, **kwargs: 92)
    sample = SimpleNamespace(st_dev=2, st_ino=9, st_uid=1000, st_gid=1000, st_mode=stat.S_IFDIR | 0o700,
                             st_nlink=2, st_size=0, st_mtime_ns=1, st_ctime_ns=1)

def test_partial_write_preserves_one_shot_and_closes_fd(monkeypatch):
    calls, closed = [], []
    monkeypatch.setattr(p.os, 'open', lambda *args, **kwargs: calls.append((args, kwargs)) or 94)
    monkeypatch.setattr(p.os, 'write', lambda *args: (_ for _ in ()).throw(OSError('modeled failure')))

def file_stat(**changes):
    values = dict(st_dev=2, st_ino=10, st_mode=stat.S_IFREG | 0o400, st_uid=1000, st_gid=1000,
                  st_nlink=1, st_size=5, st_mtime_ns=1, st_ctime_ns=1)
    values.update(changes)
    return SimpleNamespace(**values)


@pytest.mark.parametrize('change', [{'st_uid': 0}, {'st_nlink': 2},
    {'st_mode': stat.S_IFREG | 0o600}, {'st_mode': stat.S_IFIFO | 0o400},
    {'st_mode': stat.S_IFLNK | 0o400}, {'st_size': 100000000}])
def test_actual_reader_refuses_leaf_metadata_and_releases_fds(change, monkeypatch):
    closed, flags = [], []
    monkeypatch.setattr(p, 'parent_fd', lambda *args: (97, 'input', [(2, 9)]))
    monkeypatch.setattr(p.os, 'open', lambda name, value, **kwargs: flags.append(value) or 98)
    monkeypatch.setattr(p.os, 'fstat', lambda *args: file_stat(**change))
    monkeypatch.setattr(p.os, 'close', closed.append)
    with pytest.raises(RuntimeError, match='input_metadata_shape'):
        p.read_file('/modeled/input')
    assert closed == [98, 97]
    assert flags[0] & p.os.O_NOFOLLOW and flags[0] & p.os.O_NONBLOCK

@pytest.fixture
def connected_prepare(exact_consumer_view, monkeypatch):
    """Whole prepare pipeline; only host reads/syscalls are closed-data modeled.

    Real immutable consumer/package/dependency bytes are loaded before patching.
    Actual verified_ref, authority, receipt, clock, AST and final checks run.
    No root path is read, no native publication or lock is performed.
    """
    from pathlib import PurePosixPath
    view = copy.deepcopy(exact_consumer_view)
    files = dict(view['files'])
    stats, lists, resolutions, absent_calls, publishes = {}, {}, dict(view['resolution']), [], []
    real_here = Path(p.__file__).parent
    producer_files = {str(q): q.read_bytes() for q in real_here.iterdir()}
    files.update(producer_files)
    for index, (path, data) in enumerate(files.items(), 100):
        stats[path] = file_stat(st_ino=index, st_size=len(data))
    for path, row in view['contract']['dependency_metadata'].items():
        stats[path] = file_stat(st_ino=stats[path].st_ino, st_size=len(files[path]),
            st_uid=row['uid'], st_gid=row['gid'], st_nlink=row['nlink'], st_mode=stat.S_IFREG | row['mode'])
    identity = view['identity']; stage = identity['stage']
    bundle = p.unique_json(view['source_bytes']['install-bundle.json'])
    lists[str(real_here)] = [Path(q).name for q in producer_files]
    lists[str(p.SOURCE)] = list(view['source_bytes'])
    stage_names = set(bundle['root_assets']) | {'bootstrap.py', 'receipt.json', 'control.lock', 'observe.lock'}
    lists[stage] = sorted(stage_names)
    def put(path, value, uid=1000, mode=0o400):
        data = value if isinstance(value, bytes) else p.encode(value)
        files[path] = data
        stats[path] = file_stat(st_uid=uid, st_gid=uid, st_ino=1000+len(stats),
                                st_size=len(data), st_mode=stat.S_IFREG | mode)
        return dict(path=path, sha256=p.sha(data))
    receipt = dict(view['receipt'], state='EXACT_ROOT_DEPENDENCIES_INSTALLED_NOT_STARTED',
        trial_started=False, source=str(p.SOURCE), unit=identity['unit'], stage=stage,
        live=identity['live_directory'], assets={}, fragment_sha256=bundle['fragment_sha256'],
        policy_sha256=bundle['policy_sha256'], journal_path=identity['installation_journal'])
    for name in sorted(set(bundle['root_assets']) | {'bootstrap.py'}):
        original = 'install-system.py' if name == 'bootstrap.py' else name
        mode = 0o444 if name == 'bootstrap.py' else bundle['root_assets'][name]['mode']
        ref = put(stage+'/'+name, view['source_bytes'][original], 0, mode)
        s = stats[ref['path']]
        receipt['assets'][name] = dict(sha256=ref['sha256'], mode=mode, identity=[s.st_dev, s.st_ino])
    # Find the exact pinned unit fragment by content hash, never synthesize it.
    fragment = [b for b in view['source_bytes'].values() if p.sha(b) == bundle['fragment_sha256']]
    assert len(fragment) == 1
    put(bundle['fragment'], fragment[0], 0, 0o444)
    s = stats[bundle['fragment']]; receipt['fragment_identity'] = [s.st_dev, s.st_ino]
    put(identity['installation_journal'], b'{"modeled_installed_not_started":true}\n', 0, 0o444)
    s = stats[identity['installation_journal']]; receipt['journal_identity'] = [s.st_dev, s.st_ino]
    for name in ('control.lock', 'observe.lock'):
        put(stage+'/'+name, b'', 0, 0o600)
    receipt_ref = put(stage+'/receipt.json', receipt, 0, 0o444)
    live_parent = str(PurePosixPath(identity['live_directory']).parent)
    stats[live_parent] = file_stat(st_mode=stat.S_IFDIR | 0o700)
    root = str(p.OWNER_ROOT)+'/modeled-connected-prepare'
    package = producer_files[str(real_here/'package-pins.json')]
    authority = dict(sample_authority(), producer_package_sha256=p.sha(package),
        prior_settlement=put(root+'/settlement.json', {'state': 'CLOSED'}),
        producer_review=put(root+'/producer-review.json', dict(
            state='INDEPENDENT_NATIVE_ADMISSION_PRODUCER_SOURCE_ACCEPTED',
            producer_package_sha256=p.sha(package), runtime_grant=False)),
        root_receipt_sha256=receipt_ref['sha256'])
    # Deliberately noncanonical spacing/order: prepare must retain exact bytes.
    clock_bytes = (json.dumps(sample_clock(), separators=(', ', ' : '))+'\n\n').encode()
    authority['original_clock'] = put(root+'/original-clock.json', clock_bytes)
    auth_ref = put(root+'/authority.json', authority)
    control = dict(files=files, stats=stats, lists=lists, authority=authority, auth_ref=auth_ref,
        receipt=receipt, receipt_path=receipt_ref['path'], put=put, clock_bytes=clock_bytes,
        absent_calls=absent_calls, publishes=publishes, snapshot_drift=False, final_authority_drift=False)
    class ModelPath(PurePosixPath):
        def read_bytes(self):
            assert str(self) in files, 'unmodeled_path_read:'+str(self)
            return files[str(self)]
        def read_text(self, *args, **kwargs): return self.read_bytes().decode()
        def lstat(self):
            assert str(self) in stats, 'unmodeled_path_stat:'+str(self)
            s = copy.copy(stats[str(self)])
            if control['snapshot_drift'] and str(self) == str(p.SOURCE/'build.py'):
                s.st_mtime_ns += 1
            return s
        def resolve(self, strict=False):
            assert str(self) in resolutions, 'unmodeled_resolution:'+str(self)
            return ModelPath(resolutions[str(self)])
    reads = {}
    def read_file(path, uid=1000, mode=0o400, maximum=40 << 20, single_link=True):
        path = str(path); assert path in files and path in stats, 'unmodeled_read_file:'+path
        reads[path] = reads.get(path, 0)+1
        s = copy.copy(stats[path]); data = files[path]
        assert s.st_uid == uid and stat.S_IMODE(s.st_mode) == mode
        assert len(data) <= maximum and (not single_link or s.st_nlink == 1)
        if control['final_authority_drift'] and path == auth_ref['path'] and reads[path] == 2:
            data += b' '
        return data, s
    def absent(path):
        absent_calls.append(str(path))
        assert str(path) in (str(PurePosixPath(identity['runtime_admission_path']).parent), identity['live_directory'])
        p.require(str(path) not in stats, 'existing_or_consumed_path')
    def publish(prepared):
        publishes.append(prepared)
        raise AssertionError('publication_unreachable_in_prepare_controls')
    monkeypatch.setattr(p, 'Path', ModelPath)
    for name in ('HERE', 'SOURCE', 'OWNER_ROOT'):
        monkeypatch.setattr(p, name, ModelPath(str(real_here) if name == 'HERE' else str(getattr(p, name))))

@pytest.mark.parametrize('mutation,reason', [
    ('settlement', 'prior_owner_not_reconciled'), ('settlement-ref', 'external_ref_changed'),
    ('review-binding', 'producer_independent_review_required'),
    ('review-ref', 'external_ref_changed'), ('source-review-binding', 'source_review_contour'),
    ('root-payload', 'root_string_payload_map'), ('root-asset', 'installed_asset_drift'),
    ('consumed-stage', 'consumed_or_foreign_root_stage'), ('foreign-stage', 'consumed_or_foreign_root_stage'),
    ('original-clock', 'original_clock_contour'), ('clock-ref', 'external_ref_changed'),
    ('generation', 'original_generation_changed'), ('snapshot', 'snapshot_metadata_drift'),
    ('final-authority', 'authority_changed_before_publish')])
def test_connected_prepare_rejects_before_publication(connected_prepare, mutation, reason):
    m = connected_prepare; auth = m['authority']; put = m['put']; receipt = m['receipt']
    if mutation == 'settlement': auth['prior_settlement'] = put(auth['prior_settlement']['path'], {'state':'STOP_UNCONFIRMED'})
    if mutation == 'settlement-ref': auth['prior_settlement']['sha256'] = '0'*64
    if mutation == 'review-binding':
        auth['producer_review'] = put(auth['producer_review']['path'], dict(
            state='INDEPENDENT_NATIVE_ADMISSION_PRODUCER_SOURCE_ACCEPTED',
            producer_package_sha256='0'*64, runtime_grant=False))
    if mutation == 'review-ref': auth['producer_review']['sha256'] = '0'*64
    if mutation == 'source-review-binding':
        ref = put(receipt['independent_review_path'], {'state':'SCHEMA_NOT_REVIEW'})
        receipt['independent_review_sha256'] = ref['sha256']
    if mutation == 'root-payload': receipt['payloads']['build.py'] = {'sha256': '0'*64}
    if mutation == 'root-asset': receipt['assets']['build.py']['identity'][1] += 1
    if mutation in ('consumed-stage', 'foreign-stage'):
        m['lists'][receipt['stage']].append('binding.json' if mutation == 'consumed-stage' else 'foreign')
    if mutation == 'original-clock':
        clock = sample_clock(); clock['budget_seconds'] = 120
        auth['original_clock'] = put(auth['original_clock']['path'], clock)
    if mutation == 'clock-ref': auth['original_clock']['sha256'] = '0'*64
    if mutation == 'generation': auth['generation'] = 2
    if mutation == 'snapshot': m['snapshot_drift'] = True
    if mutation == 'final-authority': m['final_authority_drift'] = True
    auth['root_receipt_sha256'] = put(m['receipt_path'], receipt, 0, 0o444)['sha256']
    auth_ref = put(m['auth_ref']['path'], auth)
    with pytest.raises(RuntimeError, match=reason):
        p.prepare(auth_ref['path'], auth_ref['sha256'])
    assert m['publishes'] == []


@pytest.mark.parametrize('mutation', ['unchanged', 'external-changed', 'external-missing', 'runner-changed'])
def test_original_pin_bytes_cover_external_and_runner_after_run(mutation):
    from qualification_contract import compare_inputs
    files = {'/source/p.py': b'source', '/external/consumer.py': b'consumer', '/output/run.py': b'runner'}
    original = p.encode({path: dict(size=len(data), sha256=p.sha(data)) for path,data in files.items()})
    if mutation == 'external-changed': files['/external/consumer.py'] += b'drift'
    if mutation == 'external-missing': del files['/external/consumer.py']
    if mutation == 'runner-changed': files['/output/run.py'] += b'drift'
    comparison = compare_inputs(original, files.__getitem__)
    assert comparison['leaf_count'] == 3 and comparison['original_pins_sha256'] == p.sha(original)
    assert comparison['unchanged'] is (mutation == 'unchanged')
    assert all('expected' in r and 'actual' in r for r in comparison['rows'].values())


@pytest.mark.parametrize('mutation', ['complete', 'missing-call', 'skip', 'duplicate', 'collection-fail', 'nonzero'])
def test_terminal_qualification_never_promotes_incomplete_reports(mutation):
    from qualification_contract import validate_reports
    node = 'test_native_input_producer.py::test_example'
    result = dict(nodes=[node], collections=[dict(nodeid='', outcome='passed'), dict(nodeid='test_native_input_producer.py', outcome='passed')],
        reports=[dict(nodeid=node, when=w, outcome='passed') for w in ('setup','call','teardown')],
        exit_code=0, candidate_unchanged=True, all_inputs_unchanged=True)
    if mutation == 'missing-call': result['reports'].pop(1)
    if mutation == 'skip': result['reports'][1]['outcome'] = 'skipped'
    if mutation == 'duplicate': result['reports'].append(dict(result['reports'][1]))
    if mutation == 'collection-fail': result['collections'][0]['outcome'] = 'failed'
    if mutation == 'nonzero': result['exit_code'] = 1
    assert validate_reports(result, 1)['accepted'] is (mutation == 'complete')
