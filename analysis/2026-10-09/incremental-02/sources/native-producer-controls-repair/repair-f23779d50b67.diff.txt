+def compare_inputs(original_bytes, read_bytes):
+    pins = json.loads(original_bytes)
+    if not isinstance(pins, dict) or not pins:
+        raise ValueError('input_pin_inventory')
+    rows = {}
+    for path, expected in sorted(pins.items()):
+        try:
+            data = read_bytes(path)
+            actual = dict(sha256=hashlib.sha256(data).hexdigest(), size=len(data))
+            rows[path] = dict(expected=expected, actual=actual, unchanged=actual == expected)
+        except Exception as exc:
+            rows[path] = dict(expected=expected, actual=None, unchanged=False,
+                              error=type(exc).__name__ + ':' + str(exc))
+    return dict(original_pins_sha256=hashlib.sha256(original_bytes).hexdigest(),
+                leaf_count=len(rows), rows=rows,
+                unchanged=all(row['unchanged'] for row in rows.values()))

+def validate_reports(result, expected_count):

+    errors = []

+    if (len(nodes) != expected_count or len(set(nodes)) != expected_count

+        errors.append('exact_unique_collection')
+    collections = result.get('collections', [])
+    if not collections or any(r.get('outcome') != 'passed' for r in collections):
+        errors.append('collection_incomplete_failed_or_skipped')
+    collection_nodes = [r.get('nodeid') for r in collections]
+    if (len(collection_nodes) != 2 or set(collection_nodes) != {'', 'test_native_input_producer.py'}):
+        errors.append('collection_missing_duplicate_or_foreign')
+    reports = result.get('reports', [])
+    expected = {(node, when) for node in nodes for when in ('setup', 'call', 'teardown')}
+    actual = [(r.get('nodeid'), r.get('when')) for r in reports]
+    if len(actual) != len(expected) or len(set(actual)) != len(actual) or set(actual) != expected:
+        errors.append('terminal_reports_missing_duplicate_or_foreign')

+def compare_inputs(original_bytes, read_bytes):

+            rows[path] = dict(expected=expected, actual=actual, unchanged=actual == expected)
+        except Exception as exc:

+                              error=type(exc).__name__ + ':' + str(exc))
+    return dict(original_pins_sha256=hashlib.sha256(original_bytes).hexdigest(),
+                leaf_count=len(rows), rows=rows,
+                unchanged=all(row['unchanged'] for row in rows.values()))
