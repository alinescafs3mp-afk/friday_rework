def bootstrap_need(ok, why):
    if not ok:
        raise RuntimeError('source bootstrap refused: ' + why)

def digest_pin(pin):
    bootstrap_need(type(pin) is str and len(pin) == 64 and
                   all(c in '0123456789abcdef' for c in pin), 'SHA256 pin')
    return pin

def exact(path, pin, limit=1048576):
    digest_pin(pin)
    fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK | os.O_CLOEXEC)
    try:
        before = os.fstat(fd)
        bootstrap_need(stat.S_ISREG(before.st_mode) and
                       stat.S_IMODE(before.st_mode) == 0o600 and
                       before.st_uid == before.st_gid == 1000 and
                       before.st_nlink == 1 and 0 < before.st_size <= limit,
                       'regular private owner file metadata/size')
        # One bounded read, no stream loop or reopen. A short read refuses.
        raw = os.read(fd, limit + 1)
        after = os.fstat(fd)
        fields = ('st_dev', 'st_ino', 'st_mode', 'st_uid', 'st_gid',
                  'st_nlink', 'st_size', 'st_mtime_ns', 'st_ctime_ns')
        bootstrap_need(all(getattr(before, k) == getattr(after, k) for k in fields),
                       'same-FD metadata changed during read')
        bootstrap_need(type(raw) is bytes and len(raw) == before.st_size and
                       hashlib.sha256(raw).hexdigest() == pin, 'exact source bytes')
        return raw
    finally:
        os.close(fd)

def unique_pairs(items):
    value = {}
    for key, item in items:
        bootstrap_need(key not in value, 'duplicate JSON field')
        value[key] = item
    return value

def source_manifest(raw, pin):
    bootstrap_need(type(raw) is bytes and 0 < len(raw) <= 16384 and
                   hashlib.sha256(raw).hexdigest() == digest_pin(pin),
                   'verified manifest buffer')
    value = json.loads(raw, object_pairs_hook=unique_pairs)
    bootstrap_need(type(value) is dict and set(value) ==
                   {'status', 'assignment', 'generation', 'execution_authorized', 'files'},
                   'manifest schema')
    bootstrap_need(value['status'] ==
                   'SOURCE_ONLY_REQUIRES_INDEPENDENT_REVIEW_AND_OWNER_PRIVILEGED_EXECUTION'
                   and value['assignment'] == 'FRW011-A0-DEPENDENCY-DROPIN-G4'
                   and type(value['generation']) is int and value['generation'] == 1
                   and value['execution_authorized'] is False, 'manifest authority')
    files = value['files']
    bootstrap_need(type(files) is dict and set(files) ==
                   {'common.py', 'payload.py', 'observer.py', 'driver.py', 'recipe.json',
                    'witness.py', 'owner_launcher.py', 'parent_integration.py', 'dropins.py'},
                   'complete source allowlist')
    for pin in files.values():
        digest_pin(pin)
    return types.MappingProxyType({**value, 'files': types.MappingProxyType(files)})

class ParentCessation:
    """Actual owner transport/status and fresh native facts, never Driver report.

    This is an in-process parent callback, not a second principal. Independence
    is from Driver assertions: it uses parent's Popen inventory and its own
    fresh system-manager/unit/job/proc/cgroup reads and immutable event binding.
    """
    def __init__(self, original, names, transport, witness, session, evidence, audit_dir):
        import copy
        self.original = copy.deepcopy(original)
        self.names = copy.deepcopy(names)
        self.transport, self.witness, self.session = transport, witness, session
        self.evidence, self.audit_dir = evidence, audit_dir
        self.actors = {}
        self.cgroups = {r:set() for r in names}

    def binding(self):
        return {k:self.original[k] for k in ('assignment','generation','attempt_token','manifest_sha256',
                                             'boot_id','accepted_monotonic_ns','accepted_boottime_ns')}

    def remember(self, role, actor):
        from common import require, own_cgroup
        require(role in self.names and type(actor['pid']) is int and actor['pid']>1 and
                type(actor['starttime_ticks']) is int and actor['starttime_ticks']>0 and
                own_cgroup(role,actor), 'original actor identity/cgroup')
        key=(actor['pid'],actor['starttime_ticks'])
        if key in self.actors:
            require(self.actors[key]['role']==role,'actor role changed')
        self.actors[key]={'role':role,'identity':dict(actor)}
        for line in actor['cgroup'].splitlines():
            if line.startswith('0::'): self.cgroups[role].add(line[3:])

    def observe_capture(self, result):
        # Direct callback from common's owned Popen capture, before returning to
        # Driver. No Driver state or output fields feed this inventory.
        from common import show_argv, parse_show, identity, require
        if result['argv'] != show_argv(): return
        props=parse_show(result['stdout'])
        for role,item in props.items():
            if item.get('ControlGroup'): self.cgroups[role].add(item['ControlGroup'])
            for field in ('MainPID','ControlPID'):
                pid=int(item.get(field,'0'))
                if pid:
                    try: actor=identity(pid)
                    except FileNotFoundError: continue
                    self.remember(role,actor)

    def scan_actors(self, now):
        from common import require, read_json, identity
        observations=[]
        for path in sorted(self.evidence.glob('*.json')):
            if path.name in ('receipt.json','native-result.json') or path.name.endswith('.admission.json'): continue
            item=read_json(path)
            if 'role' not in item or 'event' not in item: continue
            require(item.get('session_binding')==self.binding() and type(item['session_binding'].get('generation')) is int,
                    'stale actor generation/nonce/boot/source')
            require(all(type(item.get(a)) is int and self.original['accepted_'+a]<=item[a]<=now[a]
                        for a in ('monotonic_ns','boottime_ns')),'actor event outside original attempt')
            self.remember(item['role'],item)
        for (pid,start),item in self.actors.items():
            try: found=identity(pid)
            except FileNotFoundError:
                observations.append({'pid':pid,'starttime_ticks':start,'observation':'PROC_ABSENT'})
                continue
            # PID reuse is not death evidence silently substituted for this actor.
            require(found['starttime_ticks']==start,'actor PID reused; independent review required')
            require(False,'original actor still present (including zombie)')
        return observations

    def lock_check(self):

                identity(os.getpid())['starttime_ticks']==parent['starttime_ticks'],'original FD9 parent changed')

        matching=[]

        # common native clients and privileged witness use close_fds without FD9.

            try: s=entry.stat()
            except FileNotFoundError: continue
