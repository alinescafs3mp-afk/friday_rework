def read(path):
    path = Path(path)
    if path in cache:
        return cache[path][0]
    assert path.is_absolute() and path.resolve(strict=True) == path and not path.is_symlink()
    for parent in path.parents:
        s = parent.lstat()
        assert stat.S_ISDIR(s.st_mode) and s.st_uid in (0, 1000) and not s.st_mode & 0o022
    fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK | os.O_CLOEXEC)
    try:
        before = os.fstat(fd)
        assert stat.S_ISREG(before.st_mode) and before.st_uid == 1000 and before.st_nlink == 1
        chunks = []
        while True:
            chunk = os.read(fd, 131072)
            if not chunk:
                break
            chunks.append(chunk)
        data = b''.join(chunks)
        after = os.fstat(fd)
    finally:
        os.close(fd)
    assert before == after == path.lstat() and len(data) == before.st_size
    pin = dict(path=str(path), sha256=hashlib.sha256(data).hexdigest(), bytes=len(data),
               uid=before.st_uid, mode=stat.S_IMODE(before.st_mode), nlink=before.st_nlink,
               device=before.st_dev, inode=before.st_ino)
    cache[path] = data, pin
    return data

def ref(path):
    read(path)
    return cache[Path(path)][1]

def check(path, expected):
    actual = ref(path)
    for key, value in expected.items():
        assert actual['bytes' if key == 'size' else key] == value, (path, key)
    return actual

def unique(pairs):
    result = {}
    for key, value in pairs:
        assert key not in result, key
        result[key] = value
    return result

def js(path):
    return json.loads(read(path), object_pairs_hook=unique)

def functions(text):
    return {n.name: ast.dump(n, include_attributes=False) for n in ast.parse(text).body
            if isinstance(n, (ast.FunctionDef, ast.ClassDef))}
