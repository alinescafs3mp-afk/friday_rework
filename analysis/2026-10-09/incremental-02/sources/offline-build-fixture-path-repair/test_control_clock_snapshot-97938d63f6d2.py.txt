def test_first_control_durable_entry_precedes_mutable_clock(memory_control):
    admin,c,a,mem,events,clock_path,admission_path=memory_control
    first=admin.Control('preflight')
    assert events.index(('create',str(admin.STAGE/'entry-preflight.json'))) < events.index(('read',clock_path))
    binding=json.loads(mem[str(admin.STAGE/'trial-binding.json')])
    assert bytes.fromhex(binding['clock_hex'])==first.clock_bytes
    assert first.clock['started_monotonic']==0 and first.clock['deadline_monotonic']==1800


def test_changed_mutable_clock_cannot_shift_start_or_dispatch(memory_control,monkeypatch):
    admin,c,a,mem,events,clock_path,admission_path=memory_control
    first=admin.Control('preflight');original=first.clock_bytes
    shifted=dict(c,started_monotonic=10,deadline_monotonic=1810)
    raw=json.dumps(shifted).encode();a['clock_sha256']=hashlib.sha256(raw).hexdigest()
    mem[clock_path]=raw;mem[admission_path]=json.dumps(a).encode()
    start=admin.Control('start');assert start.clock_bytes==original and start.clock['deadline_monotonic']==1800
    monkeypatch.setattr(start,'manager',lambda:manager())
    start.receipt['manager']=manager()
    monkeypatch.setattr(start,'run',lambda *a:pytest.fail('changed original clock dispatched start'))
    with pytest.raises(RuntimeError):start.execute()
    assert str(admin.STAGE/'start-intent.json') not in mem


@pytest.mark.parametrize('operation',['stop','cleanup'])
def test_stop_cleanup_use_pinned_original1800_without_mutable_reads(memory_control,operation):
    admin,c,a,mem,events,clock_path,admission_path=memory_control
    first=admin.Control('preflight');mem.pop(clock_path);mem.pop(admission_path);events.clear()
    control=admin.Control(operation)
    assert control.clock_bytes==first.clock_bytes and control.clock['deadline_monotonic']==1800
    assert not any(kind=='read' and name in (clock_path,admission_path) for kind,name in events)


def test_second_root_start_intent_cannot_be_recreated(memory_control,monkeypatch):
    admin,c,a,mem,events,clock_path,admission_path=memory_control
    admin.Control('preflight');start=admin.Control('start')
    monkeypatch.setattr(start,'manager',lambda:manager());start.receipt['manager']=manager()
    calls=[];monkeypatch.setattr(start,'run',lambda argv,seconds:calls.append(argv) or {'fixture':True})
    assert start.execute()['state']=='WAIT_CLIENT_ENDED'
    assert calls==[admin.fixed_argv('start')]
    with pytest.raises(FileExistsError):start.execute()
    assert len(calls)==1 and start.clock['deadline_monotonic']==1800
