import hashlib
import importlib.util
import json
from pathlib import Path
import sys

ROOT = Path(__file__).parent


def load(name):
    selection = json.loads((ROOT / 'FINAL-SOURCE-SELECTION.json').read_bytes())
    assert selection['state'] == 'EXACT_FINAL_OFFLINE_SOURCE_SELECTED'
    assert selection['runtime_grant'] is False
    assert selection['implementation_source_accept'] is True
    assert selection['source_accept'] is False
    assert selection['final_join_requires_independent_review'] is True
    assert selection['OB1_OB2_closed_by_independent_review'] is True
    pin = selection['files'][name + '.py']
    source = ROOT / (name + '.py')
    assert source.is_file() and not source.is_symlink()
    assert source.stat().st_nlink == 1
    data = source.read_bytes()
    assert len(data) == pin['bytes']
    assert hashlib.sha256(data).hexdigest() == pin['sha256']
    # Related import dependency is part of the same final-source selection.
    for leaf in ('runtime_binding.py', 'source-identity.json'):
        value = (ROOT / leaf).read_bytes()
        assert hashlib.sha256(value).hexdigest() == selection['files'][leaf]['sha256']
        assert len(value) == selection['files'][leaf]['bytes']
    sys.path.insert(0, str(ROOT))
    try:
        spec = importlib.util.spec_from_file_location('gate_' + name.replace('-', '_'), source)
        module = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(module)
        return module
    finally:
        sys.path.remove(str(ROOT))
