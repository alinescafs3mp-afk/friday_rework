def sha(data):return hashlib.sha256(data).hexdigest()
def js(p):return json.loads(p.read_bytes())
def ref(p):
    s=p.lstat();data=p.read_bytes()
    assert p==p.resolve(strict=True) and stat.S_ISREG(s.st_mode)
    assert s.st_uid==1000 and s.st_nlink==1 and stat.S_IMODE(s.st_mode)==0o400
    for parent in p.parents:
        q=parent.lstat()
        assert stat.S_ISDIR(q.st_mode) and q.st_uid in (0,1000) and not q.st_mode&0o022
    return dict(path=str(p),sha256=sha(data),bytes=len(data),mode=stat.S_IMODE(s.st_mode),uid=s.st_uid,nlink=s.st_nlink)
def parse(p):
    node=ast.parse(p.read_bytes(),filename=str(p));compile(node,str(p),'exec')
    return node
def functions(tree):return {n.name:n for n in tree.body if isinstance(n,(ast.FunctionDef,ast.ClassDef))}
def dump(node):return ast.dump(node,include_attributes=False)
def cases(node):
    result=1
    for d in node.decorator_list:
        if isinstance(d,ast.Call) and isinstance(d.func,ast.Attribute) and d.func.attr=='parametrize':
            assert isinstance(d.args[1],(ast.Tuple,ast.List))
            result*=len(d.args[1].elts)
    return result
