def ref(p):
    s=p.lstat();data=p.read_bytes()
    assert p==p.resolve(strict=True) and stat.S_ISREG(s.st_mode) and s.st_uid==1000 and s.st_nlink==1
    assert stat.S_IMODE(s.st_mode) in (0o400,0o600)
    for parent in p.parents:
        q=parent.lstat()
        assert stat.S_ISDIR(q.st_mode) and q.st_uid in (0,1000) and not q.st_mode&0o022
    return dict(path=str(p),sha256=hashlib.sha256(data).hexdigest(),bytes=len(data),mode=stat.S_IMODE(s.st_mode),uid=s.st_uid,nlink=s.st_nlink)


def put(p,value):
    assert p.is_relative_to(E)
    fd=os.open(p,os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o600)
    with os.fdopen(fd,'w') as f:
        json.dump(value,f,indent=2);f.write('\n');f.flush();os.fsync(f.fileno())
    p.chmod(0o400)
