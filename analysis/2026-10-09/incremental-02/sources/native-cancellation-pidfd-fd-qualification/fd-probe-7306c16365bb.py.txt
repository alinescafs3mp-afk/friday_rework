"""Finite real-FD predicate checks, then the unchanged reviewed source runner."""
from pathlib import Path
import ast, errno, fcntl, hashlib, json, os, resource, stat, sys, time

ROOT = Path(__file__).resolve().parent
RUN = ROOT / 'fd-probe-private'
OUTSIDE = ROOT / 'fd-probe-outside'
clock = json.loads((ROOT / 'clock.json').read_bytes())
assert time.monotonic() + 65 < clock['deadline_monotonic']
os.umask(0o077)
os.sched_setaffinity(0, {0, 1})
resource.setrlimit(resource.RLIMIT_AS, (2 << 30, 2 << 30))
resource.setrlimit(resource.RLIMIT_CPU, (45, 45))
source = (ROOT / 'run.py').read_bytes()
assert hashlib.sha256(source).hexdigest() == '53baf2aa8b1d1e281bc5a6be7b3eec4d17a7b1cbff8c3127a746384c8eb1acc4'
node = next(n for n in ast.parse(source).body if isinstance(n, ast.FunctionDef) and n.name == 'owned')
ns = dict(Path=Path, os=os, stat=stat, fcntl=fcntl, RUN=RUN,
          RAW_FSTAT=os.fstat, RAW_READLINK=os.readlink, RAW_FCNTL=fcntl.fcntl)
exec(compile(ast.Module(body=[node], type_ignores=[]), str(ROOT / 'run.py'), 'exec'), ns)
owned = ns['owned']
lock = os.open('<LOCAL_HOME>/.jericho/runtime/friday-quality-ui-exclusive.lock', os.O_RDONLY | os.O_NOFOLLOW)
s = os.fstat(lock)
assert (s.st_dev, s.st_ino, s.st_uid, s.st_gid, s.st_nlink, s.st_size, stat.S_IMODE(s.st_mode)) == (2050, 9044963, 1000, 1000, 1, 0, 0o664)
fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
RUN.mkdir(mode=0o700)
OUTSIDE.mkdir(mode=0o700)
fds = set()
results = []
def opening(p, flags, mode=0o600):
    fd = os.open(p, flags, mode); fds.add(fd); return fd
def check(label, fd, expected):
    observed = owned(fd)
    results.append(dict(case=label, fd=fd, observed=observed, expected=expected))
    assert observed is expected, label
def close(fd):
    os.close(fd); fds.remove(fd)
    try: os.fstat(fd)
    except OSError as exc: assert exc.errno == errno.EBADF
    else: raise AssertionError('descriptor still open')
try:
    p = RUN / 'private'
    fd = opening(p, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o400)
    check('private-writable-fd-despite-inode0400', fd, True)
    dup = os.dup(fd); fds.add(dup); check('same-private-inode-dup-allowed', dup, True); close(dup)
    with os.fdopen(fd, 'w', closefd=False) as out: out.write('private-owned\n'); out.flush(); os.fsync(fd)
    assert p.read_bytes() == b'private-owned\n'
    close(fd); check('closed', fd, False)
    fd = opening(p, os.O_RDONLY); check('read-only', fd, False); close(fd)
    q = OUTSIDE / 'sentinel'; q.write_bytes(b'unchanged-outside\n')
    original = q.read_bytes(); identity = (q.stat().st_dev, q.stat().st_ino)
    fd = opening(q, os.O_RDWR); check('outside-writable', fd, False); close(fd)
    fd = opening(RUN / 'deleted', os.O_RDWR | os.O_CREAT | os.O_EXCL)
    (RUN / 'deleted').unlink(); check('unlinked', fd, False); close(fd)
    p = RUN / 'hardlink'; fd = opening(p, os.O_RDWR | os.O_CREAT | os.O_EXCL)
    os.link(p, RUN / 'hardlink-alias'); check('multiple-links', fd, False); close(fd)
    p = RUN / 'moved'; fd = opening(p, os.O_RDWR | os.O_CREAT | os.O_EXCL)
    p.rename(OUTSIDE / 'moved'); p.symlink_to(q)
    check('moved-outside-and-old-name-rebound', fd, False); close(fd)
    p = RUN / 'alias'; p.symlink_to(q)
    fd = opening(p, os.O_RDWR); check('symlink-to-outside', fd, False); close(fd)
    p = RUN / 'safe-target'; fd = opening(p, os.O_RDWR | os.O_CREAT | os.O_EXCL); close(fd)
    (RUN / 'safe-alias').symlink_to(p)
    fd = opening(RUN / 'safe-alias', os.O_RDWR); check('symlink-origin-same-private-inode-allowed', fd, True); close(fd)
    check('negative', -1, False)
    assert q.read_bytes() == original and (q.stat().st_dev, q.stat().st_ino) == identity
finally:
    for fd in tuple(fds): close(fd)
    os.close(lock)
assert time.monotonic() + 60 < clock['deadline_monotonic']
with (ROOT / 'fd-probe-result.json').open('x') as out:
    json.dump(dict(state='REAL_FD_PREDICATES_PASS', cases=results, all_owned_fds_closed=True,
                   sentinel_unchanged=True, actual_audit_refusal_exercised=False,
                   pid=os.getpid(), finished_monotonic=time.monotonic()), out, indent=2)
os.execve('<PROJECT>/.runtime/hermes-telegram-test/env/bin/python',
          ['python', '-I', '-S', '-B', str(ROOT / 'run.py')], {'PATH': '/usr/bin:/bin', 'PYTEST_DISABLE_PLUGIN_AUTOLOAD': '1'})
