"""One bounded owner-only data control run; never imports candidate code."""
import ast
import copy
import datetime
import fcntl
import hashlib
import json
import os
from pathlib import Path
import resource
import stat
import sys
import time

ROOT = Path(__file__).resolve().parent
W = ROOT.parent
SOURCE = W / 'collection-owner-inspection-repair-r2/inspect_project.py'
BASIS = W / 'ordinary-skills-ancestor-metadata-collection-qualification-1/collect-project/classification-source-basis.json'
LOCK = '<LOCAL_HOME>/.jericho/runtime/friday-quality-ui-exclusive.lock'
clock = json.loads((ROOT / 'data-clock.json').read_bytes())
assert time.monotonic() + 60 < clock['deadline_monotonic']
os.umask(0o077)
os.sched_setaffinity(0, {0, 1})
resource.setrlimit(resource.RLIMIT_AS, (2 << 30, 2 << 30))
resource.setrlimit(resource.RLIMIT_CPU, (45, 45))

def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

pins = json.loads((ROOT / 'input-pins.json').read_bytes())
for path, expected in pins.items():
    assert sha(path) == expected, path
fd = os.open(LOCK, os.O_RDONLY | os.O_NOFOLLOW)
s = os.fstat(fd)
assert [s.st_dev, s.st_ino, s.st_uid, s.st_gid, s.st_nlink, s.st_size, stat.S_IMODE(s.st_mode)] == [2050, 9044963, 1000, 1000, 1, 0, 0o664]
fcntl.flock(fd, fcntl.LOCK_EX | fcntl.LOCK_NB)
state = {'status': 'RUNNING', 'started_monotonic': time.monotonic(), 'pid': os.getpid(), 'backend_pid': os.getppid(),
         'pgid': os.getpgrp(), 'pid_stat': Path('/proc/self/stat').read_text(),
         'backend_stat': Path('/proc/' + str(os.getppid()) + '/stat').read_text(),
         'resources': {'affinity': sorted(os.sched_getaffinity(0)), 'as': resource.getrlimit(resource.RLIMIT_AS), 'cpu': resource.getrlimit(resource.RLIMIT_CPU)},
         'observations': [], 'denied': [], 'candidate_imports': False, 'collection': False, 'observer': False}

def save():
    (ROOT / 'data-progress.json').write_text(json.dumps(state, indent=2) + '\n')

def owned(raw):
    if isinstance(raw, int):
        return raw in (1, 2) or Path('/proc/self/fd/' + str(raw)).resolve().is_relative_to(ROOT)
    return Path(os.fsdecode(raw)).absolute().is_relative_to(ROOT) and Path(os.fsdecode(raw)).resolve().is_relative_to(ROOT)

def deny(event):
    state['denied'].append(event)
    save()
    os.write(2, ('DATA_CONTROL_DENIED ' + event + '\n').encode())
    os._exit(125)

def audit(event, args):
    if event in ('subprocess.Popen', 'os.system', 'os.fork', 'os.forkpty', 'os.posix_spawn', 'os.exec', 'pty.spawn', 'os.kill', 'os.killpg', 'signal.pidfd_send_signal', 'os.pidfd_open') or event.startswith('socket.'):
        deny(event)
    if event == 'open' and args[2] & (os.O_WRONLY | os.O_RDWR | os.O_CREAT | os.O_TRUNC | os.O_APPEND) and not owned(args[0]):
        deny(event + ':outside_write')
    if event in ('os.mkdir', 'os.remove', 'os.rmdir', 'os.chmod', 'os.chown', 'os.utime', 'os.truncate') and not owned(args[0]):
        deny(event + ':outside_mutation')
    if event in ('os.rename', 'os.link') and not (owned(args[0]) and owned(args[1])):
        deny(event + ':outside_pair')
    if event == 'os.symlink' and not owned(args[1]):
        deny(event + ':outside_link')

sys.addaudithook(audit)
save()
rc = 2
try:
    # Execute only the reviewed owner helper, with its CLI branch disabled.
    namespace = {'__name__': 'reviewed_owner_data_functions', '__file__': str(SOURCE)}
    exec(compile(SOURCE.read_bytes(), str(SOURCE), 'exec'), namespace)
    basis = json.loads(BASIS.read_bytes())
    actual_data = {Path(r['path']).name: Path(r['path']).read_bytes() for r in basis['input_refs']}

    def fixture(name, mutate=None, closed=False):
        phase = ROOT / 'controls' / name / 'collect-project'
        phase.mkdir(mode=0o700, parents=True)
        run = phase / 'run'
        run.mkdir(mode=0o700)
        b = copy.deepcopy(basis)
        for ref in b['input_refs']:
            p = run / Path(ref['path']).name
            p.write_bytes(actual_data[p.name])
            ref['path'] = str(p)
        if mutate:
            mutate(b)
        p = phase / 'classification-source-basis.json'
        p.write_text(json.dumps(b) + '\n')
        if closed:
            (phase / 'cessation.json').write_text('{"status":"CLOSED_CONTROL"}\n')
        return p

    positive = fixture('actual-retained-positive')
    receipt = namespace['publish'](positive)
    checked = json.loads(Path(receipt['path']).read_bytes())
    assert receipt['sha256'] == sha(receipt['path'])
    assert receipt['owner_inspection'] == {'definitions': 627, 'modules': 17, 'source_files': 24, 'reference_checks': 6027, 'method': 'Actual pinned AST definitions/decorators/fixture refs and empty project skip metadata checked; no execution acceptance.'}
    assert all(r['definition_inspected'] and r['platform_skip_inspected'] for r in checked['kept'])
    assert checked['acceptance'] is False and checked['all_runtests'] == 'NOT_RUN'
    assert stat.S_IMODE(Path(receipt['path']).stat().st_mode) == 0o400
    state['observations'].append({'case': 'actual-retained-positive', 'outcome': 'PASS', 'receipt': receipt, 'retained_metadata_unchanged': [r['metadata'] for r in checked['kept']] == [r['metadata'] for r in basis['kept']]})
    save()

    def refusal(name, mutate=None, closed=False, existing=None):
        p = existing or fixture(name, mutate, closed)
        target = p.parent / 'classification-review.json'
        existed = target.exists()
        prior_hash = sha(target) if existed else None
        try:
            namespace['publish'](p)
        except Exception as error:
            unchanged = (sha(target) == prior_hash) if existed else not target.exists()
            assert unchanged, 'publication side effect after refusal'
            state['observations'].append({'case': name, 'outcome': 'REFUSED', 'error_type': type(error).__name__, 'reason': str(error), 'publication_unchanged': unchanged})
            save()
            return
        raise AssertionError('negative control unexpectedly accepted: ' + name)

    refusal('exclusive-repeat-publication', existing=positive)
    refusal('missing-definition', lambda b: b['kept'][0].pop('definition_basis'))
    refusal('tampered-source-pin', lambda b: b['kept'][0]['definition_basis'].__setitem__('sha256', '0' * 64))
    refusal('unknown-fixture-mode', lambda b: b['kept'][0]['fixture_basis'].__setitem__('env', 'unknown_fixture'))
    refusal('skip-metadata', lambda b: b['kept'][0]['metadata']['skip_reasons'].append('new-unreviewed-skip'))
    refusal('wrong-selected-module', lambda b: b['selected_modules'][0].__setitem__('path', '/unreviewed/module.py'))
    refusal('wrong-row-definition-module', lambda b: b['kept'][0]['metadata'].__setitem__('definition', '/unreviewed/module.py'))
    refusal('closed-phase', closed=True)
    refusal('decorator-drift', lambda b: b['kept'][0]['decorators_source_basis'].append('pytest.mark.skip(reason="unreviewed")'))

    # Last control: valid source refs are insufficient if bound to another row.
    def substitute_definition(b):
        for key in ('definition_basis', 'originalname_source_basis', 'decorators_source_basis', 'class_lineage_basis'):
            b['kept'][0][key] = copy.deepcopy(b['kept'][1][key])
    swapped = fixture('different-valid-definition-same-metadata', substitute_definition)
    try:
        bad_receipt = namespace['publish'](swapped)
    except Exception as error:
        assert not (swapped.parent / 'classification-review.json').exists()
        state['observations'].append({'case': 'different-valid-definition-same-metadata', 'outcome': 'REFUSED', 'error_type': type(error).__name__, 'reason': str(error)})
        state['status'] = 'ALL_DATA_CONTROLS_PASSED'
        rc = 0
    else:
        state['observations'].append({'case': 'different-valid-definition-same-metadata', 'outcome': 'FAIL_UNEXPECTED_PUBLICATION', 'receipt': bad_receipt, 'original_metadata_nodeid': basis['kept'][0]['metadata']['nodeid'], 'substituted_definition': basis['kept'][1]['definition_basis'], 'actual_metadata_unchanged': True})
        state['status'] = 'FAIL_DEFINITION_NOT_BOUND_TO_RETAINED_IDENTITY'
        rc = 1
    # No further data tests or retries after an unexpected acceptance.
finally:
    state['finished_monotonic'] = time.monotonic()
    state['exit_code'] = rc
    state['elapsed_seconds'] = state['finished_monotonic'] - state['started_monotonic']
    save()
    os.close(fd)
print(json.dumps({'status': state['status'], 'observations': len(state['observations']), 'exit_code': rc, 'elapsed_seconds': state['elapsed_seconds']}))
raise SystemExit(rc)
