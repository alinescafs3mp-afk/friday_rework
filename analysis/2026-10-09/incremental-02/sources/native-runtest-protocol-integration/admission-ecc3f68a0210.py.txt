"""Read-only exact canary, collection or per-file runtest admission. SOURCE ONLY.
Runtest refuses until separate reviewed runtime admission and source readiness.
The admitting owner pins runtime/startup BEFORE the first isolated interpreter.
Guard repeats this inside installed timeout before any fixture/FD/receipt effects.
"""
import hashlib, json, os, pathlib, re, time, runpy


def verify(source, run, name, entry, selection, collect_only=False, runtest_file=None):
    source = pathlib.Path(source).resolve(strict=True)
    run = pathlib.Path(run).resolve(strict=True)
    before = (run / 'clock.json').read_bytes()
    clock = json.loads(before)
    if type(collect_only) is not bool:
        raise SystemExit('EXACT_COLLECTION_FLAG_REQUIRED')
    runtime_unit = None
    if runtest_file is not None:
        if collect_only or not re.fullmatch(r'f[0-9]{3}', runtest_file):
            raise SystemExit('EXACT_RUNTEST_ROUTE_REQUIRED')
        runtime_bytes = (source / 'runtest-plan.json').read_bytes()
        runtime_plan = json.loads(runtime_bytes)
        runtime_sha = hashlib.sha256(runtime_bytes).hexdigest()
        scope = 'FINITE_EXACT_DONOR_FILE_RUNTEST_ONLY'
        if (clock.get('scope') != scope or clock.get('runtest_file') != runtest_file or
            clock.get('collect_only') is not False or 'canary' in clock or 'synthetic_fixture_parent' in clock):
            raise SystemExit('OLD_COLLECTION_OR_CANARY_GRANT_REFUSED')
    elif collect_only:
        if (entry, selection) not in (('guard.py', '@project'), ('native_guard.py', '@native')):
            raise SystemExit('EXACT_COLLECTION_ROUTE_REQUIRED')
        scope = 'FINITE_EXACT_CORPUS_COLLECTION_ONLY'
        if clock.get('collect_only') is not True or 'canary' in clock or 'synthetic_fixture_parent' in clock:
            raise SystemExit('EXACT_COLLECTION_CLOCK_REQUIRED')
        for key in ('start_monotonic', 'deadline_monotonic', 'original_budget_seconds'):
            value = clock.get(key)
            if type(value) not in (int, float) or not 0 < value < float('inf'):
                raise SystemExit('FINITE_ORIGINAL_COLLECTION_CLOCK_REQUIRED')
        if (clock['start_monotonic'] > time.monotonic() or
            clock['deadline_monotonic'] != clock['start_monotonic'] + clock['original_budget_seconds']):
            raise SystemExit('ORIGINAL_COLLECTION_DEADLINE_MISMATCH')
    else:
        if selection != '@canary' or entry not in ('guard.py', 'native_guard.py'):
            raise SystemExit('EXACT_CANARY_ROUTE_REQUIRED')
        scope = 'FINITE_SYNTHETIC_CANARY_ONLY'
    if not re.fullmatch(r'[a-z][a-z0-9-]{0,63}', name):
        raise SystemExit('INVALID_RUN_NAME')
    if (clock.get('execution_authorized') is not True or
        clock.get('independent_source_review') != 'PASS' or
        clock.get('scope') != scope):
        raise SystemExit('NO_SEPARATE_REVIEWED_COLLECTION_ADMISSION' if collect_only else
                         'NO_SEPARATE_REVIEWED_CANARY_ADMISSION')
    if any(clock.get(k) != v for k, v in dict(entry=entry, selection=selection,
                                            run_name=name, run_root=str(run)).items()):
        raise SystemExit('ADMISSION_ROUTE_MISMATCH')
    if not isinstance(clock.get('assignment'), str) or not clock['assignment']:
        raise SystemExit('ORIGINAL_OWNER_ASSIGNMENT_REQUIRED')
    if not isinstance(clock.get('generation'), int) or clock['generation'] < 1:
        raise SystemExit('ORIGINAL_GENERATION_REQUIRED')
    if not re.fullmatch(r'[0-9a-f]{32,64}', clock.get('nonce', '')):
        raise SystemExit('PHASE_NONCE_REQUIRED')
    if pathlib.Path('/proc/sys/kernel/random/boot_id').read_text().strip() != clock['boot_id']:
        raise SystemExit('BOOT_MISMATCH')
    if clock['deadline_monotonic'] - time.monotonic() <= 0:
        raise SystemExit('ORIGINAL_ASSIGNMENT_EXPIRED')
    source_pin_bytes = (source / 'source-pins.json').read_bytes()
    if hashlib.sha256(source_pin_bytes).hexdigest() != clock['reviewed_source_pins_sha256']:
        raise SystemExit('SOURCE_REVIEW_PIN_MISMATCH')
    pins = json.loads(source_pin_bytes)
    actual = {p.name for p in source.iterdir() if p.name != 'source-pins.json'}
    if actual != set(pins) or any('/' in p or p in ('.', '..') for p in pins):
        raise SystemExit('SOURCE_INVENTORY_DRIFT')
    for relative, expected in pins.items():
        path = source / relative
        if path.is_symlink() or hashlib.sha256(path.read_bytes()).hexdigest() != expected:
            raise SystemExit('SOURCE_DRIFT:' + relative)
    if runtest_file is not None:
        runtime_api = runpy.run_path(str(source / 'runtest_protocol.py'))
        runtime_unit = runtime_api['validate_admission'](clock, runtime_plan, runtest_file, selection, entry, runtime_sha)
        for ref in runtime_unit['conftests'] + ([runtime_unit['config']] if runtime_unit['config'] else []) + runtime_plan['runtime_plugins']:
            if hashlib.sha256(pathlib.Path(ref['path']).read_bytes()).hexdigest() != ref['sha256']:
                raise SystemExit('RUNTEST_FIXTURE_OR_PLUGIN_DRIFT')
        for ref in runtime_plan['runtime_plugins']:
            if hashlib.sha256(pathlib.Path(ref['entrypoint']).read_bytes()).hexdigest() != ref['entrypoint_sha256']:
                raise SystemExit('RUNTEST_PLUGIN_ENTRYPOINT_DRIFT')
    if runtime_unit is not None:
        phase = {'node': runtime_unit['file']['path'], 'entry_operation': 'pytest_runtest_file',
                 'guard_wall_seconds': runtime_unit['guard_wall_seconds'],
                 'outer_wall_seconds': runtime_unit['outer_wall_seconds'],
                 'requires_synthetic_descriptors': False, 'selected_files': [runtime_unit['file']['path']],
                 'unit_id': runtest_file, 'runtime_plan_sha256': runtime_sha,
                 'gate_plan_sha256': pins['gate-plan.json'], 'corpus_pins_sha256': pins['corpus-pins.json']}
    elif collect_only:
        gate_bytes = (source / 'gate-plan.json').read_bytes()
        gate_sha = hashlib.sha256(gate_bytes).hexdigest()
        if gate_sha != '14273ce2945352a75003a5e8b0066033fa9c1b4605b320c0a2bb2db2b4aef15a':
            raise SystemExit('EXACT_COLLECTION_PLAN_DRIFT')
        plan = json.loads(gate_bytes)
        phase = {'node': selection, 'entry_operation': 'pytest_collect_only',
                 'guard_wall_seconds': 120, 'outer_wall_seconds': 150,
                 'requires_synthetic_descriptors': False,
                 'selected_files': plan[selection[1:]], 'missing': plan['missing'],
                 'gate_plan_sha256': gate_sha, 'corpus_pins_sha256': pins['corpus-pins.json']}
    else:
        plan = json.loads((source / 'canary-plan.json').read_text())
        if clock.get('canary') not in plan:
            raise SystemExit('UNREVIEWED_CANARY')
        phase = plan[clock['canary']]
    if clock.get('guard_wall_seconds') != phase['guard_wall_seconds']:
        raise SystemExit('COLLECTION_WALL_MISMATCH' if collect_only else 'CANARY_WALL_MISMATCH')
    for p, expected in json.loads((source / 'runtime-pins.json').read_text()).items():
        path = pathlib.Path(p)
        if (str(path.resolve()) != expected['resolved'] or
            hashlib.sha256(path.read_bytes()).hexdigest() != expected['sha256']):
            raise SystemExit('TRUSTED_RUNTIME_DRIFT:' + p)
    for p, sha in json.loads((source / 'ancillary-pins.json').read_text()).items():
        if hashlib.sha256(pathlib.Path(p).read_bytes()).hexdigest() != sha:
            raise SystemExit('ANCILLARY_DRIFT:' + p)
    env = {'PATH': '/usr/bin:/bin', 'LANG': 'C.UTF-8',
           'HOME': str(run / 'fixtures'), 'TMPDIR': str(run / 'tmp'),
           'PYTHONDONTWRITEBYTECODE': '1', 'PYTEST_DISABLE_PLUGIN_AUTOLOAD': '1'}
    if runtest_file is not None:
        env.pop('PYTEST_DISABLE_PLUGIN_AUTOLOAD')
    if dict(os.environ) != env:
        raise SystemExit('CLEAN_PRE_INTERPRETER_ENVIRONMENT_MISMATCH')
    for root in (run, run.parent, run / 'fixtures', run / 'tmp', run / 'pytesttmp'):
        st = root.lstat()
        if root.is_symlink() or not root.is_dir() or st.st_uid != os.getuid() or st.st_mode & 0o777 != 0o700:
            raise SystemExit('PRIVATE_PREPARED_DIRECTORY_REQUIRED')
    if (run / 'clock.json').is_symlink() or (run / 'clock.json').stat().st_mode & 0o777 != 0o400:
        raise SystemExit('IMMUTABLE_PHASE_CLOCK_REQUIRED')
    path_api = runpy.run_path(str(source / 'import_path_readiness.py'))
    path_manifest = path_api['load'](source)
    path_api['verify_snapshot'](path_manifest)
    path_api['verify_startup'](path_manifest)
    # Last read-only check: no fixture/receipt effects yet. O_PATH metadata
    # descriptors above were closed; no inheritance mutation or archive open.
    if (run / 'clock.json').read_bytes() != before:
        raise SystemExit('PHASE_CLOCK_DRIFT')
    if clock['deadline_monotonic'] - time.monotonic() <= 0:
        raise SystemExit('ORIGINAL_ASSIGNMENT_EXPIRED')
    return clock, phase, hashlib.sha256(before).hexdigest()


def identity(clock, phase):
    if clock.get('runtest_file') is not None:
        return {k: clock[k] for k in ('assignment', 'generation', 'nonce', 'entry', 'selection',
                'run_name', 'run_root', 'reviewed_source_pins_sha256')} | {
                'collect_only': False, 'scope': clock['scope'], 'unit_id': phase['unit_id'],
                'node': phase['node'], 'operation': 'pytest_runtest_file',
                'runtime_plan_sha256': phase['runtime_plan_sha256'],
                'gate_plan_sha256': phase['gate_plan_sha256'], 'corpus_pins_sha256': phase['corpus_pins_sha256']}
    if clock['selection'] in ('@project', '@native'):
        return {k: clock[k] for k in ('assignment', 'generation', 'nonce', 'entry',
                'selection', 'run_name', 'run_root', 'reviewed_source_pins_sha256')} | {
                    'collect_only': True, 'scope': clock['scope'],
                    'node': phase['node'], 'operation': phase['entry_operation'],
                    'gate_plan_sha256': phase['gate_plan_sha256'],
                    'corpus_pins_sha256': phase['corpus_pins_sha256'],
                    'selected_file_count': len(phase['selected_files'])}
    return {k: clock[k] for k in ('assignment', 'generation', 'nonce', 'entry',
            'selection', 'run_name', 'run_root', 'canary', 'reviewed_source_pins_sha256')} | {
                'node': phase['node'], 'operation': phase['entry_operation']}
