def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def save(path, value, *, new=False):
    assert path.is_relative_to(E) and not path.is_symlink()
    data = (json.dumps(value, ensure_ascii=False, indent=2) + '\n').encode()
    if new:
        fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600)
    else:
        fd = os.open(path, os.O_WRONLY | os.O_TRUNC | os.O_NOFOLLOW)
    with os.fdopen(fd, 'wb') as f:
        f.write(data); f.flush(); os.fsync(f.fileno())


def refs(root):
    return {str(p.relative_to(root)): sha(p) for p in sorted(root.rglob('*')) if p.is_file()}


def relocate(value):
    if isinstance(value, dict):
        return {relocate(k): relocate(v) for k, v in value.items()}
    if isinstance(value, list):
        return [relocate(v) for v in value]
    if isinstance(value, str) and value.startswith(str(J) + '/'):
        return str(C) + value[len(str(J)):]
    return value
