"""Source-only completion of the donor's /browser command route."""
from pathlib import Path
import ast, shutil
W = Path(__file__).parent
B = Path('<SOL_WORK>/FRW026-ORDINARY-SKILLS-FINAL-GATE-native')
C = W / 'candidate/native'
routes = {
    'hermes_cli/cli_commands_mixin.py': ['_browser_connect', '_launch_default_cdp_browser',
                                      '_browser_use', '_browser_status', '_browser_disconnect'],
    'hermes_cli/browser_connect.py': ['launch_chrome_debug', 'is_browser_debug_ready',
        'discover_local_cdp_url', 'local_port_in_use', 'find_free_debug_port',
        'detect_default_chromium', 'snapshot_real_profile', 'close_browser_holding_profile'],
}
for name, names in routes.items():
    p = C / name
    shutil.copyfile(B / name, p)
    s = p.read_text()
    tree = ast.parse(s)
    body = tree.body
    insertion = 0
    for node in body:
        if (isinstance(node, ast.Expr) and isinstance(node.value, ast.Constant) and isinstance(node.value.value, str)
                or isinstance(node, ast.ImportFrom) and node.module == '__future__'):
            insertion = node.end_lineno
        else:
            break
    lines = s.splitlines(True)
    functions = {n.name: n for n in body if isinstance(n, ast.FunctionDef)}
    for target in sorted(names, key=lambda n: functions[n].lineno, reverse=True):
        lines.insert(functions[target].lineno - 1, '@browser_action_entry\n')
    lines.insert(insertion, '\nfrom hermes_cli.browser_runtime import browser_action_entry\n')
    p.write_text(''.join(lines))

p = C / 'tools/browser_tool_session.py'
s = p.read_text()
s = s.replace('''    stdout_path = os.path.join(task_socket_dir, f"_stdout_{command}")
    stderr_path = os.path.join(task_socket_dir, f"_stderr_{command}")''', '''    # Per-command files cannot overwrite another unresolved client's evidence,
    # even when two callers race before the locked creation-slot check.
    output_tag = f'{command}_{uuid.uuid4().hex}' if managed else command
    stdout_path = os.path.join(task_socket_dir, f"_stdout_{output_tag}")
    stderr_path = os.path.join(task_socket_dir, f"_stderr_{output_tag}")''', 1)
s = s.replace('''proc = _popen_agent_browser(cmd_parts, browser_env, task_socket_dir, command, stdin_payload,''',
              '''proc = _popen_agent_browser(cmd_parts, browser_env, task_socket_dir, output_tag, stdin_payload,''', 1)
p.write_text(s)
print('Source-gated /browser commands, direct CDP launch/discovery, and unique client diagnostics.')
