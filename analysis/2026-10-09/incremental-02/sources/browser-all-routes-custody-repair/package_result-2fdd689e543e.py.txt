def pin(path):
    path = Path(path); raw = path.read_bytes()
    return {'path': str(path), 'sha256': hashlib.sha256(raw).hexdigest(), 'bytes': len(raw)}
def write(name, data):
    (W / name).write_text(json.dumps(data, indent=2, ensure_ascii=False) + '\n')
def ref(rel, symbols=()):
    p = C / rel; tree = ast.parse(p.read_text())
    nodes = {n.name: n for n in ast.walk(tree) if isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef))}
    return dict(pin(p), symbols={name: nodes[name].lineno for name in symbols})
def dump_assignment(path, name):
    tree = ast.parse(path.read_text())
    value = next(n.value for n in tree.body if isinstance(n, ast.Assign)
                 and any(isinstance(t, ast.Name) and t.id == name for t in n.targets))
    return ast.dump(value, include_attributes=False)

# Reconstruct every full overlay in memory, including the project-embedded
# native patch and manifest. Source text is never imported or evaluated.
reconstructed = {}
for tree in ('native', 'project'):
    lines = (W / (tree + '.patch')).read_text().splitlines(True)
    index = 0; verified = []
    while index < len(lines):
        assert lines[index].startswith('--- ')
        old_name = lines[index][4:].strip(); index += 1
        assert lines[index].startswith('+++ b/')
        rel = lines[index][6:].strip(); index += 1
        old_path = B[tree] / rel
        original = old_path.read_text().splitlines(True) if old_name != '/dev/null' else []
        position = 0; output = []
        while index < len(lines) and not lines[index].startswith('--- '):
            match = re.match(r'@@ -(\d+)(?:,(\d+))? \+(\d+)(?:,(\d+))? @@', lines[index])
            assert match, lines[index]
            start = max(0, int(match[1]) - 1)
            assert start >= position
            output.extend(original[position:start]); position = start; index += 1
            old_count = new_count = 0
            while index < len(lines) and not lines[index].startswith(('@@ ', '--- ')):
                line = lines[index]; prefix = line[0]; content = line[1:]
                assert prefix in ' +-'
                if prefix in ' -':
                    assert original[position] == content, (tree, rel, position)
                    position += 1; old_count += 1
                if prefix in ' +':
                    output.append(content); new_count += 1
                index += 1
            assert old_count == int(match[2] if match[2] is not None else 1)
            assert new_count == int(match[4] if match[4] is not None else 1)
        output.extend(original[position:])
        assert ''.join(output).encode() == (C / tree / rel).read_bytes(), (tree, rel)
        verified.append(rel)
    reconstructed[tree] = verified

route_files = ['tools/browser_cdp_tool.py', 'tools/browser_dialog_tool.py',
               'tools/browser_use_cli.py', 'tools/browser_vault_tool.py']
routes = []
for rel in route_files:
    parsed = ast.parse((C / 'native' / rel).read_text())
    functions = {n.name: n for n in parsed.body if isinstance(n, ast.FunctionDef)}
    for node in ast.walk(parsed):
        if not isinstance(node, ast.Call) or not isinstance(node.func, ast.Attribute) or node.func.attr != 'register':
            continue
        kw = {k.arg: k.value for k in node.keywords}
        name = ast.literal_eval(kw['name'])
        fn = functions[name]
        assert any(isinstance(d, ast.Name) and d.id == 'browser_action_entry' for d in fn.decorator_list)
        routes.append({'name': name, 'module': rel, 'definition_line': fn.lineno,
                       'registration_line': node.lineno, 'toolset': ast.literal_eval(kw['toolset']),
                       'gate': 'browser_action_entry -> managed_browser_operation -> check_managed_execution -> require_execution_admission'})
facade = C / 'native/tools/browser_tool.py'
parsed = ast.parse(facade.read_text())
table = next(n.value for n in parsed.body if isinstance(n, ast.Assign)
             and any(isinstance(t, ast.Name) and t.id == '_BROWSER_TOOL_TABLE' for t in n.targets))
for row in table.elts:
    routes.append({'name': ast.literal_eval(row.elts[0]), 'module': 'tools/browser_tool.py',
                   'gate': '_managed_entry -> browser_action_entry; registry _routed_handler -> managed_browser_operation'})
assert len(routes) == 18 and len({r['name'] for r in routes}) == 18

# Trace donor discovery inputs without importing discover_builtin_tools or
# trusting its mutable cache/check_fn availability. Scan all retained modules.
discovered = []
for source in sorted(list((B['native'] / 'tools').glob('*.py')) + list((B['native'] / 'tools').glob('*/tool.py'))):
    source_rel = source.relative_to(B['native'])
    effective = C / 'native' / source_rel
    if not effective.exists(): effective = source
    for node in ast.walk(ast.parse(effective.read_text())):
        if not isinstance(node, ast.Call) or not isinstance(node.func, ast.Attribute) or node.func.attr != 'register':
            continue
        kw = {k.arg: k.value for k in node.keywords}
        name, toolset = kw.get('name'), kw.get('toolset')
        if isinstance(toolset, ast.Constant) and isinstance(toolset.value, str) and toolset.value.startswith('browser'):
            discovered.append({'module': str(source_rel), 'name': name.value if isinstance(name, ast.Constant) else 'DYNAMIC_BROWSER_TOOL_TABLE',
                               'toolset': toolset.value})
assert {r['name'] for r in discovered if r['name'] != 'DYNAMIC_BROWSER_TOOL_TABLE'} == {r['name'] for r in routes if r['module'] != 'tools/browser_tool.py'}
