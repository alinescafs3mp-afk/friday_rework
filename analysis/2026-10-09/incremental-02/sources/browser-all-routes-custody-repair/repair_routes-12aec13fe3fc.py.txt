"""Text/AST source transformation only; never import the candidate."""
from pathlib import Path
import ast

W = Path(__file__).parent
C = W / 'candidate/native'
p = C / 'hermes_cli/browser_runtime.py'
s = p.read_text()
at = s.index('\ndef friday_browser_config():')
s = s[:at] + '''
def browser_action_entry(fn=None, *, timeout_fn=None):
    """Same execution fence for facade, separate tools and direct driver routes.

    Availability/check_fn, operator identity and an existing supervisor are not
    admission. Keep legacy APIs when no managed policy is selected; a managed
    call retains its original ceiling across nested sync/async routing.
    """
    from functools import wraps
    import inspect
    if fn is None:
        return lambda target: browser_action_entry(target, timeout_fn=timeout_fn)
    signature = inspect.signature(fn)

    def operation(args, kwargs):
        if timeout_fn is not None:
            timeout = timeout_fn()
        else:
            bound = signature.bind_partial(*args, **kwargs)
            bound.apply_defaults()
            timeout = bound.arguments.get('timeout', bound.arguments.get('timeout_s', 30.0))
        return managed_browser_operation(timeout)

    if inspect.iscoroutinefunction(fn):
        @wraps(fn)
        async def async_call(*args, **kwargs):
            if local_policy() is None:
                return await fn(*args, **kwargs)
            with operation(args, kwargs):
                return await fn(*args, **kwargs)
        return async_call

    @wraps(fn)
    def call(*args, **kwargs):
        if local_policy() is None:
            return fn(*args, **kwargs)
        with operation(args, kwargs):
            return fn(*args, **kwargs)
    return call

''' + s[at:]
p.write_text(s)

p = C / 'tools/browser_tool.py'
s = p.read_text()
start, end = s.index('def _managed_entry(fn):'), s.index('\n\nfor _entry in _BROWSER_TOOL_TABLE:')
s = s[:start] + '''def _managed_entry(fn):
    from hermes_cli.browser_runtime import browser_action_entry
    return browser_action_entry(fn, timeout_fn=(
        lambda: _get_open_command_timeout(first_open=True)
        if fn.__name__ == 'browser_navigate' else _safe_command_timeout()))
''' + s[end:]
p.write_text(s)

routes = {
    'browser_cdp_tool.py': ['browser_cdp', '_resolve_cdp_endpoint', '_cdp_call', '_browser_cdp_via_supervisor'],
    'browser_dialog_tool.py': ['browser_dialog'],
    'browser_extension_router.py': ['route_browser_tool', 'routed_browser_handler'],
    'browser_tool_cdp.py': ['_resolve_cdp_override', '_get_cdp_override', '_ensure_cdp_supervisor'],
    'browser_use_cli.py': ['browser_exec', '_run_cli_killing_process_group', 'stop_harness_daemons',
        '_route_backend', '_resolve_backend_cdp', '_resolve_lightpanda_cdp',
        '_resolve_managed_chromium_cdp', '_reach_sandbox_cdp', '_resolve_real_profile_cdp',
        '_attach_vault_supervisor'],
    'browser_vault_tool.py': ['browser_vault_list', 'browser_vault_unlock', 'browser_vault_save_login',
        'browser_vault_enter_code', 'browser_vault_fill', '_fenced_page_op',
        '_eval_js', '_eval_js_secret', '_ensure_supervisor'],
}
for name, names in routes.items():
    p = C / 'tools' / name
    s = p.read_text()
    tree = ast.parse(s)
    # Insert after module docstring and __future__ imports, before any routes.
    body = tree.body
    insertion = 0
    for node in body:
        if (isinstance(node, ast.Expr) and isinstance(node.value, ast.Constant) and isinstance(node.value.value, str)
                or isinstance(node, ast.ImportFrom) and node.module == '__future__'):
            insertion = node.end_lineno
        else:
            break
    lines = s.splitlines(True)
    functions = {n.name: n for n in body if isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef))}
    for target in sorted(names, key=lambda n: functions[n].lineno, reverse=True):
        lines.insert(functions[target].lineno - 1, '@browser_action_entry\n')
    lines.insert(insertion, '\nfrom hermes_cli.browser_runtime import browser_action_entry\n')
    p.write_text(''.join(lines))
print('Repaired facade and 30 separate/direct browser action boundaries; source only.')
