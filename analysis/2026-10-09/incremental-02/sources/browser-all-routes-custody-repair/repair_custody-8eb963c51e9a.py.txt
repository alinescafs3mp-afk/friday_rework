def _require_managed_client_slot(task_id, session_info):

def _confirm_managed_client_wait(task_id, session_info, proc, returncode):
    """Record only the result of an actual wait on this exact original child."""
    with _bt._cleanup_lock:
        custody = session_info.get('_friday_direct_client')
        if (_bt._active_sessions.get(task_id) is not session_info or not custody

def _reconcile_managed_client(task_id, session_info):
    """Existing same-session stop path; no fresh identity, owner or grace clock.

                or custody['task_id'] != task_id
                or custody['session_name'] != session_info.get('session_name')
                or custody['proc'] is None or custody['proc'].pid != custody['pid']):
            raise RuntimeError('browser_original_client_custody_changed')

    # Even a successful direct wait proves nothing about forked daemon/Chrome,
    # listeners or whole scope cessation. The caller retains STOP_UNCONFIRMED.
    return custody
