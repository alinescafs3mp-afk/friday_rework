// Source only. Use once under the existing independently admitted original900.
// No actual native phase exists in this source publication.
async function inspectNativeThenObserve(phase, helper, out) {
  const inspector = "<ASTRA_WORK>/native-collection-inspection-repair/candidate/inspect_native.py";
  const observer = "<ASTRA_WORK>/native-lexical-import-repair/candidate/owner-helper/owner_observer.py";
  const quote = value => "'" + value.replace(/'/g, "'\\''") + "'";
  if (phase !== out + "/collect-native") throw Error("foreign native phase");
  if (helper !== observer) throw Error("foreign existing observer");
  let inspected = await tools.exec_command({
    cmd: "/usr/bin/gnutimeout --signal=KILL 20s /usr/bin/python3 -I -B " +
         quote(inspector) + " " + quote(phase + "/classification-source-basis.json"),
    yield_time_ms: 1000, max_output_tokens: 1200
  });
  let output = inspected.output;
  // These calls finish the single owned utility handle; they do not poll a peer.
  while (inspected.session_id && inspected.exit_code === undefined) {
    inspected = await tools.write_stdin({session_id: inspected.session_id, chars: "",
                                       yield_time_ms: 1000, max_output_tokens: 1200});
    output += inspected.output;
  }
  if (inspected.exit_code !== 0 || inspected.session_id) {
    text({stage: "native-owner-inspection", result: inspected});
    return false;
  }
  const receipt = JSON.parse(output.trim());
  if (receipt.state !== "INDEPENDENT_COLLECTION_METADATA_RECONCILED" ||
      receipt.path !== phase + "/classification-review.json" || receipt.runtime_acceptance !== false ||
      receipt.owner_inspection?.modules !== 53 ||
      !Number.isInteger(receipt.owner_inspection?.definitions) || receipt.owner_inspection.definitions < 0 ||
      receipt.owner_inspection?.deselected_definitions !== 8 || !/^[a-f0-9]{64}$/.test(receipt.sha256)) {
    throw Error("invalid native publication receipt");
  }
  // Existing observer retains independent actual backend, receipt graph, clocks,
  // immutable inputs, all selected module ordering and stop/cleanup predicates.
  let observed = await tools.exec_command({cmd: "/usr/bin/python3 -I -B " + quote(helper) + " observe " + quote(out) + " 3",
                                          yield_time_ms: 1000, max_output_tokens: 1200});
  while (observed.session_id && observed.exit_code === undefined) {
    observed = await tools.write_stdin({session_id: observed.session_id, chars: "",
                                      yield_time_ms: 1000, max_output_tokens: 1200});
  }
  text({stage: "native-observe", result: observed});
  return observed.exit_code === 0 && !observed.session_id;
}
