"""Finite SYNTHETIC layout/publication controls; never actual collection evidence.

Only this future separately admitted control creates synthetic clocks, under its
runner TMPDIR. No owner helper, corpus or canonical join is executed. Publication
inner inspection is stubbed to isolate filesystem/clock predicates; the existing
canonical four-JSON join remains unchanged and needs real collection evidence.
"""
from pathlib import Path
import hashlib
import json
import os
import runpy
import stat
import tempfile
import unittest
from unittest.mock import patch

HERE = Path(__file__).resolve().parent
PINS = json.loads((HERE / 'source-pins.json').read_bytes())
INSPECTOR = HERE / 'inspect_native.py'
assert hashlib.sha256(INSPECTOR.read_bytes()).hexdigest() == PINS['inspect_native.py']
NS = runpy.run_path(str(INSPECTOR), run_name='synthetic_layout_controls_only')
G = NS['original_clock'].__globals__


class OwnerLayoutControls(unittest.TestCase):
    def setUp(self):
        self.root = Path(tempfile.mkdtemp(prefix='synthetic-owner-layout-', dir=os.environ['TMPDIR']))
        self.addCleanup(self.cleanup_layout)
        self.out = self.root / 'owner'; self.out.mkdir(mode=0o700)
        self.phase = self.out / 'collect-native'; self.phase.mkdir(mode=0o700)
        self.run = self.phase / 'run'; self.run.mkdir(mode=0o700)
        self.map = self.root / 'lexical-qualification-binding/classification/provenance.json'
        self.map.parent.mkdir(parents=True)
        self.producer = self.root / 'producer'; self.producer.mkdir()
        self.producer_pins = self.producer / 'source-pins.json'
        self.put(self.producer_pins, {'SYNTHETIC': 'NOT_ACTUAL_SOURCE'})
        self.put(self.map, {'current_producer': {'path': str(self.producer / 'controls.py')}})
        self.reconcile = self.map.with_name('reconcile.py')
        self.reconcile.write_text('# SYNTHETIC: never executed\n')
        self.clock_path = self.out / 'original-clock.json'
        self.clock = {'assignment': 'SYNTHETIC_NOT_A_GRANT', 'generation': 1,
                      'original_budget_seconds': 900, 'start_monotonic': 1000.0,
                      'deadline_monotonic': 1900.0,
                      'boot_id': Path('/proc/sys/kernel/random/boot_id').read_text().strip(),
                      'execution_authorized': True, 'independent_source_review': 'PASS',
                      'scope': 'FINITE_TWO_POSITIVE_CANARIES_AND_EXACT_COLLECTION_ONLY',
                      'resources': {'cpu': 2, 'memory_bytes': 2 << 30, 'children': 0, 'max_phases': 4},
                      'reviewed_source_pins_sha256': self.sha(self.producer_pins)}
        self.grant_path = self.root / 'synthetic-admission.json'
        self.grant = {'state': 'INDEPENDENT_OWNER_HELPER_AND_COLLECTION_ADMITTED',
                      'execution_authorized': True, 'output_root': str(self.out),
                      'source_pins_sha256': self.sha(self.producer_pins),
                      'source_review': str(self.root / 'SYNTHETIC-REVIEW-NOT-REAL'),
                      'source_review_sha256': '0' * 64}
        self.pre = {'admission': str(self.grant_path),
                    'source_review': self.grant['source_review'],
                    'source_review_sha256': self.grant['source_review_sha256']}
        self.rebind()
        self.put(self.out / 'input-pins.json', {'hashes': {str(self.producer_pins): self.sha(self.producer_pins)}})
        self.inputs = [self.run / ('collect-native-' + suffix + '.json')
                       for suffix in ('results', 'classified', 'deselected', 'collection')]
        for path in self.inputs:
            self.put(path, {'SYNTHETIC_LAYOUT_ONLY': path.name})
        self.basis = self.phase / 'classification-source-basis.json'
        self.put(self.basis, {'input_refs': [{'path': str(p), 'sha256': self.sha(p)} for p in self.inputs]})
        self.checked = {'state': 'INDEPENDENT_COLLECTION_METADATA_RECONCILED',
                        'acceptance': False, 'SYNTHETIC_LAYOUT_ONLY': True,
                        'owner_inspection': {'modules': 53, 'definitions': 0, 'deselected_definitions': 8}}
        self.mapping = patch.dict(G, MAP=self.map, MAP_SHA=self.sha(self.map),
                                  RECONCILE=self.reconcile, RECONCILE_SHA=self.sha(self.reconcile))
        self.mapping.start(); self.addCleanup(self.mapping.stop)
        self.now = patch.object(G['time'], 'monotonic', return_value=1100.0)
        self.now.start(); self.addCleanup(self.now.stop)

    def cleanup_layout(self):
        # Use explicit absolute synthetic paths so the unchanged runner audit
        # never needs a dir_fd exception or expanded write permissions.
        for base, dirs, files in os.walk(self.root, topdown=False, followlinks=False):
            for name in files:
                (Path(base) / name).unlink()
            for name in dirs:
                path = Path(base) / name
                if path.is_symlink():
                    path.unlink()
                else:
                    path.rmdir()
        self.root.rmdir()

    @staticmethod
    def sha(path):
        return hashlib.sha256(path.read_bytes()).hexdigest()

    @staticmethod
    def put(path, value):
        # Synthetic files only; production publication remains exclusive.
        if path.exists():
            path.unlink()
        path.write_text(json.dumps(value) + '\n'); path.chmod(0o400)

    def rebind(self):
        self.put(self.clock_path, self.clock)
        self.grant['original_clock_sha256'] = self.sha(self.clock_path)
        self.put(self.grant_path, self.grant)
        self.pre.update(clock_sha256=self.sha(self.clock_path), admission_sha256=self.sha(self.grant_path))
        self.put(self.out / 'preflight.json', self.pre)

    def publish(self, inspection=None):
        with patch.dict(G, inspect_native=inspection or (lambda basis, paths: self.checked)):
            return NS['publish'](self.basis)

    def test_exact_owner_layout_accepts_original_without_root_clock_alias(self):
        self.assertFalse((self.out / 'clock.json').exists())
        self.assertEqual(NS['original_clock'](self.phase, reserve=20), self.clock)

    def test_missing_original_cannot_use_root_or_phase_clock(self):
        self.clock_path.rename(self.out / 'clock.json')
        self.put(self.run / 'clock.json', self.clock)
        with self.assertRaises(FileNotFoundError):
            NS['original_clock'](self.phase)

    def test_clock_semantic_negatives(self):
        cases = [
            ('expired', {'start_monotonic': 100.0, 'deadline_monotonic': 1000.0}, 'EXPIRED_RESET'),
            ('future', {'start_monotonic': 1200.0, 'deadline_monotonic': 2100.0}, 'EXPIRED_RESET'),
            ('reset', {'deadline_monotonic': 2000.0}, 'EXPIRED_RESET'),
            ('budget', {'original_budget_seconds': 901}, 'FOREIGN_OR_RESET'),
            ('boot', {'boot_id': 'FOREIGN'}, 'EXPIRED_RESET'),
            ('generation', {'generation': True}, 'FOREIGN_OR_RESET'),
            ('scope', {'scope': '@native'}, 'NO_EXISTING_OWNER'),
            ('grant', {'execution_authorized': False}, 'NO_EXISTING_OWNER'),
            ('review', {'independent_source_review': 'NOT_RUN'}, 'NO_EXISTING_OWNER'),
            ('resources', {'resources': {'cpu': 3}}, 'NO_EXISTING_OWNER'),
            ('producer', {'reviewed_source_pins_sha256': '0' * 64}, 'PRODUCER_PIN'),
        ]
        original = dict(self.clock)
        for name, change, reason in cases:
            with self.subTest(case=name):
                self.clock = dict(original, **change); self.rebind()
                with self.assertRaisesRegex(ValueError, reason):
                    NS['original_clock'](self.phase)

    def test_reserve_boundary_is_strict(self):
        with patch.object(G['time'], 'monotonic', return_value=1880.0):
            with self.assertRaisesRegex(ValueError, 'EXPIRED_RESET'):
                NS['original_clock'](self.phase, reserve=20)
        with patch.object(G['time'], 'monotonic', return_value=1879.0):
            self.assertEqual(NS['original_clock'](self.phase, reserve=20), self.clock)

    def test_immutable_files_and_preflight_binding(self):
        for name in ('original-clock.json', 'preflight.json', 'input-pins.json'):
            with self.subTest(mutable=name):
                path = self.out / name; path.chmod(0o600)
                with self.assertRaisesRegex(ValueError, 'FOREIGN_OR_MUTABLE'):
                    NS['original_clock'](self.phase)
                path.chmod(0o400)
        self.pre['clock_sha256'] = '0' * 64
        self.put(self.out / 'preflight.json', self.pre)
        with self.assertRaisesRegex(ValueError, 'ORIGINAL_CLOCK_BYTES_CHANGED'):
            NS['original_clock'](self.phase)

    def test_symlink_hardlink_and_missing_preflight_refused(self):
        original = self.root / 'original-copy.json'; self.clock_path.rename(original)
        self.clock_path.symlink_to(original)
        with self.assertRaisesRegex(ValueError, 'FOREIGN_OR_MUTABLE'):
            NS['original_clock'](self.phase)
        self.clock_path.unlink(); os.link(original, self.clock_path)
        with self.assertRaisesRegex(ValueError, 'FOREIGN_OR_MUTABLE'):
            NS['original_clock'](self.phase)
        original.unlink(); (self.out / 'preflight.json').unlink()
        with self.assertRaises(FileNotFoundError):
            NS['original_clock'](self.phase)

    def test_admission_and_inventory_binding(self):
        self.grant['output_root'] = str(self.root / 'foreign'); self.rebind()
        with self.assertRaisesRegex(ValueError, 'ORIGINAL_CLOCK_NOT_BOUND'):
            NS['original_clock'](self.phase)
        self.grant['output_root'] = str(self.out); self.rebind()
        self.pre['admission_sha256'] = '0' * 64; self.put(self.out / 'preflight.json', self.pre)
        with self.assertRaisesRegex(ValueError, 'OWNER_ADMISSION_CHANGED'):
            NS['original_clock'](self.phase)
        self.rebind(); self.put(self.out / 'input-pins.json', {'hashes': {}})
        with self.assertRaisesRegex(ValueError, 'PRODUCER_PIN'):
            NS['original_clock'](self.phase)

    def test_closed_owner_or_phase_refuses(self):
        for path in (self.out / 'result.json', self.phase / 'cessation.json'):
            with self.subTest(closed=path.name):
                self.put(path, {'SYNTHETIC_CLOSED': True})
                with self.assertRaisesRegex(ValueError, 'CLOSED_OWNER_OR_PHASE'):
                    self.publish()
                self.assertFalse((self.phase / 'classification-review.json').exists())
                path.unlink()

    def test_publication_is_immutable_exact_receipt_and_no_replay(self):
        receipt = self.publish(); target = self.phase / 'classification-review.json'
        self.assertEqual(json.loads(target.read_bytes()), self.checked)
        self.assertEqual(stat.S_IMODE(target.stat().st_mode), 0o400)
        self.assertEqual(receipt['sha256'], self.sha(target))
        self.assertFalse(receipt['runtime_acceptance'])
        with self.assertRaisesRegex(ValueError, 'CLOSED_OR_REVIEWED_PHASE'):
            self.publish()

    def test_bad_inputs_and_inspection_do_not_publish(self):
        self.put(self.inputs[0], {'DRIFT': True})
        with self.assertRaisesRegex(ValueError, 'ACTUAL_INPUT_DRIFT'):
            self.publish()
        self.assertFalse((self.phase / 'classification-review.json').exists())
        self.put(self.inputs[0], {'SYNTHETIC_LAYOUT_ONLY': self.inputs[0].name})
        def refuse(basis, paths):
            raise ValueError('SYNTHETIC_INNER_REFUSAL')
        with self.assertRaisesRegex(ValueError, 'SYNTHETIC_INNER_REFUSAL'):
            self.publish(refuse)
        self.assertFalse((self.phase / 'classification-review.json').exists())

    def test_input_clock_or_close_during_inspection_does_not_publish(self):
        for kind in ('input', 'clock', 'closed'):
            with self.subTest(drift=kind):
                def change(basis, paths):
                    if kind == 'input':
                        self.put(paths[0], {'DRIFT': True})
                    elif kind == 'clock':
                        self.clock['assignment'] += '-DRIFT'; self.rebind()
                    else:
                        self.put(self.phase / 'cessation.json', {})
                    return self.checked
                with self.assertRaisesRegex(ValueError, 'ACTUAL_INPUT_DRIFT|OWNER_CLOCK_DRIFT|CLOSED_OWNER_OR_PHASE'):
                    self.publish(change)
                self.assertFalse((self.phase / 'classification-review.json').exists())
                self.put(self.inputs[0], {'SYNTHETIC_LAYOUT_ONLY': self.inputs[0].name})
                closed = self.phase / 'cessation.json'
                if closed.exists():
                    closed.unlink()


if __name__ == '__main__':
    unittest.main()
