    return hashlib.sha256(data).hexdigest()

def expr(node):
    return ast.unparse(node)

def target_module(module):
    if not module:
        return None
    candidates = [ROOTS['project'] / 'tests' / (module.replace('.', '/') + '.py'),
                  ROOTS['native'] / (module.replace('.', '/') + '.py'),
                  ROOTS['project'] / (module.replace('.', '/') + '.py')]
    candidates += [p.with_suffix('') / '__init__.py' for p in candidates]
    return next((str(p) for p in candidates if str(p) in pins), None)

def location(node, qualified):
    return {'qualified_name': qualified, 'line': node.lineno, 'end_line': node.end_lineno}

def walk_scoped(node):
    yield node
    for child in ast.iter_child_nodes(node):
        if isinstance(child, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef, ast.Lambda)) and child is not node:
            continue
        yield from walk_scoped(child)

def parse_file(path, role):
    if path in files:
        return files[path]
    assert time.monotonic() < CLOCK['deadline_monotonic']
    data = pathlib.Path(path).read_bytes()
    digest = sha(data)
    assert pins.get(path) == digest, 'UNPINNED_OR_DRIFT:' + path
    text = data.decode('utf-8')
    tree = ast.parse(text, filename=path)
    trees[path] = tree
    result = {'path': path, 'sha256': digest, 'role': role, 'imports': [], 'classes': [],
              'definitions': [], 'module_declarations': [], 'dynamic_fixture_lookups': []}
    files[path] = result

    def process(nodes, scope='', enclosing_class=None, lexical_conditions=()):
        for node in nodes:
            if isinstance(node, (ast.Import, ast.ImportFrom)):
                row = {'line': node.lineno, 'scope': scope or '<module>', 'statement': expr(node)}
                if isinstance(node, ast.ImportFrom):
                    row['module'] = node.module
                    row['level'] = node.level
                    row['aliases'] = {a.asname or a.name: a.name for a in node.names}
                    row['source_candidate'] = target_module(node.module) if node.level == 0 else None
                else:
                    row['aliases'] = {a.asname or a.name.split('.')[0]: a.name for a in node.names}
                result['imports'].append(row)
            elif isinstance(node, ast.ClassDef):
                q = scope + ('.' if scope else '') + node.name
                row = location(node, q)
                row.update(bases=[expr(b) for b in node.bases], decorators=[expr(d) for d in node.decorator_list],
                           lexical_conditions=list(lexical_conditions), declarations=[])
                for n in node.body:
                    if isinstance(n, (ast.Assign, ast.AnnAssign)) and any(s in expr(n) for s in ('pytestmark', '__test__')):
                        row['declarations'].append({'line': n.lineno, 'source': expr(n)})
                result['classes'].append(row)
                process(node.body, q, q, lexical_conditions)
            elif isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
                q = scope + ('.' if scope else '') + node.name
                decorators = [expr(d) for d in node.decorator_list]
                fixtures = [d for d in node.decorator_list if expr(d).split('(')[0].split('.')[-1] == 'fixture']
                is_test = node.name.startswith('test') and '<locals>' not in scope
                kind = 'fixture' if fixtures else 'test_definition' if is_test else 'lifecycle' if node.name.startswith(('setup_', 'teardown_')) or node.name in ('setUp','tearDown','setUpClass','tearDownClass','setUpModule','tearDownModule') else 'helper'
                row = location(node, q)
                args = [a.arg for a in node.args.posonlyargs + node.args.args + node.args.kwonlyargs if a.arg not in ('self', 'cls')]
                row.update(kind=kind, async_function=isinstance(node, ast.AsyncFunctionDef),
                           arguments=args, decorators=decorators, enclosing_class=enclosing_class,
                           lexical_conditions=list(lexical_conditions),
                           source_span_sha256=sha(ast.get_source_segment(text, node).encode()),
                           calls=[], skip_calls=[], effect_signals=[], parametrize=[],
                           signature=expr(node.args))
                for d in node.decorator_list:
                    if isinstance(d, ast.Call) and expr(d.func).endswith('.parametrize'):
                        row['parametrize'].append({'line': d.lineno, 'source': expr(d),
                            'argument_names': expr(d.args[0]) if d.args else None,
                            'values_expression': expr(d.args[1]) if len(d.args) > 1 else None})
                if fixtures:
                    d = fixtures[0]
                    opts = {k.arg: expr(k.value) for k in d.keywords} if isinstance(d, ast.Call) else {}
                    row['fixture'] = {'name': node.name, 'options': opts, 'dependencies': args,
                                      'autouse_literal': opts.get('autouse') == 'True', 'scope_expression': opts.get('scope', "'function'")}
                    if 'name' in opts:
                        try: row['fixture']['name'] = ast.literal_eval(opts['name'])
                        except (ValueError, SyntaxError): row['fixture']['name_dynamic'] = opts['name']
                calls = set()
                for n in walk_scoped(node):
                    if not isinstance(n, ast.Call):
                        continue
                    call = expr(n.func)
                    calls.add(call)
                    if call.endswith(('.skip', '.importorskip')):
                        row['skip_calls'].append({'line': n.lineno, 'source': expr(n)})
                    signals = []
                    if any(v in call for v in ('subprocess.', 'multiprocessing.', 'mp.get_context', '.Process', '.Pool', 'os.fork', 'os.system', 'os.popen', 'Popen', 'create_subprocess')):
                        signals.append('process_call_candidate')
                    if any(v in call for v in ('socket', 'HTTPServer', 'TCPServer', 'serve_forever', 'requests.', 'httpx.', 'urllib.request')):
                        signals.append('network_or_http_call_candidate')
                    if call.endswith('getfixturevalue'):
                        value = expr(n.args[0]) if n.args else None
                        result['dynamic_fixture_lookups'].append({'qualified_name': q, 'line': n.lineno, 'name_expression': value})
                    if signals:
                        row['effect_signals'].append({'line': n.lineno, 'call': expr(n), 'signals': signals})
                row['calls'] = sorted(calls)
                result['definitions'].append(row)
                # Nested fixture factories are retained for installed pytest/asyncio.
                process(node.body, q + '.<locals>', enclosing_class, lexical_conditions)
            else:
                if not scope and isinstance(node, (ast.Assign, ast.AnnAssign, ast.Expr)):
                    s = expr(node)
                    if any(v in s for v in ('pytestmark', '__test__', 'pytest.skip(', 'pytest.importorskip(', 'pytest_plugins')):
                        result['module_declarations'].append({'line': node.lineno, 'source': s,
                                                             'conditions': list(lexical_conditions)})
                if isinstance(node, ast.If):
                    process(node.body, scope, enclosing_class, lexical_conditions + (expr(node.test),))
                    process(node.orelse, scope, enclosing_class, lexical_conditions + ('not (' + expr(node.test) + ')',))
                elif isinstance(node, (ast.With, ast.AsyncWith, ast.For, ast.AsyncFor, ast.While, ast.Try)):
                    for field in ('body', 'orelse', 'finalbody'):
                        process(getattr(node, field, []), scope, enclosing_class, lexical_conditions)
                    for handler in getattr(node, 'handlers', []):
                        process(handler.body, scope, enclosing_class, lexical_conditions)
    process(tree.body)
    return result

def fixture_in_module(path, name, enclosing_class=None, seen=None):
    seen = set() if seen is None else seen
    key = (path, name, enclosing_class)
    if key in seen:
        return []
    seen.add(key)
    f = files[path]
    local = [r for r in fixture_defs.get(name, []) if r['path'] == path and
             ('<locals>' not in r['qualified_name']) and
             ('.' not in r['qualified_name'] or (enclosing_class and r['qualified_name'].startswith(enclosing_class + '.')))]
    if local:
        return local
    for imp in f['imports']:
        if imp['scope'] != '<module>' or name not in imp['aliases']:
            continue
        source = imp.get('source_candidate')
        if source in files:
            refs = fixture_in_module(source, imp['aliases'][name], None, seen)
            if refs:
                return refs
    return [r for r in fixture_defs.get(name, []) if files[r['path']]['role'] == 'pinned_installed_fixture_runtime']
