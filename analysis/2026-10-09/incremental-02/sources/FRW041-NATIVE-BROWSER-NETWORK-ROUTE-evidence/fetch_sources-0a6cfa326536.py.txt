"""Trusted bounded official-source retrieval; never imports candidate modules."""
from pathlib import Path
import hashlib,json,urllib.request,os,resource,time,datetime
os.sched_setaffinity(0,{0,1})
resource.setrlimit(resource.RLIMIT_AS,(2*1024**3,2*1024**3))
E=Path(__file__).parent
D=E/'upstream'
PIN='717d1b09e1c841a4c0206033886a1a861e3ca5d9'
tree=json.loads((D/'tree-v0.26.0.json').read_text())
blobs={x['path']:x for x in tree['tree'] if x['type']=='blob'}
paths=['package.json','LICENSE','cli/Cargo.toml','cli/Cargo.lock','cli/src/native/network.rs','cli/src/native/browser.rs','cli/src/native/daemon.rs','cli/src/native/cdp/chrome.rs','cli/src/native/cdp/client.rs','cli/src/native/cdp/discovery.rs','cli/src/native/actions.rs','cli/src/native/stream/mod.rs','cli/src/native/stream/http.rs','cli/src/native/stream/websocket.rs','cli/src/native/stream/cdp_loop.rs','cli/src/native/stream/discovery.rs','cli/src/connection.rs','cli/src/flags.rs','cli/src/main.rs','docs/src/app/security/page.mdx']
records=[]
for name in paths:
 url=f'https://raw.githubusercontent.com/vercel-labs/agent-browser/{PIN}/{name}'
 try:
  with urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Friday-source-review'}),timeout=12) as r:
   body=r.read(2*1024**2+1)
   assert len(body)<=2*1024**2
   final_url=r.geturl()
  assert final_url==url
  git_blob=hashlib.sha1(b'blob '+str(len(body)).encode()+b'\0'+body).hexdigest()
  assert git_blob==blobs[name]['sha'],name
  p=D/'agent-browser-v0.26.0'/name;p.parent.mkdir(parents=True,exist_ok=True,mode=0o700)
  assert not p.exists();p.write_bytes(body);p.chmod(0o400)
  records.append({'url':url,'path':str(p),'sha256':hashlib.sha256(body).hexdigest(),'git_blob_sha1':git_blob,'bytes':len(body),'retrieved_msk':datetime.datetime.now(datetime.timezone(datetime.timedelta(hours=3))).isoformat(),'status':'RETRIEVED_TREE_BLOB_VERIFIED'})
 except Exception as ex:
  records.append({'url':url,'status':'ERROR_NO_RETRY','error':type(ex).__name__+': '+str(ex)})
  break
 assert time.monotonic()<196593.8475401
p=E/'source-retrieval.json';assert not p.exists();p.write_text(json.dumps({'commit':PIN,'records':records,'candidate_executed':False,'timeout_per_request':12,'maximum_body_bytes':2*1024**2},indent=2)+'\n');p.chmod(0o400)
print(json.dumps({'retrieved':len([r for r in records if r['status']=='RETRIEVED_TREE_BLOB_VERIFIED']),'bytes':sum(r.get('bytes',0) for r in records),'errors':[r for r in records if r['status']!='RETRIEVED_TREE_BLOB_VERIFIED'],'elapsed':time.monotonic()-194793.8475401}))
