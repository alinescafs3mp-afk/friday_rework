"""Backend-only control for the pinned native stream patch, not an admission grant.

Root must wire this into the existing managed launch *after* independently
reviewing the pinned owner dependency and the complete missing native producers.
No network, task manager, daemon or browser is created by this module.
"""
from dataclasses import dataclass, field
import hashlib
import json
import math
import secrets
import time

from tools.browser_tool_owner import BrowserOwner, BrowserOwnerRefused, bind_browser_session


def _binding(owner):
    return hashlib.sha256(json.dumps(owner.identity(), sort_keys=True,
                                    separators=(',', ':')).encode()).hexdigest()


@dataclass(frozen=True)
class StreamControl:
    owner: object = field(repr=False)
    session: str
    binding: str
    original_deadline: float
    token: str = field(repr=False)

    def check(self, owner, session):
        if (type(owner) is not BrowserOwner or owner is not self.owner
                or session != self.session or _binding(owner) != self.binding):
            raise BrowserOwnerRefused('browser_stream_original_owner_changed')
        ceiling = min(self.original_deadline, owner.check())
        if not math.isfinite(ceiling) or time.monotonic() >= ceiling:
            raise BrowserOwnerRefused('browser_stream_original_deadline_exhausted')
        return ceiling

    def launch_environment(self, owner, session, existing_env):
        self.check(owner, session)
        value = {'schema': 1, 'session': self.session, 'binding': self.binding,
                 'token': self.token, 'boot_id': owner.boot_id,
                 'deadline_monotonic': self.original_deadline}
        env = dict(existing_env)
        env['AGENT_BROWSER_FRIDAY_CONTROL_REQUIRED'] = '1'
        env['AGENT_BROWSER_FRIDAY_STREAM_CONTROL'] = json.dumps(value, separators=(',', ':'))
        return env

    def headers(self, owner, session):
        self.check(owner, session)
        # Backend headers only; never URLs, page JS, frontend assets, logs or receipts.
        return {'Authorization': 'Bearer ' + self.token, 'X-Friday-Owner': self.binding}


def stream_control(task_id, session_info):
    """Retain one capability for the already-bound original owner and daemon.

    This does not call require_browser_execution, bypass it, or claim raw CDP,
    page-network or cgroup containment. A new caller cannot renew the record.
    """
    owner = bind_browser_session(task_id, session_info)
    session = session_info.get('session_name')
    if type(session) is not str or not session or len(session) > 512:
        raise BrowserOwnerRefused('browser_stream_actual_daemon_session_required')
    with owner.fence() as ceiling:
        retained = session_info.get('_friday_stream_control')
        if retained is None:
            retained = StreamControl(owner, session, _binding(owner),
                                     min(owner.original_deadline, ceiling), secrets.token_hex(32))
            session_info['_friday_stream_control'] = retained
        if type(retained) is not StreamControl:
            raise BrowserOwnerRefused('browser_stream_original_control_required')
        retained.check(owner, session)
        return retained
