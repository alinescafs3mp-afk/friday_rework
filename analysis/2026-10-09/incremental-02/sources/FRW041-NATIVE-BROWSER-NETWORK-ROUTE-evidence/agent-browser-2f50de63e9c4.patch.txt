+//! This is NOT a Chromium egress or original-job cgroup admission producer.
+use serde::Deserialize;

+#[derive(Clone)]

+    Donor,
+    Friday(OwnerControl),

+#[derive(Clone, Deserialize)]
+#[serde(deny_unknown_fields)]
+pub(super) struct OwnerControl {
+    schema: u8,
+    session: String,
+    binding: String,
+    token: String,
+    boot_id: String,
+    deadline_monotonic: f64,

+fn lower_hex64(value: &str) -> bool {
+    value.len() == 64 && value.bytes().all(|b| b.is_ascii_digit() || (b'a'..=b'f').contains(&b))
+}

+// The issuer is the retained BrowserOwner, never page content or a request body.
+// Its secret is kept only in the launch environment and backend-only headers.
+impl StreamPolicy {
+    pub(super) fn from_env(session: &str) -> Result<Self, String> {
+        let required = std::env::var_os("AGENT_BROWSER_FRIDAY_CONTROL_REQUIRED");
+        let record = std::env::var_os("AGENT_BROWSER_FRIDAY_STREAM_CONTROL");
+        match (required, record) {
+            (None, None) => Ok(Self::Donor),
+            (Some(required), Some(record)) if required == "1" => {
+                let record = record.to_str().ok_or("Friday stream control malformed")?;
+                let boot_id = std::fs::read_to_string("/proc/sys/kernel/random/boot_id")
+                    .map_err(|_| "Friday stream boot identity unavailable")?;
+                Self::parse(record, session, boot_id.trim(), monotonic_seconds()?)
+            }
+            _ => Err("Friday stream control missing or malformed".into()),
+        }
+    }
+
+    fn parse(record: &str, session: &str, boot_id: &str, now: f64) -> Result<Self, String> {
+        if record.len() > 2048 {
+            return Err("Friday stream control malformed".into());
+        }
+        let control: OwnerControl = serde_json::from_str(record)
+            .map_err(|_| "Friday stream control malformed")?;
+        if control.schema != 1 || control.session != session || session.is_empty()
+            || control.boot_id != boot_id || !lower_hex64(&control.binding)
+            || !lower_hex64(&control.token) || !control.deadline_monotonic.is_finite()
+            || !now.is_finite() || control.deadline_monotonic <= now
+            || control.deadline_monotonic - now > 7.0 * 86400.0 {
+            return Err("Friday stream owner identity or deadline mismatch".into());
+        }
+        Ok(Self::Friday(control))
+    }
+
+    pub(super) fn io_timeout(&self) -> Result<Duration, String> {
+        Ok(self.remaining()?.unwrap_or(Duration::from_secs(5)).min(Duration::from_secs(5)))
+    }
+
+    pub(super) fn remaining(&self) -> Result<Option<Duration>, String> {
+        match self {
+            Self::Donor => Ok(None),
+            Self::Friday(control) => {
+                let remaining = control.deadline_monotonic - monotonic_seconds()?;
+                if !remaining.is_finite() || remaining <= 0.0 {
+                    return Err("Friday stream original deadline exhausted".into());
+                }
+                Ok(Some(Duration::from_secs_f64(remaining)))
+            }
+        }
+    }
+
+    pub(super) fn is_friday(&self) -> bool { matches!(self, Self::Friday(_)) }
+
+    pub(super) fn authorize(&self, request: &str) -> bool {
+        if !self.is_friday() { return true; }
+        if self.remaining().is_err() { return false; }
+        self.authorize_fields(request)
+    }
+
+    fn authorize_fields(&self, request: &str) -> bool {
+        let Self::Friday(control) = self else { return true; };
+        if request.len() > 8192 || !request.is_ascii() || !request.contains("\r\n\r\n") {
+            return false;
+        }
+        let mut lines = request.split("\r\n");
+        let Some(first) = lines.next() else { return false; };
+        let parts: Vec<_> = first.split(' ').collect();
+        // The donor-wide discovery/spawn/chat/command relay are not scoped to
+        // this owner. The existing native CLI APIs remain intact.
+        if parts.len() != 3 || parts[0] != "GET" || parts[2] != "HTTP/1.1"
+            || !matches!(parts[1], "/" | "/api/tabs") { return false; }
+        let mut auth = None;
+        let mut binding = None;
+        for line in lines {
+            if line.is_empty() { break; }
+            if line.starts_with(' ') || line.starts_with('\t') { return false; }
+            let Some((name,value)) = line.split_once(':') else { return false; };
+            if name.eq_ignore_ascii_case("origin") { return false; }
+            if name.eq_ignore_ascii_case("authorization") {
+                if auth.is_some() { return false; }
+                auth = Some(value.trim());
+            }
+            if name.eq_ignore_ascii_case("x-friday-owner") {
+                if binding.is_some() { return false; }
+                binding = Some(value.trim());
+            }
+        }
+        let expected = format!("Bearer {}", control.token);
+        constant_match(auth.unwrap_or(""), &expected)
+            && constant_match(binding.unwrap_or(""), &control.binding)
+    }
+}
+
+fn constant_match(actual: &str, expected: &str) -> bool {
+    if actual.len() != expected.len() { return false; }
+    actual.bytes().zip(expected.bytes()).fold(0u8, |diff,(a,b)| diff | (a ^ b)) == 0
+}

+#[cfg(target_os = "linux")]
+fn monotonic_seconds() -> Result<f64, String> {
+    let mut value = libc::timespec { tv_sec: 0, tv_nsec: 0 };
+    if unsafe { libc::clock_gettime(libc::CLOCK_MONOTONIC, &mut value) } != 0 {
+        return Err("Friday stream original clock unavailable".into());
+    }
+    Ok(value.tv_sec as f64 + value.tv_nsec as f64 / 1_000_000_000.0)
+}
+
+#[cfg(not(target_os = "linux"))]
+fn monotonic_seconds() -> Result<f64, String> {
+    Err("Friday stream owner clock unsupported".into())

+pub(super) async fn read_owner_head(stream: &mut TcpStream) -> Result<Vec<u8>, String> {
+    let mut head = Vec::with_capacity(512);
+    while head.len() < 4096 {
+        let byte = stream.read_u8().await.map_err(|_| "Friday stream header incomplete")?;
+        head.push(byte);
+        if head.ends_with(b"\r\n\r\n") { return Ok(head); }
+    }
+    Err("Friday stream header too large".into())
+}
+
+// The full validated header was consumed once for authentication. Give those
+// exact bytes to the existing Tungstenite handshake; body bytes stay in the TCP
+// socket. No second connection, relay task, payload parser or manager is added.
+pub(super) struct StreamIo {
+    socket: TcpStream,
+    head: Vec<u8>,
+    offset: usize,
+}
+
+impl StreamIo {
+    pub(super) fn new(socket: TcpStream, head: Vec<u8>) -> Self {
+        Self { socket, head, offset: 0 }
+    }
+}
+
+impl AsyncRead for StreamIo {
+    fn poll_read(self: Pin<&mut Self>, cx: &mut Context<'_>, out: &mut ReadBuf<'_>) -> Poll<std::io::Result<()>> {
+        let this = self.get_mut();
+        if this.offset < this.head.len() {
+            let count = out.remaining().min(this.head.len() - this.offset);
+            out.put_slice(&this.head[this.offset..this.offset+count]);
+            this.offset += count;
+            return Poll::Ready(Ok(()));
+        }
+        Pin::new(&mut this.socket).poll_read(cx,out)
+    }
+}
+
+impl AsyncWrite for StreamIo {
+    fn poll_write(self: Pin<&mut Self>, cx: &mut Context<'_>, buf: &[u8]) -> Poll<std::io::Result<usize>> {
+        Pin::new(&mut self.get_mut().socket).poll_write(cx,buf)
+    }
+    fn poll_flush(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<std::io::Result<()>> {
+        Pin::new(&mut self.get_mut().socket).poll_flush(cx)
+    }
+    fn poll_shutdown(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<std::io::Result<()>> {
+        Pin::new(&mut self.get_mut().socket).poll_shutdown(cx)
+    }
+}
