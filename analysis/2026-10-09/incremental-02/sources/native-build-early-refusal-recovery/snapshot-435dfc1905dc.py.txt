def require(ok, reason):
    if not ok: raise RuntimeError(reason)

def sha(b): return hashlib.sha256(b).hexdigest()

def metadata(p):
    p=Path(p); s=p.lstat()
    require(p.resolve(strict=True)==p,'symlink:'+str(p))
    return {'path':str(p),'identity':[s.st_dev,s.st_ino],'uid':s.st_uid,'gid':s.st_gid,
            'mode':stat.S_IMODE(s.st_mode),'size':s.st_size,'nlink':s.st_nlink,
            'mtime_ns':s.st_mtime_ns,'ctime_ns':s.st_ctime_ns,'directory':stat.S_ISDIR(s.st_mode)}

def read(p, limit=128000000):

    b=Path(p).read_bytes();require(metadata(p)==a,'read_changed:'+str(p));a['sha256']=sha(b)
    return b,a

def absent(p):

def command(argv, cap=3):
    started=time.monotonic()
    p=subprocess.run(argv,stdin=subprocess.DEVNULL,stdout=subprocess.PIPE,stderr=subprocess.PIPE,
        timeout=cap,close_fds=True,env={'PATH':'/usr/bin:/bin','LANG':'C.UTF-8','HOME':'<LOCAL_HOME>'})
    require(p.returncode==0 and not p.stderr and len(p.stdout)<=1048576,'read_command_failed:'+repr(argv))
    return {'argv':argv,'returncode':p.returncode,'stdout':p.stdout.decode(),'stderr':'',
            'started_monotonic':started,'ended_monotonic':time.monotonic(),'client_reaped':True}
