"""INERT. Defines one reviewed original-parent-only disposition; never runs on import.
No source/grant/clock edits; no native mutation, lock acquisition, unlock, stop or cleanup.
Only after exact checks: write retained-asset custody record, classify original
runtime publication as refused, and leave the existing finally to close its FD.
"""
import hashlib, json, os, stat, sys, time
from pathlib import Path

OUT=Path('<ASTRA_WORK>/native-build-early-refusal-recovery')
SNAPSHOT_SHA='ea67d958eab54e24e96dbbba95df2b37cf71715d6c9b927cc0b25a51de977784'
EVIDENCE_SHA='da0057d31fd238aa05d6828a55f5a93631677a8e921c6f645da250e4be86f9c8'

def need(ok, reason):
    if not ok: raise RuntimeError('RETAIN_STOP_UNCONFIRMED:'+reason)

def digest(b):return hashlib.sha256(b).hexdigest()

def exact_private(path):
    path=Path(path);s=path.lstat()
    need(path.resolve(strict=True)==path and stat.S_ISREG(s.st_mode) and s.st_uid==s.st_gid==1000
         and s.st_nlink==1 and stat.S_IMODE(s.st_mode)==0o400 and s.st_size<1048576,'private_review_boundary')
    b=path.read_bytes();t=path.lstat()
    need((s.st_dev,s.st_ino,s.st_mtime_ns,s.st_ctime_ns,s.st_size)==(t.st_dev,t.st_ino,t.st_mtime_ns,t.st_ctime_ns,t.st_size),'review_changed')
    return b

def release_original_parent(attempt):
    """For Astra only after independent review. Calling this function is execution.
    Any check failure leaves the original custody flags and lock untouched.
    Successful call deliberately escapes InteractiveConsole; existing retain_unknown
    catches it, sees resolved *runtime publication intent*, and returns to original
    finally. The pending original source_changed exception must still exit nonzero.
    """
    need(os.getpid()==1086621 and os.getuid()==os.geteuid()==1000,'original_parent_only')
    need(os.isatty(0) and os.ttyname(0)=='/dev/pts/4','original_terminal_only')
    own_bytes=Path(__file__).read_bytes();own_hash=digest(own_bytes)
    review_bytes=exact_private(OUT/'independent-review.json');review=json.loads(review_bytes)
    need(review=={
        'state':'INDEPENDENT_EARLY_REFUSAL_CUSTODY_RELEASE_APPROVED',
        'reviewer':'Astra','assignment':'FRW041-NATIVE-BUILD-EARLY-REFUSAL-RECOVERY','generation':1,
        'guard_sha256':own_hash,'snapshot_sha256':SNAPSHOT_SHA,'evidence_sha256':EVIDENCE_SHA,
        'parent_pid':1086621,'parent_start_ticks':21990043,'original_budget_seconds':1800,
        'admission_sha256':'7b1f372902897266a10dbe6921be0939b7a1e7e45acc219c7e310431c0ce5856',
        'clock_sha256':'644572bae620f0b3d9bacdc6624d28ddb0be5dcc95b1e7c2539d67f556cdb78f',
        'root_receipt_sha256':'69d2ebc795a78d68b740e80bb1b195d49d8d3d7caa9e16175c8820d427fa3cd5',
        'installed_inactive_asset_custodian':'Astra',
        'accept_receipt_pinned_policy_without_fresh_root_content_read_for_retention_only':True,
        'trial_outcome':'FAILED_SOURCE_ADMISSION_BEFORE_NATIVE_SUBMISSION','cleanup_claimed':False,
        'native_or_retry_authorized':False,'final_parent_reap_required':True},'independent_exact_review_required')
    need(digest((OUT/'evidence.json').read_bytes())==EVIDENCE_SHA,'evidence_changed')
    snapshot_bytes=(OUT/'snapshot.py').read_bytes();need(digest(snapshot_bytes)==SNAPSHOT_SHA,'snapshot_code_changed')
    ns={'__name__':'frw041_read_only_snapshot','__file__':str(OUT/'snapshot.py')}
    exec(compile(snapshot_bytes,str(OUT/'snapshot.py'),'exec'),ns)
    source=ns['SOURCE'];frames=[];f=sys._getframe()
    while f is not None:
        frames.append(f);f=f.f_back
    mains=[f for f in frames if f.f_code.co_filename==str(source/'run-live-unit.py') and f.f_code.co_name=='main']
    retains=[f for f in frames if f.f_code.co_filename==str(source/'run-live-unit.py') and f.f_code.co_name=='retain_unknown']
    need(len(mains)==len(retains)==1,'original_custody_stack_required');main=mains[0];local=main.f_locals
    need(local.get('attempt') is attempt and retains[0].f_locals.get('attempt') is attempt,'exact_original_attempt')

    need(not any(k in local for k in ('admission','clock','result','pending')),'admission_or_execution_completed')
    failure=local.get('failure')
    need(failure=={'state':'STOP_UNCONFIRMED','error':'source_changed','error_type':'RuntimeError','parent_pid':1086621,
        'submitted':False,'signal':None,'records':[],'automatic_retry':False,'native_cgroup_closed':False,
        'cleaned':False,'final_outcome_accepted':False,'host_sentinel':None},'original_failure_record_required')

    need(not any(k in vars(attempt) for k in ('clock_hash','root_control_unknown','recovery_disposition')),'prior_control_or_recovery')
