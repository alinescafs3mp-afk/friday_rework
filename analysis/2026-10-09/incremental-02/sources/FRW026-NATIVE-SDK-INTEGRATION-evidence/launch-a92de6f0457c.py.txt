"""Prepared NOT_RUN. Consume a separately reviewed root offline clock and admission.

Use existing GNU timeout/root process backend, no new process manager or clock.
Root retains the canonical quality lock and independently enforces aggregate
2CPU/2GiB across Node and its sole Vitest worker. This source creates no grant.
"""
import hashlib
import json
import os
from pathlib import Path
import stat
import sys
import time

ROOT = Path(__file__).resolve().parent
assert len(sys.argv) == 2
RUN = Path(sys.argv[1])
assert RUN.is_absolute() and RUN == RUN.resolve() and RUN.parent == ROOT and RUN.name.startswith('run-')
clock = json.loads((RUN / 'clock.json').read_text())
assert clock['scope'] == 'TS24_SOURCE_OFFLINE_ONLY' and clock['original_budget_seconds'] == 180
assert clock['deadline_monotonic'] == clock['started_monotonic'] + 180 and time.monotonic() + 60 < clock['deadline_monotonic']
assert clock['execution_authorized'] is False and clock['boot_id'] == Path('/proc/sys/kernel/random/boot_id').read_text().strip()
admission = json.loads((RUN / 'admission.json').read_text())
# Evidence is selected by the independently reviewed root workflow. JSON alone
# never establishes aggregate containment, native safety or whole-unit cessation.
assert admission['scope'] == 'INDEPENDENT_TS24_OFFLINE_SOURCE_QUALIFICATION_ONLY'
assert admission['runtimeGrant'] is False and admission['root_owns_execution'] is True
assert admission['independent_review_sha256'] != 'UNBOUND' and len(admission['independent_review_sha256']) == 64
assert admission['clock_sha256'] == hashlib.sha256((RUN / 'clock.json').read_bytes()).hexdigest()
assert admission['runner_pins_sha256'] == hashlib.sha256((ROOT / 'runner-pins.json').read_bytes()).hexdigest()
assert admission['support_pins_sha256'] == hashlib.sha256((ROOT / 'support-pins.json').read_bytes()).hexdigest()
assert admission['aggregate_resources_verified'] == {'cpu': 2, 'memory_bytes': 2 << 30}
assert len(os.sched_getaffinity(0)) == 2
# Root passes its already-owned quality-lock FD across the existing launch. No
# lock acquisition, bypass, alternate lock or capacity release occurs here.
lock = int(admission['root_quality_lock_fd'])
st = os.fstat(lock)
assert (st.st_dev, st.st_ino, st.st_uid, st.st_gid, st.st_nlink, st.st_size, stat.S_IMODE(st.st_mode)) == (2050, 9044963, 1000, 1000, 1, 0, 0o664)
assert Path('<LOCAL_HOME>/.jericho/runtime/friday-quality-ui-exclusive.lock').stat().st_ino == st.st_ino
inventory = json.loads((ROOT / 'support-pins.json').read_text())
for p, h in inventory['files'].items():
    with Path(p).open('rb') as f:
        assert hashlib.file_digest(f, 'sha256').hexdigest() == h, p
for p, target in inventory['links'].items():
    assert os.readlink(p) == target, p
for p, h in json.loads((ROOT / 'runner-pins.json').read_text()).items():
    assert hashlib.sha256(Path(p).read_bytes()).hexdigest() == h, p
assert time.monotonic() + 60 < clock['deadline_monotonic']
donor = '<PROJECT>/.donors/dsh'
argv = ['/usr/bin/timeout', '--signal=KILL', '60', '/usr/bin/node',
        '--permission', '--allow-child-process', '--allow-addons',
        '--max-old-space-size=256', '--max-semi-space-size=8',
        '--allow-fs-read=' + donor, '--allow-fs-read=' + str(ROOT),
        '--allow-fs-read=/proc/self', '--allow-fs-read=/proc/sys/kernel/random/boot_id',
        '--allow-fs-write=' + str(RUN), '--import', str(ROOT / 'guard.mjs'), str(ROOT / 'entry.mjs')]
env = {'FRW_TS_RUN': str(RUN), 'LANG': 'C', 'LC_ALL': 'C', 'HOME': str(RUN), 'TMPDIR': str(RUN)}
os.chdir(RUN)
os.execve(argv[0], argv, env)
