checks = []

def ck(name, value):
    assert value, name
    checks.append(name)

def tree(path):
    return ast.parse(path.read_text())

def functions(path):
    return {n.name: n for n in ast.walk(tree(path)) if isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef))}

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

for path in (N / 'caller/producer').glob('test*.py'):
    before, after = functions(path), functions(C / path.relative_to(N))
    old = {k: ast.dump(ast.Module(body=v.decorator_list, type_ignores=[])) for k, v in before.items() if k.startswith('test_')}
    new = {k: ast.dump(ast.Module(body=v.decorator_list, type_ignores=[])) for k, v in after.items() if k.startswith('test_')}
    ck('retained exact test names and parameter decorators ' + path.name, old == new)
