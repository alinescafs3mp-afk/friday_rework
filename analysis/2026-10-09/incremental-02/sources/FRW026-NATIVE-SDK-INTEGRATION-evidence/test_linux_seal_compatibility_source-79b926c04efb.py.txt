"""Authored NOT_RUN: inert ABI negatives; real kernel seals remain mandatory."""
import ast
from pathlib import Path
from types import SimpleNamespace as NS
from unittest.mock import Mock
import pytest
SOURCE = Path(__file__).parent / 'candidate/caller/owner-helper/private_network.py'
def api():
    tree=ast.parse(SOURCE.read_text())
    nodes=[n for n in tree.body if isinstance(n,ast.Assign) and any(isinstance(t,ast.Name) and t.id=='LINUX_SEAL_ABI' for t in n.targets) or isinstance(n,(ast.FunctionDef,ast.ClassDef)) and n.name in ('NetworkRefusal','require','linux_seal_constants')]
    env={'os':NS(uname=lambda:NS(sysname='Linux',machine='x86_64'),MFD_CLOEXEC=1,MFD_ALLOW_SEALING=2,memfd_create=Mock(side_effect=AssertionError('inert only')),pread=Mock(side_effect=AssertionError('inert only'))),'fcntl':NS(fcntl=Mock(side_effect=AssertionError('inert only')))}
    exec(compile(ast.Module(body=nodes,type_ignores=[]),str(SOURCE),'exec'),env)
    return env
@pytest.mark.parametrize('cut',['missing','correct','partial','wrong_add','wrong_get','wrong_write','wrong_grow','wrong_shrink','wrong_seal','bool_add','darwin','foreign_arch','no_memfd','no_pread','no_fcntl','wrong_cloexec','wrong_allow_sealing'])
def test_exact_linux_abi_missing_exports_compatibility_and_conflicts(cut):
    env=api();before=vars(env['fcntl']).copy()
    if cut=='correct':vars(env['fcntl']).update(env['LINUX_SEAL_ABI'])
    if cut=='partial':env['fcntl'].F_SEAL_WRITE=8
    names={'wrong_add':'F_ADD_SEALS','wrong_get':'F_GET_SEALS','wrong_write':'F_SEAL_WRITE','wrong_grow':'F_SEAL_GROW','wrong_shrink':'F_SEAL_SHRINK','wrong_seal':'F_SEAL_SEAL'}
    if cut in names:setattr(env['fcntl'],names[cut],99)
    if cut=='bool_add':env['fcntl'].F_ADD_SEALS=True
    if cut=='darwin':env['os'].uname=lambda:NS(sysname='Darwin',machine='x86_64')
    if cut=='foreign_arch':env['os'].uname=lambda:NS(sysname='Linux',machine='mips')
    if cut in ('no_memfd','no_pread'):setattr(env['os'],'memfd_create' if cut=='no_memfd' else 'pread',None)
    if cut=='no_fcntl':env['fcntl'].fcntl=None
    if cut=='wrong_cloexec':env['os'].MFD_CLOEXEC=0
    if cut=='wrong_allow_sealing':env['os'].MFD_ALLOW_SEALING=0
    before=vars(env['fcntl']).copy()
    if cut in ('missing','correct','partial'):assert env['linux_seal_constants']()==(1033,1034,15)
    else:
        with pytest.raises(env['NetworkRefusal']):env['linux_seal_constants']()
    assert vars(env['fcntl'])==before
    env['os'].memfd_create.assert_not_called() if callable(env['os'].memfd_create) else None
    env['os'].pread.assert_not_called() if callable(env['os'].pread) else None
    env['fcntl'].fcntl.assert_not_called() if callable(env['fcntl'].fcntl) else None
