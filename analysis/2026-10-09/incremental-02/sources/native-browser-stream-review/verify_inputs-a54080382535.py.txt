seen={}
def verify(ref):
    p=pathlib.Path(ref['path']); key=str(p)
    if key not in seen:
        h=hashlib.sha256()
        with p.open('rb') as f:
            for chunk in iter(lambda:f.read(1024*1024),b''):h.update(chunk)
        seen[key]={'path':key,'sha256':h.hexdigest(),'bytes':p.stat().st_size,'mode':oct(stat.S_IMODE(p.stat().st_mode))}
    got=seen[key]
    for k in ('sha256','bytes','mode'):
        if k in ref: assert got[k]==ref[k],(key,k,got[k],ref[k])
    return got

def graph(obj):
    if isinstance(obj,dict):
        if 'path' in obj and 'sha256' in obj:verify(obj)
        for v in obj.values():graph(v)
    elif isinstance(obj,list):
        for v in obj:graph(v)

def reconstruct(patch,base,candidate):
    lines=patch.read_text().splitlines(keepends=True);i=0;files=[]
    while i<len(lines):
        assert lines[i].startswith('--- ')
        old=lines[i][4:].strip();new=lines[i+1][4:].strip();i+=2
        assert new.startswith('b/')
        rel=new[2:];source=[] if old=='/dev/null' else (base/old[2:]).read_text().splitlines(keepends=True)
        out=[];pos=0;hunks=0
        while i<len(lines) and not lines[i].startswith('--- '):
            m=re.fullmatch(r'@@ -(\d+)(?:,(\d+))? \+(\d+)(?:,(\d+))? @@[^\n]*\n',lines[i]);assert m,lines[i]
            old_start=int(m[1]);old_count=int(m[2] or 1);new_start=int(m[3]);new_count=int(m[4] or 1);i+=1
            target=old_start-1 if old_count else old_start
            assert target>=pos
            out.extend(source[pos:target]);pos=target
            assert len(out)==(new_start-1 if new_count else new_start)
            used=added=0
            while i<len(lines) and not lines[i].startswith(('@@ ','--- ')):
                line=lines[i];i+=1;assert line[0] in ' +-'
                if line[0] in ' -': assert pos<len(source) and source[pos]==line[1:],(rel,pos);pos+=1;used+=1
                if line[0] in ' +':out.append(line[1:]);added+=1
            assert (used,added)==(old_count,new_count),(rel,used,added,old_count,new_count)
            hunks+=1
        out.extend(source[pos:]);data=''.join(out).encode()
        assert data==(candidate/rel).read_bytes(),rel
        files.append({'path':rel,'sha256':hashlib.sha256(data).hexdigest(),'hunks':hunks,'base':'empty' if old=='/dev/null' else str(base/old[2:])})
    return files
