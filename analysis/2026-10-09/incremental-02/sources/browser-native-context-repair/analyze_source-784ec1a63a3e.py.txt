"""Read/AST/hash only. Never import or execute any candidate, test or runner."""
import ast
from collections import Counter
import difflib
import hashlib
import json
import os
from pathlib import Path
import resource

os.sched_setaffinity(0, {0, 1})
resource.setrlimit(resource.RLIMIT_AS, (2 << 30, 2 << 30))
O = Path(__file__).resolve().parent
W = O.parent
S = W / 'browser-import-composed-source/source'
G = W / 'browser-close-runtime-join'


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def ref(path):
    return {'path': str(path), 'sha256': sha(path), 'bytes': path.stat().st_size}


def put(name, value):
    (O / name).write_text(json.dumps(value, indent=2, ensure_ascii=False) + '\n')


origins = {
    'project/tests/test_browser_native_host.py': S / 'project/tests/test_browser_native_host.py',
    'native/tools/browser_tool_session.py': S / 'native/tools/browser_tool_session.py',
    'qualification/private_effect_fence.py': G / 'candidate/private_effect_fence.py',
    'qualification/positive-run.py': G / 'native-preparation/positive/run.py',
    'qualification/full-run.py': G / 'candidate/run.py',
}
# This one pin is a dependency join, not a new file/read/FD authority.
fixture_hash = sha(O / 'candidate/project/tests/test_browser_native_host.py')
guard = (origins['qualification/private_effect_fence.py']).read_text()
old_pin = "OWNERSHIP_TEST_SHA='e18a07fbc382eb8632f48140ee9e1378dbefa2647815783b5db1ab2c39ac050c'"
assert guard.count(old_pin) == 1
candidate_guard = guard.replace(old_pin, 'OWNERSHIP_TEST_SHA=' + repr(fixture_hash))
(O / 'candidate/qualification/private_effect_fence.py').write_text(candidate_guard)

trees = {}
pins = []
diffs = []
for relative, origin in origins.items():
    target = O / 'candidate' / relative
    before, after = origin.read_text(), target.read_text()
    trees[relative] = (ast.parse(before), ast.parse(after))
    pins.append({'relative': relative, 'origin': ref(origin), 'candidate': ref(target)})
    diffs.extend(difflib.unified_diff(before.splitlines(keepends=True), after.splitlines(keepends=True),
                                    fromfile='baseline/' + relative, tofile='candidate/' + relative))
(O / 'source.diff').write_text(''.join(diffs))
put('candidate-pins.json', {'schema': 'friday.source-candidate-pins.v1',
    'state': 'SOURCE_ONLY_NOT_COMPOSED_NOT_ADMITTED', 'files': pins,
    'diff': ref(O / 'source.diff'),
    'product_delta': ['project/tests/test_browser_native_host.py', 'native/tools/browser_tool_session.py'],
    'qualification_templates_require_fresh_root_binding': True})


def functions(tree):
    return {n.name: n for n in tree.body if isinstance(n, ast.FunctionDef)}


def nodes(node, kind):
    return Counter(ast.dump(n, include_attributes=False) for n in ast.walk(node) if isinstance(n, kind))


old_tree, new_tree = trees['project/tests/test_browser_native_host.py']
old_tests = {k: v for k, v in functions(old_tree).items() if k.startswith('test_')}
new_tests = {k: v for k, v in functions(new_tree).items() if k.startswith('test_')}
expected = json.loads((G / 'candidate/expected-suite-cases.json').read_text())
assert set(old_tests) == set(new_tests) == set(expected['project'])
matrix = []
for name, before in old_tests.items():
    after = new_tests[name]
    assert ast.dump(before.args) == ast.dump(after.args)
    assert [ast.dump(x) for x in before.decorator_list] == [ast.dump(x) for x in after.decorator_list]
    assert not nodes(before, ast.Assert) - nodes(after, ast.Assert), name
    # Every original raises() expectation is retained, including missing-owner.
    def raises(function):
        return Counter(ast.dump(n) for n in ast.walk(function) if isinstance(n, ast.Call)
                       and isinstance(n.func, ast.Attribute) and n.func.attr == 'raises')
    assert not raises(before) - raises(after), name
    count = 1
    for dec in before.decorator_list:
        if isinstance(dec, ast.Call) and isinstance(dec.func, ast.Attribute) and dec.func.attr == 'parametrize':
            count *= len(ast.literal_eval(dec.args[1]))
    assert count == expected['project'][name]
    bindings = [ast.unparse(n) for n in ast.walk(after) if isinstance(n, ast.Call)
                and isinstance(n.func, ast.Name) and n.func.id == 'native_browser_call']
    matrix.append({'test': name, 'unchanged_case_count': count,
                   'original_assertions': sum(nodes(before, ast.Assert).values()),
                   'candidate_assertions': sum(nodes(after, ast.Assert).values()),
                   'all_original_raises_and_assertions_preserved': True,
                   'native_binding_calls': bindings,
                   'browser_fixture': any(a.arg == 'browser_users' for a in after.args.args)})
assert sum(row['unchanged_case_count'] for row in matrix) == 38
assert sum(sum(counts.values()) for counts in expected.values()) == 725
put('interaction-matrix.json', {'schema': 'friday.browser.fixture-interactions.v1',
    'evidence_kind': 'STATIC_AST_NOT_PYTEST_COLLECTION', 'cases': 38, 'function_groups': matrix})

# Product body change is confined to normalization of an already checked clock.
a, b = trees['native/tools/browser_tool_session.py']
fa, fb = functions(a), functions(b)
assert set(fa) == set(fb)
changed_product_functions = [k for k in fa if ast.dump(fa[k]) != ast.dump(fb[k])]
assert changed_product_functions == ['_get_session_info']
assert nodes(fa['_get_session_info'], ast.Assert) == nodes(fb['_get_session_info'], ast.Assert)
for suffix in ('positive-run.py', 'full-run.py'):
    a, b = trees['qualification/' + suffix]
    # Full postconditions, deadlines, negative triggers and all acceptance asserts survive.
    assert nodes(a, ast.Assert) == nodes(b, ast.Assert)
    before = functions(a)
    after = functions(b)
    allowed = {'save', 'snapshot', 'directory_profile'}
    assert set(before) == set(after)
    assert all(ast.dump(before[k]) == ast.dump(after[k]) for k in before if k not in allowed)
    if suffix == 'positive-run.py':
        for name in ('copy_metadata_profile', 'directory_profile_event', 'copy_fault_profile',
                     'directory_fault_profile', 'fault_profile', 'current_copy_snapshot'):
            assert ast.dump(before[name]) == ast.dump(after[name]), name
assert candidate_guard.replace('OWNERSHIP_TEST_SHA=' + repr(fixture_hash), old_pin) == guard

evidence = W / 'BROWSER-G18-PREP-POSITIVE-RESULT.json'
assert sha(evidence) == '6841996bb4c1cfb6437071f15c99d668c5d2f5e25f5eddc6998a4349e5fd5ddc'
assert sha(G / 'result.json') == '79c055b23a869cbfcc0d8efa30cdf98fa63a9ea7f7526d256e513223503481e1'
assert sha(W / 'browser-g18-native-review/result.json') == '08049681c38384da0477cd3d82764fb2fe1c790340456792fac2f456da195971'
progress = G / 'native-preparation/positive/project/pytest-progress.json'
raw = progress.read_bytes()
observed = json.loads(raw)

    'state': 'SOURCE_STRUCTURAL_CHECKS_SATISFIED_RUNTIME_NOT_RUN',

    'all_original_assertions_raises_decorators_and_signatures_retained': True,

    'guard_delta_is_only_exact_fixture_source_sha': True,

    'all_runner_assertions_identical': True,
    'historical_observation_not_causal_attribution': {
        'source': ref(progress), 'bytes': len(raw),
        'same_object_compact_json_bytes': len(json.dumps(observed, separators=(',', ':')).encode()),
        'nodes': len(observed['nodes']), 'reports': len(observed['reports']),
        'current_node': observed['current_node'], 'denial_record_present': 'denied' in observed,
        'histories': {k: len(v) for k, v in observed.items() if 'history' in k and type(v) is list},
        'resource_usage_fields_in_old_progress': [], 'signal137_cause': 'UNKNOWN'},

                  'new_runtime_clocks': 0, 'new_runtime_admissions': 0},
})

deps = [S / rel for rel in (
    'native/agent/web_call_context.py', 'native/agent/tool_executor.py',
    'native/tools/browser_tool_owner.py', 'native/tools/browser_tool.py',

put('input-pins.json', {'origins': [ref(p) for p in origins.values()],
    'dependencies': [ref(p) for p in deps], 'authoritative_evidence': [ref(evidence), ref(G / 'result.json'),
        ref(W / 'browser-g18-native-review/result.json'), ref(G / 'native-preparation/plan.json'),
        ref(G / 'root-runtime-contract.json'), ref(G / 'candidate/expected-suite-cases.json')]})
print(json.dumps({'source_files': len(pins), 'cases': 38, 'full_expected': 725,
                  'result': str(O / 'static-analysis.json')}))
