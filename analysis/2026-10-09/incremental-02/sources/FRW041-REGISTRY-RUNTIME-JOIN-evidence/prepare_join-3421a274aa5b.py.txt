def digest(p):
    with Path(p).open('rb') as f:
        return hashlib.file_digest(f, 'sha256').hexdigest()


def ref(p):
    return {'path': str(p), 'sha256': digest(p), 'bytes': Path(p).stat().st_size}


def read(p):
    return json.loads(Path(p).read_bytes())


def write(p, b):
    remaining()
    assert p.is_relative_to(E)
    p.parent.mkdir(parents=True, exist_ok=True, mode=0o700)
    assert not p.is_symlink()
    if p.exists():
        assert stat.S_ISREG(p.lstat().st_mode) and p.stat().st_nlink == 1
        p.chmod(0o600)
    fd = os.open(p, os.O_WRONLY | os.O_CREAT | os.O_TRUNC | os.O_NOFOLLOW, 0o600)
    with os.fdopen(fd, 'wb') as f:
        f.write(b)
        f.flush()
        os.fsync(f.fileno())


def save(p, v):
    write(p, (json.dumps(v, ensure_ascii=False, indent=2) + '\n').encode())

def rebind(v):
    if isinstance(v, str):
        if v == str(B) or v.startswith(str(B) + '/'):
            return str(C) + v[len(str(B)):]
        if v == str(OLD_SOURCE) or v.startswith(str(OLD_SOURCE) + '/'):
            return str(SOURCE) + v[len(str(OLD_SOURCE)):]
        return v
    if isinstance(v, list):
        return [rebind(x) for x in v]
    if isinstance(v, dict):
        return {rebind(k): rebind(x) for k, x in v.items()}
    return v
