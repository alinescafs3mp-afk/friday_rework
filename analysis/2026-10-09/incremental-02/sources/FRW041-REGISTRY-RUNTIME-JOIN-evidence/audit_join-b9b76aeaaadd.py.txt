cache = {}


def digest(p):
    p = Path(p)
    assert time.monotonic() < clock['deadline_monotonic'] - 60
    key = str(p)
    if key not in cache:
        with p.open('rb') as stream:
            cache[key] = hashlib.file_digest(stream, 'sha256').hexdigest()
    return cache[key]


def read(p):
    return json.loads(Path(p).read_bytes())


def reconstruct(blob, root):
    lines = blob.splitlines(True)
    output = {}
    pos = 0
    while pos < len(lines):
        assert lines[pos].startswith(b'diff --git ')
        end = pos + 1
        while end < len(lines) and not lines[end].startswith(b'diff --git '):
            end += 1
        block = lines[pos:end]
        h = next(i for i, line in enumerate(block) if line.startswith(b'--- '))
        old = block[h][4:].strip().decode()
        new = block[h + 1][4:].strip().decode()
        assert new.startswith('b/')
        name = new[2:]
        assert name not in output and '..' not in Path(name).parts
        assert old in ('a/' + name, '/dev/null')
        p = root / name
        if old == '/dev/null':
            assert not p.exists()
            before = b''
        else:
            before = p.read_bytes()
        original = before.splitlines(True)
        produced = []
        cursor = 0
        i = h + 2
        while i < len(block):
            match = re.fullmatch(rb'@@ -(\d+)(?:,(\d+))? \+(\d+)(?:,(\d+))? @@[^\n]*\n', block[i])
            assert match, block[i]
            start, count, new_start, new_count = [int(match[k] or b'1') for k in range(1, 5)]
            offset = start - 1 if count else start
            assert cursor <= offset <= len(original)
            produced.extend(original[cursor:offset])
            cursor = offset
            assert len(produced) == (new_start - 1 if new_count else new_start)
            consumed = added = 0
            i += 1
            while consumed < count or added < new_count:
                sign, data = block[i][:1], block[i][1:]
                assert sign in (b' ', b'-', b'+')
                if sign in (b' ', b'-'):
                    assert original[cursor] == data, (name, cursor)
                    cursor += 1
                    consumed += 1
                if sign in (b' ', b'+'):
                    produced.append(data)
                    added += 1
                i += 1
            assert consumed == count and added == new_count
        produced.extend(original[cursor:])
        output[name] = b''.join(produced)
        pos = end
    return output

# Every current reference in the active graph is exact. Historical refs remain

refs_checked = []
def verify_refs(v):
    if isinstance(v, dict):
        if isinstance(v.get('path'), str) and re.fullmatch(r'[0-9a-f]{64}', str(v.get('sha256', ''))):
            p = Path(v['path'])
            assert digest(p) == v['sha256'], p
            if 'bytes' in v:
                assert p.stat().st_size == v['bytes'], p
            refs_checked.append(str(p))
        for x in v.values():
            verify_refs(x)
    elif isinstance(v, list):
        for x in v:
            verify_refs(x)
