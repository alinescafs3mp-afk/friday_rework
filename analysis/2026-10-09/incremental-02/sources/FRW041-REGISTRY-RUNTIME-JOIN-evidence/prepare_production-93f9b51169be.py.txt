"""Trusted source/data/AST preparation only; never imports candidate modules."""
import ast
import difflib
import hashlib
import json
import os
from pathlib import Path
import resource
import time

E = Path(__file__).resolve().parent
W = Path('<ASTRA_WORK>')
R = Path('<PROJECT>/.worktrees/sol-registry-runtime')
B = W / 'registry-lightweight-runtime-repair/candidate'
N = W / 'browser-owner-controls-preparation/source/native'
os.umask(0o077)
os.sched_setaffinity(0, {0, 1})
resource.setrlimit(resource.RLIMIT_AS, (2 << 30, 2 << 30))
clock = json.loads((E / 'clock.json').read_bytes())


def sha(b):
    return hashlib.sha256(b).hexdigest()


def remaining():
    assert time.monotonic() < clock['deadline_monotonic']


def write(p, b, mode=0o600):
    remaining()
    p.parent.mkdir(parents=True, exist_ok=True, mode=0o700)
    fd = os.open(p, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, mode)
    with os.fdopen(fd, 'wb') as f:
        f.write(b)
        f.flush()
        os.fsync(f.fileno())


manifest_path = W / 'registry-lightweight-runtime-repair/REVIEW-MANIFEST.json'
assert sha(manifest_path.read_bytes()) == 'c26229e1f04faa00df9b8938405bc3f2e6ea9244fd9e7ee5f77451698b39199f'
for row in json.loads(manifest_path.read_bytes())['refs']:
    assert sha(Path(row['path']).read_bytes()) == row['sha256'], row['path']
review_path = W / 'registry-lightweight-runtime-review/result.json'
assert sha(review_path.read_bytes()) == '3a4925cce81fa78f7d15712b5fec15f15d73a2c7073725005adc07ba0df39dc2'
paths = ['model_tools.py', 'tools/registry.py', 'tools/dispatch_helpers.py',
         'tests/tools/test_registry_lightweight_dispatch.py']
changed = {p: (B / p).read_text() for p in paths}
t = changed[paths[-1]]
fixes = [
    ('"```json\\n<tool_call>untrusted call</tool_call>\\n```"',
     '"\\n```json\\n<tool_call>untrusted call</tool_call>\\n```"'),
    ('assert registration(lambda args: {"value": args["value"]}).dispatch(\n'
     '        "dispatch_regression", {"value": 7}) == {"value": 7}',
     'result = registration(lambda args: json.dumps({"value": args["value"]})).dispatch(\n'
     '        "dispatch_regression", {"value": 7})\n'
     '    assert json.loads(result) == {"value": 7}'),
    ('return {"value": args["value"]}', 'return json.dumps({"value": args["value"]})'),
    ('assert registry.dispatch("dispatch_regression", {"value": 1}) == {"value": 1}',
     'assert json.loads(registry.dispatch("dispatch_regression", {"value": 1})) == {"value": 1}'),
    ('assert registry.dispatch("dispatch_regression", {"value": 2}) == {"value": 2}',
     'assert json.loads(registry.dispatch("dispatch_regression", {"value": 2})) == {"value": 2}'),
]
for old, new in fixes:

assert sha(original['tools/registry.py'].encode()) == '3c57d0994eaf68bd7d9cbfc4d6d315a15be8737a671b7b982a78d6d17d09451e'
assert not original[paths[2]] and not original[paths[3]]


def functions(text):
    return {n.name: ast.dump(n, include_attributes=False)
            for n in ast.parse(text).body if isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef))}


before = functions(original['model_tools.py'])
after = functions(changed['model_tools.py'])
shared = functions(changed['tools/dispatch_helpers.py'])
moved = ['_get_tool_loop', '_get_worker_loop', '_run_async', '_sanitize_tool_error']
assert set(before) - set(after) == set(moved)
assert all(before[n] == shared[n] for n in moved)

assert all(changed[p] == (B / p).read_text() for p in paths[:-1])
patch = ''
rows = []
for p in paths:
    compile(ast.parse(changed[p], filename=p), p, 'exec', dont_inherit=True)
    prior = original[p]
    patch += 'diff --git a/' + p + ' b/' + p + '\n'
    if not prior:

    patch += ''.join(difflib.unified_diff(prior.splitlines(True), changed[p].splitlines(True),

    rows.append({'path': p, 'before_sha256': sha(prior.encode()) if prior else None,
                 'sha256': sha(changed[p].encode()), 'bytes': len(changed[p].encode()),
                 'mode': '100644', 'new_file': not bool(prior)})
    write(E / 'production' / p, changed[p].encode())
