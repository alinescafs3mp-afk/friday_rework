def ref(p):

    assert p==p.resolve(strict=True) and stat.S_ISREG(s.st_mode) and s.st_uid==1000 and s.st_nlink==1

def put(p,b):

    fd=os.open(p,os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o600)

    p.chmod(0o400)
