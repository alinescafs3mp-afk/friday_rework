def sha(b):return hashlib.sha256(b).hexdigest()
def js(p):return json.loads(p.read_bytes())
def ref(p):
    b=p.read_bytes();s=p.lstat()
    assert p==p.resolve(strict=True) and stat.S_ISREG(s.st_mode) and s.st_uid==1000 and s.st_nlink==1
    assert stat.S_IMODE(s.st_mode)==0o400
    for parent in p.parents:
        q=parent.lstat();assert stat.S_ISDIR(q.st_mode) and q.st_uid in (0,1000) and not q.st_mode&0o022
    return dict(path=str(p),sha256=sha(b),bytes=len(b),mode=stat.S_IMODE(s.st_mode),uid=s.st_uid,nlink=s.st_nlink)
def parse(p):
    tree=ast.parse(p.read_bytes(),filename=str(p));compile(tree,str(p),'exec');return tree

def cases(n):
    count=1

    return count
