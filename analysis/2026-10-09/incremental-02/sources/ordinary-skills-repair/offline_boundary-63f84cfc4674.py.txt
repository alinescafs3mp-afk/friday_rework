import os, sys, pathlib
R=pathlib.Path(__file__).parent
child_launches=0
attempts=[]
def own(p,fd=None):
 if isinstance(p,int):return os.path.realpath('/proc/self/fd/'+str(p))
 p=os.fsdecode(p)
 if not os.path.isabs(p):p=os.path.join(os.readlink('/proc/self/fd/'+str(fd)) if fd not in (None,-1) else os.getcwd(),p)
 return os.path.realpath(p)
def allowed(p):return p==str(R) or p.startswith(str(R)+'/')
def deny(reason):attempts.append(reason);raise PermissionError(reason)
def audit(event,args):
 if event=='subprocess.Popen':
  global child_launches
  argv=args[1]
  if (os.environ.get('FRIDAY_CONTROL_CHILD')=='1' or child_launches or not isinstance(argv,list)
      or argv[:2]!=[sys.executable,str(R/'admin_child.py')] or len(argv)!=4):
   deny('UNAPPROVED_CHILD')
  child_launches+=1
 if event.startswith('socket.') or event in ('os.system','os.posix_spawn','os.fork','os.forkpty','os.exec'):
  deny('OFFLINE_EFFECT:'+event)
 if event=='open':
  p=args[0]
  if isinstance(p,(str,bytes)):
   resolved=own(p);flags=args[2] or 0
   # Relative descriptor opens are checked by os.open wrapper below.
   if os.path.isabs(os.fsdecode(p)):
    if flags&(os.O_WRONLY|os.O_RDWR|os.O_CREAT|os.O_TRUNC) and not allowed(resolved):deny('OUTSIDE_WRITE')
    if pathlib.Path(resolved).name in ('.env','auth.json','credentials.json') and not allowed(resolved):deny('REAL_CREDENTIAL_READ')
 if event in ('os.mkdir','os.remove','os.rmdir','os.chmod','os.chown','os.utime'):
  fd=args[-1] if isinstance(args[-1],int) else None
  if not allowed(own(args[0],fd)):deny('OUTSIDE_MUTATION:'+event)
 if event in ('os.rename','os.link'):
  for p,fd in ((args[0],args[2]),(args[1],args[3])):
   if not allowed(own(p,fd)):deny('OUTSIDE_MUTATION:'+event)
 if event=='os.symlink':
  if not allowed(own(args[1],args[2])):deny('OUTSIDE_MUTATION:'+event)
 if event=='ctypes.dlopen' and args[0] is not None:deny('NATIVE_LIBRARY_LOAD')
sys.addaudithook(audit)
orig_open=os.open
def guarded_open(p,flags,mode=0o777,*,dir_fd=None):
 resolved=own(p,dir_fd)
 if flags&(os.O_WRONLY|os.O_RDWR|os.O_CREAT|os.O_TRUNC) and not allowed(resolved):deny('OUTSIDE_FD_WRITE')
 if pathlib.Path(resolved).name in ('.env','auth.json','credentials.json') and not allowed(resolved):deny('REAL_KEY_FD_READ')
 return orig_open(p,flags,mode,dir_fd=dir_fd)
os.open=guarded_open
# ctypes bypasses Python audit for the libc call. Confine the only admitted
# symbol before any product import; every syscall must have own directory FDs.
import ctypes
orig_cdll=ctypes.CDLL
class Library:
 def __init__(self,*args,**kw):
  if args and args[0] is not None:deny('NATIVE_LIBRARY')
  self.lib=orig_cdll(*args,**kw)
 def __getattr__(self,name):
  if name!='renameat2':deny('NATIVE_SYMBOL:'+name)
  fn=getattr(self.lib,name)
  class Rename:
   def __call__(self,a,b,c,d,e):
    if not allowed(own(b,a)) or not allowed(own(d,c)):deny('OUTSIDE_NATIVE_RENAME')
    return fn(a,b,c,d,e)
   def __setattr__(self,n,v):setattr(fn,n,v)
  return Rename()
ctypes.CDLL=Library
