def digest(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()

def ref(p):
    p = Path(p)
    return {'path': str(p), 'sha256': digest(p), 'bytes': p.stat().st_size}

def read(p):
    return json.loads(Path(p).read_text())

def write(p, value):
    p = Path(p)
    assert p.is_relative_to(O)
    p.write_text(json.dumps(value, indent=2, ensure_ascii=False) + '\n')

def historical(key):
    return key.startswith(('baseline', 'prior_', 'old_', 'historical_', 'generation7_'))

def remap_path(text):
    for root in (P, R):
        if text == str(root):
            return str(C)
        if text.startswith(str(root) + '/'):
            relative = Path(text).relative_to(root)
            if root == R and relative.name not in replacements:
                return text
            return str(C / relative)
    return text

def remap(value):
    if isinstance(value, dict):
        return {remap_path(k): (v if historical(k) else remap(v)) for k, v in value.items()}
    if isinstance(value, list):
        return [remap(v) for v in value]
    return remap_path(value) if isinstance(value, str) else value

def refresh(value):
    if isinstance(value, dict):
        result = {k: (v if historical(k) else refresh(v)) for k, v in value.items()}
        if isinstance(result.get('path'), str) and result['path'].startswith(str(C) + '/'):
            p = Path(result['path'])
            if 'sha256' in result:
                result['sha256'] = digest(p)
            if 'bytes' in result:
                result['bytes'] = p.stat().st_size
            if 'entries' in result:
                result['entries'] = len(read(p))
        for k in result:
            if k.startswith(str(C) + '/') and isinstance(result[k], str) and len(result[k]) == 64:
                result[k] = digest(k)
        return result
    if isinstance(value, list):
        return [refresh(v) for v in value]
    return value
