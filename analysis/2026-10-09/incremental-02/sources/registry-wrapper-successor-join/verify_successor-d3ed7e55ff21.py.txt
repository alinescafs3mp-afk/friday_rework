def read(p): return json.loads(Path(p).read_text())
def sha(p): return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def ref(p):
    p = Path(p)
    return {'path': str(p), 'sha256': sha(p), 'bytes': p.stat().st_size}
def write(p, data):
    assert Path(p).is_relative_to(O)
    Path(p).write_text(json.dumps(data, indent=2, ensure_ascii=False)+'\n')

def verify_ref(row):
    assert sha(row['path']) == row['sha256'], ('REF_SHA', row['path'])
    if 'bytes' in row: assert Path(row['path']).stat().st_size == row['bytes'], row['path']
    if 'entries' in row: assert len(read(row['path'])) == row['entries'], row['path']

def source_counts(path):
    output = {}
    for node in ast.parse(path.read_text()).body:
        if not isinstance(node, ast.FunctionDef) or not node.name.startswith('test_'): continue
        count = 1
        for dec in node.decorator_list:
            if isinstance(dec, ast.Call) and isinstance(dec.func, ast.Attribute) and dec.func.attr == 'parametrize':
                seq = dec.args[1]
                if isinstance(seq, (ast.List, ast.Tuple)): size = len(seq.elts)
                elif isinstance(seq, ast.Name) and seq.id == 'NODES': size = 2
                elif isinstance(seq, ast.ListComp):
                    size = 1
                    for gen in seq.generators:
                        assert isinstance(gen.iter, (ast.List, ast.Tuple)) and not gen.ifs
                        size *= len(gen.iter.elts)
                else: raise AssertionError(ast.dump(seq))
                count *= size
        output[node.name] = count
    return output

def apply_text_patch(before, patch):
    lines = before.splitlines(True); diff = patch.splitlines(True)
    assert diff[0].startswith('--- ') and diff[1].startswith('+++ ')
    output = []; cursor = 0; i = 2; hunks = 0
    while i < len(diff):
        match = re.fullmatch(r'@@ -(\d+)(?:,(\d+))? \+(\d+)(?:,(\d+))? @@[^\n]*\n', diff[i])
        assert match, diff[i]
        oldstart, oldn, newstart, newn = int(match[1]), int(match[2] or '1'), int(match[3]), int(match[4] or '1')
        start = oldstart-1 if oldn else oldstart
        assert start >= cursor
        output.extend(lines[cursor:start]); cursor = start; i += 1; oc = nc = 0
        assert len(output) == (newstart-1 if newn else newstart)
        while i < len(diff) and not diff[i].startswith('@@ '):
            line = diff[i]; assert line[0] in ' +-'; body = line[1:]
            if line[0] in ' -': assert lines[cursor] == body; cursor += 1; oc += 1
            if line[0] in ' +': output.append(body); nc += 1
            i += 1
        assert (oc, nc) == (oldn, newn)
        hunks += 1
    output.extend(lines[cursor:])
    return ''.join(output), hunks

active_refs = []; historical_refs = []; stale = []
def walk(value, location='', historical=False):
    if isinstance(value, dict):
        if isinstance(value.get('path'), str) and 'sha256' in value:
            # Local nested inputs are the changed graph; old refs are provenance and separately pinned.
            if value['path'].startswith(str(C)+'/') or value['path'].startswith(str(P)+'/'):
                verify_ref(value)
                (historical_refs if historical else active_refs).append({'at': location, **value})
        for key, child in value.items():
            old = historical or key.startswith(('baseline','prior_','old_','historical_','generation7_'))
            if isinstance(key,str) and key.startswith(str(P)+'/') and not old:
                stale.append({'at': location, 'key': key})
            walk(child, location+'.'+key, old)
    elif isinstance(value, list):
        for i, child in enumerate(value): walk(child, location+'.'+str(i), historical)
    elif isinstance(value, str) and not historical:
        if value == str(P) or value.startswith(str(P)+'/') or value.startswith(str(R)+'/'):
            stale.append({'at': location, 'value': value})
