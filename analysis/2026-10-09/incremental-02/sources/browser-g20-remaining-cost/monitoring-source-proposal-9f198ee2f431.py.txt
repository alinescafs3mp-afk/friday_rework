_COST = {}
_COST_CODES = {}
_COST_STACK = []
_COST_ERRORS = []
_COST_ACTIVE = False
_COST_TOOL = None
_COST_CPU = time.process_time_ns
_COST_WALL = time.perf_counter_ns
_COST_GET_IDENT = fence._thread.get_ident
_COST_OWNER_THREAD = None


def _cost_bad(kind):
    if len(_COST_ERRORS) < 8:
        _COST_ERRORS.append(kind)


def _cost_start(code, offset):
    key = id(code)
    row = _COST.get(key)
    if row is None or _COST_CODES[key] is not code:
        return
    if _COST_GET_IDENT() != _COST_OWNER_THREAD:
        _cost_bad('FOREIGN_TARGET_THREAD')
        return
    if _COST_ERRORS:
        return
    if len(_COST_STACK) >= 64:
        _cost_bad('STACK_BOUND')
        return
    cpu = _COST_CPU()
    wall = _COST_WALL()
    row['starts'] += 1
    _COST_STACK.append([code, cpu, wall, 0, 0])


def _cost_end(code, unwound):
    key = id(code)
    row = _COST.get(key)
    if row is None or _COST_CODES[key] is not code:
        return
    if _COST_GET_IDENT() != _COST_OWNER_THREAD:
        _cost_bad('FOREIGN_TARGET_THREAD')
        return
    if _COST_ERRORS:
        return
    cpu = _COST_CPU()
    wall = _COST_WALL()
    if not _COST_STACK or _COST_STACK[-1][0] is not code:
        _cost_bad('UNBALANCED_TARGET_EVENT')
        return
    _, began_cpu, began_wall, child_cpu, child_wall = _COST_STACK.pop()
    elapsed_cpu = cpu - began_cpu
    elapsed_wall = wall - began_wall
    if elapsed_cpu < child_cpu or elapsed_wall < child_wall:
        _cost_bad('NEGATIVE_EXCLUSIVE_INTERVAL')
        return
    row['unwinds' if unwound else 'returns'] += 1
    row['inclusive_cpu_ns'] += elapsed_cpu
    row['inclusive_wall_ns'] += elapsed_wall
    row['exclusive_target_cpu_ns'] += elapsed_cpu - child_cpu
    row['exclusive_target_wall_ns'] += elapsed_wall - child_wall
    if _COST_STACK:
        _COST_STACK[-1][3] += elapsed_cpu
        _COST_STACK[-1][4] += elapsed_wall


def _cost_return(code, offset, value):
    # Neither retain nor inspect the product's return value.
    _cost_end(code, False)


def _cost_unwind(code, offset, exception):
    # PY_UNWIND is global-only; ignore every unselected code in O(1).
    # Neither retain, stringify nor replace the original exception.
    _cost_end(code, True)


def cost_install():
    global _COST_ACTIVE, _COST_TOOL, _COST_OWNER_THREAD
    assert not _COST_ACTIVE and not _COST and not _COST_STACK
    assert sys.implementation.name == 'cpython' and sys.version_info[:3] == (3, 14, 7)
    assert fence.threading.current_thread() is fence.threading.main_thread()
    _COST_OWNER_THREAD = _COST_GET_IDENT()
    assert sys.getprofile() is None  # Original setup installs directory_profile later.
    module_targets = [
        (fence, str(O / 'private_effect_fence.py'), (
            'audit', 'canonical', 'path_ok', 'read_ok', 'check_import_open',
            'ownership_context', 'ownership_chain', 'ownership_chain_same',
            'guarded_ownership_chown', 'copy_begin', 'copy_native', 'copy_finish',
            'directory_sync_context', 'directory_sync_check',
            'directory_sync_open', 'directory_sync_fsync', 'directory_sync_close',
            'directory_sync_cessation')),
        (sys.modules.get('hermes_yaml'), str(SOURCE / 'native/hermes_yaml.py'),
            ('_load', 'safe_dump')),
        (sys.modules.get('hermes_cli.friday_user_scope'),
            str(SOURCE / 'native/hermes_cli/friday_user_scope.py'),
            ('policy', 'UserScope.check', 'check_onboarding_home')),
        (sys.modules.get('ruamel.yaml.main'),
            str(site / 'ruamel/yaml/main.py'), ('YAML.load', 'YAML.dump')),
    ]
    selected = [('runner.' + f.__name__, f, globals(), str(O / 'run.py'))
                for f in (save, snapshot, origins, sha)]
    for module, expected, names in module_targets:
        assert module is not None and module.__file__ == expected
        for name in names:
            obj = module
            for part in name.split('.'):
                obj = getattr(obj, part)
            selected.append((module.__name__ + '.' + name, obj, module.__dict__, expected))
    assert len(selected) == 29
    for label, function, namespace, expected in selected:
        code = function.__code__
        assert function.__globals__ is namespace and code.co_filename == expected
        assert not (code.co_flags & (0x20 | 0x80 | 0x200))  # No generators/coroutines.
        key = id(code)
        assert key not in _COST
        _COST_CODES[key] = code  # Strong reference; identity cannot be recycled.
        _COST[key] = dict(label=label, starts=0, returns=0, unwinds=0,
            inclusive_cpu_ns=0, inclusive_wall_ns=0,
            exclusive_target_cpu_ns=0, exclusive_target_wall_ns=0)
    mon = sys.monitoring
    tool = mon.PROFILER_ID
    assert mon.get_tool(tool) is None  # Never steal, clear or reuse another tool.
    mon.use_tool_id(tool, 'FRW041-bounded-cost-observation')
    _COST_TOOL = tool
    # Partial install failure must end the diagnostic with normal owned cessation.
    # It must not be converted into uninstrumented acceptance or a retry.
    mon.register_callback(tool, mon.events.PY_START, _cost_start)
    mon.register_callback(tool, mon.events.PY_RETURN, _cost_return)
    mon.register_callback(tool, mon.events.PY_UNWIND, _cost_unwind)
    mon.set_events(tool, mon.events.PY_UNWIND)
    for code in _COST_CODES.values():
        mon.set_local_events(tool, code, mon.events.PY_START | mon.events.PY_RETURN)
    _COST_ACTIVE = True
    assert sys.getprofile() is None


def cost_publish(label):
    # Called before existing report save and after existing collection snapshot.
    # No new file, no per-event logging, no authority fields consulted by guard.
    assert _COST_ACTIVE
    if _COST_GET_IDENT() != _COST_OWNER_THREAD:
        _cost_bad('FOREIGN_PUBLICATION_THREAD')
    expected_profile = directory_profile if state.get('directory_profile_started') else None
    assert sys.getprofile() is expected_profile
    state['cost_observation'] = {
        'label': label, 'instrumented_only': True,
        'cpu_ns': _COST_CPU(), 'wall_ns': _COST_WALL(),
        'errors': list(_COST_ERRORS),
        'open_targets': [_COST[id(item[0])]['label'] for item in _COST_STACK],
        'rows': [dict(row) for row in _COST.values()],
        'sys_profile_identity_preserved': True,
        'owner_thread_ident': _COST_OWNER_THREAD,
    }


def cost_close():
    global _COST_ACTIVE
    # Invoke only AFTER original final snapshot/effect checks/profile cessation.
    mon = sys.monitoring
    assert mon.get_tool(_COST_TOOL) == 'FRW041-bounded-cost-observation'
    if _COST_STACK:
        _cost_bad('OPEN_TARGET_AT_CLOSURE')
    # Invalid timing must not obstruct releasing our own instrumentation.
    # Retain the unmatched stack as evidence; never invent matching returns.
    mon.clear_tool_id(_COST_TOOL)
    mon.free_tool_id(_COST_TOOL)
    _COST_ACTIVE = False
    assert mon.get_tool(_COST_TOOL) is None
