def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def ref(path):
    return {'path': str(path), 'sha256': sha(path)}

def check(reference):
    path = Path(reference['path'])
    assert path.is_absolute() and path.resolve(strict=True) == path
    assert stat.S_ISREG(path.lstat().st_mode) and path.stat().st_nlink == 1
    assert sha(path) == reference['sha256'], str(path)
    return path

def put(path, value):
    data = value if isinstance(value, bytes) else (json.dumps(value, indent=2) + '\n').encode()
    assert path.is_relative_to(OUT) and not path.exists() and not path.is_symlink()
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o400)
    with os.fdopen(fd, 'wb') as stream:
        stream.write(data)
        stream.flush()
        os.fsync(stream.fileno())
    return ref(path)

def replace_source(source, name, replacements, python=False):
    original = source.read_bytes()
    text = original.decode()
    for before, after in replacements:
        assert before != after and text.count(before) == 1, (source, before)
        text = text.replace(before, after, 1)
    restored = text
    for before, after in reversed(replacements):
        assert restored.count(after) == 1
        restored = restored.replace(after, before, 1)
    assert restored.encode() == original
    entry = {'before': ref(source), 'after': put(DEST / name, text.encode()),
             'exact_replacements': [{'before': a, 'after': b} for a, b in replacements],
             'inverse_substitution_byte_equal': True}
    if python:
        old_tree = ast.parse(original, filename=str(source))
        new_tree = ast.parse(text, filename=str(DEST / name))
        mapping = {after: before for before, after in replacements}
        class Normalize(ast.NodeTransformer):
            def visit_Constant(self, node):
                if isinstance(node.value, str) and node.value in mapping:
                    node.value = mapping[node.value]
                return node
        normalized = Normalize().visit(new_tree)
        before_ast = ast.dump(old_tree, include_attributes=False)
        after_ast = ast.dump(normalized, include_attributes=False)
        assert before_ast == after_ast
        entry.update(ast_equal_except_exact_constants=True,
                     normalized_ast_sha256=hashlib.sha256(before_ast.encode()).hexdigest())
    else:
        entry['javascript_note'] = 'Exact whole-source inverse byte equality is stronger than AST equality for these literal substitutions; no JS parser or wrapper executed.'
    changes.append(entry)
    published[name] = entry['after']
    diffs.extend(difflib.unified_diff(original.decode().splitlines(True), text.splitlines(True),
                                    fromfile=str(source), tofile=str(DEST / name)))
    return entry['after']
