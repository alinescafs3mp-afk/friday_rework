"""Owner data inspection only; no collection, observer, execution grant or launch.

Publish a checked project classification document and print its receipt. The
caller must verify exit 0 and the exact receipt before reaching the observer.
A closed attempt is never repaired by this utility. Native/unknown metadata
needs owner assessment.
"""
import ast
import copy
import contextlib
import hashlib
import io
import json
import os
from pathlib import Path
import stat
import sys

MAP = Path('<ASTRA_WORK>/collection-lexical-observer-integration/classification/provenance.json')
MAP_SHA = '01bfe073139d6a6aa0c0065986a81ef126ca519b3892aa5c0d5b5d71439c7c73'
RECONCILE = MAP.with_name('reconcile.py')
RECONCILE_SHA = '4cd275cdaa0bdb2a92b16c5357445171b3cdddb59da9a24f93ea63145dc08e5d'


def require(ok, message):
    if not ok:
        raise ValueError(message)


def digest(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def read_json(data):
    def unique(pairs):
        result = {}
        for key, value in pairs:
            require(key not in result, 'DUPLICATE_JSON_KEY')
            result[key] = value
        return result
    return json.loads(data, object_pairs_hook=unique)


def canonical_basis(basis, input_paths):
    """Rejoin actual JSON with the pinned, previously reviewed owner joiner.

    This is owner data code, not a candidate import. Its exact source pin is
    checked before execution and its CLI branch stays disabled. Full equality
    binds metadata, receipts, identity and every auxiliary basis field; it is
    still not execution/receipt-graph acceptance by the owner observer.
    """
    require(len(input_paths) == 4, 'EXACT_ACTUAL_INPUTS_REQUIRED')
    require(read_json(Path(input_paths[0]).read_bytes()).get('selection') == '@project',
            'PROJECT_ONLY_INSPECTION')
    data = RECONCILE.read_bytes()
    require(hashlib.sha256(data).hexdigest() == RECONCILE_SHA, 'RECONCILE_SOURCE_DRIFT')
    namespace = {'__name__': 'pinned_owner_data_join', '__file__': str(RECONCILE)}
    exec(compile(data, str(RECONCILE), 'exec'), namespace)
    with contextlib.redirect_stdout(io.StringIO()) as output:
        exit_code = namespace['main']([str(p) for p in input_paths])
    require(exit_code == 0, 'ACTUAL_SOURCE_JOIN_UNRESOLVED')
    actual_basis = read_json(output.getvalue())
    require(basis == actual_basis, 'BASIS_NOT_BOUND_TO_ACTUAL_SOURCE_JOIN')
    return actual_basis


def parameters(node):
    """Parse names and indirect modes only; never evaluate parameter values."""
    direct, indirect, seen = set(), set(), set()
    for decorator in node.decorator_list:
        if not isinstance(decorator, ast.Call) or ast.unparse(decorator.func) != 'pytest.mark.parametrize':
            continue
        require(2 <= len(decorator.args) <= 6, 'UNSUPPORTED_PARAMETER_DECLARATION')
        options = {k.arg: k.value for k in decorator.keywords}
        require(None not in options and len(options) == len(decorator.keywords),
                'DYNAMIC_PARAMETER_OPTIONS')
        names = ast.literal_eval(decorator.args[0])
        names = [s.strip() for s in names.split(',')] if isinstance(names, str) else names
        require(isinstance(names, (list, tuple)) and bool(names)
                and all(isinstance(n, str) and n.isidentifier() for n in names)
                and len(set(names)) == len(names) and not seen.intersection(names),
                'PARAMETER_NAMES_MISMATCH')
        seen.update(names)
        require(not (len(decorator.args) > 2 and 'indirect' in options), 'DUPLICATE_INDIRECT')
        mode = ast.literal_eval(decorator.args[2]) if len(decorator.args) > 2 else (
            ast.literal_eval(options['indirect']) if 'indirect' in options else False)
        if type(mode) is bool:
            selected = set(names) if mode else set()
        else:
            require(isinstance(mode, (list, tuple)) and all(isinstance(n, str) for n in mode)
                    and len(set(mode)) == len(mode) and set(mode).issubset(names),
                    'INVALID_INDIRECT_PARAMETERS')
            selected = set(mode)
        indirect.update(selected)
        direct.update(set(names) - selected)
    return sorted(direct), sorted(indirect)


def inspect_project(basis, input_paths):
    require(digest(MAP) == MAP_SHA, 'PROVENANCE_DRIFT')
    provenance = read_json(MAP.read_bytes())
    allowed = {}

    def gather(value):
        if isinstance(value, dict):
            if 'path' in value and 'sha256' in value:
                prior = allowed.setdefault(value['path'], value['sha256'])
                require(prior == value['sha256'], 'CONFLICTING_SOURCE_PIN')
            for item in value.values():
                gather(item)
        elif isinstance(value, list):
            for item in value:
                gather(item)

    gather(provenance)
    require(basis.get('state') == 'DATA_JOIN_RESOLVED_REQUIRES_OWNER_INSPECTION'
            and basis.get('unresolved') == [] and basis.get('provenance_sha256') == MAP_SHA,
            'UNRESOLVED_OR_FOREIGN_BASIS')
    require(basis.get('all_runtests') == 'NOT_RUN' and basis.get('acceptance') is False,
            'RUNTIME_ACCEPTANCE_NOT_ALLOWED')
    result = copy.deepcopy(canonical_basis(basis, input_paths))
    cache = {}
    definitions = {}
    checks = 0

    def source(ref):
        nonlocal checks
        path = ref['path']
        require(allowed.get(path) == ref['sha256'], 'UNPINNED_SOURCE_REF')
        if path not in cache:
            p = Path(path)
            require(p.is_absolute() and not p.is_symlink(), 'FOREIGN_SOURCE_PATH')
            data = p.read_bytes()
            require(hashlib.sha256(data).hexdigest() == ref['sha256'], 'SOURCE_DRIFT')
            tree = ast.parse(data) if p.suffix == '.py' else None
            cache[path] = tree
            # Bind the full lexical class/function path, not only the leaf name.
            index = {}
            def walk(node, parents=()):
                current = parents
                if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
                    current = parents + (node.name,)
                    key = ('.'.join(current), node.lineno, node.end_lineno)
                    require(key not in index, 'AMBIGUOUS_AST_DEFINITION')
                    index[key] = node
                for child in ast.iter_child_nodes(node):
                    walk(child, current)
            if tree is not None:
                walk(tree)
            definitions[path] = index
        checks += 1
        if 'line' in ref:
            key = (ref['qualified_name'], ref['line'], ref['end_line'])
            require(key in definitions[path], 'DEFINITION_IDENTITY_MISMATCH')
            return definitions[path][key]

    def intrinsic(rule, expected_kind, name_key):
        source(rule['origin'])
        matches = [node for (name, line, _), node in definitions[rule['origin']['path']].items()
                   if name == rule[name_key] and line == rule['line']]
        require(len(matches) == 1 and isinstance(matches[0], expected_kind),
                'INTRINSIC_SOURCE_IDENTITY_MISMATCH')

    by_path = {f['path']: f for f in provenance['files']}
    by_id = {d['id']: (f, d) for f in provenance['files'] for d in f['definitions']}

    require(len(result['selected_modules']) == 17 and len(result['kept']) == 627
            and result['deselected'] == [] and result['deselected_reviews'] == [],
            'NOT_REVIEWED_PROJECT_SCOPE')
    require([m['path'] for m in result['selected_modules']] == provenance['selection']['project'],
            'FOREIGN_MODULE_SELECTION')
    for module in result['selected_modules']:
        source(module['source_basis'])
        require(module['source_basis']['path'] == module['path']
                and module['receipt']['identity']['path'] == module['path']
                and module['receipt']['outcome'] == 'passed'
                and module['disposition'] == 'COLLECTED_ONLY_NOT_RUN'
                and module['module_declarations'] == [], 'MODULE_NEEDS_OWNER_ASSESSMENT')
        module['source_and_skip_inspected'] = True
    for row in result['kept']:
        node = source(row['definition_basis'])
        metadata = row['metadata']
        # Retained-row schema intentionally has no metadata.originalname.
        require(node.name == row['originalname_source_basis']
                and metadata['definition'] == row['definition_basis']['path'], 'WRONG_DEFINITION')
        require(isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)), 'NOT_TEST_FUNCTION')
        # Rejoin binds actual parent receipts/classes and nodeid to provenance;
        # the same complete lexical identity must also exist in the pinned AST.
        class_names = []
        for ref in row['class_lineage_basis']:
            cls = source(ref)
            require(isinstance(cls, ast.ClassDef), 'NOT_PARENT_CLASS')
            class_names.append(cls.name)
            require(ref['qualified_name'] == '.'.join(class_names), 'CLASS_LINEAGE_DRIFT')
        require(row['definition_basis']['qualified_name'] == '.'.join(class_names + [node.name]),
                'QUALIFIED_TEST_IDENTITY_MISMATCH')
        parents = [p for p in metadata['parents'] if p['kind'] in ('module', 'class')]
        require(len(parents) == len(class_names) + 1 and parents[0]['kind'] == 'module'
                and all(p['path'] == metadata['definition'] for p in parents), 'PARENT_CLASS_MISMATCH')
        prefix = parents[-1]['nodeid'] + '::' + node.name
        direct, indirect = parameters(node)
        require(row['direct_parameters'] == direct and row['indirect_parameters'] == indirect,
                'PARAMETER_DECORATOR_MISMATCH')
        require(metadata['nodeid'].startswith(prefix), 'ACTUAL_NODEID_DEFINITION_MISMATCH')
        tail = metadata['nodeid'][len(prefix):]
        require((not direct and not indirect and tail == '') or
                ((direct or indirect) and tail.startswith('[') and tail.endswith(']') and len(tail) >= 2),
                'ACTUAL_PARAMETER_SUFFIX_MISMATCH')
        require([ast.unparse(d) for d in node.decorator_list] == row['decorators_source_basis'],
                'DECORATOR_DRIFT')
        require(metadata['skip_reasons'] == [] and metadata['platforms'] == []
                and row['disposition'] == 'COLLECTED_ONLY_NOT_RUN', 'SKIP_NEEDS_OWNER_ASSESSMENT')
        require(set(metadata['fixtures']) == set(row['fixture_origins']) == set(row['fixture_basis']),
                'FIXTURE_COVERAGE_MISMATCH')
        file = by_path[metadata['definition']]
        lookup_id = file['fixture_candidates_by_test'][row['definition_basis']['qualified_name']]
        lookup = provenance['fixture_candidate_sets'][lookup_id]
        for name, refs in row['fixture_origins'].items():
            mode = row['fixture_basis'][name]
            require(bool(refs) and mode in ('exact_source_fixture_lookup_candidate',
                    'direct_parameter_from_exact_decorator', 'intrinsic_request'), 'UNKNOWN_FIXTURE')
            for ref in refs:
                source(ref)
            if mode == 'exact_source_fixture_lookup_candidate':
                ids = lookup.get(name, [])
                require(len(ids) == 1 and len(refs) == 1 and refs[0].get('definition_id') == ids[0],
                        'FIXTURE_LOOKUP_MISMATCH')
                fixture_file, declaration = by_id[ids[0]]
                require(declaration['kind'] == 'fixture' and declaration['fixture']['name'] == name
                        and refs[0]['path'] == fixture_file['path'], 'FIXTURE_DECLARATION_MISMATCH')
                fixture_node = source(refs[0])
                require([ast.unparse(d) for d in fixture_node.decorator_list] == declaration.get('decorators', []),
                        'FIXTURE_DECORATOR_DRIFT')
                names = []
                for decorator in fixture_node.decorator_list:
                    call = decorator.func if isinstance(decorator, ast.Call) else decorator
                    if ast.unparse(call) in ('fixture', 'pytest.fixture'):
                        options = {k.arg: k.value for k in decorator.keywords} if isinstance(decorator, ast.Call) else {}
                        require(None not in options, 'DYNAMIC_FIXTURE_DECLARATION')
                        names.append(ast.literal_eval(options['name']) if 'name' in options else fixture_node.name)
                require(names == [name], 'FIXTURE_NAME_MISMATCH')
            if mode == 'direct_parameter_from_exact_decorator':
                require(name in row['direct_parameters'] and name not in row['indirect_parameters'],
                        'PARAMETER_MISMATCH')
                intrinsic(provenance['dynamic_rules']['direct_parameter'], ast.FunctionDef, 'function')
            if mode == 'intrinsic_request':
                require(name == 'request', 'WRONG_INTRINSIC')
                intrinsic(provenance['dynamic_rules']['intrinsic_request'], ast.ClassDef, 'qualified_name')
        row['definition_inspected'] = True
        row['platform_skip_inspected'] = True
    result.update(state='INDEPENDENT_COLLECTION_METADATA_RECONCILED',
                  owner='Astra', acceptance=False,
                  owner_inspection={'definitions': 627, 'modules': 17,
                                    'source_files': len(cache), 'reference_checks': checks,
                                    'method': 'Full basis rejoined from actual JSON and pinned provenance; qualified AST definitions, parameter modes and exact fixture/intrinsic declarations inspected; no execution acceptance.'})
    return result


def publish(basis_path):
    """One owner data publication; cannot launch or invoke the observer."""
    basis_path = Path(basis_path)
    phase = basis_path.parent
    require(basis_path.name == 'classification-source-basis.json'
            and phase.name == 'collect-project' and phase.resolve(strict=True) == phase,
            'EXACT_PROJECT_PHASE_REQUIRED')
    require(not (phase / 'cessation.json').exists()
            and not (phase / 'classification-review.json').exists(), 'CLOSED_OR_REVIEWED_PHASE')
    info = phase.stat()
    require(info.st_uid == os.getuid() and stat.S_IMODE(info.st_mode) == 0o700,
            'PRIVATE_OWNER_PHASE_REQUIRED')
    require(not basis_path.is_symlink() and basis_path.is_file(), 'REGULAR_BASIS_REQUIRED')
    basis = read_json(basis_path.read_bytes())
    expected = [phase / 'run' / ('collect-project-' + suffix + '.json')
                for suffix in ('results', 'classified', 'deselected', 'collection')]
    require([ref['path'] for ref in basis['input_refs']] == [str(p) for p in expected],
            'FOREIGN_INPUT_REFS')
    for path, ref in zip(expected, basis['input_refs']):
        require(not path.is_symlink() and path.is_file() and digest(path) == ref['sha256'],
                'ACTUAL_INPUT_DRIFT')
    checked = inspect_project(basis, expected)
    # Rejoin reads the actual inputs again; do not publish after input drift.
    for path, ref in zip(expected, basis['input_refs']):
        require(not path.is_symlink() and digest(path) == ref['sha256'], 'ACTUAL_INPUT_DRIFT')
    require(not (phase / 'cessation.json').exists(), 'CLOSED_OR_REVIEWED_PHASE')
    payload = (json.dumps(checked, indent=2) + '\n').encode()
    # No published review is opened until every inspection above has succeeded.
    target = phase / 'classification-review.json'
    fd = os.open(target, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o400)
    with os.fdopen(fd, 'wb') as stream:
        stream.write(payload)
        stream.flush()
        os.fsync(stream.fileno())
    return {'state': checked['state'], 'path': str(target),
            'sha256': hashlib.sha256(payload).hexdigest(),
            'owner_inspection': checked['owner_inspection'], 'runtime_acceptance': False}


if __name__ == '__main__':
    try:
        require(len(sys.argv) == 2, 'EXACT_BASIS_ARGUMENT_REQUIRED')
        checked = publish(sys.argv[1])
    except (ValueError, KeyError, TypeError, OSError, SyntaxError) as error:
        print(json.dumps({'state': 'OWNER_INSPECTION_REFUSED', 'reason': str(error),
                          'acceptance': False}), file=sys.stderr)
        raise SystemExit(2)
    print(json.dumps(checked, indent=2))
