const root = path.dirname(fileURLToPath(import.meta.url))
const donor = '<PROJECT>/.donors/dsh'
const source = path.join(root, 'source')
const run = process.env.FRW_TS_RUN
// Capture before Vitest imports. Even the original JS process.exit delegates to
// mutable process.reallyExit and exit listeners. Use the captured native entry
// for a refusal; its actual pinned-Node behavior still requires root review.
const raw = { append: fs.appendFileSync.bind(fs), read: fs.readFileSync.bind(fs), fork: child.fork,
  originalExit: process.exit.bind(process), stop: process.reallyExit.bind(process),
  dlopen: process.dlopen.bind(process) }
const check = (ok, reason) => { if (!ok) throw new Error(reason) }
check(run && path.isAbsolute(run) && fs.realpathSync(run) === run &&
  path.basename(run).startsWith('ts24-run-') && !run.startsWith(path.dirname(root) + '/') &&
  !root.startsWith(run + '/') && fs.statSync(run).uid === process.getuid() &&
  (fs.statSync(run).mode & 0o777) === 0o700, 'TS24_SEPARATE_OWN_RUN_REQUIRED')
const clock = JSON.parse(raw.read(path.join(run, 'clock.json')))
check(clock.original_budget_seconds === 180 && clock.execution_authorized === false &&
  clock.scope === 'TS24_SOURCE_OFFLINE_ONLY' && clock.boot_id === raw.read('/proc/sys/kernel/random/boot_id', 'utf8').trim(), 'TS24_EXACT_SOURCE_CLOCK_REQUIRED')
const now = () => Number(process.hrtime.bigint()) / 1e9
const current = () => { if (!(Number.isFinite(clock.deadline_monotonic) &&
  clock.deadline_monotonic === clock.started_monotonic + 180 && now() < clock.deadline_monotonic)) refuse('TS24_ORIGINAL_DEADLINE') }
const workerEntry = fs.realpathSync(path.join(donor, 'node_modules/vitest/dist/workers/forks.js'))
const role = process.argv[1] === path.join(root, 'entry.mjs') ? 'parent' :
  process.argv[1] === workerEntry ? 'worker' : 'unknown'
check(role !== 'unknown' && process.execPath === '/usr/bin/node' &&
  process.permission?.has('fs.write', run) && !process.permission.has('fs.write', source) &&
  !process.permission.has('fs.write', donor), 'TS24_NODE_PERMISSION_BOUNDARY_REQUIRED')
const emit = row => raw.append(path.join(run, `guard-${process.pid}.jsonl`), JSON.stringify({
  pid: process.pid, ppid: process.ppid, role, at: now(), ...row,
}) + '\n', { mode: 0o600 })
function refuse(operation) {
  // Even a failed evidence write must reach the captured original exit.
  try { emit({ event: 'DENIED', operation }) } finally { raw.stop(125) }
}
current()
emit({ event: 'START', birth: raw.read('/proc/self/stat', 'utf8'), execArgv: process.execArgv })
const inventory = JSON.parse(raw.read(path.join(root, 'support-pins.json')))
const pins = { ...inventory.files, ...JSON.parse(raw.read(path.join(root, 'runner-pins.json'))) }
const native = JSON.parse(raw.read(path.join(root, 'native-runtime-pins.json')))
Object.assign(pins, native.files)
// The preload is module-cached in the entry; this is a small existing-framework
// evidence seam, not a supervisor or an authority token.
export const guardEvidence = Object.freeze({
  refuse, pinned, emit,
  rawHash: value => crypto.createHash('sha256').update(value).digest('hex'),
})
function pinned(filename) {
  current()
  try {
    const canonical = fs.realpathSync(filename)
    const expected = pins[canonical]
    if (!expected || crypto.createHash('sha256').update(raw.read(canonical)).digest('hex') !== expected) refuse('UNPINNED_MODULE:' + canonical)
    emit({ event: 'NODE_MODULE_OR_PIN_CHECK', path: canonical, sha256: expected })
    return canonical
  } catch (error) { refuse('PIN_READ_ERROR:' + filename) }
}
function mappedObjects(phase) {
  const maps = raw.read('/proc/self/maps', 'utf8')
  const objects = [...new Set(maps.split('\n').map(line => line.match(/^\S+\s+\S+\s+\S+\s+\S+\s+\S+\s+(\/.*)$/)?.[1]).filter(Boolean))]
  for (const object of objects) {
    if (object.endsWith(' (deleted)')) refuse('DELETED_MAPPED_OBJECT')
    pinned(object)
  }
  emit({ event: 'MAPPED_OBJECTS_SNAPSHOT', phase, maps, objects })
}
process.dlopen = function (module, filename, flags) {
  const canonical = pinned(filename)
  if (!native.addons.includes(canonical)) refuse('FOREIGN_NATIVE_ADDON')
  emit({ event: 'ADDON_LOAD_REQUEST', path: canonical })
  const result = raw.dlopen(module, filename, flags)
  mappedObjects('after-addon:' + canonical)
  return result
}
mappedObjects('pre-framework')
registerHooks({
  resolve(specifier, context, next) {
    const result = next(specifier, context)
    if (!result.url.startsWith('node:')) {
      const url = new URL(result.url)
      if (url.protocol !== 'file:' || url.search || url.hash) refuse('NON_FILE_MODULE')
      pinned(fileURLToPath(url))
    }
    return result
  },
  load(url, context, next) {
    if (url.startsWith('node:')) return next(url, context)
    const filename = pinned(fileURLToPath(url))
    const result = next(url, context)
    // Vite8 may use its exact pinned native compiler: mapped libs still UNBOUND.
    // This is limited trusted source qualification, never generic native safety.
    if (filename.endsWith('.node')) return result
    if (!['module', 'module-typescript', 'commonjs', 'commonjs-typescript', 'json'].includes(result.format)) refuse('MODULE_FORMAT')
    return { ...result, source: raw.read(filename) }
  },
})
const denied = name => (...args) => refuse(name)
globalThis.fetch = denied('fetch')
for (const [object, names] of [
  [child, ['spawn', 'spawnSync', 'exec', 'execSync', 'execFile', 'execFileSync']],
  [net, ['connect', 'createConnection']], [tls, ['connect']],
  [http, ['request', 'get']], [https, ['request', 'get']],
  [dns, ['lookup', 'resolve', 'resolve4', 'resolve6']],
  [dns.promises, ['lookup', 'resolve', 'resolve4', 'resolve6']],
  [dgram, ['createSocket']], [worker, ['Worker']],
]) for (const name of names) object[name] = denied(name)
net.Socket.prototype.connect = denied('Socket.connect')
net.Server.prototype.listen = denied('Server.listen')
let active = null
let starts = 0
const argsAllowed = new Set([
  '--experimental-import-meta-resolve', '--require', fs.realpathSync(path.join(donor, 'node_modules/vitest/suppress-warnings.cjs')),
  '--no-webstorage', '--permission', '--allow-child-process', '--allow-addons', '--conditions', 'node', 'development',
  '--max-old-space-size=256', '--max-semi-space-size=8', '--import', path.join(root, 'guard.mjs'),
  `--allow-fs-read=${donor}`, `--allow-fs-read=${root}`, `--allow-fs-read=${run}`, '--allow-fs-read=/proc/self',
  '--allow-fs-read=/proc/sys/kernel/random/boot_id', `--allow-fs-write=${run}`,
  ...native.readRoots.map(p => `--allow-fs-read=${p}`),
])
child.fork = function (filename, args, options) {
  current()
  if (role !== 'parent' || active !== null || ++starts > 2 ||
      fs.realpathSync(filename) !== workerEntry || args.length !== 0 ||
      options.stdio !== 'pipe' || options.serialization !== 'advanced' || options.detached ||
      (options.execPath && options.execPath !== process.execPath) ||
      !Array.isArray(options.execArgv) || options.execArgv.some(arg => !argsAllowed.has(arg)) ||
      options.execArgv.filter(arg => arg === path.join(root, 'guard.mjs')).length !== 1 ||
      !options.execArgv.includes('--permission') || !options.execArgv.includes(`--allow-fs-write=${run}`) ||
      options.env.FRW_TS_RUN !== run || Object.hasOwn(options.env, 'NODE_OPTIONS') ||
      Object.keys(options.env).some(key => ![
        'FRW_TS_RUN', 'LANG', 'LC_ALL', 'HOME', 'TMPDIR', 'TEST', 'VITEST', 'NODE_ENV',
        'VITEST_MODE', 'FORCE_TTY', 'VITEST_WORKER_ID', 'VITEST_POOL_ID',
      ].includes(key))) refuse('FOREIGN_FORK')
  pinned(workerEntry)
  const handle = raw.fork.call(child, filename, args, options)
  active = handle
  emit({ event: 'FORK', child_pid: handle.pid, ordinal: starts })
  handle.once('exit', (code, signal) => { emit({ event: 'CHILD_EXIT', child_pid: handle.pid, code, signal }); active = null })
  return handle
}
syncBuiltinESMExports()
process.once('exit', code => emit({ event: 'EXIT', code, child_still_active: active?.pid ?? null, observed_forks: starts }))
