"""NOT_RUN. Existing root backend must supply a reviewed offline execution unit.
No unit/lock/process manager/namespace/grant is created. JSON selects expected
identities; actual kernel files/FDs/caps are checked. A working existing producer
of that boundary is UNBOUND. This preparation is not executable admission.
"""
import hashlib
import json
import os
from pathlib import Path
import stat
import sys
import time

ROOT = Path(__file__).resolve().parent
assert len(sys.argv) == 2
RUN = Path(sys.argv[1])
assert RUN.is_absolute() and RUN == RUN.resolve() and RUN.name.startswith('ts24-run-')
assert not RUN.is_relative_to(ROOT.parent) and not ROOT.is_relative_to(RUN)
rs = RUN.stat()
assert rs.st_uid == os.getuid() and stat.S_IMODE(rs.st_mode) == 0o700
clock = json.loads((RUN / 'clock.json').read_text())
assert clock['scope'] == 'TS24_SOURCE_OFFLINE_ONLY' and clock['original_budget_seconds'] == 180
assert clock['deadline_monotonic'] == clock['started_monotonic'] + 180
assert time.monotonic() + 60 < clock['deadline_monotonic']
assert clock['execution_authorized'] is False
assert clock['boot_id'] == Path('/proc/sys/kernel/random/boot_id').read_text().strip()
admission = json.loads((RUN / 'admission.json').read_text())
assert admission['scope'] == 'INDEPENDENT_TS24_OFFLINE_SOURCE_QUALIFICATION_ONLY'
assert admission['runtimeGrant'] is False
assert admission['clock_sha256'] == hashlib.sha256((RUN / 'clock.json').read_bytes()).hexdigest()
assert admission['independent_review_sha256'] != 'UNBOUND' and len(admission['independent_review_sha256']) == 64
for name in ('runner-pins', 'support-pins', 'native-runtime-pins'):
    assert admission[name.replace('-', '_') + '_sha256'] == hashlib.sha256((ROOT / (name + '.json')).read_bytes()).hexdigest()

# Root's retained backend birth is an OS observation, not a JSON ownership claim.
backend = int(admission['existing_root_backend_pid'])
bs = Path(f'/proc/{backend}/stat').read_text()
birth = int(bs[bs.rfind(')') + 2:].split()[19])
assert backend != os.getpid() and birth == admission['existing_root_backend_birth']
assert os.getsid(0) == os.getpid() and os.getpgrp() == os.getpid()
assert os.getpgrp() == admission['root_owned_pgid']

# Verify the already-held advisory lock in the kernel. Never acquire/unlock it.
lock = int(admission['root_quality_lock_fd'])
st = os.fstat(lock)
identity = (st.st_dev, st.st_ino, st.st_uid, st.st_gid, st.st_nlink, st.st_size, stat.S_IMODE(st.st_mode))
assert identity == (2050, 9044963, 1000, 1000, 1, 0, 0o664)
assert Path('<LOCAL_HOME>/.jericho/runtime/friday-quality-ui-exclusive.lock').stat().st_ino == st.st_ino
fdinfo = Path(f'/proc/self/fdinfo/{lock}').read_text()
assert any('FLOCK  ADVISORY  WRITE' in line and f'{st.st_ino} ' in line for line in fdinfo.splitlines())
assert any('FLOCK  ADVISORY  WRITE' in line and f' {backend} ' in line and f'{st.st_ino} ' in line for line in Path('/proc/locks').read_text().splitlines())

# Actual non-root cgroup limits the aggregate including native and worker threads.
# Affinity/heaps/RLIMIT_AS and a JSON flag do not establish aggregate enforcement.
relative = Path('/proc/self/cgroup').read_text().strip()
assert relative.startswith('0::/') and '\n' not in relative
cg = Path('/sys/fs/cgroup') / relative[4:]
assert cg != Path('/sys/fs/cgroup') and str(cg) == admission['existing_execution_cgroup']
assert cg.resolve() == cg and cg.stat().st_ino == admission['existing_execution_cgroup_inode']
assert os.statvfs(cg).f_flag & os.ST_RDONLY
quota, period = (cg / 'cpu.max').read_text().split()
assert quota != 'max' and 0 < int(quota) <= 2 * int(period)
memory = (cg / 'memory.max').read_text().strip()
assert memory != 'max' and 0 < int(memory) <= (2 << 30)
assert (cg / 'memory.oom.group').read_text().strip() == '1'
assert 0 < len(os.sched_getaffinity(0)) <= 2
assert set(map(int, (cg / 'cgroup.procs').read_text().split())) == {os.getpid()}
assert not any(p.is_dir() for p in cg.iterdir())

# Node permissions do not contain native code. Require a pre-existing offline
# boundary and read-only mounts. Distinct namespaces alone do not prove policy;
# root must independently qualify actual routes and namespace/mount contents.
namespaces = {}
for name in ('net', 'mnt'):
    own = Path(f'/proc/self/ns/{name}').stat().st_ino
    outside = Path(f'/proc/{backend}/ns/{name}').stat().st_ino
    assert own != outside and own == admission['existing_namespace_inodes'][name]
    namespaces[name] = own
donor = Path('<PROJECT>/.donors/dsh')
assert os.statvfs(ROOT).f_flag & os.ST_RDONLY
assert os.statvfs(donor).f_flag & os.ST_RDONLY
assert not os.statvfs(RUN).f_flag & os.ST_RDONLY
native = json.loads((ROOT / 'native-runtime-pins.json').read_text())
assert native['ld_preload_present'] is False and not Path('/etc/ld.so.preload').exists()
assert all(os.statvfs(Path(p)).f_flag & os.ST_RDONLY for p in native['files'])
inventory = json.loads((ROOT / 'support-pins.json').read_text())
for mapping in (inventory['files'], native['files'], json.loads((ROOT / 'runner-pins.json').read_text())):
    for p, h in mapping.items():
        with Path(p).open('rb') as f:
            assert hashlib.file_digest(f, 'sha256').hexdigest() == h, p
for p, target in inventory['links'].items():
    assert os.readlink(p) == target, p
for p, target in native['links'].items():
    assert str(Path(p).resolve()) == target, p
assert time.monotonic() + 60 < clock['deadline_monotonic']
observed = {'state': 'PRE_EXEC_OS_SNAPSHOT_NOT_CLOSURE', 'pid': os.getpid(), 'pgid': os.getpgrp(),
    'backend_pid': backend, 'backend_birth': birth, 'cgroup': str(cg), 'quota': [int(quota), int(period)],
    'memory_max': int(memory), 'affinity': sorted(os.sched_getaffinity(0)), 'namespaces': namespaces,
    'lock_fdinfo': fdinfo, 'deadline': clock['deadline_monotonic'], 'runtimeGrant': False}
(RUN / 'pre-exec-os.json').write_text(json.dumps(observed, indent=2) + '\n')
argv = ['/usr/bin/gnutimeout', '--signal=KILL', '60', '/usr/bin/node',
    '--permission', '--allow-child-process', '--allow-addons',
    '--max-old-space-size=256', '--max-semi-space-size=8',
    '--allow-fs-read=' + str(donor), '--allow-fs-read=' + str(ROOT), '--allow-fs-read=' + str(RUN),
    '--allow-fs-read=/proc/self', '--allow-fs-read=/proc/sys/kernel/random/boot_id',
    *['--allow-fs-read=' + p for p in native['readRoots']],
    '--allow-fs-write=' + str(RUN), '--import', str(ROOT / 'guard.mjs'), str(ROOT / 'entry.mjs')]
env = {'FRW_TS_RUN': str(RUN), 'LANG': 'C', 'LC_ALL': 'C', 'HOME': str(RUN), 'TMPDIR': str(RUN)}
os.chdir(RUN)
os.execve(argv[0], argv, env)
