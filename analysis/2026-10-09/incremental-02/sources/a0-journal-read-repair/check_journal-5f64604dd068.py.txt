def guard(event,args):
    if event in ('subprocess.Popen','os.posix_spawn'):
        assert tuple(args[1]) in allowed_commands, ('forbidden subprocess',args[1])
    elif event=='os.kill':
        assert args[0] in [p.pid for p in children] and args[1]==signal.SIGKILL
    elif event in ('os.system','os.fork','os.posix_spawn','os.exec','fcntl.flock') or event.startswith('socket.'):
        raise AssertionError('FORBIDDEN '+event)
    elif event=='open':
        path,mode,flags=args
        writes=(isinstance(mode,str) and any(x in mode for x in 'wax+')) or (isinstance(flags,int) and flags&(os.O_WRONLY|os.O_RDWR|os.O_CREAT|os.O_TRUNC))
