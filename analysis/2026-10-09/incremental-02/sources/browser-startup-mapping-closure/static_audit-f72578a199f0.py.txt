"""Finite source/data/hash/AST audit. No candidate/test code execution."""
import ast
import datetime as dt
import difflib
import hashlib
import json
import os
from pathlib import Path
import stat

O = Path(__file__).resolve().parent
C = O / 'candidate'
B = O.parent / 'browser-locale-mapping-repair/candidate'
cache = {}

def digest(p):
    p = str(p)
    if p not in cache:
        with Path(p).open('rb') as stream:
            cache[p] = hashlib.file_digest(stream, 'sha256').hexdigest()
    return cache[p]

def read(p):
    return json.loads(Path(p).read_text())

def ref(p):
    p = Path(p)
    return {'path': str(p), 'sha256': digest(p), 'bytes': p.stat().st_size}

inventory = read(O / 'baseline-source-inventory.json')
names = sorted(inventory)
assert len(names) == 27 and sorted(p.name for p in C.iterdir()) == names
for n, row in inventory.items():
    assert digest(row['path']) == row['sha256']
    s = (C / n).lstat()
    assert stat.S_ISREG(s.st_mode) and s.st_uid == os.getuid() and s.st_nlink == 1
    assert stat.S_IMODE(s.st_mode) == 0o444
assert not any((C / n).exists() for n in ('runtime-clock.json', 'runtime-admission.json', 'controls', 'native', 'project', 'owner', '__pycache__'))

python_checks = {}
for name in (n for n in names if n.endswith('.py')):
    old, new = (B / name).read_text(), (C / name).read_text()
    old_tree, new_tree = ast.parse(old), ast.parse(new)
    if name in ('run.py', 'launch.py'):
        assert old.count("admission['generation']==4") == new.count("admission['generation']==5") == 1
        normalized = ast.parse(new.replace("admission['generation']==5", "admission['generation']==4"))
        if name == 'run.py':
            old_rest = [n for n in old_tree.body if not isinstance(n, ast.FunctionDef) or n.name != 'snapshot']
            new_rest = [n for n in normalized.body if not isinstance(n, ast.FunctionDef) or n.name != 'snapshot']
            assert [ast.dump(n) for n in old_rest] == [ast.dump(n) for n in new_rest]
            python_checks[name] = 'Only snapshot body and inert admission literal4->5 changed; all other AST identical'
        else:
            assert new == old.replace("admission['generation']==4", "admission['generation']==5")
            assert ast.dump(normalized) == ast.dump(old_tree)
            python_checks[name] = 'Only inert admission literal4->5; otherwise byte-identical'
        expected = ''.join(difflib.unified_diff(old.splitlines(True), new.splitlines(True), fromfile=str(B / name), tofile=str(C / name)))
        assert (C / (name + '.delta.patch')).read_text() == expected
    else:
        assert new == old and ast.dump(new_tree) == ast.dump(old_tree)
        python_checks[name] = 'Byte-identical and AST-identical'

snapshot = next(n for n in ast.parse((C / 'run.py').read_text()).body if isinstance(n, ast.FunctionDef) and n.name == 'snapshot')
expected_snapshot = ast.parse("""def snapshot():
 save('pytest-progress.json',state)
 mapped_paths=[]
 for line in Path('/proc/self/maps').read_text().splitlines():
  fields=line.split(None,5)
  if len(fields)==6 and fields[5].startswith('/'):
   mapped_paths.append(fields[5])
 save('loaded-mapping-paths.json',mapped_paths)
 save('module-origins.json',origins())
 mapped={}
 for path in mapped_paths:
  assert not path.endswith(' (deleted)'), 'DELETED_MAPPING'
  mapped[path]=sha(path)
 save('loaded-mappings.json',mapped)
""").body[0]
assert ast.dump(snapshot) == ast.dump(expected_snapshot)
assert not {'pins', 'read_paths', 'fence', 'install'} & {n.id for n in ast.walk(snapshot) if isinstance(n, ast.Name)}
snapshot_regression = ast.parse((O / 'test_snapshot_mapping_inventory_source.py').read_text())
snapshot_tests = [n.name for n in snapshot_regression.body if isinstance(n, ast.FunctionDef) and n.name.startswith('test_')]
assert len(snapshot_tests) == 5

map_checks = {}
for name in ('input-pins.json', 'selected-pins.json', 'import-path-pins.json', 'expected-suite-cases.json'):
    assert (B / name).read_bytes() == (C / name).read_bytes()
    map_checks[name] = {'entries': len(read(C / name)), 'byte_identical': True}
old_runtime, new_runtime = read(B / 'runtime-flat-pins.json'), read(C / 'runtime-flat-pins.json')
provenance = read(O / 'startup-file-provenance.json')
additions = {p: row['sha256'] for p, row in provenance['files'].items()}
assert all(new_runtime.get(p) == h for p, h in old_runtime.items())
assert set(new_runtime) - set(old_runtime) == set(additions)
assert len(old_runtime) == 13293 and len(new_runtime) == 13295
assert all(new_runtime[p] == h == digest(p) for p, h in additions.items())
for p, row in provenance['files'].items():
    current = os.stat(p, follow_symlinks=False)
    assert {k: getattr(current, 'st_' + k) for k in row['identity_after']} == row['identity_after']
    assert row['identity_before'] == row['identity_after']
map_checks['runtime-flat-pins.json'] = {'old_entries': 13293, 'new_entries': 13295, 'old_exact_subset': True, 'additions': additions, 'removed_or_changed': []}
old_launch, new_launch = read(B / 'launch-pins.json'), read(C / 'launch-pins.json')
assert len(old_launch) == len(new_launch) == 10
assert {Path(p).name for p in old_launch} == {Path(p).name for p in new_launch}
assert all(Path(p).parent == C and digest(p) == h for p, h in new_launch.items())

counts = {k: sum(v.values()) for k, v in read(C / 'expected-suite-cases.json').items()}
assert counts == {'controls': 21, 'native': 116, 'project': 38, 'owner': 71}
diagnostic = ast.parse((C / 'test_read_refusal_diagnostic_source.py').read_text())
function_count = case_count = 0
for node in diagnostic.body:
    if isinstance(node, ast.FunctionDef) and node.name.startswith('test_'):
        function_count += 1
        cases = 1
        for dec in node.decorator_list:
            if isinstance(dec, ast.Call) and ast.unparse(dec.func) == 'pytest.mark.parametrize':
                assert isinstance(dec.args[1], (ast.List, ast.Tuple))
                cases *= len(dec.args[1].elts)
        case_count += cases
assert (function_count, case_count) == (14, 27)

    assert row['observed_path'] == row['canonical_path'] == str(Path(row['observed_path']).resolve(strict=True))
    assert digest(row['canonical_path']) == row['sha256']
    assert all(new_runtime[p] == row['sha256'] and str(Path(p).resolve(strict=True)) == row['canonical_path'] for p in row['runtime_source_paths'])

baseline_package = read(B / 'execution-package-pins.json')
package = read(C / 'execution-package-pins.json')
assert all(package.get(p) == h for p, h in baseline_package.items())
assert str(C / 'execution-package-pins.json') not in package
assert {str(C / n) for n in names if n != 'execution-package-pins.json'} <= set(package)
assert all(digest(p) == h for p, h in package.items())
owned = {str(C / n) for n in names}
owned |= {str(O / n) for n in ('startup-file-provenance.json', 'startup-mapping-readiness.json', 'mapping-input-delta.json', 'test_snapshot_mapping_inventory_source.py')}
references, edges = [], {}

def record(source, path, expected, size=None):
    assert digest(path) == expected, ('STALE_HASH', source, path)
    if size is not None:
        assert Path(path).stat().st_size == size
    references.append({'source': source, 'target': path, 'sha256': expected})
    if path in owned:
        edges.setdefault(source, set()).add(path)

def walk(source, node):
    if isinstance(node, dict):
        if isinstance(node.get('path'), str) and isinstance(node.get('sha256'), str):
            record(source, node['path'], node['sha256'], node.get('bytes'))
        for value in node.values():
            walk(source, value)
    elif isinstance(node, list):
        for value in node:
            walk(source, value)

for p in sorted(owned):
    if not p.endswith('.json') or Path(p).name in ('input-pins.json', 'selected-pins.json', 'runtime-flat-pins.json', 'import-path-pins.json'):
        continue
    data = read(p)
    if Path(p).name in ('launch-pins.json', 'execution-package-pins.json'):
        for target, expected in data.items():
            record(p, target, expected)
    else:
        walk(p, data)
seen, active = set(), set()

def visit(node):
    assert node not in active, ('CIRCULAR_REFERENCE', node)
    if node in seen:
        return
    active.add(node)
    for target in edges.get(node, ()):
        visit(target)
    active.remove(node)
    seen.add(node)

for p in sorted(owned):
    visit(p)
changed, unchanged, patches = [], [], []
for name in names:
    old, new = (B / name).read_text(), (C / name).read_text()
    if old == new:
        unchanged.append(name)
    else:
        changed.append(name)
        patches.append(''.join(difflib.unified_diff(old.splitlines(True), new.splitlines(True), fromfile=str(B / name), tofile=str(C / name))))
assert (O / 'all-source-changes.patch').read_text() == ''.join(patches)
report = {'status': 'PASS_SOURCE_DATA_HASH_AST_ONLY', 'candidate': str(C), 'package': ref(C / 'execution-package-pins.json'),
          'candidate_files': len(names), 'candidate_mode': '0444', 'python_ast_checks': python_checks,
          'snapshot_structural_check': 'Exact expected AST: one maps read, complete path inventory saved before origins/sha, existing deletion/read/hash validation preserved, no authority mutation',
          'snapshot_regression_source_cases': snapshot_tests, 'snapshot_regressions_executed': False,
          'map_checks': map_checks, 'original_suite_static_counts': counts, 'diagnostic_static_cases': case_count,
          'observed_startup_paths_covered': 11, 'prior_package_entries_preserved': len(baseline_package),
          'current_package_entries': len(package), 'verified_metadata_references': len(references), 'references': references,
          'graph_acyclic': True, 'stale_hashes': [], 'changed_files': changed, 'unchanged_files': unchanged,
          'diffs_match_actual_bytes': True, 'distinct_files_hashed': len(cache), 'whole_corpus_leaf_rehash': False,
          'candidate_imports_collection_tests': False, 'runtime_clock_or_admission_created': False,
          'actual_generation5_assigned': False, 'phase_directories_present': False,
          'finished_msk': dt.datetime.now(dt.timezone(dt.timedelta(hours=3))).isoformat()}
(O / 'static-audit.json').write_text(json.dumps(report, indent=2) + '\n')
print(json.dumps({k: report[k] for k in ('status', 'package', 'candidate_files', 'current_package_entries', 'verified_metadata_references', 'observed_startup_paths_covered', 'diagnostic_static_cases', 'distinct_files_hashed')}, indent=2))
