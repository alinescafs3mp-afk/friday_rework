def digest(p):
    with Path(p).open('rb') as f:
        return hashlib.file_digest(f, 'sha256').hexdigest()

def ref(p):
    p = Path(p)
    return {'path': str(p), 'sha256': digest(p), 'bytes': p.stat().st_size}

def read(p):
    return json.loads(Path(p).read_text())

def save(p, value):
    p = Path(p)
    assert p.is_relative_to(O)
    p.write_text(json.dumps(value, indent=2, ensure_ascii=False) + '\n')

def identity(s):
    return {k: getattr(s, 'st_' + k) for k in ('dev', 'ino', 'mode', 'uid', 'gid', 'nlink', 'size', 'mtime_ns', 'ctime_ns')}

def nofollow_pin(path):
    path = Path(path)
    assert path.is_absolute() and '..' not in path.parts
    fds, parents = [], []
    try:
        fd = os.open('/', os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
        fds.append(fd)
        parents.append({'path': '/', 'identity': identity(os.fstat(fd))})
        current = Path('/')
        for part in path.parts[1:-1]:
            current /= part
            fd = os.open(part, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=fd)
            fds.append(fd)
            parents.append({'path': str(current), 'identity': identity(os.fstat(fd))})
        for row in parents:
            s = row['identity']
            assert stat.S_ISDIR(s['mode']) and s['uid'] == s['gid'] == 0
            assert not s['mode'] & 0o022
        before = os.stat(path.name, dir_fd=fd, follow_symlinks=False)
        leaf = os.open(path.name, os.O_RDONLY | os.O_NOFOLLOW, dir_fd=fd)
        fds.append(leaf)
        opened = os.fstat(leaf)
        assert identity(before) == identity(opened)
        assert stat.S_ISREG(opened.st_mode) and opened.st_uid == opened.st_gid == 0
        assert opened.st_nlink == 1 and not opened.st_mode & 0o022
        with os.fdopen(os.dup(leaf), 'rb') as stream:
            sha = hashlib.file_digest(stream, 'sha256').hexdigest()
        after = os.fstat(leaf)
        assert identity(before) == identity(after) == identity(os.stat(path.name, dir_fd=fd, follow_symlinks=False))
        for i, row in enumerate(parents):
            assert row['identity'] == identity(os.fstat(fds[i]))
            if i:
                assert row['identity'] == identity(os.stat(Path(row['path']).name, dir_fd=fds[i-1], follow_symlinks=False))
        assert identity(os.stat('/', follow_symlinks=False)) == parents[0]['identity']
        assert path.resolve(strict=True) == path
        return {'path': str(path), 'sha256': sha, 'bytes': opened.st_size,
                'identity_before': identity(before), 'identity_after': identity(after),
                'protected_components': parents, 'nofollow_every_component': True,
                'all_component_bindings_and_fd_identities_rechecked': True}
    finally:
        for fd in reversed(fds):
            os.close(fd)
