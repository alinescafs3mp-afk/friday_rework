"""Finite AST/data checks only; never import or execute candidate/test bodies."""
import ast
import copy
import difflib
import hashlib
import json
import os
from pathlib import Path
import re
import resource

os.sched_setaffinity(0, sorted(os.sched_getaffinity(0))[:2])
resource.setrlimit(resource.RLIMIT_AS, (2 * 1024**3, 2 * 1024**3))
OUT = Path(__file__).parent
W = OUT.parent
C = Path('<SOL_WORK>/FRW041-REGISTRY-ASYNC-BOUNDARY-evidence/candidate')
PRIOR = W / 'browser-owner-controls-preparation/private_effect_fence.py'
NAME = 'test_import_read_boundary_source.py'
TARGET = 'test_process_socket_thread_and_sigint_guards_preserved'


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def ref(path):
    return {'path': str(path), 'sha256': sha(path), 'bytes': path.stat().st_size}


def dump(node):
    return ast.dump(node, include_attributes=False)


def functions(path):
    return {n.name: n for n in ast.parse(path.read_text()).body if isinstance(n, ast.FunctionDef)}


old = functions(PRIOR)
current = functions(C / 'private_effect_fence.py')
before = (C / NAME).read_text()
after = (OUT / NAME).read_text()
before_tree, after_tree = ast.parse(before), ast.parse(after)
control = functions(OUT / NAME)[TARGET]
assert before[:before.index('def ' + TARGET)] == after[:after.index('def ' + TARGET)]
assert [dump(n) for n in before_tree.body if getattr(n, 'name', None) != TARGET] == [
    dump(n) for n in after_tree.body if getattr(n, 'name', None) != TARGET]
assert len(functions(C / NAME)) == len(functions(OUT / NAME))
assert len([n for n in after_tree.body if isinstance(n, ast.FunctionDef) and n.name.startswith('test_')]) == 10

# Read only the literal insertion specification; never eval/exec the test.
additions_node = next(n.value for n in control.body if isinstance(n, ast.Assign)
                      and any(isinstance(t, ast.Name) and t.id == 'additions' for t in n.targets))
additions = ast.literal_eval(additions_node)
unchanged_loop = next(n for n in control.body if isinstance(n, ast.For) and isinstance(n.iter, ast.Tuple))
unchanged = ast.literal_eval(unchanged_loop.iter)
assert unchanged == ('guarded_bind', 'guarded_thread_start', 'guarded_joinable', 'guarded_signal',
                     'check_path', 'check_fd', 'path_ok', 'fd_path', 'canonical')
assert tuple(additions) == ('guarded_close', 'guarded_socketpair', 'guarded_send', 'guarded_recv')
assert {k: [i for i, _ in v] for k, v in additions.items()} == {
    'guarded_close': [0, 3], 'guarded_socketpair': [1], 'guarded_send': [0], 'guarded_recv': [0]}
assert ast.unparse(unchanged_loop.body[0]) == 'assert dump(new[name]) == dump(old[name]), name'
reconstruction_loop = next(n for n in control.body if isinstance(n, ast.For) and isinstance(n.target, ast.Tuple))
expected_loop = ast.parse('''for name,insertions in additions.items():
    expected=old[name]
    for index,source in insertions:
        expected.body[index:index]=ast.parse(source).body
    assert dump(new[name])==dump(expected),name
''').body[0]
assert dump(reconstruction_loop) == dump(expected_loop)
closure_tail = ast.parse('''expected=old['closed_primitives']
summary=expected.body[0].value
summary.keys.insert(0,ast.Constant(value='registry_self_pipe'))
summary.values.insert(0,ast.parse('registry_cessation()',mode='eval').body)
assert dump(new['closed_primitives'])==dump(expected),'closed_primitives'
''').body
assert [dump(n) for n in control.body[-5:]] == [dump(n) for n in closure_tail]

expected = {}
for name in unchanged:
    expected[name] = copy.deepcopy(old[name])
for name, inserts in additions.items():
    expected[name] = copy.deepcopy(old[name])
    for index, source in inserts:
        expected[name].body[index:index] = ast.parse(source).body
expected['closed_primitives'] = copy.deepcopy(old['closed_primitives'])
summary = expected['closed_primitives'].body[0].value
assert isinstance(summary, ast.Dict)
summary.keys.insert(0, ast.Constant(value='registry_self_pipe'))
summary.values.insert(0, ast.parse('registry_cessation()', mode='eval').body)
assert len(expected) == 14
for name, node in expected.items():
    assert dump(current[name]) == dump(node), name
changed = tuple(additions) + ('closed_primitives',)
assert [name for name in expected if dump(current[name]) != dump(old[name])] == list(changed)

negative = []


def rejected(name, reason, mutate):
    node = copy.deepcopy(current[name])
    mutate(node)
    assert dump(node) != dump(expected[name]), (name, reason)
    negative.append({'function': name, 'mutation': reason, 'rejected_by_exact_AST': True})


for name in unchanged:
    rejected(name, 'remove first legacy statement', lambda n: n.body.pop(0))
legacy_indices = {'guarded_close': 4, 'guarded_socketpair': 2, 'guarded_send': 2, 'guarded_recv': 2}
for name, index in legacy_indices.items():
    rejected(name, 'remove legacy guard/closure statement', lambda n, i=index: n.body.pop(i))
for name in changed:
    rejected(name, 'append unapproved statement', lambda n: n.body.append(ast.Pass()))
    rejected(name, 'change signature', lambda n: n.args.args.append(ast.arg(arg='unapproved')))
rejected('guarded_close', 'remove registry ownership check', lambda n: n.body.pop(1))
rejected('guarded_close', 'remove native close verification', lambda n: n.body.pop(3))
rejected('guarded_close', 'move ownership check after native close', lambda n: n.body.insert(2, n.body.pop(1)))
rejected('guarded_socketpair', 'remove registry phase guard', lambda n: n.body.pop(1))
rejected('guarded_socketpair', 'unconditional registry delegation', lambda n: n.body.__setitem__(1, n.body[1].body[0]))
for name in ('guarded_send', 'guarded_recv'):
    rejected(name, 'remove registry endpoint/provenance check', lambda n: n.body[1].body.pop(0))
    rejected(name, 'remove exact payload/argument check', lambda n: n.body[1].body.pop(1))
    rejected(name, 'remove per-pair budget check', lambda n: n.body[1].body.pop(3))
    rejected(name, 'remove charge before native operation', lambda n: n.body[1].body.pop(4))
rejected('closed_primitives', 'drop legacy summary entry', lambda n: (n.body[0].value.keys.pop(1), n.body[0].value.values.pop(1)))
rejected('closed_primitives', 'drop registry cessation summary', lambda n: (n.body[0].value.keys.pop(0), n.body[0].value.values.pop(0)))
rejected('closed_primitives', 'replace registry cessation call by empty dict', lambda n: n.body[0].value.values.__setitem__(0, ast.Dict(keys=[], values=[])))

patch = ''.join(difflib.unified_diff(before.splitlines(True), after.splitlines(True),
                                   fromfile='a/' + NAME, tofile='b/' + NAME))
(OUT / 'replacement.patch').write_text(patch)
delta_base = W / 'browser-owner-controls-r3' / NAME
package = json.loads((C / 'execution-package-pins.json').read_text())
assert sha(delta_base) == package[str(delta_base)]
cumulative = ''.join(difflib.unified_diff(delta_base.read_text().splitlines(True), after.splitlines(True),
                                        fromfile=str(delta_base), tofile=str(C / NAME)))
(OUT / (NAME + '.delta.patch')).write_text(cumulative)
# Strictly reconstruct from each unified hunk, checking original bytes and lengths.
for original_text, delta in ((before, patch), (delta_base.read_text(), cumulative)):
    original, result, cursor = original_text.splitlines(True), [], 0
    lines = delta.splitlines(True)
    i = 2
    while i < len(lines):
        match = re.fullmatch(r'@@ -(\d+),(\d+) \+(\d+),(\d+) @@\n', lines[i])
        assert match, lines[i]
        start, old_count, new_start, new_count = map(int, match.groups())
        result.extend(original[cursor:start - 1])
        cursor = start - 1
        assert len(result) == new_start - 1
        used_old = used_new = 0
        i += 1
        while i < len(lines) and not lines[i].startswith('@@ '):
            line = lines[i]
            assert line[0] in ' +-'
            if line[0] in ' -':
                assert original[cursor] == line[1:]
                cursor += 1
                used_old += 1
            if line[0] in ' +':
                result.append(line[1:])
                used_new += 1
            i += 1
        assert (used_old, used_new) == (old_count, new_count)
    result.extend(original[cursor:])
    assert ''.join(result) == after

snapshot = json.loads((OUT / 'input-snapshot.json').read_text())
assert all(sha(Path(path)) == expected_sha for path, expected_sha in snapshot.items())
selected = json.loads((C / 'selected-pins.json').read_text())
production = {}
for path, expected_sha in selected.items():
    if path.endswith(('/model_tools.py', '/tools/registry.py', '/tools/dispatch_helpers.py')):
        assert sha(Path(path)) == expected_sha
        production[path] = expected_sha
assert len(production) == 3
cases = json.loads((C / 'expected-suite-cases.json').read_text())
counts = {phase: sum(entries.values()) for phase, entries in cases.items()}
assert counts == {'controls': 109, 'native': 116, 'project': 38, 'owner': 71, 'registry': 7}
assert sum(counts.values()) == 341

evidence = {
    'method': 'AST parse/dump/literal_eval and data/hash comparison only; no candidate or test execution',
    'stale_comparisons_reproduced_statically': list(changed),
    'unchanged_whole_function_assertions': list(unchanged),
    'changed_whole_functions_reconstructed_exactly': list(changed),
    'legacy_and_added_AST_fully_compared': True,
    'other_module_bytes_before_target_unchanged': True,
    'all_other_module_AST_unchanged': True,
    'supplementary_test_function_count_unchanged': 10,
    'negative_controls': negative,
    'negative_control_count': len(negative),
    'patch_strict_reconstruction_exact': True,
    'cumulative_delta_patch_strict_reconstruction_exact': True,
    'candidate_direct_files_unchanged': len(snapshot),
    'unchanged_production_pins': production,
    'selected_source_manifest_counts_unchanged': counts,
    'selected_total_source_cases': 341,
    'selected_collection_or_tests_run': False,
    'refs': {key: ref(path) for key, path in {
        'legacy_fence': PRIOR, 'current_fence': C / 'private_effect_fence.py',
        'original_test': C / NAME, 'replacement_test': OUT / NAME,
        'patch': OUT / 'replacement.patch', 'checker': Path(__file__),
        'delta_base': delta_base, 'cumulative_patch': OUT / (NAME + '.delta.patch'),
        'snapshot': OUT / 'input-snapshot.json'}.items()},
    'resources': {'affinity': sorted(os.sched_getaffinity(0)),
                  'address_space_limit': list(resource.getrlimit(resource.RLIMIT_AS)),
                  'children_created': 0},
}
(OUT / 'source-checks.json').write_text(json.dumps(evidence, indent=2) + '\n')
print(json.dumps({k: evidence[k] for k in ('negative_control_count', 'patch_strict_reconstruction_exact',
                                        'candidate_direct_files_unchanged', 'selected_total_source_cases')}))
