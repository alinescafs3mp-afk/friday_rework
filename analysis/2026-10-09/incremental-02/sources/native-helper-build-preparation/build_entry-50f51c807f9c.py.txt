def build_environment():
    return {'PATH': '/usr/bin:/bin', 'HOME': str(OUTPUT / 'home'),
            'TMPDIR': str(OUTPUT / 'tmp'), 'LANG': 'C.UTF-8', 'LC_ALL': 'C.UTF-8',
            'CC': '/usr/bin/cc', 'AR': '/usr/bin/ar', 'RANLIB': '/usr/bin/ranlib',
            'PKG_CONFIG': '/usr/bin/pkgconf', 'PYTHONDONTWRITEBYTECODE': '1',
            'PYTHONNOUSERSITE': '1', 'GIT_OPTIONAL_LOCKS': '0'}


def file_sha(path, budget):
    budget.check()
    path = Path(path)
    if path.resolve() != path or not stat.S_ISREG(path.lstat().st_mode):
        raise ValueError('native_build_regular_non_symlink_input_required')
    h = hashlib.sha256()
    with path.open('rb') as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b''):
            budget.check()
            h.update(chunk)
    budget.check()
    return h.hexdigest()


def verify_file(path, sha, budget):
    if file_sha(path, budget) != sha:
        raise ValueError('native_build_input_changed')


def inventory(root, budget):
    """Entire finite copied tree; no symlink or special-file traversal."""
    def fail(error):
        raise error

    result = {}
    if root.resolve() != root or not root.is_dir():
        raise ValueError('native_build_real_directory_required')
    for directory, dirs, files in os.walk(root, followlinks=False, onerror=fail):
        budget.check()
        for name in dirs:
            item = Path(directory) / name
            if item.is_symlink():
                raise ValueError('native_build_source_symlink_refused')
        for name in files:
            path = Path(directory) / name
            result[path.relative_to(root).as_posix()] = file_sha(path, budget)
    return result


def fixed_commands():
    """Only these donor build targets. No install, tests or output execution."""
    bsrc = OUTPUT / 'source/bubblewrap-0.11.1'
    ssrc = OUTPUT / 'generated/slirp4netns-1.3.3'
    bbuild = OUTPUT / 'build/bubblewrap'
    sbuild = OUTPUT / 'build/slirp4netns'
    return [
        ('packages', ['/usr/bin/dpkg-query', '-W', '-f', '${binary:Package}\t${Version}\n',
          'meson', 'ninja-build', 'pkgconf', 'autoconf', 'automake', 'libcap-dev',
          'libglib2.0-dev', 'libslirp-dev', 'libseccomp-dev', 'libselinux-dev',
          'gcc', 'binutils', 'make', 'libc6-dev'], OUTPUT),
        ('compiler', ['/usr/bin/cc', '--version'], OUTPUT),
        ('bwrap-configure', ['/usr/bin/meson', 'setup', str(bbuild), str(bsrc),
          '--buildtype=release', '--wrap-mode=nodownload', '-Dman=disabled',
          '-Dtests=false', '-Dselinux=enabled', '-Dbash_completion=disabled',
          '-Dzsh_completion=disabled'], OUTPUT),
        ('bwrap-build', ['/usr/bin/ninja', '-C', str(bbuild), '-j', '2', 'bwrap'], OUTPUT),
        ('slirp-autoreconf', ['/usr/bin/autoreconf', '--force', '--install'], ssrc),
        ('slirp-configure', ['/bin/sh', str(ssrc / 'configure'),
          '--disable-dependency-tracking'], sbuild),
        ('slirp-build', ['/usr/bin/make', '-C', str(sbuild), '-j2', 'slirp4netns'], OUTPUT),
        ('bwrap-elf', ['/usr/bin/readelf', '-h', '-l', '-d', str(bbuild / 'bwrap')], OUTPUT),
        ('slirp-elf', ['/usr/bin/readelf', '-h', '-l', '-d', str(sbuild / 'slirp4netns')], OUTPUT),
    ]


def build(custody):
    """Reuse exact existing Containment + already-running original Budget.

    Caller preconditions are substantive: approved source and C patches,
    exact installed dependencies, fresh one-shot build ownership, resource
    enforcement, authorized filesystem effects, and current system-bwrap pin.
    Merely constructing this object or calling this function grants none.
    """
    from scripts.install_containment import Containment, Budget
    from scripts.friday_install import publish

    if type(custody) is not Containment or type(custody.budget) is not Budget:
        raise ValueError('existing_install_custody_and_original_budget_required')
    budget = custody.budget
    budget.check()
    if os.geteuid() == 0 or os.geteuid() != os.getuid():
        raise ValueError('unprivileged_native_build_required')
    if HERE.resolve() != HERE or OUTPUT.exists() or OUTPUT.is_symlink():
        raise ValueError('fresh_native_build_destination_required')
    for rel, sha in PROJECT_PINS.items():
        verify_file(R / rel, sha, budget)
    # No inherited compiler hooks, make flags, caches, proxies or credentials.
    env = build_environment()
    if custody.env != env:
        raise ValueError('exact_native_build_environment_required')
    selected = []
    for name, pristine_sha, patched_sha in DONORS:
        pristine_path = SOURCES / (name + '-pins.json')
        patched_path = SOURCES / (name + '-patched-pins.json')
        verify_file(pristine_path, pristine_sha, budget)
        verify_file(patched_path, patched_sha, budget)
        pristine = budget.call(lambda p=pristine_path: json.loads(p.read_text()))
        patched = budget.call(lambda p=patched_path: json.loads(p.read_text()))
        archive = SOURCES / (name + '.tar.gz')
        verify_file(archive, pristine['archive_sha256'], budget)
        verify_file(Path(patched['patch']['path']), patched['patch']['sha256'], budget)
        if inventory(SOURCES / name, budget) != pristine['files']:
            raise ValueError('pristine_native_source_changed')
        if inventory(SOURCES / (name + '-patched'), budget) != patched['files']:
            raise ValueError('patched_native_source_changed')
        changed = sorted(k for k in pristine['files'] if pristine['files'][k] != patched['files'].get(k))
        if set(pristine['files']) != set(patched['files']) or changed != [patched['only_changed_leaf']]:
            raise ValueError('exact_one_leaf_native_patch_required')
        selected.append((name, pristine, patched))
    # The caller must have completed the existing native custody probe and
    # bound its result before calling. This leaf does not create a second probe.
    budget.call(OUTPUT.mkdir, mode=0o700)
    for rel in ('home', 'tmp', 'source', 'generated', 'build', 'build/slirp4netns', 'provenance', 'artifacts'):
        budget.call((OUTPUT / rel).mkdir, mode=0o700)
    publish(OUTPUT / 'started.json', {
        'state': 'BUILD_STARTED_NOT_QUALIFIED', 'original_deadline': budget.deadline,
        'system_custody_pin': custody.pin, 'project_pins': PROJECT_PINS,
        'entry_sha256': file_sha(HERE / 'build_entry.py', budget),
        'source_pin_sha256': {name: {'pristine': a, 'patched': b} for name, a, b in DONORS},
        'commands': [{'phase': phase, 'argv': argv, 'cwd': str(cwd)} for phase, argv, cwd in fixed_commands()],
        'environment': env, 'runtime': 'UNBOUND', 'no_retry': True,
    }, budget=budget)
    for name, pristine, patched in selected:
        target = OUTPUT / 'source' / name
        budget.call(shutil.copytree, SOURCES / (name + '-patched'), target)
        if inventory(target, budget) != patched['files']:
            raise ValueError('copied_native_source_changed')
        _, pristine_sha, patched_sha = next(row for row in DONORS if row[0] == name)
        for path, sha in ((SOURCES / (name + '.tar.gz'), pristine['archive_sha256']),
                          (SOURCES / (name + '-pins.json'), pristine_sha),
                          (SOURCES / (name + '-patched-pins.json'), patched_sha),
                          (Path(patched['patch']['path']), patched['patch']['sha256'])):
            dest = OUTPUT / 'provenance' / path.name
            budget.call(shutil.copyfile, path, dest)
            verify_file(dest, sha, budget)
    budget.call(shutil.copytree, OUTPUT / 'source/slirp4netns-1.3.3', OUTPUT / 'generated/slirp4netns-1.3.3')
    observations = {}
    for phase, command, cwd in fixed_commands():
        # Existing run checks the system pin, reserves original-budget cleanup,
        # and requires native init-exit evidence. No exception is downgraded.
        text, observations[phase] = custody.run(command, cwd, log=OUTPUT / 'logs' / phase)
        if phase == 'packages':
            installed = dict(line.split('\t', 1) for line in text.splitlines())
            installed = {k.split(':', 1)[0]: v for k, v in installed.items()}
            if any(installed.get(k) != v for k, v in REQUIRED_PACKAGES.items()):
                raise ValueError('reviewed_native_build_dependency_version_required')
    for name, _, patched in selected:
        if inventory(OUTPUT / 'source' / name, budget) != patched['files']:
            raise ValueError('build_modified_preserved_source')
    generated = inventory(OUTPUT / 'generated/slirp4netns-1.3.3', budget)
    if any(generated.get(k) != v for k, v in selected[1][2]['files'].items()):
        raise ValueError('autoreconf_modified_donor_source')
    artifacts = {}
    for name, rel in (('bwrap', 'build/bubblewrap/bwrap'), ('slirp4netns', 'build/slirp4netns/slirp4netns')):
        source = OUTPUT / rel
        sha = file_sha(source, budget)
        if source.lstat().st_mode & (stat.S_ISUID | stat.S_ISGID):
            raise ValueError('setid_native_artifact_refused')
        target = OUTPUT / 'artifacts' / name
        budget.call(shutil.copyfile, source, target)
        budget.call(target.chmod, 0o700)
        verify_file(target, sha, budget)
        artifacts[name] = {'path': str(target), 'sha256': sha, 'build_path': str(source)}
    result = {
        'state': 'BUILT_UNQUALIFIED_RUNTIME_UNBOUND', 'runtime': 'UNBOUND',
        'original_deadline': budget.deadline, 'artifacts': artifacts,
        'source_pin_sha256': {name: {'pristine': a, 'patched': b} for name, a, b in DONORS},
        'started_sha256': file_sha(OUTPUT / 'started.json', budget),
        'preserved_provenance': inventory(OUTPUT / 'provenance', budget),
        'generated_slirp_source': generated,
        'generated_build_inputs': inventory(OUTPUT / 'build', budget),
        'observations': observations, 'log_files': inventory(OUTPUT / 'logs', budget),
        'tests': 'NOT_RUN', 'built_helper_execution': False, 'installed': False,
        'configuration_probes': 'BUILD_ONLY_MAY_HAVE_EXECUTED',
        'runtime_library_closure': 'NOT_QUALIFIED', 'release_accepted': False,
    }
    publish(OUTPUT / 'result.json', result, budget=budget)
    budget.check()
    return result
