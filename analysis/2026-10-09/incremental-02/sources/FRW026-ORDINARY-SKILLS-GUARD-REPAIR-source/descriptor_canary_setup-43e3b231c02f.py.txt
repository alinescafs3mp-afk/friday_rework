"""Proposed synthetic-FD bootstrap only; NOT_RUN, separately admitted canary only.
Reuses invocation.py in the SAME process. No fixture child, credential or daemon.
The future owner creates/pins the two fake files under its private fixture parent.
"""
import hashlib, json, os, pathlib, runpy, sys


def main():
    source = pathlib.Path(__file__).resolve().parent
    root = pathlib.Path(sys.argv[1]).resolve(strict=True)
    clock_path = root / 'clock.json'
    before = clock_path.read_bytes()
    clock = json.loads(before)
    if clock.get('execution_authorized') is not True or clock.get('independent_source_review') != 'PASS':
        raise SystemExit('NO_SEPARATE_ADMISSION')
    if not clock.get('canary', '').startswith('inherited_foreign_descriptor-'):
        raise SystemExit('NOT_DESCRIPTOR_CANARY')
    if clock.get('synthetic_fixture_parent') != str(root.parent):
        raise SystemExit('FIXTURE_PARENT_NOT_OWNED_BY_ADMISSION')
    identities = {}
    for key, leaf, flags in (('synthetic_credential_fd', 'auth.json', os.O_RDONLY),
                             ('synthetic_write_fd', 'synthetic-outside.txt', os.O_WRONLY)):
        path = root.parent / leaf
        if path.is_symlink() or path.read_bytes() != b'SYNTHETIC_GUARD_CANARY_ONLY\n':
            raise SystemExit('NOT_EXACT_SYNTHETIC_FIXTURE')
        fd = os.open(path, flags | os.O_NOFOLLOW)
        os.set_inheritable(fd, True)
        st = os.fstat(fd)
        identities[key] = {'fd': fd, 'dev': st.st_dev, 'ino': st.st_ino,
                           'mode': st.st_mode, 'source_sha256': hashlib.sha256(path.read_bytes()).hexdigest()}
        clock[key] = fd
    # Only phase descriptor metadata is added: no deadline/budget/accounting change.
    assert clock_path.read_bytes() == before
    clock_path.write_text(json.dumps(clock, indent=2) + '\n')
    (root / 'synthetic-fd-launch.json').write_text(json.dumps(identities, indent=2) + '\n')
    sys.argv[0] = str(source / 'invocation.py')
    runpy.run_path(str(source / 'invocation.py'), run_name='__main__')


if __name__ == '__main__':
    main()
