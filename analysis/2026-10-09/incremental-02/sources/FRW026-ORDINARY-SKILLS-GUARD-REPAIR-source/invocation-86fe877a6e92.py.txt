"""Small proposed exec glue for installed timeout, not a grant or process manager.
NOT_RUN. Review and a separate finite owner admission are mandatory.
Existing exec tool owns the root handle/status/reap; timeout owns its process group.
"""
import hashlib, json, os, pathlib, re, sys, time


def main():
    source = pathlib.Path(__file__).resolve().parent
    # Exact private run root populated by the independently admitting owner.
    run = pathlib.Path(sys.argv[1]).resolve(strict=True)
    name, entry, selection = sys.argv[2:5]
    if not re.fullmatch(r'[a-z][a-z0-9-]{0,63}', name):
        raise SystemExit('INVALID_RUN_NAME')
    if entry not in ('guard.py', 'native_guard.py') or selection not in ('@canary', '@project', '@native'):
        raise SystemExit('UNREVIEWED_ROUTE')
    if len(sys.argv) > 5 and sys.argv[5:] != ['--collect-only']:
        raise SystemExit('UNREVIEWED_ARGS')
    clock = json.loads((run / 'clock.json').read_text())
    # The source-only repair clock deliberately does NOT have these fields.
    if clock.get('execution_authorized') is not True or clock.get('independent_source_review') != 'PASS':
        raise SystemExit('NO_SEPARATE_REVIEWED_ADMISSION')
    if clock.get('entry') != entry or clock.get('selection') != selection:
        raise SystemExit('ADMISSION_ROUTE_MISMATCH')
    if pathlib.Path('/proc/sys/kernel/random/boot_id').read_text().strip() != clock['boot_id']:
        raise SystemExit('BOOT_MISMATCH')
    source_pins = source / 'source-pins.json'
    if hashlib.sha256(source_pins.read_bytes()).hexdigest() != clock['reviewed_source_pins_sha256']:
        raise SystemExit('SOURCE_REVIEW_PIN_MISMATCH')
    for relative, expected in json.loads(source_pins.read_text()).items():
        if hashlib.sha256((source / relative).read_bytes()).hexdigest() != expected:
            raise SystemExit('SOURCE_DRIFT:' + relative)
    # Owner preflight binds the executable/runtime before this trusted launcher.
    # Controls recheck all runtime pins under the external wall cap before pytest.
    # Do not put a large runtime inventory ahead of the effectful-unit timeout.
    if selection == '@canary':
        canary = json.loads((source / 'canary-plan.json').read_text())[clock['canary']]
        ceiling = canary['outer_wall_seconds']
        if clock['guard_wall_seconds'] != canary['guard_wall_seconds']:
            raise SystemExit('CANARY_WALL_MISMATCH')
    else:
        ceiling = 120 if sys.argv[5:] else 600
        if not 0 < clock['guard_wall_seconds'] <= ceiling:
            raise SystemExit('INVALID_GUARD_WALL')
    # Never manufacture a fresh deadline: ORIGINAL admitting task deadline only.
    # Reserve startup time; external duration is computed AFTER pin verification.
    remaining = clock['deadline_monotonic'] - time.monotonic() - 1.0
    duration = min(float(ceiling), remaining)
    if duration <= 0:
        raise SystemExit('ORIGINAL_ASSIGNMENT_EXPIRED')
    # No --foreground: installed timeout owns the whole exact process group.
    # SIGKILL is independent of pytest/Python exceptions and SIGALRM handlers.
    interpreter = '<PROJECT>/.runtime/hermes-telegram-test/env/bin/python'
    os.execv('/usr/bin/timeout', ['timeout', '--signal=KILL', f'{duration:.6f}s',
                                 interpreter, '-B', str(source / entry), name, str(run),
                                 selection, *sys.argv[5:]])


if __name__ == '__main__':
    main()
