import json, os, resource, sys, time
resource.setrlimit(resource.RLIMIT_AS, (128 << 20, 128 << 20))
resource.setrlimit(resource.RLIMIT_CPU, (2, 2))
os.sched_setaffinity(0, {0, 1})
assert sys.implementation.name == 'cpython' and sys.version_info[:3] == (3, 14, 7)
assert sys.getprofile() is None
mon = sys.monitoring
tool = mon.PROFILER_ID
assert mon.get_tool(tool) is None
calls = []
seen = []
def guard(label):
    calls.append(label)
def monitor(code, offset):
    assert code is guard.__code__
    seen.append(len(calls))
def workload():
    return None
def profile(frame, event, arg):
    if event == 'call' and frame.f_code is workload.__code__:
        guard('inside_profile')
def audit(event, args):
    if event == 'friday.source_probe.monitoring_coverage':
        guard('inside_audit')
sys.addaudithook(audit)
mon.use_tool_id(tool, 'friday-bounded-source-coverage-probe')
try:
    mon.register_callback(tool, mon.events.PY_START, monitor)
    mon.set_local_events(tool, guard.__code__, mon.events.PY_START)
    guard('ordinary')
    ordinary = len(seen)
    sys.setprofile(profile)
    try:
        workload()
    finally:
        assert sys.getprofile() is profile
        sys.setprofile(None)
    after_profile = len(seen)
    sys.audit('friday.source_probe.monitoring_coverage')
    after_audit = len(seen)
finally:
    mon.clear_tool_id(tool)
    mon.free_tool_id(tool)
assert calls == ['ordinary', 'inside_profile', 'inside_audit']
assert (ordinary, after_profile, after_audit) == (1, 1, 1)
assert sys.getprofile() is None and mon.get_tool(tool) is None
print(json.dumps({'state': 'EXACT_CPYTHON3147_MONITORING_COVERAGE_GAP_OBSERVED',
 'actual_guard_calls': calls, 'monitoring_event_counts': [ordinary, after_profile, after_audit],
 'profile_and_audit_callees_invisible': True, 'ordinary_call_observed': True,
 'limits': {'cpu_seconds': 2, 'address_space_bytes': 128 << 20, 'affinity': sorted(os.sched_getaffinity(0))},
 'product_candidate_imported': False, 'product_gate': False}))
