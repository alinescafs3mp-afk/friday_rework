def registry_audit_wrapper(obj,family,kind,proto):
 pending=getattr(LOCAL,'registry_pair',None)
 # Read only the two exact pinned implementation frames, never publish locals.
 # At the first audit a is a detached raw object, b is the live second raw.
 # At the second audit a is wrapper0, b is the same now-detached raw object.
 frame=READ_GETFRAME(1);pair_frame=None;init_frame=None
 try:
  while frame:
   if frame.f_code is SOCKET_PAIR.__code__ and frame.f_globals is socket.__dict__:
    if pair_frame is not None:registry_refuse('socketpair:registry_foreign_native_wrapper','audit')
    pair_frame=frame
   if frame.f_code is REGISTRY_SOCKET_INIT.__code__ and frame.f_globals is socket.__dict__:
    if init_frame is not None:registry_refuse('socketpair:registry_foreign_native_wrapper','audit')
    init_frame=frame
   frame=frame.f_back
  index=len(pending['objects']) if pending is not None else -1
  checks=(('pending',pending is not None),
   ('current_node',pending is not None and pending['node']==state.get('current_node')),
   ('owner_loop',pending is not None and registry_stack('_make_self_pipe',creating=True) is pending['loop']),
   ('pair_frame',pair_frame is not None),('init_frame',init_frame is not None),
   ('exact_python_object',type(obj) is SOCKET_CLASS),('not_raw_object',type(obj) is not REGISTRY_RAW_SOCKET_CLASS),
   ('family',type(family) in (int,socket.AddressFamily) and family==socket.AF_UNIX),
   ('kind',type(kind) in (int,socket.SocketKind) and kind==socket.SOCK_STREAM),
   ('proto',type(proto) is int and proto==0),('index',type(index) is int and 0<=index<2),
   ('main_thread',threading.current_thread() is threading.main_thread()))
  if (pending is None or pending['node']!=state.get('current_node')
      or registry_stack('_make_self_pipe',creating=True) is not pending['loop']
      or not registry_wrapper_allowed(family,kind,proto,index,type(obj) is REGISTRY_RAW_SOCKET_CLASS,
       type(obj) is SOCKET_CLASS,pair_frame is not None and init_frame is not None,
       threading.current_thread() is threading.main_thread())):
   registry_refuse('socketpair:registry_foreign_native_wrapper','audit',checks)
  a=pair_frame.f_locals.get('a');b=pair_frame.f_locals.get('b')
  fd=init_frame.f_locals.get('fileno')
  checks=(('init_self',init_frame.f_locals.get('self') is obj),
          ('new_identity',all(obj is not previous for previous in pending['objects'])),
          ('handoff_fd',type(fd) is int and fd>=0))
  if not all(value for _,value in checks):registry_refuse('socketpair:registry_handoff_identity','handoff',checks)
  if index==0:
   checks=(('first_raw',type(a) is REGISTRY_RAW_SOCKET_CLASS),
           ('second_raw',type(b) is REGISTRY_RAW_SOCKET_CLASS),('distinct_raw',a is not b))
   if not all(value for _,value in checks):registry_refuse('socketpair:registry_handoff_identity','handoff',checks)
   births=(registry_fd_identity(fd),registry_fd_birth(b))
   checks=(('first_raw_detached',a.fileno()==-1),('distinct_socket_births',registry_births_allowed(births)))
   if not all(value for _,value in checks):registry_refuse('socketpair:registry_handoff_identity','handoff',checks)
   pending['raw']=(a,b);pending['births']=births
  else:
   checks=(('first_wrapper',a is pending['objects'][0]),('second_raw',b is pending['raw'][1]),
           ('raw_detached',all(raw.fileno()==-1 for raw in pending['raw'])),
           ('first_birth',registry_fd_birth(pending['objects'][0])==pending['births'][0]),
           ('second_birth',registry_fd_identity(fd)==pending['births'][1]))
   if not all(value for _,value in checks):registry_refuse('socketpair:registry_handoff_identity','handoff',checks)
  pending['objects'].append(obj);COUNTS['pair_wrapping_events']+=1
 finally:del frame,pair_frame,init_frame

def diagnostic_text(value,limit):
 # Exact builtins only: never call a path object's fspath/str/repr or a subclass.
 if type(value) is str:
  return {'value':value[:limit],'truncated':len(value)>limit,'kind':'str'}
 if type(value) is bytes:
  text=value[:limit].decode('utf-8','backslashreplace')
  return {'value':text[:limit],'truncated':len(value)>limit or len(text)>limit,'kind':'bytes_utf8_backslashreplace'}
 return {'value':None,'truncated':False,'kind':'not_exact_text'}

def read_diagnostic(raw,text,resolved,stage):
 # Reuse the resolution already attempted by read_ok; no new file/FD/syscall read.
 # Frame attributes are interpreter filename/function/line only, never locals,
 # source text, globals, arguments, exception rendering or credential values.
 result={'schema':'friday.offline.read-refusal-metadata.v1','status':'AVAILABLE',
  'raw_path':diagnostic_text(raw,READ_DIAGNOSTIC_PATH_CHARS),
  'evaluated_path':diagnostic_text(text,READ_DIAGNOSTIC_PATH_CHARS),
  'canonical_path':diagnostic_text(resolved,READ_DIAGNOSTIC_PATH_CHARS),
  'resolution_stage':stage if type(stage) is str and stage in ('argument_conversion','relative_anchor','canonical_resolution','authority_decision') else 'UNKNOWN','frames':[],'frames_truncated':False}
 frame=None
 try:
  frame=READ_GETFRAME(1)
  for index in range(READ_DIAGNOSTIC_MAX_FRAMES):
   if frame is None:break
   if type(frame) is not READ_FRAME_TYPE:
    return {'schema':result['schema'],'status':'UNAVAILABLE'}
   code=frame.f_code
   result['frames'].append({'filename':diagnostic_text(code.co_filename,256),
    'function':diagnostic_text(code.co_name,96),'line':frame.f_lineno})
   frame=frame.f_back
  result['frames_truncated']=frame is not None
  # All leaves above are exact builtins from the interpreter; no default encoder.
  if len(json.dumps(result,ensure_ascii=True).encode('ascii'))>READ_DIAGNOSTIC_MAX_BYTES:
   return {'schema':result['schema'],'status':'SIZE_LIMIT_REFUSED'}
  return result
 finally:
  # Do not retain frames and their locals in shared evidence state.
  del frame

def safe_read_diagnostic(raw,text,resolved,stage):
 try:return read_diagnostic(raw,text,resolved,stage)
 except BaseException:
  # Diagnostic failure cannot permit the original refused read or delay exit.
  return {'schema':'friday.offline.read-refusal-metadata.v1','status':'UNAVAILABLE'}

def loader_diagnostic(event, args):
 # Data only: no repr/str/fspath, frame locals/globals, loader/symbol call or I/O.
 if type(event) is not str or event not in ('ctypes.dlopen', 'ctypes.dlsym'):
  return {'schema': 'friday.offline.loader-refusal-metadata.v1', 'status': 'UNAVAILABLE'}
 if type(LOADER_DIAGNOSTIC_PINS) is not dict:
  return {'schema': 'friday.offline.loader-refusal-metadata.v1', 'status': 'UNAVAILABLE'}
 result = {'schema': 'friday.offline.loader-refusal-metadata.v1',
  'status': 'AVAILABLE', 'event': event, 'arguments': [],
  'arguments_truncated': False, 'frames': [], 'frames_truncated': False,
  'unmapped_frames': 0, 'pin_kind': 'PREVERIFIED_EXPECTED_NOT_REHASHED_AT_DENIAL'}
 if type(args) is not tuple:
  return {'schema': result['schema'], 'status': 'UNAVAILABLE'}
 for value in args[:LOADER_DIAGNOSTIC_MAX_ARGS]:
  if value is None:
   result['arguments'].append({'kind': 'None', 'value': None})
  elif type(value) is int and value.bit_length() <= 64:
   result['arguments'].append({'kind': 'int', 'value': value})
  elif type(value) in (str, bytes):
   # Unknown loader text may contain credentials. Only exact preverified path
   # names can be copied; other text is represented without content or digest.
   pin = (LOADER_DIAGNOSTIC_PINS.get(value)
          if type(value) is str and len(value) <= READ_DIAGNOSTIC_PATH_CHARS else None)
   if (type(pin) is str and len(pin) == 64 and
       all(c in '0123456789abcdef' for c in pin)):
    row = diagnostic_text(value, READ_DIAGNOSTIC_PATH_CHARS)
    row['preverified_sha256'] = pin
    result['arguments'].append(row)
   else:
    result['arguments'].append({'kind': 'redacted_text',
      'type': 'str' if type(value) is str else 'bytes', 'length': len(value),
      'value': None, 'truncated': len(value) > READ_DIAGNOSTIC_PATH_CHARS})
  else:
   result['arguments'].append(diagnostic_text(value, READ_DIAGNOSTIC_PATH_CHARS))
 result['arguments_truncated'] = len(args) > LOADER_DIAGNOSTIC_MAX_ARGS
 frame = None
 try:
  frame = READ_GETFRAME(1)
  for index in range(LOADER_DIAGNOSTIC_MAX_SCAN):
   if frame is None: break
   if type(frame) is not READ_FRAME_TYPE:
    return {'schema': result['schema'], 'status': 'UNAVAILABLE'}
   code = frame.f_code
   pin = LOADER_DIAGNOSTIC_PINS.get(code.co_filename)
   if type(pin) is str and len(pin) == 64 and all(c in '0123456789abcdef' for c in pin):
    if len(result['frames']) >= LOADER_DIAGNOSTIC_MAX_FRAMES:
     result['frames_truncated'] = True
     break
    result['frames'].append({'filename': diagnostic_text(code.co_filename, 512),
      'function': diagnostic_text(code.co_name, 96), 'line': frame.f_lineno,
      'expected_sha256': pin})
   else:
    result['unmapped_frames'] += 1
   frame = frame.f_back
  result['frames_truncated'] = result['frames_truncated'] or frame is not None
  if len(json.dumps(result, ensure_ascii=True).encode('ascii')) > READ_DIAGNOSTIC_MAX_BYTES:
   return {'schema': result['schema'], 'status': 'SIZE_LIMIT_REFUSED'}
  return result
 finally:
  # Never retain frame/code objects (and therefore their locals) in the result.
  del frame

    'ownership_private_ancestor_required')

   require(ownership_identity(RAW['fstat'](leaf))==ownership_identity(before),'ownership_leaf_changed')

   else:raise RuntimeError('ownership_capability_present')

    'ownership_leaf_changed_after_effect')

   try:OWNERSHIP_RAW['getxattr'](leaf,'security.capability')

   else:raise RuntimeError('ownership_capability_appeared')
  except BaseException:

  finally:
   LOCAL.ownership=None

def copy_native(kind,src,dst,count,offset_src,offset_dst,caller):

def require(value, message):
    if not value:
        raise RuntimeError(message)

def string_path(value):
    value = RAW['fspath'](value) if not isinstance(value,(str,bytes)) else value
    return value.decode(FSENC,'surrogateescape') if isinstance(value,bytes) else value

def canonical(path, follow_leaf=True):
    """Resolve using captured syscalls, including symlinks before '..'."""
    require(path.startswith('/') and '\0' not in path, 'invalid_guard_path')
    todo = path.split('/')[1:]
    resolved = []
    links = 0
    while todo:
        part = todo.pop(0)
        if part in ('','.'):
            continue
        if part == '..':
            if resolved:
                resolved.pop()
            continue
        current = '/' + '/'.join(resolved+[part])
        if follow_leaf or any(x not in ('','.') for x in todo):
            try:
                s = RAW['stat'](current,follow_symlinks=False)
            except FileNotFoundError:
                s = None
            if s is not None and stat.S_ISLNK(s.st_mode):
                links += 1
                require(links <= 40, 'guard_symlink_cycle')
                target = RAW['readlink'](current)
                if target.startswith('/'):
                    resolved = []
                todo = target.split('/') + todo
                continue
        resolved.append(part)
    return '/' + '/'.join(resolved)

def writable(path):
    return any(path == root or path.startswith(root+'/') for root in WRITABLE)

def fd_path(fd):
    require(isinstance(fd,int) and not isinstance(fd,bool) and fd >= 0, 'invalid_guard_fd')
    st = RAW['fstat'](fd)
    path = RAW['readlink']('/proc/self/fd/'+str(fd))
    if path.endswith(' (deleted)') and st.st_nlink == 0:
        path = path[:-10]
    require(path.startswith('/'), 'nonfilesystem_guard_fd')
    return canonical(path), st

def path_ok(raw, dir_fd=None, follow_leaf=True):
    if isinstance(raw,int):
        path, _ = fd_path(raw)
        return writable(path)
    path = string_path(raw)
    if not path.startswith('/'):
        if dir_fd is None or dir_fd == -1:
            base = RAW['getcwd']()
        else:
            base, st = fd_path(dir_fd)
            require(stat.S_ISDIR(st.st_mode), 'guard_dirfd_not_directory')
        path = base+'/'+path
    return writable(canonical(path,follow_leaf=follow_leaf))

def check_path(event, raw, dir_fd=None, follow_leaf=True):
    try:
        allowed = path_ok(raw,dir_fd,follow_leaf)
    except BaseException:
        allowed = False
    if not allowed:
        deny(event+':outside_private_write_roots')

def check_fd(event, fd, stdout=False):
    if stdout and fd in stdout_ids:
        s = RAW['fstat'](fd)
        if (s.st_dev,s.st_ino,s.st_mode) == stdout_ids[fd]:
            return
    check_path(event,fd)

def fd_wrapper(name, stdout=False):
    original = RAW[name]
    def wrapped(fd,*args,**kwargs):
        check_fd('os.'+name,fd,stdout)
        return original(fd,*args,**kwargs)
    return wrapped
