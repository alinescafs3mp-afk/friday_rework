/** Source predicates only; actual memfd/mount/binary proof remains mandatory. */
import { it, expect } from 'vitest'
import { prepareHostControl } from '../src/host-control.ts'
import type { HostControlPolicy } from '@deepseek-ai/dsh-sandbox'
const policy: HostControlPolicy = {
  capability: 'host-control-v1', mode: 'workspace-write', workspaceRoot: '/w/task',
  temporaryRoot: '/w/private', readOnly: ['/usr/bin', '/usr/lib'],
  readWrite: ['/w/task', '/w/private'], protectedPaths: ['/home/operator'],
  requiredCapabilities: ['browser-own-cdp', 'web', 'local-chat', 'local-embedding'], statusFd: 8,
  privateNetwork: { driver: 'slirp4netns-1.3.3-frw-private-loopback', blockFd: 9,
    barrierProtocol: 'frw-success-byte-1-v1', resolverFd: 10,
    resolverSha256: 'a'.repeat(64), nameservers: ['192.168.12.2'] },
}
it.each([0, 2, 8, 9, NaN])('rejects wrong/colliding resolver fd %s', resolverFd => {
  expect(() => prepareHostControl(['true'], {...policy, privateNetwork: {
    ...policy.privateNetwork!, resolverFd,
  }}, '/usr/bin/landlock-run')).toThrow('PRIVATE_NETWORK')
})
it.each(['UNBOUND', '', 'A'.repeat(64), 'a'.repeat(63)])('rejects non-exact resolver digest %s', resolverSha256 => {
  expect(() => prepareHostControl(['true'], {...policy, privateNetwork: {
    ...policy.privateNetwork!, resolverSha256,
  }}, '/usr/bin/landlock-run')).toThrow('PRIVATE_NETWORK')
})
it('rejects legacy resolver path even when all new fields are also provided', () => {
  const privateNetwork = {...policy.privateNetwork!, resolverSource: '/w/private/resolver'}
  expect(() => prepareHostControl(['true'], {...policy, privateNetwork}, '/usr/bin/landlock-run')).toThrow('EXACT_PRIVATE_NETWORK_KEYS')
})
it('rejects an obsolete EOF-success protocol', () => {
  const privateNetwork = {...policy.privateNetwork!, barrierProtocol: 'eof-release'} as unknown as NonNullable<HostControlPolicy['privateNetwork']>
  expect(() => prepareHostControl(['true'], {...policy, privateNetwork}, '/usr/bin/landlock-run')).toThrow('PRIVATE_NETWORK')
})
it('keeps writable private temp but creates no resolver pathname alias there', () => {
  const prepared=prepareHostControl(['true'], policy, '/usr/bin/landlock-run')
  expect(prepared.argv).toContain('--ro-bind-data')
  const index=prepared.argv.indexOf('--ro-bind-data')
  expect(prepared.argv.slice(index,index+3)).toEqual(['--ro-bind-data','10','/etc/resolv.conf'])
  expect(prepared.network).toEqual(policy.privateNetwork)
  expect(prepared.requiredCapabilities).toEqual(policy.requiredCapabilities)
  expect(prepared.argv).not.toContain('/w/private/resolver')
  expect(prepared.runtimeGrant).toBe(false)
})
