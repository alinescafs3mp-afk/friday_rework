"""SOURCE ONLY. Explicit TS option for the existing finite Containment.run.

No admission/clock/lock/unit constructor. Caller supplies its original Budget,
held lock, declared task limit, unique scope and fresh owned output. No service
supervisor or discovery/retry. Actual installed behavior remains NOT_RUN.
"""
import hashlib
import json
import os
from pathlib import Path
import re
import stat
import time

LOCK = Path('<LOCAL_HOME>/.jericho/runtime/friday-quality-ui-exclusive.lock')
LOCK_ID = (2050, 9044963, 1000, 1000, 1, 0, 0o664)


def identity(st):
    return (st.st_dev, st.st_ino, st.st_uid, st.st_gid,
            st.st_nlink, st.st_size, stat.S_IMODE(st.st_mode))


def held(fd):
    if type(fd) is not int or fd < 3 or identity(os.fstat(fd)) != LOCK_ID:
        raise ValueError('canonical_existing_quality_fd_required')
    if LOCK.resolve() != LOCK or identity(LOCK.lstat()) != LOCK_ID:
        raise ValueError('canonical_quality_path_changed')
    text = Path(f'/proc/self/fdinfo/{fd}').read_text()
    if not any('FLOCK  ADVISORY  WRITE' in line and f'{LOCK_ID[1]} ' in line
               for line in text.splitlines()):
        raise ValueError('held_quality_open_description_required')
    return text


def birth(pid):
    raw = Path(f'/proc/{pid}/stat').read_text()
    return int(raw.rsplit(')', 1)[1].split()[19])


def scope_show(unit, budget, env):
    # Existing bounded foreground command, one read, no polling or stop/reset.
    from scripts.dsh_prepare import run
    out, observation = run(['/usr/bin/systemctl', '--user', 'show', unit,
        '--property=Id,LoadState,ActiveState,ControlGroup,InvocationID'], Path('/'),
        env=env, timeout=min(1, budget.check()), deadline=budget.deadline)
    rows = [line.split('=', 1) for line in out.splitlines()]
    if len(rows) != 5 or len({r[0] for r in rows}) != 5:
        raise ValueError('exact_scope_properties_required')
    return dict(rows), observation


def prepare(budget, command, status_fd, option, env):
    """Prepare argv/FDs only; Containment owns the single launch and receipt."""
    if set(option) != {'unit', 'tasks', 'quality_fd', 'run', 'ts_root', 'host_guard',
                        'mounts', 'pins', 'links', 'aggregate_cgroup'}:
        raise ValueError('exact_existing_owner_ts_option_required')
    if not re.fullmatch(r'friday-qualify-[a-z0-9][a-z0-9-]{0,190}\.scope', option['unit']):
        raise ValueError('unique_caller_owned_scope_required')
    if type(option['tasks']) is not int or not 0 < option['tasks'] < 2**31:
        raise ValueError('original_declared_task_count_required')
    fd = option['quality_fd']; held(fd)
    if fd == status_fd or not stat.S_ISFIFO(os.fstat(status_fd).st_mode):
        raise ValueError('distinct_status_and_quality_fd_required')
    run = Path(option['run']); root = Path(option['ts_root'])
    if (run != run.resolve() or root != root.resolve() or not run.name.startswith('ts24-run-')
            or run.is_relative_to(root.parent) or root.is_relative_to(run)
            or run.stat().st_uid != os.getuid() or stat.S_IMODE(run.stat().st_mode) != 0o700):
        raise ValueError('fresh_external_owned_output_required')
    clock = json.loads((run / 'clock.json').read_text())
    if (clock['original_budget_seconds'] != 180 or clock['execution_authorized'] is not False
            or clock['scope'] != 'TS24_SOURCE_OFFLINE_ONLY'
            or clock['deadline_monotonic'] != clock['started_monotonic'] + 180
            or clock['deadline_monotonic'] != budget.deadline
            or clock['boot_id'] != Path('/proc/sys/kernel/random/boot_id').read_text().strip()):
        raise ValueError('same_existing_original_ts180_required')
    if command != ['/usr/bin/python3', '-I', str(root / 'launch.py'), str(run), 'FRW_TS_CUSTODY_FD']:
        raise ValueError('exact_qualified_ts24_command_required')
    if len(os.sched_getaffinity(0)) > 2:
        raise ValueError('existing_owner_affinity_not_bound')
    for path, sha in option['pins'].items():
        budget.check()
        if hashlib.sha256(Path(path).read_bytes()).hexdigest() != sha:
            raise ValueError('qualifier_source_or_native_pin_changed')
    for path, target in option['links'].items():
        budget.check()
        if os.readlink(path) != target:
            raise ValueError('qualifier_link_changed')
    # The existing root-owned aggregate ancestor must already cover the finite
    # coordinator AND the future scope. A sibling scope plus per-process affinity
    # is not aggregate2CPU2GiB. No shared slice mutation or new slice is done here.
    parent_line = Path('/proc/self/cgroup').read_text().strip()
    if not parent_line.startswith('0::/') or '\n' in parent_line:
        raise ValueError('existing_aggregate_parent_cgroup_required')
    aggregate = Path(option['aggregate_cgroup'])
    own = Path('/sys/fs/cgroup') / parent_line[4:]
    if (aggregate != aggregate.resolve() or aggregate == Path('/sys/fs/cgroup')
            or not own.is_relative_to(aggregate)):
        raise ValueError('existing_owner_aggregate_ancestor_unbound')
    aggregate_caps = {k: (aggregate / k).read_text().strip()
        for k in ('memory.max','memory.swap.max','cpu.max','pids.max')}
    quota, period = aggregate_caps['cpu.max'].split()
    if (not quota.isdecimal() or not period.isdecimal() or not 0 < int(quota) <= 2 * int(period)
            or not aggregate_caps['memory.max'].isdecimal()
            or not 0 < int(aggregate_caps['memory.max']) <= 2 << 30
            or aggregate_caps['memory.swap.max'] != '0'
            or aggregate_caps['pids.max'] != str(option['tasks'])):
        raise ValueError('existing_whole_owner_aggregate_envelope_unbound')
    if re.fullmatch(r'friday-qualify-[a-z0-9][a-z0-9-]{0,190}\.slice', aggregate.name) is None:
        raise ValueError('existing_per_job_aggregate_slice_required')
    slice_props, slice_query = scope_show(aggregate.name, budget, env)
    if (slice_props['Id'] != aggregate.name or slice_props['LoadState'] != 'loaded'
            or slice_props['ActiveState'] != 'active'
            or Path('/sys/fs/cgroup') / slice_props['ControlGroup'].lstrip('/') != aggregate):
        raise ValueError('existing_aggregate_slice_not_actually_bound')
    # Pre-existing names refuse; never stop or reclaim somebody else's scope.
    before, query = scope_show(option['unit'], budget, env)
    if before['Id'] != option['unit'] or before['LoadState'] != 'not-found' or before['ControlGroup']:
        raise ValueError('scope_name_not_fresh')
    reader, writer = os.pipe2(os.O_NONBLOCK | os.O_CLOEXEC)
    context = dict(option, reader=reader, writer=writer, owner_pid=os.getpid(),
                   owner_birth=birth(os.getpid()), deadline=budget.deadline,
                   boot_id=clock['boot_id'], before_scope=before, before_query=query,
                   quality_before=held(fd), env=env, aggregate_caps=aggregate_caps,
                   aggregate_slice_before=slice_props, aggregate_query=slice_query,
                   aggregate_identity=[aggregate.stat().st_dev, aggregate.stat().st_ino])
    envelope = {key: context[key] for key in ('unit', 'tasks', 'quality_fd',
                'run', 'ts_root', 'mounts', 'owner_pid', 'owner_birth', 'deadline', 'boot_id',
                'aggregate_cgroup', 'aggregate_caps', 'aggregate_identity')}
    envelope.update(memory_bytes=2 << 30, cpu_percent=200, scope_observation_fd=writer,
                    native_status_fd=status_fd)
    args = ['/usr/bin/systemd-run', '--user', '--scope', '--quiet', '--no-ask-password',
        '--expand-environment=no', '--unit=' + option['unit'],
        '--slice=' + aggregate.name,
        '--property=MemoryMax=2147483648', '--property=MemorySwapMax=0',
        '--property=CPUQuota=200%', '--property=TasksMax=' + str(option['tasks']),
        '--property=MemoryOOMGroup=yes', '--property=KillMode=control-group',
        '--property=SendSIGKILL=yes',
        '--property=RuntimeMaxSec=' + str(min(90, budget.check(reserve=2))) + 's',
        '/usr/bin/python3', '-I', option['host_guard'], json.dumps(envelope), *command]
    return args, (status_fd, fd, writer), context


def finish(budget, context, observation, native_data):
    """One post-exit exact ownership join; uncertain facts retain stop evidence."""
    from scripts.dsh_prepare import StopUnconfirmed
    receipt = {'state': 'STOP_UNCONFIRMED', 'requested_unit': context['unit'],
               'native_status_sha256': hashlib.sha256(native_data).hexdigest(),
               'command': observation, 'runtimeGrant': False,
               'aggregate_before': {'path': context['aggregate_cgroup'],
                 'identity': context['aggregate_identity'], 'caps': context['aggregate_caps']}}
    try:
        os.close(context['writer']); context['writer'] = None
        try: raw = os.read(context['reader'], 4097)
        except BlockingIOError: raw = b''
        if not raw or len(raw) > 4096: raise ValueError('host_scope_observation_missing')
        host = json.loads(raw)
        receipt['host_before'] = host
        receipt['host_scope_observation_sha256'] = hashlib.sha256(raw).hexdigest()
        if (host['unit'] != context['unit'] or host['owner_pid'] != context['owner_pid']
                or host['owner_birth'] != context['owner_birth']
                or host['deadline'] != budget.deadline
                or host['boot_id'] != context['boot_id']
                or not re.fullmatch('[0-9a-f]{32}', host['invocation'])):
            raise ValueError('host_scope_identity_changed')
        budget.check()
        props, query = scope_show(context['unit'], budget, context['env'])
        receipt['scope_after'] = props; receipt['after_query'] = query
        cg = Path(host['cgroup'])
        if cg.exists():
            if [cg.stat().st_dev, cg.stat().st_ino] != host['cgroup_identity']:
                raise ValueError('owned_scope_inode_reused')
            events = dict(line.split() for line in (cg / 'cgroup.events').read_text().splitlines())
            receipt['cgroup_after'] = events
            if events.get('populated') != '0': raise ValueError('owned_scope_still_populated')
        else: receipt['cgroup_after'] = 'ABSENT'
        if props['LoadState'] != 'not-found':
            if (props['Id'] != host['unit'] or props['InvocationID'] != host['invocation']
                    or props['ActiveState'] not in ('inactive', 'failed')
                    or props['ControlGroup'] not in ('', host['control_group'])):
                raise ValueError('owned_scope_not_terminal')
        elif props['ControlGroup'] or cg.exists():
            raise ValueError('scope_missing_but_group_retained')
        pid = observation.get('pid')
        # Popen was retained/reaped by run. Never reopen a PID for signalling.
        if (not observation.get('namespace_init_exit_verified') or not observation.get('reaped')
                or type(pid) is not int or Path(f'/proc/{pid}').exists()):
            raise ValueError('original_native_custody_not_closed')
        try: os.killpg(pid, 0)
        except ProcessLookupError: pass
        else: raise ValueError('owned_foreground_group_retained')
        aggregate = Path(context['aggregate_cgroup'])
        aggregate_after = {k: (aggregate / k).read_text().strip() for k in context['aggregate_caps']}
        if ([aggregate.stat().st_dev, aggregate.stat().st_ino] != context['aggregate_identity']
                or aggregate_after != context['aggregate_caps']):
            raise ValueError('aggregate_owner_envelope_changed')
        receipt['aggregate_after'] = aggregate_after
        receipt['quality_after'] = held(context['quality_fd'])
        receipt['state'] = 'OWNED_NAMESPACE_AND_SCOPE_TERMINAL_LOCK_STILL_HELD'
        receipt['elapsed_terminal_observation'] = time.monotonic()
        budget.check()
    except BaseException as exc:
        receipt['failure'] = type(exc).__name__ + ':' + str(exc)
        stopped = StopUnconfirmed('STOP_UNCONFIRMED: exact TS owner scope join')
        stopped.owner_receipt = receipt
        raise stopped from exc
    finally:
        if context['writer'] is not None: os.close(context['writer'])
        os.close(context['reader'])
    # The root caller returns its lock only AFTER this receipt and its own final
    # evidence. This module never unlocks it and cannot prove that later return.
    return receipt
