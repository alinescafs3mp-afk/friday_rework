import json

from pathlib import Path
import re
import stat

p = json.loads(sys.argv[1])
assert time.monotonic()<p['deadline']
assert Path('/proc/sys/kernel/random/boot_id').read_text().strip()==p['boot_id']
line=Path('/proc/self/cgroup').read_text().strip()
assert line.startswith('0::/') and '\n' not in line
group=line[3:]; assert group.endswith('/'+p['unit'])
root=Path('/sys/fs/cgroup')/group.lstrip('/')
assert root.resolve()==root
caps={k:(root/k).read_text().strip() for k in ('memory.max','memory.swap.max','cpu.max','pids.max')}
assert caps['memory.max']==str(p['memory_bytes']) and caps['memory.swap.max']=='0'
assert caps['pids.max']==str(p['tasks'])
q,t=caps['cpu.max'].split()
assert q.isdecimal() and t.isdecimal() and int(t)>0 and int(q)*100==p['cpu_percent']*int(t)
assert time.monotonic()<p['deadline']


def original():
    assert time.monotonic() + 2 < p['deadline']


def held(fd):
    st = os.fstat(fd)
    assert (st.st_dev, st.st_ino, st.st_uid, st.st_gid, st.st_nlink,
            st.st_size, stat.S_IMODE(st.st_mode)) == (2050, 9044963, 1000, 1000, 1, 0, 0o664)
    text = Path(f'/proc/self/fdinfo/{fd}').read_text()
    assert any('FLOCK  ADVISORY  WRITE' in line and f'{st.st_ino} ' in line for line in text.splitlines())
    assert any('FLOCK  ADVISORY  WRITE' in line and f' {p["owner_pid"]} ' in line
               and f'{st.st_ino} ' in line for line in Path('/proc/locks').read_text().splitlines())
    return text


def sealed(name, raw):
    fd = os.memfd_create(name, os.MFD_ALLOW_SEALING | os.MFD_CLOEXEC)
    try:
        view = memoryview(raw)
        while view:
            n = os.write(fd, view)
            if n <= 0: raise ValueError('sealed_observation_short_write')
            view = view[n:]
        # Existing Linux ABI policy: these constants are not exported by the
        # selected CPython. Never modify fcntl globally or accept a writable FD.
        assert sys.platform == 'linux' and os.uname().machine in ('x86_64', 'aarch64')
        fcntl.fcntl(fd, 1033, 15)
        assert fcntl.fcntl(fd, 1034) == 15 and os.fstat(fd).st_nlink == 0
        os.lseek(fd, 0, os.SEEK_SET)
        os.set_inheritable(fd, True)
        return fd
    except BaseException:
        os.close(fd)
        raise


original()
aggregate = Path(p['aggregate_cgroup'])
assert root.is_relative_to(aggregate) and root != aggregate
assert [aggregate.stat().st_dev, aggregate.stat().st_ino] == p['aggregate_identity']
assert {k: (aggregate / k).read_text().strip() for k in p['aggregate_caps']} == p['aggregate_caps']
owner_stat = Path(f'/proc/{p["owner_pid"]}/stat').read_text()
assert int(owner_stat.rsplit(')', 1)[1].split()[19]) == p['owner_birth']
assert os.getpid() == os.getpgrp() and os.getsid(0) == os.getpid()
assert 0 < len(os.sched_getaffinity(0)) <= 2
lock = held(p['quality_fd'])
assert (root / 'memory.oom.group').read_text().strip() == '1'
query = subprocess.run(['/usr/bin/systemctl', '--user', 'show', p['unit'],
    '--property=Id,LoadState,ActiveState,ControlGroup,InvocationID'],
    check=True, capture_output=True, text=True, timeout=min(1, p['deadline'] - time.monotonic()))
props = dict(line.split('=', 1) for line in query.stdout.strip().splitlines())
assert props['Id'] == p['unit'] and props['LoadState'] == 'loaded'
assert props['ActiveState'] == 'active' and props['ControlGroup'] == group
assert re.fullmatch('[0-9a-f]{32}', props['InvocationID'])
snapshot = {key: p[key] for key in ('unit', 'tasks', 'owner_pid', 'owner_birth',
             'boot_id', 'deadline', 'quality_fd')}
snapshot.update(invocation=props['InvocationID'], control_group=group,
    cgroup=str(root), cgroup_identity=[root.stat().st_dev, root.stat().st_ino],
    caps=caps, affinity=sorted(os.sched_getaffinity(0)),
    host_namespace_inodes={n: Path('/proc/self/ns/' + n).stat().st_ino for n in ('pid', 'mnt', 'net')},
    scope_pid=os.getpid(), scope_birth=int(Path('/proc/self/stat').read_text().rsplit(')', 1)[1].split()[19]),
    quality_fdinfo=lock, aggregate_cgroup=p['aggregate_cgroup'],
    aggregate_identity=p['aggregate_identity'], aggregate_caps=p['aggregate_caps'], runtimeGrant=False)
metadata = json.dumps(snapshot, sort_keys=True).encode()
assert len(metadata) <= 4096
assert os.write(p['scope_observation_fd'], metadata) == len(metadata)

custody = sealed('frw-ts-host-scope-observation', metadata)

mount_ref = p['mounts']; raw = Path(mount_ref['path']).read_bytes()
assert hashlib.sha256(raw).hexdigest() == mount_ref['sha256']

args = ['--json-status-fd', str(p['native_status_fd']), '--unshare-pid',
    '--die-with-parent', '--new-session', '--as-pid-1', '--unshare-net',
    '--unshare-ipc', '--unshare-uts', '--tmpfs', '/']

for item in destinations:

    assert q.is_absolute() and '..' not in q.parts
    directories.update(str(parent) for parent in q.parents if str(parent) != '/')
for directory in sorted(directories, key=lambda value: (value.count('/'), value)):

    if directory not in links: args += ['--dir', directory]
for source in ro_roots: args += ['--ro-bind', source, source]

        args += ['--ro-bind', source, source]
for link, target in sorted(links.items()): args += ['--symlink', target, link]

         '--dev', '/dev', '--proc', '/proc', '--remount-ro', '/', '--']
