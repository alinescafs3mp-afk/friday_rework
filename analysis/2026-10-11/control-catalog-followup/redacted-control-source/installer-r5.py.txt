class Refusal(RuntimeError):

def json_exact(left, right):
    """No bool/int/float coercion at admission, evidence or custody boundaries."""
    if type(left) is not type(right):
        return False
    if type(left) is dict:
        return (left.keys() == right.keys()
                and all(type(key) is str for key in left)
                and all(json_exact(left[key], right[key]) for key in left))
    if type(left) in (list, tuple):
        return len(left) == len(right) and all(json_exact(a, b) for a, b in zip(left, right))
    if type(left) is float:
        import math
        return math.isfinite(left) and math.isfinite(right) and left == right
    return left == right


def load_json(raw):
    """Reject duplicate members and every non-finite numeric conversion."""
    def object_pairs(pairs):
        out = {}
        for key, value in pairs:
            if key in out:
                raise Refusal('DUPLICATE_JSON_MEMBER:' + key)
            out[key] = value
        return out
    def nonfinite(value):
        raise Refusal('NONFINITE_JSON_VALUE:' + value)
    def finite_float(value):
        import math
        result = float(value)
        if not math.isfinite(result):
            raise Refusal('NONFINITE_JSON_VALUE:' + value)
        return result
    return json.loads(raw, object_pairs_hook=object_pairs,
                      parse_constant=nonfinite, parse_float=finite_float)


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def now():
    return datetime.datetime.now(datetime.timezone(datetime.timedelta(hours=3))).isoformat()


def signature(info):
    return [info.st_dev, info.st_ino, info.st_uid, info.st_gid, info.st_nlink,
            stat.S_IMODE(info.st_mode), info.st_size, info.st_mtime_ns, info.st_ctime_ns]


@contextlib.contextmanager
def directory(path):
    """Walk every component with no-follow, pin directory FD, reject lexical aliases."""
    raw = os.fspath(path)
    if not raw.startswith('/') or (not json_exact(raw, str(Path(raw)))) or '..' in raw.split('/'):
        raise Refusal('NONCANONICAL_PATH')
    fd = os.open('/', os.O_RDONLY | os.O_DIRECTORY | os.O_CLOEXEC)
    try:
        for part in Path(raw).parts[1:]:
            child = os.open(part, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW | os.O_CLOEXEC, dir_fd=fd)
            os.close(fd)
            fd = child
            info = os.fstat(fd)
            if info.st_uid not in (0, os.getuid()) or info.st_mode & 0o022:
                raise Refusal('UNSAFE_DIRECTORY:' + raw)
        yield fd
    finally:
        os.close(fd)


def checked_info(info, path):
    if (not stat.S_ISREG(info.st_mode) or (not json_exact(info.st_uid, os.getuid()))
            or (not json_exact(info.st_nlink, 1)) or info.st_mode & (0o022 | stat.S_ISUID | stat.S_ISGID)):
        raise Refusal('UNSAFE_REGULAR_FILE:' + str(path))


def read_file(path, expected=None):
    path = Path(path)
    with directory(path.parent) as parent:
        fd = os.open(path.name, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK | os.O_CLOEXEC, dir_fd=parent)
        try:
            before = os.fstat(fd)
            checked_info(before, path)
            if before.st_size > 16 * 1024 * 1024:
                raise Refusal('FILE_TOO_LARGE')
            chunks = []
            while True:
                part = os.read(fd, 65536)
                if not part:
                    break
                chunks.append(part)
                if sum(map(len, chunks)) > 16 * 1024 * 1024:
                    raise Refusal('FILE_GREW_TOO_LARGE')
            after = os.fstat(fd)
            named = os.stat(path.name, dir_fd=parent, follow_symlinks=False)
            if (not json_exact(signature(before), signature(after))) or (not json_exact(signature(after), signature(named))):
                raise Refusal('FILE_DRIFT:' + str(path))
            raw = b''.join(chunks)
            digest = sha(raw)
            if expected is not None and (not json_exact(digest, expected)):
                raise Refusal('SHA_CAS_MISMATCH:' + str(path))
            return raw, {'path': str(path), 'sha256': digest, 'signature': signature(after),
                         'mode': stat.S_IMODE(after.st_mode), 'bytes': len(raw)}
        finally:
            os.close(fd)


class Ops:
    """Tests inject faults by overriding these methods, never through the real CLI."""
    def fsync(self, fd, boundary):
        os.fsync(fd)

    def replace(self, source, target, parent, boundary):
        os.replace(source, target, src_dir_fd=parent, dst_dir_fd=parent)


def write_all(fd, raw):
    view = memoryview(raw)
    while view:
        size = os.write(fd, view)
        if size <= 0:
            raise OSError('short write without progress')
        view = view[size:]


def exclusive_file(parent, name, raw, mode, ops, boundary, created=None):
    fd = os.open(name, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW | os.O_CLOEXEC, mode, dir_fd=parent)
    try:
        if created:
            created(signature(os.fstat(fd)))
        os.fchmod(fd, mode)
        write_all(fd, raw)
        # Capture original bytes/inode/mode/nlink BEFORE callbacks or fsync.
        original = {'path': str(Path(os.readlink('/proc/self/fd/' + str(parent))) / name),
                    'sha256': sha(raw), 'signature': signature(os.fstat(fd)),
                    'mode': mode, 'bytes': len(raw)}
        checked_info(os.fstat(fd), original['path'])
        ops.fsync(fd, boundary + ':file')
    finally:
        os.close(fd)
    ops.fsync(parent, boundary + ':directory')
    _, visible = read_file(original['path'], original['sha256'])
    if (not json_exact(visible, original)):
        raise Refusal('EXCLUSIVELY_CREATED_FILE_CUSTODY_DRIFT:' + original['path'])
    return original


@contextlib.contextmanager
def locks(paths):
    """Existing ordinary non-writer locks only. Never create, chmod or truncate."""
    held = []
    try:
        for path in paths:
            _, identity = read_file(path)
            with directory(Path(path).parent) as parent:
                fd = os.open(Path(path).name, os.O_RDONLY | os.O_NOFOLLOW | os.O_CLOEXEC, dir_fd=parent)
            try:
                if (not json_exact(signature(os.fstat(fd)), identity['signature'])):
                    raise Refusal('LOCK_IDENTITY_DRIFT')
                fcntl.flock(fd, fcntl.LOCK_EX | fcntl.LOCK_NB)
            except BaseException:
                os.close(fd)
                raise
            held.append((path, fd, identity))
        yield held
    finally:
        for _, fd, _ in reversed(held):
            os.close(fd)


def pin_identity(pin):
    path = Path(pin['path'])
    if json_exact(pin.get('kind'), 'symlink'):
        with directory(path.parent) as parent:
            info = os.stat(path.name, dir_fd=parent, follow_symlinks=False)
            if not stat.S_ISLNK(info.st_mode) or (not json_exact(info.st_uid, os.getuid())):
                raise Refusal('UNSAFE_RELEASE_LINK')
            target = os.readlink(path.name, dir_fd=parent)
            if (not json_exact(target, pin['link_target'])) or (not json_exact(str(path.resolve(strict=True)), pin['resolved_target'])):
                raise Refusal('CURRENT_RELEASE_LINK_DRIFT')
            return {'path': str(path), 'signature': signature(info), 'link_target': target}
    # Streaming pinned dependency read: release binaries exceed target-file limit.
    with directory(path.parent) as parent:
        fd = os.open(path.name, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK | os.O_CLOEXEC, dir_fd=parent)
        try:
            before = os.fstat(fd); checked_info(before, path)
            digest = hashlib.sha256(); size = 0
            while True:
                chunk = os.read(fd, 1024 * 1024)
                if not chunk: break
                size += len(chunk)
                if size > 512 * 1024 * 1024: raise Refusal('PIN_TOO_LARGE')
                digest.update(chunk)
            after = os.fstat(fd); named = os.stat(path.name, dir_fd=parent, follow_symlinks=False)
            if (not json_exact(signature(before), signature(after))) or (not json_exact(signature(after), signature(named))):
                raise Refusal('PIN_IDENTITY_DRIFT:' + str(path))
            if (not json_exact(digest.hexdigest(), pin['sha256'])): raise Refusal('SHA_CAS_MISMATCH:' + str(path))
            if any(k in pin and (not json_exact(pin[k], v)) for k, v in
                   [('mode', stat.S_IMODE(after.st_mode)), ('bytes', size), ('uid', after.st_uid), ('nlink', after.st_nlink)]):
                raise Refusal('PIN_METADATA_MISMATCH:' + str(path))
            if 'signature' in pin and (not json_exact(signature(after), pin['signature'])):
                raise Refusal('PIN_ORIGINAL_CUSTODY_DRIFT:' + str(path))
            return {'path': str(path), 'sha256': digest.hexdigest(), 'signature': signature(after)}
        finally: os.close(fd)


def verify_pins(pins, identities=None):
    observed = [pin_identity(pin) for pin in pins]
    if identities is not None and (not json_exact(observed, identities)):
        raise Refusal('UNCHANGED_DEPENDENCY_IDENTITY_DRIFT')
    return observed


def stable_directory_identity(path):
    with directory(path) as fd:
        s = os.fstat(fd)
        return [s.st_dev, s.st_ino, s.st_uid, s.st_gid, stat.S_IMODE(s.st_mode)]


def parent_paths(paths):
    return sorted({str(parent) for p in paths for parent in Path(p).parents if (not json_exact(str(parent), '/'))})


def verify_release_layout(audit):
    tree = audit['release_tree']; root = Path(tree['root'])
    with directory(root): pass
    actual = []
    for path in sorted(root.rglob('*'), key=lambda p: p.as_posix()):
        info = path.lstat()
        if (not json_exact(info.st_uid, os.getuid())) or info.st_mode & 0o022 or path.is_symlink():
            raise Refusal('RELEASE_TREE_UNSAFE_MEMBER')
        kind = 'directory' if stat.S_ISDIR(info.st_mode) else 'file'
        if json_exact(kind, 'file'): checked_info(info, path)
        actual.append({'path': str(path), 'kind': kind, 'mode': stat.S_IMODE(info.st_mode)})
    if (not json_exact(actual, tree['members'])): raise Refusal('RELEASE_TREE_LAYOUT_DRIFT')


def transaction(specs, journal, lock_paths, pins=(), precondition=None, ops=None, invariant=None):
    """One-shot all-backups-first CAS; per-file atomic replacement, explicit uncertainty."""
    ops = ops or Ops()
    journal = Path(journal)
    state = {'schema': 'friday.native-send-r2-local-transaction.v1', 'at_msk': now(),
             'state': 'NOT_STARTED', 'replace_attempted': [], 'replaced': [],
             'native_delivery': 'NOT_RUN', 'automatic_rollback': False,
             'durable_commit_completed': False, 'operation_id': uuid.uuid4().hex}
    staging = []
    sequence = 0
    journal_fd = None
    custody = []
    journal_custody = []

    def verify_custody():
        for original in custody:
            _, visible = read_file(original['path'], original['sha256'])
            if (not json_exact(visible, original)):
                raise Refusal('ORIGINAL_BACKUP_OR_JOURNAL_CUSTODY_DRIFT:' + original['path'])

    def save(label):
        nonlocal sequence
        sequence += 1
        state['at_msk'] = now()
        state['prior_journal_custody'] = list(journal_custody)
        raw = (json.dumps(state, indent=2, sort_keys=True) + '\n').encode()
        name = f'{sequence:03d}-{label}.json'
        original = exclusive_file(journal_fd, name, raw, 0o600, ops, 'journal:' + label)
        custody.append(original)
        journal_custody.append(original)
        verify_custody()
        return {'path': str(journal / name), 'sha256': sha(raw)}

    try:
        with locks(lock_paths) as held, contextlib.ExitStack() as stack:
            pin_identities = verify_pins(pins)
            protected_paths = [v['path'] for v in pins] + list(lock_paths) + [v[k] for v in specs for k in ('target', 'source')] + [journal]
            parent_identities = {p: stable_directory_identity(p) for p in parent_paths(protected_paths)}
            def dependencies_at_effect():
                verify_custody()
                verify_pins(pins, pin_identities)
                if invariant: invariant()
                for p, identity in parent_identities.items():
                    if (not json_exact(stable_directory_identity(p), identity)): raise Refusal('PROTECTED_PARENT_DRIFT:' + p)
                for path, fd, identity in held:
                    _, current = read_file(path)
                    if (not json_exact(current['signature'], identity['signature'])) or (not json_exact(signature(os.fstat(fd)), identity['signature'])):
                        raise Refusal('LOCK_PATH_DRIFT')
            if precondition:
                precondition()
            parents = {}
            for spec in specs:
                parent_path = str(Path(spec['target']).parent)
                if parent_path not in parents:
                    parent = stack.enter_context(directory(parent_path))
                    parents[parent_path] = (parent, signature(os.fstat(parent)))
            with directory(journal.parent) as parent:
                os.mkdir(journal.name, mode=0o700, dir_fd=parent)
                ops.fsync(parent, 'journal-create-directory')
            journal_fd = stack.enter_context(directory(journal))
            parent_identities[str(journal)] = stable_directory_identity(journal)
            images = []
            for number, spec in enumerate(specs):
                dependencies_at_effect()
                before_raw, before = read_file(spec['target'], spec['before_sha'])
                after_raw, source = read_file(spec['source'], spec['after_sha'])
                desired_mode = spec.get('mode', before['mode'])
                backup = f'{number:02d}-before.bin'
                dependencies_at_effect()
                backup_identity = exclusive_file(journal_fd, backup, before_raw, 0o600, ops, f'backup:{number}')
                custody.append(backup_identity)
                verify_custody()
                images.append({'target': str(spec['target']), 'before': before, 'source': source,
                               'after_sha': spec['after_sha'], 'after_mode': desired_mode,
                               'backup': str(journal / backup), 'backup_identity': backup_identity})
                parent = parents[str(Path(spec['target']).parent)][0]
                name = '.' + Path(spec['target']).name + '.r2-' + uuid.uuid4().hex
                parent_path = str(Path(spec['target']).parent)
                dependencies_at_effect()
                exclusive_file(parent, name, after_raw, desired_mode, ops, f'stage:{number}',
                               created=lambda identity, p=parent, n=name, d=parent_path:
                               staging.append((p, n, d, identity[:2])))
                _, staged = read_file(Path(spec['target']).parent / name, spec['after_sha'])
                if (not json_exact(staged['signature'][:2], staging[-1][3])):
                    raise Refusal('EXCLUSIVELY_CREATED_STAGE_INODE_DRIFT')
                images[-1]['staged_signature'] = staged['signature']
            state.update(state='PREPARED_DURABLE_BEFORE_IMAGES', images=images)
            save('prepared')
            if precondition:
                precondition()
            def effect_boundary():
                # No journal/fsync/write lies between this guard and replace.
                dependencies_at_effect()
                for image in images:
                    _, source = read_file(image['source']['path'], image['source']['sha256'])
                    if (not json_exact(source['signature'], image['source']['signature'])):
                        raise Refusal('SOURCE_IDENTITY_DRIFT')
                    expected = image.get('installed', image['before'])
                    _, live = read_file(image['target'], expected['sha256'])
                    if (not json_exact(live['signature'], expected['signature'])):
                        raise Refusal('PER_FILE_IDENTITY_CAS_MISMATCH')
                for index, image in enumerate(images):
                    if 'installed' in image: continue
                    parent, name, _, _ = staging[index]
                    _, staged = read_file(Path(image['target']).parent / name, image['after_sha'])
                    if (not json_exact(staged['signature'], image['staged_signature'])) or (not json_exact(staged['mode'], image['after_mode'])):
                        raise Refusal('STAGING_IDENTITY_CAS_MISMATCH')
            # Preserve the pre-effect refusal state and original43 semantics.
            effect_boundary()
            for number, image in enumerate(images):
                parent, name, _, _ = staging[number]
                state['replace_attempted'].append(image['target'])
                state['state'] = 'REPLACE_INTENT_DURABLE'
                save(f'before-replace-{number}')
                effect_boundary()
                ops.replace(name, Path(image['target']).name, parent, f'replace:{number}')
                state['replaced'].append(image['target'])
                state['state'] = 'VISIBLE_REPLACEMENT_PENDING_DIRSYNC'
                ops.fsync(parent, f'replace:{number}:directory')
                _, visible = read_file(image['target'], image['after_sha'])
                if (not json_exact(visible['mode'], image['after_mode'])):
                    raise Refusal('POSTCONDITION_MODE_MISMATCH')
                image['installed'] = visible
                save(f'after-replace-{number}')
            effect_boundary()
            for image in images:
                _, final = read_file(image['target'], image['after_sha'])
                if (not json_exact(final['signature'], image['installed']['signature'])):
                    raise Refusal('FINAL_POSTCONDITION_DRIFT')
            state['state'] = 'COMMITTED_LOCAL_FILES_ONLY'
            committed_ref = save('committed')
            effect_boundary()
            # This acknowledgement exists only after both commit fsync calls returned.
            # It is absent from the visible committed file, so that file is insufficient.
            state['durable_commit_completed'] = True
            state['commit_record_ref'] = committed_ref
            state['commit_record_identity'] = journal_custody[-1]
            return state
    except BaseException as exc:
        state['cause'] = {'type': type(exc).__name__, 'message': str(exc)}
        state['state'] = 'REFUSED_NO_REPLACE' if not state['replace_attempted'] else 'STOP_UNCERTAIN_RECONCILE_REQUIRED'
        observations = []
        for spec in specs:
            try:
                _, observed = read_file(spec['target'])
                observations.append(observed)
            except BaseException as observe_exc:
                observations.append({'path': str(spec['target']), 'observation_error': type(observe_exc).__name__})
        state['visible_target_observations'] = observations
        # The original cause wins even if the best-effort failure receipt cannot persist.
        if journal_fd is not None:
            try:
                # ExitStack has closed this FD; reopen exact owned journal for causal receipt.
                with directory(journal) as reopened:
                    journal_fd = reopened
                    save('stopped')
            except BaseException as evidence_exc:
                state['failure_receipt_error'] = {'type': type(evidence_exc).__name__, 'message': str(evidence_exc)}
        return state
    finally:
        for _, name, parent_path, owned_inode in staging:
            # Parent FDs are closed now. Remove only the inode we exclusively created.
            candidate = Path(parent_path) / name
            try:
                with directory(candidate.parent) as cleanup_parent:
                    info = os.stat(name, dir_fd=cleanup_parent, follow_symlinks=False)
                    checked_info(info, candidate)
                    if (not json_exact(signature(info)[:2], owned_inode)):
                        raise Refusal('FOREIGN_STAGING_INODE_NOT_REMOVED')
                    os.unlink(name, dir_fd=cleanup_parent)
                    ops.fsync(cleanup_parent, 'cleanup:directory')
            except FileNotFoundError:
                continue
            except BaseException as cleanup_exc:
                state.setdefault('cleanup_errors', []).append(type(cleanup_exc).__name__ + ':' + str(cleanup_exc))

        raise Refusal('FRESH_PROOF_CONTROL_SNAPSHOT_MISMATCH')
    if (not json_exact(proof.get('account_input_sha256'), before['account_input_sha256'])) or (not json_exact(proof.get('role_profile_sha256'), before['role_profile_sha256'])):
        raise Refusal('LIVE_CURRENT_ACCOUNT_OR_ROLE_BINDING_MISMATCH')
    if json_exact(proof.get('validated_at'), audit['expected_baseline']['validated_at']) or not proof.get('validated_at'):
        raise Refusal('FRESH_PROOF_VALIDATION_TIME_REQUIRED')
    # Only the exact genuine receipt is accepted. The typed manifest explains four
    # exact differences; it never grants a blanket account/config exception.
    if (not json_exact(proof, audit['expected_proof'])):
        raise Refusal('FRESH_PROOF_OTHER_GATE_INPUTS_CHANGED')
    verify_pins(pins)
    verify_release_layout(audit)
    return [*pins, *evidence_pins, review,
            {'path': str(args.acceptance_file), 'sha256': args.acceptance_sha}]


def confirmed_forward(args, attempt, committed_path, committed):
    if not all((args.forward_confirmation_file, args.forward_confirmation_sha)):
        raise Refusal('DURABLE_FORWARD_RETURN_AND_ROOT_CONFIRMATION_REQUIRED')
    raw, confirmation_identity = read_file(args.forward_confirmation_file, args.forward_confirmation_sha)
    confirmed = load_json(raw)
    required = {'schema': 'friday.native-send-r2-forward-confirmed.v1',
                'status': 'FORWARD_COMMIT_CONFIRMED', 'owner_thread': OWNER,
                'operation': 'install', 'operation_id': committed['operation_id'],
                'program_sha256': args.expected_program_sha, 'forward_exit_code': 0,
                'commit_ref': {'path': str(committed_path), 'sha256': args.expected_commit_sha},
                'installed_proof_sha256': args.proof_sha}
    if any((not json_exact(confirmed.get(k), v)) for k,v in required.items()):
        raise Refusal('FORWARD_CONFIRMATION_BINDING_MISMATCH')
    records = sorted(attempt.glob('*.json'))
    if not records or (not json_exact(records[-1], committed_path)) or any('stopped' in p.name for p in records):
        raise Refusal('ROLLBACK_REFUSES_UNCERTAIN_OR_STOPPED_FORWARD_HISTORY')
    observed = []
    for number, path in enumerate(records, 1):
        raw, identity = read_file(path)
        record = load_json(raw)
        if not path.name.startswith(f'{number:03d}-') or (not json_exact(record.get('operation_id'), committed['operation_id'])):
            raise Refusal('FORWARD_HISTORY_SEQUENCE_MISMATCH')
        if 'cause' in record or record.get('state') in ('STOP_UNCERTAIN_RECONCILE_REQUIRED', 'REFUSED_NO_REPLACE'):
            raise Refusal('ROLLBACK_REFUSES_UNCERTAIN_OR_STOPPED_FORWARD_HISTORY')
        observed.append({'path': str(path), 'sha256': identity['sha256']})
    if (not json_exact(confirmed.get('journal_records'), observed)):
        raise Refusal('FORWARD_CONFIRMED_JOURNAL_MISMATCH')
    result_ref = confirmed.get('returned_result_ref')
    if not isinstance(result_ref, dict) or (not json_exact(set(result_ref), {'path','sha256'})):
        raise Refusal('SUCCESSFUL_FORWARD_RETURN_EVIDENCE_REQUIRED')
    raw, return_identity = read_file(result_ref['path'], result_ref['sha256']); returned = load_json(raw)
    if ((not json_exact(returned.get('state'), 'COMMITTED_LOCAL_FILES_ONLY')) or returned.get('durable_commit_completed') is not True
            or (not json_exact(returned.get('commit_record_ref'), required['commit_ref']))
            or any(k in returned for k in ('cause','failure_receipt_error','cleanup_errors'))
            or (not json_exact({k:v for k,v in returned.items() if k not in ('durable_commit_completed','commit_record_ref','commit_record_identity')}, {k:v for k,v in committed.items() if k != 'durable_commit_completed'}))):
        raise Refusal('ROLLBACK_REFUSES_UNCERTAIN_FORWARD_RETURN')
    originals = committed.get('prior_journal_custody')
    commit_identity = returned.get('commit_record_identity')
    if not isinstance(originals, list) or (not json_exact(len(originals), len(records) - 1)) or not isinstance(commit_identity, dict):
        raise Refusal('ORIGINAL_FORWARD_JOURNAL_CUSTODY_REQUIRED')
    original_records = [*originals, commit_identity]
    if (not json_exact([v['path'] for v in original_records], [str(v) for v in records])):
        raise Refusal('ORIGINAL_FORWARD_JOURNAL_SET_MISMATCH')
    for original in original_records:
        _, visible = read_file(original['path'], original['sha256'])
        if (not json_exact(visible, original)): raise Refusal('ORIGINAL_FORWARD_JOURNAL_CUSTODY_DRIFT')
    backup_pins = []
    for number, image in enumerate(committed['images']):
        original = image.get('backup_identity')
        if (not isinstance(original, dict) or (not json_exact(original['path'], str(attempt / f'{number:02d}-before.bin')))
                or (not json_exact(original['path'], image['backup'])) or (not json_exact(original['sha256'], image['before']['sha256']))
                or (not json_exact(original['mode'], 0o600))): raise Refusal('ORIGINAL_BACKUP_BINDING_REQUIRED')
        _, visible = read_file(original['path'], original['sha256'])
        if (not json_exact(visible, original)): raise Refusal('ORIGINAL_FORWARD_BACKUP_CUSTODY_DRIFT')
        backup_pins.append(original)
    return [*original_records, *backup_pins, return_identity, confirmation_identity]


def approved_attempt_root(args):
    if args.attempt_root is None or args.expected_attempt_root_identity is None:
        raise Refusal('EXPLICIT_APPROVED_ROOT_ATTEMPT_DIRECTORY_REQUIRED')
    if (not json_exact(str(args.attempt_root), str(ATTEMPT_ROOT))):
        raise Refusal('EXACT_APPROVED_ATTEMPT_LOCATION_REQUIRED')
    observed = stable_directory_identity(args.attempt_root)
    expected = load_json(args.expected_attempt_root_identity)
    if ((not json_exact(observed, expected)) or (not json_exact(observed[2], os.getuid())) or (not json_exact(observed[4], 0o700))):
        raise Refusal('ROOT_ATTEMPT_DIRECTORY_UID_MODE_IDENTITY_MISMATCH')
    if json_exact(args.attempt_root, PACKAGE) or PACKAGE in args.attempt_root.parents:
        raise Refusal('IMMUTABLE_SOURCE_PACKAGE_NOT_JOURNAL_STORAGE')
    return args.attempt_root, observed


def verify_reviewed_program_location():
    if (not json_exact(Path(__file__), PROGRAM_PATH)):
        raise Refusal('EXACT_REVIEWED_PROGRAM_LOCATION_REQUIRED')
    identity = stable_directory_identity(PACKAGE)
    _, program = read_file(PROGRAM_PATH)
    if (not json_exact(identity[2], os.getuid())) or (not json_exact(identity[4], 0o500)) or (not json_exact(program['mode'], 0o500)):
        raise Refusal('IMMUTABLE_REVIEWED_SOURCE_PERMISSIONS_REQUIRED')


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('action', choices=('check', 'install', 'rollback'))
    parser.add_argument('--expected-program-sha')
    parser.add_argument('--attempt-root', type=Path)
    parser.add_argument('--expected-attempt-root-identity')
    parser.add_argument('--proof-file', type=Path)
    parser.add_argument('--proof-sha')
    parser.add_argument('--acceptance-file', type=Path)
    parser.add_argument('--acceptance-sha')
    parser.add_argument('--expected-commit-sha')
    parser.add_argument('--forward-confirmation-file', type=Path)
    parser.add_argument('--forward-confirmation-sha')
    args = parser.parse_args()
    verify_reviewed_program_location()
    audit, pins, lock_paths = real_inputs()
    if audit.get('release_tree'): verify_release_layout(audit)
    if json_exact(args.action, 'check'):
        verify_pins(pins)
        read_file(TARGET, BASE_SHA)
        read_file(PROOF_TARGET, PROOF_BASE_SHA)
        accepted_proof(args, audit, pins)
        print(json.dumps({'state': 'PRECONDITIONS_MATCH_SOURCE_ONLY', 'native_delivery': 'NOT_RUN'}))
        return 0
    if (not json_exact(os.environ.get('CODEX_THREAD_ID'), OWNER)):
        raise Refusal('EXACT_ROOT_ASTRA_CALLER_REQUIRED')
    if not args.expected_program_sha:
        raise Refusal('INDEPENDENTLY_REVIEWED_PROGRAM_SHA_REQUIRED')
    read_file(Path(__file__), args.expected_program_sha)
    pins.append({'path': str(Path(__file__)), 'sha256': args.expected_program_sha})
    attempt_root, attempt_identity = approved_attempt_root(args)
    def attempt_invariant():
        if (not json_exact(stable_directory_identity(attempt_root), attempt_identity)):
            raise Refusal('APPROVED_ATTEMPT_ROOT_IDENTITY_DRIFT')
        if audit.get('release_tree'): verify_release_layout(audit)
        verify_reviewed_program_location()
    if json_exact(args.action, 'install'):
        def precondition():
            for target, digest, mode in ((TARGET, BASE_SHA, 0o700), (PROOF_TARGET, PROOF_BASE_SHA, 0o600)):
                _, identity = read_file(target, digest)
                if (not json_exact(identity['mode'], mode)):
                    raise Refusal('EXACT_BEFORE_IMAGE_MODE_MISMATCH')
            accepted_proof(args, audit, pins)
        accepted_pins = accepted_proof(args, audit, pins)
        specs = [{'target': TARGET, 'source': SOURCE, 'before_sha': BASE_SHA, 'after_sha': R2_SHA, 'mode': 0o700},
                 {'target': PROOF_TARGET, 'source': args.proof_file, 'before_sha': PROOF_BASE_SHA,
                  'after_sha': args.proof_sha, 'mode': 0o600}]
        result = transaction(specs, attempt_root / 'INSTALLATION-ATTEMPT', lock_paths, accepted_pins, precondition,
                             invariant=attempt_invariant)
    else:
        attempt = attempt_root / 'INSTALLATION-ATTEMPT'
        committed_paths = list(attempt.glob('*-committed.json'))
        if (not json_exact(len(committed_paths), 1)):
            raise Refusal('ROLLBACK_REQUIRES_ONE_DURABLE_COMMITTED_INSTALL;RECONCILE_UNCERTAIN_MANUALLY')
        if not args.expected_commit_sha or not args.proof_sha:
            raise Refusal('ROLLBACK_REQUIRES_ROOT_REVIEWED_COMMIT_SHA_AND_INSTALLED_PROOF_SHA')
        committed_raw, _ = read_file(committed_paths[0], args.expected_commit_sha)
        committed = load_json(committed_raw)
        if (not json_exact(committed.get('state'), 'COMMITTED_LOCAL_FILES_ONLY')) or committed.get('durable_commit_completed') is not False:
            raise Refusal('ROLLBACK_COMMIT_STATE_MISMATCH')
        pins.extend(confirmed_forward(args, attempt, committed_paths[0], committed))
        original_attempt_invariant = attempt_invariant
        def rollback_invariant():
            original_attempt_invariant()
            confirmed_forward(args, attempt, committed_paths[0], committed)
        attempt_invariant = rollback_invariant
        images = committed['images']
        if (not json_exact([v['target'] for v in images], [str(TARGET), str(PROOF_TARGET)])):
            raise Refusal('ROLLBACK_TARGET_SET_MISMATCH')
        for number, (image, before_sha, after_sha, mode) in enumerate(zip(
                images, (BASE_SHA, PROOF_BASE_SHA), (R2_SHA, args.proof_sha), (0o700, 0o600))):
            if ((not json_exact(image['before']['sha256'], before_sha)) or (not json_exact(image['after_sha'], after_sha))
                    or (not json_exact(image['before']['mode'], mode)) or (not json_exact(image['after_mode'], mode))
                    or (not json_exact(image['backup'], str(attempt / f'{number:02d}-before.bin')))):
                raise Refusal('ROLLBACK_EXACT_IMAGE_BINDING_MISMATCH')
        def precondition():
            for image in images:
                _, live = read_file(image['target'], image['after_sha'])
                if (not json_exact(live['signature'], image['installed']['signature'])):
                    raise Refusal('ROLLBACK_CONCURRENT_CHANGE_REFUSED')
        # Reverse order; between-file failure remains fail-closed and explicitly uncertain.
        specs = [{'target': v['target'], 'source': v['backup'], 'before_sha': v['after_sha'],
                  'after_sha': v['before']['sha256'], 'mode': v['before']['mode']} for v in reversed(images)]
        result = transaction(specs, attempt_root / 'ROLLBACK-ATTEMPT', lock_paths, pins, precondition,
                             invariant=attempt_invariant)
    print(json.dumps(result, sort_keys=True))
    return 0 if json_exact(result['state'], 'COMMITTED_LOCAL_FILES_ONLY') and not result.get('cleanup_errors') else 2


if json_exact(__name__, '__main__'):
    try:
        raise SystemExit(main())
    except (Refusal, OSError, ValueError, KeyError) as exc:
        print(json.dumps({'state': 'REFUSED_NO_INSTALLATION', 'cause': {'type': type(exc).__name__, 'message': str(exc)},
                          'native_delivery': 'NOT_RUN'}), file=sys.stderr)
        raise SystemExit(2)
