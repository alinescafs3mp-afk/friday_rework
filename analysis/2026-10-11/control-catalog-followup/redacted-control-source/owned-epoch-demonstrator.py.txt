class Lease:
    def __init__(self,domain,state,loaded,ids,pid,lock_id):

        self.loaded_sha={k:digest(v) for k,v in loaded.items()}

    def check(self):
        if not self.active:raise Refused('LEASE_EXPIRED')
        if ident((self.domain.root/'admission.lock').lstat())!=self.lock_id:raise Refused('LOCK_DRIFT')
        if self.domain.state()!=self.state:raise Refused('EPOCH_DRIFT')
        loaded,ids,pid=self.domain.verify(self.state)
        if ids!=self.ids or pid!=self.pid or {k:digest(v) for k,v in self.loaded.items()}!=self.state['sources']:raise Refused('LOADED_OR_INODE_DRIFT')

    def effect(self,callback):
